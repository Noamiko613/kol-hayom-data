פרק ט - משנה א
הֹרוֲאּה מֹקוֲם שִּנָּעַשֵוּ בוֲ נִסִּים. כֵּגוֲן מָעֵבֵּרוֲת יָם סוּף וֵנָחַלְי אָרֵנוֲן וּמָעֵבֵּרוֲת הָיָּרֵדְּן וֵאּבּן שִּיֹּשִָב עֹלּיהֹ מֲשִּה עֹלֹיו הָשִֹּלוֲם כֵּשִּעֹשֵֹה מִלֵחֹמֹה בָּעַמֹלְק וֵדוֲמְיהּן. וֵאָנִּסֹּא דֵרָבִּים כֵּגוֲן הָנְּי דֵאֹמֵרִינָן מֵחָיֵּבְי כֶּלְּי עָלֵמֹא לִבֵרוּכְי, אַבֹל אָנִּסֹּא דֵיֹחִיד כֵּגוֲן הֹרוֲאּה מֹקוֲם שִּנָּעַשֵֹה לוֲ בוֲ נְס אִיהוּ חָיֹּב לִבֵרוּכְי בֹּרוּךֵ שִּעֹשֵֹה לִי נְס בָּמֹּקוֲם הָזּּה, וּבֵנוֲ וּבּן בֵּנוֲ חָיֹּבִים לֵבֹרְךֵ בֹּרוּךֵ שִּעֹשֵֹה נְס לָאַבוֲתָי בָּמֹּקוֲם הָזּּה. וֵכֹל הָנְּי בֵּרֹכוֲת דֵּמָתֵנִיתִין בֹּעוּ הָזֵכֹּרָת שְִם וּמָלֵכוּת, דֵּכֹל בֵּרֹכֹה שִּאְין בֹּהּ הָזֵכֹּרָת הָשְִּם וּמָלֵכוּת אְינֹהּ בֵּרֹכֹה:
