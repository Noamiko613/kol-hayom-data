פרק ז' - הזהיר בחטא זה מסייע לבנין הבית העתיד

בּוֲ יֵבֲאָר שִּהָזֹּהִיר מִזּּה הָחְטֵא הּחֹמוּר הוּא מֵסָיְּעְ לֵבִנֵיָן הָבָּיִת הּעֹתִיד.

יֹדוּעָ מָאַמָר חַזָ''ל , דֵּדוֲר שִּל בָּיִת שְִנִי, הֹיֹה בּוֲ תּוֲרֹה וּמִצֵוֲת, וֵחֶרֵבֹּנוֲ הֹיֹה בָּעַוֲן שִֵנֵאָת חִנֹּם וֵלֹשִוֲן הֹרֹע. וֵכֹתֵבוּ הֹרִאשִוֲנִים, דֵּמָה שִּּהֹיֹה כֲּחָ בּּעֹוֲן הָזּּה לֵהָחַרִיב בָּיִת הָבֹּנוּי, עָל אָחָת כָּמֹּה וֵכָמֹּה שִּלֲּא יָנִּיחָ לִבֵנוֲתוֲ מְחֹדֹשִ. וֵזּהוּ רּמּז מָה שִּּאֹמֵרוּ חַזָ''ל: כֹּל דּוֲר שִּלֲּא נִבֵנֹה בְּית הָמִּקֵדֹּשִ בֵּיֹמֹיו כֵּאִלּוּ נּחֱרָב בֵּיֹמֹיו. וֵאִם כְּן, עָל כֹּרֵחַךֹ שִּאֹנוּ צֵרִיכִין לֵהִתֵחָזְּק לֵתָקְּן הָחְטֵא הָזּּה, דֵּהָיֵנוּ לֵהִזֹּהְר שִּלֲּא לֵהִכֹּשְִל בּוֲ, דֵּעָד מֹתָי נִהֵיּה בָּגּוֲלֹה?

בּּאֱמּת נִתֵבּוֲנְן מִי יוּכָל לֵתָקְן הָחְטֵא הָזּּה. הֹאִישִ הָפֹּשִוּט בֵּוָדָּאי לֲא יוּכָל לִשִֵמֲר הֹעִנֵיֹן הָזּּה כָּהַלֹכֹה, אָחַרְי שִּאְינוֲ יוֲדְעָ מָה הוּא לֹשִוֲן הֹרֹע וּמָה הוּא רֵכִילוּת, וֵאְיזּה דֹּבֹר נִכֵנֹס בִּכֵלֹלֹם. וֵעִקָּר הָתִּקּוּן תֹּלוּי בֵּמִי שִּהוּא בּּן תּוֲרֹה. הוּא יֹכוֲל לֵהִתֵבּוֲנְן בֵּהִלֵכוֲתּיהֹ וּלֵהִזֹּהְר וּלֵקָיֵּמֹם.

אֹמֵרוּ חַזָ''ל : בֵּשִֹעֹה שִּנֹּשֵֹא שִֵלֲמֲה אּת בָּת פָּרֵעֲה וֵעֹבָר עָל ''לֲא יָרֵבּּה לוֲ נֹשִִים'', עֹלֵתֹה הָיּוּ''ד דֵּ''לֲא יָרֵבּּה'' לִפֵנְי ה' יִתֵבֹּרָךֵ וֵאֹמֵרֹה: דֵּיֹתְיקָי שִּבֹּטֵלֹה מִקֵצֹתֹהּ בֹּטֵלֹה כֶּלֹּהּ. אִם שִֵלֲמֲה יֵבָטֵּלְנִי, מִי יֵקָיֵּמְנִי? . עֹנֹה הָקֹּדוֲשִ בֹּרוּךֵ הוּא: שִֵלֲמֲה וֵאּלּף כָּיּוֲצְא בּוֲ יִהֵיוּ בֵּטְלִין, וֵאָל תִּבֹּטְל אוֲת אָחָת מִן הָתּוֲרֹה. וּמִזּּה נִרֵאּה בָּעַלִיל, שִּהָקֹּדוֲשִ בֹּרוּךֵ הוּא מָקֵפִּיד גֹּדוֲל כֵּשִּדֹּבֹר אּחֹד מִן הָתּוֲרֹה נָעַשֵּה הּפֵקְר לָכֲּל. וֵכָאַשִּר נִתֵבּוֲנְן הְיטְב אְיזּה דֹּבֹר מִן הָתּוֲרֹה הוּא הּפֵקְר, נִמֵצֹא שִּהּעֹוֲן שִּל לֹשִוֲן הֹרֹע הוּא הּפֵקְר בֵּעְינְי הָבֵּרִיוֲת שִּלֲּא טֹעַמוּ טָעָם תּוֲרֹה.

.

כְּן כָּמֹּה צֹרִיךֵ בּּן תּוֲרֹה עָל כֹּל פֹּנִים לֵהִתֵחָזְּק לִהֵיוֲת זֹהִיר בֵּאִסּוּרִים אְלּוּ, הָתֵּלוּיִם בֵּכֲחָ הָדִּבּוּר, , כֵּדְי שִּלֲּא יִהֵיוּ דִּבֵרְי ה' כֵּהּפֵקְר. וֵהִנְּה לֵפִי מָה שִּּבְּאָרֵנוּ לֵעְיל, תֹּלוּי בֹּזּה בִּנֵיָן הָבָּיִת דֵּלּעֹתִיד, , וֵכֹל אּחֹד שִּיִּתֵחָזְּק לֵתָקִּן הָחְטֵא הָזּּה, יִהֵיּה לוֲ חְלּק בָּבָּיִת הָבֹּנוּי לּעֹתִיד, דֵּבִלֵתֹּם הֹיֹה הָבָּיִת חֹרְב לֵעוֲלֹם, חָס וֵשִֹלוֲם.

אִם הֹיֵתֹה לֹנוּ רֵשִוּת לִבֵנוֲת בְּית הָמִּקֵדֹּשִ, וֵהֹיֹה צֹרִיךֵ כּּסּף לֹזּה, יֹדוּעָ דֵּכֹל אּחֹד מִיִּשֵֵרֹאְל הֹיֹה מֵנָדְּב בֵּנּפּשִ חַפְצֹה כֹּל מָה שִּּבֵּכֲחוֲ לֹזּה, כֵּדְי שִּיִּהֵיּה לוֲ גָּם כְּן חְלּק בֵּבְית הָמִּקֵדֹּשִ, וּבֵעִנֵיֹנְנוּ אְין צֹרִיךֵ כּּסּף כֵּלֹל, רָק לֵהִתֵחָזְּק לֵהִתֵרָחְק מִן הָחְטֵא הּחֹמוּר שִּל לֹשִוֲן הֹרֹע וֵשִֵנֵאָת חִנֹּם, וֵלּאֱחֲז בֵּמִדָּת הָשִֹּלוֲם, וֵעָל יֵדְי זּה יֵתֶקָּן הָחְטֵא וֵנִזֵכּּה לֵבִיאָת הָמֹּשִִיחָ וּלֵבִנֵיָן הָמִּקֵדֹּשִ.

יִתֵרוֲמְם שִֵמוֲ שִּל הֹאִישִ הָזּּה, שִּהוּא הֹיֹה הָסִּבֹּה לֵבִנֵיָן הָבָּיִת, וֵכָאַשִּר נִמֵצֹא בִּנֵחּמֵיֹה , נִרֵשִֵמוּ שִֹם לֹנּצָח שִֵמוֲת הֹאַנֹשִִים, שִּנֹּתֵנוּ חְלּק בָּחוֲמֹה שִּל יֵרוּשִֹלָיִם, וֵעָל אָחָת כָּמֹּה וֵכָמֹּה בָּבָּיִת גּוּפֹא.
