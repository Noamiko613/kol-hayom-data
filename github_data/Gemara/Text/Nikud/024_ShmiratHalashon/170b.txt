פרק ט' - ועוד מעניין המחזיק בניו לתלמוד תורה

גַּם זֶה מֵעֵנְיָן הַנִּזְכָּר לְעִיל.

נְדַבֵּר אוֹדוֹת יַעֲקֹב אָבִינוּ עָלָיו הַשָּׁלוֹם, דְּהִנֵּה יָדוּעַ, בְּעֵת שֶׁהָלַךְ מִבֵּית אָבִיו לְחָרָן, הָיָה בֶּן שִׁשִּׁים וְשָׁלֹש, כְּפֵרוּשׁ רַשִׁ''י, וּבֶן שִׁבְעִים וְשֶׁבַע הָיָה כְּשֶׁבָּא לְבֵית לָבָן, וְאַיֵּה הָיָה אַרְבַּע עֶשְׂרֵה שָׁנִים שֶׁבֵּינְתַיִם? אָמְרוּ חֲזַ''ל , שֶׁהָיָה אֵצֶל שֵׁם וְעֵבֶר וְלָמַד שָׁם בְּבֵית מִדְרָשָׁם.

לִכְאוֹרָה קַשְׁיָא עַל יַעֲקֹב אָבִינוּ עָלָיו הַשָּׁלוֹם. וְכִי זוֹ הָעֵת וְהָעוֹנָה לְתַלְמוּד תּוֹרָה, וַהֲלֹא בֶּן שְׁלֹשִים וְשָׁלֹש שָׁנָה הָיָה, וְלֹא הָיָה לוֹ עֲדַיִן שׁוּם זֶרַע, וּבִפְרָט שֶׁאֻמָּה הַיִּשְׂרְאֵלִית הַקְּדוֹשָׁה, הַצְּרִיכָה לָצֵאת עַל פְּנֵי הָאֲדָמָה, רַק הוּא מֻכְשָׁר לָזֶה, וְלֹא אִישׁ אַחֵר, וְכָל זֶה יָדוּעַ הָיָה לְיַעֲקֹב אָבִינוּ, כְּמוֹ שֶׁפֵּרֵשׁ רַשִּׁ''י עַל הַפָּסוּק : ''כִּי מָלְאוּ יָמָי'', וְאֵימָתַי אַעֲמִיד שְׁנֵים עָשָׂר שְׁבָטִים? עַיֵּן שָׁם. וְהִנֵּה אִם כֵּן בְּוַדַּאי מִצְוָה גְּדוֹלָה, שֶׁיֵּלֵךְ תֵּכֶף אֶל בֵּית לָבָן, כְּמוֹ שֶׁאָמַר לוֹ יִצְחָק, וְלִשָּׂא אִשָּׁה וְלֹא לְהִתְעַכֵּב וּלְהַפְרִישׁ אֶת עַצְמוֹ לְתַלְמוּד תּוֹרָה אַרְבַּע עֶשְׂרֵה שָׁנָה.

כָּל זֶה נִרְאֶה בִּפְשִׁיטוּת, דְּהִנֵּה יַעֲקֹב אָבִינוּ יָדַע בְּלָבָן, כִּי בְּוַדַּאי לֹא יִתְרַצֶּה לִתֵּן לוֹ בִּתּוֹ, בִּלְתִּי שֶׁיְּקַבֵּל עֲבוּר זֶה מַתָּנָה גְּדוֹלָה, וּכְמוֹ שֶׁכָּתוּב : ''הֲלוֹא נָכְרִיּוֹת נֶחְשַׁבְנוּ לוֹ'' וְגוֹ'. וְיָדוּעַ, שֶׁיַּעֲקֹב אָבִינוּ נַעֲשָׂה עָנִי בַּדֶּרֶךְ עַל יְדֵי אֱלִיפָז, אִם כֵּן בְּוַדַּאי יִהְיֶה מֻכְרָח לַעֲבֹד אֶצְלוֹ עֲבוֹדַת עֶבֶד וְלִשָּׂא מִבְּנוֹתָיו, כְּמוֹ שֶׁצִּוָּה לוֹ יִצְחָק, וּכְמוֹ שֶׁהָיָה לְבַסּוֹף. וְהָיָה יָרֵא יַעֲקֹב אָבִינוּ לָרֶדֶת אֶצְלוֹ וּלְהִתְעַכֵּב שָׁמָּה, כִּי יָדַע אֶת לָבָן כִּי רַמַּאי הוּא, וְגַם יִרְצֶה לְהַאֲבִידוֹ מִן הָעוֹלָם, וּכְמוֹ שֶׁהֵעִיד עָלָיו הַכָּתוּב לְבַסּוֹף, שֶׁרָדַף אַחֲרָיו וְרָצָה לְהָרְגוֹ, לוּלֵא שֶׁאֱלֹקֵי אַבְרָהָם הָיָה בְּעֶזְרוֹ וְהִצִּילוֹ, וְגַם בֵּיתוֹ מָלֵא טֻמְאַת תְּרָפִים, וְאֵין בְּכֹחוֹ לַעֲמֹד נֶגֶד טֻמְאָה רַבָּה כָּזוֹ, בִּלְתִּי בִּזְכוּת קְדֻשַּׁת הַתּוֹרָה, וּבְבֵית לָבָן לֹא יִהְיֶה בְּאֶפְשָׁרוּתוֹ לִהְיוֹת דָּבוּק לַתּוֹרָה.

כֵּן הִקְדִּים אֶת עַצְמוֹ וְנָס אֵצֶל שֵׁם וְעֵבֶר אַרְבַּע עֶשְׂרֵה שָׁנָה וְלָמַד שָׁם בְּהַתְמָדָה רַבָּה, כִּי כָּל הָאַרְבַּע עֶשְׂרֵה שָׁנָה לֹא שָׁכַב לִישֹׁן פַּעַם אַחַת שֵׁנַת קֶבַע, כְּמוֹ שֶׁהֵעִיד עָלָיו הַכָּתוּב : ''וַיִּשְׁכַּב בַּמָּקוֹם הַהוּא'', וּכְפֵרוּשׁ רַשִּׁ''י שָׁם. וּבְלִמּוּד כָּזֶה, אַף כִּי יִתְעַכֵּב אַחַר כָּךְ בְּבֵית לָבָן אֵיזֶה שָׁנִים, לֹא יַזִּיק לוֹ מְאוּמָה. .

אֲנַחְנוּ, שֶׁזּוֹכְרִין תָּמִיד זְכוּת אָבוֹת, צְרִיכִין לֵילֵךְ גַּם כֵּן בְּעִקְבוֹת יַעֲקֹב אָבִינוּ עָלָיו הַשָּׁלוֹם, וְכָל אָב יִרְאֶה לְהַרְגִּיל בְּנוֹ לְתוֹרָה וּלְמִצְוֹת כָּל עֵת הֱיוֹת הַבֵּן עַל שֻׁלְחָנוֹ, וְהוּא עַד שְׁנַת עֶשְׂרִים וְאַחַת, וְאָז, אֲפִלּוּ אִם יֵצֵא לוֹ הַגּוֹרָל לַעֲבוֹדַת הַקֵּיסָר יָרוּם הוֹדוֹ, וּבְוַדַּאי לֹא יוּכַל לִלְמֹד בָּעֵת הַהִיא, אַף עַל פִּי כֵן לֹא תָּסוּר מִמֶּנּוּ קְדֻשַּׁת הַתּוֹרָה, שֶׁלָּמַד עַד עַתָּה. . וְגַם בִּמְלֹאת שְׁנֵי עֲבוֹדָתו, כְּשֶׁיָּבוֹא לְבֵיתוֹ, יִהְיוּ כָּל מַעֲשָׂיו בְּתֹם וּבְיֹשֶר, וְלֹא יִתְבַּיֵּשׁ אָבִיו מִפָּנָיו, וְיַצְלִיחוֹ הַקָדוֹשׁ בָּרוּךְ הוּא עֲבוּר זֶה בְּזֶה הָעוֹלָם, כְּמוֹ שֶׁעָשָׂה לְיַעֲקֹב אָבִינוּ בִּכְלוֹת יְמֵי עֲבוֹדָתוֹ אֵצֶל לָבָן, כְּמוֹ שֶׁכָּתוּב בַּתּוֹרָה, וְגַם בָּעוֹלָם הָעֶלְיוֹן, כְּשֶׁיִּזְכֶּה לַעֲמֹד לִפְנֵי כִּסֵּא כְּבוֹדוֹ יִתְבָּרַךְ, יִהְיֶה מִקְּדוֹשֵׁי ה', בִּזְכוּת שֶׁהֶאֱמִיר אֶת ה' לִהְיוֹת לוֹ לֵאלֹקִים, כְּמוֹ שֶׁכָּתוּב : ''אֶת ה' הֶאֱמַרְתָּ הַיּוֹם לִהְיוֹת לְךָ לֵאלֹקִים וְלָלֶכֶת בִּדְרָכָיו וְלִשְׁמֹר חֻקָּיו וּמִצְוֹתָיו וְגוֹ', וְה' הֶאֱמִירְךָ הַיּוֹם וְגוֹ', וְלִהְיֹתְךָ עַם קָדֹש לַה' אֱלֹקֶיךָ'' וְגוֹ'.

אִם יִרְאֶה לְסַלֵּק מֵעָלָיו עֹל הַתּוֹרָה וְהַמִּצְוֹת בְּנַעֲרוּתוֹ כְּדֵי לִפְרֹק עֹל עֲבוֹדַת הַצָּבָא, שֶׁלֹּא יִצְטָרֵךְ לַעֲבֹד אוֹתוֹ אֶלָּא זְמַן מֻעָט, אָז יִהְיֶה בֹּשֶת לְאָבִיו, בְּרֵאשִׁיתוֹ בְּזֶה הָעוֹלָם, וְחֶרְפָּה וּכְלִמָּה מִמֶּנּוּ בְּאַחֲרִיתוֹ, בָּעוֹלָם הָעֶלְיוֹן, שֶׁהֵמִיר עוֹלָם, הָעוֹמֵד לָעַד לְעוֹלָם, בְּעוֹלָם כָּלֶה.

שֶׁיָּדוּעַ עִנְיַן הַיֵּצֶר, שֶׁהוּא מְעוֹרֵר תָּמִיד לָאָדָם לַחֲשֹׁב רַק בְּמַה שֶּׁנּוֹגֵעַ לַהַצְלָחָה הַזְּמַנִּית, לָכֵן הִקְדִּים לָנוּ הַתַּנָּא צֳרִי לָזֶה וְאָמַר : הֱוֵי מְחַשֵׁב הֶפְסֵד מִצְוָה כְּנֶגֶד שְׂכָרָהּ וּשְׂכַר עֲבֵרָה כְּנֶגֶד הֶפְסֵדָהּ. הַיְנוּ שֶׁיַּעֲרִיךְ קַטְנוּת הַהֶפְסֵד שֶׁיֵּשׁ לוֹ בָּעוֹלָם הַזֶּה מֵהַמִּצְוָה, כְּנֶגֶד עֶצֶם הַשָּׂכָר הָרוּחָנִי הַנִּצְחִי, שֶׁיִּהְיֶה לוֹ בָּעוֹלָם הָעֶלְיוֹן. וְאִם הַיֵּצֶר הָרָע מַזְמִין לוֹ לַעֲשׂוֹת אֵיזוֹ עֲבֵרָה בִּשְׁבִיל שׁוּם הֲנָאָה, יַעֲרִיךְ קַטְנוּת שָׂכָר, שֶׁיִּהְיֶה לוֹ מֵהָעֲבֵרָה בָּעוֹלָם הַזֶּה עַל זְמַן מֻעָט, נֶגֶד הָעֹנֶשׁ הַגָּדוֹל וְהַצַּעַר הַנִּצְחִי, שֶׁיִּהְיֶה לוֹ אַחַר כָּךְ.

אָנוּ נֹאמַר בְּעִנְיָנֵנוּ, כְּדֵי לְהִתְגַּבֵּר עַל הַיֵּצֶר וְהַמְסִיתִים אוֹתוֹ לְהַטּוֹת בְּנוֹ מִדֶּרֶךְ טוֹבָה לְדֶרֶךְ רָעָה, יַעֲרִיךְ לְנֶגֶד עֵינָיו הַשָּׂכָר וְהָעֹנֶשׁ הַנִּצְחִי שֶׁמַּגִּיעַ לָאָדָם עַל יְדֵי בָּנָיו, וְאָז יִרֶף מִמֶּנּוּ הַיֵּצֶר, דְּזֶה לְשׁוֹן הַזֹּהַר הַקָּדוֹשׁ פָּרָשַׁת בְּחֻקֹּתַי:

פָּתַח וְאָמַר : ''בֶּן יְכַבֵּד אָב וְעֶבֶד אֲדֹנָיו''. בֶּן יְכַבֵּד אָב כְּמָה דְּאָמוּר : ''כַּבֵּד אֶת אָבִיךָ וְאֶת אִמֶּךָ''. וְאוֹקְמוּהָ בְּמֵיכְלָא וּמִשְׁתְּיָא, וּבְכֹלָּא, הַאִי בְּחַיָּיו דְּאִתְחַיַּב בֵּהּ. בָּתַר דְּמִית, אִי תֵּימָא פָּטוּר מִנַּהּ הוּא? לָאו הָכִי, דְּאַף עַל גַּב דְּמִית, אִתְחַיַּב בִּיקָרֵהּ יָתִיר, דִּכְתִיב: ''כַּבֵּד אֶת אָבִיךָ''. דְּאִי הַהוּא בְּרָא אָזֵיל בְּאֹרַח תַּקָּלָא, וַדַּאי מְבַזֶּה לַאֲבוֹי הוּא, וַדַּאי עָבֵיד לֵהּ קְלָנָא. וְאִי הַהוּא בְּרָא אָזֵיל בְּאֹרַח מִישׁוֹר וְתַקֵּן עוֹבָדוֹי, וַדַּאי אוֹקִיר לַאֲבוֹי. אוֹקִיר לֵהּ בְּהַאי עָלְמָא גַּבֵּי בְּנֵי נָשָׁא, אוֹקִיר לֵהּ בְּהַהוּא עָלְמָא גַּבֵּי קֻדְּשָׁא בְּרִיךְ הוּא. וְקֻדְשָׁא בְּרִיךְ הוּא חָיֵיס עֲלֵהּ וְאוֹתִיב לֵהּ בְּכֻרְסַיָּא דִּיקָרֵהּ. בְּרַם זַכָּאִין אִנּוּן צַדִּיקַיָּא דְּזָכָאן לִבְנִין קַדִּישִׁין לְגִזְעִין קַדִּישִׁין, עֲלַיְהוּ אִתְקָרֵי: ''כָּל רוֹאֵיהֶם יַכִּירוּם, כִּי הֵם זֶרַע בֵּרַךְ ה'''. עַד כָּאן לְשׁוֹנוֹ.
