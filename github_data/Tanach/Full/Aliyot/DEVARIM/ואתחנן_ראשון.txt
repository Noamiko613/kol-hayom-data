וֻׁאּתָחְנְִן אּל יָהֲוֻה בִֻעֹת הְהִוא לֹאמֲׁר:
אֶדֲנֻי יַהֲוִה אְתִֻה הְׁחִלִוֲתֻ לָהְרָאוֲת אּׁת עְבָדִָךֻ אּת גִֻדָלָךֻ וָאּת יֻׁדָךֻ הְׁחֶזֻקֻה אֶשּ֖ר מִי אֹל בְִשִֻ֖מְיִם וִבֻאֻרּץ אֶשּ֖ר יְׁעֶשּ֚ה כָמְׁעֶשּ֚יךֻ וָכִגָבׁוִרֲתּׁךֻ:
אּעָבִָרֻה נִֻא וָאּרָאּה אּת הֻאֻרּץ הְטִוֲבֻה אֶשּ֖ר בִָעֹבּר הְיְִרָדִֹן הֻהֻר הְטִוֲב הְזִּה וָהְלִָבֻנֲׁן:
וְיִִתָעְבִֹר יָהֲוֻה בִִי לָמְעְנָכּם וָלֲא שֻ֖מְע אֹלֻי וְיֲִאמּר יָהֲוֻה אֹלְי רְב לֻךָ אְל תִוֲסּף דְִבִֹר אֹלְי עוֲד בְִדִֻבֻר הְזִּׁה:
עֶלֹה רֲאש֖ הְפִִסָגִֻה וָשֻ֚א עֹינּיךֻ יֻמִֻה וָצֻפֲנֻה וָתֹימֻנֻה וִמִזָרֻחֻה וִרָאֹה בָעֹינּיךֻ כִִי לֲא תְׁעֶבֲר אּת הְיְִרָדִֹן הְזִּׁה:
וָצְו אּת יָהוֲשֳ֖עְ וָחְזִָקֹהוִ וָאְמִָצֹהוִ כִִי הוִא יְׁעֶבֲר לִפָנֹי הֻעֻם הְזִּה וָהוִא יְנָחִיל אוֲתֻם אּת הֻאֻרּץ אֶשּ֖ר תִִרָאּׁה:
וְנִֹשּ֖ב בְִגִֻיָא מוִל בִֹית פִָעׁוֲר: (פ)
וָעְתִֻה יִשָ֚רֻאֹל שָ֖מְע אּל הְׁחֳקִִים וָאּל הְמִִשָ֖פִֻטִים אֶשּ֖ר אֻׁנֲכִי מָלְמִֹד אּתָכּם לְׁעֶש֚וֲת לָמְעְן תִִׁחָיוִ וִבֻאתּם וִׁירִשָ֖תִּם אּת הֻאֻרּץ אֶשּ֖ר יָהֲוֻה אַלֲהֹי אֶבֲׁתֹיכּם נֲתֹן לֻכּׁם:
לֲא תֲסִפוִ עְל הְדִֻבֻר אֶשּ֖ר אֻׁנֲכִי מָצְוִּה אּתָכּם וָלֲא תִגָרָעוִ מִמִּנִוִ לִשָ֖מֲר אּת מִצָוֲת יָהֲוֻה אַלֲׁהֹיכּם אֶשּ֖ר אֻׁנֲכִי מָצְוִּה אּתָכּׁם:
עֹׁינֹיכּם הֻׁרֲאוֲת אֹת אֶשּ֖ר עֻשֻ֚ה יָהֲוֻה בִָבְעְל פִָעוֲר כִִי כֻל הֻאִיש֖ אֶשּ֖ר הֻלְךָ אְׁחֶרֹי בְׁעְל פִָעוֲר הִשָ֖מִידוֲ יָהֲוֻה אַלֲהּיךֻ מִקִִרָבִּׁךֻ:
וָאְתִּם הְדִָבֹקִים בְִׁיהֲוֻה אַלֲׁהֹיכּם חְיִִים כֳִלִָכּם הְיִׁוֲם:
