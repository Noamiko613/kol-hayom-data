רביעי  וָטְפִָכּם אֶשּ֖ר אֶמְרָתִּם לֻבְז יִׁהָיּה וִבָנֹיכּם אֶשּ֖ר לֲא יֻׁדָעוִ הְיִוֲם טוֲב וֻרֻע הֹמִֻה יֻבֲאוִ שֻ֖מִֻה וָלֻהּם אּתִָנּנִֻה וָהֹם יִׁירֻשׁ֖וִהֻ:
וָאְתִּם פִָנוִ לֻכּם וִסָעוִ הְמִִדָבִֻרֻה דִּרּךָ יְם סׁוִף:
וְׁתְִׁעֶנוִ וְתֲִׁאמָרוִ אֹלְי חֻטֻאנוִ לְׁיהֲוֻה אֶנְחָנוִ נְׁעֶלּה וָנִלָחְמָנוִ כִָכֲל אֶשּ֖ר צִוִֻנוִ יָהֲוֻה אַלֲהֹינוִ וְׁתְִחָגִָרוִ אִיש֖ אּת כִָלֹי מִלָחְמָתִוֲ וְתִֻהִינוִ לְׁעֶלֲת הֻהֻׁרֻה:
וְיֲִאמּר יָהֲוֻה אֹלְי אַמֲר לֻהּם לֲא תְׁעֶלוִ וָלֲא תִׁלִֻׁחֶמוִ כִִי אֹינּנִִי בִָקִרָבִָכּם וָלֲא תִִנִֻׁגָפוִ לִפָנֹי אֲׁיָבֹיכּׁם:
וֻׁאֶדְבִֹר אֶלֹיכּם וָלֲא שָ֖מְעָתִּם וְתְִמָרוִ אּת פִִי יָהֲוֻה וְתִֻזִדוִ וְתְִׁעֶלוִ הֻהֻׁרֻה:
וְיִֹצֹא הֻׁאַמֲרִי הְיִשֹ֖ב בִֻהֻר הְהוִא לִקָרְאתָכּם וְיִִרָדִָפוִ אּתָכּם כְִׁאֶשּ֖ר תְִׁעֶשּ֚ינֻה הְדִָבֲרִים וְיְִכִָתוִ אּתָכּם בִָשֹ֚עִיר עְד חֻרָמֻׁה:
וְתִֻשֳ֖בוִ וְתִִבָכִוִ לִפָנֹי יָהֲוֻה וָלֲׁא שֻ֖מְע יָהֲוֻה בִָקֲלָכּם וָלֲא הּׁאַזִין אֶלֹיכּׁם:
וְתִֹׁשָ֖בוִ בָקֻדֹש֖ יֻמִים רְבִִים כְִיִֻמִים אֶשּ֖ר יָשְ֖בָתִּׁם:
וְנִֹפּן וְנִִסְִע הְמִִדָבִֻרֻה דִּרּךָ יְם סוִף כְִׁאֶשּ֖ר דִִבִּר יָהֲוֻה אֹלֻי וְנִֻסֻב אּת הְר שֹ֚עִיר יֻמִים רְבִִׁים: (ס)
