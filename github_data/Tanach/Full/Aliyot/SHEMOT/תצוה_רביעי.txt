&nbspרביעי&nbsp&nbspוָזּה הְדִֻבֻר אֶשּ֖ר תְִׁעֶשּ֚ה לֻהּם לָקְדִֹש֖ אֲתֻם לָכְהֹן לִי לָקְח פְִר אּחֻד בִּן בִֻקֻר וָאֹילִם שָ֖נְיִם תִָמִימִׁם:
וָלּחּם מְצִוֲת וָחְלֲִת מְצֲִת בִָלוִלֲת בְִשִּ֖מּן וִרָקִיקֹי מְצִוֲת מָשֳ֖חִים בְִשִֻ֖מּן סֲלּת חִטִִים תְִׁעֶשּ֚ה אֲתֻׁם:
וָנֻׁתְתִֻ אוֲתֻם עְל סְל אּחֻד וָהִקָרְבָתִֻ אֲתֻם בְִסִֻל וָאּת הְפִֻר וָאֹת שָ֖נֹי הֻׁאֹילִׁם:
וָאּת אְׁהֶרֲן וָאּת בִֻנֻיו תְִקָרִיב אּל פִּתְח אֲהּל מוֲעֹד וָרֻׁחְצָתִֻ אֲתֻם בְִמִֻׁיִם:
וָלֻׁקְחָתִֻ אּת הְבִָגֻדִים וָהִלָבְִשָ֖תִֻ אּׁת אְׁהֶרֲן אּת הְכֳִתֲִנּת וָאֹת מָעִיל הֻׁאֹפֲד וָאּת הֻׁאֹפֲד וָאּת הְחֲשּ֖ן וָאֻׁפְדָתִֻ לוֲ בִָחֹשּ֖ב הֻׁאֹפֲׁד:
וָשְ֚מָתִֻ הְמִִצָנּפּת עְל רֲאש֖וֲ וָנֻׁתְתִֻ אּת נֹזּר הְקֲִדּש֖ עְל הְמִִצָנֻׁפּת:
וָלֻׁקְחָתִֻ אּת שּ֖מּן הְמִִשָ֖חֻה וָיֻׁצְקָתִֻ עְל רֲאש֖וֲ וִמֻׁשְ֖חָתִֻ אֲתׁוֲ:
וָאּת בִֻנֻיו תְִקָרִיב וָהִלָבְִשָ֖תִֻם כֳִתִֵנֲׁת:
וָחֻׁגְרָתִֻ אֲתֻם אְבָנֹט אְׁהֶרֲן וִבֻנֻיו וָחֻׁבְשָ֖תִֻ לֻהּם מִגָבִֻעֲת וָהֻׁיָתֻה לֻהּם כִָהֳנִֻה לָחֳקְִת עוֲלֻם וִמִלִֹאתֻ יְׁד אְׁהֶרֲן וָיְד בִֻנֻׁיו:
וָהִקָרְבָתִֻ אּת הְפִֻר לִפָנֹי אֲהּל מוֲעֹד וָסֻמְךָ אְׁהֶרֲן וִבֻנֻיו אּת יָדֹיהּם עְל רֲאש֖ הְפִֻׁר:
וָשֻׁ֖חְטָתִֻ אּת הְפִֻר לִפָנֹי יָהוֲֻה פִּתְח אֲהּל מוֲעֹׁד:
וָלֻׁקְחָתִֻ מִדְִם הְפִֻר וָנֻׁתְתִֻה עְל קְרָנֲת הְמִִזָבִֹחְ בִָאּצָבִֻעּךֻ וָאּת כִֻל הְדִֻם תִִשָ֖פֲִךָ אּל יָסוֲד הְמִִזָבִֹׁחְ:
וָלֻׁקְחָתִֻ אּׁת כִֻל הְחֹלּב הְׁמָכְסִּה אּת הְקִּרּב וָאֹת הְיֲִתּרּת עְל הְכִֻבֹד וָאֹת שָ֖תִֹי הְכִָלֻיֲת וָאּת הְחֹלּב אֶשּ֖ר עֶלֹיהּן וָהִקָטְרָתִֻ הְמִִזָבִֹׁחֻה:
וָאּת בִָשְ֚ר הְפִֻר וָאּת עֲרוֲ וָאּת פִִרָש֖וֲ תִִשָ֚רֲף בִֻאֹש֖ מִחוִץ לְׁמְִׁחֶנּה חְטִֻאת הׁוִא:
וָאּת הֻאְיִל הֻׁאּחֻד תִִקִֻח וָסֻמָכוִ אְׁהֶרֲן וִבֻנֻיו אּת יָדֹיהּם עְל רֲאש֖ הֻאֻׁיִל:
וָשֻׁ֖חְטָתִֻ אּת הֻאֻיִל וָלֻׁקְחָתִֻ אּת דִֻמוֲ וָזֻׁרְקָתִֻ עְל הְמִִזָבִֹחְ סֻבִׁיב:
וָאּת הֻאְיִל תִָנְתִֹחְ לִנָתֻחֻיו וָרֻׁחְצָתִֻ קִרָבִוֲ וִכָרֻעֻיו וָנֻׁתְתִֻ עְל נָתֻחֻיו וָעְל רֲאשׁ֖וֲ:
וָהִקָטְרָתִֻ אּת כִֻל הֻאְיִל הְמִִזָבִֹחֻה עֲלֻה הוִא לְׁיהוֲֻה רֹיחְ נִיחוֲחְ אִשִּ֖ה לְׁיהוֲֻה הׁוִא:
