הלכות תשובה - פרק ששי

פְּסוּקִים הַרְבֵּה יֵשׁ בַּתּוֹרָה וּבְדִבְרֵי נְבִיאִים שֶׁהֵן נִרְאִין כְּסוֹתְרִין עִקָּר זֶה וְנִכְשָׁלִין בָּהֶן רֹב הָאָדָם וְיַעֲלֶה עַל דַּעְתָּן מֵהֶן שֶׁהַקָּדוֹשׁ בָּרוּךְ הוּא הוּא גּוֹזֵר עַל הָאָדָם לַעֲשׂוֹת רָעָה אוֹ טוֹבָה וְשֶׁאֵין לִבּוֹ שֶׁל אָדָם מָסוּר לוֹ לְהַטּוֹתוֹ לְכָל אֲשֶׁר יִרְצֶה. וַהֲרֵי אֲנִי מְבָאֵר עִקָּר גָּדוֹל שֶׁמִּמֶּנּוּ תֵּדַע פֵּרוּשׁ כָּל אוֹתָן הַפְּסוּקִים. בִּזְמַן שֶׁאָדָם אֶחָד אוֹ אַנְשֵׁי מְדִינָה חוֹטְאִים וְעוֹשֶׂה הַחוֹטֵא חֵטְא שֶׁעוֹשֶׂה מִדַּעְתּוֹ וּבִרְצוֹנוֹ כְּמוֹ שֶׁהוֹדַעְנוּ רָאוּי לְהִפָּרַע מִמֶּנּוּ וְהַקָּדוֹשׁ בָּרוּךְ הוּא יוֹדֵעַ אֵיךְ יִפָּרַע. יֵשׁ חֵטְא שֶׁהַדִּין נוֹתֵן שֶׁנִּפְרָעִים מִמֶּנּוּ עַל חֶטְאוֹ בָּעוֹלָם הַזֶּה בְּגוּפוֹ אוֹ בְּמָמוֹנוֹ אוֹ בְּבָנָיו הַקְּטַנִּים שֶׁבָּנָיו שֶׁל אָדָם הַקְּטַנִּים שֶׁאֵין בָּהֶם דַּעַת וְלֹא הִגִּיעוּ לִכְלַל מִצְוַת כְּקִנְיָנוֹ הֵן וּכְתִיב עַד שֶׁיֵּעָשֶׂה אִישׁ. וְיֵשׁ חֵטְא שֶׁהַדִּין נוֹתֵן שֶׁנִּפְרָעִין מִמֶּנּוּ לָעוֹלָם הַבָּא וְאֵין לָעוֹבֵר עָלָיו שׁוּם נֵזֶק בָּעוֹלָם הַזֶּה. וְיֵשׁ חֵטְא שֶׁנִּפְרָעִין מִמֶּנּוּ בָּעוֹלָם הַזֶּה וְלָעוֹלָם הַבָּא:

בַּמֶּה דְּבָרִים אֲמוּרִים בִּזְמַן שֶׁלֹּא עָשָׂה תְּשׁוּבָה אֲבָל אִם עָשָׂה תְּשׁוּבָה הַתְּשׁוּבָה כִּתְרִיס לִפְנֵי הַפֻּרְעָנוּת. וּכְשֵׁם שֶׁהָאָדָם חוֹטֵא מִדַּעְתּוֹ וּבִרְצוֹנוֹ כָּךְ הוּא עוֹשֶׂה תְּשׁוּבָה מִדַּעְתּוֹ וּבִרְצוֹנוֹ:

וְאֶפְשָׁר שֶׁיֶּחְטָא אָדָם חֵטְא גָּדוֹל אוֹ חֲטָאִים רַבִּים עַד שֶׁיִּתֵּן הַדִּין לִפְנֵי דַּיַן הָאֱמֶת שֶׁיְּהֵא הַפֵּרָעוֹן מִזֶּה הַחוֹטֵא עַל חֲטָאִים אֵלּוּ שֶׁעָשָׂה בִּרְצוֹנוֹ וּמִדַּעְתּוֹ שֶׁמּוֹנְעִין מִמֶּנּוּ הַתְּשׁוּבָה וְאֵין מַנִּיחִין לוֹ רְשׁוּת לָשׁוּב מֵרִשְׁעוֹ כְּדֵי שֶׁיָּמוּת וְיֹאבַד בְּחֶטְאוֹ שֶׁעָשָׂה. הוּא שֶׁהַקָּדוֹשׁ בָּרוּךְ הוּא אָמַר עַל יְדֵי יְשַׁעְיָהוּ וְגוֹ'. וְכֵן הוּא אוֹמֵר . כְּלוֹמַר חָטְאוּ בִּרְצוֹנָם וְהִרְבּוּ לִפְשֹׁעַ עַד שֶׁנִּתְחַיְּבוּ לִמְנֹעַ מֵהֶן הַתְּשׁוּבָה שֶׁהִיא הַמַּרְפֵּא. לְפִיכָךְ כָּתוּב בַּתּוֹרָה . לְפִי שֶׁחָטָא מֵעַצְמוֹ תְּחִלָּה וְהֵרֵעַ לְיִשְׂרָאֵל הַגָּרִים בְּאַרְצוֹ שֶׁנֶּאֱמַר . נָתַן הַדִּין לִמְנֹעַ הַתְּשׁוּבָה מִמֶּנּוּ עַד שֶׁנִּפְרַע מִמֶּנּוּ. לְפִיכָךְ חִזֵּק הַקָּדוֹשׁ בָּרוּךְ הוּא אֶת לִבּוֹ. וְלָמָּה הָיָה שׁוֹלֵחַ לוֹ בְּיַד משֶׁה וְאוֹמֵר שְׁלַח וַעֲשֵׂה תְּשׁוּבָה וּכְבָר אָמַר לוֹ הַקָּדוֹשׁ בָּרוּךְ הוּא אֵין אַתָּה מְשַׁלֵּחַ שֶׁנֶּאֱמַר וְגוֹ' . כְּדֵי לְהוֹדִיעַ לְבָאֵי הָעוֹלָם שֶׁבִּזְמַן שֶׁמּוֹנֵעַ הַקָּדוֹשׁ בָּרוּךְ הוּא הַתְּשׁוּבָה לַחוֹטֵא אֵינוֹ יָכוֹל לָשׁוּב אֶלָּא יָמוּת בְּרִשְׁעוֹ שֶׁעָשָׂה בִּתְחִלָּה בִּרְצוֹנוֹ. וְכֵן סִיחוֹן לְפִי עֲוֹנוֹת שֶׁהָיוּ לוֹ נִתְחַיֵּב לְמָנְעוֹ מִן הַתְּשׁוּבָה שֶׁנֶּאֱמַר . וְכֵן הַכְּנַעֲנִים לְפִי תּוֹעֲבוֹתֵיהֶן מָנַע מֵהֶן הַתְּשׁוּבָה עַד שֶׁעָשׂוּ מִלְחָמָה עִם יִשְׂרָאֵל. שֶׁנֶּאֱמַר . וְכֵן יִשְׂרָאֵל בִּימֵי אֵלִיָּהוּ לְפִי שֶׁהִרְבּוּ לִפְשֹׁעַ מָנַע מֵאוֹתָן הַמַּרְבִּים לִפְשֹׁעַ תְּשׁוּבָה שֶׁנֶּאֱמַר כְּלוֹמַר מָנַעְתָּ מֵהֶן הַתְּשׁוּבָה. נִמְצֵאתָ אוֹמֵר שֶׁלֹּא גָּזַר הָאֵל עַל פַּרְעֹה לְהָרַע לְיִשְׂרָאֵל. וְלֹא עַל סִיחוֹן לַחֲטֹא בְּאַרְצוֹ. וְלֹא עַל הַכְּנַעֲנִים לְהַתְעִיב. וְלֹא עַל יִשְׂרָאֵל לַעֲבֹד עֲבוֹדָה זָרָה. אֶלָּא כֻּלָּן חָטְאוּ מֵעַצְמָן וְכֻלָּן נִתְחַיְּבוּ לִמְנֹעַ מֵהֶן הַתְּשׁוּבָה: (-)

וּכְעִנְיָן זֶה שׁוֹאֲלִין הַצַּדִּיקִים וְהַנְּבִיאִים בִּתְפִלָּתָם מֵאֵת ה' לְעָזְרָם עַל הָאֱמֶת. כְּמוֹ שֶׁאָמַר דָּוִד . כְּלוֹמַר אַל יִמְנָעוּנִי חֲטָאַי דֶּרֶךְ הָאֱמֶת שֶׁמִּמֶּנָּה אֵדַע דַּרְכְּךָ וְיִחוּד שְׁמֶךָ. וְכֵן זֶה שֶׁאָמַר כְּלוֹמַר תַּנִּיחַ רוּחִי לַעֲשׂוֹת חֶפְצְךָ וְאַל יִגְרְמוּ לִי חֲטָאַי לְמָנְעֵנִי מִתְּשׁוּבָה אֶלָּא תִּהְיֶה הָרְשׁוּת בְּיָדִי עַד שֶׁאֶחֱזֹר וְאָבִין וְאֵדַע דֶּרֶךְ הָאֱמֶת. וְעַל דֶּרֶךְ זוֹ כָּל הַדּוֹמֶה לִפְסוּקִים אֵלּוּ:

וּמַהוּ זֶה שֶׁאָמַר דָּוִד וְגוֹ'. זֶה שֶׁשָּׁלַח נְבִיאִים לָהֶם מוֹדִיעִים דַּרְכֵי ה' וּמַחְזִירִין אוֹתָן בִּתְשׁוּבָה. וְעוֹד שֶׁנָּתַן בָּהֶם כֹּחַ לִלְמֹד וּלְהָבִין. שֶׁמִּדָּה זוֹ בְּכָל אָדָם שֶׁכָּל זְמַן שֶׁהוּא נִמְשָׁךְ בְּדַרְכֵי הַחָכְמָה וְהַצֶּדֶק מִתְאַוֶּה לָהֶן וְרוֹדֵף אוֹתָם. וְהוּא מַה שֶּׁאָמְרוּ רַבּוֹתֵינוּ זִכְרוֹנָם לִבְרָכָה בָּא לִטַּהֵר מְסַיְּעִין אוֹתוֹ כְּלוֹמַר יִמְצָא עַצְמוֹ נֶעֱזָר עַל הַדָּבָר. * וַהֲלֹא כָּתוּב בַּתּוֹרָה , הֲרֵי גָּזַר עַל הַמִּצְרִיִּים לַעֲשׂוֹת רַע. וּכְתִיב , הֲרֵי גָּזַר עַל יִשְׂרָאֵל לַעֲבֹד עֲבוֹדָה זָרָה. וְלָמָּה נִפְרַע מֵהֶן. לְפִי שֶׁלֹּא גָּזַר עַל אִישׁ פְּלוֹנִי הַיָּדוּעַ שֶׁיִּהְיֶה הוּא הַזּוֹנֶה אֶלָּא כָּל אֶחָד וְאֶחָד מֵאוֹתָן הַזּוֹנִים לַעֲבֹד עֲבוֹדָה זָרָה אִלּוּ לֹא רָצָה לַעֲבֹד לֹא הָיָה עוֹבֵד. וְלֹא הוֹדִיעוֹ הַבּוֹרֵא אֶלָּא מִנְהָגוֹ שֶׁל עוֹלָם. הָא לְמָה זֶה דּוֹמֶה לְאוֹמֵר הָעָם הַזֶּה יִהְיֶה בָּהֶן צַדִּיקִים וּרְשָׁעִים. לֹא מִפְּנֵי זֶה יֹאמַר הָרָשָׁע כְּבָר נִגְזַר עָלָיו שֶׁיִּהְיֶה רָשָׁע מִפְּנֵי שֶׁהוֹדִיעַ לְמשֶׁה שֶׁיִּהְיוּ רְשָׁעִים בְּיִשְׂרָאֵל. כָּעִנְיָן שֶׁנֶּאֱמַר . וְכֵן הַמִּצְרִיִּים כָּל אֶחָד וְאֶחָד מֵאוֹתָן הַמְּצֵרִים וְהַמֵּרֵעִים לְיִשְׂרָאֵל אִלּוּ לֹא רָצָה לְהָרַע לָהֶם הָרְשׁוּת בְּיָדוֹ. שֶׁלֹּא גָּזַר עַל אִישׁ יָדוּעַ אֶלָּא הוֹדִיעוֹ שֶׁסּוֹף זַרְעוֹ עָתִיד לְהִשְׁתַּעְבֵּד בְּאֶרֶץ לֹא לָהֶם. וּכְבָר אָמַרְנוּ שֶׁאֵין כֹּחַ בָּאָדָם לֵידַע הֵיאַךְ יֵדַע הַקָּדוֹשׁ בָּרוּךְ הוּא דְּבָרִים הָעֲתִידִין לִהְיוֹת:
