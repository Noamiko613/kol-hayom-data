הלכות טומאת צרעת - פרק ארבעה עשר צָרַעַת ] בָּתִּים כִּשְׁנֵי גְּרִיסִין זֶה בְּצַד זֶה. נִמְצָא רֹחב הַנֶּגַע כְּמוֹ מְקוֹם צְמִיחַת שֵׁשׁ שְׂעָרוֹת בַּגּוּף וְאָרְכּוֹ כִּשְׁתֵּים עֶשְׂרֵה שְׂעָרוֹת בְּרִבּוּעַ. וְהַפָּחוֹת מִשִּׁעוּר זֶה בְּבָתִּים טָהוֹר. וְכָל הַשִּׁעוּרִין הֲלָכָה לְמשֶׁה מִסִּינַי: שְׁלֹשָׁה סִימָנֵי טֻמְאָה יֵשׁ בְּבָתִּים. מַרְאֶה יְרַקְרַק וַאֲדַמְדַּם וְהַפִּשְׂיוֹן. וְכֻלָּן מְפֹרָשִׁין בַּתּוֹרָה. וּשְׁנֵי הַמַּרְאוֹת מִצְטָרְפִין זֶה עִם זֶה. וּפִשְׂיוֹן הַסָּמוּךְ כָּל שֶׁהוּא וְהָרָחוֹק כִּגְרִיס. אֲבָל הַחוֹזֵר אַחַר הִטּוֹחַ כִּשְׁנֵי גְּרִיסִין: אֵין נִגְעֵי בָּתִּים מְטַמְּאִין עַד שֶׁיִּהְיֶה מַרְאֵה הַנֶּגַע שָׁפָל מִן הַקִּיר שֶׁנֶּאֱמַר שֶׁיִּהְיוּ שׁוֹקְעִין בַּקִּירוֹת. וּבִשְׁנֵי הַמַּרְאוֹת מַסְגִּירִין אוֹ מַחְלִיטִין וּבְפִשְׂיוֹן נוֹתְצִין. וְאִם פָּשָׂה אַחֲרֵי הִטּוֹחַ נוֹתֵץ אֶת כָּל הַבַּיִת כְּמוֹ שֶׁיִּתְבָּאֵר: כְּשֶׁיֵּרָאֶה נֶגַע בַּבַּיִת אֲפִלּוּ חָכָם שֶׁיּוֹדֵעַ וַדַּאי שֶׁזֶּה נֶגַע לֹא יִגְזֹר וְיֹאמַר נֶגַע נִרְאֶה לִי בַּבַּיִת אֶלָּא אוֹמֵר לַכֹּהֵן . אֲפִלּוּ חֲבִילֵי עֵצִים וַחֲבִילֵי קָנִים וְאַחַר כָּךְ יָבוֹא הַכֹּהֵן וְיִכָּנֵס וְיִרְאֶה הַנֶּגַע: בַּיִת אָפֵל אֵין פּוֹתְחִין בּוֹ חַלּוֹנוֹת לִרְאוֹת אֶת נִגְעוֹ אֶלָּא אִם אֵין הַנֶּגַע נִרְאֶה בּוֹ טָהוֹר. וְאַחַר שֶׁיִּרְאֶה הַכֹּהֵן אֶת הַנֶּגַע יֵצֵא וְיַעֲמֹד עַל פֶּתַח הַבַּיִת בְּצַד הַמַּשְׁקוֹף וְיַסְגִּיר אוֹ יַחְלִיט אוֹ יִפְטֹר שֶׁנֶּאֱמַר . לֹא שֶׁיַּסְגִּיר וְהוּא בְּתוֹךְ בֵּיתוֹ אוֹ בְּתוֹךְ בַּיִת הַמְנֻגָּע אוֹ תַּחַת הַמַּשְׁקוֹף אֶלָּא בְּצַד פִּתְחוֹ. וְאִם עָמַד תַּחַת הַמַּשְׁקוֹף אוֹ שֶׁהָלַךְ לְבֵיתוֹ וְהִסְגִּיר הֲרֵי זֶה מֻסְגָּר: אֵין הַבַּיִת מִטַּמֵּא בִּנְגָעִים עַד שֶׁיִּהְיֶה בּוֹ אַרְבַּע אַמּוֹת עַל אַרְבַּע אַמּוֹת אוֹ יֶתֶר וִיהֵא לוֹ אַרְבָּעָה כְּתָלִים וְהוּא בָּנוּי בָּאָרֶץ בַּאֲבָנִים וְעָפָר וְעֵצִים שֶׁנֶּאֱמַר . אֲבָל אִם הָיָה בּוֹ פָּחוֹת מֵאַרְבַּע אַמּוֹת עַל אַרְבַּע אַמּוֹת אוֹ שֶׁהָיָה עָגל אוֹ שֶׁהָיָה בַּעַל שְׁלֹשָׁה כְּתָלִים אוֹ בַּעַל חֲמִשָּׁה אוֹ שֶׁהָיָה בָּנוּי בִּסְפִינָה אוֹ תָּלוּי עַל אַרְבַּע קוֹרוֹת אֵינוֹ מִטַּמֵּא בִּנְגָעִים. הָיָה בָּנוּי עַל אַרְבָּעָה עַמּוּדִים מִטַּמֵּא בִּנְגָעִים: (-) כַּמָּה אֲבָנִים יִהְיוּ בּוֹ אֵין פָּחוֹת מִשְּׁמוֹנֶה שְׁתֵּי אֲבָנִים בְּכָל כֹּתֶל כְּדֵי שֶׁיִּהְיֶה כָּל כֹּתֶל רָאוּי לְנֶגַע שֶׁאֵין הַבַּיִת מִטַּמֵּא בִּנְגָעִים עַד שֶׁיֵּרָאֶה בּוֹ כִּשְׁנֵי גְּרִיסִין עַל שְׁתֵּי אֲבָנִים שֶׁנֶּאֱמַר . וְכַמָּה עֵצִים יִהְיוּ בּוֹ כְּדֵי לִתֵּן תַּחַת הַמַּשְׁקוֹף. וְעָפָר כְּדֵי לָתֵת בֵּין פַּצִּים לַחֲבֵרוֹ. אֲבָל אִם הָיָה בּוֹ פָּחוֹת מִשִּׁעוּרִים אֵלּוּ אֵינוֹ מִטַּמֵּא בִּנְגָעִים: הַלְּבֵנִים וְהַשַּׁיִשׁ אֵינָן חֲשׁוּבִין כַּאֲבָנִים. בַּיִת שֶׁאֶחָד מִצִּדָּיו מְחֻפֶּה בְּשַׁיִשׁ וְאֶחָד בְּסֶלַע וְאֶחָד בִּלְבֵנִים וְאֶחָד בְּעָפָר אֵינוֹ מִטַּמֵּא בִּנְגָעִים: בַּיִת שֶׁלֹּא הָיוּ בּוֹ אֲבָנִים וְעֵצִים כַּשִּׁעוּר * וְנִרְאָה בּוֹ נֶגַע וְאַחַר כָּךְ הֵבִיא לוֹ אֲבָנִים וְעֵצִים וְעָפָר טָהוֹר: בַּיִת שֶׁסִּכְּכוֹ בִּזְרָעִים בִּטְּלָן הוֹאִיל וּמְשַׁמְּשִׁין תַּשְׁמִישׁ עֵץ הֲרֵי הֵן כְּעֵץ. וְאִם נִטְמָא הַבַּיִת נִטְמְאוּ עִמּוֹ טֻמְאָה חֲמוּרָה כְּמוֹ שֶׁיִּתְבָּאֵר: יְרוּשָׁלַיִם וְחוּצָה לָאָרֶץ אֵין מִטַּמְּאִין בִּנְגָעִים שֶׁנֶּאֱמַר וִירוּשָׁלַיִם לֹא נִתְחַלְּקָה לִשְׁבָטִים. וּבָתֵּי הָעַכּוּ''ם שֶׁבְּאֶרֶץ יִשְׂרָאֵל אֵין מִטַּמְּאִין בִּנְגָעִים: הַלּוֹקֵחַ בָּתִּים מִן הָעַכּוּ''ם יֵרָאֶה בַּתְּחִלָּה: בַּיִת שֶׁצִּדּוֹ אֶחָד עַכּוּ''ם וְצִדּוֹ אֶחָד יִשְׂרָאֵל צִדּוֹ אֶחָד בָּאָרֶץ וְצִדּוֹ אֶחָד בְּחוּצָה לָאָרֶץ אֵינוֹ מִטַּמֵּא בִּנְגָעִים. וּשְׁאָר כָּל הַבָּתִּים שֶׁבְּאֶרֶץ יִשְׂרָאֵל מִטַּמְּאִין בִּנְגָעִים בֵּין צְבוּעִים בִּידֵי אָדָם בֵּין צְבוּעִים בִּידֵי שָׁמַיִם: בֵּית הָאִשָּׁה בֵּית הַשֻּׁתָּפִין בֵּית הַכְּנֶסֶת אוֹ בֵּית הַמִּדְרָשׁ שֶׁיֵּשׁ בָּהֶן דִּירָה לְחַזָּנִין אוֹ
