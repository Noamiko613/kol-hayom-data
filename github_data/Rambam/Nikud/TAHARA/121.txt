הלכות כלים - פרק ששה עשר תַּנּוּר גָּדוֹל שְׁיָרָיו אַרְבָּעָה טְפָחִים וְהַקָּטָן שְׁיָרָיו רֻבּוֹ. כֵּיצַד. אִם נִשְׁבַּר בַּגָּדוֹל אַרְבָּעָה וּבַקָּטָן רֻבּוֹ מְקַבֵּל טֻמְאָה פָּחוֹת מִכָּאן אֵינוֹ מְקַבֵּל טֻמְאָה. וְכֵן אִם נִטְמָא וְנִתְּצוֹ עַד שֶׁשִּׁיֵּר בּוֹ פָּחוֹת מֵאַרְבָּעָה בַּגָּדוֹל אוֹ בְּפָחוֹת מֵרֹב בַּקָּטָן טָהוֹר. נִשְׁאַר בּוֹ אַרְבָּעָה אוֹ רֹב בַּקָּטָן עֲדַיִן הוּא בְּטֻמְאָתוֹ. וְהַכִּירָה שְׁיָרֶיהָ שָׁלֹשׁ אֶצְבָּעוֹת. הַכֻּפָּח שֶׁעֲשָׂאָהוּ לַאֲפִיָּה שִׁעוּרוֹ כְּתַנּוּר. לְבִשּׁוּל שִׁעוּרוֹ כְּכִירָה: תַּנּוּר שֶׁנִּטְמָא וְלֹא רָצָה לְכַתְּתוֹ כֵּיצַד מְטַהֲרִין אוֹתוֹ. חוֹלְקוֹ לִשְׁלֹשָׁה חֲלָקִים ] וְגוֹרֵר אֶת הַטְּפֵלָה שֶׁעַל הַחֲרָסִים עַד שֶׁנִּמְצָא כָּל חֶרֶס מֵהֶן עוֹמֵד עַל הָאָרֶץ בְּלֹא טִיט עַל גַּבָּיו. חִלְּקוֹ לִשְׁנַיִם אֶחָד גָּדוֹל וְאֶחָד קָטָן הַגָּדוֹל טָמֵא וְהַקָּטָן טָהוֹר. חִלְּקוֹ לִשְׁנַיִם בְּשָׁוֶה כָּל אֶחָד מֵהֶן טָמֵא לְפִי שֶׁאִי אֶפְשָׁר לְכַוֵּן. אֲבָל טַבְלָא שֶׁל חֶרֶס שֶׁיֵּשׁ לָהּ דֹּפֶן שֶׁנֶּחְלְקָה לִשְׁנַיִם טְהוֹרָה. וְאִם הָיָה אֶחָד גָּדוֹל וְאֶחָד קָטָן הַגָּדוֹל טָמֵא: תַּנּוּר שֶׁחִלְּקוֹ לִשְׁלֹשָׁה וְאֶחָד גָּדוֹל כִּשְׁנַיִם הַגָּדוֹל טָמֵא וְהַשְּׁנַיִם הַקְּטַנִּים טְהוֹרִין. חֲתָכוֹ חֻלְיוֹת לְרָחְבּוֹ אִם הָיָה גֹּבַהּ כָּל חֻלְיָא וְחֻלְיָא פָּחוֹת מֵאַרְבָּעָה טְפָחִים טָהוֹר. חָזַר וְסִדֵּר הַחֻלְיוֹת זוֹ עַל גַּבֵּי זוֹ וְהִמְרִיחַ עֲלֵיהֶם בְּטִיט וְהֶחֱזִירוֹ תַּנּוּר כְּשֶׁהָיָה הֲרֵי זוֹ כְּמוֹ שֶׁעָשָׂה תַּנּוּר אַחֵר וְאֵינוֹ מְקַבֵּל טֻמְאָה אֶלָּא מִכָּאן וּלְהַבָּא וְהוּא שֶׁיַּסִּיקֶנּוּ כְּדֵי לֶאֱפוֹת בּוֹ סֻפְגָּנִין אַחַר שֶׁמֵּרְחוֹ. הִרְחִיק מִמֶּנּוּ אֶת הַטְּפֵלָה וְנָתַן חוֹל אוֹ צְרוֹרוֹת בֵּין הַחֻלְיוֹת וּבֵין הַטְּפֵלָה אֵינוֹ מְקַבֵּל טֻמְאָה לְעוֹלָם וְזֶהוּ שֶׁאָמְרוּ הַנִּדָּה וְהַטְּהוֹרָה אוֹפוֹת בּוֹ וְהוּא טָהוֹר. הָיְתָה בּוֹ חֻלְיָא אַחַת שֶׁיֵּשׁ בָּהּ אַרְבָּעָה טְפָחִים הִיא מִתְטַמְּאָה בְּמַגָּע וְלֹא מִתְטַמֵּאת בַּאֲוִיר וּשְׁאָר הַחֻלְיוֹת טְהוֹרוֹת: תַּנּוּר שֶׁבָּא מְחֻתָּךְ מִבֵּית הָאֻמָּן וְעָשָׂה לוֹ סְמוֹכִין שֶׁמְּקַבְּצִין אוֹתוֹ לִהְיוֹת כְּאֶחָד וּנְתָנָם עָלָיו וְהוּא טָהוֹר וְנִטְמָא כְּשֶׁיָּסִיר אֶת הַסּוֹמְכִין אוֹתוֹ יִטְהַר וַאֲפִלּוּ הֶחְזִירָן הֲרֵי הוּא טָהוֹר. מֵרְחוֹ בְּטִיט מְקַבֵּל טֻמְאָה לְהַבָּא וְאֵינוֹ צָרִיךְ לְהַסִּיקוֹ שֶׁכְּבָר הֻסַּק: תַּנּוּר שֶׁחֲתָכוֹ חֻלְיוֹת וְנָתַן חוֹל בֵּין חֻלְיָא לְחֻלְיָא וְטָח בְּטִיט עַל הַכּל מִבַּחוּץ הֲרֵי זֶה מְקַבֵּל טֻמְאָה: יוֹרַת הָעַרְבִיִּים שֶׁהוּא חוֹפֵר בָּאָרֶץ וְטָח בְּטִיט אִם יוּכַל הַטִּיט לַעֲמֹד בִּפְנֵי עַצְמוֹ מְקַבֶּלֶת טֻמְאָה וְאִם לָאו טָהוֹר: הַמֵּבִיא שִׁבְרֵי חֶרֶס וְדִבְּקָן זֶה בָּזֶה וַעֲשָׂאָן תַּנּוּר וְעָשָׂה לוֹ טְפֵלָה מִבַּיִת וּמִבַּחוּץ וְהִסִּיקוֹ הֲרֵי זֶה מְקַבֵּל טֻמְאָה אַף עַל פִּי שֶׁאֵין בְּכָל אֶחָד מֵהֶן כַּשִּׁעוּר: פִּטָּם שֶׁקִּרְזְלוֹ וַעֲשָׂאַהוּ תַּנּוּר וְעָשָׂה לוֹ טְפֵלָה מִחוּץ אַף עַל פִּי שֶׁמְּקַבֵּל עַל דָּפְנוֹתָיו כַּשִּׁעוּר הֲרֵי זֶה טָהוֹר שֶׁכְּלִי חֶרֶס שֶׁטָּהַר אֵין לוֹ טֻמְאָה לְעוֹלָם אֶלָּא אִם כֵּן עֲשָׂאָהוּ תַּנּוּר וְעָשָׂה לוֹ טְפֵלָה מִבַּיִת וּמִבַּחוּץ: תַּנּוּר שֶׁל סְדָקִין שֶׁעָשָׂה טְפֵלָה לְכָל אֶחָד וְאֶחָד וּמְקוֹם הַסְּדָקִין מְגֻלֶּה אֵינוֹ מְקַבֵּל טֻמְאָה. נָתַן טִיט אוֹ סִיד אוֹ גַּפְסִיס עַל גַּבֵּי הַסְּדָקִין הֲרֵי זֶה מִתְטַמֵּא. נָתַן עֲלֵיהֶן חַרְסִית אוֹ זֶפֶת וְגָפְרִית שַׁעֲוָה וּשְׁמָרִים בָּצֵק אוֹ גְּלָלִים טָהוֹר. זֶה הַכְּלָל דָּבָר שֶׁאֵין עוֹשִׂין מִמֶּנּוּ תַּנּוּר אֵינוֹ מְחַבֵּר אֶת הַסְּדָקִין: סֶדֶק תַּנּוּר שֶׁנְּתָנוֹ כְּלַפֵּי זָוִית וּמֵרֵחַ בְּטִיט מִן הַצְּדָדִין טָהוֹר: דַּף שֶׁל תַּנּוּר שֶׁנְּתָנוֹ בְּזָוִית לִהְיוֹת אוֹפֶה בּוֹ טָהוֹר. וְאִם יֵשׁ בּוֹ רֹב תַּנּוּר מְקַבֵּל טֻמְאָה: תַּנּוּר שֶׁנָּתַן בּוֹ עָפָר עַד חֶצְיוֹ אִם נִטְמָא מֵאֲוִירוֹ בִּלְבַד אֵינוֹ מִתְטַמֵּא אֶלָּא מֵעָפָר וּלְמַעְלָה וְאִם נִטְמָא בְּמַגָּע וְנָגְעָה טֻמְאָה בּוֹ מִתּוֹכוֹ נִטְמָא כֻּלּוֹ אַף מֵעָפָר וּלְמַטָּה: הַכִּירָה מְקוֹם שְׁפִיתַת שְׁתֵּי קְדֵרוֹת וְהַכֻּפָּח מְקוֹם שְׁפִיתַת קְדֵרָה אַחַת. לְפִיכָךְ כִּירָה שֶׁנֶּחְלְקָה לְאָרְכָּהּ טְהוֹרָה לְרָחְבָּהּ עֲדַיִן מְקַבֶּלֶת טֻמְאָה. הַכֻּפָּח שֶׁנֶּחְלַק בֵּין לְאָרְכּוֹ בֵּין לְרָחְבּוֹ טָהוֹר: הַקְּלָתוֹת שֶׁל בַּעֲלֵי בָּתִּים שֶׁנִּפְחֲתָה קַרְקָעִיתָהּ אִם הָיָה עֹמֶק הַפְּחָת פָּחוֹת מִשְּׁלֹשָׁה טְפָחִים הֲרֵי זוֹ מְקַבֶּלֶת טֻמְאָה שֶׁאִם יַסִּיק בַּפְּחָת מִלְּמַטָּה תִּתְבַּשֵּׁל הַקְּדֵרָה מִלְּמַעְלָה. הָיָה הַפְּחָת עָמֹק שְׁלֹשָׁה אוֹ יֶתֶר אֵינָהּ מְקַבֶּלֶת טֻמְאָה שֶׁהֲרֵי הָאֵשׁ רְחוֹקָה מִן הַקְּדֵרָה וְאֵינָהּ בְּשֵׁלָה. נָתַן אֶבֶן אוֹ צְרוֹר עַל פִּי הַפְּחָת עֲדַיִן הִיא בְּטָהֳרָתָהּ. מֵרְחוֹ בְּטִיט נַעֲשֵׂת הָאֶבֶן קַרְקַע הַכִּירָה וּמְקַבֶּלֶת
