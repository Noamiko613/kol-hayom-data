הלכות שחיטה - פרק ארבעה עשר

] מִצְוַת עֲשֵׂה לְכַסּוֹת דּם שִׁחִיטַת חַיָּה טְהוֹרָה אוֹ עוֹף טָהוֹר שֶׁנֶּאֱמַר . לְפִיכָךְ חַיָּב לְבָרֵךְ קֹדֶם שֶׁיְּכַסֶּה בָּרוּךְ אַתָּה ה' אֱלֹקֵינוּ מֶלֶךְ הָעוֹלָם אֲשֶׁר קִדְּשָׁנוּ בְּמִצְוֹתָיו וְצִוָּנוּ עַל כִּסּוּי הַדָּם:

כִּסּוּי הַדָּם נוֹהֵג בִּמְזֻמָּן וּבְשֶׁאֵינוֹ מְזֻמָּן. לֹא נֶאֱמַר אֶלָּא בַּהוֹוֶה. וְנוֹהֵג בְּחֻלִּין וְלֹא בְּמֻקְדָּשִׁין. בֵּין קָדְשֵׁי הַמִּזְבֵּחַ בֵּין קָדְשֵׁי בֶּדֶק הַבַּיִת. וְאִם עָבַר וּשְׁחָטָן אֵינוֹ חַיָּב לְכַסּוֹת אֶת דָּמָן:

שָׁחַט חַיָּה וְעוֹף וְאַחַר כָּךְ הִקְדִּישָׁן אוֹ הִקְדִּישׁ אֶת הַדָּם חַיָּב לְכַסּוֹת:

כִּלְאַיִם הַבָּא מִבְּהֵמָה וְחַיָּה וְכֵן בְּרִיָּה שֶׁהִיא סְפֵק בְּהֵמָה אוֹ חַיָּה צָרִיךְ לְכַסּוֹת וְאֵינוֹ מְבָרֵךְ. ] הַשּׁוֹחֵט לְחוֹלֶה בְּשַׁבָּת חַיָּב לְכַסּוֹת לְאַחַר שַׁבָּת. וְכֵן הַשּׁוֹחֵט סָפֵק אוֹ כִּלְאַיִם בְּיוֹם טוֹב מְכַסֶּה דָּמוֹ לְאַחַר יוֹם טוֹב:

הַשּׁוֹחֵט עוֹפוֹת וּמִינֵי חַיָּה בְּמָקוֹם אֶחָד מְבָרֵךְ בְּרָכָה אַחַת לְכֻלָּן וְכִסּוּי אֶחָד לְכֻלָּן:

] דָּם שֶׁנִּתְעָרֵב בַּמַּיִם אִם יֵשׁ בּוֹ מַרְאֵה דָּם חַיָּב לְכַסּוֹת וְאִם לָאו פָּטוּר. נִתְעָרֵב בְּיַיִן אוֹ בְּדַם בְּהֵמָה רוֹאִין אוֹתָן כְּאִלּוּ הֵם מַיִם. אִם אֶפְשָׁר שֶׁיִּרְאֶה מַרְאֵה הַדָּם שֶׁחַיָּב לְכַסּוֹתוֹ כְּשִׁעוּר זֶה אִלּוּ הָיָה מַיִם חַיָּב לְכַסּוֹת הַכּל וְאִם לָאו פָּטוּר:

כִּסָּהוּ וְנִתְגַּלָּה אֵינוֹ חַיָּב לְכַסּוֹתוֹ פַּעַם אַחֶרֶת. כִּסַּתְהוּ הָרוּחַ אֵינוֹ חַיָּב לְכַסּוֹת. חָזַר וְנִתְגַּלָּה אַחַר שֶׁכִּסַּתְהוּ הָרוּחַ חַיָּב לְכַסּוֹת:

* דָּם הַנִּתָּז וְשֶׁעַל הַסַּכִּין אִם אֵין שָׁם דָּם אֶלָּא הוּא חַיָּב לְכַסּוֹת:

שָׁחַט וְנִבְלַע הַדָּם בַּקַּרְקַע אִם רִשּׁוּמוֹ נִכָּר חַיָּב לְכַסּוֹת. וְאִם לָאו הֲרֵי זֶה כְּמִי שֶׁכִּסַּתְהוּ הָרוּחַ וּפָטוּר מִלְּכַסּוֹת:

אֵין חַיָּב בְּכִסּוּי אֶלָּא דָּם שְׁחִיטָה הָרְאוּיָה לַאֲכִילָה שֶׁנֶּאֱמַר . לְפִיכָךְ הַשּׁוֹחֵט וְנִמְצֵאת טְרֵפָה. אוֹ הַשּׁוֹחֵט חֻלִּין בַּעֲזָרָה. אוֹ הַשּׁוֹחֵט חַיָּה וְעוֹף שֶׁנִּגְמַר דִּינָן לִסְקִילָה וְהַשּׁוֹחֵט וְנִתְנַבְּלָה בְּיָדוֹ פָּטוּר מִלְּכַסּוֹת. וְכֵן חֵרֵשׁ שׁוֹטֶה וְקָטָן שֶׁשָּׁחֲטוּ בֵּינָן לְבֵין עַצְמָן פְּטוּרִין מִלְּכַסּוֹת דַּם שְׁחִיטָתָן:

] בַּמֶּה מְכַסִּין בְּסִיד וּבְגַפְסִית בְּזֶבֶל דַּק וּבְחוֹל דַּק שֶׁאֵין הַיּוֹצֵר צָרִיךְ לְכָתְשׁוֹ וּבִשְׁחִיקַת אֲבָנִים וַחֲרָשִׁים וּבִנְעֹרֶת שֶׁל פִּשְׁתָּן דַּקָּה וּבִנְסֹרֶת חָרָשִׁים דַּקָּה וּבִלְבֵנָה וְחַרְסִית וּמְגוּפָה שֶׁכְּתָשָׁהּ שֶׁכָּל אֵלּוּ מִין עָפָר הֵן. אֲבָל אִם כָּפָה עָלָיו כְּלִי אוֹ כִּסָּהוּ בַּאֲבָנִים אֵין זֶה כִּסּוּי שֶׁנֶּאֱמַר :

לְפִיכָךְ אֵין מְכַסִּים בְּזֶבֶל גַּס וְחוֹל גַּס וְקֶמַח וְסֻבִּין וּמוֹרְסָן וּשְׁחִיקַת כְּלֵי מַתָּכוֹת לְפִי שֶׁאֵין אֵלּוּ מִין עָפָר. חוּץ מִשְּׁחִיקַת הַזָּהָב בִּלְבַד שֶׁמְּכַסִּין בָּהֶם מִפְּנֵי שֶׁנִּקְרָא עָפָר שֶׁנֶּאֱמַר וְאוֹמֵר :

מְכַסִּין בְּשִׁחוֹר וְהוּא פִּיחַ הַכִּבְשָׁן וּבְכוֹחַל וּבְנִקְרַת פְּסִילִים וּבְאֵפֶר בֵּין אֵפֶר עֵצִים בֵּין אֵפֶר בְּגָדִים אֲפִלּוּ אֵפֶר בָּשָׂר שֶׁנִּשְׂרַף שֶׁהֲרֵי כָּתוּב . וּמֻתָּר לְכַסּוֹת בְּעָפָר עִיר הַנִּדַּחַת:

] הַשּׁוֹחֵט צָרִיךְ לִתֵּן עָפָר לְמַטָּה וְאַחַר כָּךְ יִשְׁחֹט בּוֹ וְאַחַר כָּךְ יְכַסֶּה בְּעָפָר אֲבָל לֹא יִשְׁחֹט בִּכְלִי וִיכַסֶּה בְּעָפָר:

וּמִי שֶׁשָּׁחַט הוּא יְכַסֶּה שֶׁנֶּאֱמַר . וְאִם לֹא כִּסָּהוּ וְרָאָהוּ אַחֵר חַיָּב לְכַסּוֹת שֶׁזּוֹ מִצְוָה בִּפְנֵי עַצְמָהּ וְאֵינָהּ תְּלוּיָה בַּשּׁוֹחֵט לְבַד:

וּכְשֶׁמְּכַסֶּה לֹא יְכַסֶּה בְּרַגְלוֹ אֶלָּא בְּיָדוֹ אוֹ בְּסַכִּין אוֹ בִּכְלִי כְּדֵי שֶׁלֹּא יִנְהֹג בּוֹ מִנְהַג בִּזָּיוֹן וְיִהְיוּ מִצְוֹת בְּזוּיוֹת עָלָיו. שֶׁאֵין הַכָּבוֹד לְעַצְמָן שֶׁל מִצְוֹת אֶלָּא לְמִי שֶׁצִּוָּה בָּהֶן בָּרוּךְ הוּא וְהִצִּילָנוּ מִלְּמַשֵּׁשׁ בַּחשֶׁךְ וְעָרַךְ אוֹתָנוּ נֵר לְיַשֵּׁר הַמַּעֲקַשִּׁים וְאוֹר לְהוֹרוֹת נְתִיבוֹת הַיּשֶׁר. וְכֵן הוּא אוֹמֵר :

סְלִיקוּ לְהוּ הִלְכוֹת שְׁחִיטָה בְּסַ''ד

נִגְמַר סֵפֶר חֲמִישִׁי וְהוּא סֵפֶר קְדֻשָׁה. וּמִנְיַן פְּרָקָיו שְׁלשָׁה וַחֲמִשִׁים.

הִלְכוֹת אִסּוּרֵי בִּיאָה שְׁנַיִם וְעֶשְׂרִים פְּרָקִים: הִלְכוֹת מַאֲכָלוֹת אֲסוּרוֹת שִׁבְעָה עָשָׂר פְּרָקִים: הִלְכוֹת שְׁחִיטָה אַרְבָּעָה עָשָׂר פְּרָקִים:

וּבְכָאן נִשְׁלַם הַחֵלֶק הַשֵׁנִי. ה' בְּמַעגְּלֵי צֶדֶק יַנְחֵנִי:
