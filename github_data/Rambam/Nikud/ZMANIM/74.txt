דִּין שֶׁל אֶרֶץ יִשְׂרָאֵל קוֹבְעִין עַל חֶשְׁבּוֹן זֶה. הָיָה מִן הַדִּין שֶׁיִּהְיוּ בְּכָל הַמְּקוֹמוֹת עוֹשִׂין יוֹם טוֹב אֶחָד בִּלְבַד אֲפִלּוּ הַמְּקוֹמוֹת הָרְחוֹקוֹת שֶׁבְּחוּצָה לָאָרֶץ כְּמוֹ בְּנֵי אֶרֶץ יִשְׂרָאֵל. שֶׁהַכּל עַל חֶשְׁבּוֹן אֶחָד סוֹמְכִין וְקוֹבְעִין. אֲבָל תַּקָּנַת חֲכָמִים הוּא שֶׁיִּזָּהֲרוּ בְּמִנְהַג אֲבוֹתֵיהֶם שֶׁבִּידֵיהֶם: לְפִיכָךְ כָּל מָקוֹם שֶׁלֹּא הָיוּ שְׁלוּחֵי תִּשְׁרֵי מַגִּיעִין אֵלָיו כְּשֶׁהָיוּ הַשְּׁלוּחִין יוֹצְאִין. יַעֲשׂוּ שְׁנֵי יָמִים וַאֲפִלּוּ בַּזְּמַן הַזֶּה כְּמוֹ שֶׁהָיוּ עוֹשִׂין בִּזְמַן שֶׁבְּנֵי אֶרֶץ יִשְׂרָאֵל שֶׁבִּידֵיהֶן אִם יוֹם אֶחָד יוֹם אֶחָד וְאִם שְׁנֵי יָמִים שְׁנֵי יָמִים: מָקוֹם שֶׁבֵּינוֹ וּבֵין יְרוּשָׁלַיִם עֲשָׂרָה יָמִים אוֹ פָּחוֹת מֵעֲשָׂרָה וְהוּא סוּרְיָא אוֹ חוּצָה לָאָרֶץ וְאֵין לָהֶם מִנְהָג. אוֹ שֶׁהִיא עִיר שֶׁנִּתְחַדְּשָׁה בְּמִדְבַּר אֶרֶץ יִשְׂרָאֵל. אוֹ מָקוֹם שֶׁשָּׁכְנוּ בּוֹ יִשְׂרָאֵל עַתָּה. עוֹשִׂין שְׁנֵי יָמִים כְּמִנְהַג רֹב הָעוֹלָם. וְכָל יוֹם טוֹב שֵׁנִי מִדִּבְרֵי סוֹפְרִים וַאֲפִלּוּ יוֹם טוֹב שֵׁנִי שֶׁל רֹאשׁ הַשָּׁנָה שֶׁהַכּל עוֹשִׂין אוֹתוֹ בַּזְּמַן הַזֶּה: זֶה שֶׁאָנוּ מְחַשְּׁבִין בַּזְּמַן הַזֶּה כָּל אֶחָד וְאֶחָד בְּעִירוֹ וְאוֹמְרִין שֶׁרֹאשׁ חֹדֶשׁ יוֹם פְּלוֹנִי וְיוֹם טוֹב בְּיוֹם פְּלוֹנִי. לֹא בַּחֶשְׁבּוֹן שֶׁלָּנוּ אָנוּ קוֹבְעִין וְלֹא עָלָיו אָנוּ סוֹמְכִין. שֶׁאֵין מְעַבְּרִין שָׁנִים וְקוֹבְעִין חֳדָשִׁים בְּחוּצָה לָאָרֶץ. וְאֵין אָנוּ סוֹמְכִין אֶלָּא עַל חֶשְׁבּוֹן בְּנֵי אֶרֶץ יִשְׂרָאֵל וּקְבִיעָתָם. וְזֶה שֶׁאָנוּ מְחַשְּׁבִין לְגַלּוֹת הַדָּבָר בִּלְבַד הוּא. כֵּיוָן שֶׁאָנוּ יוֹדְעִין שֶׁעַל חֶשְׁבּוֹן זֶה הֵן סוֹמְכִין אָנוּ מְחַשְּׁבִין לֵידַע יוֹם שֶׁקָּבְעוּ בּוֹ בְּנֵי אֶרֶץ יִשְׂרָאֵל אֵי זֶה יוֹם הוּא. וּקְבִיעַת בְּנֵי אֶרֶץ יִשְׂרָאֵל אוֹתוֹ הוּא שֶׁיִּהְיֶה רֹאשׁ חֹדֶשׁ אוֹ יוֹם טוֹב. לֹא
