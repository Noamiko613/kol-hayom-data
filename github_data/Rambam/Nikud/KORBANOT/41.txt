הלכות מחוסרי כפרה - פרק חמישי ] אֵי זֶהוּ תְּנוּךְ אֹזֶן זֶה גָּדֵר הָאֶמְצָעִי. נָתַן עַל צִדֵּי הַבְּהוֹנוֹת כָּשֵׁר. נָתַן עַל צִדֵּי צִדֵּיהֶן פָּסוּל. בֵּין שֶׁנָּתַן הַשֶּׁמֶן עַל דַּם הָאָשָׁם עַצְמוֹ מִלְמַעְלָה בֵּין שֶׁנָּתַן בְּצַד הַדָּם אֲפִלּוּ נִתְקַנֵּחַ הַדָּם קֹדֶם שֶׁיִּתֵּן הַשֶּׁמֶן עַל בְּהוֹנוֹת יָצָא שֶׁנֶּאֱמַר . אֵין לוֹ בֹּהֶן יַד יְמָנִית אוֹ בֹּהֶן רֶגֶל יְמָנִית אוֹ שֶׁאֵין לוֹ אֹזֶן יְמָנִית אֵין לוֹ טָהֳרָה עוֹלָמִית: אֲשַׁם מְצֹרָע שֶׁשְּׁחָטוֹ שֶׁלֹּא לִשְׁמוֹ אוֹ שֶׁלֹּא נָתַן מִדָּמוֹ עַל גַּבֵּי בְּהוֹנוֹת הֲרֵי זֶה עוֹלֶה לַמִּזְבֵּחַ וְטָעוּן נְסָכִים כְּנִסְכֵּי אֲשַׁם מְצֹרָע וְצָרִיךְ הַמְצֹרָע אָשָׁם אַחֵר לְהַכְשִׁירוֹ: הִקְדִּים חַטָּאתוֹ לַאֲשָׁמוֹ לֹא יִהְיֶה אַחֵר מְמָרֵס בְּדָמָהּ אֶלָּא תְּעֵבַּר צוּרָתָהּ וְתֵצֵא לְבֵית הַשְּׂרֵפָה: יֵשׁ לְאָדָם לְהָבִיא אֲשָׁמוֹ הַיּוֹם וְלוֹגוֹ אַחַר עֲשָׂרָה יָמִים. וְאִם רָצָה לְשַׁנּוֹת הַלּוֹג לַאֲשַׁם מְצֹרָע אַחֵר מְשַׁנֶּה אַף עַל פִּי שֶׁקָּדַשׁ בִּכְלִי. חָסַר הַלּוֹג עַד שֶׁלֹּא יָצַק יְמַלְּאֶנּוּ. חָסַר מֵאַחַר שֶׁיָּצַק יָבִיא אַחֵר בַּתְּחִלָּה: יָצַק מִן הַשֶּׁמֶן לְכַפּוֹ וְהִתְחִיל לְהַזּוֹת וְנִשְׁפַּךְ הַלּוֹג אִם עַד שֶׁלֹּא גָּמַר מַתַּן שֶׁבַע נִשְׁפַּךְ הַלּוֹג יָבִיא אַחֵר וְיַתְחִיל בְּמַתַּן שֶׁבַע. גָּמַר מַתַּן שֶׁבַע וְנִשְׁפַּךְ הַלּוֹג יָבִיא אַחֵר וְיַתְחִיל לְכַתְּחִלָּה בַּבְּהוֹנוֹת. הִתְחִיל בַּבְּהוֹנוֹת וְנִשְׁפַּךְ הַלּוֹג קֹדֶם שֶׁיִּגְמֹר יָבִיא אַחֵר וְיַתְחִיל לְכַתְּחִלָּה בַּבְּהוֹנוֹת. גָּמַר הַבְּהוֹנוֹת וְאַחַר כָּךְ נִשְׁפַּךְ הַלּוֹג קֹדֶם שֶׁיִּתֵּן שְׁאָר הַשֶּׁמֶן שֶׁבְּכַפּוֹ עַל רֹאשׁ הַמִּטַּהֵר אֵינוֹ צָרִיךְ לְהָבִיא לוֹג אַחֵר שֶׁמַּתְּנַת הָרֹאשׁ אֵינָהּ מְעַכֶּבֶת שֶׁנֶּאֱמַר : הִקְדִּים מַתַּן שֶׁמֶן לְמַתַּן דָּם יְמַלֵּא הַלּוֹג שֶׁמֶן וְיַחְזֹר וְיִתֵּן הַשֶּׁמֶן אַחַר הַדָּם. הִקְדִּים מַתַּן שֶׁמֶן עַל הַבְּהוֹנוֹת לְמַתַּן שֶׁבַע יְמַלֵּא הַלּוֹג וְיַחְזֹר וְיִתֵּן עַל הַבְּהוֹנוֹת אַחַר מַתַּן שֶׁבַע שֶׁנֶּאֱמַר שֶׁתִּהְיֶה כָּל תּוֹרָתוֹ עַל הַסֵּדֶר: נָתַן שֶׁבַע הַמַּתָּנוֹת שֶׁלֹּא לִשְׁמָן לֹא הֻרְצָה וְהֻכְשַׁר הַמְצֹרָע: מְצֹרָע שֶׁנִּתְנַגֵּעַ אַחַר שֶׁהֵבִיא אֲשָׁמוֹ צָרִיךְ קָרְבָּן אַחֵר לְצָרַעַת שְׁנִיָּה. וְכֵן אִם הֵבִיא אֲשָׁמוֹ וְנִתְנַגֵּעַ צָרִיךְ קָרְבָּן עַל כָּל אַחַת וְאַחַת. אֲבָל אִם נִתְנַגֵּעַ וְנִרְפָּא וְהֵבִיא צִפֳּרָיו וְחָזַר וְנִתְנַגֵּעַ וְנִתְרַפֵּא וְהֵבִיא צִפֳּרָיו קָרְבָּן אֶחָד לְכֻלָּן: מְצֹרָע שֶׁהֵבִיא קָרְבַּן עָנִי וְהֶעֱשִׁיר אוֹ עָשִׁיר וְהֶעֱנִי קֹדֶם שֶׁיַּקְרִיב קָרְבְּנוֹתָיו הַכּל הוֹלֵךְ אַחַר הָאָשָׁם. אִם הָיָה בִּשְׁעַת שְׁחִיטַת הָאָשָׁם עָשִׁיר יְשַׁלֵּם קָרְבַּן עָשִׁיר וְאִם הָיָה עָנִי יְשַׁלֵּם קָרְבַּן עָנִי: שְׁנֵי מְצֹרָעִים שֶׁנִּתְעָרְבוּ קָרְבְּנוֹתֵיהֶן וְנִזְרַק דַּם אַחַת מִשְּׁתֵי הַחַטָּאוֹת שֶׁלָּהֶן וְאַחַר כָּךְ מֵת אֶחָד מִן הַמְצֹרָעִים כֵּיצַד עוֹשֶׂה זֶה הַמְצֹרָע הַחַי. לְהָבִיא חַטַּאת בְּהֵמָה אֵינוֹ יָכוֹל שֶׁמָּא זוֹ שֶׁנִּזְרַק דָּמָהּ חַטָּאתוֹ הִיא וְאֵין מְבִיאִין חַטַּאת בְּהֵמָה עַל הַסָּפֵק. וּלְהָבִיא חַטַּאת הָעוֹף אֵינוֹ יָכוֹל שֶׁעָשִׁיר שֶׁהֵבִיא קָרְבַּן עָנִי לֹא יָצָא. וְכֵיצַד יַעֲשֶׂה. יִכְתֹּב כָּל נְכָסָיו לְאַחֵר וְנִמְצָא עָנִי וְיָבִיא חַטַּאת הָעוֹף מִסָּפֵק וְלֹא תֵּאָכֵל כְּמוֹ שֶׁבֵּאַרְנוּ וְנִמְצָא אוֹכֵל בְּקָדָשִׁים: עָשִׁיר שֶׁאָמַר קָרְבָּנוֹ שֶׁל
