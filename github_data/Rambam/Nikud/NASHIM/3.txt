הלכות אישות - פרק שלישי

כֵּיצַד ] הָאִשָּׁה מִתְקַדֶּשֶׁת. אִם בְּכֶסֶף הוּא מְקַדֵּשׁ אֵין פָּחוֹת מִפְּרוּטָה ] כֶּסֶף אוֹ שְׁוֵה פְּרוּטָה. אוֹמֵר לָהּ הֲרֵי אַתְּ מְקֻדֶּשֶׁת לִי. אוֹ הֲרֵי אַתְּ מְאֹרֶסֶת לִי. אוֹ הֲרֵי אַתְּ לִי לְאִשָּׁה בָּזֶה. וְנוֹתֵן לָהּ בִּפְנֵי עֵדִים. וְהָאִישׁ הוּא שֶׁאוֹמֵר דְּבָרִים שֶׁמַּשְׁמָעָן שֶׁקּוֹנֶה אוֹתָהּ לוֹ לְאִשָּׁה וְהוּא שֶׁיִּתֵּן לָהּ הַכֶּסֶף:

נָתְנָה הִיא וְאָמְרָה לוֹ הֲרֵי אֲנִי מְקֻדֶּשֶׁת לְךָ. הֲרֵינִי מְאֹרֶסֶת לְךָ. הֲרֵינִי לְךָ לְאִנְתּוּ אוֹ בְּכָל לְשׁוֹן הַקְנָאָה אֵינָהּ מְקֻדֶּשֶׁת. וְכֵן אִם נָתְנָה הִיא לוֹ וְאָמַר ] הוּא אֵינָהּ מְקֻדֶּשֶׁת. וְאִם נָתַן o הוּא וְאָמְרָה הִיא הֲרֵי זוֹ מְקֻדֶּשֶׁת מִסָּפֵק:

וְאִם קִדֵּשׁ בִּשְׁטָר ] כּוֹתֵב עַל הַנְּיָר אוֹ עַל הַחֶרֶס אוֹ עַל הֶעָלֶה אוֹ עַל כָּל דָּבָר שֶׁיִּרְצֶה. הֲרֵי אַתְּ מְקֻדֶּשֶׁת לִי אוֹ הֲרֵי אַתְּ מְאֹרֶסֶת לִי וְכָל כַּיּוֹצֵא בִּדְבָרִים אֵלּוּ וְנוֹתְנוֹ לָהּ בִּפְנֵי עֵדִים:

וְצָרִיךְ שֶׁיִּכְתֹּב אוֹתוֹ לְשֵׁם הָאִשָּׁה הַמִּתְקַדֶּשֶׁת כְּגֵט. וְאֵינוֹ כּוֹתְבוֹ אֶלָּא מִדַּעְתָּהּ. כְּתָבוֹ שֶׁלֹּא לִשְׁמָהּ אוֹ לִשְׁמָהּ שֶׁלֹּא מִדַּעְתָּהּ אַף עַל פִּי שֶׁנְּתָנוֹ לָהּ לְדַעְתָּהּ בִּפְנֵי עֵדִים אֵינָהּ מְקֻדֶּשֶׁת

] וְאִם קִדֵּשׁ בְּבִיאָה אוֹמֵר לָהּ הֲרֵי אַתְּ מְקֻדֶּשֶׁת לִי אוֹ הֲרֵי אַתְּ מְאֹרֶסֶת לִי אוֹ הֲרֵי אַתְּ לִי לְאִשָּׁה בִּבְעִילָה זוֹ וְכָל כַּיּוֹצֵא בָּזֶה. וּמִתְיַחֵד עִמָּהּ בִּפְנֵי שְׁנֵי עֵדִים וּבוֹעֲלָהּ. וְהַמְקַדֵּשׁ בְּבִיאָה (] מִסְּתָמָא) דַּעְתּוֹ עַל גְּמַר בִּיאָה. וּכְשֶׁיִּגְמֹר בִּיאָתוֹ תִּהְיֶה מְקֻדֶּשֶׁת. (] וּבֵין שֶׁבָּא עָלֶיהָ כְּדַרְכָּהּ וּבֵין שֶׁבָּא עָלֶיהָ שֶׁלֹּא כְּדַרְכָּהּ הֲרֵי זוֹ מְקֻדֶּשֶׁת):

] הַדְּבָרִים שֶׁיֹּאמַר הָאִישׁ כְּשֶׁיְּקַדֵּשׁ צָרִיךְ שֶׁיִּהְיֶה מַשְׁמָעָם שֶׁהוּא קוֹנֶה הָאִשָּׁה וְלֹא שֶׁיְּהֵא מַשְׁמָע שֶׁהִקְנָה עַצְמוֹ לָהּ. כֵּיצַד. הֲרֵי שֶׁאָמַר לָהּ אוֹ שֶׁכָּתַב בַּשְּׁטָר שֶׁנְּתָנוֹ לָהּ. הֲרֵינִי בַּעְלֵךְ. הֲרֵינִי אֲרוּסֵךְ. הֲרֵינִי אִישֵׁךְ וְכָל כַּיּוֹצֵא בָּזֶה. אֵין כָּאן קִדּוּשִׁין כְּלָל. אָמַר לָהּ אוֹ כָּתַב לָהּ. הֲרֵי אַתְּ אִשְׁתִּי. הֲרֵי אַתְּ אֲרוּסָתִי. אוֹ הֲרֵי אַתְּ קְנוּיָה לִי. הֲרֵי אַתְּ שֶׁלִּי. הֲרֵי אַתְּ לְקוּחָתִי. הֲרֵי אֶת חֲרוּפָתִי. הֲרֵי אַתְּ בִּרְשׁוּתִי. הֲרֵי אַתְּ זְקוּקָה לִי וְכָל כַּיּוֹצֵא בָּהֶן. הֲרֵי זוֹ מְקֻדֶּשֶׁת:

אָמַר לָהּ אוֹ כָּתַב לָהּ הֲרֵי אַתְּ מְיֻחֶדֶת לִי. הֲרֵי אַתְּ מְיֹעֶדֶת לִי. הֲרֵי אַתְּ עֶזְרָתִי. הֲרֵי אַתְּ נֶגְדָּתִי. הֲרֵי אַתְּ צַלְעָתִי. הֲרֵי אַתְּ סְגוּרָתִי. הֲרֵי אַתְּ תַּחְתַּי. הֲרֵי אַתְּ ] עֲצוּרָתִי. הֲרֵי אַתְּ תְּפוּסָתִי. הֲרֵי זוֹ מְקֻדֶּשֶׁת בְּסָפֵק. וְהוּא שֶׁיִּהְיֶה מְדַבֵּר עִמָּהּ תְּחִלָּה עַל עִסְקֵי קִדּוּשִׁין. אֲבָל אִם אֵינוֹ מְדַבֵּר עִמָּהּ תְּחִלָּה עַל עִסִקֵי קִדּוּשִׁין אֵין חוֹשִׁשִׁין לִמִלּוֹת אֵלּוּ:

וְיֵשׁ לָאִישׁ לְקַדֵּשׁ הָאִשָּׁה בְּכָל לָשׁוֹן שֶׁהִיא מַכֶּרֶת בּוֹ. וְיִהְיֶה מַשְׁמַע הַדְּבָרִים בְּאוֹתָהּ הַלָּשׁוֹן שֶׁקְּנָאָהּ כְּמוֹ שֶׁבֵּאַרְנוּ. ] הָיָה מְדַבֵּר עִם הָאִשָּׁה עַל עִסְקֵי הַקִּדּוּשִׁין וְרָצְתָה וְעָמַד וְקִדֵּשׁ וְלֹא פֵּרֵשׁ וְלֹא אָמַר לָהּ כְּלוּם אֶלָּא נָתַן בְּיָדָהּ אוֹ בָּעַל הוֹאִיל וְהֵן עֲסוּקִין בָּעִנְיָן דַּיּוֹ וְאֵינוֹ צָרִיךְ לְפָרֵשׁ. וְכֵן עֵדֵי הַקִּדּוּשִׁין וְהַגֵּרוּשִׁין אֵינוֹ צָרִיךְ לוֹמַר לָהֵם אַתֵּם עֵדַי. אֵלָּא כֵּיוָן שֵׁגֵּרֵשׁ אוֹ קִדֵּשׁ בִּפִנֵיהֵם הֲרֵי זוֹ מְקֻדֶּשֶׁת אוֹ מִגֹרֵשֵׁת:

הָאוֹמֵר ] לְאִשָּׁה הִתְקַדְּשִׁי לְחֶצְיִי הֲרֵי זוֹ מְקֻדֶּשֶׁת. הָא לְמַה זֶּה דּוֹמֶה לְאוֹמֵר לָהּ תְּהִי אִשְׁתִּי אַתְּ וְאַחֶרֶת שֶׁנִּמְצָא שֶׁאֵין לָהּ אֶלָּא חֲצִי אִישׁ. אֲבָל אִם אָמַר חֶצְיֵךְ מְקֻדֶּשֶׁת לִי אֵינָהּ מְקֻדֶּשֶׁת שֶׁאֵין אִשָּׁה אַחַת רְאוּיָה לִשְׁנַיִם. וְכֵן אִם אוֹמֵר הֲרֵי אַתְּ מְקֻדֶּשֶׁת לִי וְלָזֶה אֵינָהּ מְקֻדֶּשֶׁת:

אָמַר לָהּ הֲרֵי חֶצְיֵךְ מְקֻדֶּשֶׁת לִי בִּפְרוּטָה וְחֶצְיֵךְ בִּפְרוּטָה. אוֹ שֶׁאָמַר לָהּ חֶצְיֵךְ מְקֻדֶּשֶׁת לִי בַּחֲצִי פְּרוּטָה וְחֶצְיֵךְ הָאַחֶרֶת מְקֻדֶּשֶׁת בַּחֲצִי פְּרוּטָה ] הֲרֵי זוֹ מְקֻדֶּשֶׁת. * אָמַר לָהּ חֶצְיֵךְ מְקֻדֶּשֶׁת לִי בִּפְרוּטָה הַיּוֹם וְחֶצְיֵךְ בִּפְרוּטָה לְמָחָר. שְׁתֵּי חֲצָיַיִךְ בִּפְרוּטָה. שְׁתֵּי בְּנוֹתֶיךָ לִשְׁנֵי בָּנַי בִּפְרוּטָה. בִּתְּךָ מְקֻדֶּשֶׁת לִי וּפָרָתְךָ מְכוּרָה לִי בִּפְרוּטָה. אוֹ בִּתְּךָ וְקַרְקָעָתְךָ לִי בִּפְרוּטָה. בְּכָל אֵלּוּ מְקֻדֶּשֶׁת בְּסָפֵק:

] הָאָב מְקַדֵּשׁ אֶת בִּתּוֹ שֶׁלֹּא לְדַעְתָּהּ כָּל זְמַן שֶׁהִיא קְטַנָּה. וְכֵן כְּשֶׁהִיא נַעֲרָה רְשׁוּתָהּ בְּיָדוֹ שֶׁנֶּאֱמַר . וְקִדּוּשֶׁיהָ לְאָבִיהָ. וְכֵן הוּא זַכַּאי בִּמְצִיאָתָהּ וּבְמַעֲשֵׂה יָדֶיהָ וּבִכְתֻבָּתָהּ אִם נִתְגָּרְשָׁה אוֹ נִתְאַלְמְנָה מִן הָאֵרוּסִין הוּא זַכַּאי בַּכּל עַד שֶׁתִּבָּגֵר. לְפִיכָךְ מְקַבֵּל הָאָב קִדּוּשֵׁי בִּתּוֹ מִיּוֹם שֶׁתִּוָּלֵד עַד שֶׁתִּבָּגֵר. וַאֲפִלּוּ הָיְתָה חֵרֶשֶׁת אוֹ שׁוֹטָה וְקִדְּשָׁהּ הָאָב הֲרֵי הִיא אֵשֶׁת אִישׁ גְּמוּרָה. וְאִם הָיְתָה בַּת שָׁלֹשׁ שָׁנִים וְיוֹם אֶחָד מִתְקַדֶּשֶׁת בְּבִיאָה מִדַּעַת אָבִיהָ. פָּחוֹת מִכֵּן אִם קִדְּשָׁהּ אָבִיהָ בְּבִיאָה אֵינָהּ מְקֻדֶּשֶׁת:

בָּגְרָה הַבַּת אֵין לְאָבִיהָ בָּהּ רְשׁוּת וַהֲרֵי הִיא כִּשְׁאָר כָּל הַנָּשִׁים שֶׁאֵינָם מִתְקַדְּשׁוֹת אֶלָּא לְדַעְתָּן. וְכֵן אִם הִשִּׂיאָהּ אָבִיהָ וְנִתְאַלְמְנָה אוֹ נִתְגָּרְשָׁה בְּחַיֵּי אָבִיהָ הֲרֵי הִיא בִּרְשׁוּת עַצְמָהּ וְאַף עַל פִּי שֶׁעֲדַיִן הִיא קְטַנָּה. כֵּיוָן שֶׁנִּשֵּׂאת אֵין לְאָבִיהָ בָּהּ רְשׁוּת לְעוֹלָם:

נִתְקַדְּשָׁה קֹדֶם שֶׁתִּבְגֹּר שֶׁלֹּא לְדַעַת אָבִיהָ אֵינָהּ מְקֻדֶּשֶׁת o וַאֲפִלּוּ אִם נִתְרַצָּה הָאָב אַחַר שֶׁנִּתְקַדְּשָׁה. וַאֲפִלּוּ אִם ] נִתְאַלְמְנָה אוֹ נִתְגָּרְשָׁה מִן אוֹתָן הַקִּדּוּשִׁין אֵינָהּ אֲסוּרָה לְכֹהֵן. וּבֵין הִיא וּבֵין אָבִיהָ יְכוֹלִין לְעַכֵּב. בֵּין אִם נִתְקַדְּשָׁה בְּפָנָיו בֵּין שֶׁנִּתְקַדְּשָׁה שֶׁלֹּא בְּפָנָיו אֵינָהּ מְקֻדֶּשֶׁת:

] הָיְתָה הַבַּת סָפֵק בּוֹגֶרֶת. בֵּין שֶׁקִּדְּשָׁהּ אָבִיהָ שֶׁלֹּא לְדַעְתָּהּ בֵּין שֶׁקִּדְּשָׁה הִיא עַצְמָהּ שֶׁלֹּא לְדַעַת אָבִיהָ. הֲרֵי זוֹ מְקֻדֶּשֶׁת בְּסָפֵק לְפִיכָךְ צְרִיכָה גֵּט מִסָּפֵק. יֵשׁ ] לָאִישׁ לַעֲשׂוֹת שָׁלִיחַ לְקַדֵּשׁ לוֹ אִשָּׁה בֵּין אִשָּׁה פְּלוֹנִית בֵּין אִשָּׁה מִשְּׁאָר הַנָּשִׁים. וְכֵן הָאִשָּׁה הַגְּדוֹלָה עוֹשָׂה שָׁלִיחַ לְקַבֵּל קִדּוּשֶׁיהָ בֵּין מֵאִישׁ פְּלוֹנִי בֵּין מֵאִישׁ מִשְּׁאָר אֲנָשִׁים. וְכֵן ] הָאָב עוֹשֶׂה שָׁלִיחַ לְקַבֵּל קִדּוּשֵׁי בִּתּוֹ כְּשֶׁהִיא בִּרְשׁוּתוֹ. וְאוֹמֵר הָאָב לְבִתּוֹ הַקְּטַנָּה צְאִי וְקַבְּלִי קִדּוּשַׁיִךְ:
