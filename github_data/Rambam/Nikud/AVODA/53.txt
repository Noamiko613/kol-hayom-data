אֵין ] הַמַּעֲלֶה בַּחוּץ חַיָּב עַד שֶׁיַּעֲלֶה לְרֹאשׁ הַמִּזְבֵּחַ שֶׁיַּעֲשֶׂה בַּחוּץ. אֲבָל אִם הֶעֱלָה עַל הַסֶּלַע אוֹ עַל הָאֶבֶן פָּטוּר. שֶׁאֵין קָרוּי קָרְבָּן אֶלָּא עַל הַמִּזְבֵּחַ וְאַף עַל פִּי שֶׁהוּא בַּחוּץ שֶׁנֶּאֱמַר . וְאֵינוֹ חַיָּב עַד שֶׁיַּעֲלֶה לַשֵּׁם שֶׁנֶּאֱמַר עַד שֶׁיִּתְכַּוֵּן לַשֵּׁם:

אֵינוֹ חַיָּב אֶלָּא עַל הַעֲלָאַת דָּבָר הָרָאוּי לָאִשִּׁים וְלַמִּזְבֵּחַ כְּגוֹן הָעוֹלָה שֶׁנֶּאֱמַר מָה עוֹלָה שֶׁרְאוּיָה לָאִשִּׁים אַף כָּל שֶׁרָאוּי לָאִשִּׁים הוּא שֶׁחַיָּבִין עַל הַעֲלָאָתוֹ בַּחוּץ:

מִכָּאן אָמְרוּ הַזּוֹרֵק אֶת הַדָּם אוֹ הַמַּקְטִיר אֵיבְרֵי עוֹלָה אוֹ אֵימוּרִין אוֹ קֹמֶץ אוֹ לְבוֹנָה אוֹ קְטֹרֶת אוֹ מִנְחַת כֹּהֲנִים אוֹ מִנְחַת נְסָכִים אוֹ הַמְנַסֵּךְ שְׁלֹשָׁה לוֹגִין יַיִן אוֹ מַיִם בַּחוּץ חַיָּב שֶׁנֶּאֱמַר כָּל הַמִּתְקַבֵּל בִּפְנִים חַיָּבִין עָלָיו בַּחוּץ:

אֲבָל הַזּוֹרֵק שְׁיָרֵי הַדָּם בַּחוּץ אֲפִלּוּ שְׁיָרֵי דָּמִים הַפְּנִימִים פָּטוּר. שֶׁזְּרִיקַת שְׁיָרֵי הַדָּם שְׁיָרֵי מִצְוָה הֵם וְאֵינָן מְעַכְּבִין. וְכֵן הַמְנַסֵּךְ יַיִן אוֹ מַיִם פָּחוֹת מִשְּׁלֹשָׁה לוֹגִין בַּחוּץ פָּטוּר בֵּין בֶּחָג בֵּין בִּשְׁאָר יְמוֹת הַשָּׁנָה. ] הוֹאִיל וְחָסֵר הַשִּׁעוּר הֲרֵי אֵינָן רְאוּיִין לְהִתְקַבֵּל בִּפְנִים. וְכֵן הַמַּעֲלֶה מִבְּשַׂר חַטָּאת מִבְּשַׂר אָשָׁם מִבְּשַׂר שְׁלָמִים בֵּין שֶׁל יָחִיד בֵּין שֶׁל צִבּוּר מִשְּׁיָרֵי מְנָחוֹת מִשְּׁתֵי הַלֶּחֶם מִלֶּחֶם הַפָּנִים בַּחוּץ פָּטוּר. שֶׁכָּל אֵלּוּ רְאוּיִין לַאֲכִילָה לֹא לָאִשִּׁים:

הַמַּעֲלֶה אֶת הַבְּהֵמָה כֻּלָּהּ בַּחוּץ חַיָּב מִפְּנֵי הָאֵימוּרִין. וְאַף עַל פִּי שֶׁלֹּא הִפְרִישָׁן אֵין בְּשַׂר הַזֶּבַח חוֹצֵץ. וּכְאִלּוּ הִקְטִיר הָאֵימוּרִים בִּפְנֵי עַצְמָן. אֲבָל הַמַּעֲלֶה מִנְחָה שֶׁלֹּא נִקְמְצָה פָּטוּר. שֶׁאֵין הַקֹּמֶץ בָּרוּר וּמֻבְדָּל. קְמָצָהּ וְחָזַר קֻמְצָהּ לְתוֹכָהּ וְהִקְרִיב כֻּלָּהּ בַּחוּץ חַיָּב:

הַיּוֹצֵק וְהַבּוֹלֵל וְהַפּוֹתֵת וְהַמּוֹלֵחַ וְהַמֵּנִיף וְהַמַּגִּישׁ וְהַמְסַדֵּר אֶת הַשֻּׁלְחָן וְהַמֵּטִיב אֶת הַנֵּרוֹת וְהַקּוֹמֵץ וְהַמְקַבֵּל דָּמִים בַּחוּץ פָּטוּר. לְפִי שֶׁכָּל אֶחָד מֵאֵלּוּ אֵינוֹ גְּמַר עֲבוֹדָה וְנֶאֱמַר . מַה הַעֲלָאָה שֶׁהִיא גְּמַר עֲבוֹדָה אַף כָּל שֶׁהוּא גְּמַר עֲבוֹדָה חַיָּבִין עָלָיו:

# הלכות תמידין ומוספין

יֵשׁ בִּכְלָלָן תְשַׁע עֶשְׂרֵה מִצְוֹת. שְׁמוֹנֶה עֶשְׂרֵה מִצְוֹת עֲשֵׂה. וְאַחַת מִצְוַת לֹא תַעֲשֶׂה. וְזֶה הוּא פְּרָטָן:

לְהַקְרִיב שְׁנֵי כְּבָשִׂים בְּכָל יוֹם עוֹלוֹת. לְהַדְלִיק אֵשׁ עַל הַמִּזְבֵּחַ בְּכָל יוֹם. שֶׁלֹּא לְכַבּוֹתָהּ. לְהָרִים אֶת הַדֶּשֶׁן בְּכָל יוֹם. לְהַקְטִיר קְטֹרֶת בְּכָל יוֹם. לְהַדְלִיק נֵרוֹת בְּכָל יוֹם. שֶׁיַּקְרִיב כֹּהֵן גָּדוֹל מִנְחָה בְּכָל יוֹם וְהִיא הַנִּקְרֵאת חֲבִיתִין. לְהוֹסִיף שְׁנֵי כְּבָשִׂים עוֹלוֹת בְּשַׁבָּת. לַעֲשׂוֹת לֶחֶם הַפָּנִים. מוּסַף רָאשֵׁי חֳדָשִׁים. מוּסַף הַפֶּסַח. לְהַקְרִיב עֹמֶר הַתְּנוּפָה. מוּסַף עֲצֶרֶת. לְהָבִיא שְׁתֵּי הַלֶּחֶם עִם הַקָּרְבָּנוֹת הַבָּאוֹת בִּגְלַל הַלֶּחֶם בְּיוֹם עֲצֶרֶת. מוּסַף רֹאשׁ הַשָּׁנָה. מוּסַף יוֹם צוֹם. מוּסַף הֶחָג. מוּסַף שְׁמִינִי עֲצֶרֶת. לִסְפֹּר כָּל אִישׁ וְאִישׁ שִׁבְעָה שָׁבוּעוֹת מִיּוֹם הַקְרָבַת הָעֹמֶר:

וּבֵאוּר מִצְוֹת אֵלוּ בִּפְרָקִים אֵלוּ:
