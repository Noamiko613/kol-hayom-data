הלכות פסולי המוקדשין - פרק אחד עשר

כָּל הַמְּנָחוֹת שֶׁקְּמָצָן אֶחָד מִן הַפְּסוּלִין לַעֲבוֹדָה הֲרֵי הֵם פְּסוּלוֹת. וְכֵן אִם לִקֵּט הַלְּבוֹנָה פָּסַל אַף עַל פִּי שֶׁלֹּא קָמַץ:

קָמַץ הַכָּשֵׁר וְנָתַן לַפָּסוּל. קָמַץ בִּימִינוֹ וְנָתַן לִשְׂמֹאלוֹ וְאַחַר כָּךְ נְתָנוֹ לִכְלִי. קָמַץ מִכְּלִי קֹדֶשׁ וְנָתַן לִכְלִי חֹל. פָּסַל:

קָמַץ וְעָלָה בְּיָדוֹ צְרוֹר אוֹ גַּרְגִּיר מֶלַח אוֹ קֹרֶט לְבוֹנָה פָּסוּל:

קָמַץ עַד שֶׁהוּא בַּחוּץ וְנִכְנַס לְפָנִים. יַחְזֹר וְיִקְמֹץ בִּפְנִים וְכָשֵׁר:

נִתְפַּזֵּר הַקֹּמֶץ עַל גַּבֵּי הָרִצְפָּה. יַחְזֹר וְיַאַסְפֶנּוּ:

הָיְתָה הַמִּנְחָה שֶׁלֹּא בִּכְלִי שָׁרֵת. אוֹ שֶׁהָיְתָה קֻמְצָהּ שֶׁלֹּא בִּכְלִי שָׁרֵת. אוֹ שֶׁהֶעֱלָהוּ לַמִּזְבֵּחַ שֶׁלֹּא בִּכְלִי שָׁרֵת. אוֹ שֶׁבְּלָלָהּ בְּשַׁמְנָהּ חוּץ לָעֲזָרָה. פְּסוּלָה עַד שֶׁתִּהְיֶה בְּלִילָתָהּ בִּפְנִים:

כָּל הַמְּנָחוֹת שֶׁיָּצַק עֲלֵיהֶן הַשֶּׁמֶן פָּסוּל לַעֲבוֹדָה כְּגוֹן הַזָּר וְכַיּוֹצֵא בּוֹ אוֹ שֶׁבְּלָלָן אוֹ פְּתָתָם אוֹ מְלָחָן כְּשֵׁרוֹת. הִגִּישָׁן אוֹ הֱנִיפָן חוֹזֵר הַכֹּהֵן וּמַגִּישׁ אוֹ מֵנִיף. וְאִם לֹא הִגִּישׁ וְלֹא הֵנִיף הַכֹּהֵן כְּשֵׁרוֹת. שֶׁנֶּאֱמַר . מִקְּמִיצָה וְאֵילָךְ מִצְוַת כְּהֻנָּה לִמֵּד עַל יְצִיקָה וּבְלִילָה שֶׁכְּשֵׁרָה בְּזָר:

מִנְחָה שֶׁנָּפַל לְתוֹכָהּ שֶׁמֶן מִנְחָה אַחֶרֶת אוֹ שֶׁמֶן חֻלִּין כָּל שֶׁהוּא נִפְסְלָה. חִסֵּר שַׁמְנָהּ פְּסוּלָה. חִסֵּר לְבוֹנָתָהּ כְּשֵׁרָה. וְהוּא שֶׁיִּהְיוּ עָלֶיהָ שֵׁנִי קוֹרְטֵי לְבוֹנָה אֲבָל קֹרֶט אֶחָד פְּסוּלָה שֶׁנֶּאֱמַר אֶת כָּל לְבוֹנָתָהּ:

רִבָּה שַׁמְנָהּ וּלְבוֹנָתָהּ עַד שְׁנֵי לוֹגִין לְכָל עִשָּׂרוֹן וּשְׁנֵי קֻמְצֵי לְבוֹנָה לְכָל מִנְחָה כְּשֵׁרָה. שְׁנֵי לוֹגִין אוֹ שְׁנֵי קְמָצִין אוֹ יֶתֶר עַל זֶה פְּסוּלָה:

מִנְחַת חוֹטֵא שֶׁנָּתַן עָלֶיהָ שֶׁמֶן אוֹ עַל הַקֹּמֶץ שֶׁלָּהּ נִפְסְלָה. נָתַן עָלֶיהָ לְבוֹנָה יִלְקְטֶנָּה. הָיְתָה שְׁחוּקָה הֲרֵי זוֹ פְּסוּלָה מִסָּפֵק. שֶׁהֲרֵי אִי אֶפְשָׁר לְלַקֵּט:

נָתַן שֶׁמֶן עַל שְׁיָרֶיהָ אַחַר שֶׁקָּמַץ. אֵינוֹ לוֹקֶה וְלֹא פְּסָלָהּ שֶׁהֲרֵי הַקֹּמֶץ כָּשֵׁר:

נָתַן מַשֶּׁהוּ שֶׁמֶן עַל גַּבֵּי כְּזַיִת מִן הַמִּנְחָה פְּסָלָהּ מִסָּפֵק. אֲבָל אִם נָתַן הַשֶּׁמֶן עַל פָּחוֹת מִכְּזַיִת לֹא פְּסָלָהּ. וְאֵינוֹ פּוֹסֵל הַלְּבוֹנָה עַד שֶׁיִּתֵּן כְּזַיִת לְבוֹנָה:

אֲפִלּוּ נָתַן הַלְּבוֹנָה עַל כָּל שֶׁהוּא מִן הַמִּנְחָה פָּסַל עַד שֶׁיְּלַקֵּט:

גִּבְּלָהּ בְּמַיִם וְקָמַץ כְּשֵׁרָה. לֹא נֶאֱמַר אֶלָּא מִשֶּׁמֶן:

מִנְחָה שֶׁקְּמָצָהּ פַּעֲמַיִם כְּשֵׁרָה, אֲפִלּוּ פְּעָמִים רַבּוֹת. וְהוּא שֶׁיַּקְטִיר כְּזַיִת בְּבַת אַחַת. שֶׁאֵין הַקְטָרָה פְּחוּתָה מִכְּזַיִת:

הִקְרִיב הַקֹּמֶץ בְּלֹא מֶלַח פְּסוּלָה. שֶׁהַמֶּלַח מְעַכֵּב בַּמִּנְחָה כְּמוֹ שֶׁבֵּאַרְנוּ. מִנְחָה שֶׁחָסְרָה קֹדֶם קְמִיצָה. יָבִיא מִתּוֹךְ בֵּיתוֹ וִימַלְּאֶנָּה. שֶׁהַקְּמִיצָה הִיא הַקּוֹבַעַת לֹא נְתִינָתָהּ בִּכְלִי שָׁרֵת:

הִתְנַדֵּב קֹמֶץ לְבוֹנָה בִּפְנֵי עַצְמוֹ. אִם חָסַר כָּל שֶׁהוּא פָּסוּל. וְכֵן שְׁנֵי בְּזִיכֵי לְבוֹנָה שֶׁעִם הַלֶּחֶם אִם חָסַר אֶחָד מֵהֶן כָּל שֶׁהוּא פְּסוּלִין. עַד שֶׁיִּהְיוּ שְׁנֵי קְמָצִין מִתְּחִלָּה וְעַד סוֹף:

הִפְרִישׁ שְׁנֵי קְמָצִין לְמִנְחָה אַחַת וְאָבַד אֶחָד מֵהֶן קֹדֶם קְמִיצָה לֹא הֻקְבַּע. לְאַחַר קְמִיצָה הֻקְבַּע וּפְסוּלָה מִפְּנֵי שֶׁרִבָּה לְבוֹנָתָהּ. וְכֵן אִם הִפְרִישׁ אַרְבָּעָה קְמָצִים לִשְׁנֵי בְּזִיכֵי לֶחֶם וְאָבְדוּ שְׁנַיִם מֵהֶם קֹדֶם סִלּוּק הַבְּזִיכִים לֹא הֻקְבְּעוּ וּכְשֵׁרִים. לְאַחַר סִלּוּק הַבְּזִיכִין הֻקְבְּעוּ וּפְסוּלִין מִפְּנֵי הָרִבּוּי:

קֹמֶץ מִנְחָה שֶׁנִּטְמָא וְהִקְטִירוֹ. הַצִּיץ מְרַצֶּה שֶׁנֶּאֱמַר . יָצָא הַקֹּמֶץ חוּץ לָעֲזָרָה וְהִכְנִיסוֹ וְהִקְטִירוֹ אֵין הַצִּיץ מְרַצֶּה. שֶׁהַצִּיץ מְרַצֶּה עַל הַטָּמֵא וְאֵינוֹ מְרַצֶּה עַל הַיּוֹצֵא:

קָמַץ אֶת הַמִּנְחָה וְאַחַר כָּךְ נִטְמְאוּ שְׁיָרֶיהָ כֻּלָּן אוֹ נִשְׂרְפוּ אוֹ יָצְאוּ חוּץ לָעֲזָרָה אוֹ אָבְדוּ. לֹא יַקְטִיר הַקֹּמֶץ וְאִם הִקְטִיר הֻרְצָה. נִשְׁאַר מְעַט מִן הַשְּׁיָרִים בְּכַשְׁרוּתָן. יַקְטִיר הַקֹּמֶץ וְאוֹתָן הַשְּׁיָרִים שֶׁנִּשְׁאֲרוּ אֲסוּרִין בַּאֲכִילָה:

הָיְתָה מְחִצָּה מִלְּמַטָּה בִּכְלִי שֶׁיֵּשׁ בּוֹ עִשָּׂרוֹן שֶׁל מִנְחָה. אַף עַל פִּי שֶׁהוּא מְעֹרָב מִלְמַעְלָה לֹא יִקְמֹץ. וְאִם קָמַץ פְּסוּלָה:

הָיָה הַכְּלִי חָלוּק בִּמְחִצָּה מִלְמַעְלָה וּמְעֹרָב מִלְּמַטָּה. קוֹמֵץ מִמֶּנּוּ:

חָלַק הָעִשָּׂרוֹן בִּכְלִי אֶחָד וְאֵין חֲלָקָיו נוֹגְעִין זֶה בָּזֶה וְאֵין בֵּינֵיהֶן מְחִצָּה. הֲרֵי זֶה סָפֵק אִם מְצָרֵף הַכְּלִי לַקְּמִיצָה אוֹ אֵינוֹ מְצָרֵף. לְפִיכָךְ לֹא יִקְמֹץ. וְאִם קָמַץ לֹא יַקְטִיר. וְאִם הִקְטִיר הֻרְצָה. וְלֹא יֵאָכְלוּ הַשְּׁיָרִים:

קָמַץ וְנָתַן הַקֹּמֶץ לְמַעְלָה עַל הַשֻּׁלְחָן כְּנֶגֶד גֹּבַהּ מַעֲרֶכֶת לֶחֶם הַפָּנִים קִדְּשׁוֹ הַשֻּׁלְחָן לְהִפָּסֵל. אֲבָל אֵינוֹ מְקַדְּשׁוֹ לִקָּרֵב. וְאֵינוֹ קָרֵב עַד שֶׁיִּתְקַדֵּשׁ בִּכְלִי שָׁרֵת הָרָאוּי לְקֹמֶץ:

הִדְבִּיק הַקֹּמֶץ לְדֹפֶן הַכְּלִי וְקָמַץ. אוֹ שֶׁהָפַךְ הַכְּלִי עַל יָדוֹ וְקָמַץ מִתּוֹכוֹ וּפִיו לְמַטָּה. לֹא יַקְטִיר וְאִם הִקְטִיר הֻרְצָה:

עִשָּׂרוֹן שֶׁחִלְּקוֹ וְאָבַד אֶחָד מֵחֲלָקָיו וְהִפְרִישׁ חֵלֶק אַחֵר תַּחְתָּיו וְנִמְצָא הָאָבוּד וַהֲרֵי שְׁלָשְׁתָּן מֻנָּחִין בִּכְלִי אֶחָד וְאֵין נוֹגְעִין זֶה בָּזֶה. נִטְמָא זֶה שֶׁאָבַד הֲרֵי הוּא מִצְטָרֵף עִם חֵלֶק רִאשׁוֹן וְנִפְסְלוּ. וְזֶה שֶׁהִפְרִישׁ אֵינוֹ מִצְטָרֵף אֶלָּא יַשְׁלִים עָלָיו. נִטְמָא הַמֻּפְרָשׁ. מֻפְרָשׁ וְרִאשׁוֹן מִצְטָרְפִין וְנִפְסְלוּ. וְזֶה שֶׁנִּמְצָא אֵינוֹ מִצְטָרֵף עִמָּהֶן. נִטְמָא הַחֵלֶק הָרִאשׁוֹן הֲרֵי הָאָבוּד וְהַמֻּפְרָשׁ תַּחְתָּיו מִצְטָרְפִין:

וְכֵן לְעִנְיַן קְמִיצָה. קָמַץ מִן הַנִּמְצָא. שְׁיָרָיו עִם הַחֵלֶק הָרִאשׁוֹן נֶאֱכָלִין. וְהַמֻּפְרָשׁ אֵינוֹ נֶאֱכָל. קָמַץ מִן הַמֻּפְרָשׁ. שְׁיָרָיו נֶאֱכָלִין וְהַנִּמְצָא אֵינוֹ נֶאֱכָל. קָמַץ מִן הָרִאשׁוֹן. שְׁנֵיהֶן אֵין נֶאֱכָלִין. לְפִי שֶׁשְּׁנֵיהֶן שְׁיָרִים יְתֵרִים שֶׁהֲרֵי הֵן עִשָּׂרוֹן שָׁלֵם וְדוֹמִין לְמִנְחָה שֶׁלֹּא נִקְמְצָה שֶׁהִיא אֲסוּרָה. וְהֵיאַךְ קָרֵב הַקֹּמֶץ הַזֶּה וַהֲרֵי לְפָנָיו עִשָּׂרוֹן וּמֶחֱצָה. מִפְּנֵי שֶׁהַקְּמִיצָה תְּלוּיָה בְּדַעַת הַכֹּהֵן וּבְעֵת שֶׁקּוֹמֵץ אֵין דַּעְתּוֹ אֶלָּא עַל הָעִשָּׂרוֹן בִּלְבַד. וַהֲרֵי הַחֲלָקִים אֵינָן נוֹגְעִין זֶה בָּזֶה:

קֹמֶץ מִנְחָה שֶׁנִּתְעָרֵב בְּקֹמֶץ מִנְחָה אַחֶרֶת. מַקְטִיר שְׁנֵיהֶן כְּאַחַת וְהֵן כְּשֵׁרוֹת. וְכֵן אִם נִתְעָרֵב הַקֹּמֶץ בְּמִנְחַת כֹּהֲנִים אוֹ בְּמִנְחַת הַנְּסָכִים אוֹ בַּחֲבִיתֵי כֹּהֵן גָּדוֹל. הֲרֵי אֵלּוּ כְּשֵׁרוֹת וְיַקְטִיר הַכּל כְּאֶחָד שֶׁהַכּל לָאִשִּׁים:

שְׁתֵּי מְנָחוֹת שֶׁלֹּא נִקְמְצוּ שֶׁנִּתְעָרְבוּ זוֹ בָּזוֹ אִם יָכוֹל לִקְמֹץ מִזּוֹ בִּפְנֵי עַצְמוֹ וּמִזּוֹ בִּפְנֵי עַצְמוֹ כְּשֵׁרוֹת וְאִם לָאו פְּסוּלוֹת:

קֹמֶץ שֶׁנִּתְעָרֵב בְּמִנְחָה שֶׁלֹּא נִקְמְצָה לֹא יַקְטִיר. וְאִם הִקְטִיר הַכּל. זוֹ שֶׁנִּקְמְצָה עָלְתָה לַבְּעָלִים. וְזוֹ שֶׁלֹּא נִקְמְצָה לֹא עָלְתָה לַבְּעָלִים:

נִתְעָרֵב קֻמְצָהּ בִּשְׁיָרֶיהָ. אוֹ שֶׁנִּתְעָרְבוּ שְׁיָרֶיהָ בִּשְׁיָרֵי חֲבֶרְתָּהּ. לֹא יַקְטִיר. וְאִם הִקְטִיר עָלְתָה לַבְּעָלִים:
