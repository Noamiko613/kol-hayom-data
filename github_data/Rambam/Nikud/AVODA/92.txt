הַמַּקְדִּישׁ בַּיִת בָּנוּי הַדָּר בּוֹ לֹא מָעַל. אֲבָל הַמַּקְדִּישׁ עֵצִים וַאֲבָנִים וּבָנָה בָּהֶן בַּיִת הַדָּר שָׁם מָעַל כְּמוֹ שֶׁיִּתְבָּאֵר: גִּדּוּלֵי הֶקְדֵּשׁ מוֹעֲלִין בָּהֶן. כֵּיצַד. הִקְדִּישׁ שָׂדֶה וְהוֹצִיאָה עֲשָׂבִים אִילָן וְעָשָׂה פֵּרוֹת מוֹעֲלִין בָּהֶן. אֲבָל הַמַּקְדִּישׁ אֶת הַבּוֹר רֵיקָן וְאַחַר כָּךְ נִתְמַלֵּא מַיִם. הִקְדִּישׁ אַשְׁפָּה וְאַחַר כָּךְ נִתְמַלֵּאת
