הלכות מעשה הקרבנות - פרק עשירי

אֲכִילַת ] הַחַטָּאת וְהָאָשָׁם מִצְוַת עֲשֵׂה שֶׁנֶּאֱמַר . הַכֹּהֲנִים אוֹכְלִים וּבְעָלִים מִתְכַּפְּרִים. וְהוּא הַדִּין לִשְׁאָר הַקָּדָשִׁים שֶׁאוֹכְלִין אוֹתָן הַכֹּהֲנִים שֶׁאֲכִילָתָן מִצְוָה:

וְכֵן אֲכִילַת שְׁיָרֵי מְנָחוֹת מִצְוַת עֲשֵׂה שֶׁנֶּאֱמַר :

וְאֵין חַטָּאת וְאָשָׁם וּשְׁיָרֵי מְנָחוֹת נֶאֱכָלִין אֶלָּא לְזִכְרֵי כְּהֻנָּה בָּעֲזָרָה וְאִם נֶאֶכְלוּ בַּהֵיכָל נֶאֶכְלוּ שֶׁנֶּאֱמַר . וְכֵן שַׁלְמֵי צִבּוּר הֲרֵי הֵן כַּחַטָּאת וְכָאָשָׁם כְּמוֹ שֶׁבֵּאַרְנוּ:

חָזֶה וְשׁוֹק שֶׁל שְׁלָמִים נֶאֱכָלִין לְכֹהֲנִים זְכָרִים וּנְקֵבוֹת שֶׁנֶּאֱמַר בָּהֶם . וְכֵן הַמּוּרָם מִתּוֹדָה וְהַמּוּרָם מֵאֵיל נָזִיר שֶׁנֶּאֱמַר :

וְכֵן הַבְּכוֹר שֶׁהֲרֵי נֶאֱמַר בּוֹ . וְכָל אֵלּוּ הַנֶּאֱכָלִים לַכֹּהֲנוֹת נֶאֱכָלִים לְעַבְדֵי הַכֹּהֲנִים וְלִנְשֵׁיהֶם כִּתְרוּמָה. וְכֻלָּן נֶאֱכָלִים בְּכָל הָעִיר שֶׁנֶּאֱמַר לֹא נֶאֱמַר בָּהֶן בְּמָקוֹם קָדוֹשׁ שֶׁהִיא הָעֲזָרָה אֶלָּא טָהוֹר שֶׁהוּא כָּל מַחֲנֵה יִשְׂרָאֵל שֶׁכְּנֶגְדוֹ לְדוֹרוֹת יְרוּשָׁלַיִם. וְהוּא הַדִּין לְמַעֲשֵׂר וּפֶסַח שֶׁהֲרֵי הֵן קָדָשִׁים קַלִּים כִּשְׁלָמִים. וְהַחַלּוֹנוֹת וָעֳבִי הַחוֹמָה כִּלְפָנִים:

הַשְּׁלָמִים נֶאֱכָלִים בְּיוֹם הַזְּבִיחָה וְכָל הַלַּיְלָה וְכָל יוֹם הַמָּחֳרָת עַד שֶׁתִּשְׁקַע הַחַמָּה שֶׁנֶּאֱמַר וְגוֹ'. נִמְצֵאתָ לָמֵד שֶׁהֵן נֶאֱכָלִין לִשְׁנֵי יָמִים וְלַיְלָה אֶחָד בֵּין חֵלֶק הַכֹּהֲנִים בֵּין חֵלֶק הַבְּעָלִים וְהוּא הַדִּין לִבְכוֹר וּמַעֲשֵׂר שֶׁהֲרֵי הֵן קָדָשִׁים קַלִּים כִּשְׁלָמִים:

אֲבָל הַתּוֹדָה אַף עַל פִּי שֶׁהִיא קָדָשִׁים קַלִּים אֵינָהּ נֶאֱכֶלֶת אֶלָּא בְּיוֹם הַזְּבִיחָה עִם הַלַּיְלָה שֶׁנֶּאֱמַר בָּהּ . וְכֵן אֵיל נָזִיר וְהַלֶּחֶם הַבָּא עִמָּהֶן וְכַיּוֹצֵא בָּהֶן בֵּין חֵלֶק הַכֹּהֲנִים בֵּין חֵלֶק הַבְּעָלִים וְהוּא הַדִּין לְחַטָּאת וּלְאָשָׁם וּלְשַׁלְמֵי צִבּוּר וְלִשְׁיָרֵי מְנָחוֹת שֶׁהַכּל נֶאֱכָל לְיוֹם וְלַיְלָה שֶׁנֶּאֱמַר בְּיוֹם קָרְבָּנוֹ יֵאָכֵל כָּל הַקָּרְבָּנוֹת בְּמַשְׁמָע חוּץ מִן הַשְּׁלָמִים שֶׁפֵּרֵשׁ בָּהֶן הַכָּתוּב וּבְכוֹר וּמַעֲשֵׂר הַדּוֹמִים לָהֶם: (-)

כָּל אֵלּוּ הַנֶּאֱכָלִין לְיוֹם וְלַיְלָה דִּין תּוֹרָה שֶׁהֵן נֶאֱכָלִין עַד שֶׁיַּעֲלֶה עַמּוּד הַשַּׁחַר וּכְדֵי לְהַרְחִיק מִן הָעֲבֵרָה אָמְרוּ חֲכָמִים שֶׁאֵין נֶאֱכָלִין אֶלָּא עַד חֲצוֹת הַלַּיְלָה:

כָּל הַקָּרְבָּנוֹת בֵּין קָדְשֵׁי קָדָשִׁים בֵּין קָדָשִׁים קַלִּים אֵין אוֹכְלִין אוֹתָם אֶלָּא הַטְּהוֹרִים בִּלְבַד הַמּוּלִין. אֲפִלּוּ הֶעֱרִיב שִׁמְשׁוֹ וְלֹא הֵבִיא כַּפָּרָתוֹ אֵינוֹ אוֹכֵל בְּקָדָשִׁים. וְהַטֻּמְטוּם אָסוּר לֶאֱכל בְּקָדָשִׁים לְפִי שֶׁהוּא סְפֵק עָרֵל אֲבָל הָאַנְדְּרוֹגִינוּס יֵרָאֶה לִי שֶׁאוֹכֵל בְּקָדָשִׁים קַלִּים:

] וּמֻתָּר לֶאֱכל אֶת הַקָּדָשִׁים בְּכָל מַאֲכָל. אֲפִלּוּ הַכֹּהֲנִים מֻתָּרִין לֶאֱכל חֶלְקָם בֵּין מִקָּדָשִׁים קַלִּים בֵּין מִקָּדְשֵׁי קָדָשִׁים בְּכָל מַאֲכָל. וּלְשַׁנּוֹת בַּאֲכִילָתָן וּלְאָכְלָם צְלוּיִים שְׁלוּקִים וּמְבֻשָּׁלִים וְלָתֵת לְתוֹכָן תַּבְלִין שֶׁל חֻלִּין אֲבָל לֹא תַּבְלִין שֶׁל תְּרוּמָה שֶׁלֹּא יָבִיאוּ אֶת הַתְּרוּמָה לִידֵי פִּסּוּל. וְהָעֲצָמוֹת הַנִּשְׁאָרוֹת מֻתָּרוֹת וְעוֹשֶׂה אָדָם מֵהֶם כָּל כֵּלִים שֶׁיִּרְצֶה:

הָיְתָה לָהֶם אֲכִילָה מוּעֶטֶת אוֹכְלִין עִמָּהּ חֻלִּין וּתְרוּמוֹת כְּדֵי שֶׁתִּהְיֶה נֶאֱכֶלֶת עִם הַשֹּׂבַע. הָיְתָה לָהֶן אֲכִילָה מְרֻבָּה אֵין אוֹכְלִין עִמָּהּ חֻלִּין וּתְרוּמָה כְּדֵי שֶׁלֹּא תִּהְיֶה נֶאֱכֶלֶת אֲכִילָה גַּסָּה. וְכֵן בִּשְׁיָרֵי הַמְּנָחוֹת:

אֵין מְבַשְּׁלִין חַטָּאת אוֹ אָשָׁם עִם מוּרָם מִתּוֹדָה אוֹ מֵאֵיל נָזִיר מִפְּנֵי שֶׁמְּמַעֵט בְּאוֹכְלֵיהֶן וּבִמְקוֹם אֲכִילָתָן. וְלֹא מוּרָם מִתּוֹדָה וּמֵאֵיל נָזִיר עִם בְּכוֹר אוֹ עִם חָזֶה וְשׁוֹק שֶׁל שְׁלָמִים מִפְּנֵי שֶׁמְּמַעֵט זְמַן אֲכִילָתָן. וְלֹא מוּרָם מִשְּׁלָמִים שֶׁל אֶמֶשׁ עִם חַטָּאת וְאָשָׁם שֶׁל יוֹם מִפְּנֵי שֶׁמְּמַעֵט בְּאוֹכְלֵיהֶן וּבִמְקוֹם אֲכִילָתָן וּבִזְמַן אֲכִילָתָן. אֲבָל חַטָּאת וְאָשָׁם מִתְבַּשְּׁלִין זֶה עִם זֶה וְתוֹדָה וְאֵיל נָזִיר זֶה עִם זֶה וְהַבְּכוֹר וְחָזֶה וְשׁוֹק זֶה עִם זֶה. חֲתִיכָה שֶׁל קָדְשֵׁי קָדָשִׁים אוֹ שֶׁל פִּגּוּל אוֹ נוֹתָר שֶׁנִּתְבַּשְּׁלָה עִם חֲתִיכוֹת אֲחֵרוֹת. אוֹתָן הַחֲתִיכוֹת אֲסוּרוֹת לְזָרִים * וּמֻתָּרוֹת לַכֹּהֲנִים:

בְּשַׂר קָדְשֵׁי קָדָשִׁים אוֹ בְּשַׂר קָדָשִׁים קַלִּים שֶׁנִּתְבַּשֵּׁל עִם בְּשַׂר תַּאֲוָה הֲרֵי בְּשַׂר הַתַּאֲוָה אָסוּר לִטְמֵאִים וּמֻתָּר לִטְהוֹרִים:

] נֶאֱמַר בְּעוֹלָה . וְנֶאֱמַר בְּחַטָּאת . וְנֶאֱמַר בְּאָשָׁם . וְנֶאֱמַר בִּשְׁלָמִים . וְנֶאֱמַר בְּמִנְחָה . אֵין הַכָּתוּב מְדַבֵּר בְּכָל אֵלּוּ אֶלָּא בְּרָאוּי. שֶׁהַכֹּהֵן הָרָאוּי לַעֲבוֹדָה זֶהוּ שֶׁיֵּשׁ לוֹ חֵלֶק לֶאֱכל. וּמִי שֶׁאֵינוֹ רָאוּי בִּשְׁעַת הַקָּרְבָּן כְּגוֹן שֶׁהָיָה טָמֵא אֵין לוֹ חֵלֶק לֶאֱכל אֲפִלּוּ כְּשֶׁיִּטְהַר לָעֶרֶב. אֲבָל לְעִנְיַן חֲלוּקָה הַכּל לְאַנְשֵׁי בֵּית אָב שֶׁמַּקְרִיבִין בְּאוֹתוֹ הַיּוֹם. וְכֻלָּם חוֹלְקִים בְּכָל קָדְשֵׁי הַמִּקְדָּשׁ אִישׁ כְּאָחִיו. בֵּין זֶה שֶׁהִקְרִיב בֵּין אָחִיו שֶׁעִמּוֹ בַּמִּקְדָּשׁ שֶׁלֹּא הִקְרִיב:

וְלָמָּה חִלֵּק הַכָּתוּב בִּמְנָחוֹת בֵּין מְנָחוֹת הָאֲפוּיוֹת וּמִנְחַת הַסּלֶת. שֶׁהֲרֵי בַּאֲפוּיוֹת הוּא אוֹמֵר . וּבְמִנְחַת הַסּלֶת הוּא אוֹמֵר . שֶׁהָאֲפוּיָה כְּשֶׁחוֹלְקִין אוֹתָהּ אַנְשֵׁי בֵּית אָב כָּל שֶׁיַּגִּיעַ לוֹ חֶלְקוֹ אֲפִלּוּ כְּזַיִת פַּת הֲרֵי הוּא רָאוּי לוֹ שֶׁהֲרֵי אוֹכְלוֹ מִיָּד. אֲבָל הַסּלֶת אִם יְחַלְּקוּהָ בֵּינֵיהֶן נִמְצָא מַגִּיעַ לָזֶה מְלֹא כַּפּוֹ סלֶת אוֹ פָּחוֹת שֶׁאֵינוֹ רָאוּי לֹא לָלוּשׁ אוֹתוֹ וְלֹא לֶאֱפוֹתוֹ לְפִיכָךְ הָיָה עוֹלֶה עַל הַדַּעַת שֶׁיַּחְלְקוּ מִנְחָה כְּנֶגֶד מִנְחָה וְלֹא יַחְלְקוּ מִנְחָה זוֹ בִּפְנֵי עַצְמָהּ עַל כָּל אַנְשֵׁי בֵּית אָב. הֻצְרַךְ הַכָּתוּב לוֹמַר בָּהּ לְכָל בְּנֵי אַהֲרֹן תִּהְיֶה אִישׁ כְּאָחִיו לוֹמַר שֶׁיְּחַלְּקוּהָ בִּפְנֵי עַצְמָהּ. מִכָּאן אָמְרוּ חֲכָמִים ] אֵין חוֹלְקִין מִנְחָה כְּנֶגֶד מִנְחָה אֲפִלּוּ מַעֲשֵׂה מַחֲבַת כְּנֶגֶד מַעֲשֵׂה מַחֲבַת אוֹ סלֶת כְּנֶגֶד סלֶת. אֶלָּא כָּל אַחַת וְאַחַת חוֹלְקִין שְׁיָרֶיהָ בִּפְנֵי עַצְמָהּ: (-)

וְכֵן אֵין חוֹלְקִין עוֹף כְּנֶגֶד עוֹף. וְלֹא חַטָּאת כְּנֶגֶד חַטָּאת. וְלֹא חָזֶה וְשׁוֹק כְּנֶגֶד חָזֶה וְשׁוֹק. אֶלָּא כָּל חֵלֶק וְחֵלֶק חוֹלְקִין אוֹתוֹ בֵּינֵיהֶן בְּשָׁוֵה:

אֵין הַקָּטָן חוֹלֵק אֲפִלּוּ בְּקָדָשִׁים קַלִּים אַף עַל פִּי שֶׁמֻּתָּר לְהַאֲכִילוֹ קָדְשֵׁי קָדָשִׁים. וְכֵן אִשָּׁה וְאַנְדְּרוֹגִינוּס אֵין חוֹלְקִין לָהֶם בְּקָדְשֵׁי הַמִּקְדָּשׁ כְּלָל שֶׁנֶּאֱמַר . אֲבָל בַּעַל מוּם בֵּין קָבוּעַ בֵּין עוֹבֵר בֵּין שֶׁנּוֹלַד בְּמוּמוֹ בֵּין שֶׁהָיָה תָּמִים וְנִפְסַל חוֹלֵק וְאוֹכֵל שֶׁנֶּאֱמַר וְגוֹ'. וְהוּא שֶׁיִּהְיֶה רָאוּי לַאֲכִילָה. אֲבָל אִם הָיָה טָמֵא אֵינוֹ חוֹלֵק לֶאֱכל לָעֶרֶב:

וְכֹהֵן גָּדוֹל אוֹכֵל שֶׁלֹּא בַּחֲלוּקָה אֶלָּא נוֹטֵל כָּל מַה שֶּׁיִּרְצֶה:

כָּל הָרָאוּי לַאֲכִילַת הַקָּדָשִׁים בִּשְׁעַת הַקָּרְבָּן חוֹלֵק לֶאֱכל. וְכָל מִי שֶׁאֵינוֹ רָאוּי לֶאֱכל בִּשְׁעַת הָעֲבוֹדָה אַף עַל פִּי שֶׁהוּא רָאוּי לַעֲבוֹדָה וַהֲרֵי הוּא רָאוּי לֶאֱכל לָעֶרֶב אֵינוֹ חוֹלֵק כְּדֵי לְהָנִיחַ חֶלְקוֹ לָעֶרֶב. כֵּיצַד. טְבוּל יוֹם וּמְחֻסַּר כִּפּוּרִים וְהָאוֹנֵן בְּיוֹם הַקְּבוּרָה בֵּין כֹּהֵן גָּדוֹל בֵּין כֹּהֵן הֶדְיוֹט אֵין חוֹלְקִין לֶאֱכל לָעֶרֶב:

* כָּל שֶׁאֵינוֹ רָאוּי לֶאֱכל אֵינוֹ רָאוּי לַעֲבֹד. חוּץ מִכֹּהֵן גָּדוֹל הָאוֹנֵן שֶׁהוּא מַקְרִיב וְאֵינוֹ אוֹכֵל כְּמוֹ שֶׁבֵּאַרְנוּ. וְכָל שֶׁאֵינוֹ רָאוּי לַעֲבֹד אֵינוֹ רָאוּי לֶאֱכל חוּץ מִבַּעַל מוּם שֶׁהֲרֵי נִתְפָּרֵשׁ בַּתּוֹרָה:

כָּל שֶׁאֵין לוֹ חֵלֶק בַּבָּשָׂר אֵין לוֹ חֵלֶק בָּעוֹרוֹת. אֲפִלּוּ טָמֵא בִּשְׁעַת זְרִיקַת דָּמִים וְטָהוֹר בִּשְׁעַת הֶקְטֵר חֲלָבִים אֵינוֹ חוֹלֵק בַּבָּשָׂר שֶׁנֶּאֱמַר . עַד שֶׁיִּהְיֶה טָהוֹר וְרָאוּי לַעֲבוֹדָה מִשְּׁעַת זְרִיקָה עַד ] אַחַר הֶקְטֵר חֲלָבִים:

נִטְמָא אַחַר זְרִיקָה וְטָבַל וַהֲרֵי הוּא טָהוֹר אַף בִּשְׁעַת הֶקְטֵר חֲלָבִים הַדָּבָר סָפֵק אִם חוֹלֵק אִם אֵינוֹ חוֹלֵק לְפִיכָךְ אִם תָּפַס אֵין מוֹצִיאִין מִיָּדוֹ:
