הלכות מעשה הקרבנות - פרק ראשון כָּל הַקָּרְבָּנוֹת שֶׁל מִינֵי נֶפֶשׁ חַיָּה בָּאִין מֵחֲמִשָּׁה מִינִין בִּלְבַד. מִן הַבֹּקֶר וּמִן הַכְּבָשִׂים וּמִן הָעִזִּים וּמִן הַתּוֹרִים וּמִן בְּנֵי הַיּוֹנָה: וְכָל הַקָּרְבָּנוֹת בֵּין שֶׁל צִבּוּר בֵּין שֶׁל יָחִיד אַרְבָּעָה מִינִין. עוֹלָה. וְחַטָּאת. וְאָשָׁם. וּשְׁלָמִים: וְעוֹד יֵשׁ שָׁם שְׁלֹשָׁה מִינֵי קָרְבַּן יָחִיד וְהֵם הַפֶּסַח. וְהַבְּכוֹר. וְהַמַּעֲשֵׂר: כָּל קָרְבָּנוֹת הַצִּבּוּר הֵן עוֹלָה אוֹ חַטָּאת. וְאֵין בְּקָרְבְּנוֹת הַצִּבּוּר שְׁלָמִים חוּץ מִשְּׁנֵי כְּבָשִׂים הַבָּאִים עִם לֶחֶם הַתְּנוּפָה בַּעֲצֶרֶת. וְהֵם הַנִּקְרָאִים זִבְחֵי שַׁלְמֵי צִבּוּר. וְאֵין הַצִּבּוּר מַקְרִיבִין אָשָׁם לְעוֹלָם וְלֹא עוֹף: קָרְבְּנוֹת הַצִּבּוּר הֵם שְׁנֵי תְּמִידִין שֶׁל כָּל יוֹם. וּמוּסְפֵי שַׁבָּתוֹת וְרָאשֵׁי חֳדָשִׁים וְהַמּוֹעֲדוֹת וּשְׂעִיר חַטָּאת שֶׁל יוֹם הַכִּפּוּרִים. וְכֵן אִם שָׁגְגוּ בֵּית דִּין וְהוֹרוּ בַּעֲבוֹדָה זָרָה מֵבִיא כָּל שֵׁבֶט וְשֵׁבֶט פַּר וְשָׂעִיר. הַפָּר עוֹלָה וְהַשָּׂעִיר חַטָּאת. וְאֵלּוּ הַשְּׂעִירִים הֵם הַנִּקְרָאִים שְׂעִירֵי עֲבוֹדָה זָרָה. וְאִם שָׁגְגוּ וְהוֹרוּ בִּשְׁאָר הַמִּצְוֹת מְבִיאִין פַּר לְחַטָּאת וְהוּא הַנִּקְרָא פַּר הֶעְלֵם דָּבָר שֶׁל צִבּוּר: קָרְבְּנוֹת הַיָּחִיד הֵם הַבְּכוֹר וְהַמַּעֲשֵׂר וְהַפֶּסַח. וְהַחֲגִיגָה וְהִיא שְׁלָמִים. וְהָרְאִיָּה וְהִיא עוֹלוֹת. וְקָרְבַּן הַגֵּר וְהוּא עוֹלָה מִן הַבְּהֵמָה. אוֹ שְׁנֵי בְּנֵי יוֹנָה אוֹ שְׁתֵּי תּוֹרִים וּשְׁנֵיהֶן עוֹלָה. אוֹ שְׁתֵּי בְּהֵמוֹת אַחַת עוֹלָה וְאַחַת שְׁלָמִים. וְהַנּוֹדֵר אוֹ הַמִּתְנַדֵּב עוֹלָה אוֹ שְׁלָמִים. וּשְׁלָמִים הַבָּאִין עִם הַלֶּחֶם הֵם הַנִּקְרָאִים תּוֹדָה. וְכֵן קָרְבְּנוֹת הַנָּזִיר וְהֵן עוֹלָה וְחַטָּאת וּשְׁלָמִים וְקָרְבְּנוֹת מְצֹרָע וְהֵן חַטָּאת וְאָשָׁם וְעוֹלָה. וְקָרְבְּנוֹת זָבִים וְיוֹלְדוֹת וְהֵן חַטָּאת וְעוֹלָה. וְקָרְבַּן הַשּׁוֹגֵג בְּמִצְוַת לֹא תַּעֲשֶׂה שֶׁיֵּשׁ בָּהּ כָּרֵת וְהוּא חַטָּאת. וְאִם נִסְתַּפֵּק לוֹ אִם עָשָׂה אוֹ לֹא עָשָׂה אוֹתוֹ הַחוֹטֵא מֵבִיא אָשָׁם וְהוּא הַנִּקְרָא אָשָׁם תָּלוּי. וְיֵשׁ עֲבֵרוֹת שֶׁמֵּבִיא עֲלֵיהֶן אָשָׁם וְהוּא הַנִּקְרָא אָשָׁם וַדַּאי. וְכֵן אֵיל הָעוֹלָה וּפַר הַחַטָּאת שֶׁמַּקְרִיב כֹּהֵן גָּדוֹל מִשֶּׁלּוֹ בְּיוֹם הַכִּפּוּרִים הֲרֵי הֵן קָרְבַּן יָחִיד. וּפַר זֶה הוּא הַנִּקְרָא פַּר יוֹם הַכִּפּוּרִים. וְכָל אֵלּוּ הַקָּרְבָּנוֹת מְפֹרָשִׁין הֵן בַּתּוֹרָה וְכָל אֶחָד מֵהֶן יִתְבָּאֲרוּ דִּינָיו בִּמְקוֹמוֹ: כָּל קָרְבְּנוֹת הַיָּחִיד חַיָּב בְּאַחֲרָיוּתָן וּבְאַחֲרָיוּת נִסְכֵּיהֶן חוּץ מִן הַנְּדָבָה. וְכָל קָרְבְּנוֹת הַצִּבּוּר אֵינָן חַיָּבִין בְּאַחֲרָיוּתָן וְלֹא בְּאַחֲרָיוּת נִסְכֵּיהֶן. וְאִם קָרַב הַזֶּבַח חַיָּבִין בְּאַחֲרָיוּת נִסְכֵּיהֶם. הַכְּבָשִׂים וּמִן הָעִזִּים וּמִן הַבָּקָר בֵּין גְּדוֹלִים בֵּין קְטַנִּים. וּמִן הַתּוֹרִים וּבְנֵי יוֹנָה וְאֶחָד בָּהֶן הַזָּכָר וְהַנְּקֵבָה: הַחַטָּאת בָּאָה מֵחֲמֵשֶׁת הַמִּינִים הָאֵלּוּ מִן הַקְּטַנִּים: הַשְּׁלָמִים בָּאִים מִן הַכְּבָשִׂים וּמִן הָעִזִּים וּמִן הַבָּקָר מִזְּכָרִים וּמִנְּקֵבוֹת מִן הַגְּדוֹלִים וּמִן הַקְּטַנִּים. וְאֵין הָעוֹף בָּא שְׁלָמִים. הַקְּטַנִּים הֵם מִבֶּן וְשָׁלֹשׁ לַצִּבּוּר: כָּל הַקָּרְבָּנוֹת הָאֵלּוּ נִקְרָאִים זְבָחִים. וְכָל הָעוֹלוֹת וְהַחַטָּאוֹת וְהָאֲשָׁמוֹת וּשְׁנֵי כַּבְשֵׂי שְׁלָמִים שֶׁל עֲצֶרֶת נִקְרָאִים קָדְשֵׁי קָדָשִׁים. אֲבָל שְׁלָמִים שֶׁל יָחִיד וְהַבְּכוֹר וְהַמַּעֲשֵׂר וְהַפֶּסַח נִקְרָאִים קָדָשִׁים קַלִּים: הָאֵיבָרִים שֶׁשּׂוֹרְפִין אוֹתָן עַל גַּבֵּי הַמִּזְבֵּחַ מִן הַחַטָּאוֹת הַנֶּאֱכָלוֹת וּמִן הָאֲשָׁמוֹת וּמִן הַשְּׁלָמִים הֵן הַנִּקְרָאִין אֵימוּרִין. וְאֵלּוּ הֵן הָאֵימוּרִין שֶׁל שׁוֹר אוֹ שֶׁל עֵז. הַחֵלֶב אֲשֶׁר עַל הַקֶּרֶב וּבִכְלָלוֹ חֵלֶב שֶׁעַל גַּבֵּי הַקֵּבָה וּשְׁתֵּי הַכְּלָיוֹת וְחֵלֶב שֶׁעֲלֵיהֶן עִם הַחֵלֶב אֲשֶׁר עַל הַכְּסָלִים וְיוֹתֶרֶת הַכָּבֵד וְנוֹטֵל מִן הַכָּבֵד מְעַט עִם הַיּוֹתֶרֶת. וְאִם הָיָה הַקָּרְבָּן מִמִּין הַכְּבָשִׂים מוֹסִיף עַל אֵלּוּ הָאַלְיָה תְּמִימָה עִם הַחֻלְיוֹת מִן הַשִּׁדְרָה עַד מְקוֹם הַכְּלָיוֹת שֶׁנֶּאֱמַר . וְכָל הָאֵימוּרִין נִשְׂרָפִין עַל מִזְבֵּחַ הַחִיצוֹן: הָיְתָה הַבְּהֵמָה מְעֻבֶּרֶת אַף עַל פִּי שֶׁכָּלוּ חֳדָשָׁיו שֶׁל עֻבָּר וַאֲפִלּוּ נִמְצָא חַי אֵינוֹ מַעֲלֶה חֶלְבּוֹ עִם חֵלֶב אִמּוֹ אֶלָּא חֵלֶב אִמּוֹ בִּלְבַד. וַהֲרֵי הָעֻבָּר כְּאֶחָד מֵאֵיבָרֶיהָ:
