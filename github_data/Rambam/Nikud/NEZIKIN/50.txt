הלכות רוצח ושמירת נפש - פרק ראשון

כָּל ] הוֹרֵג נֶפֶשׁ בֶּן אָדָם עוֹבֵר בְּלֹא תַּעֲשֶׂה שֶׁנֶּאֱמַר . וְאִם רָצַח בְּזָדוֹן בִּפְנֵי עֵדִים מִיתָתוֹ בְּסַיִף שֶׁנֶּאֱמַר . מִפִּי הַשְּׁמוּעָה לָמְדוּ שֶׁזּוֹ מִיתַת סַיִף. בֵּין שֶׁהָרַג אֶת חֲבֵרוֹ בְּבַרְזֶל בֵּין שֶׁשְּׂרָפוֹ בָּאֵשׁ מִיתָתוֹ בְּסַיִף: (-)

מִצְוָה בְּיַד גּוֹאֵל הַדָּם שֶׁנֶּאֱמַר . וְכָל הָרָאוּי לִירֻשָּׁה הוּא גּוֹאֵל הַדָּם. לֹא רָצָה גּוֹאֵל הַדָּם אוֹ שֶׁלֹּא הָיָה יָכוֹל לַהֲמִיתוֹ אוֹ שֶׁאֵין לוֹ גּוֹאֵל דָּם בֵּית דִּין מְמִיתִין אֶת הָרוֹצֵחַ בְּסַיִף:

הָאָב שֶׁהָרַג אֶת בְּנוֹ. אִם הָיָה בֶּן לַנֶּהֱרָג הֲרֵי זֶה הוֹרֵג אֲבִי אָבִיו מִפְּנֵי שֶׁהוּא גּוֹאֵל. וְאִם אֵין לוֹ בֵּן אֵין אֶחָד מִן הָאַחִין נַעֲשֶׂה גּוֹאֵל הַדָּם לְהָמִית אֶת אָבִיו אֶלָּא בֵּית דִּין מְמִיתִין אוֹתוֹ. וְאֶחָד הַזָּכָר וְאֶחָד הַנְּקֵבָה בִּגְאֻלַּת הַדָּם:

וּמֻזְהָרִין בֵּית דִּין ] שֶׁלֹּא לִקַּח כֹּפֶר מִן הָרוֹצֵחַ. וַאֲפִלּוּ נָתַן כָּל מָמוֹן שֶׁבָּעוֹלָם וַאֲפִלּוּ רָצָה גּוֹאֵל הַדָּם לְפָטְרוֹ. שֶׁאֵין נַפְשׁוֹ שֶׁל זֶה הַנֶּהֱרָג קִנְיַן גּוֹאֵל הַדָּם אֶלָּא קִנְיַן הַקָּדוֹשׁ בָּרוּךְ הוּא שֶׁנֶּאֱמַר . וְאֵין לְךָ דָּבָר שֶׁהִקְפִּידָה תּוֹרָה עָלָיו כִּשְׁפִיכוּת דָּמִים שֶׁנֶּאֱמַר וְגוֹ'. וְגוֹ':

רוֹצֵחַ שֶׁהָרַג בְּזָדוֹן אֵין מְמִיתִין אוֹתוֹ הָעֵדִים וְלֹא הָרוֹאִים אוֹתוֹ עַד שֶׁיָּבוֹא לְבֵית דִּין וִידִינוּהוּ לְמִיתָה. שֶׁנֶּאֱמַר . וְהוּא הַדִּין לְכָל מְחֻיְּבֵי מִיתוֹת בֵּית דִּין שֶׁעָבְרוּ וְעָשׂוּ שֶׁאֵין מְמִיתִין אוֹתָן עַד שֶׁיִּגָּמֵר דִּינָם בְּבֵית דִּין:

בַּמֶּה דְּבָרִים אֲמוּרִים בְּשֶׁעָבַר וְעָשָׂה הֶעָוֹן שֶׁחַיָּב עָלָיו מִיתַת בֵּית דִּין. אֲבָל הָרוֹדֵף אַחַר חֲבֵרוֹ לְהָרְגוֹ אֲפִלּוּ הָיָה הָרוֹדֵף קָטָן הֲרֵי כָּל יִשְׂרָאֵל מְצֻוִּין לְהַצִּיל הַנִּרְדָּף מִיַּד הָרוֹדֵף וַאֲפִלּוּ בְּנַפְשׁוֹ שֶׁל רוֹדֵף:

כֵּיצַד. אִם הִזְהִירוּהוּ וַהֲרֵי הוּא רוֹדֵף אַחֲרָיו אַף עַל פִּי שֶׁלֹּא קִבֵּל עָלָיו הַתְרָאָה כֵּיוָן שֶׁעֲדַיִן הוּא רוֹדֵף הֲרֵי זֶה נֶהֱרָג. וְאִם יְכוֹלִים לְהַצִּילוֹ בְּאֵיבָר מֵאֵיבְרֵי הָרוֹדֵף כְּגוֹן שֶׁיַּכּוּ אוֹתוֹ בְּחֵץ אוֹ בְּאֶבֶן אוֹ בְּסַיִף וְיִקְטְעוּ אֶת יָדוֹ אוֹ יִשְׁבְּרוּ אֶת רַגְלוֹ אוֹ יְסַמּוּ אֶת עֵינוֹ עוֹשִׂין. וְאִם יְכוֹלִין לְכַוֵּן וּלְהַצִּילוֹ אֶלָּא אִם כֵּן הֲרָגוּהוּ לַרוֹדֵף הֲרֵי אֵלּוּ הוֹרְגִין אוֹתוֹ אַף עַל פִּי שֶׁעֲדַיִן לֹא הָרַג * שֶׁנֶּאֱמַר :

אֶחָד בִּמְבוּשָׁיו וְאֶחָד כָּל דָּבָר שֶׁיֵּשׁ בּוֹ סַכָּנַת נְפָשׁוֹת. אֶחָד הָאִישׁ אוֹ הָאִשָּׁה. ] עִנְיַן הַכָּתוּב שֶׁכָּל הַחוֹשֵׁב לְהַכּוֹת חֲבֵרוֹ הַכָּאָה הַמְּמִיתָה אוֹתוֹ מַצִּילִין אֶת הַנִּרְדָּף בְּכַפּוֹ שֶׁל רוֹדֵף. וְאִם אֵינָן יְכוֹלִין מַצִּילִין אוֹתוֹ אַף בְּנַפְשׁוֹ שֶׁנֶּאֱמַר : (-)

אַף זוֹ מִצְוַת לֹא תַּעֲשֶׂה שֶׁלֹּא לָחוּס עַל נֶפֶשׁ הָרוֹדֵף. לְפִיכָךְ הוֹרוּ חֲכָמִים שֶׁהָעֻבָּרָה שֶׁהִיא מַקְשָׁה לֵילֵד מֻתָּר לַחְתֹּךְ הָעֵבָּר בְּמֵעֶיהָ בֵּין בְּסַם בֵּין בְּיָד מִפְּנֵי שֶׁהוּא כְּרוֹדֵף אַחֲרֶיהָ לְהָרְגָהּ. וְאִם מִשֶּׁהוֹצִיא רֹאשׁוֹ אֵין נוֹגְעִין בּוֹ שֶׁאֵין דּוֹחִין נֶפֶשׁ מִפְּנֵי נֶפֶשׁ וְזֶהוּ טִבְעוֹ שֶׁל עוֹלָם:

אֶחָד הָרוֹדֵף אַחַר חֲבֵרוֹ לְהָרְגוֹ אוֹ רוֹדֵף אַחַר נַעֲרָה מְאֹרָסָה לְאָנְסָהּ. שֶׁנֶּאֱמַר . וַהֲרֵי הוּא אוֹמֵר . הָא יֵשׁ לָהּ מוֹשִׁיעַ מוֹשִׁיעָהּ בְּכָל דָּבָר שֶׁיָּכוֹל לְהוֹשִׁיעַ וַאֲפִלּוּ בַּהֲרִיגַת הָרוֹדֵף: (-)

וְהוּא הַדִּין ] לִשְׁאָר כָּל הָעֲרָיוֹת חוּץ מִן הַבְּהֵמָה. אֲבָל הַזָּכוּר מַצִּילִין אוֹתוֹ בְּנֶפֶשׁ הָרוֹדֵף כִּשְׁאָר כָּל הָעֲרָיוֹת. אֲבָל הָרוֹדֵף אַחַר הַבְּהֵמָה לְרָבְעָהּ. אוֹ שֶׁרָדַף לַעֲשׂוֹת מְלָאכָה בְּשַׁבָּת אוֹ לַעֲבֹד עֲבוֹדָה זָרָה. אַף עַל פִּי שֶׁהַשַּׁבָּת וַעֲבוֹדָה זָרָה עִקְּרֵי הַדָּת אֵין מְמִיתִין אוֹתוֹ עַד שֶׁיַּעֲשֶׂה וִיבִיאוּהוּ לְבֵית דִּין וִידִינוּהוּ וְיָמוּת:

רָדַף אַחַר עֶרְוָה וּתְפָשָׂהּ וְשָׁכַב עִמָּהּ וְהֶעֱרָה. אַף עַל פִּי שֶׁלֹּא גָּמַר בִּיאָתוֹ אֵין מְמִיתִין אוֹתוֹ עַד עָמְדוֹ בַּדִּין. רָדַף אַחַר עֶרְוָה וַאֲחֵרִים הָיוּ רוֹדְפִין אַחֲרָיו לְהַצִּילָהּ. וְאָמְרָה לָהֶם הֲנִיחוּהוּ כְּדֵי שֶׁלֹּא יַהַרְגֵנִי. אֵין שׁוֹמְעִין לָהּ אֶלָּא מַבְהִילִין אוֹתוֹ וּמוֹנְעִין אוֹתוֹ מִלִּבְעל בְּאֵיבָרָיו. וְאִם אֵינָן יְכוֹלִים לְמָנְעוֹ בְּאֵיבָרָיו אֲפִלּוּ בְּנַפְשׁוֹ כְּמוֹ שֶׁבֵּאַרְנוּ:

o כָּל הַיָּכוֹל לְהַצִּיל בְּאֵיבָר מֵאֵיבָרָיו וְלֹא טָרַח בְּכָךְ אֶלָּא הִצִּיל בְּנַפְשׁוֹ שֶׁל רוֹדֵף וַהֲרָגוֹ הֲרֵי זֶה שׁוֹפֵךְ דָּמִים וְחַיָּב מִיתָה אֲבָל אֵין בֵּית דִּין מְמִיתִין אוֹתוֹ:

כָּל הַיָּכוֹל לְהַצִּיל וְלֹא הִצִּיל עוֹבֵר עַל . וְכֵן הָרוֹאֶה אֶת חֲבֵרוֹ טוֹבֵעַ בַּיָּם. אוֹ לִסְטִים בָּאִים עָלָיו. אוֹ חַיָּה רָעָה בָּאָה עָלָיו. וְיָכוֹל לְהַצִּילוֹ הוּא בְּעַצְמוֹ. אוֹ שֶׁיִּשְׂכֹּר אֲחֵרִים לְהַצִּילוֹ וְלֹא הִצִּיל. אוֹ שֶׁשָּׁמַע עוֹבְדֵי כּוֹכָבִים אוֹ מוֹסְרִים מְחַשְּׁבִים עָלָיו רָעָה אוֹ טוֹמְנִין לוֹ פַּח וְלֹא גִּלָּה אֹזֶן חֲבֵרוֹ וְהוֹדִיעוֹ. אוֹ שֶׁיָּדַע בְּעַכּוּ''ם אוֹ בְּאוֹנֵס שֶׁהוּא בָּא עַל חֲבֵרוֹ וְיָכוֹל לְפַיְּסוֹ בִּגְלַל חֲבֵרוֹ לְהָסִיר מַה שֶּׁבְּלִבּוֹ וְלֹא פִּיְּסוֹ. וְכָל כַּיּוֹצֵא בִּדְבָרִים אֵלּוּ. הָעוֹשֶׂה אוֹתָם עוֹבֵר עַל לֹא תַעֲמֹד עַל דַּם רֵעֶךָ: (-)

הָרוֹאֶה רוֹדֵף אַחַר חֲבֵרוֹ לְהָרְגוֹ אוֹ אַחַר עֶרְוָה לְבָעֳלָהּ וְיָכוֹל לְהַצִּיל וְלֹא הִצִּיל. הֲרֵי זֶה בִּטֵּל מִצְוַת עֲשֵׂה שֶׁהִיא . וְעָבַר עַל שְׁנֵי לָאוִין עַל וְעַל :

אַף עַל פִּי שֶׁאֵין לוֹקִין עַל לָאוִין אֵלּוּ מִפְּנֵי שֶׁאֵין מַעֲשֶׂה בָּהֶן חֲמוּרִים הֵם. ] שֶׁכָּל הַמְאַבֵּד נֶפֶשׁ אַחַת מִיִּשְׂרָאֵל כְּאִלּוּ אִבֵּד כָּל הָעוֹלָם כֻּלּוֹ. וְכָל הַמְקַיֵּם נֶפֶשׁ אַחַת מִיִּשְׂרָאֵל כְּאִלּוּ קִיֵּם כָּל הָעוֹלָם כֻּלּוֹ:
