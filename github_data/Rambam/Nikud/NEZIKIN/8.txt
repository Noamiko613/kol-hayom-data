הלכות נזקי ממון - פרק שמיני

] שׁוֹר שֶׁל יִשְׂרָאֵל שֶׁנָּגַח שׁוֹר שֶׁל הֶקְדֵּשׁ אוֹ שׁוֹר שֶׁל הֶקְדֵּשׁ שֶׁנָּגַח שׁוֹר שֶׁל יִשְׂרָאֵל פָּטוּר שֶׁנֶּאֱמַר . וְכָל הַקָּדָשִׁים שֶׁחַיָּבִין בָּהֶן מְעִילָה אֵין בָּהֶן דִּין נְזָקִין. וּפְסוּלֵי הַמֻּקְדָּשִׁין יֵשׁ בָּהֶן דִּין נְזָקִין בֵּין שֶׁהִזִּיקוּ בֵּין שֶׁהֻזְּקוּ שֶׁהֲרֵי יָצְאוּ לַפִּדְיוֹן וְלִהְיוֹתָם חֻלִּין:

שְׁלָמִים שֶׁהִזִּיקוּ גּוֹבֶה מִבְּשָׂרָם וְאֵינוֹ גּוֹבֶה מִן הַבָּשָׂר כְּנֶגֶד אֵימוּרֵיהֶן. שֶׁהָאֵימוּרִים שֶׁל קָדָשִׁים קַלִּים מוֹעֲלִין בָּהֶן כְּמוֹ שֶׁבֵּאַרְנוּ בְּהִלְכוֹת מְעִילָה. וְכֵן תּוֹדָה שֶׁהִזִּיקָה גּוֹבֶה מִבְּשָׂרָהּ וְאֵינוֹ גּוֹבֶה מִן הַלֶּחֶם הַבָּא עִמָּהּ שֶׁאֵין הַלֶּחֶם מִכְּלַל בָּשָׂר:

וְכֵיצַד גּוֹבֶה. שֶׁיֹּאכַל הַנִּזָּק וַחֲבוּרָתוֹ מִן הַבָּשָׂר בִּקְדֻשָּׁה כְּנֶגֶד חֲצִי נֵזֶק שֶׁלּוֹ. וְכֵיצַד אֵינוֹ גּוֹבֶה כְּנֶגֶד הָאֵימוּרִין. שֶׁאִם הָיָה לוֹ לִגְבּוֹת בַּחֲצִי נִזְקוֹ שְׁוֵה דִּינָר וְהָיָה כָּל הַבָּשָׂר עִם הָאֵימוּרִין שָׁוֶה שְׁנֵי דִּינָרִים וְהַבָּשָׂר בְּלֹא אֵימוּרִין שָׁוֶה דִּינָר וָחֵצִי אֵינוֹ גּוֹבֶה שְׁנֵי שְׁלִישִׁי הַבָּשָׂר אֶלָּא חֲצִי הַבָּשָׂר בִּלְבַד:

וְכֵן שׁוֹר הֶפְקֵר שֶׁהִזִּיק פָּטוּר שֶׁנֶּאֱמַר עַד שֶׁיִּהְיוּ הַנְּכָסִים מְיֻחָדִים לַבְּעָלִים. כֵּיצַד. שׁוֹר הַהֶפְקֵר שֶׁנָּגַח וְקֹדֶם שֶׁיִּתְפּשֹׁ אוֹתוֹ הַנִּזָּק בָּא אַחֵר וְזָכָה בּוֹ הֲרֵי זֶה פָּטוּר. וְלֹא עוֹד אֶלָּא שׁוֹר הַמְיֻחָד לַבְּעָלִים שֶׁהִזִּיק וְאַחַר שֶׁהִזִּיק הַקְדִּישׁוֹ אוֹ הִפְקִירוֹ הֲרֵי זֶה פָּטוּר עַד שֶׁיִּהְיוּ לוֹ בְּעָלִים בִּשְׁעַת הֶזֵּקוֹ וּבִשְׁעַת הַעֲמָדָה בַּדִּין:

* שׁוֹר שֶׁל יִשְׂרָאֵל שֶׁנָּגַח שׁוֹר שֶׁל עַכּוּ''ם בֵּין תָּם בֵּין מוּעָד פָּטוּר. לְפִי שֶׁאֵין הָעַכּוּ''ם מְחַיְּבִין אֶת הָאָדָם עַל בְּהֶמְתּוֹ שֶׁהִזִּיקָה וַהֲרֵי אָנוּ דָּנִין לָהֶם כְּדִינֵיהֶם. וְשׁוֹר שֶׁל עַכּוּ''ם שֶׁנָּגַח שֶׁל יִשְׂרָאֵל בֵּין תָּם בֵּין מוּעָד מְשַׁלֵּם נֵזֶק שָׁלֵם. קְנָס הוּא זֶה לְעַכּוּ''ם לְפִי שֶׁאֵינָן זְהִירִין בְּמִצְוֹת וְאֵינָן מְסַלְּקִין הַנֵּזֶק וְאִם לֹא תְּחַיֵּב אוֹתָן עַל נִזְקֵי בְּהֶמְתָּן אֵין מְשַׁמְּרִין אוֹתָהּ וּמַפְסִידִין מָמוֹן הַבְּרִיּוֹת:

] שׁוֹר תָּם שֶׁהִזִּיק אִם מְכָרוֹ הַמַּזִּיק עַד שֶׁלֹּא עָמַד בַּדִּין אַף עַל פִּי שֶׁהוּא מָכוּר הֲרֵי הַנִּזָּק גּוֹבֶה הֵימֶנּוּ וְחוֹזֵר הַלּוֹקֵחַ וְגוֹבֶה מִן הַמַּזִּיק שֶׁמָּכַר לוֹ. שֶׁכֵּיוָן שֶׁנָּגַח קוֹל יֵשׁ לוֹ וְלֹא הָיָה לוֹ לַלּוֹקֵחַ לִקַּח עַד שֶׁיִּגְבֶּה הַנִּזָּק:

הִקְדִּישׁוֹ הַמַּזִּיק הֲרֵי זֶה מֻקְדָּשׁ כְּדֵי שֶׁלֹּא יֹאמְרוּ הֶקְדֵּשׁ יוֹצֵא בְּלֹא פִּדְיוֹן. שְׁחָטוֹ גּוֹבֶה מִבְּשָׂרוֹ. נְתָנוֹ בְּמַתָּנָה מַה שֶּׁעָשָׂה עָשׂוּי וְיִגְבֶּה הַנִּזָּק מִמֶּנּוּ:

הִזִּיק וְעָמַד בַּדִּין וְאַחַר כָּךְ מְכָרוֹ אֵינוֹ מָכוּר. הִקְדִּישׁוֹ אֵינוֹ מֻקְדָּשׁ. נְתָנוֹ בְּמַתָּנָה לֹא עָשָׂה וְלֹא כְּלוּם. o קָדְמוּ בַּעֲלֵי חוֹבוֹת שֶׁל מַזִּיק וּתְפָסוּהוּ בֵּין שֶׁחָב עַד שֶׁלֹּא הִזִּיק בֵּין הִזִּיק עַד שֶׁלֹּא חָב לֹא זָכוּ אֶלָּא הַנִּזָּק גּוֹבֶה מִמֶּנּוּ שֶׁאֲפִלּוּ הָיָה שֶׁל בַּעֲלֵי חוֹבוֹת מִתְּחִלָּה וְהִזִּיק הֲרֵי זֶה גּוֹבֶה מִגּוּפוֹ:

מוּעָד שֶׁהִזִּיק בֵּין עָמַד בַּדִּין בֵּין שֶׁלֹּא עָמַד בַּדִּין וְהִקְדִּישׁוֹ אוֹ מְכָרוֹ אוֹ נְתָנוֹ בְּמַתָּנָה אוֹ שְׁחָטוֹ. מַה שֶּׁעָשָׂה עָשׂוּי. קָדְמוּ בַּעֲלֵי חוֹבוֹת וְהִנְהִיגוּהוּ בֵּין חָב עַד שֶׁלֹּא הִזִּיק בֵּין הִזִּיק עַד שֶׁלֹּא חָב זָכוּ בּוֹ. לְפִי שֶׁאֵין מִשְׁתַּלֵּם הַנִּזָּק אֶלָּא מִן הַמְעֻלֶּה שֶׁבְּנִכְסֵי הַמַּזִּיק וַהֲרֵי כָּל נְכָסָיו מְשֻׁעְבָּדִין לְנֶזֶק זֶה:

] כְּשֶׁבֵּית דִּין נִזְקָקִין לִגְבּוֹת לַנִּזָּק מִנִּכְסֵי הַמַּזִּיק גּוֹבִין מִן הַמִּטַּלְטְלִין תְּחִלָּה וְאִם לֹא הָיוּ לוֹ מִטַּלְטְלִין כְּלָל אוֹ שֶׁלֹּא הָיוּ לוֹ מִטַּלְטְלִין כְּנֶגֶד כָּל הַנֵּזֶק גּוֹבִין הַשְּׁאָר מִן הַקַּרְקַע הַמְעֵלָּה שֶׁבְּנִכְסֵי הַמַּזִּיק. וְכָל זְמַן שֶׁיִּמְצְאוּ מִטַּלְטְלִין וַאֲפִלּוּ סֻבִּין אֵין נִזְקָקִין לַקַּרְקַע:

מֵת הַמַּזִּיק קֹדֶם שֶׁיְּשַׁלֵּם אֵין בֵּית דִּין נִזְקָקִין לַמִּטַּלְטְלִין שֶׁל יְתוֹמִים אֶלָּא לַקַּרְקַע. וְגוֹבִין לַנִּזָּק מִן הַזִּבּוּרִית. מִפְּנֵי שֶׁהַנִּזָק נַעֲשָׂה כְּבַעַל חוֹב וְהַמִּטַּלְטְלִין אֵינָן מְשֻׁעְבָּדִין לְבַעַל חוֹב. וְאִם תָּפַס הַנִּזָּק הַמִּטַּלְטְלִין בְּחַיֵּי הַמַּזִּיק גּוֹבִין לוֹ מֵהֶם לְאַחַר מוֹתוֹ:

כְּבָר תִּקְּנוּ הַגְּאוֹנִים לִגְבּוֹת בַּעַל חוֹב מִן הַמִּטַּלְטְלִין. וּפָשְׁטָה תַּקָּנָה זוֹ בְּכָל בָּתֵּי דִּינִין. לְפִיכָךְ מְגַבִּין הַנְּזָקִין מִן הַמִּטַּלְטְלִין שֶׁל יְתוֹמִים. וְאִם לֹא הִנִּיחַ מִטַּלְטְלִין גּוֹבִין לוֹ מִן הַזִּבּוּרִית שֶׁכָּל הַבָּא לִפָּרַע מִנִּכְסֵי יְתוֹמִים לֹא יִפָּרַע אֶלָּא מִן הַזִּבּוּרִית כְּמוֹ שֶׁבֵּאַרְנוּ:

] * אֵין הַנְּזָקִין מִשְׁתַּלְּמִין וְאֵין חַיָּבִין בְּכֹפֶר וְאֵין הַבְּהֵמָה נֶהֱרֶגֶת אֶלָּא בִּרְאָיָה בְּרוּרָה וּבְעֵדִים הַכְּשֵׁרִים לְהָעִיד. שֶׁלֹּא תֹּאמַר הוֹאִיל וְאֵין מְצוּיִין בְּאֻרְווֹת הַסּוּסִים וּבְרֶפֶת הַבָּקָר וְגִדְרוֹת צֹאן אֶלָּא הָעֲבָדִים וְהָרוֹעִים וְכַיּוֹצֵא בָּהֶן אִם הֵעִידוּ שֶׁבְּהֵמָה זוֹ הִזִּיקָה אֶת זוֹ שׁוֹמְעִין לָהֶם וְאִם הֵעִידוּ קְטַנִּים אוֹ נָשִׁים שֶׁאָדָם זֶה חָבַל אֶת זֶה אוֹ הֵעִידוּ בִּשְׁאָר נְזָקִין סוֹמְכִין עֲלֵיהֶן. אֵין הַדָּבָר כֵּן. אֶלָּא אֵין מְחַיְּבִין לְעוֹלָם מָמוֹן עַל פִּי עֵדִים עַד שֶׁיִּהְיוּ עֵדִים הַכְּשֵׁרִים לְהָעִיד שְׁאָר עֵדֻיּוֹת וְיָעִידוּ וִיחַיְּבוּ בֵּית דִּין הַמַּזִּיק לְשַׁלֵּם:

שׁוֹר שֶׁהָיָה רוֹעֶה עַל גַּבֵּי הַנָּהָר וְנִמְצָא ] שׁוֹר הָרוּג בְּצִדּוֹ. אַף עַל פִּי שֶׁזֶּה מְנֻגָּח וְזֶה מוּעָד לִגַּח זֶה מְנֻשָּׁךְ וְזֶה מוּעָד לִשַּׁךְ. אֵין אוֹמְרִים בְּיָדוּעַ שֶׁזֶּה נְשָׁכוֹ וְזֶה נְגָחוֹ. וַאֲפִלּוּ גָּמָל הָאוֹחֵר בֵּין הַגְּמַלִּים וְנִמְצָא הָרוּג בְּצִדּוֹ אֵין אוֹמְרִים בְּיָדוּעַ שֶׁזֶּה הֲרָגוֹ עַד שֶׁרָאוּהוּ עֵדִים כְּשֵׁרִים:
