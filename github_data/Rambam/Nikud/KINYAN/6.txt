אֵינוֹ נִקְנֶה בְּקִנְיָן. נִמְצָא הַמַּטְבֵּעַ אֵינוֹ נִקְנֶה בְּקִנְיָן וְאֵינוֹ נַעֲשֶׂה הוּא עַצְמוֹ קִנְיָן לִקְנוֹת כִּשְׁאָר דְּבָרִים: לְשׁוֹנוֹת שֶׁל זָהָב וְשֶׁל כֶּסֶף הֲרֵי הֵן כַּעֲשָׁשִׁיּוֹת שֶׁל בַּרְזֶל אוֹ שֶׁל נְחשֶׁת. ] וְהַכּל כִּשְׁאָר הַמִּטַּלְטְלִין הֵן. וְנִקְנִין בְּקִנְיָן וְקוֹנִין זֶה אֶת זֶה בְּהַחְלָפָה כְּמוֹ שֶׁבֵּאַרְנוּ. אֲבָל הַמַּטְבְּעוֹת שֶׁל כֶּסֶף אוֹ דִּינָרִין שֶׁל זָהָב . אוֹ הַמָּעוֹת שֶׁל נְחשֶׁת. הֲרֵי הֵן כֻּלָּן דָּמִים כְּנֶגֶד שְׁאָר הַמִּטַּלְטְלִין. וְהַנּוֹתֵן אֶת אֶחָד מֵהֶן דְּמֵי הַמִּטַּלְטְלִין לֹא קָנָה עַד שֶׁיִּמְשֹׁךְ אוֹ יַגְבִּיהַּ כְּמוֹ שֶׁבֵּאַרְנוּ. וְאֵין אֶחָד מֵהֶן נִקְנֶה בְּקִנְיָן וְלֹא נַעֲשֶׂה קִנְיָן: בַּמֶּה דְּבָרִים אֲמוּרִים בִּזְמַן שֶׁקּוֹנֶה שְׁאָר מִטַּלְטְלִין בְּאֶחָד מִמִּינֵי מַטְבְּעוֹת אֵלּוּ אוֹ עֲבָדִים וְקַרְקָעוֹת. אֲבָל הַדִּינָרִין שֶׁל זָהָב לְגַבֵּי מַטְבְּעוֹת שֶׁל כֶּסֶף הֲרֵי הֵן כְּפֵרוֹת. ] וְכֵן הַמָּעוֹת שֶׁל נְחשֶׁת כְּמוֹ פֵּרוֹת לְגַבֵּי מַטְבְּעוֹת שֶׁל כֶּסֶף: כֵּיצַד. נָתַן לוֹ דִּינָר שֶׁל זָהָב בְּעֶשְׂרִים וַחֲמִשָּׁה דִּינָר שֶׁל כֶּסֶף נִקְנֶה הַכֶּסֶף אַף עַל פִּי שֶׁעֲדַיִן לֹא בָּא הַכֶּסֶף לְיָדוֹ. וְחַיָּב לִתֵּן לוֹ עֶשְׂרִים וַחֲמִשָּׁה דִּינָר שֶׁל כֶּסֶף כְּמוֹ שֶׁפָּסַק עִמּוֹ. אִם חֲדָשִׁים חֲדָשִׁים אִם יְשָׁנִים יְשָׁנִים. אֲבָל אִם נָתַן לוֹ עֶשְׂרִים וַחֲמִשָּׁה כֶּסֶף בְּדִינַר זָהָב לֹא קָנָה עַד שֶׁיִּקַּח הַדִּינָר שֶׁל זָהָב. וְכָל אֶחָד יָכוֹל לַחְזֹר בּוֹ: נָתַן לוֹ שְׁלֹשִׁים אִיסָר שֶׁל נְחשֶׁת בְּדִינָר שֶׁל כֶּסֶף. נִקְנֶה הַדִּינָר שֶׁל כֶּסֶף וְחַיָּב לִתֵּן לוֹ כְּמוֹ שֶׁפָּסַק עִמּוֹ. אִם דִּינָר חָדָשׁ חָדָשׁ אִם יָשָׁן יָשָׁן. אֲבָל אִם נָתַן לוֹ דִּינָר שֶׁל כֶּסֶף בִּשְׁלֹשִׁים אִיסָר. לֹא קָנָה עַד שֶׁיִּקַּח הָאִיסָרוֹת שֶׁל נְחשֶׁת. וְכָל אֶחָד מֵהֶן יָכוֹל לַחְזֹר בּוֹ: וְכֵן מָעוֹת הָרָעוֹת שֶׁפְּסָלָתָן מַלְכוּת אוֹ הַמְּדִינָה אוֹ דִּינָרִין שֶׁאֵינָן יוֹצְאִין בְּאוֹתָהּ מְדִינָה וְאֵין נוֹשְׂאִין וְנוֹתְנִין בָּהֶן עַד שֶׁמְּשַׁנִּין אוֹתָן לְמַטְבֵּעַ
