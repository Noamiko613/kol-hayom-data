הלכות עבדים - פרק שלישי

כָּל עֶבֶד עִבְרִי הֲרֵי הָאָדוֹן חַיָּב בִּמְזוֹנוֹת אִשְׁתּוֹ הַנְּשׂוּאָה לוֹ וְלֹא אֲרוּסָה וְשׁוֹמֶרֶת יָבָם. וְהוּא שֶׁתִּהְיֶה מֻתֶּרֶת לוֹ. אֲבָל אִם הָיְתָה מֵאִסּוּרֵי לָאוִין אֲפִלּוּ שְׁנִיָּה אֵינוֹ חַיָּב בִּמְזוֹנוֹתֶיהָ שֶׁנֶּאֱמַר אִשְׁתּוֹ עִמּוֹ אִשָּׁה הָרְאוּיָה לַעֲמֹד עִמּוֹ. וְכֵן חַיָּב בִּמְזוֹנוֹת בָּנָיו וּבְנוֹתָיו. בִּמְכָרוּהוּ בֵּית דִּין נֶאֱמַר . וְכִי תַּעֲלֶה עַל דַּעְתְּךָ שֶׁכֵּיוָן שֶׁנִּקְנָה זֶה נִשְׁתַּעְבְּדָה אִשְׁתּוֹ. אֲבָל לֹא בָּא לְלַמֵּד אֶלָּא שֶׁהָאָדוֹן חַיָּב בִּמְזוֹנוֹתֶיהָ. וּבְמוֹכֵר עַצְמוֹ נֶאֱמַר . וּבְנִמְכָּר לְעַכּוּ''ם נֶאֱמַר . אֶחָד אִשָּׁה וּבָנִים שֶׁהָיוּ לוֹ בִּשְׁעַת מְכִירָה אוֹ אִשָּׁה וּבָנִים שֶׁהָיוּ לוֹ אַחַר מְכִירָתוֹ. וְהוּא שֶׁיִּקָּחֶנָּה מִדַּעַת הָאָדוֹן. אֲבָל אִם לְקָחָהּ שֶׁלֹּא מִדַּעַת רַבּוֹ אֵינוֹ חַיָּב בִּמְזוֹנוֹתֶיהָ: (-)

אַף עַל פִּי שֶׁהָאָדוֹן חַיָּב בִּמְזוֹנוֹת אִשְׁתּוֹ וּבָנָיו אֵין לוֹ בְּמַעֲשֵׂה יְדֵיהֶם כְּלוּם אֶלָּא הֲרֵי מַעֲשֵׂה הָאִשָּׁה וּמְצִיאָתָהּ לְבַעֲלָהּ. וְכָל שֶׁזּוֹכֶה הַבַּעַל בְּאִשְׁתּוֹ זוֹכֶה זֶה אַף עַל פִּי שֶׁהוּא עֶבֶד עִבְרִי:

מִי שֶׁמְּכָרוּהוּ בֵּית דִּין יֵשׁ לְרַבּוֹ לִתֵּן לוֹ שִׁפְחָה כְּנַעֲנִית. בֵּין הָאָדוֹן בֵּין בְּנוֹ שֶׁל אֲדוֹן אִם מֵת אָבִיו הֲרֵי זֶה נוֹתֵן לוֹ שִׁפְחָה. וְכוֹפֵהוּ עַל זֶה כְּדֵי שֶׁיּוֹלִיד מִמֶּנָּה עֲבָדִים. וַהֲרֵי הִיא מֻתֶּרֶת לוֹ כָּל יְמֵי עַבְדוּתוֹ שֶׁנֶּאֱמַר וְגוֹ'. וְהַמּוֹכֵר עַצְמוֹ אָסוּר בְּשִׁפְחָה כְּנַעֲנִית כִּשְׁאָר כָּל יִשְׂרָאֵל:

אֵין עֶבֶד עִבְרִי מֻתָּר בְּשִׁפְחָה כְּנַעֲנִית עַד שֶׁתִּהְיֶה לוֹ אִשָּׁה יִשְׂרְאֵלִית וּבָנִים. אֲבָל אִם אֵין לוֹ אִשָּׁה וּבָנִים אֵין רַבּוֹ מוֹסֵר לוֹ שִׁפְחָה כְּנַעֲנִית. וְדָבָר זֶה קַבָּלָה הוּא אֲפִלּוּ הָיָה הַנִּמְכָּר כֹּהֵן הֲרֵי זֶה מֻתָּר בְּשִׁפְחָה כְּנַעֲנִית כָּל יְמֵי עַבְדוּתוֹ:

אִם הָיוּ לוֹ אִשָּׁה וּבָנִים אַף עַל פִּי שֶׁמּוֹסֵר לוֹ רַבּוֹ שִׁפְחָה כְּנַעֲנִית אֵינוֹ יָכוֹל לְהַפְרִישׁוֹ מֵאִשְׁתּוֹ וּבָנָיו שֶׁנֶּאֱמַר . וְאֵינוֹ יָכוֹל לִתֵּן לוֹ שְׁתֵּי שְׁפָחוֹת. וְלֹא לִתֵּן שִׁפְחָה אַחַת לִשְׁנֵי עֲבָדָיו הָעִבְרִים כְּדֶרֶךְ שֶׁנּוֹתֵן לַכְּנַעֲנִיִּים שֶׁנֶּאֱמַר . נִרְצָע שֶׁמָּסַר לוֹ רַבּוֹ שִׁפְחָה כְּנַעֲנִית וְהִגִּיעַ יוֹבֵל וְהָיָה רַבּוֹ מְסָרֵב בּוֹ לָצֵאת וְאֵינוֹ רוֹצֶה לָצֵאת וְחָבַל בּוֹ פָּטוּר שֶׁהֲרֵי נֶאֱסַר בְּשִׁפְחָה:

הַמּוֹכֵר ] עַצְמוֹ אֵינוֹ נִרְצָע. אֲבָל מְכָרוּהוּ בֵּית דִּין וְעָבַד שֵׁשׁ וְלֹא רָצָה לָצֵאת הֲרֵי זֶה נִרְצָע וְעוֹבֵד עַד שְׁנַת הַיּוֹבֵל אוֹ עַד שֶׁיָּמוּת הָאָדוֹן:

אַף עַל פִּי שֶׁהִנִּיחַ בֵּן אֵין הַנִּרְצָע עוֹבֵד אֶת הַבֵּן מִפִּי הַשְּׁמוּעָה לָמְדוּ , לוֹ וְלֹא לִבְנוֹ. , לְעוֹלָמוֹ שֶׁל יוֹבֵל. נִמְצֵאתָ לָמֵד שֶׁהַנִּרְצָע אֵינוֹ קוֹנֶה אֶת עַצְמוֹ אֶלָּא בַּיּוֹבֵל וּבְמִיתַת הָאָדוֹן:

עֶבֶד עִבְרִי כֹּהֵן אֵינוֹ נִרְצָע מִפְּנֵי שֶׁנַּעֲשָׂה בַּעַל מוּם וַהֲרֵי הוּא אוֹמֵר לַחֲזָקָה שֶׁהָיָה בָּהּ. וְאֵינוֹ שָׁב לַשְּׂרָרָה שֶׁהָיָה בָּהּ:

כֵּיצַד רוֹצְעִין. מְבִיאוֹ לְבֵית דִּין שֶׁל שְׁלֹשָׁה וְאוֹמֵר דְּבָרָיו לִפְנֵיהֶם וּמַגִּישׁוֹ לְסוֹף שֵׁשׁ אֶל הַדֶּלֶת אוֹ אֶל הַמְּזוּזָה כְּשֶׁהֵן עוֹמְדִין בְּבִנְיָן. בֵּין דֶּלֶת וּמְזוּזָה שֶׁל אָדוֹן בֵּין שֶׁל כָּל אָדָם. וְנוֹקֵב אֶת אָזְנוֹ הַיְמָנִית בְּגוּפָהּ שֶׁל אֹזֶן בְּמַרְצֵעַ שֶׁל מַתֶּכֶת עַד שֶׁיַּגִּיעַ לַדֶּלֶת שֶׁנֶּאֱמַר . וְלֹא נֶאֱמַר אֶלָּא שֶׁיִּהְיֶה עוֹמֵד בֵּין אֵצֶל דֶּלֶת בֵּין אֵצֶל מְזוּזָה. וּמָה מְזוּזָה כְּשֶׁהִיא עוֹמֶדֶת אַף דֶּלֶת כְּשֶׁהִיא עוֹמֶדֶת. אֲבָל הָרְצִיעָה בַּדֶּלֶת אַף עַל פִּי שֶׁאֵין שָׁם מְזוּזָה. הָאָדוֹן ] הוּא שֶׁרוֹצֵעַ בְּעַצְמוֹ שֶׁנֶּאֱמַר לֹא בְּנוֹ וְלֹא שְׁלוּחוֹ וְלֹא שְׁלִיחַ בֵּית דִּין. וְאֵין רוֹצְעִין שְׁנֵי עֲבָדִים כְּאֶחָד שֶׁאֵין עוֹשִׂין מִצְוֹת חֲבִילוֹת חֲבִילוֹת: (-)

עַד שֶׁיֹּאמַר וְיִשְׁנֶה. הָעֶבֶד עַד שֶׁיֹּאמַר כְּשֶׁהוּא עֶבֶד אֲבָל אִם אָמַר אַחַר שֵׁשׁ אֵינוֹ נִרְצָע עַד שֶׁיֹּאמַר וְיִשְׁנֶה בְּסוֹף שֵׁשׁ בִּתְחִלַּת פְּרוּטָה אַחֲרוֹנָה. כֵּיצַד. כְּגוֹן שֶׁנִּשְׁאַר מִן הַיּוֹם שְׁוֵה פְּרוּטָה מִדְּמֵי מְכִירָתוֹ אוֹ יֶתֶר מְעַט. אֲבָל אִם נִשְׁאַר פָּחוֹת מִשְּׁוֵה פְּרוּטָה הֲרֵי זֶה כְּאוֹמֵר אַחַר שֵׁשׁ: (-)

הָיְתָה לָעֶבֶד שִׁפְחָה כְּנַעֲנִית וְלוֹ מִמֶּנָּה בָּנִים וּלְרַבּוֹ אֵין אִשָּׁה וּבָנִים אֵינוֹ נִרְצָע שֶׁנֶּאֱמַר . לְרַבּוֹ אִשָּׁה וּבָנִים וְלוֹ אֵין אִשָּׁה וּבָנִים אֵינוֹ נִרְצָע שֶׁנֶּאֱמַר . הוּא אוֹהֵב אֶת רַבּוֹ וְרַבּוֹ אֵינוֹ אוֹהֲבוֹ אֵינוֹ נִרְצָע שֶׁנֶּאֱמַר כִּי טוֹב לוֹ עִמָּךְ. רַבּוֹ אוֹהֲבוֹ וְהוּא אֵינוֹ אוֹהֵב אֶת רַבּוֹ אֵינוֹ נִרְצָע שֶׁנֶּאֱמַר כִּי אֲהֵבְךָ. הוּא חוֹלֶה וְרַבּוֹ אֵינוֹ חוֹלֶה אֵינוֹ נִרְצָע שֶׁנֶּאֱמַר כִּי טוֹב לוֹ עִמָּךְ. רַבּוֹ חוֹלֶה וְהוּא אֵינוֹ חוֹלֶה אוֹ שֶׁהָיוּ שְׁנֵיהֶם חוֹלִים אֵינוֹ נִרְצָע שֶׁנֶּאֱמַר עַד שֶׁיִּהְיוּ שְׁנֵיהֶם בְּטוֹבָה: (-)

מַה ] בֵּין מוֹכֵר עַצְמוֹ לִמְכָרוּהוּ בֵּית דִּין. מוֹכֵר עַצְמוֹ אֵינוֹ נִרְצָע וּמְכָרוּהוּ בֵּית דִּין נִרְצָע. מוֹכֵר עַצְמוֹ אָסוּר בְּשִׁפְחָה כְּנַעֲנִית וּמְכָרוּהוּ בֵּית דִּין רַבּוֹ מוֹסֵר לוֹ שִׁפְחָה כְּנַעֲנִית. מוֹכֵר עַצְמוֹ נִמְכָּר לְעַכּוּ''ם וּמְכָרוּהוּ בֵּית דִּין אֵינוֹ נִמְכָּר אֶלָּא לְיִשְׂרָאֵל שֶׁנֶּאֱמַר אֵין בֵּית דִּין מוֹכְרִין אוֹתוֹ אֶלָּא לְךָ. מוֹכֵר עַצְמוֹ נִמְכָּר לְשֵׁשׁ וְיֶתֶר עַל שֵׁשׁ מְכָרוּהוּ בֵּית דִּין אֵינוֹ נִמְכָּר אֶלָּא לְשֵׁשׁ. מוֹכֵר עַצְמוֹ אֵין מַעֲנִיקִין לוֹ מְכָרוּהוּ בֵּית דִּין מַעֲנִיקִין לוֹ:

מִפִּי ] הַשְּׁמוּעָה לָמְדוּ שֶׁאֵין הָאִשָּׁה נִרְצַעַת וְכֵן יֵרָאֶה מֵעִנְיַן הַכָּתוּב שֶׁהֲרֵי אוֹמֵר בְּנִרְצָע . וּמַהוּ שֶׁנֶּאֱמַר . לְהַעֲנִיק. שֶׁכְּשֵׁם שֶׁמִּצְוַת עֲשֵׂה לְהַעֲנִיק עֶבֶד עִבְרִי כָּךְ מַעֲנִיק אָמָה הָעִבְרִיָּה:

כָּל הַמְשַׁלֵּחַ עַבְדּוֹ וַאֲמָתוֹ רֵיקָם עוֹבֵר בְּלֹא תַּעֲשֶׂה שֶׁנֶּאֱמַר וַהֲרֵי הַכָּתוּב נִתְּקוֹ לַעֲשֵׂה שֶׁנֶּאֱמַר ] . אֶחָד הַיּוֹצֵא בְּסוֹף שֵׁשׁ אוֹ שֶׁיָּצָא בְּיוֹבֵל אוֹ בְּמִיתַת הָאָדוֹן וְכֵן הָאָמָה הָעִבְרִיָּה שֶׁיָּצָאת בְּאֶחָד מִכָּל אֵלּוּ אוֹ בְּסִימָנִין הֲרֵי אֵלּוּ מַעֲנִיקִין לָהֶם. אֲבָל הַיּוֹצֵא בְּגִרְעוֹן כֶּסֶף אֵין מַעֲנִיקִין לָהֶם שֶׁנֶּאֱמַר וְזֶה לֹא שִׁלְּחוֹ חָפְשִׁי אֶלָּא הָעֶבֶד נָתַן שְׁאָר הַדָּמִים שֶׁנִּשְׁאַר לוֹ לַעֲבֹד בָּהֶן וְאַחַר כָּךְ יָצָא. בְּעִנְיַן צֹאן גֹּרֶן וְיֶקֶב דְּבָרִים שֶׁיֵּשׁ בָּהֶם בְּרָכָה מֵחֲמַת עַצְמָן הוּא שֶׁחַיָּב לְהַעֲנִיק לוֹ מֵהֶם אֲבָל הַכְּסָפִים וְהַבְּגָדִים אֵינוֹ חַיָּב לִתֵּן לוֹ מֵהֶם. וְכַמָּה נוֹתֵן לוֹ אֵין פָּחוֹת מִשְּׁוֵה שְׁלֹשִׁים סֶלַע בֵּין מִמִּין אֶחָד בֵּין מִמִּינִין הַרְבֵּה כִּשְׁלֹשִׁים שֶׁל קְנַס הָעֶבֶד שֶׁנֶּאֱמַר בּוֹ . בֵּין שֶׁנִּתְבָּרֵךְ הַבַּיִת בִּגְלָלוֹ בֵּין שֶׁלֹּא נִתְבָּרֵךְ שֶׁנֶּאֱמַר הַעֲנֵיק תַּעֲנִיק לוֹ מִכָּל מָקוֹם. אִם כֵּן לָמָּה נֶאֱמַר הַכּל לְפִי הַבְּרָכָה תֵּן לוֹ: (-)

אִם בָּרַח וּפָגַע בּוֹ יוֹבֵל כְּשֶׁהוּא בּוֹרֵחַ וְיָצָא לְחֵרוּת אֵינוֹ חַיָּב לְהַעֲנִיקוֹ שֶׁנֶּאֱמַר וְגוֹ'. עֶנֶק עֶבֶד עִבְרִי לְעַצְמוֹ וְאֵין בַּעַל חוֹב גּוֹבֶה הֵימֶנּוּ. וְעֶנֶק אָמָה הָעִבְרִיָּה וְכֵן מְצִיאָתָהּ לְאָבִיהָ. וְאִם מֵת אָבִיהָ קֹדֶם שֶׁיָּבוֹא לְיָדוֹ הֲרֵי הֵן שֶׁל עַצְמָהּ וְאֵין לְאַחֶיהָ בָּהֶם כְּלוּם שֶׁאֵין אָדָם מוֹרִישׁ זְכוּת שֶׁיֵּשׁ לוֹ בְּבִתּוֹ לִבְנוֹ:
