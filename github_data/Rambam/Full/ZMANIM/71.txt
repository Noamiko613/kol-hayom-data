וְאֵין מְעִידִין. אָב וּבְנוֹ שֶׁרָאוּ אֶת הַיָּרֵחַ יֵלְכוּ לְבֵית דִּין לְהָעִיד. לֹא מִפְּנֵי שֶׁעֵדוּת הַחֹדֶשׁ כְּשֵׁרָה בִּקְרוֹבִים אֶלָּא שֶׁאִם יִמָּצֵא אֶחָד מֵהֶן פָּסוּל מִפְּנֵי שֶׁהוּא גַּזְלָן וְכַיּוֹצֵא בּוֹ מִשְּׁאָר הַפַּסְלָנוּת יִצְטָרֵף הַשֵּׁנִי עִם אַחֵר וְיָעִידוּ. וְכָל הַפָּסוּל לְעֵדוּת מִדִּבְרֵי סוֹפְרִים אַף עַל פִּי שֶׁהוּא כָּשֵׁר מִן הַתּוֹרָה פָּסוּל לְעֵדוּת הַחֹדֶשׁ: דִּין תּוֹרָה שֶׁאֵין מְדַקְדְּקִין בְּעֵדוּת הַחֹדֶשׁ. שֶׁאֲפִלּוּ קִדְּשׁוּ אֶת הַחֹדֶשׁ עַל פִּי עֵדִים וְנִמְצְאוּ זוֹמְמִין בְּעֵדוּת זוֹ הֲרֵי זֶה מְקֻדָּשׁ. לְפִיכָךְ הָיוּ
