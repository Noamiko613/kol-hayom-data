הלכות נזקי ממון - פרק אחד עשר

כַּמָּה הוּא הַכֹּפֶר. כְּמוֹ שְׁרָאוּ הַדַּיָּנִין שֶׁהוּא דְּמֵי הַנֶּהֱרָג הַכּל לְפִי שָׁוְיוֹ שֶׁל נֶהֱרָג שֶׁנֶּאֱמַר . וְכֹפֶר הָעֲבָדִים ] בֵּין גְּדוֹלִים בֵּין קְטַנִּים בֵּין זְכָרִים בֵּין נְקֵבוֹת הוּא הַקְּנָס הַקָּצוּב בַּתּוֹרָה שְׁלֹשִׁים סֶלַע כֶּסֶף יָפֶה בֵּין שֶׁהָיָה הָעֶבֶד שָׁוֶה מֵאָה מָנֶה בֵּין שֶׁאֵינוֹ שָׁוֶה אֶלָּא דִּינָר. וְכָל הַמְעֻכָּב גֵּט שִׁחְרוּר אֵין לוֹ קְנָס הוֹאִיל וְאֵין לוֹ אָדוֹן שֶׁהֲרֵי יָצָא לְחֵרוּת:

לְמִי נוֹתְנִין הַכֹּפֶר. לְיוֹרְשֵׁי הַנֶּהֱרָג. וְאִם הֵמִית אִשָּׁה הַכֹּפֶר לְיוֹרְשֶׁיהָ מֵאָבִיהָ וְאֵינוֹ לַבַּעַל. הֵמִית מִי שֶׁחֶצְיוֹ עֶבֶד וְחֶצְיוֹ בֶּן חוֹרִין נוֹתֵן חֲצִי קְנָס לְרַבּוֹ וְהַחֵצִי הָאַחֵר רָאוּי לִתֵּן וְאֵין מִי יִקָּחֶנּוּ:

שׁוֹר שֶׁנָּגַח אִשָּׁה וְיָצְאוּ יְלָדֶיהָ אַף עַל פִּי שֶׁהוּא מוּעָד לִגַּח הַבְּעָלִים פְּטוּרִין מִדְּמֵי וְלָדוֹת. שֶׁלֹּא חִיְּבָה תּוֹרָה בִּדְמֵי וְלָדוֹת אֶלָּא לָאָדָם:

] נָגַח שִׁפְחָה וְיָצְאוּ יְלָדֶיהָ מְשַׁלֵּם דְּמֵי וְלָדוֹת. שֶׁזֶּה כְּמִי שֶׁנָּגַח חֲמוֹר מְעֻבֶּרֶת. וְאִם הָיָה תָּם מְשַׁלֵּם חֲצִי דְּמֵי וְלָדוֹת מִגּוּפוֹ:

כֵּיצַד שָׁמִין אוֹתָהּ. אוֹמְדִין כַּמָּה הָיְתָה שִׁפְחָה זוֹ שָׁוָה כְּשֶׁהָיְתָה מְעֵבֶּרֶת וְכַמָּה הִיא שָׁוָה עַתָּה וְנוֹתֵן לִבְעָלֶיהָ הַפְּחָת אוֹ חֶצְיוֹ. וְאִם הֵמִית הַשִּׁפְחָה מְשַׁלֵּם הַכֹּפֶר הַקָּצוּב בַּתּוֹרָה בִּלְבַד כְּמוֹ שֶׁבֵּאַרְנוּ:

שׁוֹר שֶׁנִּתְכַּוֵּן לִבְהֵמָה וְנָגַח אֶת הָאָדָם אַף עַל פִּי שֶׁאִם הֱמִיתוֹ פָּטוּר כְּמוֹ שֶׁבֵּאַרְנוּ אִם חָבַל בּוֹ חַיָּב בְּנֵזֶק. וְאִם תָּם הוּא מְשַׁלֵּם חֲצִי נֵזֶק מִגּוּפוֹ. וְאִם מוּעָד נֵזֶק שָׁלֵם:

שׁוֹר תָּם שֶׁהֵמִית וְהִזִּיק דָּנִין אוֹתוֹ דִּינֵי נְפָשׁוֹת וְאֵין דָּנִין אוֹתוֹ דִּינֵי מָמוֹנוֹת. וּמוּעָד שֶׁהֵמִית וְהִזִּיק דָּנִין אוֹתוֹ דִּינֵי מָמוֹנוֹת וְחוֹזְרִין וְדָנִין אוֹתוֹ דִּינֵי נְפָשׁוֹת. קָדְמוּ וְדָנוּהוּ דִּינֵי נְפָשׁוֹת תְּחִלָּה חוֹזְרִין וְדָנִין אוֹתוֹ דִּינֵי מָמוֹנוֹת:

* וּמֵהֵיכָן מִשְׁתַּלֵּם מִן הַשֶּׁבַח שֶׁהִשְׁבִּיחַ בִּרְדִיָּתוֹ אַחַר שֶׁנִּגְמַר דִּינוֹ. שֶׁכֵּיוָן שֶׁנִּגְמַר דִּינוֹ לִסְקִילָה אֵין לוֹ בְּעָלִים שֶׁיִּתְחַיְּבוּ בִּנְזָקָיו. דָּנוּהוּ דִּינֵי נְפָשׁוֹת וּבָרַח אֵין דָּנִין אוֹתוֹ דִּינֵי מָמוֹנוֹת:

שׁוֹר שֶׁהֵמִית אֶת הָאָדָם וְהִקְדִּישׁוֹ בְּעָלָיו אֵינוֹ קָדוֹשׁ. וְכֵן אִם הִפְקִירוֹ אֵינוֹ מֻפְקָר. מְכָרוֹ אֵינוֹ מָכוּר. הֶחֱזִירוֹ שׁוֹמֵר לִבְעָלָיו אֵינָהּ חֲזָרָה. שְׁחָטוֹ בְּשָׂרוֹ אָסוּר בַּהֲנָאָה. בַּמֶּה דְּבָרִים אֲמוּרִים אַחַר שֶׁנִּגְמַר דִּינוֹ לִסְקִילָה. אֲבָל עַד שֶׁלֹּא נִגְמָר דִּינוֹ לִסְקִילָה אִם הִקְדִּישׁוֹ מֻקְדָּשׁ וְאִם הִפְקִירוֹ הֲרֵי הוּא מֻפְקָר. מְכָרוֹ הֲרֵי זֶה מָכוּר. הֶחֱזִירוֹ שׁוֹמֵר לִבְעָלָיו הֲרֵי זֶה מֻחְזָר. וְאִם קָדַם וּשְׁחָטוֹ הֲרֵי זֶה מֻתָּר בַּאֲכִילָה:

שׁוֹר שֶׁלֹּא נִגְמַר דִּינוֹ שֶׁנִּתְעָרֵב בִּשְׁוָרִים אֲחֵרִים כֻּלָּן פְּטוּרִין לְפִי שֶׁאֵין גּוֹמְרִין דִּינוֹ שֶׁל שׁוֹר אֶלָּא בִּפְנֵי הַשּׁוֹר כְּדִין הָאָדָם. נִגְמַר דִּינוֹ וְאַחַר כָּךְ נִתְעָרֵב בַּאֲחֵרִים אֲפִלּוּ בְּאֶלֶף כֻּלָּן נִסְקָלִים וְנִקְבָּרִין וַאֲסוּרִין בַּהֲנָאָה כְּדִין בְּהֵמָה נִסְקֶלֶת:

פָּרָה מְעֻבֶּרֶת שֶׁהֵמִיתָה אֶת הָאָדָם וְכֵן כָּל בְּהֵמָה שֶׁנֶּעֶבְדָה בָּהּ עֲבֵרָה הֲרֵי עֵבָּרָהּ כָּמוֹהָ. הִיא וְעֵבָּרָהּ נָגְחָה הִיא וְעֻבָּרָהּ נִרְבְּעָה:

נָגְחָה וְהֵמִיתָה וְאַחַר כָּךְ נִתְעַבְּרָה. אִם עַד שֶׁלֹּא נִגְמַר דִּינָהּ נִתְעַבְּרָה וְיָלְדָה קֹדֶם גְּמַר דִּין וְלָדָהּ מֻתָּר וְאִם יָלְדָה אַחַר גְּמַר דִּין וְלָדָהּ אָסוּר שֶׁהָעֻבָּר יֶרֶךְ אִמּוֹ הוּא. וְאִם נִתְעָרֵב וָלָד זֶה בַּאֲחֵרִים כּוֹנְסִין אֶת כֻּלָּם בְּכִפָּה עַד שֶׁיָּמוּתוּ שָׁם:

שׁוֹר הַנִּסְקָל שֶׁהוּזַמּוּ עֵדָיו כָּל הַקּוֹדֵם בּוֹ זָכָה שֶׁהֲרֵי מִשֶּׁנִּגְמַר דִּינוֹ הִפְקִירוּהוּ בְּעָלָיו. וְאִם הֵעִידוּ שֶׁבְּעָלָיו רִבְּעָהוּ וְהֵזִים אֶת הָעֵדִים הֲרֵי הַשּׁוֹר לִבְעָלָיו וְהַקּוֹדֵם וּמְשָׁכוֹ לֹא זָכָה בּוֹ שֶׁכֵּיוָן שֶׁהַבַּעַל יוֹדֵעַ בְּעַצְמוֹ שֶׁלֹּא חָטָא וְשֶׁאֵלּוּ עֵדֵי שֶׁקֶר הֵם וַהֲרֵי הוּא מְיַחֵל לַהֲזִימָם לֹא הִפְקִיר:
