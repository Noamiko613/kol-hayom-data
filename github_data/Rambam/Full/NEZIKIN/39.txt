הלכות גזלה ואבדה - פרק ששה עשר

הַמּוֹצֵא מְחָטִים וְצִנּוֹרוֹת וּמַסְמְרִים וְכַיּוֹצא בָּהֶן אִם מְצָאָם אֶחָד אֶחָד הֲרֵי אֵלּוּ שֶׁלּוֹ. שְׁנַיִם שְׁנַיִם אוֹ יֶתֶר חַיָּב לְהַכְרִיז שֶׁמִּנְיָן סִימָן:

וְכֵן הַמּוֹצֵא מָעוֹת מְפֻזָּרִים הֲרֵי אֵלּוּ שֶׁלּוֹ. אֲפִלּוּ הָיוּ מִקְצָת מַטְבְּעוֹת זֶה עַל גַּבֵּי זֶה הֲרֵי הֵן כִּמְפֻזָּרִין. אֲבָל אִם מָצָא צִבּוּר מָעוֹת חַיָּב לְהַכְרִיז. מָצָא שְׁלֹשָׁה מַטְבְּעוֹת זֶה עַל גַּב זֶה וְהֵן עֲשׂוּיִין כְּמִגְדָּל. אוֹ שֶׁהָיוּ אֶחָד מִיכַּן וְאֶחָד מִיכַּן וְאֶחָד עַל גַּבֵּיהֶן. אוֹ שֶׁהָיוּ מִקְצָת זֶה עַל מִקְצָת זֶה כְּדֵי שֶׁאִם יַכְנִיס קֵיסָם בֵּינֵיהֶן יִנָּטְלוּ בְּבַת אַחַת חַיָּב לְהַכְרִיז. הָיוּ עֲשׂוּיִים כְּשִׁיר אוֹ כְּשׁוּרָה אוֹ כַּחֲצוּבָה אוֹ כְּסֻלָּם הֲרֵי זֶה סָפֵק וְלֹא יִטּל:

] הַמּוֹצֵא מָעוֹת בַּכִּיס אוֹ כִּיס כְּמוֹת שֶׁהוּא חַיָּב לְהַכְרִיז. מָצָא כִּיס וּלְפָנָיו מָעוֹת מְפֻזָּרִים הֲרֵי אֵלּוּ שֶׁלּוֹ. וְאִם מַרְאִין הַדְּבָרִים שֶׁהַכִּיס וְהַמָּעוֹת שֶׁל אָדָם אֶחָד וּמִן הַכִּיס נָפְלוּ חַיָּב לְהַכְרִיז:

* הַמּוֹצֵא מָעוֹת בַּחֲנוּת אִם הָיוּ בֵּין תֵּבָה לַחֶנְוָנִי הֲרֵי הֵן שֶׁל בַּעַל הַחֲנוּת. וְאִם מְצָאָן עַל הַתֵּבָה וְאֵין צָרִיךְ לוֹמַר מִתֵּבָה וְלַחוּץ הֲרֵי הֵן שֶׁל מוֹצְאָן. וְלָמָּה לֹא תִּקְנֶה הַחֲנוּת לִבְעָלֶיהָ לְפִי שֶׁאֵינָהּ חָצֵר הַמִּשְׁתַּמֶּרֶת. וְאַף עַל פִּי שֶׁבְּעָלֶיהָ בְּתוֹכָהּ צָרִיךְ לוֹמַר תִּקְנֶה לִי חֲנוּתִי כְּמוֹ שֶׁיִּתְבָּאֵר:

] מָצָא מָעוֹת בַּחֲנוּת הַשֻּׁלְחָנִי בֵּין כִּסֵּא o לַשֻּׁלְחָנִי הֲרֵי אֵלּוּ שֶׁל שֻׁלְחָנִי. מְצָאָן עַל הַכִּסֵּא לִפְנֵי הַשֻּׁלְחָנִי אֲפִלּוּ הָיוּ צְרוּרִין וּמֻנָּחִין עַל הַשֻּׁלְחָן הֲרֵי אֵלּוּ שֶׁל מוֹצְאָן. וְהוּא שֶׁיִּהְיוּ רֹב עַכּוּ''ם כְּמוֹ שֶׁבֵּאַרְנוּ. אֲבָל אִם הָיוּ רֹב יִשְׂרָאֵל חַיָּב לְהַכְרִיז. מִפְּנֵי שֶׁהֵן צְרוּרִים יֵשׁ לָהֶם סִימָן:

הַלּוֹקֵחַ פֵּרוֹת מֵחֲבֵרוֹ אוֹ שֶׁשָּׁלַח לוֹ חֲבֵרוֹ פֵּרוֹת וּמָצָא בְּתוֹכָן מָעוֹת צְרוּרוֹת נוֹטֵל וּמַכְרִיז. מְצָאָן מְפֻזָּרוֹת הֲרֵי אֵלּוּ שֶׁלּוֹ. בַּמֶּה דְּבָרִים אֲמוּרִים בְּשֶׁהָיוּ הַפֵּרוֹת מִן הַתַּגָּר אוֹ מִבַּעַל הַבַּיִת שֶׁלְּקָחָן מִן הַתַּגָּר. אֲבָל אִם בַּעַל הַבַּיִת דָּשׁ הַפֵּרוֹת לְעַצְמוֹ אוֹ עַל יְדֵי עֲבָדָיו וְשִׁפְחוֹתָיו הַכְּנַעֲנִים הֲרֵי זֶה חַיָּב לְהַחְזִיר:

] הַמּוֹצֵא מַטְמוֹן בְּגַל אוֹ בְּכֹתֶל יָשָׁן הֲרֵי אֵלּוּ שֶׁלּוֹ שֶׁאֲנִי אוֹמֵר שֶׁל אֱמוֹרִיִּים הַקַּדְמוֹנִים הֵן. וְהוּא שֶׁיִּמְצָאֵם מַטָּה מַטָּה כְּדֶרֶךְ כָּל הַמַּטְמוֹנוֹת הַיְשָׁנוֹת. אֲבָל אִם מַרְאִין הַדְּבָרִים שֶׁהֵן מַטְמוֹן חָדָשׁ אֲפִלּוּ נִסְתַּפֵּק לוֹ הַדָּבָר הֲרֵי זֶה לֹא יִגַּע בָּהֶן שֶׁמָּא מֻנָּחִים הֵם o שָׁם:

וְהוֹאִיל וַחֲצֵרוֹ שֶׁל אָדָם קוֹנָה לוֹ שֶׁלֹּא מִדַּעְתּוֹ כְּמוֹ שֶׁיִּתְבָּאֵר לָמָּה לֹא יִקְנֶה בַּעַל הֶחָצֵר זֶה הַמַּטְמוֹן שֶׁבְּתוֹךְ הַכֹּתֶל הַיָּשָׁן אַף עַל פִּי שֶׁהוּא שֶׁל אֱמוֹרִיִּים וְתִהְיֶה מְצִיאָה זוֹ לְבַעַל הֶחָצֵר. מִפְּנֵי שֶׁאֵינוֹ יָדוּעַ לוֹ וְלֹא לַאֲחֵרִים וַהֲרֵי זֶה הַמַּטְמוֹן אָבוּד מִמֶּנּוּ וּמִכָּל אָדָם וּלְפִיכָךְ הוּא שֶׁל מוֹצְאוֹ. וּמַה אֲבֵדָה שֶׁל אָדָם אָמְרָה תּוֹרָה מִי שֶׁאֲבוּדָה מִמֶּנּוּ וּמְצוּיָה אֵצֶל כָּל אָדָם יָצָאת זוֹ שֶׁנָּפְלָה לַיָּם שֶׁאֲבוּדָה מִמֶּנּוּ וּמִכָּל אָדָם קַל וָחֹמֶר לְמַטְמוֹן קַדְמוֹנִי שֶׁלֹּא הָיָה שֶׁלּוֹ מֵעוֹלָם וְהוּא אָבוּד מִמֶּנּוּ וּמִכָּל אָדָם לְפִיכָךְ * הוּא שֶׁל מוֹצְאוֹ:

מָצָא מַטְמוֹן בְּכֹתֶל חָדָשׁ ] אִם הַמַּטְמוֹן מוֹכִיחַ שֶׁהוּא לְבַעַל הַבַּיִת הֲרֵי הוּא שֶׁלּוֹ וְאִם מוֹכִיחַ שֶׁהוּא שֶׁל אַחֵר מִן הַשּׁוּק הֲרֵי הוּא שֶׁל מוֹצְאוֹ. כֵּיצַד. הַסַּכִּין הֲרֵי הַנִּצָּב שֶׁלּוֹ מוֹכִיחַ. וְהַכִּיס פִּיו מוֹכִיחַ. וְאִם נִמְצָא תּוֹךְ הַכֹּתֶל מָלֵא מֵהֶן חוֹלְקִין:

הָיוּ בְּתוֹךְ הַכֹּתֶל מָעוֹת אוֹ לְשׁוֹנוֹת שֶׁל זָהָב שֶׁאֵין שָׁם מוֹכִיחַ. מֵחֶצְיוֹ וְלַחוּץ שֶׁל מוֹצֵא. מֵחֶצְיוֹ וּלְפָנִים שֶׁל בַּעַל הַבַּיִת:

וְיֵרָאֶה לִי שֶׁאֵין הַדְּבָרִים אֲמוּרִים ] אֶלָּא כְּשֶׁטָּעַן בַּעַל הַבַּיִת שֶׁהַמַּטְמוֹן שֶׁלּוֹ. אוֹ שֶׁהָיָה יוֹרֵשׁ שֶׁאָנוּ טוֹעֲנִין לוֹ שֶׁמָּא שֶׁל אָבִיו הֵן. אֲבָל אִם הוֹדָה שֶׁהֵן מְצִיאָה הֲרֵי הֵן שֶׁל מוֹצֵא. לְפִיכָךְ אִם הָיָה * מַשְׂכִּיר בֵּיתוֹ לַאֲחֵרִים הֲרֵי הֵן שֶׁל שׂוֹכֵר אַחֲרוֹן. וְאִם הִשְׂכִּירוֹ לִשְׁלֹשָׁה עַכּוּ''ם כְּאֶחָד הֲרֵי עֲשָׂאָהוּ פֻּנְדָּק וְכָל הַנִּמְצָא בּוֹ אֲפִלּוּ בְּתוֹךְ הַבַּיִת הֲרֵי הוּא שֶׁל מוֹצְאָן מִפְּנֵי שֶׁאֵין אֶחָד יָכוֹל לִטְעֹן שֶׁהֵן שֶׁלּוֹ אוֹ שֶׁהוּא טָמַן שֶׁהֲרֵי עֲשָׂאָהוּ פֻּנְדָּק:
