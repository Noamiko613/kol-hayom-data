הלכות שמיטה ויובל - פרק אחד עשר אֶרֶץ יִשְׂרָאֵל הַמִּתְחַלֶּקֶת לִשְׁבָטִים אֵינָהּ נמְכֶּרֶת לִצְמִיתוּת שֶׁנֶּאֱמַר . וְאִם מָכַר לִצְמִיתוּת שְׁנֵיהֶם עוֹבְרִין בְּלֹא תַּעֲשֶׂה. וְאֵין מַעֲשֵׂיהֶן מוֹעִילִין אֶלָּא תַּחֲזֹר הַשָּׂדֶה לִבְעָלֶיהָ בַּיּוֹבֵל: וְהַמּוֹכֵר שָׂדֵהוּ לְשִׁשִּׁים שָׁנָה אֵינָהּ יוֹצְאָה בַּיּוֹבֵל שֶׁאֵין חוֹזֵר בַּיּוֹבֵל אֶלָּא דָּבָר הַנִמְכָּר סְתָם אוֹ הַנִמְכָּר לִצְמִיתוּת: לֹא יִמְכֹּר אָדָם בֵּיתוֹ אוֹ שְׂדֵה אֲחֻזָּתוֹ אַף עַל פִּי שֶׁהֵם חוֹזְרִין אַחַר זְמַן אֶלָּא אִם כֵּן הֶעֱנִי שֶׁנֶּאֱמַר . אֲבָל לִמְכֹּר וּלְהַנִּיחַ הַדָּמִים בְּכִיסוֹ אוֹ לַעֲשׂוֹת בָּהֶן סְחוֹרָה אוֹ לִקַּח בָּהֶן כֵּלִים אוֹ עֲבָדִים וּבְהֵמָה אֵינוֹ רַשַּׁאי אֶלָּא לִמְזוֹנוֹת בִּלְבַד. וְאִם עָבַר וּמָכַר מִכָּל מָקוֹם הֲרֵי אֵלּוּ מְכוּרִין: וְדָנִין בְּבַיִת דִּין בָּתֵּי עָרֵי חוֹמָה וּבְשָׂדֶה דִּין שְׂדֵה אֲחֻזָּה. דִּין מוֹכֵר שְׂדֶה אֲחֻזָּתוֹ לְחַשֵּׁב אֶת הַדָּמִים לְפִי הַשָּׁנִים הַנִּשְׁאָרוֹת לַיּוֹבֵל. וּבְכָל עֵת שֶׁיִּרְצֶה לִפְדּוֹת מְחַשֵּׁב עִם הַלּוֹקֵחַ עַל הַשָּׁנִים שֶׁאָכַל. וְגוֹרֵעַ מִדְּמֵי הַמּוֹכֵר וּמַחֲזִיר לוֹ הַשְּׁאָר: כֵּיצַד. הֲרֵי הַנִּשְׁאָר לַיּוֹבֵל עֶשֶׂר שָׁנִים וּמָכַר לוֹ שָׂדֶה בְּמֵאָה דִּינָר. אֲכָלָהּ הַלּוֹקֵחַ שָׁלֹשׁ שָׁנִים וְרָצָה הַמּוֹכֵר לִגְאל נוֹתֵן לוֹ שִׁבְעִים דִּינָר וּמַחֲזִיר שָׂדֵהוּ. וְכֵן אִם אֲכָלָהּ שֵׁשׁ שָׁנִים נוֹתֵן לוֹ אַרְבָּעִים דִּינָר וּמַחֲזִיר לוֹ שָׂדֵהוּ. לֹא גְאָלָהּ אֶלָּא הִנִּיחָהּ בְּיַד הַלּוֹקֵחַ עַד שְׁנַת הַיּוֹבֵל תַּחֲזֹר לַבְּעָלִים בְּלֹא דָּמִים שֶׁנֶּאֱמַר : מְכָרָהּ לוֹ וְהִיא מְלֵאָה פֵּרוֹת וּלְאַחַר שְׁתֵּי שָׁנִים גְאָלָהּ אֵינוֹ יָכוֹל לוֹמַר לוֹ הַחֲזִירֶנָּה לִי מְלֵאָה פֵּרוֹת כְּמוֹ שֶׁמָּכַרְתִּי לְךָ. לְפִיכָךְ אִם מְכָרָהּ לוֹ מְלֵאָה פֵּרוֹת לִפְנֵי רֹאשׁ הַשָּׁנָה וּגְאָלָהּ אַחַר שְׁתֵּי שָׁנִים הֲרֵי זֶה אוֹכֵל שָׁלֹשׁ תְּבוּאוֹת בִּשְׁתֵּי הַשָּׁנִים וְאֵינוֹ מְחַשֵּׁב עִמּוֹ אֶלָּא לְפִי שְׁתֵּי שָׁנִים בִּלְבַד שֶׁנֶּאֱמַר לֹא עַל פִּי הַתְּבוּאוֹת: הַקָּנִים וְהַזְּמוֹרוֹת וּפֵרוֹת שִׁקְמָה שֶׁבְּתוֹכָהּ הֲרֵי הֵן שֶׁל לוֹקֵחַ כִּשְׁאָר הַפֵּרוֹת שֶׁלָּהּ. אֲבָל אִילָן שֶׁנִּכְסַח אוֹ שֶׁיָּבַשׁ שְׁנֵיהֶן אֲסוּרִין בּוֹ. כֵּיצַד יַעֲשֶׂה. יִמָּכֵר וְיִלָּקַח בְּדָמָיו קַרְקַע וְהַלּוֹקֵחַ אוֹכֵל פֵּרוֹתֶיהָ עַד שֶׁתִּגָּאֵל הַשָּׂדֶה מִיָּדוֹ: הַלּוֹקֵחַ שְׂדֵה אֲחֻזָּה וּנְטָעָהּ אִילָנוֹת וְהִשְׁבִּיחָה כְּשֶׁהִיא חוֹזֶרֶת בַּיּוֹבֵל שָׁמִין שֶׁבַח הָאִילָנוֹת שֶׁבְּתוֹכָהּ. וְנוֹתֵן בַּעַל הַשָּׂדֶה דְּמֵי הַשֶּׁבַח לַלּוֹקֵחַ שֶׁנֶּאֱמַר מִמְכָּר חוֹזֵר וְלֹא הַשֶּׁבַח: הַמּוֹכֵר אֶת שָׂדֵהוּ בִּזְמַן שֶׁהַיּוֹבֵל נוֹהֵג כְּמוֹ שֶׁבֵּאַרְנוּ אֵינוֹ מֻתָּר לִגְאל לְפָחוֹת מִשְּׁתֵּי שָׁנִים שֶׁנֶּאֱמַר . וַאֲפִלּוּ רָצָה הַלּוֹקֵחַ אָסוּר שֶׁנֶּאֱמַר . אֵין פָּחוֹת מִשְּׁתֵּי שָׁנִים מֵעֵת לְעֵת מִיּוֹם הַמְּכִירָה: וְצָרִיךְ שֶׁיֹּאכַל הַלּוֹקֵחַ שְׁתֵּי תְּבוּאוֹת בִּשְׁתֵּי שָׁנִים וְאַחַר כָּךְ יִגְאַל שֶׁנֶּאֱמַר . לְפִיכָךְ אִם הָיְתָה אַחַת מִשְּׁתֵּי הַשָּׁנִים שְׁבִיעִית אוֹ שְׁנַת שִׁדָּפוֹן אוֹ יֵרָקוֹן אֵינָהּ עוֹלָה מִן הַמִּנְיָן: הִנִּיחָהּ הַלּוֹקֵחַ בּוּרָה שָׁנָה וַאֲכָלָהּ שָׁנָה. אוֹ אֲכָלָהּ שָׁנָה וְנָרָהּ שָׁנָה וְלֹא זְרָעָהּ הֲרֵי אֵלּוּ עוֹלִין לַמִּנְיָן. מְכָרָהּ בִּשְׁנַת יוֹבֵל עַצְמָהּ אֵינָהּ נִמְכֶּרֶת וְהַדָּמִים חוֹזְרִין לַבְּעָלִים: מְכָרָהּ שָׁנָה אַחַת לִפְנֵי הַיּוֹבֵל הֲרֵי הַלּוֹקֵחַ אוֹכֵל אוֹתָהּ שָׁנָה שְׁנִיָּה אַחַר הַיּוֹבֵל שֶׁנֶּאֱמַר : מָכַר נְקָעִים מְלֵאִים מַיִם אוֹ סְלָעִים שֶׁאֵינָם רְאוּיִין לִזְרִיעָה הֲרֵי זֶה פּוֹדֶה בְּפָחוֹת מִשְּׁתֵּי שָׁנִים שֶׁנֶּאֱמַר . שָׂדֶה הָרָאוּי לַתְּבוּאָה הִיא שֶׁאֵינָהּ נִגְאֶלֶת אֶלָּא אַחַר שְׁתֵּי שָׁנִים. וְאִם לֹא גְּאָלָהּ אַף עַל פִּי שֶׁאֵינָהּ רְאוּיָה לִזְרִיעָה חוֹזֶרֶת לַבְּעָלִים בַּיּוֹבֵל: מָכַר אִילָנוֹת אֵין נִגְאָלִין לְפָחוֹת מִשְּׁתֵּי שָׁנִים שֶׁהֲרֵי רְאוּיִים לִתְבוּאוֹת. וְאִם לֹא גְּאָלָן אֵינָן חוֹזְרִין לַבְּעָלִים בַּיּוֹבֵל שֶׁנֶּאֱמַר וְלֹא לָאִילָנוֹת: מָכַר שָׂדֵהוּ לְרִאשׁוֹן וְרִאשׁוֹן מָכַר לְשֵׁנִי וְשֵׁנִי לִשְׁלִישִׁי אֲפִלּוּ מֵאָה זֶה אַחַר זֶה בִּשְׁנַת הַיּוֹבֵל תַּחֲזֹר לָאָדוֹן הָרִאשׁוֹן. שֶׁנֶּאֱמַר : מְכָרָהּ לְרִאשׁוֹן בְּמֵאָה דִּינָר וְרִאשׁוֹן לְשֵׁנִי בְּמָאתַיִם וְרָצָה הָאָדוֹן לִגְאל אֵינוֹ מְחַשֵּׁב אֶלָּא עִם הָרִאשׁוֹן שֶׁנֶּאֱמַר . מְכָרָהּ לְרִאשׁוֹן בְּמָאתַיִם וְרִאשׁוֹן לְשֵׁנִי בְּמֵאָה הֲרֵי זֶה מְחַשֵּׁב עִם הָאַחֲרוֹן. וְכֵן אִם מָכַר בְּמֵאָה וְהִשְׁבִּיחָה בְּיַד הַלּוֹקֵחַ וַהֲרֵי הִיא רְאוּיָה לְהִמָּכֵר בְּמָאתַיִם מְחַשֵּׁב לְפִי מַה שֶּׁמָּכַר. וְאִם מְכָרָהּ בְּמָאתַיִם וְהִכְסִיפָה וַהֲרֵי הִיא רְאוּיָה לְהִמָּכֵר בְּמֵאָה מְחַשֵּׁב לְפִי מַה שֶּׁהִיא. וּלְעוֹלָם מְיַפִּים כֹּחַ מוֹכֵר שְׂדֵה אֲחֻזָּה וּמְרֵעִין כֹּחַ הַלּוֹקֵחַ: הַמּוֹכֵר שְׂדֵה אֲחֻזָּתוֹ וְהָיוּ לוֹ שָׂדוֹת אֲחֵרוֹת וּמָכַר מֵאוֹתָם הַשָּׂדוֹת כְּדֵי לִגְאל שָׂדֶה שֶׁמָּכַר אֵין שׁוֹמְעִין לוֹ. שֶׁנֶּאֱמַר עַד שֶׁיִּמְצָא דָּבָר שֶׁאֵינוֹ מָצוּי לוֹ בְּשָׁעָה שֶׁמָּכַר. וְכֵן אִם לָוָה לִגְאל אֵין שׁוֹמְעִין לוֹ שֶׁנֶּאֱמַר לֹא שֶׁיִּלְוֶה: מָצָא מְעַט וְרָצָה לִגְאל חֲצִי הַשָּׂדֶה שֶׁמָּכַר אֵין שׁוֹמְעִין לוֹ שֶׁנֶּאֱמַר . אוֹ גּוֹאֵל אֶת כֻּלָּהּ אוֹ אֵינוֹ גּוֹאֵל. וְאִם רָצוּ קְרוֹבָיו לִגְאל גּוֹאֲלִים שֶׁנֶּאֱמַר : הַנּוֹתֵן שָׂדֵהוּ מַתָּנָה הֲרֵי זוֹ חוֹזֶרֶת לוֹ בַּיּוֹבֵל שֶׁנֶּאֱמַר לְרַבּוֹת אֶת הַמַּתָּנָה: הָאַחִין שֶׁחָלְקוּ כְּלָקוֹחוֹת הֵן וּמַחְזִירִין זֶה לָזֶה חֶלְקוֹ בַּיּוֹבֵל. לֹא תִּבָּטֵל חֲלוּקָתָן מִכְּמוֹת שֶׁהָיְתָה. וְכֵן הַבְּכוֹר וְהַמְיַבֵּם אֵשֶׁת אָחִיו מַחֲזִיר בַּיּוֹבֵל חֵלֶק שֶׁנָּטַל וְנוֹטֵל הַחֵלֶק שֶׁכְּנֶגְדּוֹ: אֲבָל הַיּוֹרֵשׁ אֶת אִשְׁתּוֹ אַף עַל פִּי שֶׁיְּרֻשַּׁת הַבַּעַל מִדִּבְרֵיהֶם עָשׂוּ חִזּוּק לְדִבְרֵיהֶם כְּשֶׁל תּוֹרָה וְאֵינוֹ מַחֲזִיר בַּיּוֹבֵל. וְאִם יָרַשׁ מִמֶּנָּה בֵּית הַקְּבָרוֹת יַחֲזִיר לִבְנֵי מִשְׁפָּחָה מִשּׁוּם פְּגַם מִשְׁפָּחָה וְיִתְּנוּ לוֹ דָּמֶיהָ וּמְנַכִּין לוֹ דְּמֵי קֶבֶר אִשְׁתּוֹ שֶׁהֲרֵי חַיָּב בִּקְבוּרָתָהּ:
