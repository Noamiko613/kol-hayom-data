וּבְנוֹ אֵינוֹ פָּדוּי. נָתַן לוֹ כְּלִי שֶׁאֵינוֹ שָׁוֶה בַּשּׁוּק חֲמִשָּׁה סְלָעִים וְקִבְּלוֹ הַכֹּהֵן בַּחֲמִשָּׁה סְלָעִים הֲרֵי בְּנוֹ פָּדוּי. נָתַן חֲמִשָּׁה סְלָעִים לַעֲשָׂרָה כֹּהֲנִים בֵּין בְּבַת אַחַת בֵּין בָּזֶה אַחַר זֶה יָצָא: רָצָה הַכֹּהֵן לְהַחְזִיר לוֹ הַפִּדְיוֹן מַחְזִיר. וְלֹא יִתֵּן לוֹ הוּא וְדַעְתּוֹ שֶׁיַּחְזִיר. וְאִם עָשָׂה כֵּן וְהֶחְזִיר לוֹ אֵין בְּנוֹ פָּדוּי עַד שֶׁיִּגְמֹר בְּלִבּוֹ לִתֵּן לוֹ מַתָּנָה גְּמוּרָה. וְאִם רָצָה הַכֹּהֵן אַחַר כָּךְ לְהַחְזִיר יַחְזִיר. וְכֵן אִם פֵּרֵשׁ וְנָתַן לוֹ עַל מְנָת לְהַחְזִיר הֲרֵי בְּנוֹ פָּדוּי: כֹּהֲנִים וּלְוִיִּם פְּטוּרִים מִפִּדְיוֹן הַבֵּן מִקַּל וָחֹמֶר. אִם פָּטְרוּ שֶׁל יִשְׂרָאֵל בַּמִּדְבָּר דִּין הוּא שֶׁיִּפְטְרוּ עַצְמָן: יִשְׂרָאֵל הַבָּא מִן הַכֹּהֶנֶת וּמִן הַלְּוִיָּה פָּטוּר. שֶׁאֵין הַדָּבָר תָּלוּי בָּאָב אֶלָּא בָּאֵם שֶׁנֶּאֱמַר פֶּטֶר רֶחֶם בְּיִשְׂרָאֵל: לְוִיָּה הַמְעֻבֶּרֶת מֵעַכּוּ''ם בְּנָהּ פָּטוּר. וְכֹהֶנֶת הַמְעֻבֶּרֶת מֵעַכּוּ''ם בְּנָהּ חַיָּב שֶׁהֲרֵי נִפְסְלָה אִמּוֹ מִן הַכְּהֻנָּה בִּבְעִילַת הָעַכּוּ''ם: כֹּהֵן שֶׁנּוֹלַד לוֹ בֶּן חָלָל. מֵת הָאָב בְּתוֹךְ שְׁלֹשִׁים יוֹם הַבֵּן חַיָּב לִפְדּוֹת אֶת עַצְמוֹ. שֶׁלֹּא זָכָה הָאָב בְּפִדְיוֹנוֹ. מֵת אַחַר שְׁלֹשִׁים יוֹם אֵינוֹ חַיָּב הַבֵּן לִפְדּוֹת אֶת עַצְמוֹ. שֶׁהֲרֵי זָכָה הָאָב בְּפִדְיוֹנוֹ: הַשִּׁפְחָה שֶׁנִּשְׁתַּחְרְרָה וְכוּתִית שֶׁנִּתְגַּיְּרָה כְּשֶׁהֵן מְעֻבָּרוֹת וְיָלְדוּ. אַף עַל פִּי שֶׁהוֹרָתוֹ שֶׁלֹּא בִּקְדֻשָּׁה הוֹאִיל וְנוֹלַד בִּקְדֻשָּׁה חַיָּב. שֶׁנֶּאֱמַר פֶּטֶר רֶחֶם בְּיִשְׂרָאֵל וַהֲרֵי פֶּטֶר וְרֶחֶם בְּיִשְׂרָאֵל. אֵין יָדוּעַ אִם קֹדֶם שֶׁנִּתְגַּיְּרָה יָלְדָה אוֹ אַחַר שֶׁנִּתְגַּיְּרָה יָלְדָה הַמּוֹצִיא מֵחֲבֵרוֹ עָלָיו הָרְאָיָה: הַכּוּתִית וְהַשִּׁפְחָה שֶׁיָּלְדוּ וְאַחַר כָּךְ נִתְגַּיְּרוּ וְנִשְׁתַּחְרְרוּ וְיָלְדוּ וָלָד אַחֵר הֲרֵי זֶה פָּטוּר שֶׁנֶּאֱמַר וְאֵין זֶה פֶּטֶר רֶחֶם. וְכֵן הַבָּא אַחַר הַנְּפָלִים כָּל נֵפֶל שֶׁאִמּוֹ טְמֵאָה לֵדָה הַבָּא אַחֲרָיו אֵינוֹ פֶּטֶר רֶחֶם. וְכָל נֵפֶל שֶׁאֵין אִמּוֹ טְמֵאָה לֵדָה כְּמוֹ וְיָלְדָה. זֶה שֶׁיָּלְדָה אֵינוֹ פֶּטֶר רֶחֶם. שֶׁהֲרֵי נִפְטַר בְּרֹאשׁוֹ שֶׁל רִאשׁוֹן. וּמִשֶּׁתֵּצֵא פַּדַּחְתּוֹ פּוֹטֵר הַבָּא אַחֲרָיו: יוֹצֵא דֹּפֶן וְהַבָּא אַחֲרָיו כְּדַרְכּוֹ שְׁנֵיהֶם שְׁלֹשִׁים וַאֲפִלּוּ בְּיוֹם שְׁלֹשִׁים. וְכֵן אִם נַעֲשָׂה טְרֵפָה. אֵינוֹ חַיָּב בַּחֲמִשָּׁה סְלָעִים. וְאִם הִקְדִּים וְנָתַן לַכֹּהֵן יַחְזִיר לוֹ הַפִּדְיוֹן. וְאִם לֹא נָתַן ] יִתֵּן: מִי שְׁלֹשִׁים יוֹם: מִי שֶׁהוּא סָפֵק אִם חַיָּב בְּפִדְיוֹן אוֹ לֹא הֲרֵי זֶה פָּטוּר שֶׁהַמּוֹצִיא מֵחֲבֵרוֹ עָלָיו הָרְאָיָה. מֵת הָאָב בְּתוֹךְ שְׁלֹשִׁים יוֹם הֲרֵי הַבֵּן בְּחֶזְקַת שֶׁלֹּא אֵין כָּאן לַכֹּהֵן כְּלוּם שֶׁאֲנִי אוֹמֵר נְקֵבָה נוֹלְדָה תְּחִלָּה וְאַחֲרֶיהָ זָכָר: שְׁתֵּי נָשָׁיו אַחַת בִּכְּרָה וְאַחַת לֹא בִּכְּרָה וְיָלְדוּ שְׁנֵי זְכָרִים וְנִתְעָרְבוּ. נוֹתֵן חֲמִשָּׁה סְלָעִים לַכֹּהֵן. מֵת אֶחָד מֵהֶם בְּתוֹךְ שְׁלֹשִׁים יוֹם הָאָב פָּטוּר. מֵת הָאָב יִנָּתֵן מִן הַנְּכָסִים חֲמִשָּׁה סְלָעִים: יָלְדוּ * זָכָר וּנְקֵבָה אוֹ שְׁנֵי זְכָרִים וּנְקֵבָה אֵין כָּאן לַכֹּהֵן כְּלוּם. שֶׁאֲנִי אוֹמֵר זוֹ שֶׁלֹּא בִּכְּרָה יָלְדָה נְקֵבָה תְּחִלָּה וְאַחֲרֶיהָ זָכָר וְזוֹ שֶׁבִּכְּרָה יָלְדָה זָכָר: שְׁתֵּי נָשִׁים שֶׁל שְׁנֵי אֲנָשִׁים שֶׁלֹּא בִּכְּרוּ וְיָלְדוּ שְׁנֵי זְכָרִים וְנִתְעָרְבוּ זֶה נוֹתֵן חֲמִשָּׁה סְלָעִים וְזֶה נוֹתֵן חֲמִשָּׁה סְלָעִים. נָתְנוּ וְאַחַר כָּךְ מֵת אֶחָד מִן הַבָּנִים בְּתוֹךְ שְׁלֹשִׁים יוֹם. אִם לִשְׁנֵי כֹּהֲנִים נָתְנוּ אֵינָן יְכוֹלִין לְהוֹצִיא מִיָּדָן. וְאִם לְכֹהֵן אֶחָד נָתְנוּ כּוֹתֵב אֶחָד מֵהֶן לַחֲבֵרוֹ הַרְשָׁאָה וְיֵלֵךְ זֶה בְּהַרְשָׁאָה וְיַחְזִיר מִן הַכֹּהֵן חֲמִשָּׁה סְלָעִים: יָלְדוּ זָכָר וּנְקֵבָה וְנִתְעָרְבוּ הָאָבוֹת פְּטוּרִין וְהַבֵּן חַיָּב לִפְדּוֹת אֶת עַצְמוֹ. וְכֵן מְבַכֶּרֶת שֶׁלֹּא שָׁהֲתָה אַחַר בַּעְלָהּ שְׁלֹשָׁה חֳדָשִׁים וְיָלְדָה וְאֵין יָדוּעַ אִם בֶּן תִּשְׁעָה לָרִאשׁוֹן אוֹ בֶּן שִׁבְעָה לָאַחֲרוֹן שְׁנֵיהֶן פְּטוּרִין. וְהַבֵּן חַיָּב לִפְדּוֹת אֶת עַצְמוֹ: יָלְדוּ שְׁתֵּי נְקֵבוֹת וְזָכָר אוֹ שְׁתֵּי נְקֵבוֹת וּשְׁנֵי זְכָרִים אֵין כָּאן לַכֹּהֵן כְּלוּם: שְׁתֵּי נָשִׁים שֶׁל שְׁנֵי אֲנָשִׁים אַחַת בִּכְּרָה וְאַחַת לֹא בִּכְּרָה וְיָלְדוּ שְׁנֵי זְכָרִים. זֶה שֶׁלֹּא בִּכְּרָה אִשְׁתּוֹ נוֹתֵן חָמֵשׁ סְלָעִים לַכֹּהֵן. יָלְדוּ זָכָר וּנְקֵבָה
