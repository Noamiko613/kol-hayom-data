הלכות מעשר - פרק עשירי

הַמְקַבֵּל ] עָלָיו לִהְיוֹת נֶאֱמָן עַל הַמַּעַשְׂרוֹת וְלֹא יִהְיוּ פֵּרוֹתָיו דְּמַאי צָרִיךְ לִהְיוֹת מְעַשֵּׂר אֶת שֶׁהוּא אוֹכֵל וְאֶת שֶׁהוּא מוֹכֵר וְאֶת שֶׁהוּא לוֹקֵחַ. וְאֵינוֹ מִתְאָרֵחַ אֵצֶל עַם הָאָרֶץ. וְצָרִיךְ שֶׁיְּקַבֵּל עָלָיו דְּבָרִים אֵלּוּ בָּרַבִּים. וּכְשֶׁיָּבוֹאוּ עֵדִים נֶאֱמָנִין שֶׁקִּבֵּל דְּבָרִים אֵלּוּ בָּרַבִּים וְשֶׁהוּא רָגִיל בָּהֶם תָּמִיד הֲרֵי זֶה נֶאֱמָן עַל פֵּרוֹתָיו לוֹמַר מְעֵשָּׂרִין הֵן:

כָּל תַּלְמִיד חָכָם לְעוֹלָם נֶאֱמָן וְאֵינוֹ צָרִיךְ בְּדִיקָה אַחֲרָיו. וּבָנָיו וּבְנֵי בֵּיתוֹ וַעֲבָדָיו ] הֲרֵי הֵם כָּמוֹהוּ. תַּלְמִיד חָכָם שֶׁמֵּת וְהִנִּיחַ פֵּרוֹת אֲפִלּוּ כְּנָסָם בְּאוֹתוֹ הַיּוֹם הֲרֵי הֵן בְּחֶזְקַת מְתֻקָּנִין:

בַּת עַם הָאָרֶץ אוֹ אִשְׁתּוֹ שֶׁנִּסֵּת לְחָבֵר וְעַבְדּוֹ שֶׁנִּמְכָּר לְחָבֵר צְרִיכִין לְקַבֵּל עֲלֵיהֶם בַּתְּחִלָּה. וּבַת חָבֵר אוֹ אִשְׁתּוֹ שֶׁנִּסֵּת לְעַם הָאָרֶץ וְעַבְדּוֹ שֶׁנִּמְכָּר לְעַם הָאָרֶץ הֲרֵי הֵן בְּחֶזְקָתָן עַד שֶׁיֵּחָשְׁדוּ. בְּנוֹ אוֹ עַבְדּוֹ שֶׁל חָבֵר שֶׁהָיוּ לְמוּדִין אֵצֶל עַם הָאָרֶץ צְרִיכִין לְקַבֵּל. בְּנוֹ אוֹ עַבְדּוֹ שֶׁל עַם הָאָרֶץ שֶׁהָיוּ לְמוּדִים אֵצֶל חָבֵר כָּל זְמַן שֶׁהֵן אֶצְלוֹ הֲרֵי הֵן כְּחָבֵר. יָצְאוּ הֲרֵי הֵן כְּעַם הָאָרֶץ:

מִי שֶׁאֵינוֹ נֶאֱמָן וְהָיָה אֶחָד מִבָּנָיו אוֹ מֵעֲבָדָיו נֶאֱמָן אוֹ אֶחָד מִמִּשְׁפַּחְתּוֹ לוֹקְחִין מִמֶּנּוּ וְאוֹכְלִין עַל פִּיהֶן וְאֵינָן חוֹשְׁשִׁין לְדָבָר:

הָיָה הוּא נֶאֱמָן וְאִשְׁתּוֹ אֵינָהּ נֶאֱמֶנֶת לוֹקְחִים מִמֶּנּוּ וְאֵין מִתְאָרְחִין אֶצְלוֹ. :

לֹא יְשַׁמֵּשׁ חָבֵר לֹא בְּמִשְׁתֵּה עַם הָאָרֶץ וְלֹא בִּסְעֵדָּתוֹ אֶלָּא אִם כֵּן הָיָה הַכּל מְעֻשָּׂר וּמְתֻקָּן מִתַּחַת יָדוֹ. לְפִיכָךְ הָרוֹאֶה חָבֵר מְשַׁמֵּשׁ בְּמִשְׁתֵּה עַם הָאָרֶץ אוֹ בִּסְעֻדָּתוֹ הֲרֵי הַכּל בְּחֶזְקַת מְתֻקָּן וּמְעֻשָּׂר. רָאוּהוּ מֵסֵב עִם עַמֵּי הָאָרֶץ אֵין הַסְּעֻדָּה בְּחֶזְקַת מְעֻשֶּׂרֶת. שֶׁמָּא הֶחָבֵר עַל הַתְּנָאִים שֶׁבְּלִבּוֹ הוּא סוֹמֵךְ:

כְּשֵׁם שֶׁאָדָם סוֹעֵד אֵצֶל עַם הָאָרֶץ וְסוֹמֵךְ עַל תְּנָאוֹ כָּךְ הוּא צָרִיךְ לִהְיוֹת מַתְנֶה עַל בְּנוֹ וַאֲפִלּוּ הָיָה בְּנוֹ בְּמָקוֹם אַחֵר. אֲבָל אֵינוֹ צָרִיךְ לִהְיוֹת מַתְנֶה עַל אַחֵר חוּץ מִבְּנוֹ וַאֲפִלּוּ הָיָה עִמּוֹ בְּמִשְׁתֶּה. לְפִיכָךְ בֶּן חָבֵר שֶׁהָיָה מֵסֵב בְּמִשְׁתֶּה עַם הָאָרֶץ אֵין הַסְּעֻדָּה בְּחֶזְקַת מְעֻשֶּׂרֶת שֶׁמָּא אָבִיו הִתְנָה עָלָיו:

עַם הָאָרֶץ שֶׁנָּתַן מָעָה לְחָבֵר וְאָמַר לוֹ קַח לִי אֲגֻדַּת יָרָק אַחַת אוֹ גְּלֻסְקָא אַחַת לוֹקֵחַ לוֹ סְתָם וּפָטוּר מִלְּעַשֵּׂר. וְאִם הֶחֱלִיף אֶת הַמָּעָה חַיָּב לְעַשֵּׂר וְאַחַר כָּךְ יָבִיא לוֹ. וְאִם פֵּרֵשׁ הֶחָבֵר וְלֹא לְקָחָן סְתָם אֶלָּא אָמַר לַחֶנְוָנִי אֲגֻדָּה זוֹ שֶׁאֲנִי לוֹקֵחַ מִמְּךָ לַחֲבֵרִי אֲנִי קוֹנֶה אוֹתָהּ וְזוֹ לְעַצְמִי אֲנִי קוֹנֶה אוֹתָהּ. זוֹ שֶׁלָּקַח לְעַצְמוֹ חַיָּב לְעַשְּׂרָהּ. וְזוֹ שֶׁלָּקַח לַחֲבֵרוֹ אֵינוֹ צָרִיךְ לְעַשְּׂרָהּ. וְאִם נִתְעָרְבוּ אֲפִלּוּ נִתְעָרְבָה אַחַת שֶׁלּוֹ בְּמֵאָה שֶׁל חֲבֵרוֹ מְתַקֵּן הַכּל דְּמַאי. וְאַחַר כָּךְ יִתֵּן לַחֲבֵרוֹ שֶׁשְּׁלָחוֹ לִקַּח לוֹ:

חֲמִשָּׁה שֶׁאָמְרוּ לְאֶחָד צֵא וְהָבֵא לָנוּ חָמֵשׁ אֲגֻדּוֹת שֶׁל יָרָק. וְהֵבִיא לְכָל אֶחָד וְאֶחָד בִּפְנֵי עַצְמוֹ. חֲבֵרִים שֶׁבָּהֶן אֵין צְרִיכִין לְעַשֵּׂר אֶלָּא עַל חֶלְקָן בִּלְבַד. הֵבִיא לָהֶן בְּעִרְבּוּב חֲבֵרִים שֶׁבָּהֶם צְרִיכִין לְעַשֵּׂר הַכּל:

עַם הָאָרֶץ שֶׁאָמַר לְחָבֵר צֵא וְלַקֵּט לִי תְּאֵנִים מִתְּאֵנָתִי אוֹכֵל מֵהֶן הֶחָבֵר עַרְאַי וּמְעַשְּׂרָן ] . וְאִם אָמַר הֶחָבֵר לְעַם הָאָרֶץ לִלְקֹט לוֹ וְשָׁמַע חָבֵר אַחֵר זֶה הָאַחֵר אוֹכֵל וְאֵינוֹ צָרִיךְ לְעַשֵּׂר שֶׁאֵין הֶחָבֵר מוֹצִיא דָּבָר שֶׁאֵינוֹ מְתֻקָּן מִתַּחַת יָדוֹ וְחֶזְקָתוֹ שֶׁהִפְרִישׁ עָלֶיהָ מִמָּקוֹם אַחֵר. אַף עַל פִּי שֶׁלֹּא נֶחְשְׁדוּ חֲבֵרִים לִתְרֹם שֶׁלֹּא מִן הַמֻּקָּף, כְּדֵי לְסַלֵּק הַמִּכְשׁוֹל מִלִּפְנֵי עַם הָאָרֶץ תּוֹרְמִין:

מֻתָּר לְהַאֲכִיל אֶת הָעֲנִיִּים וְאֶת הָאוֹרְחִים דְּמַאי וְצָרִיךְ לְהוֹדִיעָן. וְהֶעָנִי עַצְמוֹ וְהָאוֹרֵחַ אִם רָצוּ לְתַקֵּן מְתַקְּנִין:

גַּבָּאֵי צְדָקָה גּוֹבִין סְתָם מִכָּל אָדָם וּמְחַלְּקִים סְתָם וְהָרוֹצֶה לְתַקֵּן יְתַקֵּן:

רוֹפֵא חָבֵר שֶׁהָיָה מַאֲכִיל לְחוֹלֵה עַם הָאָרֶץ מִפֵּרוֹת עַם הָאָרֶץ נוֹתֵן לְתוֹךְ יָדוֹ אֲבָל לֹא לְתוֹךְ פִּיו. וְאִם הָיָה הַדְּמַאי שֶׁל רוֹפֵא אֲפִלּוּ לְתוֹךְ יָדוֹ לֹא יִתֵּן. וְכֵן אִם יָדַע שֶׁהוּא טֶבֶל וַדַּאי אֲפִלּוּ לְתוֹךְ יָדוֹ אָסוּר:
