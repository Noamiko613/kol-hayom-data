בָּזֶה: הַמּוֹכֵר יַיִן לַחֲבֵרוֹ וּנְתָנוֹ הַלּוֹקֵחַ בְּקַנְקַנָּיו וְהֶחְמִיץ מִיָּד אֵינוֹ חַיָּב בְּאַחֲרָיוּתוֹ. וְאַף עַל פִּי שֶׁאָמַר לוֹ לְתַבְשִׁיל אֲנִי צָרִיךְ לוֹ. וְאִם יָדַע שֶׁיֵּינוֹ מַחְמִיץ הֲרֵי זֶה מִקַּח טָעוּת. מָכַר לוֹ יַיִן וַהֲרֵי הוּא בְּקַנְקַנָּיו שֶׁל מוֹכֵר וְהֶחְמִיץ. אִם אָמַר לוֹ לְמִקְפָּה אֲנִי צָרִיךְ וְהֶחְמִיץ מַחְזִיר וְאוֹמֵר לוֹ הֲרֵי יֵינְךָ וְקַנְקַנְּךָ שֶׁאֲנִי לֹא קָנִיתִי לִשְׁתּוֹתוֹ אֶלָּא לְבַשֵּׁל מְעַט מְעַט. וְאִם לֹא אָמַר לְמִקְפָּה הוּא אֵינוֹ יָכוֹל לְהַחְזִיר. שֶׁהֲרֵי אוֹמֵר לוֹ לָמָּה לֹא שָׁתִיתָ אוֹתוֹ. וְלֹא הָיָה פְּלוֹנִי לְמָכְרָהּ שָׁם וְקֹדֶם שֶׁהִגִּיעַ שָׁם הוּזַל הַיַּיִן אוֹ הֶחְמִיצָה הֲרֵי זֶה בִּרְשׁוּת הַמּוֹכֵר. מִפְּנֵי שֶׁהֶחָבִית וְהַיַּיִן שֶׁלּוֹ. וְכֵן כָּל כַּיּוֹצֵא בָּזֶה: הָאוֹמֵר וְצָרִיךְ שֶׁיַּעֲמֹד וְלֹא יַחְמִיץ עַד הֶחָג. וּבְמָקוֹם שֶׁיֵּשׁ מִנְהָג יָדוּעַ הַכּל כְּמִנְהַג הַמְּדִינָה: הָאוֹמֵר לַחֲבֵרוֹ מַרְתֵּף זֶה שֶׁל יַיִן אֲנִי מוֹכֵר לְךָ לְמִקְפָּה. אוֹ שֶׁמָּכַר לוֹ מַרְתֵּף שֶׁל יַיִן סְתָם. הֲרֵי הַלּוֹקֵחַ מְקַבֵּל עָלָיו עֲשָׂרָה קַנְקַנִּים בְּכָל מֵאָה שֶׁלֹּא יִהְיֶה יֵינָם טוֹב אֶלָּא כְּבָר הִתְחִיל לְהִשְׁתַּנּוֹת. יֶתֶר עַל זֶה לֹא יְקַבֵּל: אָמַר נוֹתֵן לוֹ יַיִן הַנִּמְכָּר בַּחֲנוּת שֶׁהֲרֵי הוּא בֵּינוֹנִי לֹא רַע וְלֹא יָפֶה. אָמַר לוֹ מַרְתֵּף זֶה אֲנִי מוֹכֵר לְךָ וְלֹא הִזְכִּיר יַיִן אֲפִלּוּ כֻּלּוֹ חֹמֶץ הִגִּיעוֹ. וְכֵן כָּל
