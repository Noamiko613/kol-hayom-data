הֲרֵי זוֹ סְפֵק זָבָה שֶׁמָּא יוֹם אֶחָד מֵהֶן קֹדֶם נִדָּתָהּ וּשְׁנַיִם בִּתְחִלַּת הַנִּדָּה. וְכֵן אִם רָאֲתָה אַרְבָּעָה שֶׁמָּא שְׁנַיִם קֹדֶם הַנִּדָּה וּשְׁנַיִם מִתְּחִלַּת הַנִּדָּה וְיוֹשֶׁבֶת חֲמִשָּׁה תַּשְׁלוּם יְמֵי נִדָּה וְאַחַד עָשָׂר יְמֵי זִיבָה אַחַר הַחֲמִשָּׁה: וְכֵן אִם רָאֲתָה תִּשְׁעָה יָמִים הֲרֵי זוֹ סְפֵק זָבָה שֶׁמָּא שְׁנַיִם קֹדֶם יְמֵי נִדָּה וְשִׁבְעָה שֶׁל נִדָּה וּמַתְחֶלֶת לִמְנוֹת אַחַד עָשָׂר יוֹם מֵאַחַר הַתִּשְׁעָה שֶׁפָּסַק הַדָּם. וְכֵן אִם רָאֲתָה אַחַד עָשָׂר יוֹם הֲרֵי זוֹ סְפֵק זָבָה. שֶׁמָּא שְׁנַיִם קֹדֶם הַנִּדָּה וְשִׁבְעָה שֶׁל נִדָּה וּשְׁנַיִם שֶׁל אַחַר הַנִּדָּה וְנִשְׁאַר לָהּ מִימֵי זִיבָתָהּ תִּשְׁעָה: רָאֲתָה שְׁנֵים עָשָׂר יוֹם הֲרֵי זוֹ זָבָה וַדָּאִית. שֶׁאֲפִלּוּ הָיוּ מֵהֶן שְׁנַיִם לִפְנֵי הַנִּדָּה וְשִׁבְעָה שֶׁל נִדָּה הֲרֵי שְׁלֹשָׁה לְאַחַר הַנִּדָּה וְיִשָּׁאֵר לָהּ מִימֵי זִיבָתָהּ שְׁמוֹנָה. וְכֵן אִם רָאֲתָה שְׁלֹשָׁה עָשָׂר יוֹם יִשָּׁאֵר לָהּ מִימֵי זִיבָתָהּ שִׁבְעָה וְהֵן יְמֵי הַסְּפִירָה: מָשְׁכָה בִּרְאִיַּת הַדָּם אֲפִלּוּ רָאֲתָה אֶלֶף יוֹם כְּשֶׁיִּפְסֹק הַדָּם סוֹפֶרֶת שִׁבְעַת יָמִים נְקִיִּים. וְאַחַר הַשִּׁבְעָה יַתְחִילוּ יְמֵי הַנִּדָּה לְזוֹ שֶׁטָּעֲתָה: הִנֵּה לָמַדְתָּ שֶׁכָּל הַטּוֹעָה אֵינָהּ מוֹנָה מִשֶּׁיִּפְסֹק הַדָּם פָּחוֹת מִשִּׁבְעָה וְלֹא יוֹתֵר עַל י''ז וְיָבוֹאוּ יְמֵי נִדָּתָהּ. כֵּיצַד. רָאֲתָה יוֹם אֶחָד וּפָסַק הַדָּם מוֹנָה י''ז שִׁשָּׁה לְתַשְׁלוּם נִדָּתָהּ וְי''א יְמֵי זִיבָתָהּ וְיָבוֹאוּ יְמֵי נִדָּתָהּ. וְאִם רָאֲתָה י''ג אוֹ יֶתֶר מוֹנָה שִׁבְעָה מִשֶּׁיִּפְסֹק הַדָּם וְיָבוֹאוּ יְמֵי נִדָּתָהּ כְּמוֹ שֶׁבֵּאַרְנוּ:
