הלכות איסורי מזבח - פרק שלישי

אֵין הַמּוּמִין פּוֹסְלִין בָּעוֹף וְכֵן אֶחָד הַזָּכָר וְאֶחָד הַנְּקֵבָה בָּעוֹף שֶׁלֹּא נֶאֱמַר אֶלָּא בִּבְהֵמָה בִּלְבַד. בַּמֶּה דְּבָרִים אֲמוּרִים בְּמוּמִין קְטַנִּים. אֲבָל עוֹף שֶׁיָּבַשׁ גַּפּוֹ אוֹ נִסְמֵית עֵינוֹ אוֹ נִקְטְעָה רַגְלוֹ אָסוּר לְגַבֵּי הַמִּזְבֵּחַ שֶׁאֵין מַקְרִיבִין חָסֵר כְּלָל. וְכֵן אִם נוֹלַד בּוֹ אַחַת מִן הַטְּרֵפוֹת שֶׁאוֹסְרִין אוֹתָהּ בַּאֲכִילָה הֲרֵי זֶה נִפְסָל לְקָרְבָּן:

תּוֹרִים קְטַנִּים פְּסוּלִין וּבְנֵי יוֹנָה גְּדוֹלִים פְּסוּלִין שֶׁנֶּאֱמַר . תְּחִלַּת הַצִּהוּב בָּזֶה וּבָזֶה פָּסוּל. וְעַד מָתַי יִהְיוּ בְּנֵי יוֹנָה כְּשֵׁרִים כָּל זְמַן שֶׁעוֹקֵר כָּנָף וּמִתְמַלֵּא מְקוֹם עִקָּרוֹ דָּם. וְהַתּוֹרִים כְּשֵׁרִים מִשֶּׁיִּזְהֲבוּ:

הַטֻּמְטוּם וְהָאַנְדְּרוֹגִינוּס אַף עַל פִּי שֶׁאֵין לְךָ מוּם גָּדוֹל מֵהֶן הֲרֵי הֵן פְּסוּלִין לַמִּזְבֵּחַ מִדֶּרֶךְ אַחֶרֶת לְפִי שֶׁהֵן סָפֵק זָכָר סָפֵק נְקֵבָה הֲרֵי הֵן כְּמִין אַחֵר. וּבַקָּרְבָּנוֹת נֶאֱמַר וּ עַד שֶׁיִּהְיֶה זָכָר וַדַּאי אוֹ נְקֵבָה וַדָּאִית. לְפִיכָךְ אַף הָעוֹף שֶׁהוּא טֻמְטוּם אוֹ אַנְדְּרוֹגִינוּס פָּסוּל לַמִּזְבֵּחַ: (-)

וְכֵן הַכִּלְאַיִם וְיוֹצֵא דֹּפֶן וּמְחֻסַּר זְמַן פְּסוּלִין אַף עַל פִּי שֶׁאֵין בָּהֶם מוּם שֶׁנֶּאֱמַר עַד שֶׁיִּהְיֶה כָּל מִין וּמִין בִּפְנֵי עַצְמוֹ לֹא שֶׁיִּהְיֶה מְעֹרָב מִכֶּבֶשׂ וְעֵז. פְּרָט לְיוֹצֵא דֹּפֶן. פְּרָט לִמְחֻסַּר זְמַן. פְּרָט לְיָתוֹם שֶׁנּוֹלַד אַחַר שֶׁנִּשְׁחֲטָה אִמּוֹ:

הַנִּדְמֶה אַף עַל פִּי שֶׁאֵינוֹ כִּלְאַיִם הֲרֵי הוּא פָּסוּל לַמִּזְבֵּחַ. כֵּיצַד. רָחֵל שֶׁיָּלְדָה כְּמִין עֵז וְעֵז שֶׁיָּלְדָה כְּמִין כֶּבֶשׂ. אַף עַל פִּי שֶׁיֵּשׁ בּוֹ מִקְצָת סִימָנִין הוֹאִיל וְהוּא דּוֹמֶה לְמִין אַחֵר פָּסוּל כְּבַעַל מוּם קָבוּעַ שֶׁאֵין לְךָ מוּם קָבוּעַ גָּדוֹל מִן הַשִּׁנּוּי:

וְכֵן הָרוֹבֵעַ וְהַנִּרְבָּע וְהַמֻּקְצֶה לַעֲבוֹדָה זָרָה וְהַנֶּעֱבָד אַף עַל פִּי שֶׁהֵן מֻתָּרִין בַּאֲכִילָה הֲרֵי הֵן פְּסוּלִין לְגַבֵּי הַמִּזְבֵּחַ שֶׁנֶּאֱמַר כָּל שֶׁיֵּשׁ בּוֹ הַשְׁחָתָה פָּסוּל. וּבַעֲבֵרָה הוּא אוֹמֵר . וּבַעֲבוֹדָה זָרָה כְּתִיב . וְכֵן בְּהֵמָה וְעוֹף שֶׁהָרְגוּ אֶת הָאָדָם הֲרֵי הֵן כְּרוֹבֵעַ אוֹ נִרְבָּע וּפְסוּלִין לַמִּזְבֵּחַ: (-)

וְיֵרָאֶה לִי שֶׁאַף עַל פִּי שֶׁאֵין כָּל הַפְּסוּלִין הָאֵלּוּ רְאוּיִין לַמִּזְבֵּחַ לְקָרְבָּן אִם עָבַר וְהִקְרִיבָן אֵינוֹ לוֹקֶה מִן הַתּוֹרָה לְפִי שֶׁלֹּא נִתְפָּרְשָׁה אַזְהָרָתָן. כֶּלֶב אֲסוּרִין לְגַבֵּי הַמִּזְבֵּחַ וְהַמַּקְרִיב אֶחָד מֵהֶן אוֹ שְׁנֵיהֶן כְּאֶחָד לוֹקֶה אַחַת שֶׁנֶּאֱמַר . וּמִפְּנֵי מָה לוֹקֶה אַחַת עַל שְׁנֵיהֶן מִפְּנֵי שֶׁנֶּאֶמְרוּ בְּלָאו אֶחָד: (-)

מִצְוַת עֲשֵׂה לְהַקְרִיב כָּל הַקָּרְבָּנוֹת מִיּוֹם הַשְּׁמִינִי וָהָלְאָה שֶׁנֶּאֱמַר . וְכָל שִׁבְעַת הַיָּמִים נִקְרָא מְחֻסַּר זְמַן. וְאַף עַל פִּי שֶׁמְּחֻסַּר זְמַן פָּסוּל אִם עָבַר וְהִקְרִיבוֹ אֵינוֹ לוֹקֶה מִפְּנֵי שֶׁהוּא לָאו הַבָּא מִכְּלַל עֲשֵׂה וְלֹא נִרְצָה הַקָּרְבָּן:

תּוֹרִים שֶׁלֹּא הִגִּיעַ זְמַנָּן שֶׁהֵן כִּמְחֻסַּר זְמַן בִּבְהֵמָה וּבְנֵי יוֹנָה שֶׁעָבַר זְמַנָּן הַכּל כְּבַעַל מוּם וְהַמַּקְרִיבָן אֵינוֹ לוֹקֶה אַף עַל פִּי שֶׁהַקָּרְבָּן פָּסוּל וְלֹא נִרְצָה:

הַמַּקְדִּישׁ טֻמְטוּם וְאַנְדְּרוֹגִינוּס וּטְרֵפָה וְכִלְאַיִם וְיוֹצֵא דֹּפֶן לַמִּזְבֵּחַ הֲרֵי זֶה כְּמַקְדִּישׁ עֵצִים וַאֲבָנִים. לְפִי שֶׁאֵין קְדֻשָּׁה חָלָה עַל גּוּפָן וַהֲרֵי הֵן חֻלִּין לְכָל דָּבָר. וְיִמָּכְרוּ וְיָבִיא בִּדְמֵיהֶם * כָּל קָרְבָּן שֶׁיִּרְצֶה. וְאֵינָן כְּבַעַל מוּם שֶׁבַּעַל מוּם יֵשׁ בְּמִינוֹ קָרְבָּן. אֲבָל הַמַּקְדִּישׁ רוֹבֵעַ וְנִרְבָּע וּמֻקְצֶה וְנֶעֱבָד וְאֶתְנָן וּמְחִיר הֲרֵי זֶה כְּמַקְדִּישׁ בַּעַל מוּם עוֹבֵר. וְיִרְעוּ עַד שֶׁיִּפּל בָּהֶן מוּם קָבוּעַ וְיִפָּדוּ עָלָיו. וְכֵן הַמַּקְדִּישׁ מְחֻסַּר זְמַן הֲרֵי זֶה כְּמַקְדִּישׁ בַּעַל מוּם עוֹבֵר וְאֵינוֹ לוֹקֶה כְּמוֹ שֶׁבֵּאַרְנוּ:

נִמְצְאוּ כָּל הָאִסּוּרִין לַמִּזְבֵּחַ הֲרֵי הֵן י''ד וְאֵלּוּ הֵן. בַּעַל מוּם. וְשֶׁאֵינוֹ מִן הַמֻּבְחָר. וּמְחֻסַּר אֵיבָר מִבִּפְנִים. וּטְרֵפָה. וְכִלְאַיִם. וְיוֹצֵא דֹּפֶן. וְרוֹבֵעַ. * וְנִרְבָּע. וְשֶׁהֵמִית הָאָדָם. הַנֶּעֱבָד. הַמֻּקְצֶה. הָאֶתְנָן. הַמְּחִיר. מְחֻסַּר זְמַן:

כָּל הָאֲסוּרִין לְגַבֵּי הַמִּזְבֵּחַ אוֹסְרִין בְּכָל שֶׁהֵן אֲפִלּוּ נִתְעָרֵב אֶחָד בְּרִבּוֹא נִפְסַד הַכּל וְנִפְסַל לַמִּזְבֵּחַ. וְכֻלָּן וַלְדוֹתֵיהֶן מֻתָּרִין לַמִּזְבֵּחַ חוּץ מִוְּלַד נִרְבַּעַת וְנֶעֱבֶדֶת וּמֻקְצֵית וְשֶׁהֵמִיתָה אֶת הָאָדָם שֶׁוְּלָדָן אֲסוּרִין לַמִּזְבֵּחַ כְּמוֹתָן:

בַּמֶּה דְּבָרִים אֲמוּרִים כְּשֶׁנֶּעֶבְדָה בָּהּ עֲבֵרָה אוֹ שֶׁהֵמִיתָה כְּשֶׁהָיְתָה מְעֵבֶּרֶת שֶׁהֲרֵי הַוָּלָד מָצוּי עִמָּהּ בְּעֵת שֶׁנִּפְסְלָה וְהָיָה כְּאֵיבָר מֵאֵיבָרֶיהָ. אֲבָל אִם נִתְעַבְּרָה אַחַר שֶׁנֶּעֶבְדָה בָּהּ עֲבֵרָה אוֹ אַחַר שֶׁהֵמִיתָה הֲרֵי וְלָדָהּ כָּשֵׁר לַמִּזְבֵּחַ אֲפִלּוּ נִרְבְּעָה כְּשֶׁהִיא מֻקְדֶּשֶׁת וְאַחַר כָּךְ נִתְעַבְּרָה. וְאֵין צָרִיךְ לוֹמַר אִם נִרְבְּעָה וְהִיא חֻלִּין וְאַחַר כָּךְ הִקְדִּישָׁהּ וְנִתְעַבְּרָה שֶׁוְּלָדָהּ מֻתָּר. וְכֵן אֶפְרוֹחַ בֵּיצַת טְרֵפָה מֻתָּר לַמִּזְבֵּחַ:

הַמִּשְׁתַּחֲוֶה לַקָּמָה חִטֶּיהָ מֻתָּרִין לִמְנָחוֹת שֶׁהֲרֵי נִשְׁתַּנּוּ וְנִדְמוּ לִוְלָדוֹת שֶׁל אִסּוּר מִזְבֵּחַ שֶׁהֵן מֻתָּרִין. וְכֵן בְּהֵמָה שֶׁפִּטְּמָהּ בְּכַרְשִׁינֵי עֲבוֹדָה זָרָה מֻתֶּרֶת לַמִּזְבֵּחַ שֶׁהֲרֵי נִשְׁתַּנּוּ:

לוֹקְחִין כָּל הַקָּרְבָּנוֹת מִן הָעַכּוּ''ם וְאֵין חוֹשְׁשִׁין לָהֶם לֹא מִשּׁוּם רוֹבֵעַ וְנִרְבָּע וְלֹא מִשּׁוּם מֻקְצֶה וְנֶעֱבָד. עַד שֶׁיִּוָּדַע בְּוַדַּאי שֶׁזֶּה נִפְסָל. הֲרֵי הוּא אוֹמֵר :
