וְנָתַן קֻמְצָהּ לִכְלִי שָׁרֵת הֲרֵי זוֹ נִשְׂרֶפֶת: מִזְבֵּחַ שֶׁנִּפְגַּם נִפְסְלוּ כָּל הַקָּדָשִׁים שֶׁהָיוּ שָׁם שְׁחוּטִין בַּמִּקְדָּשׁ שֶׁעֲדַיִן לֹא נִזְרַק דָּמָם. שֶׁהֲרֵי אֵין שָׁם מִזְבֵּחַ לִזְרֹק עָלָיו וְנֶאֱמַר כְּלוֹמַר תִּזְבַּח וְהוּא עוֹמֵד בְּתִקּוּנוֹ לֹא פָּגוּם: אֲבָל קָדָשִׁים חַיִּים שֶׁהָיוּ שָׁם בָּעֲזָרָה כְּשֶׁנִּפְגַּם לֹא נִפְסְלוּ. אֶלָּא כְּשֶׁיִּבָּנֶה הַמִּזְבֵּחַ יִקָּרְבוּ שֶׁאֵין בַּעֲלֵי חַיִּים נִדְחִין: הִקְדִּישׁ בְּהֵמוֹת עַד שֶׁלֹּא נִבְנָה הַמִּזְבֵּחַ. כְּשֶׁיִּבָּנֶה מַקְרִיבִין אוֹתָן שֶׁהַדָּחוּי מֵעִקָּרוֹ אֵינוֹ דָּחוּי: וְכֵן אֵין אוֹכְלִין קָדָשִׁים וְהַמִּזְבֵּחַ פָּגוּם. שֶׁנֶּאֱמַר . וְהוּא הַדִּין לְקָדָשִׁים קַלִּים שֶׁאֵין אוֹכְלִין אוֹתָן בִּירוּשָׁלַיִם וְהַמִּזְבֵּחַ פָּגוּם עַד שֶׁיִּבָּנֶה:
