הלכות סנהדרין והעונשין המסורים להם - פרק שני

] אֵין מַעֲמִידִין בְּסַנְהֶדְרִין בֵּין בִּגְדוֹלָה בֵּין בִּקְטַנָּה אֶלָּא אֲנָשִׁים חֲכָמִים וּנְבוֹנִים. מֻפְלָגִין בְּחָכְמַת הַתּוֹרָה בַּעֲלֵי דֵּעָה מְרֻבָּה. יוֹדְעִים קְצָת מִשְּׁאָר חָכְמוֹת כְּגוֹן רְפוּאוֹת וְחֶשְׁבּוֹן וּתְקוּפוֹת וּמַזָּלוֹת וְאִצְטַגְנִינוּת וְדַרְכֵי הַמְעוֹנְנִים וְהַקּוֹסְמִים וְהַמְכַשְּׁפִים וְהַבְלֵי עֲבוֹדָה זָרָה וְכַיּוֹצֵא בְּאֵלּוּ כְּדֵי שֶׁיִּהְיוּ יוֹדְעִים לָדוּן אוֹתָם. וְאֵין מַעֲמִידִין בְּסַנְהֶדְרִין אֶלָּא כֹּהֲנִים לְוִיִּים וְיִשְׂרְאֵלִים הַמְיֻחָסִים הָרְאוּיִים לְהַשִּׂיא לִכְהֻנָּה. שֶׁנֶּאֱמַר בְּדוֹמִין לְךָ בְּחָכְמָה וּבְיִרְאָה וּבְיַחַס:

וּמִצְוָה לִהְיוֹת בְּסַנְהֶדְרִין גְּדוֹלָה כֹּהֲנִים וּלְוִיִּים שֶׁנֶּאֱמַר . וְאִם לֹא מָצְאוּ אֲפִלּוּ הָיוּ כֻּלָּם יִשְׂרְאֵלִים הֲרֵי זֶה מֻתָּר:

] אֵין מַעֲמִידִין בְּכָל הַסַּנְהֶדְרִין לֹא זָקֵן מֻפְלָג בְּשָׁנִים וְלֹא סָרִיס מִפְּנֵי שֶׁיֵּשׁ בָּהֶן אַכְזָרִיּוּת. וְלֹא מִי שֶׁאֵין לוֹ בָּנִים כְּדֵי שֶׁיְּהֵא רַחֲמָן:

ואֵין מוֹשִׁיבִין מֶלֶךְ יִשְׂרָאֵל בְּסַנְהֶדְרִין שֶׁאָסוּר לַחְלֹק עָלָיו וְלַמְרוֹת אֶת דְּבָרוֹ. אֲבָל מוֹשִׁיבִין כֹּהֵן גָּדוֹל אִם הָיָה רָאוּי בְּחָכְמָה:

מַלְכֵי בֵּית דָּוִד אַף עַל פִּי שֶׁאֵין מוֹשִׁיבִין אוֹתָם בְּסַנְהֶדְרִין יוֹשְׁבִין וְדָנִים הֵם אֶת הָעָם. וְדָנִים אוֹתָם אִם יֵשׁ עֲלֵיהֶן דִּין. אֲבָל מַלְכֵי יִשְׂרָאֵל אֵין דָּנִין וְאֵין דָּנִין אוֹתָם לְפִי שֶׁאֵין נִכְנָעִים לְדִבְרֵי תּוֹרָה שֶׁמָּא תָּבוֹא מֵהֶן תַּקָּלָה:

כְּשֵׁם שֶׁבֵּית דִּין מְנֻקִּין בְּצֶדֶק כָּךְ צְרִיכִין לִהְיוֹת מְנֻקִּין מִכָּל מוּמֵי הַגּוּף. וְצָרִיךְ לְהִשְׁתַּדֵּל וְלִבְדֹּק וּלְחַפֵּשׂ שֶׁיִּהְיוּ כֻּלָּן בַּעֲלֵי שֵׂיבָה. בַּעֲלֵי קוֹמָה. בַּעֲלֵי מַרְאֶה. נְבוֹנֵי לַחַשׁ. וְשֶׁיֵּדְעוּ בְּרֹב הַלְּשׁוֹנוֹת כְּדֵי שֶׁלֹּא תְּהֵא סַנְהֶדְרִין שׁוֹמַעַת מִפִּי הַתֻּרְגְּמָן:

] בֵּית דִּין שֶׁל שְׁלֹשָׁה אַף עַל פִּי שֶׁאֵין מְדַקְדְּקִין בָּהֶן בְּכָל אֵלּוּ הַדְּבָרִים צָרִיךְ שֶׁיְּהֵא בְּכָל אֶחָד מֵהֶן שִׁבְעָה דְּבָרִים וְאֵלּוּ הֵן. חָכְמָה. וַעֲנָוָה. וְיִרְאָה. וְשִׂנְאַת מָמוֹן. וְאַהֲבַת הָאֱמֶת. וְאַהֲבַת הַבְּרִיּוֹת לָהֶן. וּבַעֲלֵי שֵׁם טוֹב. וְכָל אֵלּוּ הַדְּבָרִים מְפֹרָשִׁין הֵן בַּתּוֹרָה. הֲרֵי הוּא אוֹמֵר הֲרֵי בַּעֲלֵי חָכְמָה אָמוּר. אֵלּוּ שֶׁרוּחַ הַבְּרִיּוֹת נוֹחָה מֵהֶם. וּבַמֶּה יִהְיוּ אֲהוּבִים לַבְּרִיּוֹת בִּזְמַן שֶׁיִּהְיוּ בַּעֲלֵי עַיִן טוֹבָה. וְנֶפֶשׁ שְׁפָלָה. וְחֶבְרָתָן טוֹבָה. וְדִבּוּרָן וּמַשָּׂאָן בְּנַחַת עִם הַבְּרִיּוֹת. וּלְהַלָּן הוּא אוֹמֵר אֵלּוּ שֶׁהֵן גִּבּוֹרִים בְּמִצְוֹת וּמְדַקְדְּקִים עַל עַצְמָם וְכוֹבְשִׁין אֶת יִצְרָן עַד שֶׁלֹּא יְהֵא לָהֶן שׁוּם גְּנַאי וְלֹא שֵׁם רַע וִיהֵא פִּרְקָן נָאֶה. וּבִכְלַל אַנְשֵׁי חַיִל שֶׁיִּהְיֶה לָהֶן לֵב אַמִּיץ לְהַצִּיל עָשׁוּק מִיַּד עוֹשְׁקוֹ כָּעִנְיָן שֶׁנֶּאֱמַר . וּמָה משֶׁה רַבֵּנוּ עָנָו אַף כָּל דַּיָּן צָרִיךְ לִהְיוֹת עָנָו. כְּמַשְׁמָעוֹ. אַף מָמוֹן שֶׁלָּהֶם אֵינָן נִבְהָלִין עָלָיו. וְלֹא רוֹדְפִין לְקַבֵּץ הַמָּמוֹן. שֶׁכָּל מִי שֶׁהוּא נִבְהָל לַהוֹן חֶסֶר יְבוֹאֶנּוּ. שֶׁיִּהְיוּ רוֹדְפִין אַחַר הַצֶּדֶק מֵחֲמַת עַצְמָן בְּדַעְתָּן. אוֹהֲבִין אֶת הָאֱמֶת וְשׂוֹנְאִין אֶת הֶחָמָס וּבוֹרְחִין מִכָּל מִינֵי הֶעָוֶל: (-)

אָמְרוּ חֲכָמִים שֶׁמִּבֵּית דִּין הַגָּדוֹל הָיוּ שׁוֹלְחִים בְּכָל אֶרֶץ יִשְׂרָאֵל וּבוֹדְקִין כָּל מִי שֶׁיִּמְצָאוּהוּ חָכָם וִירֵא חָטָא וְעָנָו וְשָׁפוּי וּפִרְקוֹ נָאֶה וְרוּחַ הַבְּרִיּוֹת נוֹחָה הֵימֶנּוּ. עוֹשִׂין אוֹתוֹ דַּיָּן בְּעִירוֹ. וּמִשָּׁם מַעֲלִין אוֹתוֹ לְפֶתַח הַר הַבַּיִת וּמִשָּׁם מַעֲלִין אוֹתוֹ לְפֶתַח הָעֲזָרָה. וּמִשָּׁם מַעֲלִין אוֹתוֹ לְבֵית דִּין הַגָּדוֹל:

] בֵּית דִּין שֶׁל שְׁלֹשָׁה שֶׁהָיָה אֶחָד מֵהֶן גֵּר הֲרֵי זֶה פָּסוּל. עַד שֶׁתִּהְיֶה אִמּוֹ מִיִּשְׂרָאֵל. הָיָה אֶחָד מַמְזֵר אֲפִלּוּ שְׁלָשְׁתָּן מַמְזֵרִים הֲרֵי אֵלּוּ כְּשֵׁרִין לָדוּן. וְכֵן אִם הָיָה כָּל אֶחָד מֵהֶם סוּמָא בְּאֶחָד מֵעֵינָיו כָּשֵׁר מַה שֶּׁאֵין כֵּן בְּסַנְהֶדְרִין. אֲבָל הַסּוּמָא בִּשְׁתֵּי עֵינָיו פָּסוּל לַכּל:

אַף עַל פִּי שֶׁאֵין בֵּית דִּין פָּחוֹת מִשְּׁלֹשָׁה מֻתָּר ] לְאֶחָד לָדוּן מִן הַתּוֹרָה שֶׁנֶּאֱמַר . וּמִדִּבְרֵי סוֹפְרִים עַד שֶׁיִּהְיוּ שְׁלֹשָׁה. וּשְׁנַיִם שֶׁדָּנוּ אֵין דִּינֵיהֶן דִּין:

אֶחָד שֶׁהָיָה מֻמְחֶה לָרַבִּים אוֹ שֶׁנָּטַל רְשׁוּת מִבֵּית דִּין הֲרֵי זֶה מֻתָּר לוֹ לָדוּן יְחִידִי אֲבָל אֵינוֹ חָשׁוּב בֵּית דִּין. וְאַף עַל פִּי שֶׁהוּא מֻתָּר מִצְוַת חֲכָמִים הוּא שֶׁמּוֹשִׁיב עִמּוֹ אֲחֵרִים שֶׁהֲרֵי אָמְרוּ אַל תְּהִי דָּן יְחִידִי שֶׁאֵין דָּן יְחִידִי אֶלָּא אֶחָד:

] יֵשׁ לְאָדָם לַעֲשׂוֹת דִּין לְעַצְמוֹ אִם יֵשׁ בְּיָדוֹ כֹּחַ הוֹאִיל וְכַדָּת וְכַהֲלָכָה הוּא עוֹשֶׂה אֵינוֹ חַיָּב לִטְרֹחַ וְלָבוֹא לְבֵית דִּין אַף עַל פִּי שֶׁלֹּא הָיָה שָׁם הֶפְסֵד בִּנְכָסָיו אִלּוּ נִתְאַחֵר וּבָא לְבֵית דִּין. לְפִיכָךְ אִם קִבֵּל עָלָיו בַּעַל דִּינוֹ וֶהֱבִיאוֹ לְבֵית דִּין וְדָרְשׁוּ וּמָצְאוּ שֶׁעָשָׂה כַּהֲלָכָה וְדִין אֱמֶת דָּן לְעַצְמוֹ אֵין סוֹתְרִין אֶת דִּינוֹ:

אַף עַל פִּי שֶׁבֵּית דִּין שֶׁל שְׁלֹשָׁה בֵּית דִּין שָׁלֵם הוּא כָּל זְמַן שֶׁהֵן רַבִּים הֲרֵי זֶה מְשֻׁבָּח. וּמוּטָב שֶׁיֵּחָתֵךְ הַדִּין בְּאַחַד עָשָׂר יוֹתֵר מֵעֲשָׂרָה וּצְרִיכִין שֶׁיִּהְיוּ הַיּוֹשְׁבִים כֻּלָּם שָׁם בְּבֵית דִּין תַּלְמִידֵי חֲכָמִים וּרְאוּיִין:

וְאָסוּר לְאָדָם ] חָכָם שֶׁיֵּשֵׁב בַּדִּין עַד שֶׁיֵּדַע עִם מִי יֵשֵׁב שֶׁמָּא יִצְטָרֵף עִם אֲנָשִׁים שֶׁאֵינָן הֲגוּנִים וְנִמְצָא בִּכְלַל קֶשֶׁר בּוֹגְדִים לֹא בִּכְלַל בֵּית דִּין:
