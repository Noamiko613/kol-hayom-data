כָּל אֶחָד מִשְּׁנֵיהֶן נִשְׁבָּע בִּנְקִיטַת חֵפֶץ שֶׁאֵין לוֹ בְּזֶה הַדָּבָר פָּחוֹת מֵחֶצְיוֹ וְיַחְלֹקוּ. וּשְׁבוּעָה זוֹ תַּקָּנַת חֲכָמִים הִיא כְּדֵי שֶׁלֹּא יִהְיֶה כָּל אֶחָד תּוֹפֵס בְּטַלִּיתוֹ שֶׁל חֲבֵרוֹ וְנוֹטֵל בְּלֹא שְׁבוּעָה: זֶה אוֹמֵר כֻּלָּהּ שֶׁלִּי וְזֶה אוֹמֵר חֶצְיָהּ שֶׁלִּי הָאוֹמֵר כֻּלָּהּ שֶׁלִּי יִשָּׁבַע שֶׁאֵין לוֹ בָּהּ פָּחוֹת מִשְּׁלֹשָׁה חֲלָקִים. וְהָאוֹמֵר חֶצְיָהּ שֶׁלִּי יִשָּׁבַע שֶׁאֵין לוֹ בָּהּ פָּחוֹת מֵרְבִיעַ. וְזֶה נוֹטֵל שְׁלֹשָׁה חֲלָקִים וְזֶה נוֹטֵל רְבִיעַ . מִכָּאן אַתָּה לָמֵד לְכָל הַנִּשְׁבָּעִין לִטּל בֵּין שְׁבוּעָה קַלָּה בֵּין שְׁבוּעָה חֲמוּרָה שֶׁאֵינוֹ נִשְׁבָּע עַל מַה שֶּׁטּוֹעֵן אֶלָּא עַל מַה שֶּׁנּוֹטֵל אַף עַל פִּי שֶׁטּוֹעֵן יוֹתֵר: הָיוּ שְׁנַיִם אֲדוּקִין בְּטַלִּית זֶה אוֹמֵר כֻּלָּהּ שֶׁלִּי וְזֶה אוֹמֵר כֻּלָּהּ שֶׁלִּי. זֶה נוֹטֵל עַד מָקוֹם שֶׁיָּדוֹ מַגַּעַת וְזֶה נוֹטֵל עַד מָקוֹם שֶׁיָּדוֹ מַגַּעַת וְהַשְּׁאָר חוֹלְקִין בְּשָׁוֶה אַחַר שֶׁנִּשְׁבָּעִין. וְיֵשׁ לְכָל אֶחָד לְגַלְגֵּל עַל חֲבֵרוֹ שֶׁכָּל מַה שֶּׁנָּטַל כַּדִּין נָטַל: הָיָה זֶה אוֹחֵז בַּחוּטִין שֶׁבִּשְׂפַת הַטַּלִּית וְזֶה בַּחוּטִין שֶׁבִּשְׂפַת הָאַחֶרֶת חוֹלְקִין כֻּלָּהּ בְּשָׁוֶה אַחַר שֶׁנִּשְׁבָּעִין. וְכָל
