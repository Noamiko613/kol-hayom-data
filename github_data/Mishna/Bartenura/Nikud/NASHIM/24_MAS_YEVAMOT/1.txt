פרק א

חֲמֵשׁ עֶשְׂרֵה נָשִׁים פּוֹטְרוֹת צָרוֹתֵיהֶן. שֶׁאִם הָיְתָה אַחַת מֵהֶן נְשׂוּאָה לְאָחִיו וְלוֹ שְׁתֵּי נָשִׁים וּמֵת בְּלֹא בָנִים, שְׁתֵּיהֶן פְּטוּרוֹת, שֶׁנֶּאֱמַר (וַיִּקְרָא יח) וְאִשָּׁה אֶל אֲחוֹתָהּ לֹא תִקַּח לִצְרֹר לְגַלּוֹת עֶרְוָתָהּ עָלֶיהָ, שֶׁאֵין תַּלְמוּד לוֹמַר עָלֶיהָ, אֶלָּא לְלַמֶּדְךָ שֶׁאֲפִלּוּ יְבִמְתּוֹ שֶׁנֶּאֱמַר בָּהּ (דְּבָרִים כה) יְבָמָהּ יָבֹא עָלֶיהָ, אִם הִיא אִשָּׁה אֶל אֲחוֹתָהּ, לֹא תִקַּח. וְהוּא הַדִּין לִשְׁאָר עֲרָיוֹת שֶׁיֵּשׁ בָּהֶן כָּרֵת. וְאֵין לִי אֶלָּא הִיא, צָרָתָהּ מִנַּיִן, תַּלְמוּד לוֹמַר לִצוֹר. צָרַת צָרָתָהּ מִנַּיִן, תַּלְמוּד לוֹמַר לִצְרֹר. דְּמַשְׁמַע לֹא תִקַּח לֹא הִיא וְלֹא צָרָתָהּ וְלֹא צָרַת צָרָתָהּ: בִּתּוֹ וּבַת בִּתּוֹ וְכוּ'. בְּבִתּוֹ מֵאֲנוּסָתוֹ, וְכֵן בַּת בִּתּוֹ וּבַת בְּנוֹ, דְּאִלּוּ בִּתּוֹ מֵאִשְׁתּוֹ הַיְנוּ בַּת אִשְׁתּוֹ. דְּכֵיוָן דִּכְתִיב (וַיִּקְרָא יח) עֶרְוַת אִשָּׁה וּבִתָּהּ, לֹא שְׁנָא הֵימֶנּוּ וְלֹא שְׁנָא מֵאִישׁ אַחֵר. אֲבָל בִּתּוֹ מֵאֲנוּסָתוֹ לֹא נָפֵיק מֵהַאי קְרָא, דְּאִשָּׁה עַל יְדֵי קִדּוּשִׁין מַשְׁמַע, אֶלָּא נָפְקָא מֵעֶרְוַת בַּת בִּנְךָ אוֹ בַּת בִּתְּךָ (שָׁם): חֲמוֹתוֹ וְאֵם חֲמוֹתוֹ וְאֵם חָמִיו. כָּל אֵלּוּ אֲסוּרוֹת לוֹ מִשּׁוּם עֶרְוַת אִשָּׁה וּבִתָּהּ לֹא תְגַלֶּה אֶת בַּת בְּנָהּ וְאֶת בַּת בִּתָּהּ, וְיֵשׁ בִּכְלַל זֶה חֲמוֹתוֹ וְאֵם חֲמוֹתוֹ וְאֵם חָמִיו: אֲחוֹתוֹ מֵאִמּוֹ. שֶׁנִּשֵּׂאת לְאָחִיו מֵאָבִיו וּמֵת. וְכֵן אֲחוֹת אִמּוֹ. שֶׁאֵין יִבּוּם אֶלָּא בְּאָחִיו מִן הָאָב, דְּיָלְפִינַן אַחֲוָה אַחֲוָה מִבְּנֵי יַעֲקֹב , מַה לְּהַלָּן מִן הָאָב אַף עַל פִּי שֶׁאֵין מִן הָאֵם, הָכָא נַמִּי לֹא שְׁנָא: וְאֵשֶׁת אָחִיו מֵאִמּוֹ. שֶׁמֵּת וּנְשָׂאָהּ אָחִיו מֵאָבִיו, שֶׁהָיְתָה נָכְרִית אֶצְלוֹ, וּמֵת בְּלֹא בָנִים וְנָפְלָה לְפָנָיו לְיִבּוּם. וַאֲסוּרָה לוֹ לְפִי שֶׁהָיְתָה תְּחִלָּה אֵשֶׁת אָחִיו מֵאִמּוֹ וַאֲסוּרָה לוֹ עוֹלָמִית, דִּכְתִיב (שָׁם) עֶרְוַת אֵשֶׁת אָחִיךָ, וְדָרְשִׁינַן אָחִיךָ בֵּין מִן הָאָב בֵּין מִן הָאֵם: וְאֵשֶׁת אָחִיו שֶׁלֹּא הָיָה בְעוֹלָמוֹ. כְּגוֹן רְאוּבֵן שֶׁמֵּת בְּלֹא בָנִים, וְנוֹלַד לוֹ אָח לְאַחַר מִיתָה וּשְׁמוֹ לֵוִי , וְיִבֵּם שִׁמְעוֹן אֶת אִשְׁתּוֹ וְלוֹ אִשָּׁה אַחֶרֶת, וּמֵת בְּלֹא בָנִים, וְנָפְלוּ שְׁתֵּיהֶן לִפְנֵי לֵוִי, שְׁתֵּיהֶן פְּטוּרוֹת, לְפִי שֶׁאֵשֶׁת רְאוּבֵן שֶׁנִּשֵּׂאת לְשִׁמְעוֹן אֲסוּרָה לְלֵוִי בְּכָרֵת, לְפִי שֶׁמִּעֲטָהּ הַכָּתוּב מִיִּבּוּם לֵוִי, שֶׁנֶּאֱמַר (דְּבָרִים כה) כִּי יֵשְׁבוּ אַחִים יַחְדָּו, שֶׁהָיְתָה לָהֶם יְשִׁיבָה אַחַת בָּעוֹלָם . הִלְכָּךְ כְּשֶׁמֵּת רְאוּבֵן, נֶאֶסְרָה עַל לֵוִי אִסּוּר עוֹלָם כְּאֵשֶׁת אָח שֶׁיֵּשׁ לוֹ בָּנִים. וְאַף עַל פִּי שֶׁעַתָּה נָפְלָה מִכֹּחַ נִשּׂוּאֵי שִׁמְעוֹן שֶׁהָיָה בְּעוֹלָמוֹ, אֲסוּרָה לוֹ מֵחֲמַת רְאוּבֵן. וּכְשֵׁם שֶׁהִיא אֲסוּרָה, כָּךְ צָרָתָהּ אֲסוּרָה: וְכַלָּתוֹ. שֶׁמֵּת בְּנוֹ, וּנְשָׂאָהּ אָחִיו. וְכַלָּתוֹ אֲסוּרָה עָלָיו אִסּוּר עוֹלָם וַאֲפִלּוּ אַחַר שֶׁמֵּת בְּנוֹ: וְכֻלָּן אִם מֵתוּ. כְּגוֹן שֶׁמֵּתָה בִּתּוֹ קֹדֶם מִיתַת אָחִיו אוֹ שֶׁמֵּאֲנָה בּוֹ. וְאַף עַל פִּי שֶׁאֵין מֵאוּן אֶלָּא בִּיתוֹמָה קְטַנָּה שֶׁהִשִּׂיאוּהָ אִמָּהּ וְאַחֶיהָ, מַשְׁכַּחַת לָהּ שֶׁמֵּאֲנָה בְּחַיֵּי אָבִיהָ, כְּגוֹן שֶׁהִשִּׂיאָהּ אָבִיהָ לְאַחֵר וְנִתְגָּרְשָׁה בְּעוֹדָהּ קְטַנָּה, שֶׁשּׁוּב אֵין לְאָבִיהָ רְשׁוּת בָּהּ, וְנִשֵּׂאת לַאֲחִי אָבִיהָ בְּעוֹדָהּ קְטַנָּה, אָז יוֹצְאָה בְּמֵאוּן, דְּקָרִינַן לָהּ לְקַמָּן יְתוֹמָה בְּחַיֵּי הָאָב. וְאִם מֵאֲנָה בְּאָחִיו אוֹ נִתְגָּרְשָׁה מִמֶּנּוּ, אוֹ נִמְצְאָה אַיְלוֹנִית שֶׁמִּקְחָהּ מֶקַּח טָעוּת וּכְאִלּוּ לֹא הָיְתָה אֵשֶׁת אָחִיו מֵעוֹלָם , צָרָתָהּ מִתְיַבֶּמֶת: אַיְלוֹנִית. לְשׁוֹן אַיִל, הַזָּכָר מִן הַצֹּאן. וְסִימָנֶיהָ מְפֹרָשִׁים, שֶׁאֵין לָהּ שָׁדַיִם כְּנָשִׁים, וְאֵין לָהּ סִימָנִים, וְקוֹלָהּ עָבֶה כְּאִישׁ, וְאֵין לָהּ שִׁפּוּלֵי מֵעַיִם כְּנָשִׁים, כְּלוֹמַר שֶׁאוֹתוֹ מָקוֹם אֵינוֹ בּוֹלֵט מִגּוּפָהּ כְּעֶרְוַת שְׁאָר הַנָּשִׁים: וְאִי אַתָּה יָכוֹל לוֹמַר וְכוּ'. שֶׁכְּבָר יָלְדוּ מֵאַחֵר קֹדֶם שֶׁנִּשְּׂאוּ לְאָחִיו, הִלְכָּךְ לָאו אַיְלוֹנִיּוֹת נִינְהוּ. וּמֵאוּן נַמִּי לֵיכָּא, שֶׁהֲרֵי גְּדוֹלוֹת הֵן, וְאֵין מְמָאֶנֶת אֶלָּא קְטַנָּה. וְתַנָּא דִּידַן דְּלֹא חָשִׁיב אִמּוֹ בִּכְלָל הַנָּשִׁים שֶׁפּוֹטְרוֹת צָרוֹתֵיהֶן, סָבַר לָהּ כְּמַאן דְּאָמַר שֶׁאֵין אָדָם נוֹשֵׂא אֲנוּסַת אָבִיו וּמְפֻתַּת אָבִיו. אֲבָל לְדִבְרֵי הָאוֹמֵר נוֹשֵׂא אָדָם אֲנוּסַת אָבִיו וּמְפֻתַּת אָבִיו, וְהִיא הֲלָכָה, אֶפְשָׁר שֶׁתִּנָּשֵׂא אִמּוֹ לְאָחִיו מֵאָבִיו, וּכְשֶׁיָּמוּת בְּלֹא בָנִים נִמְצֵאת אִמּוֹ נוֹפֶלֶת לִפְנֵי בְּנָהּ לְיִבּוּם, וְנִמְצְאוּ שֵׁשׁ עֶשְׂרֵה נָשִׁים פּוֹטְרוֹת צָרוֹתֵיהֶן, וְאִמּוֹ אַחַת מֵהֶן. וְכֵן הֲלָכָה:

כְּשֵׁם שֶׁבִּתּוֹ פְּטוּרָה. שֶׁהֲרֵי אֲסוּרָה לוֹ, וְאֵין שָׁם אָח אֶלָּא הוּא, כָּךְ צָרָתָהּ פְּטוּרָה: הָלְכָה צָרַת בִּתּוֹ וְנִשֵּׂאת לְאָחִיו הַשֵּׁנִי. הֵיכָא דְּיֵשׁ אָח אַחֵר שֶׁשְּׁתֵּיהֶן מֻתָּרוֹת לוֹ, וְאֵינָן פְּטוּרוֹת, שֶׁהֲרֵי יֵשׁ כָּאן מִצְוַת יִבּוּם, וּמִתְיַבֶּמֶת הָאַחַת וְהַשְּׁנִיָּה פְּטוּרָה, דִּכְתִיב (דְּבָרִים כה) אֶת בֵּית אָחִיו, בַּיִת אֶחָד הוּא בּוֹנֶה וְאֵין בּוֹנֶה שְׁתֵּי בָּתִּים. וְאִם יִבֵּם אָחִיו הַשֵּׁנִי אֶת צָרַת בִּתּוֹ וְלוֹ אִשָּׁה אַחֶרֶת וּמֵת בְּלֹא בָנִים וְחָזְרוּ וְנָפְלוּ לְפָנָיו: כְּשֵׁם שֶׁצָּרַת בִּתּוֹ פְּטוּרָה. שֶׁנֶּאֶסְרָה עָלָיו מִשְּׁעַת נְפִילַת אָחִיו הָרִאשׁוֹן, כָּךְ צָרַת צָרָתָהּ, אֵשֶׁת אָחִיו הַשֵּׁנִי הָאַחֶרֶת, פְּטוּרָה, שֶׁזּוֹ פּוֹטַרְתָּהּ, שֶׁצָּרַת עֶרְוָה פּוֹטֶרֶת צָרָתָהּ: וַאֲפִלּוּ הֵם מֵאָה. אַחִים, וְהָלְכָה צָרַת צָרָתָהּ שֶׁל בִּתּוֹ וְנִתְיַבְּמָה לְאָחִיו הַשְּׁלִישִׁי, וְלוֹ אִשָּׁה אַחֶרֶת וּמֵת בְּלֹא בָּנִים, שְׁתֵּיהֶן אֲסוּרוֹת עַל זֶה. וְכֵן לְעוֹלָם: צָרָתָהּ מֻתֶּרֶת. לְהִתְיַבֵּם. שֶׁהֲרֵי בִּשְׁעַת זִקַּת יִבּוּם אֵינָהּ צָרָתָהּ: וְכָל הַיְּכוֹלָה לְמָאֵן. שֶׁהִיא קְטַנָּה הָעֶרְוָה, וִיכוֹלָה לְמָאֵן וְלֹא מֵאֲנָה, וּמֵת אָחִיו, הוֹאִיל וְקִדּוּשֶׁיהָ אֵינָן אֶלָּא מִדְּרַבָּנָן וְזִקָּה שֶׁלָּהּ אֵינָהּ אֶלָּא מִדְּרַבָּנָן, אֵינָהּ פּוֹטֶרֶת צָרָתָהּ מִן הַחֲלִיצָה. וּלְהִתְיַבֵּם אֲסוּרָה, שֶׁנִּרְאֵית כְּצָרַת עֶרְוָה:

שֵׁשׁ עֲרָיוֹת חֲמוּרוֹת מֵאֵלּוּ. וּמַה הִיא חֻמְרָא שֶׁלָּהֶן, שֶׁנְּשׂוּאוֹת לַאֲחֵרִים, וְאֵינָן יְכוֹלוֹת לְהִנָּשֵׂא לְאָחִיו שֶׁל זֶה מֵאָבִיו, וְאִם מֵתוּ בַּעֲלֵיהֶן שֶׁהֵן נָכְרִים אֵצֶל זֶה, צָרוֹתֵיהֶן מֻתָּרוֹת לִנָּשֵׂא לָזֶה, שֶׁאֵין צָרַת עֶרְוָה אֲסוּרָה אֶלָּא בְּנוֹפֶלֶת לְפָנָיו לְיִבּוּם מֵאָחִיו . וְרַבֵּנוּ מֹשֶׁה בַּר מַיְמוֹן פֵּרֵשׁ, אִם נִשְּׂאוּ בַּעֲבֵרָה לְאָחִיו, וּמֵת בְּלֹא בָנִים וְנָפְלוּ לְיִבּוּם לְפָנָיו, צָרוֹתֵיהֶן מֻתָּרוֹת, שֶׁאֵין אֵלּוּ צָרוֹת עֶרְוָה, הוֹאִיל וְלֹא הָיוּ קִדּוּשִׁין תּוֹפְסִין לְאָחִיו בְּאוֹתָן עֲרָיוֹת: אִמּוֹ. אֵינָהּ יְכוֹלָה לְהִנָּשֵׂא לְאָחִיו מֵאָבִיו, שֶׁהֲרֵי הִיא לוֹ אֵשֶׁת הָאָב, לְדִבְרֵי הָאוֹמֵר שֶׁאֲנוּסַת אָבִיו אֲסוּרָה. וּלְעֵיל פֵּרַשְׁתִּי שֶׁאֵין כֵּן הֲלָכָה, אֶלָּא אִמּוֹ פּוֹטֶרֶת צָרָתָהּ, שֶׁהִיא יְכוֹלָה לְהִנָּשֵׂא לְאָחִיו מֵאָבִיו: אֵשֶׁת אָבִיו, אֵשֶׁת אָחִיו מֵאָבִיו. וַאֲחוֹת אָבִיו, כֻּלָּן אֲסוּרוֹת לְאָחִיו כְּמוֹ שֶׁאֲסוּרוֹת עָלָיו, וְאֵין בָּהֶם צַד יִבּוּם עַל זֶה לְעוֹלָם. וְאִם נְשָׂאוּם אֲחֵרִים וְלָהֶם נָשִׁים אֲחֵרוֹת וּמֵתוּ, צָרוֹתֵיהֶן מֻתָּרוֹת לִנָּשֵׂא לָזֶה. וְכֵן אִם נָשָׂא אוֹתָהּ אָחִיו בַּעֲבֵרָה וּמֵת בְּלֹא בָנִים, צָרוֹתֵיהֶן מֻתָּרוֹת:

בֵּית שַׁמַּאי מַתִּירִין. צָרַת עֶרְוָה לְהִתְיַבֵּם לְאָחִיו. דְּלֵית לְהוּ הַךְ דְּרָשָׁה דְּלִצְרֹר: חָלְצוּ. הַצָּרוֹת מִן הָאַחִין: בֵּית שַׁמַּאי פּוֹסְלִין. הַצָּרוֹת מִן הַכְּהֻנָּה, שֶׁחֲלִיצָתָן חֲלִיצָה: וּבֵית הִלֵּל מַכְשִׁירִין. שֶׁחֲלִיצָתָן שֶׁלֹּא לְצֹרֶךְ הָיְתָה, וַהֲרֵי הִיא כְּחוֹלֶצֶת מִן נָכְרִי: נִתְיַבְּמוּ. לָאַחִין: בֵּית שַׁמַּאי מַכְשִׁירִין. אוֹתָן לַכֹּהֲנִים אִם נִתְאַלְמְנוּ מִיְבָמֵיהֶם: וּבֵית הִלֵּל פּוֹסְלִים. שֶׁנִּבְעֲלוּ לָאָסוּר לָהֶם, וְהַנִּבְעֶלֶת לָאָסוּר לָהּ עֲשָׂאָהּ זוֹנָה, וְזוֹנָה אֲסוּרָה לְכֹהֵן: לֹא נִמְנְעוּ. וְאַף עַל פִּי שֶׁבְּנֵי הַצָּרוֹת שֶׁנִּתְיַבְּמוּ כְּדִבְרֵי בֵּית שַׁמַּאי מַמְזֵרִים הֵם לְבֵית הַלֵּל, שֶׁהֲרֵי בְּאִסּוּר אֵשֶׁת אָח הֵם עֲלֵיהֶם וְאֵשֶׁת אָח בְּכָרֵת, וּבְנֵי חַיָּבֵי כְּרִיתוּת מַמְזֵרִים הֵם, אַף עַל פִּי כֵן לֹא נִמְנְעוּ בֵּית הִלֵּל מִלִּשָּׂא נָשִׁים מִבֵּית שַׁמַּאי, לְפִי שֶׁהָיוּ מוֹדִיעִים לָהֶם אוֹתָן הַבָּאוֹת מִן הַצָּרוֹת וּפוֹרְשִׁין מֵהֶן: אֵלּוּ עַל גַּבֵּי אֵלּוּ. מַשְׁאִילִים כְּלֵיהֶם אֵלּוּ לְאֵלּוּ. וּבַגְּמָרָא פָּרֵיךְ אֲהָא דִּתְנַן בֵּית שַׁמַּאי מַתִּירִין אֶת הַצָּרוֹת לָאַחִים וּבֵית הִלֵּל אוֹסְרִים, קָרֵי כָּאן לֹא תִתְגּוֹדְדוּ, לֹא תַעֲשׂוּ אֲגוּדוֹת אֲגוּדוֹת . וּמְשָׁנֵי, כְּגוֹן בֵּית דִּין אֶחָד בְּעִיר אַחַת, פְּלַג מוֹרִים כְּבֵית שַׁמַּאי וּפְלַג מוֹרִים כְּבֵית הִלֵּל. אֲבָל שְׁנֵי בָּתֵּי דִּינִים בְּעִיר אַחַת, וְכָל שֶׁכֵּן שְׁנֵי בָּתֵּי דִּינִים בִּשְׁתֵּי עֲיָרוֹת, לֵית לָן בָּהּ:
