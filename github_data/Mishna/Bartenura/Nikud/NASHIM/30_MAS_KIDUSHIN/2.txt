פרק ב

הָאִישׁ מְקַדֵּשׁ בּוֹ וּבִשְׁלוּחוֹ. בּוֹ תְחִלָּה וְאַחַר כָּךְ בִּשְׁלוּחוֹ, שֶׁמִּצְוָה בּוֹ יוֹתֵר מִבִּשְׁלוּחוֹ, דְּכִי עָסִיק גּוּפוֹ בַּמִּצְוָה מְקַבֵּל שָׂכָר טְפֵי. וְיָלְפִינַן שְׁלוּחוֹ שֶׁל אָדָם כְּמוֹתוֹ מִקְּרָא, דִּכְתִיב (שְׁמוֹת יב) וְשָׁחֲטוּ אֹתוֹ כֹּל קְהַל עֲדַת יִשְׂרָאֵל, וְכִי כָּל יִשְׂרָאֵל שׁוֹחֲטִין , אֶלָּא מִכָּאן שֶׁשְּׁלוּחוֹ שֶׁל אָדָם כְּמוֹתוֹ: הָאִישׁ מְקַדֵּשׁ אֶת בִּתּוֹ כְּשֶׁהִיא נַעֲרָה. וְכָל שֶׁכֵּן כְּשֶׁהִיא קְטַנָּה. וְהַאי דְנָקַט נַעֲרָה, אוֹרַח אַרְעָא אַשְׁמוֹעִינַן, שֶׁאָסוּר לָאָדָם לְקַדֵּשׁ אֶת בִּתּוֹ כְּשֶׁהִיא קְטַנָּה עַד שֶׁתַּגְדִּיל וְתֹאמַר בִּפְלוֹנִי אֲנִי רוֹצָה: וְאִם לָאו אֵינָהּ מְקֻדֶּשֶׁת. דְּכֵיוָן דְּאָמַר הִתְקַדְּשִׁי הִתְקַדְּשִׁי, כָּל חַד הֲוֵי קִדּוּשִׁין בְּאַפֵּי נַפְשַׁיְהוּ: עַד שֶׁיְּהֵא בְאַחַת מֵהֶן שָׁוֶה פְרוּטָה. הַאי אַחַת מֵהֶן לֹא מִתּוֹקְמָא אֶלָּא בָאַחֲרוֹנָה שֶׁבָּהֶן, דְּכִי אָמַר לָהּ הִתְקַדְּשִׁי לִי בָּזוֹ וּבָזוֹ וּבָזוֹ וְהָיְתָה אוֹכֶלֶת אַחַת אַחַת, כָּל אוֹתָן שֶׁאָכְלָה הֲווּ מִלְוֶה גַבָּהּ, וְכִי מָטְיָא אַחֲרוֹנָה שֶׁבָּהּ נִגְמָרִין הַקִּדּוּשִׁין, אִי אִית בָּהּ שָׁוֶה פְרוּטָה הֲוֵי לֵיהּ מְקַדֵּשׁ בְּמִלְוֶה וּפְרוּטָה, וְקָיְמָא לָן הַמְקַדֵּשׁ בְּמִלְוֶה וּפְרוּטָה דַּעְתָּהּ אַפְּרוּטָה וּמְקֻדֶּשֶׁת. אֲבָל אִי לֵית בָּאַחֲרוֹנָה שָׁוֶה פְרוּטָה, אַף עַל גַּב דְּאִיכָּא בְּקַמָּאֵי שָׁוֶה פְרוּטָה, כִּי מָטֵי גְמַר קִדּוּשִׁין הֲוֵי לֵיהּ מְקַדֵּשׁ בְּמִלְוֶה, וְהַמְקַדֵּשׁ בְּמִלְוֶה אֵינָהּ מְקֻדֶּשֶׁת :

שֶׁל זָהָב וְנִמְצָא שֶׁל כֶּסֶף כוּ'. דְּאִיכָּא דְנִיחָא לָהּ בְּהָא וְאִיכָּא דְנִיחָא לָהּ בְּהָא : רַבִּי שִׁמְעוֹן אוֹמֵר אִם הִטְעָהּ לְשֶׁבַח הֲרֵי זוֹ מְקֻדֶּשֶׁת. לֹא פָלִיג רַבִּי שִׁמְעוֹן אֶלָּא עַל שֶׁבָח דְּמָמוֹן, דְּמִסְּתָּמָא נִיחָא לָהּ בְּאוֹתוֹ שֶׁבָח. אֲבָל עַל שֶׁבָח דְּיוֹחֲסִין, כְּגוֹן לֵוִי וְנִמְצָא כֹהֵן, מוֹדֶה רַבִּי שִׁמְעוֹן דַּאֲפִלּוּ הִטְעַתוּ אוֹ הִטְעָה הוּא אוֹתָהּ לְשֶׁבָח אֵינָהּ מְקֻדֶּשֶׁת, דְּלֹא נִיחָא לָהּ בִּשְׁבָחוֹ מִפְּנֵי שֶׁמִּתְגָּאֶה עָלֶיהָ. וְאֵין הֲלָכָה כְּרַבִּי שִׁמְעוֹן:

שִׁפְחָה גוֹדֶלֶת. קוֹלַעַת שְׂעַר הַנָּשִׁים . פֵּרוּשׁ אַחֵר, דַּבְּרָנִית וּבַעֲלַת לָשׁוֹן, כְּמוֹ שֶׁנֶּאֱמַר (יְחֶזְקֵאל לה) וַתַּגְדִּילוּ עָלַי בְּפִיכֶם: אַף עַל פִּי שֶׁאָמְרָה בְּלִבִּי הָיָה לְהִתְקַדֵּשׁ כוּ'. דִּדְבָרִים שֶׁבַּלֵּב אֵינָם דְּבָרִים:

וְהָלַךְ וְקִדְּשָׁהּ בְּמָקוֹם אַחֵר כוּ'. דְּמַרְאֶה מָקוֹם הוּא לוֹ. וְאֵין זֶה תְנַאי:

תֵּצֵא שֶׁלֹּא בִכְתֻבָּה. דְּאָמַר אִי אֶפְשִׁי בְאִשָּׁה נַדְרָנִית. אֲבָל גִּטָּא בָּעְיָא מִסָּפֵק, כֵּיוָן דְּלֹא פֵּרֵשׁ, דִּלְמָא דַעְתֵּיהּ נַמִּי אַנַּדְרָנִית: הַפּוֹסְלִים בַּכֹּהֲנִים. מְפֹרָשׁ בִּכְתֻבּוֹת :

סִבְלוֹנוֹת. דּוֹרוֹנוֹת שֶׁדֶּרֶךְ חָתָן לִשְׁלֹחַ לַאֲרוּסָתוֹ: אֵינָהּ מְקֻדֶּשֶׁת. וְלֹא אָמְרִינַן יוֹדֵעַ הָיָה שֶׁאֵין קִדּוּשָׁיו קִדּוּשִׁין וְגָמַר וְשָׁלַח סִבְלוֹנוֹת לְשֵׁם קִדּוּשִׁין, אֶלָּא אָמְרִינַן מֵחֲמַת קִדּוּשִׁין הָרִאשׁוֹנִים שָׁלַח: וְכֵן קָטָן שֶׁקִּדֵּשׁ. וְשָׁלַח סִבְלוֹנוֹת מִשֶּׁהִגְדִּיל:

אֵינָן מְקֻדָּשׁוֹת. דְּאָמַר קְרָא (וַיִּקְרָא יח) וְאִשָּׁה אֶל אֲחֹתָהּ לֹא תִקָּח לִצְרֹר, בְּשָׁעָה שֶׁנַּעֲשׂוּ צָרוֹת זוֹ לְזוֹ אֵין לְךָ לִקּוּחִין אֲפִלּוּ בְּאַחַת מֵהֶן. וְהוּא הַדִּין לְכָל שְׁאָר עֲרָיוֹת שֶׁיֵּשׁ בָּהֶן כָּרֵת, שֶׁאֵין קִדּוּשִׁין תּוֹפְסִין בָּהֶם : וְשֶׁלָּהֶן הָיְתָה. שָׁמְעִינַן מֵהָא מַתְנִיתִין דְּהַמְקַדֵּשׁ בְּגָזֵל וַאֲפִלּוּ בְּגָזֵל דִּידָהּ אֵינָהּ מְקֻדֶּשֶׁת, וְלֹא אָמְרִינַן מִדְּקַבִּילְתֵּיהּ אַחִילְתֵּיהּ. מִדְּקָתָנֵי וְשֶׁלָּהֶן הָיְתָה וְשֶׁל שְׁבִיעִית הָיְתָה, מִשּׁוּם דְּשֶׁל שְׁבִיעִית הָיְתָה שֶׁהַפֵּרוֹת הֶפְקֵר מִשׁוּם הָכִי הַנָּשִׁים מְקֻדָּשׁוֹת, אֶלָּא שֶׁאֵין אֲחָיוֹת מְקֻדָּשׁוֹת, אֲבָל אִם לֹא הָיְתָה שְׁבִיעִית, הוֹאִיל וְשֶׁלָּהֶן הָיְתָה לֹא הָיוּ קִדּוּשִׁין תּוֹפְסִים בְּאַחַת מֵהֶן. וְדַוְקָא בְּגָזֵל שֶׁלִּפְנֵי יֵאוּשׁ בְּעָלִים אָמְרִינַן דְּלֹא הֲווּ קִדּוּשִׁין, אֲבָל הַמְקַדֵּשׁ בְּגָזֵל שֶׁלְּאַחַר יֵאוּשׁ, קִדּוּשָׁיו קִדּוּשִׁין. וְשָׁמְעִינַן נַמִּי שֶׁהַמְקַדֵּשׁ בְּפֵרוֹת שְׁבִיעִית מְקֻדֶּשֶׁת, וְלֹא אָמְרִינַן אֵינָן מָמוֹנוֹ לִזְכּוֹת בָּהֶן, אֶלָּא כֵיוָן שֶׁזָּכָה בָּהֶן מָמוֹנוֹ הֵם לְכָל דָּבָר. וְשָׁמְעִינַן נַמִּי שֶׁאִשָּׁה נַעֲשֵׂית שָׁלִיחַ לַחֲבֶרְתָּהּ, וַאֲפִלּוּ בִּמְקוֹם שֶׁנַּעֲשֵׂית לָהּ צָרָה. וְאַף עַל גַּב דִּבְכָל עֵדוּת שֶׁהָאִשָּׁה כְשֵׁרָה לָהּ אֵין הַצָּרָה כְשֵׁרָה לָהּ, שְׁלִיחוּתָא מִיהָא כֵיוָן דְּעָבְדָא עָבְדָא, דְּהָא הָכָא בַקִּדּוּשִׁין הַלָּלוּ הֵן נַעֲשׂוֹת צָרוֹת עַל יְדֵי זוֹ שֶׁקִּבַּלְתָּהּ, וְקָתָנֵי אֵין הָאֲחָיוֹת מְקֻדָּשׁוֹת, הָא נָכְרִיּוֹת מְקֻדָּשׁוֹת. וּמַתְנִיתִין מִתָּרְצָא בַגְּמָרָא אַלִיבָּא דְהִלְכְתָא הָכִי, הַמְקַדֵּשׁ אִשָּׁה וּבִתָּהּ אוֹ אִשָּׁה וַאֲחוֹתָהּ, שְׁתֵּיהֶן כְּאַחַת אֵינָן מְקֻדָּשׁוֹת, הָא אַחַת מֵאִשָּׁה וּבִתָּהּ, אַחַת מֵאִשָּׁה וַאֲחוֹתָהּ, כְּגוֹן שֶׁאָמַר לִשְׁתֵּיהֶן אַחַת מִכֶּם מְקֻדֶּשֶׁת לִי וְלֹא פֵרֵשׁ לְאֵיזוֹ מֵהֶן מְקַדֵּשׁ, וְקִבְּלָה אַחַת מֵהֶן הַקִּדּוּשִׁין עַל יְדֵי חֲבֶרְתָּהּ , אוֹ שְׁתֵּיהֶן קִבְּלוּ קִדּוּשִׁין כְּאַחַת, מְקֻדָּשׁוֹת, וּשְׁתֵּיהֶן צְרִיכוֹת גֵּט. וְאִם אָמַר הָרְאוּיָה מִכֶּם לְבִיאָה מְקֻדֶּשֶׁת לִי, אֵינָן מְקֻדָּשׁוֹת, דְּכָל אַחַת מֵהֶן אִיכָּא לְסַפּוּקָהּ בַּאֲחוֹת אִשְׁתּוֹ וְאֵין שׁוּם אַחַת מֵהֶן רְאוּיָה לְבִיאָה. וּמַעֲשֶׂה נַמִּי בְחָמֵשׁ נָשִׁים וּבָהֵן שְׁתֵּי אֲחָיוֹת וְלִקֵּט אֶחָד כַּלְכָּלָה שֶׁל תְּאֵנִים וְאָמַר הָרְאוּיָה מִכֶּם לְבִיאָה תִתְקַדֵּשׁ לִי, וְאָמְרוּ חֲכָמִים אֵין אֲחָיוֹת מִתְקַדְּשׁוֹת, הָא נָכְרִיּוֹת שֶׁהָיוּ רְאוּיוֹת לְבִיאָה, מִתְקַדְּשׁוֹת. אֲבָל אִם אָמַר כֻּלְּכֶם מִתְקַדְּשׁוֹת לִי, לֹא הָיְתָה שׁוּם אַחַת מֵהֶן מְקֻדֶּשֶׁת, דְּכִי הֵיכִי דְאֵין אֲחָיוֹת מְקֻדָּשׁוֹת כָּךְ אֵין הָאֲחֵרוֹת מְקֻדָּשׁוֹת הֵיכָא דְאָמַר כֻּלְּכֶם :

הַמְקַדֵּשׁ בְּחֶלְקוֹ. שֶׁחָלַק עִם אֶחָיו הַכֹּהֲנִים: אֵינָהּ מְקֻדֶּשֶׁת. דְּכֹהֲנִים מִשֻּׁלְחָן גָּבוֹהַּ קָא זָכוּ, וְאָמַר קְרָא (בְּמִדְבָּר יח) זֶה יִהְיֶה לְךָ מִקֹּדֶשׁ הַקָּדָשִׁים מִן הָאֵשׁ, מָה אֵשׁ אֵין אַתָּה מִשְׁתַּמֵּשׁ בּוֹ אֶלָּא לַאֲכִילָה , כָּךְ מַתָּנוֹת הַלָּלוּ לֹא תִשְׁתַּמֵּשׁ בָּהֶן אֶלָּא לַאֲכִילָה, וּבְמַעֲשֵׂר כְּתִיב לַה' הוּא, בַּהֲוָיָתוֹ יְהֵא : רַבִּי יְהוּדָה אוֹמֵר מֵזִיד קִדֵּשׁ. בְּמַעֲשֵׂר שֵׁנִי. הוֹאִיל וְיוֹצֵא לְחֻלִּין עַל יְדֵי פִדְיוֹן, וַהֲרֵי הוֹצִיאוֹ לְחֻלִּין עַל יְדֵי קִדּוּשִׁין הַלָּלוּ. וְרַבִּי מֵאִיר סָבַר לֵית דֶּרֶךְ חִלּוּל בְּכָךְ: וּבְהֶקְדֵּשׁ. שֶׁל בֶּדֶק הַבַּיִת: מֵזִיד קִדֵּשׁ. שֶׁכֵּיוָן שֶׁיָּדַע שֶׁהוּא הֶקְדֵּשׁ וְהוֹצִיאוֹ לַחֻלִּין בְּכַוָּנָה, נִתְחַלְלָה קְדֻשָּׁתוֹ : וּבְשׁוֹגֵג. דְּלֹא יָדַע שֶׁהוּא הֶקְדֵּשׁ וְלֹא נִיחָא לֵיהּ דְּלִתְחַלֵּל הֶקְדֵּשׁ עַל יְדֵיהּ, לֹא נִתְחַלְלָה קְדֻשָּׁתוֹ, וְאֵינָהּ מְקֻדֶּשֶׁת. וְרַבִּי יְהוּדָה סָבַר אִפְּכָא. וַהֲלָכָה כְּרַבִּי מֵאִיר בְּמַעֲשֵׂר, וּכְרַבִּי יְהוּדָה בְּהֶקְדֵּשׁ:

הַמְקַדֵּשׁ בְּעָרְלָה וְכוּ' אֵינָהּ מְקֻדֶּשֶׁת. דְּכֻלְּהוּ אִסּוּרֵי הֲנָאָה נִינְהוּ. עָרְלָה, דִּכְתִיב לֹא יֵאָכֵל, אֶחָד אִסּוּר אֲכִילָה וְאֶחָד אִסּוּר הֲנָאָה בְּמַשְׁמָע: וּבְכִלְאֵי הַכֶּרֶם. דִּכְתִיב (דְּבָרִים כב) פֶּן תִּקְדַּשׁ, פֶּן תּוּקַד אֵשׁ: וּבְשׁוֹר הַנִּסְקָל. דִּכְתִיב בֵּיהּ (שְׁמוֹת כא) וְלֹא יֵאָכֵל אֶת בְּשָׂרוֹ : וּבְעֶגְלָה עֲרוּפָה. דִּכְתִיב בָּהּ כַּפָּרָה כְּקָדָשִׁים , שֶׁנֶּאֱמַר (דְּבָרִים כא) וְנִכַּפֵּר לָהֶם הַדָּם: בְּצִפֳּרֵי מְצֹרָע. דְּתַנְיָא, נֶאֱמַר מַכְשִׁיר וּמְכַפֵּר בִּפְנִים, אֲשָׁמוֹ שֶׁל מְצֹרָע שֶׁהוּא מַכְשִׁיר אוֹתוֹ לֶאֱכֹל בְּקָדָשִׁים, וְחַטָּאתוֹ שֶׁמְּכַפֵּר, וּשְׁנֵיהֶם נַעֲשִׂים בִּפְנִים. וְנֶאֱמַר מַכְשִׁיר וּמְכַפֵּר בַּחוּץ, מַכְשִׁיר, צִפֳּרֵי מְצֹרָע שֶׁמַּכְשִׁירִין הַמְצֹרָע לָבֹא אֶל תּוֹךְ הַמַּחֲנֶה, וּמְכַפֵּר, עֶגְלָה עֲרוּפָה, שֶׁנֶּאֱמַר בָּהּ וְנִכַּפֵּר לָהֶם הַדָּם, וּשְׁנֵיהֶם נַעֲשִׂים חוּץ לָעֲזָרָה. מַה מַּכְשִׁיר וּמְכַפֵּר הָאָמוּר בִּפְנִים, שֶׁהֵם אֲשָׁמוֹ וְחַטָּאתוֹ שֶׁל מְצֹרָע, עָשָׂה בוֹ מַכְשִׁיר כִּמְכַפֵּר, דְּתַרְוַיְהוּ קָדָשִׁים נִינְהוּ וַאֲסוּרִים בַּהֲנָאָה, אַף מַכְשִׁיר וּמְכַפֵּר הָאֲמוּרִים בַּחוּץ, שֶׁהֵם צִפֳּרֵי מְצֹרָע וְעֶגְלָה עֲרוּפָה, עָשָׂה בּוֹ מַכְשִׁיר כִּמְכַפֵּר לִהְיוֹת אֲסוּרִים בַּהֲנָאָה. וּמֵאֵימָתַי אֲסוּרִין צִפֳּרֵי מְצֹרָע בַּהֲנָאָה, מִשְּׁעַת שְׁחִיטָה. וְהַצִּפּוֹר הַשְּׁחוּטָה בִלְבַד הִיא שֶׁאֲסוּרָה בַּהֲנָאָה . וְעֶגְלָה עֲרוּפָה, יְרִידָתָהּ אֶל נַחַל אֵיתָן אוֹסְרַתָּהּ: וּבִשְׂעַר הַנָּזִיר. דְּאָמַר קְרָא (בְּמִדְבָּר ו) קָדֹשׁ יִהְיֶה גַּדֵּל פֶּרַע שְׂעַר רֹאשׁוֹ, גִּדּוּלוֹ פֶּרַע שְׂעַר רֹאשׁוֹ יִהְיֶה קָדוֹשׁ : וּפֶטֶר חֲמוֹר. נֶאֱמַר בּוֹ (שְׁמוֹת יג) וַעֲרַפְתּוֹ, וְנֶאֱמַר בְּעֶגְלָה עֲרוּפָה וְעָרְפוּ, מָה עֶגְלָה עֲרוּפָה אֲסוּרָה בַּהֲנָאָה, אַף פֶּטֶר חֲמוֹר אָסוּר בַּהֲנָאָה. וְאֵינוֹ אָסוּר בַּהֲנָאָה אֶלָּא אַחַר עֲרִיפָה : וּבָשָׂר בְּחָלָב. נֶאֱמַר שָׁלֹשׁ פְּעָמִים לֹא תְבַשֵּׁל גְּדִי בַּחֲלֵב אִמּוֹ, אֶחָד לְאִסּוּר אֲכִילָה, וְאֶחָד לְאִסּוּר הֲנָאָה, וְאֶחָד לְאִסּוּר בִּשּׁוּל: וּבְחֻלִּין שֶׁנִּשְׁחֲטוּ בָעֲזָרָה. דִּכְתִיב (דְּבָרִים יב) כִּי יִרְחַק מִמְּךָ הַמָּקוֹם וְזָבַחְתָּ, בְּרִחוּק מָקוֹם אַתָּה זוֹבֵחַ וְאִי אַתָּה זוֹבֵחַ בְּקֵרוּב מָקוֹם . יָכוֹל לֹא יֹאכְלֶנּוּ אֲבָל יַשְׁלִיכֶנּוּ לַכְּלָבִים, תַּלְמוּד לוֹמַר (שְׁמוֹת כב) לַכֶּלֶב תַּשְׁלִכוּן אֹתוֹ, אוֹתוֹ אַתָּה מַשְׁלִיךְ לַכֶּלֶב וְאִי אַתָּה מַשְׁלִיךְ לַכֶּלֶב חֻלִּין שֶׁנִּשְׁחֲטוּ בָּעֲזָרָה: מְכָרָן וְקִדֵּשׁ בִּדְמֵיהֶן מְקֻדֶּשֶׁת. שֶׁאֵין לְךָ דָּבָר שֶׁתּוֹפֵס אֶת דָּמָיו לִהְיוֹת כָּמוֹהוּ אֶלָּא עֲבוֹדָה זָרָה וּשְׁבִיעִית. עֲבוֹדָה זָרָה, שֶׁנֶּאֱמַר (דְּבָרִים ז) וְהָיִיתָ חֵרֶם כָּמֹהוּ, כָּל מַה שֶּׁאַתָּה מְהַוֶּה מִמֶּנּוּ הֲרֵי הוּא כָמוֹהוּ. וּשְׁבִיעִית, שֶׁנֶּאֱמַר קֹדֶשׁ הִיא, תּוֹפֶסֶת דָּמֶיהָ כְּהֶקְדֵּשׁ. וַהֲווּ עֲבוֹדָה זָרָה וּשְׁבִיעִית שְׁנֵי כְתוּבִים הַבָּאִים כְּאֶחָד, וְכָל שְׁנֵי כְתוּבִים הַבָּאִין כְּאֶחָד אֵין מְלַמְּדִין:

הַמְקַדֵּשׁ בִּתְרוּמוֹת. תְּרוּמָה גְדוֹלָה וּתְרוּמַת מַעֲשֵׂר: וּבְמַעַשְׂרוֹת. מַעֲשֵׂר רִאשׁוֹן וּמַעְשַׂר עָנִי: וּמַתָּנוֹת. הַזְּרֹעַ וְהַלְּחָיַיִם וְהַקֵּבָה: וּבְמֵי חַטָּאת. וּבְמֵי אֵפֶר חַטָּאת . וּרְאוּיִים לְמָכְרָם לַטְּמֵאִים לִקַּח מֵהֶם שְׂכַר הֲבָאָה וּשְׂכַר מִלּוּי הַמַּיִם. אֲבָל שְׂכַר הַזָּאָה וּשְׂכַר קִדּוּשׁ, וְהוּא נְתִינַת הָאֵפֶר בַּמַּיִם, אָסוּר : וַאֲפִלּוּ יִשְׂרָאֵל. הָכִי קָאָמַר, וַאֲפִלּוּ יִשְׂרָאֵל שֶׁנָּפְלוּ לוֹ תְּרוּמוֹת וּמַתָּנוֹת מִבֵּית אֲבִי אִמּוֹ כֹּהֵן, שֶׁזָּכָה בָּהֶם וְיָכוֹל לְמָכְרָם לַכֹּהֲנִים , אִם קִדֵּשׁ בָּהֶם אֶת הָאִשָּׁה, מְקֻדֶּשֶׁת. וַאֲפִלּוּ לֹא נָפְלוּ לוֹ תְרוּמוֹת אֶלָּא טְבָלִים שֶׁעֲדַיִן לֹא הוּרְמוּ, הוֹאִיל וּמוֹרִישׁוֹ כֹהֵן הָיָה עוֹמֵד לַהֲרִימָם וְהַתְּרוּמוֹת שֶׁלּוֹ, וְזֶה הַיִּשְׂרָאֵל שֶׁיָּרַשׁ אוֹתָם נַמִּי יַפְרִישׁ מֵהֶם הַתְּרוּמָה וְהִיא שֶׁלּוֹ, וְיָכוֹל לְמָכְרָהּ לַכֹּהֲנִים, דְּמַתָּנוֹת שֶׁלֹּא הוּרְמוּ כְּמִי שֶׁהוּרְמוּ דָּמוּ:
