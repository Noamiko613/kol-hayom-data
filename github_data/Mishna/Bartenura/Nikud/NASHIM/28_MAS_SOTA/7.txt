פרק ז

אֵלּוּ נֶאֱמָרִין. פָּרָשַׁת סוֹטָה. דִּכְתִיב (בְּמִדְבָּר ה) וְאָמַר אֶל הָאִשָּׁה, בְּכָל לָשׁוֹן שֶׁהוּא אוֹמֵר : וּוִדּוּי מַעֲשֵׂר. בִּעַרְתִּי הַקֹּדֶשׁ מִן הַבַּיִת כוּ', אוֹמֵר בְּכָל לָשׁוֹן שֶׁיִּרְצֶה, דִּכְתִיב (דְּבָרִים כו) וְאָמַרְתָּ לִפְנֵי ה' אֱלֹהֶיךָ, בְּכָל לָשׁוֹן שֶׁאַתָּה אוֹמֵר : קְרִיאַת שְׁמַע. דִּכְתִיב (שָׁם ו) שְׁמַע יִשְׂרָאֵל, בְּכָל לָשׁוֹן שֶׁאַתָּה שׁוֹמֵעַ : וּתְפִלָּה. צִבּוּר שֶׁמִּתְפַּלְּלִין, מִתְפַּלְּלִין בְּכָל לָשׁוֹן. אֲבָל יָחִיד אֵינוֹ מִתְפַּלֵּל אֶלָּא בִּלְשׁוֹן הַקֹּדֶשׁ : וּבִרְכַּת הַמָּזוֹן. דִּכְתִיב (שָׁם ח) וּבֵרַכְתָּ, בְּכָל לָשׁוֹן שֶׁאַתָּה מְבָרֵךְ : וּשְׁבוּעַת הָעֵדוּת וּשְׁבוּעַת הַפִּקָּדוֹן. בְּכָל לָשׁוֹן שֶׁהִשְׁבִּיעוּם וְלֹא הוֹדוּ, חַיָּבִין. שְׁבוּעַת הָעֵדוּת יָלְפִינַן לַהּ דִּכְתִיב בָּהּ (וַיִּקְרָא ה) וְשָׁמְעָה קוֹל אָלָה , בְּכָל לָשׁוֹן שֶׁאַתָּה שׁוֹמֵעַ. וּשְׁבוּעַת הַפִּקָּדוֹן גָּמַר תֶּחֱטָא תֶחֱטָא מִשְּׁבוּעַת הָעֵדוּת:

מִקְרָא בִכּוּרִים. שֶׁהָיָה קוֹרֵא מֵאֲרַמִּי אוֹבֵד אָבִי (דְּבָרִים כו) עַד סוֹף הַפָּרָשָׁה: וַחֲלִיצַת יְבָמָה. הִיא אוֹמֶרֶת (שָׁם כה) מֵאֵן יְבָמִי לְהָקִים לְאָחִיו שֵׁם בְּיִשְׂרָאֵל לֹא אָבָה יַבְּמִי, וְהוּא אוֹמֵר לֹא חָפַצְתִּי לְקַחְתָּהּ, וְאַחַר הַחֲלִיצָה אוֹמֶרֶת כָּכָה יֵעָשֶׂה לָאִישׁ: בְּרָכוֹת וּקְלָלוֹת. שֶׁאָמְרוּ יִשְׂרָאֵל בְּהַר גְּרִזִּים וּבְהַר עֵיבָל: בִּרְכַּת כֹּהֲנִים. נְשִׂיאַת כַּפַּיִם: בִּרְכוֹת כֹּהֵן גָּדוֹל. בְּיוֹם הַכִּפּוּרִים אַחַר עֲבוֹדַת הַיּוֹם שֶׁלִּפְנַי וְלִפְנִים הוּא קוֹרֵא אֶת הַפָּרָשָׁה בַּתּוֹרָה, וּמְבָרֵךְ שְׁמֹנֶה בְרָכוֹת הַמְּפֹרָשׁוֹת בְּמַתְנִיתִין: וּפָרָשַׁת הַמֶּלֶךְ. הִיא פָּרָשַׁת הַקְהֵל, כְּדִמְפָרֵשׁ בְּסוֹף פִּרְקִין: וּפָרָשַׁת עֶגְלָה עֲרוּפָה. (שָׁם כא) יָדֵינוּ לֹא שָׁפְכוּ אֶת הַדָּם הַזֶּה, כַּפֵּר לְעַמְּךָ יִשְׂרָאֵל וְגוֹ': פָּרָשַׁת מְשׁוּחַ מִלְחָמָה. (שָׁם כ) וְהָיָה כְּקָרָבְכֶם וְגוֹ' שְׁמַע יִשְׂרָאֵל אַתֶּם קְרֵבִים הַיּוֹם וְגוֹ':

כֵּיצַד. כְּמוֹ מִנַּיִן : וּלְהַלָּן הוּא אוֹמֵר. בַּבְּרָכוֹת וּקְלָלוֹת שֶׁל הַר גְּרִזִּים: מָה עֲנִיָּה הָאֲמוּרָה לְהַלָּן בִּלְשׁוֹן הַקֹּדֶשׁ. דִּכְתִיב (שָׁם כז) וְאָמְרוּ אֶל כָּל אִישׁ יִשְׂרָאֵל קוֹל רָם, וּכְתִיב הָתָם (שְׁמוֹת יט) וְהָאֱלֹהִים יַעֲנֶנּוּ בְּקוֹל, מַה לְּהַלָּן בִּלְשׁוֹן הַקֹּדֶשׁ, אַף כָּאן בִּלְשׁוֹן הַקֹּדֶשׁ:

וְעָנְתָה וְאָמְרָה כָּכָה. אֵינוֹ צָרִיךְ לְלַמְּדָהּ בִּגְזֵרָה שָׁוָה, שֶׁמֵּעַצְמָהּ הִיא לְמֵדָה, וְעָנְתָה וְאָמְרָה כָּכָה, כַּלָּשׁוֹן הַזֶּה תֹּאמַר:

שֶׁבְּשׁוֹמְרוֹן. אֵצֶל שׁוֹמְרוֹן: מִכָּאן וּמִכָּאן. עַל הֶהָרִים: בְּשִׁבְעִים לָשׁוֹן. בִּכְתָב שֶׁל שִׁבְעִים אֻמּוֹת , וְכָל הָרוֹצֶה לְלָמְדָהּ יָבֹא וְיִלְמֹד. שֶׁלֹּא יִהְיֶה פִּתְחוֹן פֶּה לָאֻמּוֹת לוֹמַר לֹא הָיָה לָנוּ מֵהֵיכָן לִלְמֹד: אֶת הָאֲבָנִים. סָתְרוּ אֶת הַמִּזְבֵּחַ לְאַחַר שֶׁהֶעֱלוּ הָעוֹלוֹת וְהַשְּׁלָמִים : וְלָנוּ בִמְקוֹמָן. בְּבֵית מְלוֹנָם בַּגִּלְגָּל, וְשָׁם הֵקִימוּ אֶת הָאֲבָנִים:

בַּמְּדִינָה אוֹמֵר אוֹתָהּ שָׁלֹשׁ בְּרָכוֹת. שֶׁמַפְסִיקִין הַכֹּהֲנִים בֵּין פָּסוּק לְפָסוּק, וְעוֹנִים הַצִּבּוּר אָמֵן: וּבַמִּקְדָּשׁ בְּרָכָה אֶחָת. לְפִי שֶׁאֵין עוֹנִים אָמֵן בַּמִּקְדָּשׁ וְאֵין כָּאן שׁוּם הֶפְסֵק : כִּכְתָבוֹ. בְּיוּ"ד הֵ"א: בְּכִנּוּיוֹ. בְּאַלֶ"ף דָּלֶ"ת. שֶׁאֵין מַזְכִּירִין הַשֵּׁם כִּכְתָבוֹ אֶלָּא בַּמִּקְדָּשׁ בִּלְבַד, שֶׁנֶּאֱמַר (שְׁמוֹת כ) בְּכָל הַמָּקוֹם אֲשֶׁר אַזְכִּיר אֶת שְׁמִי אָבֹא אֵלֶיךָ וּבֵרַכְתִּיךָ, סָרֵס הַמִּקְרָא וְדָרְשֵׁהוּ, בְּכָל הַמָּקוֹם אֲשֶׁר אָבֹא אֵלֶיךָ וּבֵרַכְתִּיךָ, דְּהַיְנוּ בַּמִּקְדָּשׁ, שָׁם אַזְכִּיר אֶת שְׁמִי: כְּנֶגֶד כִּתְפֵיהֶן. מִפְּנֵי שֶׁצָּרִיךְ נְשִׂיאַת כַּפַּיִם , דִּכְתִיב וַיִּשָּׂא אַהֲרֹן אֶת יָדָיו אֶל הָעָם וַיְבָרְכֵם, וּכְתִיב (דְּבָרִים יח) הוּא וּבָנָיו כָּל הַיָּמִים , מָה הוּא בִּנְשִׂיאוּת כַּפַּיִם, אַף בָּנָיו בִּנְשִׂיאַת כַּפַּיִם כָּל הַיָּמִים: וּבַמִּקְדָּשׁ. שֶׁמְּבָרְכִים אֶת הָעָם בַּשֵּׁם הַמְּפֹרָשׁ וּשְׁכִינָה לְמַעְלָה מִקִּשְׁרֵי אֶצְבְּעוֹתֵיהֶם, מַגְבִּיהִים יְדֵיהֶם לְמַעְלָה מֵרָאשֵׁיהֶם: שֶׁאֵין מַגְבִּיהַּ יָדָיו לְמַעְלָה מִן הַצִּיץ. מִפְּנֵי שֶׁהַשֵּׁם כָּתוּב בּוֹ. וְאֵין הֲלָכָה כְּרַבִּי יְהוּדָה:

בִּרְכוֹת כֹּהֵן גָּדוֹל. בְּיוֹם הַכִּפּוּרִים, לְאַחַר שֶׁכָּלְתָה עֲבוֹדַת הַיּוֹם קוֹרֵא בַתּוֹרָה וּמְבָרֵךְ שְׁמֹנֶה בְּרָכוֹת: חַזַּן הַכְּנֶסֶת. שַׁמַּשׁ הַכְּנֶסֶת שֶׁטֹּרַח עִסְקֵי הַכְּנֶסֶת עָלָיו לְהַכְנִיס וּלְהוֹצִיא לְהַפְשִׁיט אֶת הַתֵּבָה וּלְהָכִין : רֹאשׁ הַכְּנֶסֶת. עַל פִּיו נֶחְתָּכִין דִּבְרֵי הַכְּנֶסֶת, מִי יִפְרֹס עַל שְׁמַע, מִי יַפְטִיר בַּנָּבִיא, מִי יֵרֵד לִפְנֵי הַתֵּבָה: סְגָן. הַמּוּכָן תַּחַת כֹּהֵן גָּדוֹל. שֶׁאִם יֶאֱרַע בּוֹ פְּסוּל יְשַׁמֵּשׁ תַּחְתָּיו: וְהַסְּגָן לְכֹהֵן גָּדוֹל. וְכֻלָּן מִשּׁוּם כְּבוֹדוֹ שֶׁל כֹּהֵן גָּדוֹל, דִּכְתִיב (מִשְׁלֵי יד) בְּרָב עָם הַדְרַת מֶלֶךְ: וְכֹהֵן גָּדוֹל עוֹמֵד וּמְקַבֵּל. מִכְּלָל דְּיוֹשֵׁב הָיָה, שְׁמַע מִנָּהּ בְּעֶזְרַת נָשִׁים הָיָה קוֹרֵא. דְּאִי בְּעֶזְרַת יִשְׂרָאֵל קָיְמָא לָן אֵין יְשִׁיבָה בָּעֲזָרָה אֶלָּא לְמַלְכֵי בֵּית דָּוִד בִּלְבַד: וְקוֹרֵא אַחֲרֵי מוֹת. שֶׁהוּא מִסֵּדֶר יוֹם הַכִּפּוּרִים: וְאַךְ בֶּעָשׂוֹר. שֶׁבְּפָרָשַׁת שׁוֹר אוֹ כֶּשֶׂב, דִּסְמוּכָה הִיא לְפָרָשַׁת אַחֲרֵי מוֹת וְיָכוֹל לִגְלֹל סֵפֶר תּוֹרָה לְשָׁם בְּעוֹד שֶׁהַמְּתֻרְגְּמָן מְתַרְגֵּם פָּסוּק אַחֲרוֹן שֶׁל פָּרָשַׁת אַחֲרֵי מוֹת, וְאֵין כָּאן שָׁהוּת כְּדֵי שֶׁיַּפְסִיק הַמְּתֻרְגְּמָן: וְגוֹלֵל אֶת הַתּוֹרָה. וְאַף עַל פִּי שֶׁעֲדַיִן לֹא קָרָא בֶּעָשׂוֹר שֶׁל חֻמָּשׁ הַפְּקוּדִים, יִקְרָא אוֹתָהּ עַל פֶּה וְלֹא יִגְלֹל סֵפֶר תּוֹרָה לְשָׁם, מִפְּנֵי שֶׁהוּא רָחוֹק וְיֵשׁ שָׁהוּת שֶׁיַּפְסִיק הַתֻּרְגְּמָן וְאִיכָּא גְּנַאי לַצִּבּוּר. וְלִקְרוֹת בִּשְׁנֵי סִפְרֵי תּוֹרָה אִי אֶפְשָׁר, לְפִי שֶׁאֵין אָדָם אֶחָד קוֹרֵא בִּשְׁנֵי סִפְרֵי תּוֹרָה בַּצִּבּוּר, מִשּׁוּם פְּגָמוֹ שֶׁל סֵפֶר תּוֹרָה הָרִאשׁוֹן: יוֹתֵר מִמַּה שֶּׁקָּרִיתִי לִפְנֵיכֶם. מַה שֶּׁאֲנִי עָתִיד לִקְרוֹת עוֹד עַל פֶּה, אַל תֹּאמְרוּ שֶׁחָסֵר מִן הַסֵּפֶר הַזֶּה לְכָךְ אֵינִי גּוֹלְלוֹ שָׁם. וְכָל כָּךְ לָמָּה, שֶׁלֹּא לְהוֹצִיא לַעַז עַל סֵפֶר תּוֹרָה: עַל הַתּוֹרָה. בְּרָכָה שֶׁלְּאַחֲרֶיהָ: וְעַל הָעֲבוֹדָה. רְצֵה ה' אֱלֹהֵינוּ: וְעַל הַהוֹדָיָה. מוֹדִים אֲנַחְנוּ לָךְ: וְעַל מְחִילַת עָוֹן. אַתָּה בְחַרְתָּנוּ, שֶׁחוֹתֵם בָּהּ מֶלֶךְ מוֹחֵל וְסוֹלֵחַ לַעֲוֹנוֹתֵינוּ: וְעַל הַמִּקְדָּשׁ. שֶׁתִּשְׁרֶה שְׁכִינָה בְּבֵית הַמִּקְדָּשׁ, וְחוֹתֵם בָּהּ בָּרוּךְ אַתָּה ה' שׁוֹכֵן בְּצִיּוֹן: וְעַל הַכֹּהֲנִים. שֶׁיְּקֻבַּל קָרְבְּנוֹתֵיהֶם בְּרָצוֹן. וְחוֹתֵם בָּהּ מְקַדֵּשׁ הַכֹּהֲנִים: וְעַל שְׁאָר הַתְּפִלָּה. שֶׁיְּהוּ יִשְׂרָאֵל נוֹשָׁעִין מֵאוֹיְבֵיהֶם. וְחוֹתֵם בָּהּ עַמְּךָ יִשְׂרָאֵל צְרִיכִים יְשׁוּעָה בָּרוּךְ אַתָּה ה' שׁוֹמֵעַ תְּפִלָּה:

בַּשְּׁמִינִי. בְּשָׁנָה שְׁמִינִית שֶׁל שְׁמִיטָה שֶׁעָבְרָה, שֶׁהִיא מוֹצָאֵי שְׁבִיעִית זוֹ. בְּמוֹצָאֵי יוֹם טוֹב רִאשׁוֹן, דִּכְתִיב (דְּבָרִים לא) בְּחַג הַסֻּכּוֹת, שׁוֹמֵעַ אֲנִי אֲפִלּוּ בְּיוֹם טוֹב אַחֲרוֹן שֶׁל חָג, תַּלְמוּד לוֹמַר שָׁם בְּבֹא כָּל יִשְׂרָאֵל לֵרָאוֹת, מִתְּחִלַּת חֹל הַמּוֹעֵד : אַגְרִיפָּס הַמֶּלֶךְ. מִזַּרְעוֹ שֶׁל הוֹרוֹדוּס הָיָה, וּבְיָמָיו נֶחְרַב בַּיִת שֵׁנִי: זָלְגוּ עֵינָיו דְּמָעוֹת. שֶׁהַמִּקְרָא הַזֶּה פְּסָלוֹ מִן הַמַּלְכוּת: אָחִינוּ אָתָּה. שֶׁאִמּוֹ מִיִּשְׂרָאֵל: וְקוֹרֵא מִתְּחִלַּת אֵלֶּה הַדְּבָרִים עַד שְׁמַע. שֶׁבְּוָאֶתְחַנַּן. וּמְחַבֵּר עִם פָּרָשַׁת שְׁמַע פָּרָשַׁת וְהָיָה אִם שָׁמֹעַ, שֶׁיֵּשׁ בְּפָרָשַׁת שְׁמַע קַבָּלַת מַלְכוּת שָׁמַיִם וּבְפָרָשַׁת וְהָיָה אִם שָׁמֹעַ קַבָּלַת עֹל מִצְוֹת. וּמְדַלֵּג, וְקוֹרֵא עַשֵּׂר תְּעַשֵּׂר. וּמִשָּׁם מְדַלֵּג וְקוֹרֵא כִּי תְכַלֶּה לַעְשֵׂר, מִפְּנֵי שֶׁהוּא זְמַן אָסִיף וּמַתְּנוֹת עֲנִיִּים וְהַפְרָשַׁת תְּרוּמוֹת וּמַעַשְׂרוֹת. וְאַף עַל פִּי שֶׁפָּרָשַׁת הַמֶּלֶךְ בֵּין עַשֵּׂר לְכִי תְכַלֶּה לַעְשֵׂר הָיָה קוֹרֵא אוֹתָם יַחַד , שֶׁלֹּא לְהַפְסִיק בְּמַעַשְׂרוֹת. וְאַחַר כָּךְ בְּרָכוֹת וּקְלָלוֹת, שֶׁהֵן קַבָּלַת גְּזֵרוֹת וָעֳנָשִׁים שֶׁל תּוֹרָה. וְחוֹזֵר וְקוֹרֵא פָּרָשַׁת הַמֶּלֶךְ , שׂוֹם תָּשִׂים עָלֶיךָ מֶלֶךְ:
