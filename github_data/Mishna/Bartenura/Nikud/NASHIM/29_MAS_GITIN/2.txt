פרק ב - משנה א
הַמֵּבִיא. נֶחְתַּם חֶצְיוֹ. אֶחָד מִן הָעֵדִים חָתַם בְּפָנַי: בְּפָנַי נִכְתַּב חֶצְיוֹ וְכוּ' פָּסוּל. וְדַוְקָא חֶצְיוֹ הָאַחֲרוֹן. אֲבָל בְּפָנַי נִכְתַּב חֶצְיוֹ הָרִאשׁוֹן שֶׁהוּא שֵׁם הָאִישׁ וְהָאִשָּׁה וְהַזְּמַן, כָּשֵׁר . וְחֶצְיוֹ הָרִאשׁוֹן נַמִּי לֹא צָרִיךְ שֶׁיִּרְאֶה הַכְּתִיבָה עַצְמָהּ, אֶלָּא אִם שָׁמַע קוֹל הַקֻּלְמוֹס בִּלְבַד עוֹבֵר עַל הַנְּיָר בִּשְׁעַת כְּתִיבָה, תּוּ לֹא צָרִיךְ: אֶחָד אוֹמֵר בְּפָנַי נִכְתָּב וְאֶחָד אוֹמֵר בְּפָנַי נֶחְתָּם פָּסוּל. בִּזְמַן שֶׁהַגֵּט יוֹצֵא מִתַּחַת יַד אֶחָד מֵהֶם, דְּאַצְרְכוּהוּ רַבָּנָן לְשָׁלִיחַ הַמֵּבִיא גֵּט לְמֵימַר תַּרְוַיְהוּ. אֲבָל אִם הַגֵּט יוֹצֵא מִתַּחַת יְדֵי שְׁנֵיהֶם, כָּשֵׁר , שֶׁשְּׁנַיִם שֶׁהֵבִיאוּ גֵּט אֵין צָרִיךְ שֶׁיֹּאמְרוּ בְּפָנֵינוּ נִכְתַּב וּבְפָנֵינוּ נֶחְתַּם : שְׁנַיִם אוֹמְרִים בְּפָנֵינוּ נִכְתָּב וְאֶחָד אוֹמֵר בְּפָנַי נֶחְתָּם פָּסוּל. בִּזְמַן שֶׁהַגֵּט יוֹצֵא מִתַּחַת יַד אֶחָד מֵהֶם . אֲבָל אִם יוֹצֵא מִתַּחַת יַד שְׁנֵיהֶם, כָּשֵׁר: וְרַבִּי יְהוּדָה מַכְשִׁיר. וַאֲפִלּוּ יוֹצֵא מִתַּחַת יָדוֹ שֶׁל אֶחָד מֵהֶם. וְאֵין הֲלָכָה כְּרַבִּי יְהוּדָה:

פרק ב - משנה ב
בַּלַּיְלָה וְנֶחְתַּם בַּיּוֹם כָּשֵׁר. שֶׁהַיּוֹם הוֹלֵךְ אַחַר הַלַּיְלָה וְאֵין זֶה מֻקְדָּם: בַּיּוֹם וְנֶחְתַּם בַּלַּיְלָה פָּסוּל. שֶׁהֲרֵי מֻקְדָּם הוּא. וְתִקְּנוּ חֲכָמִים זְמַן בְּגִטִּין, גְּזֵרָה שֶׁמָּא יִהְיֶה נָשׂוּי עִם בַּת אֲחוֹתוֹ וְתִזְנֶה עָלָיו , וְחָס עָלֶיהָ שֶׁלֹּא תֵּחָנֵק וְנוֹתֵן לָהּ גֵּט בְּלֹא זְמַן, וּכְשֶׁמְּעִידִים עָלֶיהָ בְּבֵית דִּין מוֹצִיאָה גִּטָּהּ וְאוֹמֶרֶת גְּרוּשָׁה הָיִיתִי וּפְנוּיָה בְּאוֹתָהּ שָׁעָה : וְרַבִּי שִׁמְעוֹן מַכְשִׁיר. דְּסָבַר רַבִּי שִׁמְעוֹן דַּחֲכָמִים תִּקְּנוּ זְמַן בְּגִטִּין מִשּׁוּם פֵּירֵי , שֶׁאִם לֹא יִהְיֶה זְמַן בַּגֵּט יִהְיֶה הַבַּעַל מוֹכֵר וְהוֹלֵךְ פֵּרוֹת נִכְסֵי מְלוֹג שֶׁל אִשְׁתּוֹ לְאַחַר גֵּרוּשִׁין, וּכְשֶׁתִּתְבְּעוֹ בַּדִּין יֹאמַר קֹדֶם גֵּרוּשִׁין מָכַרְתִּי. וּלְהָכִי מַכְשִׁיר רַבִּי שִׁמְעוֹן בְּנִכְתַּב בַּיּוֹם וְנֶחְתַּם בַּלַּיְלָה אַף עַל פִּי שֶׁהוּא מֻקְדָּם, דְּסָבַר מִשָּׁעָה שֶׁנָּתַן עֵינָיו לְגָרֵשׁ אַף עַל פִּי שֶׁלֹּא גֵּרֵשׁ שׁוּב אֵין לוֹ לַבַּעַל פֵּרוֹת. וְאֵין הֲלָכָה כְּרַבִּי שִׁמְעוֹן:

פרק ב - משנה ג
בִּדְיוֹ. אלמיד"ר בַּעֲרָבִי: סִקְרָא. צֶבַע אָדֹם: קוֹמוֹס. שְׂרַף הָאִילָן: קַנְקַנְתּוֹם. וידריאול"ו בְּלַעַ"ז: עַל הֶעָלֶה שֶׁל זַיִת. תְּלוּשָׁה: וְנוֹתֵן לָהּ אֶת הַפָּרָה. שֶׁאֵינוֹ יָכוֹל לְקַצְּצוֹ אַחַר כְּתִיבָה, דִּכְתִיב (דְּבָרִים כד) וְכָתַב לָהּ סֵפֶר כְּרִיתוּת וְנָתַן בְּיָדָהּ, מִי שֶׁאֵינוֹ מְחֻסָּר אֶלָּא כְּתִיבָה וּנְתִינָה, יָצָא זֶה שֶׁמְּחֻסָּר כְּתִיבָה קְצִיצָה וּנְתִינָה: רַבִּי יוֹסֵי הַגְּלִילִי אוֹמֵר כוּ'. דְּרַחֲמָנָא קַרְיֵהּ לְגֵט סֵפֶר, מַה סֵּפֶר מְיֻחָד שֶׁאֵין בּוֹ רוּחַ חַיִּים וְאֵינוֹ אוֹכֵל, אַף כָּל שֶׁאֵין בּוֹ רוּחַ חַיִּים וְאֵינוֹ אוֹכֵל. וְרַבָּנָן אָמְרֵי, אִי כָּתַב בַּסֵּפֶר כִּדְקְאָמַרְתְּ, הַשְׁתָּא דִּכְתִיב סֵפֶר, לִסְפִירַת דְּבָרִים הוּא דַּאֲתָא . וַהֲלָכָה כַּחֲכָמִים:

פרק ב - משנה ד
אֵין כּוֹתְבִין בִּמְחֻבָּר. מִפְּנֵי שֶׁמְּחֻסָּר קְצִיצָה: כְּתָבוֹ עַל הַמְּחֻבָּר וְכוּ'. הָכִי קָאָמַר, כָּתַב בִּמְחֻבָּר הַטֹּפֶס, שֶׁהוּא כָּל הַגֵּט כֻּלּוֹ חוּץ מִמְּקוֹם הָאִישׁ וּמְקוֹם הָאִשָּׁה וְהַזְּמַן : וּתְלָשׁוֹ וַחֲתָמוֹ. כְּלוֹמַר שֶׁכָּתַב הַתּוֹרֵף שֶׁהוּא מְקוֹם הָאִישׁ וְהָאִשָּׁה וְהַזְּמַן אַחַר שֶׁתְּלָשׁוֹ: כָּשֵׁר. דְּכֵיוָן שֶׁנִּכְתַּב הַתּוֹרֵף בְּתָלוּשׁ, אַף עַל פִּי שֶׁהַטֹּפֶס נִכְתַּב בִּמְחֻבָּר, כָּשֵׁר: עַל הַנְּיָר הַמָּחוּק. שֶׁיָּכוֹל לַחֲזֹר וּלְמוֹחֲקוֹ עַד הָעֵדִים וְלִכְתֹּב עָלָיו מַה שֶּׁיִּרְצֶה, וְלֹא מִנַּכְּרָא מִלְּתָא, שֶׁהֲרֵי עֵדִים נַמִּי עַל הַמָּחוּק הֵם חֲתוּמִים : וְדִפְתְּרָא. אֵין הַמָּחוּק שֶׁלּוֹ נִכָּר. דִּפְתְּרָא, דְּמָלִיחַ וְקָמִיחַ וְלֹא עָפִיץ: וַחֲכָמִים מַכְשִׁירִין. בְּגִטִּין בִּלְבַד. דִּסְבִירָא לְהוּ עֵדֵי מְסִירָה כָּרְתֵי, וְהָעֵדִים שֶׁהַגֵּט נִמְסָר בִּפְנֵיהֶם הֵם עִקַּר הַגֵּרוּשִׁין, לֹא עֵדֵי חֲתִימָה. אֲבָל בִּשְׁאָר שְׁטָרוֹת דְּסַמְכִינַן אַעֵדֵי חֲתִימָה , מוֹדִים חֲכָמִים שֶׁאֵין נִכְתָּבִים לֹא עַל נְיָר מָחוּק וְלֹא עַל דִּפְתְּרָא. וַהֲלָכָה כַּחֲכָמִים:

פרק ב - משנה ה
וַאֲפִלּוּ חֵרֵשׁ שׁוֹטֶה וְקָטָן. וְהוּא שֶׁגָּדוֹל עוֹמֵד עַל גַּבָּיו וְאוֹמֵר לוֹ כְּתֹב לְשֵׁם פְּלוֹנִי. אֲבָל נָכְרִי וְעֶבֶד, אֲפִלּוּ גָּדוֹל עוֹמֵד עַל גַּבָּיו, לְכַתְּחִלָּה לֹא יִכְתֹּב גֵּט, מִפְּנֵי שֶׁהֵם בְּנֵי דַעַת וְאַדַּעְתָּא דְּנַפְשַׁיְהוּ עָבְדֵי, שֶׁאֲפִלּוּ גָּדוֹל אוֹמֵר לוֹ כְּתֹב לְשֵׁם פְּלוֹנִי הוּא לֹא יִכְתֹּב אֶלָּא לְדַעַת עַצְמוֹ. וְאִם נָכְרִי וְעֶבֶד כָּתְבוּ טֹפֶס הַגֵּט, וְיִשְׂרָאֵל בֶּן דַּעַת כָּתַב הַתּוֹרֵף שֶׁהוּא שֵׁם הָאִישׁ וְהָאִשָּׁה וְהַזְּמַן, שֶׁכָּל אֵלּוּ צְרִיכִין לִשְׁמָן, הַגֵּט כָּשֵׁר. וְכֵן חֵרֵשׁ שׁוֹטֶה וְקָטָן דְּאָמְרִינַן בְּמַתְנִיתִין שֶׁכְּשֵׁרִים לִכְתֹּב אֶת הַגֵּט, הָנֵי מִלֵּי טֹפֶס , אֲבָל תּוֹרֵף אֵינוֹ כָּשֵׁר אֶלָּא שֶׁיִּכְתְּבֶנּוּ יִשְׂרָאֵל גָּדוֹל וּבֶן דַּעַת: חוּץ מֵחֵרֵשׁ שׁוֹטֶה וְקָטָן. דְּלָאו בְּנֵי דֵעָה נִינְהוּ: וְסוּמָא. פָּסוּל לְהָבִיא גֵּט מֵחוּצָה לָאָרֶץ, שֶׁאֵינוֹ יָכוֹל לוֹמַר בְּפָנַי נִכְתַּב וּבְפָנַי נֶחְתָּם . אֲבָל לְהָבִיא גֵּט בְּאֶרֶץ יִשְׂרָאֵל שֶׁאֵין צָרִיךְ לוֹמַר בְּפָנַי נִכְתַּב וּבְפָנַי נֶחְתָּם, אוֹ אֲפִלּוּ בְּחוּץ לָאָרֶץ אִם הַגֵּט מְקֻיָּם בְּחוֹתְמָיו , אוֹ לִהְיוֹת שָׁלִיחַ שֶׁל אִשָּׁה לְקַבֵּל גִּטָּהּ, לְכָל אֵלּוּ הַסּוּמָא כָּשֵׁר: וְנָכְרִי. דְּלֵיתֵיהּ בְּתוֹרַת גִּטִּין וְקִדּוּשִׁין, וּבְמִידֵי דִּלְנַפְשֵׁיהּ לֹא חָזֵי לֹא מָצֵי לְמֵעֲבַד שְׁלִיחוּתָא לְאַחְרִינֵי:

פרק ב - משנה ו
קִבֵּל הַקָּטָן. הַגֵּט מִיַּד הַבַּעַל: וְהִגְדִּיל. קֹדֶם שֶׁמְּסָרוֹ לָהּ: פָּתוּחַ וְנִסְתַּמֵּא וְחָזַר וְנִתְפַּתֵּחַ. אֲפִלּוּ לֹא חָזַר וְנִתְפַּתֵּחַ, הוֹאִיל וְהָיָה פָּתוּחַ בִּשְׁעַת קַבָּלַת הַגֵּט שַׁפִּיר מָצֵי לְמֵהָוֵי שָׁלִיחַ, דְּהָא יָכוֹל לוֹמַר בְּפָנַי נִכְתַּב וּבְפָנַי נֶחְתָּם. אֶלָּא אַיְדֵי דְּבָעֵי לְמִתְנֵי סֵיפָא וְחָזַר וְנִשְׁתַּפָּה, שֶׁצָּרִיךְ שֶׁיִּהְיֶה בֶּן דַּעַת בִּשְׁעַת נְתִינָה, תָּנָא נַמִּי רֵישָׁא וְחָזַר וְנִתְפַּתֵּחַ. וְכָל פְּסוּלֵי עֵדוּת בַּעֲבֵרָה, פְּסוּלִין לְהָבִיא אֶת הַגֵּט, שֶׁאֵין נֶאֱמָנִים לוֹמַר בְּפָנַי נִכְתַּב וּבְפָנַי נֶחְתָּם. וְאִם הַגֵּט מְקֻיָּם בְּחוֹתְמָיו, כְּשֵׁרִים לַהֲבִיאוֹ :

פרק ב - משנה ז
חֲמוֹתָהּ וּבַת חֲמוֹתָהּ כוּ'. אֵינָן נֶאֱמָנוֹת לוֹמַר מֵת בַּעְלָהּ, מִשּׁוּם דְּסַנְיָן לָהּ וּמְכַוְּנִין לְקַלְקְלָהּ: וּבִלְבַד שֶׁהִיא צְרִיכָה לוֹמַר בְּפָנַי נִכְתַּב וּבְפָנַי נֶחְתָּם. וְדַוְקָא כְּשֶׁהִתְנָה הַבַּעַל עִמָּהּ בְּשָׁעָה שֶׁמָּסַר הַגֵּט לְיָדָהּ וְאָמַר לָהּ לֹא תִתְגָּרְשִׁי אֶלָּא בְּבֵית דִּינוֹ שֶׁל פְּלוֹנִי וְתֹאמְרִי בְּפָנַי נִכְתַּב וּבְפָנַי נֶחְתָּם . וְהַבֵּית דִּין לוֹקְחִין אוֹתוֹ מִיָּדָהּ אַחַר שֶׁאָמְרָה בְּפָנַי נִכְתַּב וּבְפָנַי נֶחְתָּם, וּמְמַנִּין שָׁלִיחַ שֶׁיַּחֲזֹר וְיִתֵּן אוֹתוֹ לָהּ. אֲבָל אִשָּׁה שֶׁגִּטָּהּ יוֹצֵא מִתַּחַת יָדָהּ בְּכָל מָקוֹם שֶׁהִיא, הֲרֵי זוֹ מְגֹרֶשֶׁת, וַאֲפִלּוּ שֶׁאֵין הַגֵּט מְקֻיָּם בְּחוֹתְמָיו, וְאֵינָהּ צְרִיכָה שֶׁתֹּאמַר בְּפָנַי נִכְתַּב וּבְפָנַי נֶחְתָּם :
