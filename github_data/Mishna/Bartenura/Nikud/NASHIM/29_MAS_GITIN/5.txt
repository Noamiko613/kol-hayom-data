פרק ה - משנה א
הַנִּזָּקִין שָׁמִין לָהֶם בְּעִדִּית. אַף עַל גַּב דְּמִדְּאוֹרַיְתָא הִיא , דִּכְתִיב (שְׁמוֹת כב) מֵיטַב שָׂדֵהוּ וּמֵיטַב כַּרְמוֹ יְשַׁלֵּם, הַאי תַּנָּא סָבַר דְּמֵיטַב שָׂדֵהוּ דְּנִזָּק קָאָמַר קְרָא, וּמִפְּנֵי תִקּוּן הָעוֹלָם אָמְרוּ חֲכָמִים שֶׁיְּשַׁלֵּם הַמַּזִּיק מֵעִדִּית שֶׁבִּנְכָסָיו, אַף עַל פִּי שֶׁהֵם טוֹבִים מֵעִדִּית שֶׁל נִזָּק, כְּדֵי שֶׁיְּהוּ נִזְהָרִים מִלְּהַזִּיק: וּבַעַל חוֹב בְּבֵינוֹנִית. דְּמִדְּאוֹרַיְתָא לֹא גָבֵי אֶלָּא מִזִּבּוּרִית, דִּכְתִיב (דְּבָרִים כד) וְהָאִישׁ אֲשֶׁר אַתָּה נוֹשֶׁה בוֹ יוֹצִיא אֵלֶיךָ אֶת הָעֲבוֹט הַחוּצָה, וְאֵין דַּרְכּוֹ שֶׁל אָדָם לְהוֹצִיא אֶלָּא הַפָּחוּת שֶׁבִּנְכָסָיו, וּמִפְּנֵי תִקּוּן הָעוֹלָם אָמְרוּ בַּעַל חוֹב בְּבֵינוֹנִית, כְּדֵי שֶׁלֹּא תִנְעֹל דֶּלֶת בִּפְנֵי לֹוִין : וּכְתֻבַּת אִשָּׁה בְּזִבּוּרִית. דְּלֵיכָּא לְמֵיחַשׁ בָּהּ לִנְעִילַת דֶּלֶת. דְּיוֹתֵר מִמַּה שֶּׁהָאִישׁ רוֹצֶה לִשָּׂא, אִשָּׁה רוֹצָה לְהִנָּשֵׂא : רַבִּי מֵאִיר אוֹמֵר כוּ'. וְאֵין הֲלָכָה כְּרַבִּי מֵאִיר:

פרק ה - משנה ב
אֵין נִפְרָעִים מִנְּכָסִים מְשֻׁעְבָּדִים. כְּגוֹן לֹוֶה שֶׁמָּכַר בֵּינוֹנִית שֶׁלּוֹ שֶׁהִיא מְשֻׁעְבֶּדֶת לְבַעַל חוֹב, אֵין הַבַּעַל חוֹב יָכוֹל לִטְרֹף אוֹתָהּ מִן הַלּוֹקֵחַ אִם נִשְׁאַר אֵצֶל הַלֹּוֶה נְכָסִים בְּנֵי חוֹרִין, וְאַף עַל פִּי שֶׁאֵינָן אֶלָּא זִבּוּרִית :

פרק ה - משנה ג
אֵין מוֹצִיאִין לַאֲכִילַת פֵּרוֹת וּלְשֶׁבַח קַרְקָעוֹת. הַגּוֹזֵל שָׂדֶה וּמְכָרָהּ לְאַחֵר, וּזְרָעָהּ וְצָמְחָה וְעָשְׂאָה פֵּרוֹת, וּבָא נִגְזָל וּגְבָאָהּ עִם פֵּרוֹתֶיהָ מִן הַלּוֹקֵחַ, אֵינוֹ מְשַׁלֵּם לַלּוֹקֵחַ כִּי אִם הַיְּצִיאָה, וְחוֹזֵר הַלּוֹקֵחַ עַל הַמּוֹכֵר וְגוֹבֶה דְּמֵי הַקַּרְקַע מִנְּכָסִים מְשֻׁעְבָּדִים, שֶׁהֲרֵי מְכָרָהּ לוֹ בְּאַחֲרָיוּת וְכָתַב לוֹ שְׁטַר מְכִירָה וַהֲרֵי הִיא מִלְוֶה בִשְׁטָר , וְאֶת הַפֵּרוֹת מִנְּכָסִים בְּנֵי חוֹרִין וְלֹא מִמְּשֻׁעְבָּדִים: וְכֵן לְשֶׁבַח קַרְקָעוֹת. אִם הִשְׁבִּיחָהּ הַלּוֹקֵחַ בִּנְטִיעַת הָאִילָנוֹת אוֹ בְּזֶבֶל וְכַיּוֹצֵא בָזֶה: וְלִמְזוֹן הָאִשָּׁה וְהַבָּנוֹת. דִּתְנַאי כְּתֻבָּה הוּא, וְאַתְּ תְּהֵא יָתְבָא בְּבֵיתִי וּמִתְּזְנָא מִנְּכָסַי, וּבְנָן נוּקְבִין דִּיְהָוְיָן לִיכִי מִנָּאִי יְהָוְיָן יַתְבִין בְּבֵיתִי וּמִתְּזְנִין מִנְּכָסַי וְכוּ', כְּשֶׁבָּאוֹת לִגְבּוֹת מְזוֹנוֹתֵיהֶן אֵין גּוֹבִים אֶלָּא מִנְּכָסִים בְּנֵי חוֹרִין וְלֹא מִמְּשֻׁעְבָּדִים: מִפְּנֵי תִקּוּן הָעוֹלָם. שֶׁהֵם דְּבָרִים שֶׁאֵין לָהֶם קִצְבָה וְלֹא יֵדַע כַּמָּה הֵם וְלֹא מָצֵי מִזְדַּהֵר: וְהַמּוֹצֵא מְצִיאָה. וְהֶחְזִירָהּ, וְהַבְּעָלִים אוֹמְרִים שֶׁלֹּא הֶחְזִיר כֻּלָּהּ: לֹא יִשָּׁבַע מִפְּנֵי תִקּוּן הָעוֹלָם. שֶׁאִם אַתָּה אוֹמֵר יִשָּׁבַע, אֵין לְךָ אָדָם שֶׁמְּטַפֵּל בִּמְצִיאָה לְהַחְזִירָהּ:

פרק ה - משנה ד
יְתוֹמִים שֶׁסָּמְכוּ אֵצֶל בַּעַל הַבַּיִת. לַעֲשׂוֹת מַעֲשֵׂיהֶם עַל פִּיו, וְלֹא שֶׁנִּתְמַנָּה לָהֶם אַפּוֹטְרוֹפּוֹס, אֲפִלּוּ הָכִי כְּאַפּוֹטְרוֹפּוֹס הוּא. אַפּוֹטְרוֹפּוֹס, בִּלְשׁוֹן רוֹמִי קוֹרִין לָאָב פאט"ר וְלַקְּטַנִּים פוטו"ס. וּפֵרוּשׁ אַפּוֹטְרוֹפּוֹס אֲבִיהֶן שֶׁל קְטַנִּים: שֶׁמִּנָּהוּ אֲבִי יְתוֹמִים יִשָּׁבֵעַ. דְּאִי לָאו דְּאִית לֵיהּ הַנָּאָה מִנֵּיהּ לֹא הֲוָה לֵיהּ אַפּוֹטְרוֹפּוֹס, וּמִשּׁוּם שְׁבוּעָה לֹא אָתֵי לְאַמְּנוּעֵי: מִנָּהוּ בֵית דִּין לֹא יִשָּׁבֵעַ. דְּטוֹבַת חִנָּם הִיא שֶׁזֶּה עוֹשֶׂה לְבֵית דִּין לְקַבֵּל דִּבְרֵיהֶם וְלִטְרֹחַ חִנָּם, וְאִי רָמִית עֲלֵיהּ שְׁבוּעָה אָתֵי לְאִמָּנוּעֵי: אַבָּא שָׁאוּל אוֹמֵר חִלּוּף הַדְּבָרִים. מִנָּהוּ בֵּית דִּין יִשָּׁבֵעַ, דִּבְהַהִיא הֲנָאָה דְקָא נָפֵיק עֲלֵיהּ קָלָא דֶאֱינַשׁ מְהֵימָנָא הוּא דְּהָא סָמְכֵי עֲלֵיהּ בֵּי דִינָא, מִשּׁוּם שְׁבוּעָה לֹא אָתֵי לְאִמָּנוּעֵי. מִנָּהוּ אֲבִי יְתוֹמִים לֹא יִשָּׁבֵעַ, דְּטוֹבַת חִנָּם עוֹשֶׂה לוֹ לִטְרֹחַ עִם בָּנָיו, וְאִי רָמִית עֲלֵיהּ שְׁבוּעָה אָתֵי לְאִמָּנוּעֵי. וַהֲלָכָה כְּאַבָּא שָׁאוּל: הַמְטַמֵּא. טָהֳרוֹתָיו שֶׁל חֲבֵרוֹ: וְהַמְדַמֵּעַ. מְעָרֵב תְּרוּמָה בְּחֻלִּין שֶׁל חֲבֵרוֹ, וּמַפְסִידוֹ, שֶׁצָּרִיךְ לְמָכְרוֹ לַכֹּהֲנִים בְּזוֹל: וְהַמְנַסֵּךְ. שֶׁמְּעָרֵב יֵין נֶסֶךְ בְּיַיִן כָּשֵׁר וְאוֹסְרוֹ בַּהֲנָאָה: בְּמֵזִיד חַיָּב. וּבְדִין הוּא שֶׁיְּהֵא פָטוּר, דְּהֶזֵּק שֶׁאֵינוֹ נִכָּר לָאו שְׁמֵיהּ הֶזֵּק , אֶלָּא מִפְּנֵי תִּקּוּן הָעוֹלָם, שֶׁלֹּא יְהֵא כָּל אֶחָד וְאֶחָד הוֹלֵךְ וּמְטַמֵּא טָהֳרוֹתָיו שֶׁל חֲבֵרוֹ וְאוֹמֵר פָּטוּר אֲנִי: הַכֹּהֲנִים שֶׁפִּגְלוּ. קָרְבָּנוֹת, שֶׁשָּׁחֲטוּ וְזָרְקוּ דָמָם בְּמַחֲשֶׁבֶת לֶאֱכֹל מֵהֶם חוּץ לִזְמַנּוֹ. וְלֹא הוּרְצוּ לַבְּעָלִים: מְזִידִים. שֶׁיָּדְעוּ שֶׁפְּסוּלִים בְּכָךְ: חַיָּבִים. לְשַׁלֵּם דְּמֵיהֶן לַבְּעָלִים, שֶׁהֲרֵי צְרִיכִין לְהָבִיא אֲחֵרִים, וְאִי נַמִּי נְדָבָה הִיא שֶׁאֵינוֹ חַיָּב בְּתַשְׁלוּמִין, מִכָּל מָקוֹם קָשֶׁה בְּעֵינָיו שֶׁלֹּא הִקְרִיב קָרְבָּנוֹ שֶׁהֲרֵי לְהָבִיא דּוֹרוֹן הָיָה מְבַקֵּשׁ:

פרק ה - משנה ה
עַל הַחֵרֶשֶׁת שֶׁהִשִּׂיאָהּ אָבִיהָ. וְאַף עַל גַּב דְּהַוְיָא אֵשֶׁת אִישׁ גְּמוּרָה שֶׁהֲרֵי קִבֵּל אָבִיהָ קִדּוּשֶׁיהָ כְּשֶׁהִיא קְטַנָּה, אֲפִלּוּ הָכִי יוֹצְאָה בְּגֵט, וּמְקַבֶּלֶת אֶת גִּטָּהּ כְּשֶׁהִיא חֵרֶשֶׁת, וְאַף עַל גַּב דְּלֵית בַּהּ דַּעְתָּא, דְּאִשָּׁה מִתְגָּרֶשֶׁת בְּעַל כָּרְחָהּ, הִלְכָּךְ לֹא בָּעִינַן דַּעְתָּהּ: וְעַל קְטַנָּה בַת יִשְׂרָאֵל שֶׁנִּשֵּׂאת לְכֹהֵן. וְהִיא יְתוֹמָה, דְּלֹא הָוֵי נִשּׂוּאִים אֶלָּא מִדְּרַבָּנָן: שֶׁאוֹכֶלֶת בַּתְּרוּמָה. דְּרַבָּנָן . וְלֹא גָזְרִינַן תְּרוּמָה דְּרַבָּנָן אַטּוּ תְּרוּמָה דְּאוֹרַיְתָא: מָרִישׁ. קוֹרָה: בִּירָה. בַּיִת גָּדוֹל: מִפְּנֵי תַקָּנַת הַשָּׁבִים. שֶׁאִם אַתָּה מַצְרִיכוֹ לְקַעְקֵעַ בִּירָתוֹ וּלְהַחְזִיר הַמָּרִישׁ עַצְמוֹ, יִמָּנַע מִלַּעֲשׂוֹת תְּשׁוּבָה: שֶׁלֹּא נוֹדְעָה לָרַבִּים. שֶׁהִיא גְזוּלָה: שֶׁהִיא מְכַפֶּרֶת. וְאֵין צָרִיךְ לְהָבִיא אַחֶרֶת: מִפְּנֵי תִקּוּן מִזְבֵּחַ. שֶׁלֹּא יִהְיוּ כֹּהֲנִים עֲצֵבִים שֶׁאָכְלוּ חֻלִּין שֶׁנִּשְׁחֲטוּ בָּעֲזָרָה, וְנִמְצָא מִזְבֵּחַ בָּטֵל, שֶׁנִּמְנָעִים מִלַּעֲבֹד עֲבוֹדָה:

פרק ה - משנה ו
לֹא הָיָה סִיקָרִיקוֹן. נָכְרִי רוֹצֵחַ. כְּלוֹמַר לֹא דָּנוּ דִּין סִיקָרִיקוֹן, לוֹמַר שֶׁהַקּוֹנֶה קַרְקַע שֶׁל יִשְׂרָאֵל מִיַּד הַנָּכְרִי רוֹצֵחַ, יִהְיֶה חַיָּב לִדּוֹן עִם הַבְּעָלִים: בִּשְׁעַת הֲרוּגֵי הַמִּלְחָמָה. בְּשָׁעָה שֶׁהָיְתָה הַגְּזֵרָה קָשָׁה עַל יִשְׂרָאֵל לֵהָרֵג בַּמִּלְחָמָה. שֶׁהַלּוֹקֵחַ מִמֶּנּוּ בְּאוֹתָהּ שָׁעָה הָיָה מִקָּחוֹ קַיָּם וְלֹא הָיָה צָרִיךְ לִדּוֹן עִם הַיִשְׂרָאֵל בַּעַל הַקַּרְקַע, מִשּׁוּם דְּאַגַּב אוֹנְסֵיהּ דְּיִשְׂרָאֵל הֲוָה גָּמַר וּמַקְנֶה לְסִיקָרִיקוֹן. וְקָיְמָא לָן תָּלוּהוּ וְזָבִין זְבִינֵיהּ זְבִינֵי: אֲבָל מֵהֲרוּגֵי מִלְחָמָה וְאֵילָךְ. שֶׁלֹּא הָיְתָה הַגְּזֵרָה לֵהָרֵג, דָּנוּ דִּין סִיקָרִיקוֹן, לוֹמַר שֶׁהַקּוֹנֶה מִמֶּנּוּ יַעֲשֶׂה דִּין עִם הַבְּעָלִים. כְּמוֹ שֶׁמְּפֹרָשׁ בַּמִּשְׁנָה: מִקָּחוֹ בָטֵל. דְּאָמְרִינַן מִיִּרְאָה עָבַד: לָקַח מִן הָאִישׁ. קַרְקַע הַמְיֻחָד לִכְתֻבַּת אִשְׁתּוֹ : מִקָּחוֹ בָטֵל. דְּאָמְרָה נַחַת רוּחַ עָשִׂיתִי לְבַעֲלִי: נוֹתֵן לַבְּעָלִים רְבִיעַ. שֶׁשִּׁעֲרוּ דְּסִיקָרִיקוֹן מִתּוֹךְ שֶׁבְּחִנָּם בָּא לְיָדוֹ מוֹזִיל גַּבֵּיהּ רִבְעָא:

פרק ה - משנה ז
רוֹמֵז וְנִרְמָז. מַה שֶּׁהוּא רוֹמֵז אוֹ שֶׁאֲחֵרִים רוֹמְזִים לוֹ וְנִתְרַצָּה, הַכֹּל קַיָּם . רְמִיזָה, בְּיָדָיו וּבְרֹאשׁוֹ. קְפִיצָה, עֲקִימַת שְׂפָתַיִם, שֶׁנֶּאֱמַר (אִיּוֹב ה) קָפְצָה פִּיהָ. וְאֵינוֹ סִימָן נִכָּר כְּמוֹ רְמִיזָה: בְּמִטַּלְטְלִין. אִם מָכַר מִטַּלְטְלִין. וְאֵין הֲלָכָה כְּבֶן בְּתֵירָה: הַפָּעוֹטוֹת. הַקְּטַנִּים כְּבֶן שֶׁבַע כְּבֶן שְׁמֹנֶה אִם הוּא חָרִיף וְיוֹדֵעַ בְּטִיב מַשָּׂא וּמַתָּן, אוֹ כְּבֶן תֵּשַׁע וּבֶן עֶשֶׂר אִם אֵינוֹ חָרִיף כָּל כָּךְ: מִקָּחָן מִקָּח וּמִמְכָּרָן מִמְכָּר בְּמִטַּלְטְלִין. וּמַתְּנָתָן מַתָּנָה. אֶחָד מַתְּנַת בָּרִיא, וְאֶחָד מַתְּנַת שְׁכִיב מֵרַע. אֶחָד מַתָּנָה מְרֻבָּה, וְאֶחָד מַתָּנָה מֻעֶטֶת:

פרק ה - משנה ח
כֹּהֵן קוֹרֵא רִאשׁוֹן וְכוּ'. כְּשֶׁהֵן שָׁוִין בְּחָכְמָה. אֲבָל אִם הַיִּשְׂרָאֵל גָּדוֹל בְּחָכְמָה, קוֹדֵם לַכֹּהֵן וְלַלֵּוִי, דְּמַמְזֵר תַּלְמִיד חָכָם קוֹדֵם לְכֹהֵן גָּדוֹל עַם הָאָרֶץ. הֲדֵין הוּא דִינָא דִגְמָרָא. וְהָאִידְנָא נָהוּג שֶׁכֹּהֵן אֲפִלּוּ עַם הָאָרֶץ קוֹדֵם לְחָכָם גָּדוֹל שֶׁבְּיִשְׂרָאֵל: מִפְּנֵי דַרְכֵי שָׁלוֹם. דְּמִדְּאוֹרַיְתָא יָכוֹל הַכֹּהֵן לָתֵת רְשׁוּת לְמִי שֶׁיִּרְצֶה שֶׁיִּקְרָא בַּתּוֹרָה לְפָנָיו, וּמִפְּנֵי דַרְכֵי שָׁלוֹם אָמְרוּ שֶׁיִּקְרָא הוּא רִאשׁוֹן וְלֹא יִתֵּן רְשׁוּת לְאַחֵר שֶׁיִּקְרָא, דְּלֹא לֵיתֵי לְאִינְּצוֹיֵי וְלוֹמַר מִפְּנֵי מָה הִרְשָׁה לָזֶה וְלֹא לְאַחֵר. וְלֹא שְׁנָא בְּשַׁבָּתוֹת וְיָמִים טוֹבִים דִּשְׁכִיחֵי רַבִּים, וְלֹא שְׁנָא בְּשֵׁנִי וּבַחֲמִישִׁי בַּזְּמַן הַזֶּה , לְעוֹלָם כֹּהֵן קוֹרֵא רִאשׁוֹן וְאֵינוֹ יָכוֹל לָתֵת רְשׁוּת לְיִשְׂרָאֵל שֶׁיִּקְרָא לְפָנָיו, כִּי הֵיכִי דְּלֹא לֵיתוּ לְאִינְּצוֹיֵי. וְאִם אֵין שָׁם כֹּהֵן, נִתְפָּרְדָה הַחֲבִילָה, וְלֹא יִקְרָא לֵוִי שֵׁנִי אֶלָּא כְּפִי חֲשִׁיבוּתוֹ, וְאִית דְּאָמְרֵי לֹא יִקְרָא לֵוִי בַּתּוֹרָה כְלָל , וְכֵן הַמִּנְהָג: מְעָרְבִין בְּבַיִת יָשָׁן. בְּנֵי חָצֵר שֶׁרְגִילִים לִתֵּן עֵרוּבָן בְּכָל שַׁבָּת בְּבַיִת אֶחָד, אֵין מְשַׁנִּין אֶת מְקוֹמָן לִתְּנוֹ בְּבַיִת אַחֵר: מִפְּנֵי דַרְכֵי שָׁלוֹם. שֶׁבְּנֵי אָדָם שֶׁהָיוּ רְגִילִים לִרְאוֹת הָעֵרוּב בְּאוֹתוֹ בַּיִת, עַכְשָׁיו שֶׁאֵין רוֹאִים אוֹתוֹ יֹאמְרוּ שֶׁמְּטַלְטְלִין בְּלֹא עֵרוּב, וְאִיכָּא חֲשָׁדָא : בּוֹר שֶׁהוּא קָרוֹב. לְמוֹצָא אַמַּת הַמַּיִם הַבָּא מִן הַנָּהָר: מִתְמַלֵּא רִאשׁוֹן . וְאַחֲרָיו יִתְמַלְּאוּ הַתַּחְתּוֹנִים: מְצוּדוֹת חַיָּה. שֶׁאֵין לָהֶם תּוֹךְ דְּלִיקְנֵי לֵיהּ כֵּלָיו: יֵשׁ בָּהֶן גָּזֵל מִפְּנֵי דַרְכֵי שָׁלוֹם. וְלֹא נָפֵיק בְּדַיָּנִים: רַבִּי יוֹסֵי אוֹמֵר גָּזֵל גָּמוּר. מִדִּבְרֵיהֶם. וְנָפֵיק בְּדַיָּנִים . וּמִיהוּ מוֹדֶה רַבִּי יוֹסֵי דְּלֹא הָוֵי גָּזֵל דְּאוֹרַיְתָא לַעֲבֹר עָלָיו בְּלָאו. וְאֵין הֲלָכָה כְּרַבִּי יוֹסֵי: הַמְנַקֵּף. חוֹתֵךְ . כְּמוֹ וְנָקַף סִבְכֵי הַיַּעַר (יְשַׁעְיָה י):

פרק ה - משנה ט
הַחֲשׁוּדָה עַל הַשְּׁבִיעִית. לִשְׁמֹר פֵּרוֹת שְׁבִיעִית וּלְהַצְנִיעָם מִן הַבִּעוּר וָאֵילָךְ: לֹא תָבֹר וְלֹא תִטְחַן עִמָּהּ. לְסַיְּעָהּ, מִפְּנֵי שֶׁאָסוּר לְסַיֵּעַ בַּיָּדַיִם יְדֵי עוֹבְרֵי עֲבֵרָה בִּשְׁעַת הָעֲבֵרָה: לְאֵשֶׁת עַם הָאָרֶץ. הַחֲשׁוּדָה עַל הַמַּעַשְׂרוֹת: וּבוֹרֶרֶת וְטוֹחֶנֶת. לְפִי שֶׁרֹב עַמֵּי הָאָרֶץ מְעַשְּׂרִין הֵן: אֲבָל מִשֶּׁתַּטִּיל הַמַּיִם. בָּעִסָּה: לֹא תִגַּע עִמָּהּ. לְפִי שֶׁמִּשֶּׁגִּלְגְּלָה הֻטְבְּלָה לְחַלָּה, וְקָא מִטַּמְּאָה מֵחֲמַת כֵּלִים טְמֵאִים שֶׁהֲרֵי הֻכְשְׁרָה לְקַבֵּל טֻמְאָה, וְזוֹ מְסַיְּעָהּ לְגַלְגֵּל, וְאָסוּר לִגְרֹם טֻמְאָה לַחַלָּה: וְכֻלָּן לֹא אָמְרוּ כוּ'. לֹא הִתִּירוּ לְהַשְׁאִילָם כֵּלִים וּלְסַיְּעָם בְּלֹא שְׁעַת עֲבֵרָה עַצְמָהּ, אֶלָּא מִשּׁוּם דַּרְכֵי שָׁלוֹם: וּמַחֲזִיקִים יְדֵי גוֹיִם. לוֹמַר לָהֶם תֶּחֱזַקְנָה יְדֵיכֶם: וְשׁוֹאֲלִים בִּשְׁלוֹמָם. כָּל הַיָּמִים וַאֲפִלּוּ בְּיוֹם חַגָּם. וְאַף עַל פִּי שֶׁמֵּטִיל שֵׁם שָׁמַיִם עַל הַנָּכְרִי, שֶׁשָּׁלוֹם אֶחָד מִשְּׁמוֹתָיו שֶׁל הַקָּדוֹשׁ בָּרוּךְ הוּא:
