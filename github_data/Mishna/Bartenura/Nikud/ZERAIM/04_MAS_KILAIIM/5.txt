פרק ה - משנה א
כֶּרֶם שֶׁחָרַב. עֶשֶׂר גְּפָנִים לְבֵית סְאָה. שֶׁיֵּשׁ בַּקַּרְקַע שֶׁל בֵּית סְאָה עֶשֶׂר גְּפָנִים לֹא פָחוֹת מִכֵּן: וּנְטוּעוֹת כְּהִלְכָתָן. דְּהַיְנוּ שְׁתַּיִם כְּנֶגֶד שְׁתַּיִם וְאַחַת יוֹצְאָה זָנָב, וְאֵין רְחוֹקוֹת זוֹ מִזּוֹ שֵׁשׁ עֶשְׂרֵה אַמּוֹת וְלֹא פָחוֹת מֵאַרְבַּע אַמּוֹת : כֶּרֶם דַּל. שֶׁגְּפָנָיו מֻעָטִים, וְאַף עַל פִּי כֵן דִּין כֶּרֶם יֵשׁ לוֹ. וְאֵין זֶה קָרַחַת הַכֶּרֶם, דְּאוֹתוֹ חָרַב מֵאֶמְצָעוֹ וְזֶה נֶחֱרַב מִכָּל צַד: עִרְבּוּבְיָא. שֶׁאֵין הַגְּפָנִים מְכֻוָּנִים כְּשׁוּרָה: אִם יָכוֹל לְכַוֵּן. בְּכָל הַכֶּרֶם שְׁתַּיִם כְּנֶגֶד שָׁלֹשׁ, כְּגוֹן שֶׁיָּבִיא חוּט וּמוֹתֵחַ וּמוֹצֵא הַגֶּזַע מְכֻוָּן שְׁתַּיִם כְּנֶגֶד שְׁתַּיִם וְאֶחָד יוֹצֵא זָנָב, אַף עַל פִּי שֶׁהַנּוֹף אֵינוֹ מְכֻוָּן הֲרֵי זֶה כֶרֶם. אֲבָל אִם מוֹצֵא הַנּוֹף מְכֻוָּן וְהַגֶּזַע אֵינוֹ מְכֻוָּן אֵין זֶה כֶרֶם: כְּתַבְנִית כְּרָמִים הֲרֵי זֶה כָרֶם. וּבָעֵי הַרְחָקָה שֵׁשׁ עֶשְׂרֵה אַמּוֹת, וְאֵין הֲלָכָה כְּרַבִּי מֵאִיר:

פרק ה - משנה ב
כֶּרֶם שֶׁהוּא נָטוּעַ עַל פָּחוֹת מֵאַרְבַּע אַמּוֹת. שֶׁאֵין בֵּין שׁוּרָה לְשׁוּרָה אַרְבַּע אַמּוֹת: אֵינוֹ כֶרֶם. וּמַרְחִיק שִׁשָּׁה טְפָחִים וְזוֹרֵעַ אֶת הַשְּׁאָר, דְּכֵיוָן שֶׁאֵין יָכוֹל לַחֲרֹשׁ בִּשְׁוָרִים בֵּינֵיהֶם אֵין שֵׁם כֶּרֶם עָלָיו: וַחֲכָמִים אוֹמְרִים כֶּרֶם. דְּרוֹאִים אֶת הָאֶמְצָעִיִּים כְּאִלּוּ אֵינָם וּמִצְטָרְפִין הַחִיצוֹנִים אִם יֵשׁ בֵּינֵיהֶם רֶוַח הָרָאוּי לָהֶם. לְפִי שֶׁדֶּרֶךְ בְּנֵי אָדָם לִטַּע שׁוּרוֹת רְצוּפוֹת הַרְבֵּה בְּיַחַד, וְאוֹתָן שֶׁרְאוּיִין לְהִתְקַיֵּם וּלְהֵעָשׂוֹת כֶּרֶם יִשָּׁאֲרוּ כֶּרֶם, וְהַשְּׁאָר יִהְיוּ לְעֵצִים. הִלְכָּךְ אֵלּוּ הָאֶמְצָעִיִּים הַנְּטוּעִים בְּתוֹךְ הָרֶוַח הָרְאוּיִם לִהְיוֹת בֵּין שׁוּרָה לְשׁוּרָה רוֹאִים אוֹתָם כְּאִלּוּ הֵם עֵצִים בְּעַלְמָא, וְהַחִיצוֹנִים מִצְטָרְפִים לִהְיוֹת כֶּרֶם:

פרק ה - משנה ג
וְזוֹרְעִים בְּתוֹכוֹ. דְּחָשֵׁיב הֶפְסֵק וּרְשׁוּת אַחֶרֶת כֵּיוָן דְּעָמֹק עֲשָׂרָה וְרָחָב אַרְבָּעָה וּמְפֻלָּשׁ. אֲבָל פָּחוּת מֵעֲשָׂרָה אוֹ מֵאַרְבָּעָה אוֹ אֲפִלּוּ עָמֹק עֲשָׂרָה וְרָחָב אַרְבָּעָה וְאֵינוֹ מְפֻלָּשׁ אֶלָּא שֶׁהַכֶּרֶם מַקִּיפוֹ מִשָּׁלֹשׁ רוּחוֹת, אֵין זוֹרְעִים בְּתוֹכוֹ, שֶׁהֲרֵי הִיא כְּגַת שֶׁנֶּחְלְקוּ בָהּ רַבִּי אֱלִיעֶזֶר וְרַבָּנָן לְקַמָּן, וְרַבִּי אֱלִיעֶזֶר בֶּן יַעֲקֹב סָבַר כְּרַבָּנָן דְּאָסְרֵי לִזְרֹעַ בְּגַת: שׁוֹמֵרָה שֶׁבַּכֶּרֶם. תֵּל גָּבוֹהַּ שֶׁהַשּׁוֹמֵר עוֹמֵד עָלָיו לִשְׁמֹר הַכֶּרֶם וְרוֹאֶה לְמֵרָחוֹק: זוֹרְעִים בְּתוֹכָהּ. וּבְהָא לֹא פְּלִיגֵי רַבָּנָן עַל רַבִּי אֱלִיעֶזֶר, לְפִי שֶׁהִיא גְבוֹהָה וְאֵין אֲוִיר הַכֶּרֶם מַקִּיפָהּ, אֲבָל בְּגַת שֶׁאֲוִיר הַכֶּרֶם מַקִּיפָהּ אָסְרֵי רַבָּנָן, וְכֵן הֲלָכָה: וְאִם הָיָה שֵׂעָר כּוֹתֵשׁ. שֶׁהַשָּׂרִיגִים שֶׁל הַגְּפָנִים שֶׁהֵן קְרוּיִין שֵׂעָר הָיוּ מִתְעָרְבִים וּמִסְתַּבְּכִים לְמַעְלָה עַל גַּבֵּי הַשּׁוֹמֵרָה זֶה עַל גַּבֵּי זֶה כַּעֲלִי בַּמַּכְתֵּשׁ :

פרק ה - משנה ד
נֶקַע. בְּקָעִים שֶׁבַּשָּׂדוֹת כְּעֵין גֻּמּוֹת: נוֹתְנִים לָהּ עֲבוֹדָתָהּ. שִׁשָּׁה טְפָחִים שֶׁזּוֹ הִיא עֲבוֹדַת גֶּפֶן יְחִידִית: וְזוֹרֵעַ אֶת הַמּוֹתָר. וְלֹא אָמְרִינַן דְּמֵחֲזֵי כְּכִלְאַיִם, כֵּיוָן שֶׁהַגֶּפֶן וְהַזְּרָעִים בְּתוֹךְ הַגַּת אוֹ בְתוֹךְ הַנֶּקַע: אִם אֵין שָׁם אַרְבַּע אַמּוֹת אָסוּר. כְּדִין מְחוֹל הַכֶּרֶם דְּפִרְקִין דִּלְעֵיל דְּנוֹתֵן אַרְבַּע אַמּוֹת לַעֲבוֹדַת הַכֶּרֶם . וְאֵין הֲלָכָה כְּרַבִּי יוֹסֵי: וְהַבַּיִת שֶׁבַּכֶּרֶם. שֶׁסְּבִיבוֹת הַבַּיִת יֵשׁ גְּפָנִים: זוֹרְעִים בְּתוֹכוֹ. וְלֹא אָמְרִינַן הוֹאִיל וְהַגְּפָנִים מַקִּיפִים אוֹתוֹ מֵאַרְבַּע רוּחוֹתָיו אֵין הַמְּחִצּוֹת מוֹעִילוֹת :

הַנּוֹטֵעַ יָרָק בַּכֶּרֶם אוֹ מְקַיֵּם. שֶׁרוֹאֵהוּ שֶׁצּוֹמֵחַ וְיַנִּיחֵהוּ: הֲרֵי זֶה מְקַדֵּשׁ וְכוּ'. כְּלַל מִשְׁנָתֵנוּ הוּא שֶׁלְּעוֹלָם מְקַדֵּשׁ שֵׁשׁ עֶשְׂרֵה אַמָּה לְכָל רוּחַ עֲגֻלּוֹת. וְאַף עַל גַּב דִּתְנַן לְעֵיל הַזּוֹרֵעַ אַרְבַּע אַמּוֹת שֶׁל עֲבוֹדַת הַכֶּרֶם אֵינוֹ מְקַדֵּשׁ אֶלָּא שְׁתֵּי שׁוּרוֹת, הֵיכָא דְזוֹרֵעַ בֵּין הַגְּפָנִים מְקַדֵּשׁ שֵׁשׁ עֶשְׂרֵה כְּשִׁעוּר כֶּרֶם גָּדוֹל דְּמִצְטָרֵף עַד שֵׁשׁ עֶשְׂרֵה. אָמְנָם אֵיזֶה כֶּרֶם שֶׁיִּהְיֶה מְרֻבָּע שְׁלֹשִׁים וּשְׁתַּיִם אַמּוֹת עַל שְׁלֹשִׁים וּשְׁתַּיִם אַמּוֹת וְהַיָּרָק נָטוּעַ בִּנְקֻדַּת אַמָּה לְכָל רוּחַ מִן הַיָּרָק. וְהִנֵּה אִם הַגְּפָנִים נְטוּעוֹת עַל אַרְבַּע אַרְבַּע. פֵּרוּשׁ שֶׁיֵּשׁ בֵּין כָּל שׁוּרָה וְשׁוּרָה אַרְבַּע אַמּוֹת, וְהִפְרַשְׁנוּ מִזֶּה הַכֶּרֶם מְרֻבָּע אַמָּה בִּלְבָד, לָמָּה הִפְרִישׁוֹ הַתַּנָּא מִכְּלַל שֵׁשׁ עֶשְׂרֵה אַמָּה לְכָל רוּחַ דְּבַחֲלוּקָה דְסֵיפָא דְּנֵימָא הָיוּ נְטוּעוֹת עַל אַרְבַּע אַרְבַּע אוֹ עַל שֵׁשׁ שֵׁשׁ, וְצֵרֵף אוֹתוֹ עִם כְּשֶׁאַגִּיעַ לוֹ. אַף עַל פִּי שֶׁהוֹסִיף מָאתַיִם עַד שֶׁלֹּא הִגִּיעַ לוֹ מֻתָּר, מֵאַחַר שֶׁאֵינוֹ מִתְעַצֵּל בַּכִּלְאַיִם, אֲבָל כְּשֶׁאֶחֱזֹר אָסוּר מִשּׁוּם דְּנִתְעַצֵּל וְנִתְיָאֵשׁ, וְדִכְוָתָהּ גַּבֵּי מְחִצַּת הַכֶּרֶם שֶׁנִּפְרְצָה , דִּכְשֶׁנִּתְיָאֵשׁ וְלֹא גְדָרָהּ קִדֵּשׁ, אֲבָל לֹא נִתְיָאֵשׁ שֶׁמִּתְעַסֵּק כָּל שָׁעָה לִגְדֹּר וְלֹא הִסְפִּיק עַד שֶׁהוֹסִיף מָאתַיִם לֹא קִדֵּשׁ. וְכֵיצַד מְשַׁעֲרִין אִם הוֹסִיף מָאתַיִם, רוֹאִים כְּשֶׁחוֹתְכִים יָרָק זֶה אוֹ מִין שֶׁל תְּבוּאָה זוֹ מִן הַמְחֻבָּר, בְּכַמָּה זְמָן יָבַשׁ שֶׁלֹּא תִשָּׁאֵר בּוֹ לֵיחָה, הַגַּע בְּעַצְמְךָ שֶׁיָּבַשׁ בְּמֵאָה שָׁעָה, אִם נִשְׁתַּהָא בָּאָרֶץ חֲצִי שָׁעָה מִשֶּׁהִגִּיעַ לוֹ וְלֹא לִקְּטוֹ הֲרֵי הוֹסִיף מָאתַיִם וְאָסוּר, שֶׁכְּשֵׁם שֶׁהוּא יָבֵשׁ מִלַּחוּתוֹ בְּכָל חֲצִי שָׁעָה חֵלֶק מִמָּאתַיִם כְּשֶׁהוּא בְּתָלוּשׁ, כָּךְ מוֹסִיף בַּחֲצִי שָׁעָה חֵלֶק מִמָּאתַיִם כְּשֶׁהוּא בִּמְחֻבָּר :

עִם הַזְּבָלִים אוֹ עִם הַמַּיִם. כְּשֶׁמְּזַבֵּל כַּרְמוֹ אוֹ כְּשֶׁפּוֹתֵחַ אַמַּת הַמַּיִם לְהַשְׁקוֹת הַכֶּרֶם פְּעָמִים יֵשׁ עִמָּהֶם זְרָעִים, וְכָל הַנֵּי שֶׁלֹּא בְכַוָּנָה נִזְרְעוּ שָׁרֵי, דְּלֹא תִזְרַע כַּרְמְךָ בְּכַוָּנָה מַשְׁמַע: סִעֲרַתּוּ הָרוּחַ. שֶׁהָיָה עוֹמֵד בַּשָּׂדֶה וְזוֹרֵעַ וְהוֹלִיכָה רוּחַ סְעָרָה אֶת הַזֶּרַע בַּכֶּרֶם: לְאַחֲרָיו מֻתָּר. דְּשֶׁלֹּא בְכַוָּנָה הָיָה וְלֹא רָאָה שֶׁהוֹלִיךְ הָרוּחַ אֶת הַזֶּרַע לְתוֹךְ הַכֶּרֶם: וּלְפָנָיו אָסוּר. אִם סִעֲרַתּוּ הָרוּחַ לְפָנָיו וְרָאָה אָסוּר: כֵּיצַד יַעֲשֶׂה אִם עֲשָׂבִים יוֹפַךְ. יַחֲרוֹשׁ וִיהַפֵּךְ שָׁרְשֵׁי הָעֲשָׂבִים וְשׁוּב לֹא יִצְמְחוּ: אִם אָבִיב. שֶׁכְּבָר צָמַח הַזֶּרַע אֶלָּא שֶׁלֹּא הֵבִיא שְׁלִישׁ: יְנַפֵּץ. הַזֶּרַע מִן הַשִּׁבֹּלֶת, וְהַקַּשִּׁין מֻתָּרִין וְהַדָּגָן אָסוּר: וְאִם הֵבִיאָה דָגָן תִּדָּלֵק. וְהַכֹּל אָסוּר:

פרק ה - משנה ח
רַבִּי אֱלִיעֶזֶר אוֹמֵר קִדֵּשׁ. שֶׁכֵּן בְּעַרְבִיָּא מְקַיְּמִין קוֹצִים בַּכֶּרֶם כְּדֵי שֶׁיִּרְעוּ גְמַלֵּיהֶם, וּמֵאַחַר שֶׁמִּקְצָת בְּנֵי אָדָם מְקַיְּמִים אוֹתָם בִּשְׂדוֹתֵיהֶם הֲווּ כִלְאַיִם בְּכָל מָקוֹם, דְּשַׁפִּיר קָרִינַן בּוֹ אֲשֶׁר תִּזְרַע, דְּמַשְׁמַע מַה שֶּׁבְּנֵי אָדָם נוֹהֲגִים לִזְרֹעַ וְרוֹצִים בְּקִיּוּמָן: לֹא קִדֵּשׁ אֶלָּא דָבָר שֶׁכָּמוֹהוּ מְקַיְּמִים. בְּאוֹתוֹ מָקוֹם, כְּגוֹן קוֹצִים בְּעַרְבִיָּא לְפִי שֶׁלְּשָׁם מְקַיְּמִים אוֹתָם, אֲבָל בִּשְׁאָר מְקוֹמוֹת שֶׁאֵין מְקַיְּמִין אוֹתָן מֻתָּר. וַהֲלָכָה כַּחֲכָמִים: הָאִירוּס. מִין יָרָק שֶׁעָלָיו רְחָבִים וְנוֹתְנִין אוֹתוֹ בַּתַּבְשִׁיל, וְקוֹרִין לוֹ בַּעֲרָבִ"י סוסנב"ר: וְהַקִּסּוֹם. אדר"א בְּלַעַ"ז שֶׁמַּדְלִין אוֹתוֹ עַל הַחֲנֻיּוֹת וְעַל הַחַלּוֹנוֹת: וְשׁוֹשַׁנַּת הַמֶּלֶךְ. רוזי"ן בְּלַעַ"ז: וְכָל מִינֵי זְרָעִים אֵינָן כִּלְאַיִם בַּכָּרֶם. מִן הַתּוֹרָה, אֲבָל מִדְּרַבָּנָן הֲווּ כִלְאַיִם, דְּהָכִי מוּכָח בִּמְנָחוֹת : קַנְבּוֹס. קנאפ"ו בְּלַעַ"ז וּבַעֲרָבִי קינ"ב: וַחֲכָמִים אוֹמְרִים כִּלְאָיִם. וַהֲלָכָה כַּחֲכָמִים: הַקִּנְרָס. בַּעֲרָבִי כרפו"ש וּבְלַעַ"ז קרד"ו, וְהוּא דַּרְדַּר הָאָמוּר בַּתּוֹרָה (בְּרֵאשִׁית ג) וְקוֹץ וְדַרְדַּר תַּצְמִיחַ לָךְ:
