פרק א - משנה א
קָדְשֵׁי קָדָשִׁים שֶׁשְּׁחָטָן בַּדָּרוֹם. אַף עַל פִּי שֶׁדִּינָן לְהִשָּׁחֵט בַּצָּפוֹן , לֹא תֵימָא כְמַאן דְּחַנְקִינְהוּ דָּמוּ וְהָווּ כְקָדָשִׁים שֶׁמֵּתוּ שֶׁיָּצְאוּ יְדֵי מְעִילָה דְבַר תּוֹרָה, קָמַשְׁמַע לָן דְּקָדָשִׁים שֶׁמֵּתוּ לֹא חֲזוּ כְלָל , אֲבָל דָּרוֹם נְהִי דְאֵינוֹ רָאוּי לְקָדְשֵׁי קָדָשִׁים, אֲבָל רָאוּי הוּא לְקָדָשִׁים קַלִּים, הִלְכָּךְ מוֹעֲלִין בָּהֶן , וּמִי שֶׁנֶּהֱנָה מֵהֶם שָׁוֶה פְרוּטָה מֵבִיא קָרְבַּן מְעִילָה. וְלֹא בִלְבַד כִּי נִשְׁחֲטוּ בַדָּרוֹם וְקִבֵּל דָּמָן בַּצָּפוֹן הוּא דְּמוֹעֲלִין בָּהֶן, מִשּׁוּם דְּעִקַּר עֲבוֹדָה בַצָּפוֹן הַוְיָא , דְּמִקַּבָּלָה וָאֵילָךְ מִצְוַת כְּהֻנָּה, אֶלָּא אֲפִלּוּ שְׁחָטָן בַּצָּפוֹן וְקִבֵּל דָּמָן בַּדָּרוֹם, אַף עַל גַּב דְּעִקַּר עֲבוֹדָה הַוְיָא בַדָּרוֹם שֶׁלֹּא כַדִּין, אֲפִלּוּ הָכִי מוֹעֲלִין בָּהֶן : וְזָרַק דָּמָן בַּלַּיְלָה. אַף עַל גַּב דְּלַיְלָה לָאו זְמַן הַקְרָבָה, מוֹעֲלִין בָּהֶן: בַּלַּיְלָה וְזָרַק דָּמָן בַּיּוֹם. הַאי כָל שֶׁכֵּן הוּא דְמוֹעֲלִין, דְּהָא זָרַק בַּיּוֹם דְּעִקַּר עֲבוֹדָה הוּא. אֶלָּא זוֹ וְאֵין צָרִיךְ לוֹמַר זוֹ קָתָנֵי: אוֹ שֶׁשְּׁחָטָן. בַּצָּפוֹן, וְחָשַׁב עֲלֵיהֶן לְאָכְלָן חוּץ לִזְמַנָּן שֶׁהוּא פִגּוּל שֶׁיֶּשׁ בּוֹ כָרֵת, אוֹ חוּץ לִמְקוֹמָן שֶׁהוּא פְסוּל שֶׁאֵין בּוֹ כָרֵת, מוֹעֲלִים בָּהֶן: כְּלָל אָמַר רַבִּי יְהוֹשֻׁעַ כֹּל שֶׁהָיָה לָהּ שְׁעַת הֶתֵּר לַכֹּהֲנִים. אַף עַל גַּב דְּשׁוּב נִפְסְלָה, וְאֵין רַשָּׁאִים לְאָכְלָן, אֲפִלּוּ הָכִי אֵין מוֹעֲלִין בָּהֶן: שֶׁלָּנָה. לְאַחַר זְרִיקָה, אוֹ שֶׁנִּטְמְאָה אוֹ שֶׁיָּצְאָה חוּץ לָעֲזָרָה לְאַחַר זְרִיקָה. אַף עַל גַּב דְּאֵינָהּ רְאוּיָה לַכֹּהֲנִים, כֵּיוָן דְּהָיָה לָהּ שָׁעָה אַחַת הֶתֵּר קֹדֶם שֶׁלָּנָה, אֵין מוֹעֲלִים בָּהּ, דְּלָאו קָדְשֵׁי ה' קָרִינַן בָּהּ, דְּהָא חֲזוּ לַכֹּהֲנִים: וְשֶׁקִּבְּלוּ פְסוּלִים וְזָרְקוּ אֶת דָּמָהּ. קִבְּלוּ פְּסוּלִים אֶת דָּמָהּ אַף עַל פִּי שֶׁזְּרָקוּהוּ כְּשֵׁרִים, אוֹ שֶׁזְּרָקוּהוּ פְּסוּלִים אַף עַל פִּי שֶׁקִּבְּלוּהוּ כְּשֵׁרִים. וְאִם לְאַחַר שֶׁקִּבְּלוּ פְּסוּלִים אֶת הַדָּם וּזְרָקוּהוּ, חָזְרוּ כְּשֵׁרִים וְקִבְּלוּ שְׁאָר דַּם הַנֶּפֶשׁ וּזְרָקוּהוּ, הֲרֵי זְרִיקַת הַכְּשֵׁרִים מוֹעֶלֶת וּמַתֶּרֶת הַבָּשָׂר לַכֹּהֲנִים וְאֵין בּוֹ מִשּׁוּם מְעִילָה. וְהָנֵי מִלֵּי שְׁאָר פְּסוּלִים חוּץ מִן הַטָּמֵא, אֲבָל טָמֵא שֶׁקִּבֵּל אֶת הַדָּם וּזְרָקוֹ, אַף עַל פִּי שֶׁחָזְרוּ כְּשֵׁרִים וְקִבְּלוּ שְׁאָר דַּם הַנֶּפֶשׁ וּזְרָקוּהוּ, אֵין לַבָּשָׂר שְׁעַת הֶתֵּר, וּמוֹעֲלִים בּוֹ, שֶׁהַטָּמֵא הוֹאִיל וְרָאוּי לַעֲבוֹדַת צִבּוּר, שֶׁקָּרְבַּן צִבּוּר דּוֹחֶה אֶת הַטֻּמְאָה, כְּשֶׁזָּרַק הַדָּם נַעֲשָׂה שְׁאָר הַדָּם שְׁיָרִים, וְשׁוּב אֵין זְרִיקַת הַכְּשֵׁרִים מוֹעֶלֶת לְהַתִּיר הַבָּשָׂר. וְאֵין לְךָ בַּפְּסוּלִים מִי שֶׁעוֹשֶׂה הַדָּם שְׁיָרִים אֶלָּא הַטָּמֵא בִלְבַד:

פרק א - משנה ב
בְּשַׂר קָדְשֵׁי קָדָשִׁים שֶׁיָּצָא לִפְנֵי זְרִיקַת דָּמִים. וְאַחַר כָּךְ נִכְנַס וְאַחַר כָּךְ זָרַק אֶת הַדָּם: רַבִּי אֱלִיעֶזֶר אוֹמֵר מוֹעֲלִים בּוֹ. אַף עַל פִּי שֶׁזָּרַק אֶת הַדָּם. דִּסְבִירָא לֵיהּ לְרַבִּי אֱלִיעֶזֶר דִּזְרִיקָה לֹא מַהֲנֵי לַיּוֹצֵא לַאֲפוֹקֵי מִידֵי מְעִילָה: וְאֵין חַיָּבִין עָלָיו מִשּׁוּם פִּגּוּל נוֹתָר וְטָמֵא. הוֹאִיל וְיָצָא. דִּזְרִיקָה כְשֵׁרָה קָא קָבְעָה לְפִגּוּל, וְלֹא פְסוּלָה : רַבִּי עֲקִיבָא סָבַר אֵין מוֹעֲלִים בּוֹ. דְּקָסָבַר דִּזְרִיקָה מְהַנְּיָא לַיּוֹצֵא לְאַפּוֹקֵי מִמְּעִילָה : אֲבָל חַיָּבִין כוּ'. אֲבָל, לְשׁוֹן בְּרַם, כְּמוֹ בֶאֱמֶת. וְכִי אָמַר רַבִּי עֲקִיבָא דִזְרִיקָה מַהֲנֵי לַיּוֹצֵא, כְּגוֹן שֶׁיָּצָא מִקְצָת הַבָּשָׂר וְלֹא כֻלּוֹ, דְּמִגּוֹ דִמְהַנְּיָא לְהַהוּא מִקְצָת שֶׁבִּפְנִים, מְהַנְּיָא נַמִּי לְהַהוּא מִקְצָת שֶׁיָּצָא חוּץ. וַהֲלָכָה כְּרַבִּי עֲקִיבָא: אָמַר רַבִּי עֲקִיבָא וַהֲרֵי הַמַּפְרִישׁ חַטָּאתוֹ וְאָבְדָה. הַשְׁתָּא מַיְתֵי רְאָיָה לְמַאי דְקָאָמַר דִּזְרִיקָה מוֹעֶלֶת לַיּוֹצֵא: וַהֲרֵי שְׁתֵּיהֶן עוֹמְדוֹת. שְׁחוּטוֹת וְנִתְקַבֵּל דָּמָן בִּשְׁתֵּי כוֹסוֹת וְזָרַק דָּמָהּ שֶׁל אַחַת מֵהֶן: לֹא כְשֵׁם שֶׁדָּמָהּ. שֶׁל אוֹתָהּ: פּוֹטֵר אֶת בְּשָׂרָהּ. מִן הַמְּעִילָה: כָּךְ הוּא פוֹטֵר אֶת בְּשַׂר חֲבֶרְתָּהּ. שֶׁלֹּא נִזְרַק דָּמָהּ, מִן הַמְּעִילָה, הוֹאִיל וְהָיָה יָכוֹל לִזְרֹק דָּמָהּ שֶׁל אֵיזוֹ שֶׁיִּרְצֶה: וְאִם פּוֹטֵר דָּמָהּ אֶת בְּשַׂר חֲבֶרְתָּהּ מִן הַמְּעִילָה. אַף עַל גַּב דִּפְסוּלָה הִיא, דְּמוֹתַר חַטָּאת הִיא, אֵינוֹ דִין שֶׁיִּפְטֹר בְּשָׂרָהּ שֶׁל עַצְמָהּ, אַף עַל פִּי שֶׁנִּפְסְלָה בְּיוֹצֵא. וְלֹא אָמַר רַבִּי עֲקִיבָא כְשֵׁם שֶׁדָּמָהּ פּוֹטֵר אֶת בְּשָׂרָהּ כָּךְ הוּא פוֹטֵר אֶת בְּשַׂר חֲבֶרְתָּהּ, אֶלָּא כְשֶׁשָּׁחַט שְׁתֵּי הַחַטָּאוֹת כְּאַחַת , לְפִי שֶׁאִם רָצָה מִזֶּה זוֹרֵק רָצָה מִזֶּה זוֹרֵק, אֲבָל בָּזֶה אַחַר זֶה לֹא אָמַר רַבִּי עֲקִיבָא שֶׁיִּהְיֶה דָמָהּ פּוֹטֵר בְּשַׂר חֲבֶרְתָּהּ:

פרק א - משנה ג
אֵמוּרֵי קָדָשִׁים קַלִּים. אֵין מוֹעֲלִים בְּאֵמוּרֵי קָדָשִׁים קַלִּים אֶלָּא לְאַחַר זְרִיקַת דָּמִים, כִּדְאָמְרִינַן בְּשִׁלְהֵי פִרְקִין. וְאִם יָצְאוּ לִפְנֵי זְרִיקַת דָּמִים, רַבִּי אֱלִיעֶזֶר אוֹמֵר אֵין מוֹעֲלִין בָּהֶן. דְּכִי הֵיכִי דְאִית לֵיהּ לְרַבִּי אֱלִיעֶזֶר דְּאֵין זְרִיקָה מוֹעֶלֶת לַיּוֹצֵא לְאַפּוֹקֵי בְשַׂר קָדְשֵׁי קָדָשִׁים מִמְּעִילָה, הָכִי נַמִּי אֵין מוֹעֶלֶת לַיּוֹצֵא לְהָבִיא אֵמוּרֵי קָדָשִׁים קַלִּים לִידֵי מְעִילָה . וְכִי הֵיכִי דְאִית לֵיהּ לְרַבִּי עֲקִיבָא דִזְרִיקָה מוֹעֶלֶת לַיּוֹצֵא לְאַפּוֹקֵי בְשַׂר קָדְשֵׁי קָדָשִׁים מִידֵי מְעִילָה, מוֹעֶלֶת נַמִּי לַיּוֹצֵא לְהָבִיא אֵמוּרֵי קָדָשִׁים קַלִּים לִידֵי מְעִילָה:

פרק א - משנה ד
מַעֲשֵׂה דָמִים. זוֹ זְרִיקַת הַדָּם: וְאֵין מוֹעֲלִין בַּבָּשָׂר. לְאַחַר זְרִיקַת דָּמִים אֵין מוֹעֲלִין בַּבָּשָׂר, דִּכְבָר יֶשׁ בּוֹ שְׁעַת הֶתֵּר לַכֹּהֲנִים . וְהַיְנוּ לְהָקֵל, דְּמִשּׁוּם זְרִיקַת דָּמִים אָתְיָא לֵיהּ קֻלָּא דְאֵין מוֹעֲלִין בּוֹ: וְעַל זֶה וְעַל זֶה. בֵּין עַל הָאֵמוּרִין בֵּין עַל הַבָּשָׂר לְאַחַר זְרִיקָה: חַיָּבִין מִשּׁוּם פִּגּוּל. אִם פִּגֵּל בְּאַחַת מֵאַרְבַּע עֲבוֹדוֹת. דִּזְרִיקָה קֹבַעַת בְּפִגּוּל וְקָבְעָה נַמִּי לְנוֹתָר וְטָמֵא. וְהַיְנוּ לְהַחְמִיר: וּבְקָדָשִׁים קַלִּים. לְאַחַר זְרִיקָה: כֻּלּוֹ לְהַחְמִיר. דְּמוֹעֲלִין בָּאֵמוּרִים, דִּכְבָר חֲזוּ לְגָבוֹהַּ וְקָדָשִׁים נִינְהוּ וְלֹא מָמוֹן בְּעָלִים: וְעַל זֶה וְעַל זֶה. בֵּין עַל הָאֵמוּרִים בֵּין עַל הַבָּשָׂר חַיָּבִין עָלָיו מִשּׁוּם פִּגּוּל נוֹתָר וְטָמֵא. וְהַיְנוּ כֻלּוֹ לְהַחְמִיר:
