פרק ז

חַטַּאת הָעוֹף שֶׁעֲשָׂאָהּ לְמַטָּה. מִחוּט הַסִּקְרָא, כְּהִלְכָתָהּ שֶׁהִיא נַעֲשֵׂית לְמַטָּה, כְּדַאֲמַרַן בְּפִרְקִין דִּלְעֵיל: כְּמַעֲשֵׂה חַטָּאת. בְּהַזָּאָה וּמִצּוּי: לְשֵׁם חַטָּאת. אַף עַל פִּי שֶׁשִּׁנָּה בְּסֵדֶר מְלִיקָה, כְּגוֹן שֶׁמָּלַק וְהִבְדִּיל, וּבְחַטָּאת נֶאֱמַר וְלֹא יַבְדִּיל : כְּשֵׁרָה. הוֹאִיל וְנַעֲשֵׂית לְמַטָּה וּבִשְׁאָר עֲבוֹדוֹת כְּהִלְכָתָהּ: כְּמַעֲשֵׂה חַטָּאת לְשֵׁם עוֹלָה וְכוּ' פְּסוּלָה. דְּחַטָּאת שֶׁלֹּא לִשְׁמָהּ פְּסוּלָה: כְּמַעֲשֵׂה כֻלָּם. כְּלוֹמַר כְּמַעֲשֶׂה אֶחָד מִכֻּלָּם הָאֲמוּרִים כָּאן, וַאֲפִלּוּ כְּמַעֲשֵׂה חַטָּאת לְשֵׁם חַטָּאת, הוֹאִיל וַעֲשָׂאָהּ לְמַעְלָה פְּסוּלָה :

עוֹלַת הָעוֹף שֶׁעֲשָׂאָהּ לְמַעְלָה. כְּהִלְכָתָהּ שֶׁהִיא נַעֲשֵׂית לְמַעְלָה: כְּמַעֲשֵׂה עוֹלָה לְשֵׁם עוֹלָה. אַף עַל פִּי שֶׁשִּׁנָּה בִמְלִיקָתָהּ שֶׁלֹּא הִבְדִּיל, וּמִן הַדִּין הָיָה לוֹ לְהַבְדִּיל, אֲפִלּוּ הָכִי כְּשֵׁרָה : כְּמַעֲשֵׂה עוֹלָה לְשֵׁם חַטָּאת כְּשֵׁרָה. דְּכָל הַזְּבָחִים שֶׁנִּזְבְּחוּ שֶׁלֹּא לִשְׁמָן כְּשֵׁרִים אֶלָּא שֶׁלֹּא עָלוּ לַבְּעָלִים לְשֵׁם חוֹבָה, חוּץ מִן הַפֶּסַח וּמִן הַחַטָּאת: עֲשָׂאָהּ לְמַטָּה כְּמַעֲשֶׂה כֻלָּן. כְּמַעֲשֶׂה אֶחָד מִכֻּלָּן, וַאֲפִלּוּ כְּמַעֲשֶׂה עוֹלָה לְשֵׁם עוֹלָה: פְּסוּלָה. הוֹאִיל וַעֲשָׂאָהּ לְמַטָּה פְּסוּלָה:

וְכֻלָּן אֵינָן מְטַמְּאוֹת בְּבֵית הַבְּלִיעָה. אַף עַל פִּי שֶׁנִּפְסְלוּ, מְלִיקָתָן מְטַהֲרְתָּן מִידֵי נְבֵלָה, וְאֵין מְטַמְּאִין בְּבֵית הַבְּלִיעָה כְּדִין נִבְלַת עוֹף טָהוֹר שֶׁהָאוֹכְלָהּ אַף עַל פִּי שֶׁלֹּא נָגַע בָּהּ כְּגוֹן שֶׁתְּחָבָהּ לוֹ חֲבֵרוֹ בְּבֵית הַבְּלִיעָה נִטְמָא טֻמְאָה חֲמוּרָה לְטַמֵּא בְגָדִים שֶׁעָלָיו, אֲבָל אֵלּוּ אַף עַל פִּי שֶׁאֵין מְלִיקָתָן מַתִּירָתָן בַּאֲכִילָה מִפְּנֵי הַפְּסוּל שֶׁאֵרַע בָּהֶן, מִכָּל מָקוֹם מְטַהֲרָתָן וּמוֹצִיאָתָן מִידֵי נְבֵלָה: וּמוֹעֲלִין בָּהֶן. הַנֶּהֱנֶה מֵהֶן בְּשָׁוֶה פְרוּטָה מֵבִיא קָרְבַּן מְעִילָה כְּדִין כָּל נֶהֱנֶה מִן הַהֶקְדֵּשׁ. וַאֲפִלּוּ חַטָּאת, הוֹאִיל וְנִפְסְלָה וְלֹא הָיָה בָהּ שְׁעַת הֶתֵּר לַכֹּהֲנִים, לֹא יָצְאָה מִידֵי מְעִילָה: חוּץ מֵחַטַּאת הָעוֹף שֶׁעֲשָׂאָהּ לְמַטָּה כְּמַעֲשֵׂה חַטָּאת לְשֵׁם חַטָּאת. דִּכְשֵׁרָה הִיא וְיֵשׁ בָּהּ שְׁעַת הֶתֵּר לַכֹּהֲנִים, וְשׁוּב אֵין בָּהּ מְעִילָה אֲפִלּוּ לְזָר:

רַבִּי אֱלִיעֶזֶר אוֹמֵר מוֹעֲלִים בָּהּ. דְּהָא עוֹלָה הִיא וּמִי הוֹצִיאָהּ מִמְּעִילָתָהּ, וַהֲלֹא שְׁעַת הֶתֵּר לַכֹּהֲנִים לֹא הָיָה לָהּ: רַבִּי יְהוֹשֻׁעַ אוֹמֵר אֵין מוֹעֲלִים בָּהּ. דְּכֵיוָן דְּשִׁנָּה שְׁמָהּ וּמַעֲשֶׂיהָ וּמְקוֹמָהּ לְשֵׁם חַטָּאת, נַעֲשֵׂית חַטָּאת : וּמַה חַטַּאת הָעוֹף שֶׁאֵין מוֹעֲלִים בָּהּ. כְּשֶׁשָּׁחֲטָהּ לִשְׁמָהּ, שֶׁהֲרֵי נֶאֱכֶלֶת לַכֹּהֲנִים: שִׁנָּה אֶת שְׁמָהּ. נִפְסְלָה וְלֹא בָאָה לִכְלַל הֶתֵּר: מוֹעֲלִים בָּהּ. דְּהָא רַבִּי יְהוֹשֻׁעַ גּוּפֵיהּ לֹא פָּלֵיג עָלֶיהָ: לְדָבָר שֶׁיֶּשׁ בּוֹ מְעִילָה. לְשֵׁם עוֹלָה: שֶׁשָּׁחֲטָן בַּדָּרוֹם. לְשֵׁם שְׁלָמִים, יוֹכִיחוּ: שֶׁשִּׁנָּה שְׁמָן וּמַעֲשֵׂיהֶן לְדָבָר שֶׁאֵין בּוֹ מְעִילָה. שֶׁקָּדָשִׁים קַלִּים אֵין מְעִילָה אֶלָּא בְּאֵמוּרֵיהֶן: וּמוֹעֲלִים בָּהֶן. מִפְּנֵי שֶׁנִּפְסְלוּ בְּשִׁנּוּי מְקוֹמָן וְלֹא הוֹצִיאָתָן זְרִיקָתָן מִידֵי מְעִילָה : אִסּוּר וְהֶתֵּר. קָדָשִׁים קַלִּים יֵשׁ בְּאֵמוּרֵיהֶן מִשּׁוּם מְעִילָה וְאֵין בִּבְשָׂרָן מִשּׁוּם מְעִילָה: בְּדָבָר שֶׁכֻּלּוֹ הֶתֵּר. בְּחַטַּאת הָעוֹף שֶׁאֵין בּוֹ צַד מְעִילָה . וַהֲלָכָה כְּרַבִּי יְהוֹשֻׁעַ:

מָלַק בִּשְׂמֹאל. דִּמְלִיקָתוֹ פְסוּלָה. שֶׁכָּל מָקוֹם שֶׁנֶּאֱמַר אֶצְבַּע וּכְהֻנָּה אֵינוֹ אֶלָּא שֶׁל יָמִין: אוֹ בַלַּיְלָה. דְּאֵין מְלִיקָה בַלַּיְלָה, דְּרַחֲמָנָא אָמַר בְּיוֹם צַוֹּתוֹ: אֵינָן מְטַמְּאִין. הוֹאִיל וּפְסוּלָן בַּקֹּדֶשׁ וְאַהֲנֵי מְלִיקָה דִּידְהוּ שֶׁאִם עָלוּ לֹא יֵרְדוּ : שָׁחַט חֻלִּין בִּפְנִים וְקָדָשִׁים בַּחוּץ אֵין מְטַמְּאִין בְּבֵית הַבְּלִיעָה. וְאַף עַל פִּי שֶׁאֲסוּרִים בַּאֲכִילָה: מָלַק בְּסַכִּין. לָאו מְלִיקָה הִיא וְלָאו שְׁחִיטָה הִיא , דְּהָא מְלִיקָה בְּצִפֹּרֶן הִיא בְּעַצְמוֹ שֶׁל כֹּהֵן: מָלַק חֻלִּין בִּפְנִים. דְּלֹא שַׁיָּךְ מְלִיקָה בְּחֻלִּין אֶלָּא בְּקָדָשִׁים בִּפְנִים, וְהַמּוֹלֵק חֻלִּין בְּכָל מָקוֹם אוֹ קָדָשִׁים בַּחוּץ הָוֵי כְּאִלּוּ נוֹחֵר אוֹ מְעַקֵּר, וְאֵין אוֹתָהּ מְלִיקָה מוֹצִיאָה מִידֵי נְבֵלָה: תּוֹרִין שֶׁלֹּא הִגִּיעַ זְמַנָּן. תּוֹרִים גְּדוֹלִים כְּשֵׁרִים, קְטַנִּים פְּסוּלִים. בְּנֵי יוֹנָה, קְטַנִּים כְּשֵׁרִים, גְּדוֹלִים פְּסוּלִים: וְשֶׁיָּבַשׁ גַּפָּה. דִּמְחֻסָּר אֵבָר פָּסוּל אַף בָּעוֹפוֹת . דְּהָא דְּקַיְמָא לָן אֵין תַּמּוּת וְזִכְרוּת בְּעוֹפוֹת, הָנֵי מִלֵּי כְּגוֹן דּוֹקִין שֶׁבָּעַיִן שֶׁאֵין מוּמָן נִכָּר, אֲבָל מְחֻסַּר אֵבָר, הַקְרִיבֵהוּ נָא לְפֶחָתֶךָ : מְטַמֵּא בְבֵית הַבְּלִיעָה. דְּלֹא מַהֲנֵי מְלִיקָה דִּידְהוּ לְמִידִי, שֶׁאֲפִלּוּ עָלוּ יֵרְדוּ: כֹּל שֶׁאֵין פְּסוּלָן בַּקֹּדֶשׁ. שֶׁנִּפְסְלוּ קֹדֶם שֶׁבָּאוּ לָעֲזָרָה. וְכָל שֶׁאֵין פְּסוּלָן בַּקֹּדֶשׁ אָמְרִינַן לְקַמָּן בְּפֶרֶק הַמִּזְבֵּחַ מְקַדֵּשׁ, דְּאִם עָלָה יֵרֵד. אֲבָל אוֹתָן שֶׁפְּסוּלָן בַּקֹּדֶשׁ כְּגוֹן הַקֳּדָשִׁים הָרְאוּיִין אֶלָּא שֶׁנִּפְסְלוּ בַּעֲבוֹדוֹתֵיהֶן, אֵין מְטַמְּאִין, דְּהָא אַהֲנֵי לְהוּ מְלִיקָה דְּאִם עָלוּ לֹא יֵרְדוּ: כָּל הַפְּסוּלִים. כְּגוֹן זָר וְאוֹנֵן וּטְבוּל יוֹם וּמְחֻסַּר כִּפּוּרִים וְכָל אוֹתָן הַשְּׁנוּיִין בְּרֵישׁ פֶּרֶק ב': וְאֵינָן מְטַמְּאוֹת בְּבֵית הַבְּלִיעָה. שֶׁאַף עַל פִּי שֶׁאֵין מְלִיקָתָן מַתֶּרֶת קָדָשִׁים בַּאֲכִילָה, מִכָּל מָקוֹם מוֹצִיאָתָן מִידֵי נְבֵלָה:

אֵינָהּ מְטַמְּאָה בְבֵית הַבְּלִיעָה. דִּמְלִיקָתָהּ מְטַהַרְתָּהּ מִידֵי נְבֵלָה: רַבִּי יְהוּדָה אוֹמֵר מְטַמְּאָה בְבֵית הַבְּלִיעָה. דִּסְבִירָא לֵיהּ לְרַבִּי יְהוּדָה דְלָא אַהֲנֵי שְׁחִיטָה בְּעוֹף שֶׁל חֻלִּין וְלֹא מְלִיקָה בְּקָדָשִׁים בִּטְרֵפָה בָּעוֹף לְהוֹצִיאָהּ מִידֵי נְבֵלָה: אֵינוֹ דִין שֶׁתְּהֵא שְׁחִיטָה כוּ'. וְכֵיוָן דְּיָלַפְתְּ שֶׁשְּׁחִיטָה מְטַהֵר אֶת הַטְּרֵפָה בָּעוֹף בְּחֻלִּין מִקַּל וָחֹמֶר יָלְפִינַן מְלִיקַת קָדָשִׁים מִנַּהּ בְּבִנְיַן אָב, מַה מָּצִינוּ בִּשְׁחִיטָה כוּ' אַף מְלִיקָה הַמַּכְשֶׁרֶת בַּאֲכִילָה תְּטַהֵר טְרֵיפָתָהּ מִטֻּמְאָתָהּ: רַבִּי יוֹסֵי אוֹמֵר דַּיָּהּ כוּ'. כֵּיוָן דְּלֹא אַשְׁכְּחַן בְּהֶדְיָא דִּשְׁחִיטָה מוֹצִיאָה מִידֵי נְבֵלָה בָּעוֹף, אֶלָּא מִקַּל וָחֹמֶר מִנִּבְלַת בְּהֵמָה אַתָּה בָא לִלְמֹד, דַּיָּהּ שֶׁתְּהֵא כְּנִבְלַת בְּהֵמָה שֶׁשְּׁחִיטָתָהּ מְטַהַרְתָּהּ מִידֵי נְבֵלָה וְלֹא מְלִיקָתָהּ, דְּדַיּוֹ לַבָּא מִן הַדִּין לִהְיוֹת כַּנִּדּוֹן . וְשָׁלֹשׁ מַחֲלוֹקוֹת בַּדָּבָר, רַבִּי מֵאִיר סָבַר, אֶחָד שְׁחִיטָה וְאֶחָד מְלִיקָה מוֹצִיאָה בָּעוֹף מִידֵי נְבֵלָה. וְרַבִּי יְהוּדָה סָבַר, בֵּין שְׁחִיטָה בֵּין מְלִיקָה לֹא מַהֲנֵי בִּטְרֵפָה בָּעוֹף לְהוֹצִיאָהּ מִידֵי נְבֵלָה. וְרַבִּי יוֹסֵי סָבַר, שְׁחִיטָה מְהַנְּיָא, מְלִיקָה לֹא מְהַנְּיָא. וַהֲלָכָה כְּרַבִּי יוֹסֵי:
