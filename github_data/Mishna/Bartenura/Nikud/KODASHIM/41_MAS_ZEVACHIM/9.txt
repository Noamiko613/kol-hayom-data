פרק ט

הַמִּזְבֵּחַ מְקַדֵּשׁ אֶת הָרָאוּי לוֹ. וּפְלִיגֵי תַנָּאֵי בְּפֵרוּשָׁא דְּמַתְנִיתִין מַאי נִיהוּ הָרָאוּי לוֹ: רַבִּי יְהוֹשֻׁעַ אוֹמֵר כָּל הָרָאוּי לָאִשִּׁים. לְהִשָּׂרֵף עַל גַּבֵּי הָאֵשׁ, אַף עַל פִּי שֶׁנִּפְסַל, אִם עָלָה לַמִּזְבֵּחַ קִדְּשׁוֹ הַמִּזְבֵּחַ וְנַעֲשָׂה לַחְמוֹ, וְאִם עָלָה לֹא יֵרֵד. אֲבָל דָּבָר שֶׁאֵין רָאוּי לָאֵשׁ, כְּגוֹן דָּם פָּסוּל וּנְסָכִים פְּסוּלִים, אֲפִלּוּ אִם עָלוּ יֵרְדוּ: הִיא הָעֹלָה עַל מוֹקְדָה. דָּבָר שֶׁהוּא לַמּוֹקֵד וְעָלָה, הֲרֵי הוּא בַּהֲוָיָתוֹ וְלֹא יֵרֵד. הִיא, מַשְׁמַע בַּהֲוָיָתָהּ תְּהֵא: רַבָּן גַּמְלִיאֵל אוֹמֵר כָּל הָרָאוּי לַמִּזְבֵּחַ. וַאֲפִלּוּ דָּם וּנְסָכִים שֶׁנִּפְסְלוּ, אִם עָלוּ לֹא יֵרְדוּ. וְאֵין לְךָ לְמַעֵט אֶלָּא דָבָר שֶׁלֹּא הוּבְרַר לְחֶלְקוֹ שֶׁל מִזְבֵּחַ מֵעוֹלָם, כְּגוֹן קְמָצִים שֶׁלֹּא קָדְשׁוּ בִּכְלִי אַחַר קְמִיצָה, דְּאַף עַל גַּב דְּהַמִּנְחָה כֻלָּהּ קָדְשָׁה בִּכְלִי אֵין זֶה חֵלֶק גָּבוֹהַּ בָּרוּר, שֶׁאֵין הַמִּנְחָה קְרֵבָה כֻלָּהּ. וּבִקְמִיצָה בְלֹא מַתַּן כְּלִי, אֵין זֶה מְבֹרָר לַמִּזְבֵּחַ, וְאֵינוֹ מִתְקַדֵּשׁ אֲפִלּוּ עָלָה עַל גַּבֵּי הַמִּזְבֵּחַ: אֶלָּא הַדָּם וְהַנְּסָכִים. שֶׁהֵם רְאוּיִים לַמִּזְבֵּחַ וְאֵינָם רְאוּיִם לָאִשִּׁים: רַבִּי שִׁמְעוֹן אוֹמֵר הַזֶּבַח כָּשֵׁר וְהַנְּסָכִים פְּסוּלִים. רַבִּי שִׁמְעוֹן סְבִירָא לֵיהּ בִּנְסָכִים שֶׁל זֶבַח כְּרַבִּי יְהוֹשֻׁעַ וּבְדָם וּבִנְסָכִים הַבָּאִים בִּפְנֵי עַצְמָן כְּרַבָּן גַּמְלִיאֵל. וְלֹא מִמּוֹקְדָה יָלֵיף, אֶלָּא מִקְּרָא דִּכְתִיב (שְׁמוֹת כט) כָּל הַנּוֹגֵעַ בַּמִּזְבֵּחַ יִקְדָּשׁ, וּכְתִיב בַּתְרֵיהּ וְזֶה אֲשֶׁר תַּעֲשֶׂה עַל הַמִּזְבֵּחַ כְּבָשִׂים בְּנֵי שָׁנָה תְמִימִים שְׁנַיִם לַיּוֹם עוֹלָה, אֵין הַמִּזְבֵּחַ מְקַדֵּשׁ אֶלָּא דוּמְיָא דְּעוֹלָה, מָה עוֹלָה הַבָּאָה בִּגְלַל עַצְמָהּ, אַף כָּל הַבָּאִים בִּגְלַל עַצְמָן, יָצְאוּ נְסָכִים הַבָּאִים בִּגְלַל הַזֶּבַח. הִלְכָּךְ בֵּין שֶׁהַזֶּבַח כָּשֵׁר וְהַנְּסָכִים פְּסוּלִים כְּגוֹן שֶׁיָּצְאוּ אוֹ נִטְמְאוּ, בֵּין שֶׁהַזֶּבַח פָּסוּל וְהַנְּסָכִים כְּשֵׁרִים, דְּהוֹאִיל וְהֵן בָּאִים בִּגְלָלוֹ נִפְסָלִים עִמּוֹ וּבְטֵלָה תּוֹרַת הֶקְדֵּשׁ מֵהֶן: וַאֲפִלּוּ זֶה וְזֶה פְּסוּלִים הַזֶּבַח לֹא יֵרֵד. שֶׁהַמִּזְבֵּחַ מְקַדְּשׁוֹ: וְהַנְּסָכִים יֵרֵדוּ. כֵּיוָן שֶׁאֵין בָּאִים בִּגְלַל עַצְמָן אֵינָן דּוּמְיָא דְעוֹלָה וְאֵין מִזְבֵּחַ מְקַדְּשָׁן. וַהֲלָכָה כְּרַבִּי יְהוֹשֻׁעַ:

הַלָּן. בֵּין דָּם בֵּין אֵמוּרִים שֶׁלָּנוּ לַיְלָה חוּץ לַמִּזְבֵּחַ: רַבִּי יְהוּדָה אוֹמֵר שֶׁנִּשְׁחֲטָה בַלַּיְלָה כוּ' אִם עָלְתָה תֵּרֵד. דִּכְתִיב זֹאת תּוֹרַת הָעוֹלָה הִיא הָעוֹלָה, זֹאת הִיא הָעוֹלָה, הֲרֵי כָאן שְׁלשָׁה מִעוּטִים , מִעֵט נִשְׁחֲטָה בַלַּיְלָה וְשֶׁנִּשְׁפַּךְ דָּמָהּ וְשֶׁיָּצָא דָּמָהּ חוּץ לַקְּלָעִים, שֶׁאִם עָלְתָה תֵּרֵד: רַבִּי שִׁמְעוֹן אוֹמֵר לֹא תֵרֵד. דְּדָרִישׁ זֹאת תּוֹרַת הָעוֹלָה, תּוֹרָה אַחַת לְכָל הָעוֹלִים, שֶׁאִם עָלוּ לֹא יֵרְדוּ. וַהֲלָכָה כְּרַבִּי יְהוּדָה:

מַכְשִׁיר בְּבַעֲלֵי מוּמִין. דְּאִם עָלוּ לֹא יֵרְדוּ. וְדַוְקָא בְדוֹקִין שֶׁבָּעַיִן וְכַיּוֹצֵא בָהֶן הִכְשִׁיר רַבִּי עֲקִיבָא, הוֹאִיל וְהֵן כְּשֵׁרִים לְכַתְּחִלָּה בְּעוֹפוֹת. אֲבָל בִּמְחֻסָּר אֵבָר דְּפָסוּל בְּעוֹפוֹת, לֹא: דּוֹחֶה הָיָה אַבָּא אֶת בַּעֲלֵי מוּמִין. שֶׁאִם עָלוּ דּוֹחֶה הָיָה אוֹתָן כִּלְאַחַר יָד וְלֹא הָיָה מוֹרִידָן דֶּרֶךְ בִּזָּיוֹן בְּפַרְהֶסְיָא, וּבְהָא פָלֵיג אַתַּנָּא קַמָּא דְּאָמַר יֵרְדוּ. וַהֲלָכָה כְּרַבִּי חֲנִינָא בֶן אַנְטִיגְנוֹס :

כְּשֵׁם שֶׁאִם עָלוּ לֹא יֵרְדוּ. אַכָּל הָנָךְ פְּסוּלִין דִּתְנַן בְּהוּ לֹא יֵרְדוּ קָאֵי: כָּךְ אִם יָרְדוּ. לְאַחַר שֶׁעָלוּ, לֹא יַעֲלוּ עוֹד: וְכֻלָּן שֶׁעָלוּ חַיִּים בְּרֹאשׁ הַמִּזְבֵּחַ. רַבִּי עֲקִיבָא קָאָמַר לַהּ, דְּמַכְשִׁיר בְּבַעֲלֵי מוּמִין. וּמוֹדֶה שֶׁאִם עָלוּ חַיִּים בְּרֹאשׁ הַמִּזְבֵּחַ יֵרְדוּ, שֶׁאֵין הַמִּזְבֵּחַ מְקַדֵּשׁ בַּעֲלֵי מוּמִין חַיִּים: עוֹלָה שֶׁעָלְתָה חַיָּה לְרֹאשׁ הַמִּזְבֵּחַ תֵּרֵד. בְּעוֹלָה כְשֵׁרָה מַיְרֵי, וְדִבְרֵי הַכֹּל. וְלֹא נָקַט לַהּ אֶלָּא מִשּׁוּם סֵיפָא, לְלַמֶּדְךָ שֶׁרֹאשׁ הַמִּזְבֵּחַ מָקוֹם רָאוּי לְהֶפְשֵׁט וְנִתּוּחַ הֵיכָא דְּעָבַר וּשְׁחָטָהּ בְּרֹאשׁ הַמִּזְבֵּחַ:

וְאֵלּוּ אִם עָלוּ יֵרְדוּ. וַאֲפִלּוּ הֵן כְּשֵׁרִים וּטְהוֹרִים. מִשּׁוּם דְּלֹא שַׁיָּךְ בְּהוּ מִזְבֵּחַ כְּלָל: בְּשַׂר קָדְשֵׁי קָדָשִׁים כוּ'. דְּלַאֲכִילָה קָיְמֵי וְלֹא לְמִזְבֵּחַ: וְהַקְּטֹרֶת. לֹא רְאוּיָה הִיא לַמִּזְבֵּחַ הַחִיצוֹן: הַצֶּמֶר שֶׁבְּרָאשֵׁי כְבָשִׂים. גָּרְסִינַן. וְרֵישָׁא דְבָבָא אַחֲרִיתָא הִיא, וּבַהֲדֵי עֲצָמוֹת וְגִידִין קָתָנֵי לַהּ וּמִשְׁפָּט אֶחָד לָהֶם. הַצֶּמֶר שֶׁבְּרָאשֵׁי כְבָשִׂים, שֶׁל עוֹלָה. שֶׁהָרֹאשׁ אֵינוֹ בִכְלַל הֶפְשֵׁט וְקָרֵב עִם עוֹרוֹ, וְהָכִי אָמְרִינַן בִּשְׁחִיטַת חֻלִּין , מִנַּיִן לְרַבּוֹת אֶת הָרֹאשׁ שֶׁכְּבָר הֻתַּז בִּשְׁחִיטָה שֶׁאַף עַל פִּי שֶׁאֵינוֹ בִּכְלַל הֶפְשֵׁט יֶשְׁנוֹ בִכְלַל הַקְטָרָה, תַּלְמוּד לוֹמַר אֶת הָרֹאשׁ. וְיֵשׁ צֶמֶר בְּגֹבַהּ רֹאשׁוֹ שֶׁל כֶּבֶשׂ אֵצֶל עָרְפּוֹ, וְכֵן הַשֵּׂעָר שֶׁבִּזְקַן הַתְּיָשִׁים בִּזְמַן שֶׁהֵן מְחֻבָּרִים, כֻּלָּן עוֹלִים:

וְכֻלָּם. בֵּין פְּסוּלִים שֶׁעָלוּ דִּתְנַן בְּהוּ לֹא יֵרְדוּ, בֵּין עֲצָמוֹת וְגִידִין שֶׁהֶעֱלָן מְחֻבָּרִים וַאֲכָלָתָם הָאֵשׁ: וּפָקְעוּ מֵעַל הַמִּזְבֵּחַ. לָאָרֶץ. אסקלטי"ר בְּלַעַ"ז: לֹא יַחֲזִיר. אֵין צָרִיךְ לְהַחֲזִיר : וְכֵן גַּחֶלֶת שֶׁפָּקְעָה. אֵין צָרִיךְ לְהַחֲזִיר: אֵבָרִים שֶׁפָּקְעוּ כוּ'. אִי דְאִית בְּהוּ מַמָּשׁ, שֶׁבְּשָׂרָם נִכָּר, אֲפִלּוּ לְאַחַר חֲצוֹת נַמִּי יַחֲזִיר, דְּלֹא הָוֵי עִכּוּל. וְאִי דְלֵית בְּהוּ מַמָּשׁ, שֶׁנִּשְׂרְפוּ כֻלָּן וְנַעֲשׂוּ גַּחֶלֶת, אֲפִלּוּ קֹדֶם חֲצוֹת לֹא יַחֲזִיר. אֶלָּא הָכָא בְּמַאי עָסְקִינַן, כְּגוֹן שֶׁנִּתְקַשּׁוּ מֵחֲמַת הָאֵשׁ שֶׁשָּׁלְטָה בְכֻלָּן וְנִשְׂרְפוּ וְלֹא נַעֲשׂוּ פֶּחָם אֲבָל כְּעֵצִים יְבֵשִׁים הֵם מִתּוֹכָם: קֹדֶם לַחֲצוֹת וּלְאַחַר חֲצוֹת. יָלְפִינַן מִקְּרָא דִכְתִיב כָּל הַלַּיְלָה עַד הַבֹּקֶר, מִמַּשְׁמַע שֶׁנֶּאֱמַר כָּל הַלַּיְלָה אֵינִי יוֹדֵעַ שֶׁעַד הַבֹּקֶר, אֶלָּא תֵּן בֹּקֶר לְבָקְרוֹ שֶׁל לַיְלָה, שֶׁאֵין צְרִיכִין לִהְיוֹת עוֹד עַל הַמּוֹקֵד, וְאֵיזֶה בֹקֶר נוֹסַף עַל בָּקְרוֹ שֶׁל לַיְלָה שֶׁאֵין צְרִיכִין לִהְיוֹת עוֹד עַל הַמּוֹקֵד, זֶה חֲצוֹת. דְּבֹקֶר שֶׁל לַיְלָה הוּא עַמּוּד הַשַּׁחַר: וּמוֹעֲלִין בָּהֶן. דְּאַכַּתִּי בְּנֵי מִזְבֵּחַ נִינְהוּ: וְאֵין מוֹעֲלִין בָּהֶן. דְּכֵיוָן דַּהֲווּ מְעֻכָּלִים הֲווּ לְהוּ דָבָר שֶׁנַּעֲשֵׂית מִצְוָתוֹ וְאֵין מוֹעֲלִים בּוֹ:

הַכֶּבֶשׁ מְקַדֵּשׁ. דִּכְתִיב (שְׁמוֹת מ) וּמָשַׁחְתָּ אֶת הַמִּזְבֵּחַ וְגוֹ', אֶת, לְרַבּוֹת אֶת הַכֶּבֶשׁ: הַכֵּלִים מְקַדְּשִׁים. דִּבְכֵלִים נַמִּי כְתִיב (שָׁם ל) כָּל הַנּוֹגֵעַ בָּהֶם יִקְדָּשׁ: כְּלֵי הַלַּח. קְעָרוֹת וּמִזְרָקוֹת לְדָם וּלְיַיִן וּלְשֶׁמֶן: מִדּוֹת הַיָּבֵשׁ. שְׁתֵּי מִדּוֹת שֶׁל יָבֵשׁ הָיוּ שָׁם, עִשָּׂרוֹן, וַחֲצִי עִשָּׂרוֹן : שֶׁהָיוּ עוֹשִׂין וְהֵן שְׁלֵמִים. שֶׁהָיוּ עוֹשִׂים כְּשֶׁהֵן שְׁלֵמִים. וּמִיהוּ כָּל כְּלֵי שָׁרֵת שֶׁנִּקְּבוּ אוֹ שֶׁנִּשְׁבְּרוּ אֵין מְתַקְּנִין שִׁבְרֵיהֶן אֶלָּא עוֹשִׂים אוֹתָן חֲדָשִׁים. וְכֵן סַכִּין שֶׁנִּפְגַּם אֵין מַשְׁחִיזִין אוֹתוֹ לְהָסִיר הַפְּגָם. וּבִגְדֵי כְהֻנָּה שֶׁנִּתְלַכְלְכוּ אֵין מְכַבְּסִין אוֹתָן אֶלָּא עוֹשִׂים חֲדָשִׁים וּמַנִּיחִין אֵלּוּ לְצֹרֶךְ פְּתִילוֹת שֶׁל מְנוֹרָה וְשִׂמְחַת בֵּית הַשּׁוֹאֵבָה. וְכָל כָּךְ לָמָּה, לְפִי שֶׁאֵין עֲנִיּוּת בִּמְקוֹם עֲשִׁירוּת. וּמִזְבֵּחַ שֶׁנִּפְגַּם פָּסוּל. וְכָל פְּגִימָה הַפּוֹסֶלֶת בְּסַכִּין שֶׁל שְׁחִיטָה, פּוֹסֶלֶת בַּמִּזְבֵּחַ. וְכָל זְמַן שֶׁהַמִּזְבֵּחַ פָּגוּם כָּל קָדָשִׁים הַנִּשְׁחָטִין בָּעֲזָרָה כֻּלָּם פְּסוּלִין, דִּכְתִיב (שָׁם כ) וְזָבַחְתָּ עָלָיו אֶת עוֹלוֹתֶיךָ וְאֶת שְׁלָמֶיךָ, וְאִי אֶפְשָׁר לוֹמַר מִקְרָא כְּמַשְׁמָעוֹ שֶׁהֲרֵי שְׁחִיטַת קָדָשִׁים בָּעֲזָרָה הִיא וְלֹא עַל הַמִּזְבֵּחַ, אֶלָּא עָלָיו, כְּלוֹמַר כָּל זְמַן שֶׁהוּא שָׁלֵם אַתָּה זוֹבֵחַ, וְאֵין אַתָּה זוֹבֵחַ כְּשֶׁהוּא פָגוּם. וְכָל קָדָשִׁים הַנִּמְצָאִים בָּעֲזָרָה בְּשָׁעָה שֶׁהַמִּזְבֵּחַ פָּגוּם אַף עַל פִּי שֶׁנִּשְׁחֲטוּ בְכַשְׁרוּת , כֻּלָּם פְּסוּלִים. וַאֲפִלּוּ שְׁיָרֵי מְנָחוֹת אֵינָם נֶאֱכָלִים, דִּכְתִיב (וַיִּקְרָא י) וְאִכְלוּהָ מַצּוֹת אֵצֶל הַמִּזְבֵּחַ, מִפִּי הַשְּׁמוּעָה לָמְדוּ בִּזְמַן שֶׁהַמִּזְבֵּחַ שָׁלֵם וְלֹא בִּזְמַן שֶׁהוּא פָגוּם:
