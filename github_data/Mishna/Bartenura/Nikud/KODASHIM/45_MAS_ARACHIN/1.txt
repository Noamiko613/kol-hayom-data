פרק א

הַכֹּל מַעֲרִיכִין. הַכֹּל לַאֲתוֹיֵי בֶּן שְׁלֹשׁ עֶשְׂרֵה שָׁנָה וְיוֹם אֶחָד וְלֹא הֵבִיא שְׁתֵּי שְׂעָרוֹת, שֶׁהוּא מַעֲרִיךְ. וְהוּא מֻפְלָא הַסָּמוּךְ לְאִישׁ. וּמֻפְלָא אִקְּרֵי לֵיהּ לְפִי שֶׁבּוֹדְקִים אוֹתוֹ אִם יוֹדֵעַ לְהַפְלוֹת וּלְפָרֵשׁ לְשֵׁם מִי נָדַר וּלְשֵׁם מִי הִקְדִּישׁ. אַף עַל גַּב דְּאִישׁ כִּי יַפְלִיא כְתִיב (וַיִּקְרָא כז), כְּשֶׁהוּא סָמוּךְ לְפִרְקוֹ קָרוּי אִישׁ לְעִנְיָן זֶה. וְהַכֹּל דְּקָאֵי נַמִּי עַל נֶעֱרָכִים, לַאֲתוֹיֵי מְנֻוָּל וּמֻכֶּה שְׁחִין, שֶׁאֵין שָׁוִין דָּמִים, דְּסָלְקָא דַעְתָּךְ אֲמֵינָא נֶדֶר בְּעֶרְכְּךָ כְּתִיב (שָׁם), כָּל שֶׁיֶּשְׁנוֹ בְדָמִים יֶשְׁנוֹ בַעֲרָכִין, וְכָל שֶׁאֵינוֹ בְדָמִים אֵינוֹ בַעֲרָכִין, קָמַשְׁמַע לָן . וְהַכֹּל דְּקָאֵי עַל נוֹדְרִים, מִשּׁוּם נִדָּרִים נָקַט לֵיהּ. וְנִדָּרִים גּוּפֵיהּ אִצְטְרִיךְ, לַאֲתוֹיֵי פָּחוּת מִבֶּן חֹדֶשׁ שֶׁהוּא נִדָּר אַף עַל פִּי שֶׁאֵינוֹ נֶעֱרָךְ : מַעֲרִיכִין. אִם אָמַר עַל אֶחָד, עֵרֶךְ פְּלוֹנִי עָלָי, נוֹתֵן עֵרֶךְ אוֹתוֹ פְּלוֹנִי כְּפִי שָׁנָיו כְּמוֹ שֶׁקָּצוּב בְּפָרָשַׁת עֲרָכִין, שֶׁהָעֵרֶךְ נוֹתֵן לְפִי שָׁנִים שֶׁל נֶעֱרָכִין: וְנֶעֱרָכִין. אִם אָמַר אַחֵר עָלָיו עֵרֶךְ פְּלוֹנִי זֶה עָלָי, אוֹ אָמַר הוּא עַצְמוֹ עֶרְכִּי עָלָי: נוֹדְרִים. אָמַר דְּמֵי פְּלוֹנִי עָלָי, צָרִיךְ שֶׁיִּתֵּן כְּמוֹ שֶׁהוּא יָפֶה לִמָּכֵר בַּשּׁוּק: וְנִדָּרִין. אָמַר אַחֵר עָלָיו דְּמֵי פְּלוֹנִי עָלָי, אוֹ אָמַר הוּא עַצְמוֹ דָּמַי עָלָי: כֹּהֲנִים וּלְוִיִּם. כֹּהֲנִים אִצְטְרִיכָא לֵיהּ לְאַשְׁמְעִינַן דְּנֶעֱרָכִין, דְּסָלְקָא דַעְתָּךְ אֲמֵינָא הוֹאִיל וּכְתִיב (שָׁם) וְהֶעֱמִידוֹ לִפְנֵי הַכֹּהֵן וְהֶעֱרִיךְ אוֹתוֹ הַכֹּהֵן , יִשְׂרָאֵל לִפְנֵי כֹהֵן וְלֹא כֹהֵן לִפְנֵי כֹהֵן, הִלְכָּךְ כֹּהֵן לֵיתֵיהּ בַּעֲרָכִין, דְּאִי עָנִי הוּא לֹא קָרֵינָא בֵּיהּ וְהֶעֱמִידוֹ לִפְנֵי הַכֹּהֵן, קָמַשְׁמַע לָן דְּכֹהֲנִים אִיתַנְהוּ בְתוֹרַת עֲרָכִין. וְאַיְדֵי דִתְנַן כֹּהֲנִים, תְּנַן לְוִיִּם וְיִשְׂרְאֵלִים: נָשִׁים וַעֲבָדִים. וְהָאִשָּׁה מְשַׁלֶּמֶת כְּשֶׁתִּתְאַלְמֵן אוֹ תִּתְגָּרֵשׁ, וְהָעֶבֶד כְּשֶׁיִּשְׁתַּחְרֵר : טֻמְטוּם וְאַנְדְּרוֹגִינוֹס נוֹדְרִים וְנִדָּרִים. שֶׁהֲרֵי יֵשׁ לָהֶם דָּמִים: וּמַעֲרִיכִין. אֶת אֲחֵרִים, אִם אָמְרוּ עֵרֶךְ פְּלוֹנִי עָלָי, נוֹתְנִים עֵרֶךְ אוֹתוֹ פְּלוֹנִי: אֲבָל לֹא נֶעֱרָכִין. אִם אָמַר עֶרְכִּי עָלָי, אוֹ אָמַר אַחֵר עָלֵיהֶם עֵרֶךְ פְּלוֹנִי עָלָי, לֹא אָמַר כְּלוּם. דְּזָכָר וּנְקֵבָה אֲמוּרִים בְּפָרָשַׁת עֲרָכִין, עַד שֶׁיִּהְיֶה זָכָר וַדָּאִי אוֹ נְקֵבָה וַדָּאִית: פָּחוּת מִבֶּן חֹדֶשׁ נִדָּר. אִם אָמַר דָּמָיו עָלָי. שֶׁהֲרֵי שָׁוֶה דָּמִים כָּל שֶׁהֵן: אֲבָל לֹא נֶעֱרָךְ. שֶׁלֹּא נֶאֱמַר עֵרֶךְ בַּפָּרָשָׁה אֶלָּא מִבֶּן חֹדֶשׁ וָמַעְלָה:

הַנָּכְרִי רַבִּי מֵאִיר אוֹמֵר נֶעֱרָךְ. תְּרֵי קְרָאֵי כְתִיבֵי בְּפָרָשַׁת עֲרָכִין, חַד רִבּוּי וְחַד מִעוּט. בְּנֵי יִשְׂרָאֵל, מִעֵט אֶת הַנָּכְרִים. אִישׁ כִּי יַפְלִיא, רִבָּה כָל אִישׁ וַאֲפִלּוּ נָכְרִי בְּמַשְׁמַע. רַבִּי מֵאִיר סָבַר נָכְרִי נֶעֱרָךְ וְלֹא מַעֲרִיךְ, שֶׁכֵּן מָצִינוּ שֶׁרִבָּה הַכָּתוּב בַּנֶּעֱרָכִים יוֹתֵר מִבַּמַּעֲרִיכִים , שֶׁהֲרֵי חֵרֵשׁ שׁוֹטֶה וְקָטָן נֶעֱרָכִין, אֲבָל לֹא מַעֲרִיכִין, הִלְכָּךְ שְׁדֵי מִעוּטָא לַמַּעֲרִיךְ וְרִבּוּיָא לַנֶּעֱרָךְ: רַבִּי יְהוּדָה אוֹמֵר נָכְרִי מַעֲרִיךְ וְלֹא נֶעֱרָךְ. שֶׁכֵּן מָצִינוּ טֻמְטוּם וְאַנְדְּרוֹגִינוֹס מַעֲרִיכִים וְלֹא נֶעֱרָכִים. וַהֲלָכָה כְּרַבִּי יְהוּדָה . הִלְכָּךְ נָכְרִי שֶׁאָמַר עֵרֶךְ יִשְׂרָאֵל פְּלוֹנִי עָלָי, חַיָּב לִתֵּן כְּפִי שָׁנָיו הָעֵרֶךְ הַקָּצוּב בַּפָּרָשָׁה. וְיִשְׂרָאֵל שֶׁאָמַר עֵרֶךְ נָכְרִי פְּלוֹנִי עָלָי, אוֹ נָכְרִי שֶׁאָמַר עֶרְכִּי עָלָי, לֹא אָמַר כְּלוּם:

הַגּוֹסֵס. לֹא נִדָּר, דְּלָאו בַּר דָּמִים הוּא, דִּלְמִיתָה הוּא עוֹמֵד: וְהַיּוֹצֵא לֵהָרֵג. שֶׁנִּגְמַר דִּינוֹ בְּבֵית דִּין שֶׁל יִשְׂרָאֵל . אֲבָל יוֹצֵא לֵהָרֵג עַל פִּי הַמַּלְכוּת, לֹא שְׁנָא מַלְכוּת יִשְׂרָאֵל, וְלֹא שְׁנָא מַלְכוּת שֶׁל נָכְרִים, דִּבְרֵי הַכֹּל מַעֲרִיךְ וְנֶעֱרָךְ : לֹא נִדָּר. דְּלָאו בַּר דָּמִים הוּא: וְלֹא נֶעֱרָךְ. דִּכְתִיב וְהֶעֱמִידוֹ וְהֶעֱרִיכוֹ, וְהַאי לָאו בַּר הַעֲמָדָה וְהַעֲרָכָה הוּא : וְלֹא נֶעֱרָךְ. דִּכְתִיב (וַיִּקְרָא כז) כָּל חֵרֶם אֲשֶׁר יָחֳרַם מִן הָאָדָם לֹא יִפָּדֶה . וְרַבִּי חֲנִינָא בֶּן עֲקַבְיָא דְּאָמַר נֶעֱרָךְ, מוֹקֵי לְהַאי קְרָא דְּכָל חֵרֶם אֲשֶׁר יָחֳרַם לִדְרָשָׁא אַחֲרִיתִי : רַבִּי יוֹסֵי אוֹמֵר כוּ'. רַבִּי יוֹסֵי וְתַנָּא קַמָּא לֹא פְלִיגֵי בְנוֹדֵר וּמַעֲרִיךְ וּמַקְדִּישׁ. כִּי פְלִיגֵי בְאִם הִזִּיק . תַּנָּא קַמָּא סָבַר, אִם הִזִּיק פָּטוּר מִן הַתַּשְׁלוּמִים. וְאַף עַל פִּי שֶׁהַמַּזִּיק חַיָּב בְּתַשְׁלוּמִין מִן הַתּוֹרָה, מִלְוָה הַכְּתוּבָה בַּתּוֹרָה לָאו כִּכְתוּבָה בִשְׁטָר דָּמְיָא, אֶלָּא כְמִלְוָה עַל פֶּה, וּמִלְוָה עַל פֶּה אֵינוֹ גוֹבֶה מִן הַיּוֹרְשִׁים. וְרַבִּי יוֹסֵי סָבַר, מִלְוָה הַכְּתוּבָה בַּתּוֹרָה כְּמִלְוָה הַכְּתוּבָה בִשְׁטָר דָּמְיָא, וְגוֹבֶה מִן הַיּוֹרְשִׁים. וַהֲלָכָה כְתַנָּא קַמָּא. אֲבָל לְאַחַר שֶׁתִּקְּנוּ דְמִלְוָה עַל פֶּה גוֹבֶה מִן הַיּוֹרְשִׁים , בֵּית דִּין גּוֹבִין מִמָּמוֹנוֹ וּמְשַׁלְּמִים מַה שֶּׁהִזִּיק:

אֵין מַמְתִּינִין לָהּ עַד שֶׁתֵּלֵד. דְּמַהוּ דְתֵימָא וְלָדוֹת מָמוֹנָא דְבַעַל הֵן כְּדִכְתִיב (שְׁמוֹת כא) כַּאֲשֶׁר יָשִׁית עָלָיו בַּעַל הָאִשָּׁה, וְלֹא נַפְסְדִּינְהוּ מִנֵּיהּ, קָמַשְׁמַע לָן דִּכְתִיב (דְּבָרִים כב) וּמֵתוּ גַּם שְׁנֵיהֶם, גַּם לְרַבּוֹת אֶת הַוָּלָד: יָשְׁבָה עַל הַמַּשְׁבֵּר. מְקוֹם מוֹשַׁב הָאִשָּׁה הַיּוֹלֶדֶת קָרוּי מַשְׁבֵּר: מַמְתִּינִין לָהּ עַד שֶׁתֵּלֵד. דְּכֵיוָן דְּנֶעֱקַר לָצֵאת, גּוּפָא אַחֲרִינָא הוּא וְאֵינוֹ כְגוּף אִמּוֹ: נֶהֱנִין בִּשְׂעָרָהּ. לָאו שְׂעָרָהּ מַמָּשׁ קָאָמַר, אֶלָּא פֵּאָה שֶׁהָיְתָה לָהּ מִשְּׂעַר אִשָּׁה אַחֶרֶת קְשׁוּרָה לִשְׂעָרָהּ. וְדַוְקָא כְּשֶׁאָמְרָה תְנוּ אוֹתָהּ לְבִתִּי אוֹ לִפְלוֹנִית, דְּכֵיוָן דְּאָמְרָה תְנוּ גַּלְיָא דַעְתָּהּ דְּלֹא נִיחָא לַהּ דְּתֶהֱוֵי כְגוּפָה לְאִתְּסוֹרֵי, וְכִנְטוּלָה מֵחַיִּים דָּמְיָא. אֲבָל בְּעִנְיָן אַחֵר אָסוּר, שֶׁנּוֹיֵי הַמֵּת אֲסוּרִים בַּהֲנָאָה : בְּהֵמָה שֶׁנֶּהֶרְגָה אֲסוּרָה בַהֲנָאָה. אֲפִלּוּ שְׂעָרָהּ :
