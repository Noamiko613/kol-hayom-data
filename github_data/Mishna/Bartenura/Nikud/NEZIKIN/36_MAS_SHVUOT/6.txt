פרק ו

שְׁבוּעַת הַדַּיָּנִין. הַשְּׁבוּעָה שֶׁהַדַּיָּנִים מַשְׁבִּיעִים עַל יְדֵי הוֹדָאָה בְמִקְצָת הַטַּעֲנָה , צָרִיךְ שֶׁלֹּא תְהֵא הַטַּעֲנָה פָּחוֹת מִשְּׁתֵּי מָעוֹת כֶּסֶף, שֶׁהוּא שְׁלִישׁ דִּינָר, שֶׁהַדִּינָר שֵׁשׁ מָעוֹת וְהוּא מִשְׁקַל תִּשְׁעִים וְשֵׁשׁ שְׂעֹרוֹת בֵּינוֹנִיּוֹת, נִמְצְאוּ שְׁתֵּי מָעוֹת מִשְׁקַל שְׁלֹשִׁים וּשְׁתַּיִם שְׂעֹרוֹת: וְהַהוֹדָאָה. שֶׁמְּחַיַּבְתּוֹ לִשָּׁבַע, לֹא תְהֵא פָחוֹת מִשָּׁוֶה פְרוּטָה. שֶׁאִם הָיָה מַה שֶּׁכָּפַר לוֹ פָּחוֹת מִשְּׁתֵּי כֶסֶף, אוֹ מַה שֶּׁהוֹדָה פָּחוֹת מִשָּׁוֶה פְרוּטָה, אֵינוֹ חַיָּב שְׁבוּעָה דְאוֹרַיְתָא. אֲבָל מַשְׁבִּיעִין אוֹתוֹ שְׁבוּעַת הֶסֵּת מִדִּבְרֵי סוֹפְרִים. וְהַמְחֻיָּב שְׁבוּעָה דְאוֹרַיְתָא נָקֵיט חֶפְצָא בְּיָדֵיהּ כְּגוֹן סֵפֶר תּוֹרָה אוֹ תְפִלִּין כְּשֶׁנִּשְׁבָּע. וְהַמְחֻיָּב שְׁבוּעַת הֶסֵּת לֹא נָקֵיט הַנִּשְׁבָּע חֶפְצָא בִידֵיהּ, אֶלָּא חַזָּן הַכְּנֶסֶת אוֹ הַמַּשְׁבִּיעוֹ נָקֵיט חֶפְצָא בִידֵיהּ בְּשָׁעָה שֶׁחֲבֵרוֹ נִשְׁבָּע לוֹ. וּשְׁלֹשָׁה מִינֵי שְׁבוּעוֹת הֵן דְּאוֹרַיְתָא, וְלֹא יוֹתֵר. הַמּוֹדֶה בְמִקְצָת הַטַּעֲנָה, וּמִי שֶׁעֵד אֶחָד מֵעִיד כְּנֶגְדּוֹ שֶׁהוּא נִשְׁבָּע לְהַכְחִישׁ הָעֵד, וּשְׁבוּעַת הַשּׁוֹמְרִים. וְכָל שְׁאָר שְׁבוּעוֹת הַנִּזְכָּרִים בַּמִּשְׁנָה, הֵם בְּתַקָּנַת חֲכָמִים וְהֵם כְּעֵין שֶׁל תּוֹרָה בִּנְקִיטַת חֵפֶץ. וְאֵין הֶפְרֵשׁ בֵּין שְׁבוּעָה דְאוֹרַיְתָא לַשְּׁבוּעוֹת הַנִּזְכָּרוֹת בַּמִּשְׁנָה, אֶלָּא שֶׁהַמְחֻיָּב שְׁבוּעָה דְאוֹרַיְתָא וְלֹא רָצָה לִשָּׁבַע, בֵּית דִּין יוֹרְדִין לִנְכָסָיו וְנִפְרָעִים מִמֶּנּוּ מִשָּׁלֵם. וְהַמְחֻיָּב שְׁבוּעָה מִדִּבְרֵיהֶם בְּתַקָּנַת חֲכָמִים וְלֹא רָצָה לִשָּׁבַע, מְשַׁמְּתִין אוֹתוֹ עַד שֶׁיִּפְרַע אוֹ עַד שֶׁיִּשָּׁבַע. וְאִם עָמַד בְּנִדּוּיוֹ שְׁלֹשִׁים יוֹם וְלֹא רָצָה לִשָּׁבַע וְלֹא לְשַׁלֵּם, מַכִּין אוֹתוֹ מַכַּת מַרְדּוּת וּמַתִּירִין לוֹ נִדּוּיוֹ וְהוֹלֵךְ לוֹ, וְאֵין יוֹרְדִין לִנְכָסָיו: אֵין לְךָ בְיָדִי אֶלָּא פְרוּטָה פָּטוּר. שֶׁאֵין הַהוֹדָאָה מִמִּין הַטַּעֲנָה , שֶׁזֶּה טְעָנוֹ כֶסֶף וְזֶה הוֹדָה לוֹ נְחֹשֶׁת. וְדַוְקָא שֶׁלֹּא טְעָנוֹ אֶלָּא מִשְׁקַל שְׁתֵּי מָעוֹת כֶּסֶף אוֹ יוֹתֵר, אֲבָל אִם טְעָנוֹ מַטְבֵּעַ כֶּסֶף, הֲרֵי הוֹדָה לוֹ בַמַּטְבֵּעַ: שְׁתֵּי כֶסֶף וּפְרוּטָה. קָא סָבַר טְעָנוֹ חִטִּים וּשְׂעֹרִים וְהוֹדָה לוֹ עַל אַחַת מֵהֶן חַיָּב: מִפְּנֵי שֶׁהוּא כְמֵשִׁיב אֲבֵדָה. שֶׁהוּא פָטוּר מִשְּׁבוּעָה, כְּדִתְנַן הַמּוֹצֵא מְצִיאָה לֹא יִשָּׁבַע מִפְּנֵי תִקּוּן הָעוֹלָם. וְדַוְקָא כְּשֶׁאֵין הַבֵּן טוֹעֵן וַדַּאי מָנֶה לְאַבָּא בְיָדֶךָ, אֶלָּא שֶׁמָּא. אֲבָל טְעָנוֹ וַדַּאי, וְאָמַר הוּא לֹא הָיָה לוֹ אֶצְלוֹ אֶלָּא חֲמִשִּׁים, חַיָּב שְׁבוּעָה דְאוֹרַיְתָא, דְּאֵין זֶה מֵשִׁיב אֲבֵדָה:

מָנֶה לִי בְיָדֶךָ וְאָמַר לוֹ הֵן. בִּפְנֵי עֵדִים. וּכְגוֹן שֶׁאָמַר אַתֶּם עֵדַי, דְּשׁוּב אֵינוֹ יָכוֹל לִטְעֹן מְשַׁטֶּה הָיִיתִי בְךָ : פָּטוּר. מִשְּׁבוּעָה דְאוֹרַיְתָא. אֲבָל מַשְׁבִּיעִין אוֹתוֹ הֶסֵּת: אֵין לְךָ בְיָדִי. לֹא הָיוּ דְבָרִים מֵעוֹלָם : חַיָּב. לְשַׁלֵּם. וְאֵינוֹ נֶאֱמָן בִּשְׁבוּעָה, שֶׁהֲרֵי הֻחְזַק כַּפְרָן . וְהוּא שֶׁכָּפַר בְּבֵית דִּין, אֲבָל שֶׁלֹּא בְבֵית דִּין לֹא הֻחְזַק כַּפְרָן:

אֶלָּא לִיטְרָא כָסֶף פָּטוּר. שֶׁאֵין כָּאן הוֹדָאָה מִמִּין הַטַּעֲנָה : לֶתֶךְ. חֲצִי כוֹר. חֲמֵשׁ עֶשְׂרֵה סְאִין: כַּדֵּי שֶׁמֶן. כַּדִּים מְלֵאִים שֶׁמֶן : וְהוֹדָה לוֹ בַקַּנְקַנִּים. כַּדִּים רֵיקִים בְּלֹא שֶׁמֶן: הוֹאִיל וְהוֹדָה בְמִקְצָת הַטַּעֲנָה. דַּהֲוָה לֵיהּ כְּמוֹ טְעָנוֹ חִטִּין וּשְׂעֹרִין וְהוֹדָה לוֹ בְּאֶחָד מֵהֶן. וַהֲלָכָה כְּאַדְמוֹן, אֲבָל אֵין הֲלָכָה כְּרַבָּן גַּמְלִיאֵל דִּמְחַיֵּב בִּטְעָנוֹ חִטִּין וְהוֹדָה לוֹ בִשְׂעֹרִים: פָּטוּר. מִשְּׁבוּעָה דְאוֹרַיְתָא, לֹא עַל הַקַּרְקָעוֹת וְלֹא עַל הַכֵּלִים. שֶׁאֵין הוֹדָאַת קַרְקָעוֹת מְבִיאַתּוּ לִידֵי שְׁבוּעָה, שֶׁאֵין דִּין שְׁבוּעָה בְּקַרְקָעוֹת : הוֹדָה בְמִקְצָת כֵּלִים. דְּאִיכָּא הוֹדָאָה וּכְפִירָה בְּלָאו קַרְקָעוֹת: חַיָּב. לִשָּׁבַע אַף עַל הַקַּרְקָעוֹת עַל יְדֵי גִלְגּוּל שְׁבוּעָה: זוֹקְקִין. גּוֹרְרִין:

אֵין נִשְׁבָּעִין עַל טַעֲנַת חֵרֵשׁ. כְּגוֹן שֶׁטּוֹעֲנוֹ חֵרֵשׁ בִּרְמִיזָה. דְּחֵרֵשׁ שֶׁאָמְרוּ חֲכָמִים בְּכָל מָקוֹם, אֵינוֹ שׁוֹמֵעַ וְאֵינוֹ מְדַבֵּר: וְטַעֲנַת קָטָן. כְּגוֹן שֶׁטּוֹעֲנוֹ קָטָן. דְּאָמַר קְרָא (שְׁמוֹת כב) כִּי יִתֵּן אִישׁ אֶל רֵעֵהוּ, וְאֵין בִּנְתִינַת קָטָן כְּלוּם. וְהַשּׁוֹטֶה כְּקָטָן הוּא. וְדַוְקָא שְׁבוּעָה דְאוֹרַיְתָא הוּא שֶׁאֵין נִשְׁבָּעִין עַל טַעֲנַת קָטָן, אֲבָל שְׁבוּעַת הֶסֵּת חַיָּבִין לִשָּׁבַע עַל טַעֲנָתוֹ : אֲבָל נִשְׁבָּעִין לַקָּטָן וְלַהֶקְדֵּשׁ. הַבָּא לִפָּרַע מִנִּכְסֵי קָטָן, לֹא יִפָּרַע אֶלָּא בִשְׁבוּעָה . וְכֵן הַמַּקְדִּישׁ נְכָסָיו וְיָצָא עָלָיו שְׁטַר חוֹב וּבָא בַעַל חוֹב לִפָּרַע מִן הַנְּכָסִים שֶׁל הֶקְדֵּשׁ, צָרִיךְ שְׁבוּעָה :

הָעֲבָדִים וְהַשְּׁטָרוֹת וְהַקַּרְקָעוֹת וְכוּ'. דִּכְתִיב (שְׁמוֹת כב) עַל כָּל דְּבַר פֶּשַׁע, כְּלָל. עַל שׁוֹר עַל חֲמוֹר עַל שֶׂה עַל שַׂלְמָה, פְּרָט. עַל כָּל אֲבֵדָה, חָזַר וְכָלַל. כְּלָל וּפְרָט וּכְלָל אִי אַתָּה דָן אֶלָּא כְעֵין הַפְּרָט, מַה הַפְּרָט מְפֹרָשׁ דָּבָר הַמִּטַּלְטֵל וְגוּפוֹ מָמוֹן, אַף כָּל דָּבָר הַמִּטַּלְטֵל וְגוּפוֹ מָמוֹן, יָצְאוּ קַרְקָעוֹת שֶׁאֵינָן מִטַּלְטְלִין, יָצְאוּ עֲבָדִים שֶׁהֻקְּשׁוּ לְקַרְקָעוֹת, יָצְאוּ שְׁטָרוֹת שֶׁאַף עַל פִּי שֶׁהֵם מִטַּלְטְלִין אֵין גּוּפָן מָמוֹן. שֶׁכָּל אֵלּוּ אֵין נִשְׁבָּעִין עֲלֵיהֶן. וְהֶקְדֵּשׁ אֵין נִשְׁבָּעִין עָלָיו דִּכְתִיב (שָׁם) כִּי יִתֵּן אִישׁ אֶל רֵעֵהוּ, וְלֹא שֶׁל הֶקְדֵּשׁ: אֵין בָּהֶן תַּשְׁלוּמֵי כֶפֶל. דִּבְהָנֵי דַוְקָא דִּמְרַבִּינַן כְּעֵין הַפְּרָט דִּכְתִיב בְּאוֹתָהּ הַפָּרָשָׁה יְשַׁלֵּם שְׁנַיִם: וְלֹא תַשְׁלוּמֵי אַרְבָּעָה וַחֲמִשָּׁה. דְּכָל הֵיכָא דְלֵיכָּא כֶפֶל אֵינוֹ מְשַׁלֵּם אַרְבָּעָה וַחֲמִשָּׁה דְּכֵיוָן דְּלֵיכָּא כֶפֶל הָווּ לְהוּ תַּשְׁלוּמֵי שְׁלֹשָׁה וְאַרְבָּעָה, וְתַשְׁלוּמֵי אַרְבָּעָה וַחֲמִשָּׁה אָמַר רַחֲמָנָא וְלֹא תַשְׁלוּמֵי שְׁלֹשָׁה וְאַרְבָּעָה: שׁוֹמֵר חִנָּם אֵינוֹ נִשְׁבָּע. שׁוֹמֵר חִנָּם אֵין לוֹ עָלָיו אֶלָּא שְׁבוּעָה, דִּכְתִיב (שָׁם) וְנִקְרַב בַּעַל הַבַּיִת אֶל הָאֱלֹהִים אִם לֹא שָׁלַח יָדוֹ, וְאוֹתָהּ הַפָּרָשָׁה בְּשׁוֹמֵר חִנָּם נֶאֶמְרָה. וְעַל הָעֲבָדִים וְעַל הַקַּרְקָעוֹת וְעַל הַשְּׁטָרוֹת אֵינוֹ נִשְׁבָּע, דְּיָלְפִינַן לַהּ מִדִּכְתִיב כִּי יִתֵּן אִישׁ אֶל רֵעֵהוּ, כְּלָל. כֶּסֶף אוֹ כֵלִים, פְּרָט. לִשְׁמֹר, חָזַר וְכָלַל. כְּלָל וּפְרָט וּכְלָל אִי אַתָּה דָן אֶלָּא כְּעֵין הַפְּרָט, מַה הַפְּרָט מְפֹרָשׁ דָּבָר הַמִּטַּלְטֵל וְגוּפוֹ מָמוֹן, אַף כָּל וְכוּ', יָצְאוּ קַרְקָעוֹת דְּלָאו מִטַּלְטְלִים וְכוּ'. וְכָל הָנֵי דְּאָמְרִינַן בְּהוּ אֵינוֹ נִשְׁבָּע, דַּוְקָא שְׁבוּעָה דְאוֹרַיְתָא, אֲבָל שְׁבוּעַת הֶסֵּת לְעוֹלָם חַיָּבִין לִשָּׁבַע: נוֹשֵׂא שָׂכָר אֵינוֹ מְשַׁלֵּם. גְּנֵבָה וַאֲבֵדָה הַכָּתוּב בּוֹ לְחִיּוּב, דִּכְתִיב (שָׁם) אִם גָּנֹב יִגָּנֵב מֵעִמּוֹ יְשַׁלֵּם לִבְעָלָיו. וַעֲבָדִים וְקַרְקָעוֹת וּשְׁטָרוֹת אֵינוֹ מְשַׁלֵּם, דִּכְתִיב בְּשׁוֹמֵר שָׂכָר (שָׁם) כִּי יִתֵּן אִישׁ אֶל רֵעֵהוּ, כְּלָל. חֲמוֹר אוֹ שׁוֹר אוֹ שֶׂה וְכָל בְּהֵמָה, פְּרָט. לִשְׁמֹר, חָזַר וְכָלַל וְכוּ', כִּדְאָמְרִינַן לְעֵיל. רֵעֵהוּ כְתִיב בְּשׁוֹמֵר חִנָּם וּבְשׁוֹמֵר שָׂכָר, דְּמַשְׁמַע רֵעֵהוּ וְלֹא הֶקְדֵּשׁ. וְשׁוֹאֵל וְשׂוֹכֵר לֹא הֻזְכְּרוּ כָאן לִמְעוּטֵי בְהוּ עֲבָדִים וְקַרְקָעוֹת וּשְׁטָרוֹת וְהֶקְדֵּשׁוֹת, דְּלֹא שַׁיָּךְ שְׁאֵלָה בְּקַרְקָעוֹת , וְלֹא בִשְׁטָרוֹת עַל הָרֹב, וְכָל שֶׁכֵּן שְׂכִירוּת שֶׁאֵינוֹ שַׁיָּךְ בִּשְׁטָרוֹת, וְכֵן לֹא שַׁיָּךְ שְׁאֵלָה וּשְׂכִירוּת בְּהֶקְדֵּשׁ, שֶׁאָסוּר לְהַשְׁאִיל וּלְהַשְׂכִּיר הַהֶקְדֵּשׁוֹת:

רַבִּי מֵאִיר אוֹמֵר יֵשׁ דְּבָרִים שֶׁהֵם כַּקַּרְקַע. עֲנָבִים הָעוֹמְדוֹת לִבָּצֵר הוּא דְאִיכָּא בֵּין חֲכָמִים לְרַבִּי מֵאִיר, דְּרַבִּי מֵאִיר סָבַר עֲנָבִים הָעוֹמְדוֹת לִבָּצֵר כִּבְצוּרוֹת דָּמְיָן, וַחֲכָמִים סָבְרֵי לָאו כִּבְצוּרוֹת דָּמְיָן. וַהֲלָכָה כַּחֲכָמִים. וְדַוְקָא בְּדִין שׁוֹמְרִים , אֲבָל גַּבֵּי מִקָּח וּמִמְכָּר וְאוֹנָאָה וּמוֹדֶה בְמִקְצָת הַטַּעֲנָה, בְּכָל הָנֵי קַיְמָא לָן דְּדָבָר הָעוֹמֵד לִבָּצֵר כְּבָצוּר דָּמִי . וְכֵן הֲלָכָה: עַד הַזִּיז. קוֹרָה שֶׁל עֲלִיָּה הַבּוֹלֶטֶת מִתּוֹךְ הַבַּיִת. וּכְלָלָא דְמִלְּתָא, לְעוֹלָם אֵינוֹ חַיָּב שְׁבוּעָה דְאוֹרַיְתָא עַד שֶׁיִּטְעֲנֶנּוּ דָבָר שֶׁבַּמִּדָּה אוֹ שֶׁבַּמִּשְׁקָל אוֹ שֶׁבַּמִּנְיָן, וְיוֹדֶה לוֹ בְמִקְצָת הַמִּדָּה אוֹ בְמִקְצָת הַמִּשְׁקָל אוֹ מִקְצָת מִנְיָן:

הַמַּלְוֶה אֶת חֲבֵרוֹ עַל הַמַּשְׁכּוֹן. נַעֲשֶׂה עָלָיו שׁוֹמֵר שָׂכָר, לֹא שְׁנָא מִשְׁכְּנוֹ בִשְׁעַת הַלְוָאָתוֹ, לֹא שְׁנָא מִשְׁכְּנוֹ שֶׁלֹּא בִּשְׁעַת הַלְוָאָתוֹ. וְאִם אָבַד אוֹ נִגְנַב וְהָיָה הַמַּשְׁכּוֹן כְּנֶגֶד הַחוֹב, יָצָא מַשְׁכּוֹנוֹ בְחוֹבוֹ וְאֵין לָהֶם זֶה עַל זֶה כְּלוּם. וְאִם הַחוֹב יָתֵר עַל הַמַּשְׁכּוֹן, מְשַׁלֵּם לֹוֶה לַמַּלְוֶה מַה שֶּׁהָיָה הַחוֹב יוֹתֵר. וְאִם הַמַּשְׁכּוֹן יוֹתֵר עַל הַחוֹב, מְשַׁלֵּם מַלְוֶה לַלֹּוֶה. וְאִם נֶאֱבַד בְּאֹנֶס, שֶׁשּׁוֹמֵר שָׂכָר פָּטוּר, גַּם הַמַּלְוֶה פָטוּר, וְנִשְׁבַּע שֶׁבְּאֹנֶס נֶאֱבַד וְגוֹבֶה חוֹבוֹ כֻלּוֹ: שֶׁקֶל. חֲצִי סֶלַע: לֹא כִי. לֹא כֵן הָיָה: אֶלָּא סֶלַע הִלְוִיתַנִי עָלָיו וּשְׁלֹשָׁה דִינָרִים הָיָה שָׁוֶה חַיָּב. שֶׁהֲרֵי הוּא הוֹדָה בְמִקְצָת הַטַּעֲנָה , שֶׁהַסֶּלַע שָׁוֶה אַרְבָּעָה דִינָרִין. וְכֵיוָן שֶׁהַלֹּוֶה חַיָּב לִשָּׁבַע כַּמָּה הָיָה שָׁוֶה הַמַּשְׁכּוֹן שֶׁלּוֹ, וְהַמַּלְוֶה חַיָּב לִשָּׁבַע עַל הַמַּשְׁכּוֹן שֶׁאֵינוֹ בִרְשׁוּתוֹ, וְאַף עַל פִּי שֶׁמְּשַׁלֵּם דָּמָיו, מִשּׁוּם דְּחָיְשִׁינַן שֶׁמָּא עֵינָיו נָתַן בּוֹ, בֵּית דִּין מַשְׁבִּיעִין אֶת הַמַּלְוֶה תְּחִלָּה שֶׁאֵין הַמַּשְׁכּוֹן בִּרְשׁוּתוֹ, וְאַחַר כָּךְ מַשְׁבִּיעִין אֶת הַלֹּוֶה כַּמָּה הָיָה שָׁוֶה. שֶׁמָּא יִשָּׁבַע הַלֹּוֶה תְּחִלָּה וְלֹא דִקְדֵּק בַּשּׁוּמָא וְיוֹצִיא הַמַּלְוֶה אֶת הַמַּשְׁכּוֹן וְיִפְסְלֶנּוּ לְעֵדוּת וְלִשְׁבוּעָה . וְהָא דִתְנִינַן בַּסֵּיפָא מִי נִשְׁבָּע, מִי נִשְׁבָּע תְּחִלָּה קָאָמַר: מִי שֶׁהַפִּקָּדוֹן אֶצְלוֹ. הַמַּלְוֶה שֶׁהָיָה הַמַּשְׁכּוֹן אֶצְלוֹ נִשְׁבָּע שֶׁאֵין הַמַּשְׁכּוֹן בִּרְשׁוּתוֹ:
