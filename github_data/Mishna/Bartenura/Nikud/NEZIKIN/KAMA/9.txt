פרק ט

פרק ט - משנה א
הַגּוֹזֵל. מְשַׁלֵּם כִּשְׁעַת הַגְּזֵלָה. דְּמֵי עֵצִים וְצֶמֶר. וְאֵין חַיָּב לְהַחְזִיר לוֹ כֵּלִים , דְּקָנֵי בְּשִׁנּוּי: דְּמֵי רָחֵל הָעוֹמֶדֶת לִגָּזֵז. וְהָעֹדֶף שֶׁשָּׁוֶה יוֹתֵר הַוָּלָד וְהַגִּזָּה, שֶׁלּוֹ הוּא, דִּקְנַנְּהוּ בְּשִׁנּוּי : זֶה הַכְּלָל. לַאֲתוֹיֵי גָּנַב טָלֶה וְנַעֲשָׂה אַיִל, עֵגֶל וְנַעֲשָׂה שׁוֹר, וּטְבָחוֹ אוֹ מְכָרוֹ, שֶׁהוּא פָּטוּר מִתַּשְׁלוּמֵי אַרְבָּעָה וַחֲמִשָּׁה , שֶׁכֵּיוָן שֶׁנַּעֲשָׂה שִׁנּוּי בְּיָדוֹ, קְנָאוֹ, וְשֶׁלּוֹ הוּא טוֹבֵחַ וְשֶׁלּוֹ הוּא מוֹכֵר:

פרק ט - משנה ב
בַּעֲבָדִים אוֹמֵר לוֹ הֲרֵי שֶׁלְּךָ לְפָנֶיךָ. דִּכְקַרְקַע דָּמוּ וּבִרְשׁוּתָא דְּמָרַיְהוּ קַיְּמֵי. וַהֲלָכָה כְּרַבִּי מֵאִיר : מַטְבֵּעַ וְנִסְדַּק. שִׁנּוּי הַנִּכָּר הוּא: פֵּרוֹת וְהִרְקִיבוּ. כְּשֶׁהִרְקִיבוּ מִקְצָתָן . אֲבָל הִרְקִיבוּ כֻּלָּם, מְשַׁלֵּם כִּשְׁעַת הַגְּזֵלָה: מַטְבֵּעַ וְנִפְסַל. שֶׁאֵינוֹ עוֹבֵר בְּאוֹתָהּ מְדִינָה, אֲבָל עוֹבֵר הוּא בִּמְדִינָה אַחֶרֶת : וְנֶעֶבְדָה בָהּ עֲבֵרָה. שֶׁנִּרְבְּעָה אוֹ נֶעֶבְדָה שֶׁהִיא פְּסוּלָה לְקָרְבָּן: שֶׁנִּפְסְלָה לַמִּזְבֵּחַ. בְּמוּם שֶׁאֵינוֹ נִכָּר כְּגוֹן בְּדוֹקִין שֶׁבָּעַיִן:

פרק ט - משנה ג
נָתַן לְאֻמָּנִין לְתַקֵּן. נָתַן לָהֶם עֵצִים לַעֲשׂוֹת כְּלִי. וּלְאַחַר שֶׁנַּעֲשָׂה הַכְּלִי קִלְקְלוּהוּ, חַיָּבִים לְשַׁלֵּם דְּמֵי הַכְּלִי וְלֹא דְּמֵי עֵצִים בִּלְבַד : נָתַן לְחָרָשׁ. הַשְׁתָּא אַשְׁמוֹעִינַן שֶׁאִם נָתַן כְּלִי עָשׂוּי לְאֻמָּן כְּדֵי לְתַקְּנוֹ, וְקִלְקְלוֹ, חַיָּב לְשַׁלֵּם דְּמֵי הַכְּלִי. וְתָנָא סֵיפָא לְגִלּוּיֵי רֵישָׁא, דְּלָא תֵימָא רֵישָׁא מַיְרֵי בִּכְלִי עָשׂוּי: שִׁדָּה. עֲגָלָה שֶׁל עֵץ הָעֲשׂוּיָה לְמֶרְכַּב נָשִׁים:

פרק ט - משנה ד
וְהִקְדִּיחָתוֹ יוֹרָה. שֶׁשְּׂרָפָתוֹ יוֹרָה. שֶׁהִרְתִּיחוֹ יוֹתֵר מִדַּאי: נוֹתֵן לוֹ דְּמֵי צַמְרוֹ. וְהָכָא לֵיכָּא שְׁבָחָא כְּלָל, שֶׁהֲרֵי נִשְׂרַף לְגַמְרֵי, וְלֵיכָּא לְמֵימַר אִם הַשֶּׁבַח יָתֵר: צְבָעוֹ כָאוּר. כְּמוֹ כָּעוּר. שֶׁצְּבָעוֹ בִּפְסֹלֶת שֶׁל צֶבַע. וּמַזִּיק בְּכַוָּנָה הוּא, לְפִיכָךְ יָדוֹ עַל הַתַּחְתּוֹנָה לְדִבְרֵי הַכֹּל: אִם הַשֶּׁבַח. שֶׁהִשְׁבִּיחַ הַצֶּמֶר: יָתֵר עַל הַיְצִיאָה. שֶׁל צֶבַע. נוֹתֵן לוֹ לַצַּבָּע אֶת הַיְצִיאָה וְלֹא שָׂכָר שָׁלֵם , וּמְקַבֵּל צַמְרוֹ. וּדְמֵי צַמְרוֹ לֹא אָמְרִינַן דְּנֵיתִיב לֵיהּ דְּנִקְנֵי אִיהוּ שְׁבָחָא דְּצֶמֶר, דְּהָא בְּצֶבַע שֶׁהִתְנָה עִמּוֹ צָבַע וְלֵיכָּא שִׁנּוּיָא דְּנִקְנֵי: לִצְבֹּעַ לוֹ אָדֹם וְכוּ'. קַנְיֵהּ בְּשִׁנּוּי. לְרַבִּי מֵאִיר לֹא יָהֵיב אֶלָּא דְּמֵי צַמְרוֹ, אוֹ זֶה יִתֵּן שְׂכָרוֹ מִשָּׁלֵם וְיִקַּח הַצֶּמֶר : רַבִּי יְהוּדָה אוֹמֵר וְכוּ'. דְּקָנֵיס לֵיהּ לְהַאי דְּשִׁנָּה לִהְיוֹת יָדוֹ עַל הַתַּחְתּוֹנָה וְלֹא נִתְהַנֵּי מִשְּׁבָחָא, וְאַגְרָא נַמִּי לֹא נִשְׁקֹל אֶלָּא יְצִיאָה. וְאִם יְצִיאָה יְתֵרָה עַל הַשֶּׁבַח, יִתֵּן לוֹ הַשֶּׁבַח שֶׁהִשְׁבִּיחַ. וְאִם יִרְצֶה לָתֵת אֶת שְׂכָרוֹ מִשָּׁלֵם, שֶׁשֶּׁבַח יוֹתֵר עַל הַשָּׂכָר, יִתֵּן שְׂכָרוֹ. וַהֲלָכָה כְּרַבִּי יְהוּדָה:

פרק ט - משנה ה
נִשְׁבַּע לוֹ. עַל שֶׁקֶר, וְהוֹדָה: יוֹלִיכֶנּוּ אַחֲרָיו אֲפִלּוּ לְמָדַי. דְּאֵין לוֹ כַּפָּרָה עַד שֶׁיַּחְזִיר לַנִּגְזָל עַצְמוֹ. דְּגַבֵּי נִשְׁבַּע עַל שֶׁקֶר כְּתִיב (וַיִּקְרָא ה) לַאֲשֶׁר הוּא לוֹ יִתְּנֶנּוּ : לֹא יִתֵּן לֹא לִבְנוֹ. שֶׁל נִגְזָל . דְּלֹא הָוֵי הֲשָׁבָה עַד דְּמָטֵי לְיָדֵיהּ: לִשְׁלִיחַ בֵּית דִּין. תַּקָּנָה הוּא דְּעַבוּד רַבָּנָן מִפְּנֵי תַּקָּנַת הַשָּׁבִים, שֶׁלֹּא נְחַיְּבֶנּוּ לָזֶה לְהוֹצִיא מָנֶה בְּהוֹצָאַת הַדֶּרֶךְ : וְאִם מֵת. הַנִּגְזָל:

פרק ט - משנה ו
חוּץ מִפָּחוֹת מִשָּׁוֶה פְרוּטָה וְכוּ'. וְלֹא חָיְשִׁינַן שֶׁמָּא יִתְיַקֵּר וְיָשׁוּב עַל שָׁוֶה פְּרוּטָה, וַאֲפִלּוּ הַגָּזֵל בְּעֵין אֵינוֹ צָרִיךְ לְהוֹלִיכוֹ אַחֲרָיו:

פרק ט - משנה ז
וְנִשְׁבַּע לוֹ עַל הַחֹמֶשׁ. שְׁבוּעָה שְׁנִיָּה, שֶׁנְּתָנוֹ לוֹ, וְהוֹדָה שֶׁלֹּא נְתָנוֹ: הֲרֵי זֶה מְשַׁלֵּם חֹמֶשׁ עַל חֹמֶשׁ. חֻמְשׁוֹ שֶׁל חֹמֶשׁ, דְּחֹמֶשׁ רִאשׁוֹן נַעֲשֶׂה קֶרֶן: עַד שֶׁיִּתְמַעֵט וְכוּ'. שֶׁאִם חָזַר וְנָתַן לוֹ חֹמֶשׁ רִאשׁוֹן וְנִשְׁבַּע עַל הַשֵּׁנִי וְהוֹדָה, מְשַׁלֵּם חֻמְשׁוֹ וְחֻמְשׁוֹ שֶׁל חֹמֶשׁ שֵׁנִי, וְכֵן לְעוֹלָם. שֶׁנֶּאֱמַר (שָׁם) וַחֲמִשִׁתָיו יוֹסֵף עָלָיו, הַתּוֹרָה רִבְּתָה חֲמִישִׁיּוֹת הַרְבֵּה לְקֶרֶן אֶחָד: תְּשׂוּמֶת יָד. הַלְוָאָה: עָשַׁק אֶת עֲמִיתוֹ. שְׂכַר שָׂכִיר: מְשַׁלֵּם קֶרֶן. דְּאֵין חֹמֶשׁ וְאָשָׁם אֶלָּא אִם כֵּן הוֹדָה. דִּכְתִיב בְּגֶזֶל הַגֵּר בְּפָרָשַׁת נָשֹׂא, וְהִתְוַדָּה:

פרק ט - משנה ח
תַּשְׁלוּמֵי כֶפֶל. וְאִם הוֹדָה מֵעַצְמוֹ אֵינוֹ מְשַׁלֵּם כֶּפֶל, דִּכְתִיב (שְׁמוֹת כב) אֲשֶׁר יַרְשִׁיעוּן אֱלֹהִים, פְּרָט לְמַרְשִׁיעַ אֶת עַצְמוֹ:

פרק ט - משנה ט
הֲרֵי זֶה מְשַׁלֵּם קֶרֶן וָחֹמֶשׁ לְבָנָיו אוֹ לְאֶחָיו. אִם אֵין לוֹ בָּנִים. וְאַף עַל גַּב דְּנָפְלָה יְרֻשָּׁה קַמֵּיהּ דְּהַאִיךְ, בָּעֵי לְמֶעֱבַד הֲשָׁבָה וְאֵין מְעַכֵּב אֶצְלוֹ אֲפִלּוּ כְּנֶגֶד חֶלְקוֹ. דִּכְתִיב (וַיִּקְרָא ה) וְהֵשִׁיב אֶת הַגְּזֵלָה, אֵין לוֹ תַּקָּנָה עַד שֶׁיּוֹצִיא גְּזֵלוֹ מִתַּחַת יָדוֹ. וּבִלְבַד שֶׁתְּהֵא הַגְּזֵלָה בְּעֵין, דְּלֹא קַנְיֵהּ בְּשִׁנּוּי : אֵין לוֹ. נְכָסִים כָּל כָּךְ שֶׁיָּכוֹל לְוַתֵּר עַל חֶלְקוֹ: אוֹ שֶׁאֵינוֹ רוֹצֶה. לְהַפְסִיד חֶלְקוֹ: לֹוֶה. מֵאֲחֵרִים, וּמַחְזִיר אֶת הַגָּזֵל לְאֶחָיו לְקַיֵּם מִצְוַת הֲשָׁבָה: וּבַעֲלֵי הַחוֹב. שֶׁלָּוָה זֶה הַגַּזְלָן מֵהֶם: בָּאִים וְנִפְרָעִים. מִן הַגְּזֵלָה הַזֹּאת הַחֵלֶק שֶׁיֵּשׁ לַגַּזְלָן בָּהּ. וְאִם אֵין יוֹרֵשׁ לְאָבִיו אֶלָּא הוּא , נוֹתֵן הוּא עַצְמוֹ הַגְּזֵלָה לְבַעַל חוֹבוֹ בְּפִרְעוֹן הַחוֹב, וְצָרִיךְ לְהוֹדִיעוֹ וְלוֹמַר לוֹ זֶה גָּזַל אַבָּא. אוֹ נוֹתֵן עַל דֶּרֶךְ זֶה לִכְתֻבַּת אִשְׁתּוֹ אוֹ לְקֻפַּת הַצְּדָקָה. וּבְכֻלָּן צָרִיךְ שֶׁיּוֹדִיעַ שֶׁזֶּה גֶּזֶל אָבִיו:

פרק ט - משנה י
וְאִם אֵין לוֹ. מַה יֹּאכַל , לֹוֶה וְאוֹכֵל. וּבַעֲלֵי חוֹב בָּאִים וְנִפְרָעִים מִן הַיְרֻשָּׁה אֶת חֶלְקוֹ. וְלֹא חֲשִׁיבָא הֲנָאָה מַה שֶּׁפּוֹרֵעַ חוֹבוֹ בְּאוֹתָם מָעוֹת, דְּהָא תְּנַן בִּנְדָרִים הַמּוּדָר הֲנָאָה מֵחֲבֵרוֹ פּוֹרֵעַ לוֹ חוֹבוֹ:

פרק ט - משנה יא
שֶׁנֶּאֱמַר וְאִם אֵין לָאִישׁ גֹּאֵל. וְאֵין לְךָ אָדָם בְּיִשְׂרָאֵל שֶׁאֵין לוֹ גּוֹאֲלִים לְמַעְלָה עַד יַעֲקֹב אָבִינוּ אֶלָּא זֶה הַגֵּר שֶׁמֵּת וְאֵין לוֹ יוֹרְשִׁין: וּמֵת. הַגַּזְלָן: הַכֶּסֶף יִנָּתֵן לְבָנָיו. שֶׁל גַּזְלָן. שֶׁכְּבָר זָכָה בָּהֶן מִמִּיתַת הַגֵּר, אֶלָּא דַּהֲשָׁבָה בָּעֵי לְמֶעֱבַד כִּי הֵיכִי דְּתֶהֱוֵי לֵיהּ כַּפָּרָה אַשְּׁבוּעֲתֵיהּ, וְהַשְׁתָּא תּוּ לֵיכָּא כַּפָּרָה כֵּיוָן דְּמִית לֵיהּ: וְהָאָשָׁם יִרְעֶה. כְּדִין אָשָׁם שֶׁמֵּתוּ בְּעָלָיו, דְּקָיְמָא לָן כָּל שֶׁבְּחַטָּאת מֵתָה בְּאָשָׁם רוֹעֶה: עַד שֶׁיִּסְתָּאֵב. עַד שֶׁיִּפֹּל בּוֹ מוּם: וְיִפְּלוּ דָמָיו לִנְדָבָה. לְקֵיץ הַמִּזְבֵּחַ לִקְנוֹת מֵהֶם עוֹלוֹת:

פרק ט - משנה יב
לְאַנְשֵׁי מִשְׁמָר וּמֵת. קֹדֶם שֶׁהֵבִיא קָרְבָּן. אֵין יוֹרְשֵׁי הַגַּזְלָן יְכוֹלִים לְהוֹצִיא מִיַּד הַכֹּהֲנִים, מֵאַחַר שֶׁזָּכוּ בָּהֶן כְּבָר : נָתַן. הַגַּזְלָן אֶת הַכֶּסֶף לִיהוֹיָרִיב בְּמִשְׁמַרְתּוֹ, וְאַחַר כָּךְ נָתַן הָאָשָׁם לִידַעְיָה בְּמִשְׁמַרְתּוֹ, יָצָא. כְּדִמְפָרֵשׁ וְאָזֵיל, שֶׁהַמֵּבִיא גְּזֵלוֹ עַד שֶׁלֹּא הֵבִיא קָרְבַּן אֲשָׁמוֹ, יָצָא. וְזֶה הַמִּשְׁמָר זָכָה בְּשֶׁלּוֹ, וְזֶה בְּשֶׁלּוֹ. אֲבָל נָתַן אָשָׁם לִיהוֹיָרִיב, שֶׁהֵבִיא אֲשָׁמוֹ תְּחִלָּה, וְאַחַר כָּךְ הֵבִיא גְּזֵלוֹ וְנָתַן לַמִּשְׁמָר שֶׁלְּאַחֲרָיו, אִם הָאָשָׁם קַיָּם, שֶׁלֹּא הִקְרִיבוּהוּ בְּנֵי יְהוֹיָרִיב, יַקְרִיבֻהוּ בְּנֵי יְדַעְיָה וִיהֵא גְּזֵלוֹ וַאֲשָׁמוֹ לִידַעְיָה. וְאִם אֵין אֲשָׁמוֹ קַיָּם, לֹא יָצָא בָּאָשָׁם שֶׁנָּתַן לִיהוֹיָרִיב, כֵּיוָן שֶׁלֹּא נָתַן גְּזֵלוֹ. וְיַחְזֹר וְיָבִיא אָשָׁם אַחֵר: לִיהוֹיָרִיב. הִיא מִשְׁמָרָה רִאשׁוֹנָה שֶׁל עֶשְׂרִים וְאַרְבַּע מִשְׁמְרוֹת כְּהֻנָּה שֶׁבַּמִּקְדָּשׁ. וְשֶׁל יְדַעְיָה אַחֲרֶיהָ: נָתַן הַקֶּרֶן. לַכֹּהֲנִים: אֵין הַחֹמֶשׁ מְעַכֵּב. מִלְּהַקְרִיב אֶת הָאָשָׁם, אִם לֹא נָתַן עֲדַיִן וּלְבַסּוֹף נָתַן :
