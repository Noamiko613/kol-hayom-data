פרק ג - משנה א
בַּכֹּל מְעָרְבִין. עֵרוּבֵי תְּחוּמִין . דְּאִלּוּ עֵרוּבֵי חֲצֵרוֹת אֵין מְעָרְבִין אֶלָּא בַּפַּת. וְהַאי בַּכֹּל דִּתְנַן בְּמַתְנִיתִין, לָאו דַּוְקָא, כִּדְאָמְרִינַן בַּגְּמָרָא אֵין לְמֵדִין מִן הַכְּלָלוֹת אֲפִלּוּ בְּמָקוֹם שֶׁנֶּאֱמַר בָּהֶן חוּץ, דְּהָא הָכָא תְּנַן בַּכֹּל מְעָרְבִין חוּץ מִן הַמַּיִם וּמִן הַמֶּלַח, וְאִיכָּא נַמִּי כְּמֵהִין וּפִטְרִיּוֹת דְּאֵין מְעָרְבִין וְלֹא מִשְׁתַּתְּפִים בָּהֶם, דְּלָאו מָזוֹן נִינְהוּ, כְּמוֹ מַיִם וּמֶלַח: וּמִשְׁתַּתְּפִים. שִׁתּוּפֵי מְבוֹאוֹת: הַכֹּל נִקָּח בְּכֶסֶף מַעֲשֵׂר. כְּדִכְתִיב (דְּבָרִים יד) וְנָתַתָּ הַכֶּסֶף בְּכָל אֲשֶׁר תְּאַוֶּה נַפְשֶׁךָ: חוּץ מִן הַמַּיִם וּמִן הַמֶּלַח. דְּלָאו פְּרִי מִפְּרִי הוּא: הַנּוֹדֵר מִן הַמָּזוֹן. לֹא דְּאָמַר יֵאָסֵר עָלַי מָזוֹן, דְּאֵין נִקְרָא מָזוֹן אֶלָּא חֲמֵשֶׁת הַמִּינִים דְּזָיְנֵי וְסַעֲדֵי, אֶלָּא כְּגוֹן דְּאָמַר כָּל הַזָּן עָלַי קוֹנָם, וְכָל מִלֵּי זָיְנֵי וּמַשְׂבִּיעִים לְפִי שָׁעָה חוּץ מִן הַמַּיִם וְהַמֶּלַח: מְעָרְבִין לְנָזִיר בְּיַיִן. דְּאַף עַל גַּב דְּלָא חֲזִי לְדִידֵיהּ, חֲזִי לְאַחֲרִינֵי: וּלְיִשְׂרָאֵל בִּתְרוּמָה. דְּהָא חַזְיָא לַכֹּהֲנִים: סוּמְכוֹס אוֹמֵר בְּחֻלִּין. דְּמִידֵי דַּחֲזִי לֵיהּ בָּעִינַן. וּבַיַּיִן לַנָּזִיר לֹא פָלֵיג סוּמְכוֹס, מִפְּנֵי שֶׁאֶפְשָׁר שֶׁיִּשָּׁאֵל הַנָּזִיר עַל נִדְרוֹ וְיִהְיֶה הַיַּיִן מֻתָּר לוֹ בְּאוֹתָהּ שַׁבָּת, אֲבָל תְּרוּמָה אִי אֶפְשָׁר שֶׁתִּהְיֶה רְאוּיָה לְיִשְׂרָאֵל, שֶׁאֲפִלּוּ יִשָּׁאֵל הַמַּפְרִישׁ הַתְּרוּמָה עָלֶיהָ וַהֲרֵי הִיא כְּאִלּוּ לֹא הוּרְמָה, הֲרֵי הִיא חוֹזֶרֶת לְטִבְלָהּ וְאָסוּר לֶאֱכֹל מִמֶּנָּה עַד שֶׁיַּחְזֹר וְיַפְרִישׁ תְּרוּמָה אַחֶרֶת, וְאֵין מַפְרִישִׁין תְּרוּמָה בְּשַׁבָּת אֲפִלּוּ בֵּין הַשְּׁמָשׁוֹת, הִלְכָּךְ לֹא חַזְיָא לֵיהּ. וְאֵין הֲלָכָה כְּסוּמְכוֹס: וּלְכֹהֵן בְּבֵית הַפְּרָס. סְתָמָא הִיא, וְלָאו סוּמְכוֹס קָאֲמַר לַהּ. בֵּית הַפְּרַס, שָׂדֶה שֶׁנֶּחְרַשׁ בָּהּ קֶבֶר מֻתָּר לְכֹהֵן לִכָּנֵס שָׁם כְּשֶׁהוּא מְנַפֵּחַ וְהוֹלֵךְ וְרוֹאֶה שֶׁאֵינוֹ נוֹגֵעַ בְּעֶצֶם כִּשְׂעוֹרָה. וּמִשּׁוּם הָכִי מֻתָּר לִתֵּן עֵרוּבוֹ שָׁם, שֶׁהֲרֵי יָכוֹל לְהִכָּנֵס בַּמָּקוֹם שֶׁנָּתַן שָׁם עֵרוּבוֹ וְהָוֵי הוּא וְעֵרוּבוֹ בְּמָקוֹם אֶחָד: לָחוֹץ. לַעֲשׂוֹת מְחִצָּה בֵּינוֹ לַקֶּבֶר שֶׁלֹּא יַאֲהִיל, כְּגוֹן שֶׁיִּכָּנֵס שָׁם בְּשִׁדָּה תֵּבָה וּמִגְדָּל , אַלְמָא חֲזִי לֵיהּ. וּפְלֻגְתָּא דְּרַבִּי יְהוּדָה וְרַבָּנָן מוֹקְמִינַן בַּגְּמָרָא דַּאֲפִלּוּ בְּיִשְׂרָאֵל פְּלִיגֵי רַבָּנָן עֲלֵיהּ דְּרַבִּי יְהוּדָה וְאָמְרֵי דְּאֵין מַנִּיחִין עֵרוּבֵי תְּחוּמִין בְּבֵית הַקְּבָרוֹת, וְלֹא נָקַט כֹּהֵן אֶלָּא לְהוֹדִיעֲךָ כֹּחוֹ דְּרַבִּי יְהוּדָה דַּאֲפִלּוּ בְּכֹהֵן שָׁרֵי. וּבְהָא קָמִפְלְגֵי, דְּרַבִּי יְהוּדָה סָבַר אַף עַל גַּב דְּאָסוּר לֶהֱנוֹת וּלְהִשְׁתַּמֵּשׁ בְּבֵית הַקְּבָרוֹת , מֻתָּר לְהַנִּיחַ שָׁם עֵרוּב, דְּאֵין מְעָרְבִין עֵרוּבֵי תְּחוּמִין אֶלָּא לִדְבַר מִצְוָה וּמִצְווֹת לָאו לֶהֱנוֹת נִתְּנוּ, וְאַף עַל פִּי שֶׁמִּשְׁתַּמֵּר שָׁם הָעֵרוּב אַחַר קְנִיָּתוֹ, דְּהָא בֵּין הַשְּׁמָשׁוֹת קוֹנֶה וּכְבָר נַעֲשֵׂית הַמִּצְוָה וְהוּא מִשְׁתַּמֵּר שָׁם כָּל הַשַּׁבָּת, קָסָבַר רַבִּי יְהוּדָה אֵין אָדָם מַקְפִּיד עַל עֵרוּבוֹ אִם אָבַד אוֹ נִגְנַב לְאַחַר שֶׁקָּנָה. וְרַבָּנָן סָבְרֵי אָדָם מַקְפִּיד עַל עֵרוּבוֹ לְאַחַר שֶׁקָּנָה, וְרוֹצֶה שֶׁלֹּא יִגָּנֵב, הִלְכָּךְ כְּשֶׁהָעֵרוּב מִשְׁתַּמֵּר בְּבֵית הַקְּבָרוֹת כָּל הַשַּׁבָּת לְאַחַר שֶׁקָּנָה בֵּין הַשְּׁמָשׁוֹת וְנִגְמְרָה מִצְוָתוֹ נִמְצָא מִשְׁתַּמֵּשׁ בְּאִסּוּרֵי הֲנָאָה, שֶׁהַקְּבָרוֹת אֲסוּרִים בַּהֲנָאָה, לְפִיכָךְ לֹא יַנִּיחַ אָדָם עֵרוּב בְּבֵית הַקְּבָרוֹת. וַהֲלָכָה כַּחֲכָמִים:

מְעָרְבִין בִּדְמַאי. בְּכִכָּר שֶׁלְּקָחָהּ מֵעַם הָאָרֶץ וְלֹא הִפְרִישׁ מִמֶּנָּה תְּרוּמַת מַעֲשֵׂר שֶׁל דְּמַאי: וּבְמַעֲשֵׂר רִאשׁוֹן כוּ' אֲבָל לֹא בְטֶבֶל וְכוּ'. כֻּלָּן מְפֹרָשִׁין בְּמַסֶּכֶת שַׁבָּת פֶּרֶק מְפַנִּין: הַשּׁוֹלֵחַ אֶת עֵרוּבוֹ בְּיַד חֵרֵשׁ כוּ'. לְהוֹלִיכוֹ לְסוֹף אַלְפַּיִם אַמָּה. וְדַוְקָא בְּעֵרוּבֵי תְּחוּמִין אֵינוֹ עֵרוּב אִם שְׁלָחוֹ בְּיַד קָטָן, אֲבָל בְּעֵרוּבֵי חֲצֵרוֹת קָיְמָא לָן קָטָן גּוֹבֶה אֶת הָעֵרוּב : בְּיַד מִי שֶׁאֵינוֹ מוֹדֶה בָעֵרוּב. כְּגוֹן כּוּתִי אוֹ צְדוֹקִי: וְאִם אָמַר לְאַחֵר. כָּשֵׁר, לְקַבְּלוֹ מִן הַפָּסוּל וְיוֹלִיכֶנּוּ הַכָּשֵׁר לְסוֹף אַלְפַּיִם אַמָּה: הֲרֵי זֶה עֵרוּב. וּבִלְבַד שֶׁיְּהֵא הוּא עוֹמֵד וְרוֹאֶה בְּשָׁעָה שֶׁהַפָּסוּל נוֹתְנוֹ לְיַד הַכָּשֵׁר, אַף עַל פִּי שֶׁאֵינוֹ רוֹאֶה כְּשֶׁהַכָּשֵׁר מוֹלִיכוֹ, דַּחֲזָקָה שָׁלִיחַ עוֹשֶׂה שְׁלִיחוּתוֹ:

נְתָנוֹ בְאִילָן. הָעוֹמֵד בִּרְשׁוּת הָרַבִּים וְיֵשׁ בּוֹ אַרְבָּעָה טְפָחִים עַל אַרְבָּעָה טְפָחִים אוֹ יוֹתֵר: לְמַעְלָה מֵעֲשָׂרָה אֵין עֵרוּבוֹ עֵרוּב. דְּכֵיוָן דְּהָאִילָן רָחָב אַרְבָּעָה, לְמַעְלָה מֵעֲשָׂרָה הָוֵי רְשׁוּת הַיָּחִיד, וְהוּא קָנָה שְׁבִיתָתוֹ בִּרְשׁוּת הָרַבִּים, וְכֵיוָן דְּאִי הָוֵי בָּעֵי לְמִשְׁקַל עֵרוּבֵיהּ וּמֵיכְלֵיהּ בְּשָׁעָה שֶׁהָעֵרוּב זוֹכֶה לוֹ, דְּהַיְנוּ בֵּין הַשְּׁמָשׁוֹת, לֹא מָצֵי שָׁקֵיל לֵיהּ דְּמַיְתֵי מֵרְשׁוּת הַיָּחִיד לִרְשׁוּת הָרַבִּים, לֹא הָוֵי עֵרוּב: לְמַטָּה מֵעֲשָׂרָה הֲרֵי זֶה עֵרוּב. דִּלְמַטָּה מֵעֲשָׂרָה כַּרְמְלִית הִיא, דְּכָל מִשְּׁלֹשָׁה וְעַד תִּשְׁעָה בְּרֹחַב אַרְבָּעָה כַּרְמְלִית הִיא, וְכֵיוָן דְּאִי שָׁקֵיל לֵיהּ לְעֵרוּבֵיהּ אֵין כָּאן אֶלָּא אִסּוּר שְׁבוּת, הֲרֵי זֶה עֵרוּב. וּמַתְנִיתִין רַבִּי הִיא דְּאָמַר כָּל דָּבָר שֶׁהוּא מִשּׁוּם שְׁבוּת לֹא גָּזְרוּ עָלָיו בֵּין הַשְּׁמָשׁוֹת . וּבְשָׁעָה שֶׁהָעֵרוּב זוֹכֶה לוֹ דְּהַיְנוּ בֵּין הַשְּׁמָשׁוֹת מָצֵי שָׁקֵיל לֵיהּ, וְנִמְצָא הוּא וְעֵרוּבוֹ בְּמָקוֹם אֶחָד, הִלְכָּךְ הָוֵי עֵרוּב: נְתָנוֹ בְּבוֹר. שֶׁבַּכַּרְמְלִית כְּגוֹן שֶׁבַּבִּקְעָה וְשֶׁבַּשָּׂדוֹת, וְנִתְכַּוֵּן לִשְׁבֹּת בַּבִּקְעָה אוֹ בַּשָּׂדֶה, הֲרֵי זֶה עֵרוּב, דְּבוֹר עַצְמוֹ רְשׁוּת הַיָּחִיד וְהוּא קָנָה שְׁבִיתָה בַּכַּרְמְלִית, וּבְשָׁעָה שֶׁהָעֵרוּב זוֹכֶה לוֹ דְּהַיְנוּ בֵּין הַשְּׁמָשׁוֹת מָצֵי שָׁקֵיל לֵיהּ, כְּרַבִּי דְּאָמַר כָּל דָּבָר שֶׁהוּא מִשּׁוּם שְׁבוּת לֹא גָּזְרוּ עָלָיו בֵּין הַשְּׁמָשׁוֹת: אֲפִלּוּ גָּבוֹהַּ מֵאָה אַמָּה. וְאֵין לְמַטָּה רֹחַב אַרְבָּעָה, דְּלָאו רְשׁוּת הַיָּחִיד הוּא, וְאַף עַל גַּב דִּלְמַעְלָה רָחָב אַרְבָּעָה, דְּהָא עֵרוּב עַל גַּבֵּי מְקוֹם אַרְבָּעָה בָּעִינַן: תָּלוּשׁ וְנָעוּץ. הוּא דַּהֲוֵי עֵרוּב, אֲבָל מְחֻבָּר לֹא הָוֵי עֵרוּב, מִשּׁוּם דְּכִי שָׁקֵיל לֵיהּ גְּזֵרָה שֶׁמָּא יִקְטֹם, לְפִי שֶׁהַקָּנֶה רַךְ הוּא אִיכָּא לְמִגְזַר שֶׁמָּא יִקְטֹם כִּי שָׁקֵיל לֵיהּ, אֲבָל אִילָן קָשֶׁה הוּא וּבֵין הַשְּׁמָשׁוֹת לְשֶׁמָּא יַעֲלֶה וְיִתְלֹשׁ לֹא חָיְשִׁינַן, אֲבָל קְטִימַת קָנֶה אַיְדֵי דִּרְכִיכָא וַדַּאי מִקְּטְמָא וּמִחַיַּב מִשּׁוּם קוֹצֵר. אִי נַמִּי קָנֶה וְקֻנְדָּס מִחְלַף לֵיהּ מְחֻבָּר בְּתָלוּשׁ, דְּקָנִים הַרְבֵּה תְּלוּשִׁין נְעוּצִים וְנִרְאִין כִּמְחֻבָּרִים, מִשּׁוּם הָכִי אִיכָּא לְמִגְזַר שֶׁמָּא יִקְטֹם בִּמְחֻבָּר וְיַחְשֹׁב שֶׁהוּא תָּלוּשׁ, אֲבָל בְּאִילָן לֵיכָּא לְמִגְזַר שֶׁמָּא יַעֲלֶה וְיִתְלֹשׁ וְיִטְעֶה לַחְשֹׁב דְּתָלוּשׁ הוּא: נְתָנוֹ בְמִּגְדָּל. שֶׁל עֵץ: וְאָבַד הַמַּפְתֵּחַ הֲרֵי זֶה עֵרוּב. וּכְגוֹן שֶׁהַמַּנְעוּל קָשׁוּר בְּחוּטִים וַחֲבָלִים, שֶׁאִם לֹא יִמָּצֵא הַמַּפְתֵּחַ אִי אֶפְשָׁר לְחָתְכָן אֶלָּא בְּסַכִּין. תַּנָּא קַמָּא סָבַר הוֹאִיל וְאֶפְשָׁר לִפְתֹּחַ עַל יְדֵי שֶׁיַּחְתֹּךְ הַחֲבָלִים בְּסַכִּין אֵין כָּאן אִסּוּר סְקִילָה אֶלָּא אִסּוּר שְׁבוּת, דִּמְקַלְקֵל הוּא בַּחֲתִיכַת הַחֲבָלִים וְכָל הַמְקַלְקְלִים פְּטוּרִים, וְאִסּוּר שְׁבוּת בֵּין הַשְּׁמָשׁוֹת בִּשְׁעַת קְנִיַּת הָעֵרוּב לֹא גָּזְרוּ בּוֹ כִּדְאָמְרָן, הִלְכָּךְ הָוֵי עֵרוּב, דְּמָצֵי מַיְתֵי סַכִּין וְיַחְתֹּךְ לַחֲבָלִים וְשָׁקֵיל לְעֵרוּבֵיהּ. וְרַבִּי אֱלִיעֶזֶר סָבַר אֵין כְּלִי נִטָּל אֶלָּא לְצֹרֶךְ תַּשְׁמִישׁוֹ, וְאָסוּר לְטַלְטֵל הַסַּכִּין כְּדֵי לַחְתֹּךְ הַחֲבָלִים, שֶׁאֵין תַּשְׁמִישׁוֹ אֶלָּא לַחְתֹּךְ הָאֳכָלִים, וְהוֹאִיל וְאִיכָּא תַּרְתֵּי, טִלְטוּל הַסַּכִּין, וַחֲתִיכַת הַחֲבָלִים, אֲפִלּוּ לְרַבִּי דְּאָמַר כָּל דָּבָר שֶׁהוּא מִשּׁוּם שְׁבוּת בֵּין הַשְּׁמָשׁוֹת לֹא גָזְרוּ, בִּכְהַאי גַּוְנָא מוֹדֶה דְּגָזְרוּ. וְאֵין הֲלָכָה כְּרַבִּי אֱלִיעֶזֶר:

פרק ג - משנה ד
נִתְגַּלְגֵּל חוּץ לַתְּחוּם אֵינוֹ עֵרוּב. הוֹאִיל וְיֵשׁ מִבֵּיתוֹ שֶׁהוּא לָן שָׁם עַד עֵרוּבוֹ יוֹתֵר מֵאַלְפַּיִם אַמָּה, לֹא מָצֵי לְמֵיזַל וּלְמִשְׁקְלֵיהּ. וְהוּא שֶׁנִּתְגַּלְגֵּל שְׁתֵּי אַמּוֹת חוּץ לְאַלְפַּיִם אַמָּה, שֶׁכָּל אָדָם יֵשׁ לוֹ אַרְבַּע אַמּוֹת מִמְּקוֹם עֵרוּבוֹ, שְׁתֵּי אַמּוֹת מִצַּד מִזְרָחוֹ שֶׁל עֵרוּב וּשְׁתֵּי אַמּוֹת מִצַּד מַעֲרָבוֹ: נָפַל עָלָיו גַּל. וְהוּא דְּבָעֵי מָרָא וַחֲצִינָא לְפִנּוּיֵהּ, דְּהַוְיָא מְלָאכָה וְלֹא שְׁבוּת: תְּרוּמָה וְנִטְמֵאת. דְּהַשְׁתָּא לֹא חַזְיָא לֹא לְדִידֵיהּ וְלֹא לְאַחֲרִינֵי. וְאִצְטְרִיךְ תַּנָּא לְאַשְׁמְעִינַן נִתְגַּלְגֵּל חוּץ לַתְּחוּם וְנָפַל עָלָיו גַּל, דְּאִי אַשְׁמְעִינַן נִתְגַּלְגֵּל, מִשּׁוּם דְּלֵיתֵיהּ גַּבֵּיהּ הָוֵי הוּא בְּמָקוֹם אֶחָד וְעֵרוּבוֹ בְּמָקוֹם אַחֵר, אֲבָל נָפַל עָלָיו גַּל דְּאִיתֵיהּ גַּבֵּיהּ אֵימָא לֶהֱוֵי עֵרוּב. וְאִי אַשְׁמְעִינַן נָפַל עָלָיו גַּל דְּלֹא מָצֵי שָׁקֵיל לֵיהּ אֶלָּא בִּמְלָאכָה בְּמָרָא וַחֲצִינָא, אֲבָל נִתְגַּלְגֵּל דְּאֶפְשָׁר דְּנָשֵׁיב זִיקָא וּמַהְדַּר לֵיהּ לְתוֹךְ הַתְּחוּם אֵימָא לֵיהֱוֵי עֵרוּב, קָא מַשְׁמַע לָן. וְתָנָא נִשְׂרַף לְהוֹדִיעֲךָ כֹּחוֹ דְּרַבִּי יוֹסֵי, דְּאַף עַל גַּב דְּאֵינוֹ בָּעוֹלָם מִשּׁוּם סָפֵק לֹא מִתְסַר. וְתָנָא תְּרוּמָה שֶׁנִּטְמֵאת לְהוֹדִיעֲךָ כֹּחוֹ דְּרַבִּי מֵאִיר, דְּאַף עַל גַּב דְּיֶשְׁנוֹ בָּעוֹלָם וְיֵשׁ לוֹמַר הַעֲמִידֶנָּה בֵּין הַשְּׁמָשׁוֹת עַל חֶזְקָתָהּ שֶׁל קֹדֶם לָכֵן וּטְהוֹרָה הָיְתָה, אֲפִלּוּ הָכִי לֹא אָמְרִינַן חֲזָקָה לְקֻלָּא: מִשֶּׁחֲשֵׁכָה הֲרֵי זֶה עֵרוּב. דִּלְאַחַר שֶׁקָּנָה בֵּין הַשְּׁמָשׁוֹת אֵין חוֹשְׁשִׁין לוֹ אִם נֶאֱבַד: הֲרֵי זֶה חַמָּר גַּמָּל. דִּמְסַפְקָא לָן אִי קָנָה לֵיהּ עֵרוּב וְהָכָא הַוְיָא בֵּיתוֹ וּמֵהָכָא יֵשׁ לוֹ אַלְפַּיִם לְכָל רוּחַ, וְהִפְסִיד אַלְפַּיִם שֶׁמֵּעֵבֶר בֵּיתוֹ וָהָלְאָה, אוֹ שֶׁמָּא לֹא קָנָה עֵרוּב וּמִבֵּיתוֹ יֵשׁ לוֹ אַלְפַּיִם לְכָל רוּחַ, וְלֹא קָנָה לְעֵבֶר עֵרוּבוֹ כְּלוּם, וּמֵחֲמַת סָפֵק זֶה נֶאֱסַר לָלֶכֶת אֶלָּא אַלְפַּיִם אַמָּה שֶׁבֵּין בֵּיתוֹ לָעֵרוּב, דְּמִמַּה נַּפְשָׁךְ בְּהָנָךְ מִשְׁתְּרֵי, אֲבָל בָּאַלְפַּיִם דְּמִעֵרוּבוֹ וָאֵילָךְ לֹא, דִּלְמָא לֹא קָנָה עֵרוּב. וּמִבֵּיתוֹ וָאֵילָךְ נַמִּי לֹא, דִּלְמָא קָנָה עֵרוּב. נִמְצָא זֶה מוֹשְׁכוֹ לְכָאן וְזֶה מוֹשְׁכוֹ לְכָאן, כְּאָדָם הַמַּנְהִיג חֲמוֹר וְגָמָל, שֶׁהַחֲמוֹר הוֹלֵךְ לְפָנָיו וְזֶה מַנְהִיגוֹ, וְהַגָּמָל מֵאֲחוֹרָיו וְהוּא מוֹשְׁכוֹ, וְצָרִיךְ לִפְנוֹת לְפָנָיו וּלְאַחֲרָיו: רַבִּי יוֹסֵי וְרַבִּי שִׁמְעוֹן אוֹמְרִים סְפֵק עֵרוּב כָּשֵׁר. דְּאָמְרִינַן הַעֲמֵד עֵרוּב עַל חֶזְקָתוֹ, וּכְשֶׁהֵנִיחוֹ הָיָה בְּתוֹךְ הַתְּחוּם וְהָיָה טָהוֹר וְלֹא הָיָה עָלָיו גַּל, לְפִיכָךְ עֵרוּבוֹ עֵרוּב. וְכֵן הֲלָכָה:

פרק ג - משנה ה
מַתְנֶה אָדָם עַל עֵרוּבוֹ. מֵנִיחַ שְׁנֵי עֵרוּבִין , אֶחָד לְסוֹף אַלְפַּיִם לְמִזְרַח בֵּיתוֹ וְאֶחָד לְסוֹף אַלְפַּיִם לְמַעֲרַב בֵּיתוֹ, וְאוֹמֵר אִם בָּאוּ עוֹבְדֵי כּוֹכָבִים לַמִּזְרָח וְצָרִיךְ אֲנִי לִבְרֹחַ מִפְּנֵיהֶם יִקְנֶה לִי עֵרוּבִי שֶׁבַּמַּעֲרָב וְיִהְיוּ לִי לְמַעֲרַב בֵּיתִי אַרְבַּעַת אֲלָפִים אַמָּה. וְאַף עַל גַּב דְּלֹא אָתוּ עַד לְמָחָר, אָמְרִינַן יֵשׁ בְּרֵרָה , דְּבֵין הַשְּׁמָשׁוֹת קָנָה לֵיהּ עֵרוּבָא דְּאִידָךְ גִּיסָא: הֲרֵינִי כִבְנֵי עִירִי. אַלְפַּיִם מֵעִירִי לְכָל רוּחַ, וְאֵינִי צָרִיךְ לְהִשְׂתַּכֵּר לְכָאן וּלְהַפְסִיד מִכָּאן: אִם בָּא חָכָם לַמִּזְרָח. חוּץ לִתְחוּם עִירִי וְרוֹצֶה אֲנִי לִלְמֹד מִפִּיו, וְעַכְשָׁיו אֵינִי יוֹדֵעַ לְאֵיזֶה צַד יָבֹא, וּלְמָחָר אֶשְׁמַע מִבְּנֵי אָדָם הַבָּאִים מִשָּׁם לְכָאן עַל יְדֵי עֵרוּב: רַבִּי יְהוּדָה אוֹמֵר. אִם בָּאוּ מִכָּאן וּמִכָּאן, וְאֶחָד מֵהֶם רַבּוֹ, יֵלֵךְ אֵצֶל רַבּוֹ וְלֹא אֵצֶל הָאַחֵר, דְּקִים לָן בְּגַוֵּיהּ דְּבִשְׁעַת קְנִיַּת עֵרוּב בֵּין הַשְּׁמָשׁוֹת דַּעְתֵּיהּ דְּלִקְנֵי לֵיהּ הַהוּא עֵרוּב דִּלְצַד רַבֵּיהּ. וְאֵין הֲלָכָה כְּרַבִּי יְהוּדָה, דְּזִמְנִין דְּנִיחָא לֵיהּ לֶאֱינַשׁ בְּחַבְרֵיהּ טְפֵי מֵרַבֵּיהּ:

פרק ג - משנה ו
מְעָרֵב אָדָם שְׁנֵי עֵרוּבִין. אִם הָיָה צָרִיךְ לֵילֵךְ יוֹם רִאשׁוֹן לְכָאן וְיוֹם שֵׁנִי לְכָאן, יָכוֹל לְעָרֵב לַמִּזְרָח וְלַמַּעֲרָב עֶרֶב יוֹם הָרִאשׁוֹן , וְאוֹמֵר עֵרוּבִי שֶׁל מִזְרָח יִקְנֶה לִי הַיּוֹם לְצֹרֶךְ מָחָר, וְעֵרוּבִי שֶׁל מַעֲרָב יִקְנֶה לִי בֵּין הַשְּׁמָשׁוֹת דִּלְמָחָר לְצֹרֶךְ יוֹם שֵׁנִי. דְּקָסָבַר רַבִּי אֱלִיעֶזֶר שַׁבָּת וְיוֹם טוֹב לָאו כְּחַד יוֹמָא אֲרִיכָא נִינְהוּ אֶלָּא שְׁתֵּי קְדֻשּׁוֹת , וּבֵין הַשְּׁמָשׁוֹת דְּקַמָּא לְדִידֵיהּ הוּא דְּקָנֵי וְלֹא לְיוֹם שֵׁנִי: עֵרוּבִי בָרִאשׁוֹן וּבַשֵּׁנִי כִּבְנֵי עִירִי. כְּלוֹמַר וְאִם לֹא הָיָה צָרִיךְ לֵילֵךְ אֶלָּא בְּיוֹם רִאשׁוֹן, וּבַשֵּׁנִי אֵין צָרִיךְ לָזוּז מִמְּקוֹמוֹ וְאֵינוֹ רוֹצֶה לְהַפְסִיד לֹא אַלְפַּיִם שֶׁל תְּחוּמוֹ מִכָּאן וְלֹא אַלְפַּיִם שֶׁל תְּחוּמוֹ מִכָּאן , יְעָרֵב עֵרוּב אֶחָד לַצַּד שֶׁהוּא רוֹצֶה לֵילֵךְ בּוֹ בְּיוֹם רִאשׁוֹן, וְאוֹמֵר עֵרוּבִי זֶה יִקְנֶה לִי לְצֹרֶךְ מָחָר, וּבַשֵּׁנִי הֲרֵינִי כִּבְנֵי עִירִי שֶׁלֹּא עֵרְבוּ. אוֹ אִם הָיָה צָרִיךְ לֵילֵךְ בַּשֵּׁנִי וְלֹא בָּרִאשׁוֹן, יֹאמַר עֵרוּב זֶה יִקְנֶה לִי בֵּין הַשְּׁמָשׁוֹת שֶׁל מָחָר וּבָרִאשׁוֹן אֶהֱיֶה כִּבְנֵי עִירִי: אוֹ מְעָרֵב לִשְׁנֵי הַיָּמִים. הַיְנוּ לְרוּחַ אַחַת דְּאָמְרֵי לְעֵיל. וְהָא תּוּ לָמָּה לִי, אֶלָּא הָכִי קָאָמְרֵי לֵיהּ רַבָּנָן לְרַבִּי אֱלִיעֶזֶר, אִי אַתָּה מוֹדֶה לְיוֹם אֶחָד דְּאוֹ מְעָרֵב לְרוּחַ אַחַת אוֹ אֵינוֹ מְעָרֵב כָּל עִקָּר, שֶׁאֵינוֹ יָכוֹל לוֹמַר חֲצִי יוֹם עֵרוּבִי לַמִּזְרָח וַחֲצִי יוֹם עֵרוּבִי לַמַּעֲרָב. לִשְׁנֵי הַיָּמִים נַמִּי, דְּהַיְנוּ לְשַׁבָּת וְיוֹם טוֹב, אוֹ מְעָרֵב כְּמוֹ לְיוֹם אֶחָד אוֹ אֵינוֹ מְעָרֵב כָּל עִקָּר. דִּסְפֵקָא לְהוּ לְרַבָּנָן שַׁבָּת וְיוֹם טוֹב אִי כְּיוֹמָא אֲרִיכָא דָּמוּ אִי לֹא, וְעָבְדֵי הָכָא לְחֻמְרָא דְּאֵין יָכוֹל לְעָרֵב לִשְׁתֵּי רוּחוֹת דִּלְמָא חֲדָא קְדֻשָּׁה הִיא, וְהָכָא לְחֻמְרָא דְּאָמְרֵי לְקַמָּן נֶאֱכַל בָּרִאשׁוֹן אֵין עֵרוּב לַשֵּׁנִי, דִּלְמָא שְׁתֵּי קְדֻשּׁוֹת הֵן וְלֹא כְּיוֹמָא אֲרִיכָא דָּמוּ: מוֹלִיכוֹ. הַשָּׁלִיחַ עֶרֶב יוֹם טוֹב שֶׁלִּפְנֵי שַׁבָּת וּמַחְשִׁיךְ עָלָיו עַד שֶׁיִּקְנֶה הָעֵרוּב: וְנוֹטְלוֹ וּבָא לוֹ. שֶׁמָּא יֵאָבֵד וְשׁוּב אֵין לוֹ עֵרוּב לַשֵּׁנִי, כִּדְקָתָנֵי נֶאֱכַל עֵרוּבוֹ בָּרִאשׁוֹן עֵרוּב לָרִאשׁוֹן וְאֵין עֵרוּב לַשֵּׁנִי: בַּשֵּׁנִי. חוֹזֵר וּמוֹלִיכוֹ בַּשֵּׁנִי. דְּקָיְמָא לָן הַמְעָרֵב בְּפַת בָּרִאשׁוֹן וְרוֹצֶה לְעָרֵב בְּפַת בַּשֵּׁנִי צָרִיךְ שֶׁיְּעָרֵב בְּאוֹתוֹ עֵרוּב עַצְמוֹ שֶׁקָּרָא עָלָיו שֵׁם מֵאֶתְמוֹל, וְעַתָּה שׁוֹתֵק, אֲבָל לֹא בְּפַת אַחֶרֶת, דְּבָעֵי לְמִקְרָא עֲלֵיהּ הַשְׁתָּא שֵׁם עֵרוּב, וְהָוֵי מֵכִין מִיּוֹם טוֹב לְשַׁבָּת: וְנִמְצָא מִשְׂתַּכֵּר בַּהֲלִיכָתוֹ. כְּלוֹמַר קוֹנֶה לוֹ הֲלִיכָה לְיוֹם מָחָר: וּמִשְׂתַּכֵּר בְּעֵרוּבוֹ. שֶׁאוֹכְלוֹ. וְיוֹם טוֹב אַחַר הַשַּׁבָּת דְּלֹא אֶפְשָׁר בְּהָכִי, מוֹלִיכוֹ בָּרִאשׁוֹן וְאֵינוֹ אוֹכְלוֹ, וְחוֹזֵר וְהוֹלֵךְ שָׁם בַּשֵּׁנִי לִרְאוֹת, אִם קַיָּם הָעֵרוּב מַחְשִׁיךְ עָלָיו, וְאִם רוֹצֶה יֹאכְלֶנּוּ אַחֲרֵי כֵן: מוֹדִים אַתֶּם לִי שֶׁהֵן שְׁתֵּי קְדֻשּׁוֹת. דְּאָמְרִיתוּ נֶאֱכַל עֵרוּב בָּרִאשׁוֹן אֵין עֵרוּב לַשֵּׁנִי, וְאִי חֲדָא קְדֻשָּׁה נִינְהוּ הֲוָה לֵיהּ כְּחַד יוֹמָא אֲרִיכְתָא וְיִקְנֶה בֵּין הַשְּׁמָשׁוֹת שֶׁל רִאשׁוֹן לִשְׁנֵיהֶם, וְכֵיוָן דִּשְׁתֵּי קְדֻשּׁוֹת נִינְהוּ יָכוֹל לְעָרֵב נַמִּי לִשְׁתֵּי רוּחוֹת. וְרַבָּנָן אָזְלֵי הָכָא לְחֻמְרָא וְהָכָא לְחֻמְרָא כִּדְאָמְרָן, מִשּׁוּם דִּמְסַפְּקָא לְהוּ. וַהֲלָכָה כְּרַבִּי אֱלִיעֶזֶר דְּשַׁבָּת וְיוֹם טוֹב שְׁתֵּי קְדֻשּׁוֹת הֵן:

פרק ג - משנה ז
שֶׁמָּא תִּתְעַבֵּר. שֶׁמָּא יַעֲשׂוּ בֵּית דִּין הַגָּדוֹל אֱלוּל מְעֻבָּר וְיִהְיוּ שְׁנֵי יָמִים טוֹבִים, יוֹם ל' שֶׁל אֱלוּל וְיוֹם ל"א, וְהוּא צָרִיךְ לֵילֵךְ בָּרִאשׁוֹן לְצַד אֶחָד וּבַשֵּׁנִי לְצַד אַחֵר, מְעָרֵב שְׁנֵי עֵרוּבִין וּמְנִיחָן בְּעֶרֶב יוֹם טוֹב זֶה לְכָאן וְזֶה לְכָאן וְאוֹמֵר וְכוּ': וְלֹא הוֹדוּ לוֹ חֲכָמִים. דְּקָסָבְרֵי קְדֻשָּׁה אַחַת הֵן. וַהֲלָכָה כַּחֲכָמִים בִּשְׁנֵי יָמִים טוֹבִים שֶׁל רֹאשׁ הַשָּׁנָה. דְּלָאו מִשּׁוּם סְפֵקָא בִּלְבַד אִתְּקַן שֶׁמָּא קִדְּשׁוּ בֵּית דִּין אֶת יוֹם שְׁלֹשִׁים אוֹ יוֹם ל"א וְאֶחָד מֵהֶם חֹל, אֶלָּא מִשּׁוּם דְּשֶׁמָּא בָּאוּ עֵדִים מִן הַמִּנְחָה וּלְמַעְלָה, שֶׁנּוֹהֲגִין אוֹתוֹ הַיּוֹם קֹדֶשׁ וּלְמָחָר קֹדֶשׁ וְתַרְוַיְהוּ קְדֻשָּׁה אַחַת. אֲבָל בִּשְׁאָר יָמִים טוֹבִים שֶׁל גָּלֻיּוֹת דְּלֹא אַתְקוּן אֶלָּא מִסְּפֵיקָא דְּלֹא יָדְעִינַן מָתַי קִדְּשׁוּ בֵּית דִּין הַגָּדוֹל אֶת הַחֹדֶשׁ, מוֹדִים חֲכָמִים לְרַבִּי יְהוּדָה דִּשְׁתֵּי קְדֻשּׁוֹת הֵן. וַחֲכָמִים הָאֲמוּרִים כָּאן הוּא רַבִּי יוֹסֵי :

פרק ג - משנה ח
מַתְנֶה אָדָם עַל הַכַּלְכָּלָה. סַל שֶׁיֵּשׁ בּוֹ פֵּרוֹת טְבָלִים, מַתְנֶה עָלָיו בְּיוֹם טוֹב רִאשׁוֹן שֶׁל רֹאשׁ הַשָּׁנָה וְאוֹמֵר, אִם הַיּוֹם חֹל תְּהֵא זוֹ תְּרוּמָה עַל אֵלּוּ, וְאִם הַיּוֹם קֹדֶשׁ אֵין בִּדְבָרַי כְּלוּם. שֶׁאֵין מַגְבִּיהִין תְּרוּמָה בְּיוֹם טוֹב. וּלְמָחָר הוּא אוֹמֵר אִם אֶתְמוֹל קֹדֶשׁ וְהַיּוֹם חֹל תְּהֵא זוֹ שֶׁאָמַרְתִּי אֶתְמוֹל תְּרוּמָה עַל אֵלּוּ, וְאִם הַיּוֹם קֹדֶשׁ וְאֶתְמוֹל חֹל כְּבָר הִיא תְּרוּמָה. וְאוֹכֵל כַּלְכָּלָה הַמְּתֻקֶּנֶת וּמְשַׁיֵּר הַתְּרוּמָה: וְלֹא הוֹדוּ לוֹ חֲכָמִים. הוּא רַבִּי יוֹסֵי כִּדְאָמְרָן, דְּסָבַר קְדֻשָּׁה אַחַת הֵן וְאֵין מַגְבִּיהִים בָּהֶם תְּרוּמָה. וְדַוְקָא בִּשְׁנֵי יָמִים טוֹבִים שֶׁל רֹאשׁ הַשָּׁנָה כְּדִפְרִישִׁית לְעֵיל: וְכֵן בֵּיצָה שֶׁנּוֹלְדָה בָרִאשׁוֹן. שֶׁל רֹאשׁ הַשָּׁנָה: תֵּאָכֵל בַּשֵּׁנִי. דְּאִם הָרִאשׁוֹן קֹדֶשׁ הַשֵּׁנִי חֹל, וּבֵיצָה שֶׁנּוֹלְדָה בְּיוֹם טוֹב מֻתֶּרֶת בְּחֹל, דְּאֵין הֲכָנָה לְחֹל. וְאִם הָרִאשׁוֹן חֹל וְהַשֵּׁנִי קֹדֶשׁ נִמְצָא שֶׁחוֹל מֵכִין לְיוֹם טוֹב וְשַׁפִּיר דָּמִי: וְלֹא הוֹדוּ לוֹ חֲכָמִים. בִּשְׁנֵי יָמִים טוֹבִים שֶׁל רֹאשׁ הַשָּׁנָה בִּלְבַד, דִּקְדֻשָּׁה אַחַת הֵן. אֲבָל בִּשְׁנֵי יָמִים טוֹבִים שֶׁל גָּלֻיּוֹת מוֹדִים לוֹ :

פרק ג - משנה ט
וְהַחֲלִיצֵנוּ. זָרְזֵנוּ וְהַחֲלִיצֵנוּ כֹּחַ, כְּמוֹ נֵחָלֵץ חוּשִׁים (בְּמִדְבָּר לב). פֵּרוּשׁ אַחֵר הַצִּילֵנוּ וּמַלְּטֵנוּ, כְּמוֹ חַלְּצֵנִי ה' מֵאָדָם רַע (תְּהִלִּים קמ): אִם הַיּוֹם אִם מָחָר. אִם הַיּוֹם הוּא הַחֲלִיצֵנוּ הַיּוֹם, וְאִם מָחָר הוּא הַחֲלִיצֵנוּ מָחָר: וְלֹא הוֹדוּ לוֹ חֲכָמִים. לֹא לְהַזְכִּיר שֶׁל רֹאשׁ חֹדֶשׁ בְּרֹאשׁ הַשָּׁנָה וְלֹא לְהַתְנוֹת אִם הַיּוֹם וְאִם מָחָר, אֶלָּא אוֹמֵר וְהַחֲלִיצֵנוּ סְתָמָא בִּשְׁנֵי הַיָּמִים וְאֵינוֹ מַזְכִּיר שֶׁל רֹאשׁ חֹדֶשׁ כָּל עִקָּר. וַהֲלָכָה כַּחֲכָמִים:
