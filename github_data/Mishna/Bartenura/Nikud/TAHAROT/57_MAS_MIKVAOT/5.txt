פרק ה - משנה א
מַעְיָן שֶׁהֶעֱבִירוֹ עַל גַּבֵּי הַשֹּׁקֶת. שֶׁהוּא כְלִי, וְלֹא שֹׁקֶת שֶׁבַּסֶּלַע: פָּסוּל. לְהַטְבִּיל בַּמַּיִם שֶׁבְּתוֹךְ הַשֹּׁקֶת וּבַמַּיִם שֶׁיּוֹצְאִים וְעוֹבְרִים מִן הַשֹּׁקֶת לַחוּץ, הוֹאִיל וּמֵי הַמַּעְיָן כֻּלָּם שׁוֹפְכִים לְתוֹךְ הַשֹּׁקֶת. אֲבָל אִם מֵי הַמַּעְיָן עוֹבְרִים עַל שְׂפַת הַשֹּׁקֶת כָּל שֶׁהוּא, אַף עַל פִּי שֶׁמִּקְצָתָם עוֹבְרִים בְּתוֹךְ הַשֹּׁקֶת, כָּשֵׁר לְהַטְבִּיל בַּמַּיִם שֶׁעַל שְׂפַת הַשֹּׁקֶת אוֹ חוּצָה לָהּ, דְּמַעְיָן מְטַהֵר בְּכָל שֶׁהוּא: וְהִפְסִיקוֹ. שֶׁאֵין מֵי הַבְּרֵכָה מְחֻבָּרִין לַמַּעְיָן: הֲרֵי הוּא כְמִקְוֶה. וְאֵין מְטַהֵר בְּכָל שֶׁהוּא, עַד שֶׁיְּהֵא שָׁם אַרְבָּעִים סְאָה : חָזַר וְהִמְשִׁיכוֹ. שֶׁחִבֵּר מֵי הַמַּעְיָן לַבְּרֵכָה: פָּסוּל לַזָּבִים. דִּכְבָר פָּסַק חַיּוּתָן. וְזָבִים וּמְצֹרָעִים וּמֵי חַטָּאת כְּתִיב בְּהוּ מַיִם חַיִּים:

פרק ה - משנה ב
הֶעֱבִירוֹ עַל גַּבֵּי כֵלִים. שֶׁהֶעֱבִיר הַמַּעְיָן עַל דָּפְנֵי הַכְּלִי מִבַּחוּץ : הֲרֵי הוּא כְמוֹ שֶׁהָיָה. לְטַהֵר בְּזוֹחֲלִים וּבְכָל שֶׁהוּא: הֲרֵי הוּא כְמִקְוֶה. דְּבָעֵי אַרְבָּעִים סְאָה וּבְאַשְׁבֹּרֶן: וּבִלְבַד שֶׁלֹּא יַטְבִּיל עַל גַּבֵּי הַסַּפְסָל. גְּזֵרָה שֶׁמָּא יַטְבִּיל בְּכֵלִים . וַהֲלָכָה כְּרַבִּי יוֹסֵי:

פרק ה - משנה ג
שֶׁהוּא מָשׁוּךְ כְּנָדָל. שֶׁמֵּימָיו זוֹחֲלִים בִּמְקוֹמוֹת הַרְבֵּה, כְּשֶׁרֶץ זֶה שֶׁמַּרְבֶּה רַגְלַיִם, וְקוֹרִין לוֹ בְלַעַ"ז צנטופיד"י: רִבָּה עָלָיו. מַיִם שְׁאוּבִים, עַד שֶׁנִּמְשְׁכוּ וְנִתְגַּדְּלוּ הַמְּקוֹמוֹת שֶׁמֵּימָיו זוֹחֲלִים שָׁם: הֲרֵי הוּא כְמוֹ שֶׁהָיָה. הַמְּקוֹמוֹת הַלָּלוּ מְטַהֲרִים בְּזוֹחֲלִים וּבְכָל שֶׁהוּא כְּמוֹ שֶׁהָיוּ בַתְּחִלָּה: הָיָה הַמַּעְיָן עוֹמֵד. וְלֹא הָיוּ מֵימָיו נִזְחָלִים, וְרִבָּה עָלָיו מַיִם שְׁאוּבִים וְנִתְגַּדֵּל הַמַּעְיָן עַד שֶׁמֵּימָיו נִזְחָלִים אֵילָךְ וָאֵילָךְ מֵחֲמַת הַמַּיִם שְׁאוּבִין: שָׁוֶה וְכוּ' לְמַעְיָן. הַמְּקוֹמוֹת הַלָּלוּ שֶׁהַמַּיִם נִזְחָלִים בָּהֶם, שָׁוִין לְמַעְיָן שֶׁמְּטַהֲרִים בְּכָל שֶׁהֵן, וְשָׁוִים לְמִקְוֶה שֶׁאֵין מְטַהֲרִים בְּזוֹחֲלִין אֶלָּא בְאַשְׁבֹּרֶן. פֵּרוּשׁ אַחֵר, רִבָּה עָלָיו וְהִמְשִׁיכוֹ, הִרְחִיב הַבְּרֵכוֹת שֶׁהָיוּ מוֹשְׁכוֹת מַיִם מִן הַמַּעְיָן הַהוּא וְהִגְדִּילָן , הֲרֵי הוּא כְמוֹת שֶׁהָיָה, דְּבַמָּקוֹם שֶׁהָיוּ מֵימָיו זוֹחֲלִים קֹדֶם שֶׁרִבָּה עֲלֵיהֶן וְהִרְחִיבָן, שָׁם מְטַהֵר בְּזוֹחֲלִין, וּמִן הַצְּדָדִים אֵין מְטַהֵר אֶלָּא בְאַשְׁבֹּרֶן. אֲבָל אִם לֹא הָיוּ מֵימָיו זוֹחֲלִים אֶלָּא עוֹמְדִים, וְרִבָּה עָלָיו, שֶׁהִרְחִיב מְקוֹם עֲמִידָתָן, וְהִמְשִׁיכָם, שֶׁעָשָׂה מְקוֹם עֲמִידָתָן זוֹחֲלִין, שָׁוֶה לְמִקְוֶה לְטַהֵר בְּאַשְׁבֹּרֶן, וְאֵין מְטַהֵר בְּזוֹחֲלִין, מֵאַחַר דְּעִקַּר מְקוֹם הַמַּעְיָן אֵינָן זוֹחֲלִין:

פרק ה - משנה ד
הַיָּם הַגָּדוֹל כְּמִקְוֶה. לֹא קָרָא הַכָּתוּב מִקְוֶה אֶלָּא לַיָּם הַגָּדוֹל, דְּבֵיהּ מִשְׁתַּעֵי קְרָא בְּמַעֲשֵׂה בְרֵאשִׁית, שֶׁשָּׁם נִקְווּ כָל מֵימֵי בְרֵאשִׁית: לֹא נֶאֱמַר יַמִּים. בִּלְשׁוֹן רַבִּים, אֶלָּא מִפְּנֵי שֶׁמְּעֹרָבִין בּוֹ מִינֵי יַמִּים הַרְבֵּה, שֶׁכָּל הַנְּחָלִים הוֹלְכִים אֵלָיו: רַבִּי יוֹסֵי אוֹמֵר כָּל הַיַּמִּים. וְיָם הַגָּדוֹל נַמִּי תוֹרַת מַעְיָן עֲלֵיהֶם לָעִנְיָן שֶׁהֵם מְטַהֲרִין בְּזוֹחֲלִין, מִפְּנֵי שֶׁהַנְּחָלִים הוֹלְכִים וְזוֹחֲלִין עֲלֵיהֶם. אֲבָל פְּסוּלִין הֵן לְתוֹרַת מַיִם חַיִּים, שֶׁהַכָּתוּב קְרָאָן מִקְוֶה. וַהֲלָכָה כְּרַבִּי יוֹסֵי:

פרק ה - משנה ה
הַזּוֹחֲלִין. כְּגוֹן נְהָרוֹת שֶׁהֵן זוֹחֲלִין, הֲרֵי הֵן כְּמַעְיָן וּמְטַהֲרִין בְּזוֹחֲלִין וּבְכָל שֶׁהֵן: וְהַנּוֹטְפִים. כְּגוֹן מֵי גְשָׁמִים: כְּמִקְוֶה. לְטַהֵר בְּאַרְבָּעִים סְאָה וּבְאַשְׁבֹּרֶן: שֶׁהֵם כְּשֵׁרִים. כְּזוֹחֲלִין. הוֹאִיל וְנוֹטְפִין מֻעָטִים. וַאֲפִלּוּ בְמָקוֹם שֶׁאֵין כְּדֵי טְבִילָה בַזּוֹחֲלִין אֶלָּא אִם כֵּן נוֹטְפִין מַשְׁלִימִים לָהֶם : וְנוֹטְפִין שֶׁעֲשָׂאָן זוֹחֲלִין. כְּגוֹן מִקְוֶה שֶׁפָּרַץ עַל שְׂפָתוֹ וּמֵימָיו יוֹצְאִין וְזוֹחֲלִין : סוֹמֵךְ אֲפִלּוּ מַקֵּל. אוֹ קָנֶה, אוֹ זָב אוֹ זָבָה סוֹמְכִין בַּיָּד אוֹ בָרֶגֶל, וְסוֹתְמִין מְקוֹם יְצִיאַת הַמַּיִם עַד שֶׁיַּעַמְדוּ הַמַּיִם בְּמָקוֹם אֶחָד וְנַעֲשִׂים אַשְׁבֹּרֶן, וְיוֹרֵד הַטָּמֵא וְטוֹבֵל: כָּל דָּבָר שֶׁהוּא מְקַבֵּל טֻמְאָה. לֹא מִבַּעְיָא זָב וְזָבָה שֶׁהֵן טְמֵאִין, שֶׁאֵינָן יְכוֹלִים לִסְתֹּם מְקוֹם יְצִיאַת הַמַּיִם בִּידֵיהֶם אוֹ בְרַגְלֵיהֶם כְּדֵי לְהַעֲמִיד הַמַּיִם וְלַעֲשׂוֹתָן אַשְׁבֹּרֶן, אֶלָּא אֲפִלּוּ טָהוֹר גָּמוּר, הוֹאִיל וְהוּא מְקַבֵּל טֻמְאָה: אֵין מַזְחִילִין בּוֹ. כְּלוֹמַר אֵין מַעֲמִידִין בּוֹ מְקוֹם הַזְּחִילָה כְּדֵי שֶׁיֵּעָשֶׂה אַשְׁבֹּרֶן וְיוּכַל לִטְבֹּל בּוֹ. וְטַעֲמָא דְרַבִּי יוֹסֵי, דְּאָמַר קְרָא מִקְוֵה מַיִם יִהְיֶה טָהוֹר, הֲוָיָתָן תְּהֵא עַל יְדֵי טָהֳרָה. וַהֲלָכָה כְּרַבִּי יוֹסֵי:

פרק ה - משנה ו
גַּל שֶׁנִּתְלַשׁ. מִן הַיָּם שֶׁהוֹלֵךְ וְסוֹעֵר: וְנָפַל עַל הָאָדָם וְעַל הַכֵּלִים טְהוֹרִים. לְחֻלִּין בִּלְבַד, דְּחֻלִּין לֹא בָעוּ כַוָּנָה. אֲבָל לְמַעֲשֵׂר שֵׁנִי וְלִתְרוּמָה, לֹא טָהַר הַטּוֹבֵל עַד שֶׁיְּכַוֵּן: טוֹבְלִין. אִישׁ וְאִשָּׁה: וּמַטְבִּילִין. כֵּלִים אוֹ יָדַיִם בְּמָקוֹם שֶׁצָּרִיךְ טְבִילַת יָדַיִם, כְּגוֹן לַקֹּדֶשׁ דְּלֹא סַגִּי בִנְטִילָה: חֲרִיצִין. רְחָבִים וּמְרֻבָּעִים כִּמְעָרָה, אֶלָּא שֶׁאֵינָן מְקֹרִין: נְעִיצִין. קְצָרִים מִלְּמַטָּה וּרְחָבִים מִלְּמַעְלָה: פַּרְסַת חֲמוֹר. וְהוּא הַדִּין פַּרְסַת כָּל בְּהֵמָה: הַמְעֹרֶבֶת בַּבִּקְעָה. גֻּמָּא שֶׁנַּעֲשֵׂית בַּקַּרְקַע עַל יְדֵי פַרְסַת רֶגֶל שֶׁל בְּהֵמָה וְנִתְקַבְּצוּ בָהּ מַיִם, וְהִיא מְעֹרֶבֶת לְמִקְוֶה שֶׁל אַרְבָּעִים סְאָה עַל יְדֵי נֶקֶב כִּשְׁפוֹפֶרֶת הַנּוֹד, מַטְבִּילִין בָּהּ: חַרְדָּלִית. כְּמוֹ הַר דָּלִית. זֶרֶם מַיִם הַבָּאִים מִדָּלְיוֹ שֶׁל הַר, פֵּרוּשׁ מִגֹּבַהּ הָהָר. וַאֲפִלּוּ אֵין מִתְּחִלָּתָן וְעַד סוֹפָן אֶלָּא אַרְבָּעִים סְאָה, מַטְבִּילִין בָּהֶן לְבֵית שַׁמַּאי: וּבֵית הִלֵּל אוֹמְרִים אֵין מַטְבִּילִין. עַד שֶׁיִּהְיוּ אַרְבָּעִים סְאָה בְּמָקוֹם אֶחָד, שֶׁאֵין חַרְדָּלִית מְטַהֶרֶת אֶלָּא בְאַשְׁבֹּרֶן: שֶׁגּוֹדֵר כֵּלִים וְטוֹבֵל בָּהֶם. עוֹשֶׂה מְחִצָּה בְכֵלִים כְּדֵי שֶׁיְּעַכְּבוּ הִלּוּךְ הַמַּיִם וְלֹא יִהְיוּ זוֹחֲלִין, וְטוֹבֵל בְּאוֹתָן הַמַּיִם שֶׁבְּתוֹךְ מְחִצַּת הַכֵּלִים שֶׁהֵן כְּמֵי מִקְוֶה: וְכֵלִים שֶׁגָּדַר בָּהֶן לֹא הֻטְבְּלוּ. לְפִי שֶׁלֹּא הֻטְבַּל כָּל הַכְּלִי בַּמַּיִם שֶׁבְּתוֹךְ הַמְּחִצָּה, כִּי אִם הַצַּד שֶׁכְּלַפֵּי הַחַרְדָּלִית וְלֹא צַד שֶׁכְּלַפֵּי חוּץ. וְאִי קַשְׁיָא, וְהָא אָמַר רַבִּי יוֹסֵי לְעֵיל כָּל דָּבָר שֶׁהוּא מְקַבֵּל טֻמְאָה אֵין מַזְחִילִין בּוֹ, וְהֵיכִי פָלֵיג אַבֵּית הִלֵּל שֶׁמּוֹדִים שֶׁגּוֹדֵר כֵּלִים וְטוֹבֵל בָּהֶן. יֵשׁ לוֹמַר, דְּהַאי וּמוֹדִים אַסֵּיפָא קָאֵי, דְּאָמַר וְכֵלִים שֶׁגָּדַר בָּהֶן לֹא הֻטְבְּלוּ, וְהָכִי קָאָמַר, מוֹדִים בֵּית שַׁמַּאי לְבֵית הִלֵּל שֶׁהַגּוֹדֵר כֵּלִים וְטָבַל בָּהֶן, כֵּלִים שֶׁגָּדַר בָּהֶן לֹא הֻטְבְּלוּ. אֲבָל בֵּית הִלֵּל לְעוֹלָם סָבְרֵי שֶׁאֵין גּוֹדֵר כֵּלִים וְטוֹבֵל בָּהֶן, כְּרַבִּי יוֹסֵי דְאָמַר כָּל דָּבָר שֶׁהוּא מְקַבֵּל טֻמְאָה אֵין מַזְחִילִין בּוֹ. וְהָכִי מַשְׁמַע בַּתּוֹסֶפְתָּא:
