פרק ו

כָּל הַמְעֹרָב. חוֹרֵי הַמְּעָרָה. חוֹרִין שֶׁבְּתוֹךְ הַמִּקְוֶה. וּמִשּׁוּם שֶׁרֹב טְבִילוֹת שֶׁלָּהֶן בִּמְעָרוֹת, קָרֵי לְמִקְוֶה מְעָרָה: כְּמָה שֶׁהֵן. אַף עַל פִּי שֶׁאֵין נְקוּבִין כִּשְׁפוֹפֶרֶת הַנּוֹד: עֻקַּת הַמְּעָרָה. גֻּמָּא שֶׁבְּשׁוּלֵי הַמִּקְוֶה, אֵין מֵימֶיהָ נֶחֱשָׁבִין עִם מֵימֵי הַמִּקְוֶה שֶׁלְּמַעְלָה מֵהֶן לִהְיוֹת כְּשֵׁרִים לִטְבֹּל בָּהֶן, אֶלָּא אִם כֵּן יֵשׁ בֵּינֵיהֶן כִּשְׁפוֹפֶרֶת הַנּוֹד לְחַבְּרָן: אָמַר רַבִּי יְהוּדָה אֵימָתַי, לֹא בָא רַבִּי יְהוּדָה לַחֲלֹק אֶלָּא לְפָרֵשׁ: בִּזְמַן שֶׁהִיא מַעֲמֶדֶת אֶת עַצְמָהּ. כְּשֶׁבָּא לִטְבֹּל בָּעֻקָּה, הָעֻקָּה הִיא מַעֲמֶדֶת אֶת עַצְמָהּ, וְאֵין הַגַּג הַמַּפְסִיק בֵּינָהּ וּבֵין הַמְּעָרָה נוֹפֵל מֵאֵלָיו: אֲבָל אִם אֵינָהּ מַעֲמֶדֶת אֶת עַצְמָהּ. וּבְשָׁעָה שֶׁבָּא לִטְבֹּל, נוֹפֵל גַּג הָעֻקָּה, שֶׁהוּא תַּחְתִּית הַמִּקְוֶה, וּמִתְעָרְבִים מֵימֵי הַמְּעָרָה בְמֵי הָעֻקָּה: מַטְבִּילִין בָּהּ כְּמָה שֶׁהִיא. אֲפִלּוּ אֵין בַּנֶּקֶב כִּשְׁפוֹפֶרֶת הַנּוֹד:

וְהִטְבִּילָן הֲרֵי אֵלּוּ טְהוֹרִים. וַאֲפִלּוּ אֵין פִּי הַדְּלִי רָחָב כִּשְׁפוֹפֶרֶת הַנּוֹד. וּכְגוֹן שֶׁהַדְּלִי עַצְמוֹ טָמֵא וְצָרִיךְ טְבִילָה, דְּמִגּוֹ דְסָלְקָא לֵיהּ טְבִילָה לַדְּלִי, סָלְקָא לֵיהּ נַמִּי לְמַה שֶּׁבְּתוֹכוֹ: וְאִם לֹא טָבַל. גָּרְסִינַן. כְּלוֹמַר וְאִם הַדְּלִי טָהוֹר וְאֵינוֹ מַטְבִּילוֹ, שֶׁאֵינוֹ צָרִיךְ טְבִילָה: עַד שֶׁיְּהוּ מְעֹרָבִין כִּשְׁפוֹפֶרֶת הַנּוֹד. אֵין הַכֵּלִים שֶׁבְּתוֹכוֹ טְהוֹרִים עַד שֶׁיִּהְיֶה בְפִי הַדְּלִי כִשְׁפוֹפֶרֶת הַנּוֹד:

בָּזֶה עֶשְׂרִים וּבָזֶה עֶשְׂרִים. מַיִם כְּשֵׁרִים, וּבַשְּׁלִישִׁי עֶשְׂרִים סְאָה מַיִם שְׁאוּבִין: וְטָבְלוּ וְנִתְעָרְבוּ. עַל יְדֵי טְבִילָתָן צָפוּ הַמַּיִם וְהִשִּׁיקוּ וְנִתְחַבְּרוּ שְׁלָשְׁתָּן זֶה בָזֶה: הַמִּקְוָאוֹת טְהוֹרִים. אֲפִלּוּ הַשָּׁאוּב. שֶׁהֲרֵי שְׁנֵי הַכְּשֵׁרִים מִתְעָרְבִים יַחַד וְיֵשׁ כָּאן שִׁעוּר מִקְוֶה אַרְבָּעִים סְאָה מַיִם כְּשֵׁרִים, וּכְשֶׁמִּתְחַבֵּר לָהֶן הַשָּׁאוּב שֶׁהַמַּיִם צָפִין עַל גַּבֵּיהֶן מֵחֲמַת הַטּוֹבְלִים, הֻכְשְׁרוּ מֵימָיו לִטְבִילָה: הָיָה הַשָּׁאוּב בָּאֶמְצַע. מֵחֲמַת שֶׁהַשָּׁאוּב מַפְסִיק בֵּין שְׁנַיִם הַחִיצוֹנִים הַכְּשֵׁרִים, אֵין הַכְּשֵׁרִים מִתְעָרְבִין זֶה עִם זֶה: הַמִּקְוָאוֹת כְּמוֹת שֶׁהָיוּ. שֶׁהַכְּשֵׁרִים כְּשֵׁרִים לְהִקָּווֹת עֲלֵיהֶן לְהַשְׁלִים אַרְבָּעִים סְאָה כְּשֵׁרִים, וְהַשָּׁאוּב אֵין כָּשֵׁר לְהִקָּווֹת עָלָיו. וְאַף עַל פִּי שֶׁמֵּימֵי הַשָּׁאוּב מִתְעָרְבִים עִם כָּל אֶחָד, לֹא מִפַּסְלֵי לְהִקָּווֹת עֲלֵיהֶן, דְּדֶרֶךְ הַמְשָׁכָה נִתְעָרְבוּ וְלֹא נָפְלוּ מִן הַכְּלִי לַכְּשֵׁרִין, וּמַה שֶּׁיָּרְדוּ מִמֶּנּוּ לְכָל אֶחָד , נִתְבַּטֵּל בְּרֹב קֹדֶם שֶׁיָּרְדוּ לַמִּקְוֶה :

שֶׁהָיוּ בָהֶן שְׁלֹשָׁה לֻגִּין מַיִם. בְּלוּעִין וּדְבוּקִין בָּהֶן , אֲבָל אֵינָן נִרְאִין בְּעֵין, הִלְכָּךְ לֹא גָזְרוּ רַבָּנָן, דְּאֵין זֶה מִקְּרֵי נָפְלוּ:

הַשִּׁדָּה וְהַתֵּבָה. כֵּלִים גְּדוֹלִים שֶׁל עֵץ וּמֻנָּחִים בְּתוֹךְ מִקְוֶה כָשֵׁר: אֶלָּא אִם כֵּן הָיוּ נְקוּבִים כִּשְׁפוֹפֶרֶת הַנּוֹד. דּוּמְיָא דְעֻקָּה דְאָמְרִינַן לְעֵיל שֶׁצָּרִיךְ שֶׁתְּהֵא נְקוּבָה כִשְׁפוֹפֶרֶת הַנּוֹד: בִּכְלִי גָדוֹל. כְּגוֹן בַּר תִּשְׁעָה טְפָחִים אוֹ יוֹתֵר, דְּמִעוּטוֹ הָוֵי אַרְבָּעָה טְפָחִים, סַגִּי בִנְקִיבַת אַרְבָּעָה טְפָחִים . וְאֵין הֲלָכָה כְּרַבִּי יְהוּדָה: שַׂק אוֹ קֻפָּה. סַל עָשׂוּי מִנְּצָרִים שֶׁל עֲרָבָה קְלוּפִין: מַטְבִּילִין בָּהֶן כְּמוֹ שֶׁהֵן. אִם הֵן בַּיָּם. וְלֹא בָעוּ נְקִיבָה כִשְׁפוֹפֶרֶת הַנּוֹד, שֶׁכֻּלָּם מְלֵאִים נְקָבִים וּבְטֵלִים מַיִם שֶׁבְּתוֹכָן לְגַבֵּי מֵי מִקְוֶה: אֵינָן פּוֹסְלִין אֶת הַמִּקְוֶה. מִשּׁוּם מַיִם שְׁאוּבִין: וּמַעֲלִין אוֹתָן כְּדַרְכָּן. כְּגוֹן בְּמִקְוֶה שֶׁיֶּשׁ בּוֹ אַרְבָּעִים סְאָה מְצֻמְצָמוֹת דְּאָמְרִינַן לְקַמָּן דְּכַר וְכֶסֶת שֶׁל עוֹר כֵּיוָן שֶׁהִגְבִּיהַּ שִׂפְתוֹתֵיהֶם מִן הַמַּיִם, הַמַּיִם שֶׁבְּתוֹכָן שְׁאוּבִין, כֵּיצַד יַעֲשֶׂה, מַטְבִּילָן וּמַעֲלֶה אוֹתָן דֶּרֶךְ שׁוּלֵיהֶן, וּבְשַׂק וְקֻפָּה לֹא חָיְשִׁינַן, דְּלָא חֲשִׁיבֵי מַיִם שֶׁבְּתוֹכָן שְׁאוּבִין לִפְסֹל אֶת הַמִּקְוֶה:

עֵרוּב מִקְוָאוֹת. מִקְוֶה חָסֵר שֶׁבְּצַד מִקְוֶה שָׁלֵם, אִי נַמִּי בָּזֶה עֶשְׂרִים סְאָה וּבָזֶה עֶשְׂרִים סְאָה וּמִתְעָרְבִין הַמַּיִם דֶּרֶךְ חָרִיץ אוֹ נֶקֶב, שִׁעוּרוֹ לְצָרְפָן וּלְהַכְשִׁירָן כְּאִלּוּ יֵשׁ בְּכָל אֶחָד אַרְבָּעִים סְאָה: כִּשְׁפוֹפֶרֶת הַנּוֹד כְּעָבְיָהּ וְכַחֲלָלָהּ. שֶׁל קָנֶה שֶׁנּוֹתְנִין בְּפִי הַנּוֹד, וְהַיְנוּ כִשְׁתֵּי אֶצְבָּעוֹת חוֹזְרוֹת לִמְקוֹמָן. מִתְהַפְּכוֹת בְּרֶוַח בַּחֲלַל הַנֶּקֶב. וּשְׁתֵּי אֶצְבָּעוֹת שֶׁאָמְרוּ הֵן שְׁתֵּי אֶצְבָּעוֹת הַסְּמוּכוֹת לַגּוּדָל, הָרִאשׁוֹנוֹת שֶׁבְּפַס הַיָּד, דְּהַיְנוּ אֶצְבַּע וְאַמָּה: מִפְּנֵי שֶׁהִיא מִן הַתּוֹרָה. חִיּוּב הַטְּבִילָה בְּמִקְוֵה מַיִם הָוֵי מִן הַתּוֹרָה: וְכֵן כַּזַּיִת מִן הַמֵּת. סָפֵק יֵשׁ בָּהֶן כַּשִּׁעוּר סָפֵק אֵין בָּהֶן כַּשִּׁעוּר, סְפֵקָן טָמֵא. וְדַוְקָא בִרְשׁוּת הַיָּחִיד. אֲבָל בִּרְשׁוּת הָרַבִּים, סְפֵקָן טָהוֹר. וְאַף עַל גַּב דַּאֲפִלּוּ בִדְרַבָּנָן סְפֵק טֻמְאָה בִרְשׁוּת הַיָּחִיד סְפֵקוֹ טָמֵא, מִכָּל מָקוֹם הָכָא אִלּוּ הָווּ שִׁעוּרֵי דְרַבָּנָן, הָיִינוּ מְטַהֲרִין אוֹתוֹ: כֹּל שֶׁיַּעֲמֹד. בֵּין עָפָר בֵּין צְרוֹרוֹת בֵּין יַבְחוּשִׁים אֲדֻמִּים הַגְּדֵלִין בַּמַּיִם, אִם עוֹמְדִין עַל פִּי הַנֶּקֶב מְמַעֲטִין בִּשְׁפוֹפֶרֶת הַנּוֹד וְאֵין הַמִּקְוָאוֹת מִתְעָרְבִין: כֹּל שֶׁהוּא מִבְּרִיַּת הַמַּיִם. כְּגוֹן יַבְחוּשִׁים אֲדֻמִּים. וְאֵין הֲלָכָה כְּרַבָּן שִׁמְעוֹן בֶּן גַּמְלִיאֵל:

הָעֶלְיוֹן מִן הַתַּחְתּוֹן. כְּגוֹן שֶׁהָיָה הָעֶלְיוֹן מַיִם שְׁאוּבִים וְתַחְתּוֹן מַיִם כְּשֵׁרִים : וְהָרָחוֹק מִן הַקָּרוֹב. וְלֹא חָיְשִׁינַן שֶׁמָּא בָא אָדָם וְהִפְסִיק הַשָּׁקָתוֹ וְלֹא עָלְתָה לוֹ טְבִילָה: מֵבִיא סִלּוֹן שֶׁל חֶרֶס אוֹ שֶׁל אֲבָר. וְהוּא הַדִּין שֶׁל עֵץ אוֹ שֶׁל עֶצֶם אוֹ שֶׁל זְכוּכִית: אֲבָר. עוֹפֶרֶת: וּמַנִּיחַ יָדוֹ תַּחְתָּיו. שֶׁל הַסִּלּוֹן כְּדֵי שֶׁלֹּא יֵצְאוּ מַיִם לַחוּץ כְּשֶׁהוּא מִתְמַלֵּא עַד שָׁעָה שֶׁיַּשִּׁיק וְיִתְחַבְּרוּ עִם הַמִּקְוֶה הָאַחֵר: אֲפִלּוּ כַשַּׂעֲרָה. דְּהֵקֵלּוּ בִשְׁאוּבָה שֶׁהִיא דְרַבָּנָן . אֲבָל אִם הָיָה הָעֶלְיוֹן חָסֵר מִכַּשִּׁעוּר וּבָא לְהַכְשִׁירוֹ עַל יְדֵי הַשָּׁקָה לְהַשְׁלִימוֹ לְכַשִּׁעוּר, אֵין דַּי בְהַשָּׁקָה כַשַּׂעֲרָה, דַּאֲפִלּוּ רַבִּי יְהוּדָה דְמֵקֵל טְפֵי בְּפֶרֶק ב' דְּגִטִּין , בָּעֵי טוֹפֵחַ עַל מְנָת לְהַטְפִּיחַ: מְמַלֵּא בַכָּתֵף. דְּמֵאַחַר שֶׁיֵּשׁ בַּמִּקְוֶה אַרְבָּעִים סְאָה מַיִם כְּשֵׁרִים, אֲפִלּוּ כָל מַיִם שְׁאוּבִין שֶׁבָּעוֹלָם אֵין פּוֹסְלִין אוֹתוֹ :

לַשְּׁתִי. מִלְּמַעְלָה לְמַטָּה: לָעֵרֶב. מִיָּמִין לִשְׂמֹאל: חִלּוּף הַדְּבָרִים. מִלְּמַעְלָה לְמַטָּה, אֵין מִצְטָרֵף. מִיָּמִין לִשְׂמֹאל, מִצְטָרֵף. וְאֵין הֲלָכָה כְּרַבִּי יְהוּדָה: נִפְרְצוּ זֶה בְתוֹךְ זֶה. הַיְנוּ מִלְּמַעְלָה בְרֹאשׁ הַכֹּתֶל. דְּדַוְקָא כְשֶׁנִּסְדַּק הַכֹּתֶל אוֹ נִקַּב וּלְמַעְלָה בְרֹאשׁוֹ שָׁלֵם, הוּא דְבָעִינַן כִּשְׁפוֹפֶרֶת הַנּוֹד, אֲבָל כְּשֶׁנִּפְרַץ הַכֹּתֶל בְּרֹאשׁוֹ וְהַמַּיִם מִתְחַבְּרִים שָׁם, סַגִּי בְרוּם כִּקְלִפַּת הַשּׁוּם וּבְרֹחַב כִּשְׁפוֹפֶרֶת הַנּוֹד:

הָאָבִיק שֶׁבַּמֶּרְחָץ. כְּלִי שֶׁל מַתֶּכֶת מְשֻׁקָּע בָּאַמְבָּטִי וְיֶשׁ בּוֹ נֶקֶב, דִּכְשֶׁנִּמְאָסִין מֵי הָאַמְבָּטִי מֵחֲמַת הַטְּבִילוֹת פּוֹתְחִים הַנֶּקֶב שֶׁבָּאָבִיק וְיוֹצְאִים הַמַּיִם שֶׁבָּאַמְבָּטִי וּמַמְשִׁיכִין לוֹ מַיִם אֲחֵרִים נְקִיִּים וְסוֹתְמִים הַנֶּקֶב שֶׁלֹּא יֵצְאוּ הַמַּיִם, וְהַמַּיִם הַנִּמְשָׁכִים לָאַמְבָּטִי הֵן כְּשֵׁרִים לִטְבִילָה, דִּלְאַחַר שֶׁרָחֲצוּ בְחַמִּין שֶׁבַּמֶּרְחָץ נִכְנָסִים לָאַמְבָּטִי וְטוֹבְלִים שָׁם: בִּזְמַן שֶׁהוּא בָאֶמְצַע פּוֹסֵל. לְפִי שֶׁכָּל הַמַּיִם גְּרוּרִים עַל גַּבָּיו, וּמַיִם הַגְּרוּרִים עַל גַּבֵּי כֵלִים הֲרֵי אֵלּוּ פְסוּלִים: מִן הַצַּד אֵינוֹ פוֹסֵל. דְּאֵין הַמַּיִם גְּרוּרִים עַל גַּבָּיו, וְאֵין כָּאן שָׁאוּב אֶלָּא מַה שֶּׁבְּתוֹכוֹ, וּשְׁאָר הַמַּיִם הָווּ לְהוּ מִקְוֶה כָשֵׁר סָמוּךְ לַשָּׁאוּב, וּמַיִם שְׁאוּבִין בְּצַד מַיִם כְּשֵׁרִים לֹא פָסְלֵי וְאַף עַל פִּי שֶׁנּוֹגְעִים זֶה בָזֶה: אִם מְקַבֶּלֶת הָאַמְבָּטִי רְבִיעִית. מַיִם כְּשֵׁרִים קֹדֶם שֶׁיַּגִּיעוּ לָאָבִיק, כְּאִלּוּ מְקַבֶּלֶת אַרְבָּעִים סְאָה דָמֵי, מִשּׁוּם דִּרְבִיעִית שִׁעוּר מִקְוֶה הִיא מִדְּאוֹרַיְתָא לְהַטְבִּיל בָּהּ מְחָטִין וְצִנּוֹרִיּוֹת : אִם מְקַבֵּל הָאָבִיק. כְּלוֹמַר כֵּיוָן דְּאָבִיק הוּא הֶעָשׂוּי לְקַבָּלָה כָל שֶׁהוּא, פָּסוּל בְּכָל עִנְיָן שֶׁיִּהְיֶה, בֵּין בָּאֶמְצַע, בֵּין מִן הַצַּד, בֵּין מְקַבֶּלֶת הָאַמְבָּטִי רְבִיעִית קֹדֶם שֶׁבָּא לָאָבִיק, בֵּין אֵינָהּ מְקַבֶּלֶת:

הַמַּטְהֶרֶת שֶׁבַּמֶּרְחָץ. שִׁקֲתוֹת הַמַּיִם הָעֲשׂוּיוֹת בַּמֶּרְחָץ שֶׁנִּמְשָׁכִים לָהֶם מַיִם צוֹנְנִים לְהִשְׁתַּטֵּף בָּהֶן כְּדֵי לְטַהֵר עַצְמָן לְאַחַר שֶׁרָחֲצוּ בְחַמִּין, וּפְעָמִים שֶׁהֵן שְׁתֵּי שְׁקָתוֹת זוֹ בְצַד זוֹ אֶלָּא שֶׁהָאַחַת עֶלְיוֹנָה מֵחֲבֶרְתָּהּ, וְיֵשׁ נֶקֶב בַּכֹּתֶל שֶׁבֵּין שְׁתֵּיהֶן שֶׁהַמַּיִם יוֹצְאִים מִזּוֹ לָזוֹ: אִם יֵשׁ כְּנֶגֶד הַנֶּקֶב שְׁלֹשָׁה לֻגִּין. שֶׁהַנֶּקֶב רָחָב וְאָרֹךְ כְּשִׁעוּר שֶׁיִּכָּנְסוּ בוֹ שְׁלֹשָׁה לֻגִּין, נִפְסְלָה הָעֶלְיוֹנָה מֵחֲמַת הַמַּיִם שְׁאוּבִין שֶׁבַּתַּחְתּוֹנָה, דַּחֲשִׁיב כְּאִלּוּ הַמַּיִם שֶׁכְּנֶגֶד חֲלַל הַנֶּקֶב שְׁאוּבִין הֵן בְּאֶמְצַע הָעֶלְיוֹנָה, לֹא מִן הַצַּד, וּשְׁלֹשָׁה לֻגִּין מַיִם שְׁאוּבִין פּוֹסְלִים כְּשֶׁהֵן בָּאֶמְצַע אַף עַל פִּי שֶׁאֵין פּוֹסְלִין מִן הַצַּד. וּרְבוּתָא נָקַט כְּשֶׁיִּהְיֶה הַכָּשֵׁר לְצַד תַּחְתּוֹן: כַּמָּה יְהֵא בַנֶּקֶב. הַשְׁתָּא אֲתָא לְאַשְׁמוֹעִינַן כַּמָּה שִׁעוּרוֹ שֶׁל נֶקֶב שֶׁיֵּצְאוּ מִמֶּנּוּ שְׁלֹשָׁה לֻגִּין בְּבַת אַחַת: אַחַד מִשְּׁלֹשׁ שֶׁמַּחֲזִיק אַרְבָּעִים סְאָה, אַמָּה עַל אַמָּה בְּרוּם שָׁלֹשׁ אַמּוֹת, הֲרֵי שִׁעוּר הַנֶּקֶב הַמַּחֲזִיק שְׁלֹשָׁה לֻגִּין הוּא אֶחָד מִשְּׁלֹשׁ מֵאוֹת וְעֶשְׂרִים מֵאַמָּה עַל אַמָּה בְּרוּם שָׁלֹשׁ אַמּוֹת הַמַּחֲזִיק אַרְבָּעִים סְאָה. וְרַמְבַּ"ם כָּתַב, שֶׁאִם הָיְתָה הַבְּרֵכָה פָּחוֹת אוֹ יוֹתֵר מֵאַרְבָּעִים סְאָה, מְחַשְּׁבִין הַנֶּקֶב בְּפָחוֹת וּבְיוֹתֵר לְפִי חֶשְׁבּוֹן זֶה, כְּגוֹן אִם הָיָה בַבְּרֵכָה עֶשְׂרִים סְאָה, צָרִיךְ לִהְיוֹת הַנֶּקֶב אֶחָד מִמֵּאָה וְשִׁשִּׁים : אֲפִלּוּ הַתַּחְתּוֹנָה מְלֵאָה כְשֵׁרִים וְהָעֶלְיוֹנָה מְלֵאָה שְׁאוּבִין. שֶׁדֶּרֶךְ מַיִם לֵירֵד, אֲפִלּוּ הָכִי אֵין שְׁלֹשָׁה לֻגִּין שְׁאוּבִין שֶׁבַּנֶּקֶב פּוֹסְלִין אֶת הַתַּחְתּוֹנָה: שֶׁלֹּא אָמְרוּ אֶלָּא שְׁלֹשָׁה לֻגִּין שֶׁנָּפְלוּ. וְאֵין הַשְּׁאוּבִין פּוֹסְלִין אֶת הַמִּקְוֶה עַד שֶׁיִּפְּלוּ לְתוֹכוֹ. וַהֲלָכָה כְּרַבִּי יוֹסֵי:
