פרק א - משנה א
שַׁמַּאי אוֹמֵר כָּל הַנָּשִׁים דַּיָּן שְׁעָתָן. מִשּׁוּם דְּרַבִּי אֱלִיעֶזֶר אוֹמֵר אַרְבַּע נָשִׁים דַּיָּן שְׁעָתָן, תָּנֵי נַמִּי שַׁמַּאי אוֹמֵר כָּל הַנָּשִׁים, לַאֲפוֹקֵי מִדְּרַבִּי אֱלִיעֶזֶר: דַּיָּן שְׁעָתָן. נָשִׁים הָרוֹאוֹת דָּם דַּיָּן שְׁעָתָן לְטַמֵּא טָהֳרוֹת שֶׁנָּגְעוּ בָּהֶן מִשְּׁעַת רְאִיָּתָן וָאֵילָךְ, וְלֹא אָמְרִינַן מִקַּמֵּי הָכִי נַמִּי הָיָה דָּם וְכָתְלֵי בֵּית הָרֶחֶם הֶעֱמִידוּהוּ וְהָיְתָה טְמֵאָה מִקֹּדֶם. דְּכָל הַנָּשִׁים מְטַמְּאוֹת בַּבַּיִת הַחִיצוֹן וְאַף עַל פִּי שֶׁלֹּא יָצָא הַדָּם לַחוּץ, דִּכְתִיב (וַיִּקְרָא טו) דָּם יִהְיֶה זֹבָהּ בִּבְשָׂרָהּ, אַף עַל פִּי שֶׁהוּא בַּבָּשָׂר וְלֹא יָצָא לַחוּץ, טְמֵאָה. וְטַעֲמָא דְּשַׁמַּאי דְּלֹא חָיֵשׁ דִּלְמָא מִקַּמֵּי הָכִי הֲוָה דָּם, שֶׁאִם אַתָּה אוֹמֵר כֵּן, לִבּוֹ שֶׁל אָדָם נוֹקְפוֹ תָּמִיד בִּשְׁעַת בִּיאָה וּפוֹרֵשׁ מֵאִשְׁתּוֹ וְנִמְצֵאתָ מְבַטֵּל בְּנוֹת יִשְׂרָאֵל מִפְּרִיָּה וּרְבִיָּה: מִפְּקִידָה לִפְקִידָה. בָּדְקָה הַיּוֹם וּמָצְאָה טְהוֹרָה, וּבָדְקָה לְסוֹף שָׁבוּעַ וּמָצְאָה טְמֵאָה, חוֹשְׁשִׁין לְמַגָּעָהּ מִבְּדִיקָה רִאשׁוֹנָה וָאֵילָךְ, שֶׁמָּא עִם סִלּוּק יָדֶיהָ רָאֲתָה וְכָתְלֵי בֵּית הָרֶחֶם הֶעֱמִידוּהוּ. וּלְבִטּוּל פְּרִיָּה וּרְבִיָּה לֹא חָיְשִׁינַן, דְּדַוְקָא לְטָהֳרוֹת הוּא דִּמְטַמִּינַן לְהוּ מִפְּקִידָה לִפְקִידָה, וְלֹא לְבַעְלָהּ. וְשַׁמַּאי אוֹמֵר, לְטָהֳרוֹת נַמִּי לֹא מְטַמִּינַן, דְּאִם כֵּן לִבּוֹ נוֹקְפוֹ וּפוֹרֵשׁ נַמִּי מִתַּשְׁמִישׁ: לֹא כְדִבְרֵי זֶה. שַׁמַּאי דְּמֵקֵל טְפֵי וְאֵינוֹ עוֹשֶׂה סְיָג לִדְבָרָיו: וְלֹא כְדִבְרֵי זֶה. הִלֵּל דְּמַחְמִיר יוֹתֵר מִדַּאי. דְּוַדַּאי כֻּלֵּי הַאי יָמִים רַבִּים לֹא מוֹקְמִי כָּתְלֵי בֵּית הָרֶחֶם דָּם: מֵעֵת לְעֵת מְמַעֵט עַל יַד מִפְּקִידָה לִפְקִידָה. שְׁנֵי זְמַנִּים הֻזְכְּרוּ בְּאִשָּׁה לְטַמֵּא טָהֳרוֹת לְמַפְרֵעַ, וְהַלֵּךְ אַחַר הַקַּל שֶׁבִּשְׁנֵיהֶן. אִם מִפְּקִידָה לִפְקִידָה יוֹתֵר עַל מֵעֵת לְעֵת, הַלֵּךְ אַחַר מֵעֵת לְעֵת וְלֹא תְּטַמֵּא אֶלָּא טָהֳרוֹת שֶׁנָּגְעָה מֵאֶתְמוֹל בַּשָּׁעָה הַזֹּאת. וְאִם מֵעֵת לְעֵת יוֹתֵר עַל מִפְּקִידָה לִפְקִידָה, כְּגוֹן שֶׁבָּדְקָה עַצְמָהּ שַׁחֲרִית וּמָצְאָה טְהוֹרָה וְעַרְבִית מָצְאָה טְמֵאָה, לֹא מְטַמְּאִינַן אֶלָּא הָנָךְ טָהֳרוֹת דְּמִבְּדִיקַת שַׁחֲרִית וָאֵילָךְ. וְטַעֲמַיְהוּ דְּרַבָּנָן, לְפִי שֶׁחֲכָמִים תִּקְּנוּ לִבְנוֹת יִשְׂרָאֵל שֶׁיִּהְיוּ בּוֹדְקוֹת שַׁחֲרִית וְעַרְבִית, שַׁחֲרִית לְהַכְשִׁיר טָהֳרוֹת שֶׁל לַיְלָה, וְעַרְבִית לְהַכְשִׁיר טָהֳרוֹת שֶׁל יוֹם. וְאִם בָּדְקָה שַׁחֲרִית וּמָצְאָה טְהוֹרָה, עַרְבִית וּמָצְאָה טְמֵאָה, הִפְסִידָה טָהֳרוֹת שֶׁל יוֹם, וְזוֹ שֶׁיָּשְׁבָה יָמִים רַבִּים וְלֹא בָּדְקָה כְּתַקָּנַת חֲכָמִים, הִפְסִידָה עוֹנָה אַחַת יְתֵרָה, מִשּׁוּם קְנָס. וּכְדֵי לְהַשְׁווֹת מִדּוֹתֵיהֶם שֶׁל חֲכָמִים שֶׁלֹּא תַּחֲלֹק בֵּין הֵיכָא דְּבָדְקָה שַׁחֲרִית לְהֵיכָא דְּבָדְקָה בַּצָהֳרָיִם , לֹא רָצוּ לְחַלֵּק וּלְמַעֵט וּלְהוֹסִיף בַּזְּמַן, וְנָתְנוּ זְמַן קָבוּעַ מֵעֵת לְעֵת, שֶׁלֹּא לִתֵּן דִּבְרֵיהֶם לְשִׁעוּרִין. וַהֲלָכָה כַּחֲכָמִים: עַל יַד. אַחַר. כְּמוֹ וְעַל יָדוֹ הֶחֱזִיק, שֶׁהוּא כְּמוֹ וְאַחֲרָיו, בְּסֵפֶר עֶזְרָא : וֶסֶת. קָבוּעַ, שֶׁנִּקְבַּע לָהּ זְמַן הָאֹרַח שָׁלֹשׁ פְּעָמִים וּבָדְקָה בִּשְׁעַת וֶסְתָּהּ וּמָצְאָה טְמֵאָה: דַּיָּהּ שְׁעָתָהּ. וְלֹא חָיְשִׁינַן דִּלְמָא מִקַּמֵּי הָכִי הֲוָה, דְּוַדַּאי אֹרַח בִּזְמַנּוֹ בָּא : וְהַמְשַׁמֶּשֶׁת בְּעִדִּים. שֶׁבָּדְקָה עַצְמָהּ בְּעִדִּים, דְּהַיְנוּ בְּסָדִין, קֹדֶם תַּשְׁמִישׁ וּלְאַחַר תַּשְׁמִישׁ: הֲרֵי זוֹ כִפְקִידָה. בַּגְּמָרָא מוֹקֵי לְמַתְנִיתִין דְּתַרְתֵּי קָתָנֵי וְהָכִי קָאָמַר, וּמְשַׁמֶּשֶׁת בְּעִדִּים כְּלוֹמַר מִצְוָה עַל אִשָּׁה שֶׁתְּשַׁמֵּשׁ בִּשְׁנֵי עִדִּים, אֶחָד לִפְנֵי תַּשְׁמִישׁ וְאֶחָד לְאַחַר תַּשְׁמִישׁ. וַהֲרֵי זוֹ כִּפְקִידָה, הָעֵד שֶׁל אַחַר תַּשְׁמִישׁ הֲרֵי הוּא כִּפְקִידָה. וּמְמַעֶטֶת עַל יַד מֵעֵת לְעֵת וְעַל יַד מִפְּקִידָה לִפְקִידָה, דְּמַהוּ דְתֵימָא שֶׁמָּא תִּרְאֶה טִפַּת דָּם כְּחַרְדָּל וּתְחַפֶּנָּה שִׁכְבַת זֶרַע וְלֹא לֶהֱוֵי כִּפְקִידָה, קָמַשְׁמַע לָן. וְאִם שִׁמְּשָׁה בַּצָהֳרַיִם בְּעִדִּים, וּבָדְקָה אַחֲרֵי כֵן בָּעֶרֶב וּמָצְאָה טְמֵאָה, לֹא מְטַמְּאִינַן טָהֳרוֹת דְּמִבְּדִיקַת שַׁחֲרִית עַד תַּשְׁמִישׁ, שֶׁהֲרֵי בִּשְׁעַת תַּשְׁמִישׁ בָּדְקָה וּמָצְאָה טְהוֹרָה. וְאַף עַל גַּב דִּבְדִיקָה שֶׁקֹּדֶם תַּשְׁמִישׁ לָא חֲשִׁיבָא בְּדִיקָה מְעַלְּיָתָא, שֶׁמִּתּוֹךְ שֶׁהִיא בְּהוּלָה לְשַׁמֵּשׁ אֵינָהּ מַכְנֶסֶת הָעֵד לַחוֹרִים וְלַסְּדָקִין, בְּדִיקָה בָּעֵד שֶׁלְאַחַר תַּשְׁמִישׁ בְּדִיקָה מְעַלְּיָתָא הִיא וַחֲשִׁיבָא כִּפְקִידָה כְּדַאֲמַרַן:

פרק א - משנה ב
הָיְתָה יוֹשֶׁבֶת בַּמִּטָּה וַעֲסוּקָה בְטָהֳרוֹת. הָא דְּלֹא תָּנֵי הָיְתָה עֲסוּקָה בְּטָהֳרוֹת וּפָרְשָׁה וְרָאֲתָה, וְקָתָנֵי הָיְתָה יוֹשֶׁבֶת בַּמִּטָּה, לְאַשְׁמוֹעִינַן דְּמִשּׁוּם דְּיֵשׁ לָהּ וֶסֶת וְדַיָּהּ שַׁעְתָּהּ, הַמִּטָּה נַמִּי טְהוֹרָה, כִּדְּקָתָנֵי וְכֻלָּן טְהוֹרוֹת, הָא אִם אֵין לָהּ וֶסֶת, דִּמְטַמִּינַן מֵעֵת לְעֵת, הַמִּטָּה נַמִּי טְמֵאָה , כְּמִשְׁכַּב נִדָּה עַצְמָהּ שֶׁמְּטַמְּאָה טֻמְאָה חֲמוּרָה לְטַמֵּא אָדָם לְטַמֵּא בְּגָדִים שֶׁעָלָיו: אַף עַל פִּי שֶׁאָמְרוּ. אִשָּׁה שֶׁאֵין לָהּ וֶסֶת מְטַמְּאָה מֵעֵת לְעֵת: אֵינָהּ מוֹנָה. יְמֵי נִדָּה אֶלָּא מִשָּׁעָה שֶׁרָאֲתָה:

פרק א - משנה ג
אַרְבַּע נָשִׁים דַּיָּן שְׁעָתָן. דְּאֵינָן רְגִילוֹת לִרְאוֹת דָּם: אֲבָל הֲלָכָה כְּרַבִּי אֱלִיעֶזֶר. וְהָכִי מַסִּיק בַּגְּמָרָא דַּהֲלָכָה כְּרַבִּי אֱלִיעֶזֶר:

פרק א - משנה ד
מִשֶּׁיִּוָּדַע עֻבָּרָהּ. לְסוֹף שְׁלֹשָׁה חֳדָשִׁים מִיּוֹם שֶׁנִּתְעַבְּרָה: עַד שֶׁתִּגְמֹל. עַד שֶׁיַּשְׁלִים לִבְנָהּ עֶשְׂרִים וְאַרְבָּעָה חֹדֶשׁ. וַאֲפִלּוּ מֵת בְּתוֹךְ עֶשְׂרִים וְאַרְבָּעָה חֹדֶשׁ, דַּיָּהּ שַׁעְתָּהּ , שֶׁהַיּוֹלֶדֶת אֵבָרֶיהָ מִתְפָּרְקִים וְאֵין דָּמָהּ חוֹזֵר עַד אַחַר עֶשְׂרִים וְאַרְבָּעָה חֹדֶשׁ. וּמִשָּׁם וָאֵילָךְ הֲרֵי הִיא כְּכָל הַנָּשִׁים וּמְטַמְּאָה מֵעֵת לְעֵת וּמִפְּקִידָה לִפְקִידָה, וְאַף עַל פִּי שֶׁמֵּנִיקָה וְהוֹלֶכֶת אַרְבַּע וְחָמֵשׁ שָׁנִים: גְּמָלַתּוּ אוֹ מֵת רַבִּי מֵאִיר אוֹמֵר מְטַמְּאָה מֵעֵת לְעֵת. דְּסָבַר רַבִּי מֵאִיר דָּם נֶעְכָּר וְנַעֲשָׂה חָלָב . וּכְשֶׁאֵינָהּ מֵנִיקָה, אֲפִלּוּ קֹדֶם עֶשְׂרִים וְאַרְבָּעָה חֹדֶשׁ חוֹזֵר הַדָּם לִמְקוֹמוֹ וְחוֹזֶרֶת לִהְיוֹת רוֹאָה. וְאֵין הֲלָכָה כְּרַבִּי מֵאִיר:

פרק א - משנה ה
כֹּל שֶׁעָבְרוּ עָלֶיהָ שָׁלֹשׁ עוֹנוֹת. עוֹנָה בֵּינוֹנִית הִיא שְׁלֹשִׁים יוֹם. וּמִכֵּיוָן שֶׁעָבְרוּ עָלֶיהָ שָׁלֹשׁ עוֹנוֹת בֵּינוֹנִיּוֹת אַחַר שֶׁהִגִּיעָה סָמוּךְ לְזִקְנָתָהּ, דַּיָּהּ שַׁעְתָּהּ. וְסָמוּךְ לְזִקְנָתָהּ נִקְרָא, כָּל שֶׁקּוֹרִים לָהּ אִימָא אִימָא מֵחֲמַת זִקְנָתָהּ וְאֵינָהּ מַקְפֶּדֶת: כָּל אִשָּׁה. וַאֲפִלּוּ אֵינָהּ זְקֵנָה, וְעָבְרוּ עָלֶיהָ שָׁלֹשׁ עוֹנוֹת וְלֹא רָאֲתָה, דַּיָּהּ שַׁעְתָּהּ: מְעֻבֶּרֶת וּמֵנִיקָה שֶׁעָבְרוּ עֲלֵיהֶן שָׁלֹשׁ עוֹנוֹת. אֲבָל לֹא עָבְרוּ עֲלֵיהֶן, לֹא. וּדְלֹא כְּתַנָּא קַמָּא דְּאָמַר מְעֻבֶּרֶת מִשֶּׁיִּוָּדַע הָעֻבָּר , וּמֵנִיקָה מִשֶּׁנּוֹלַד. וְאֵין הֲלָכָה לֹא כְּרַבִּי אֱלִיעֶזֶר וְלֹא כְּרַבִּי יוֹסֵי:

פרק א - משנה ו
וּבַמֶּה אָמְרוּ. אַרְבַּע נָשִׁים דַּיָּן שְׁעָתָן: בִּרְאִיָה רִאשׁוֹנָה. שֶׁרָאֲתָה הַבְּתוּלָה, וְהַמְעֻבֶּרֶת וְהַמֵּנִיקָה, וְהַזְּקֵנָה לְאַחַר שֶׁעָבְרוּ עָלֶיהָ שָׁלֹשׁ עוֹנוֹת מֵחֲמַת זִקְנָתָהּ: אֲבָל בַּשְּׁנִיָּה מְטַמְּאָה מֵעֵת לְעֵת. שֶׁהֲרֵי חָזְרָה לְקַדְמוּתָהּ לִהְיוֹת דָּמֶיהָ מְצוּיִין בָּהּ: מֵאֹנֶס. כְּגוֹן דְּקָפְצָה וְרָאֲתָה, אוֹ מֵחֲמַת פַּחַד, אוֹ שֶׁאָכְלָה דָּבָר הַמַּגִּיר דָּם:

פרק א - משנה ז
אַף עַל פִּי שֶׁאָמְרוּ. אִשָּׁה שֶׁיֵּשׁ לָהּ וֶסֶת דַּיָּהּ שַׁעְתָּהּ: צְרִיכָה לִהְיוֹת בּוֹדֶקֶת. בְּכָל יוֹם שַׁחֲרִית וְעַרְבִית כִּשְׁאָר נָשִׁים: חוּץ מִן הַנִּדָּה. מִשֶּׁרָאֲתָה יוֹם אֶחָד, טְמֵאָה שִׁבְעָה, וְאֵינָהּ צְרִיכָה בְּדִיקָה כָּל שִׁבְעָה, שֶׁהֲרֵי בְּלֹא רְאִיָּה הִיא טְמֵאָה כָּל שִׁבְעָה: וְהַיּוֹשֶׁבֶת עַל דַּם טֹהַר. דְּלָמָּה לָהּ בְּדִיקָה, אֲפִלּוּ תִּרְאֶה, דַּם טֹהַר הוּא: וּמְשַׁמֶּשֶׁת בְּעִדִּים. אַף עַל פִּי שֶׁיֵּשׁ לָהּ וֶסֶת, צְרִיכָה לִבְדֹּק לִפְנֵי תַּשְׁמִישׁ וּלְאַחַר תַּשְׁמִישׁ. וְדַוְקָא כְּשֶׁהִיא עֲסוּקָה בְּטָהֳרוֹת, דְּמִגּוֹ דְּבָעֲיָא בְּדִיקָה לְטָהֳרוֹת, בַּעְיָא נַמִּי בְּדִיקָה לְבַעְלָהּ. אֲבָל אִשָּׁה שֶׁאֵינָהּ עֲסוּקָה בְּטָהֳרוֹת, לְבַעְלָהּ לֹא בַּעְיָא בְּדִיקָה : וּבְתוּלָה שֶׁדָּמֶיהָ טְהוֹרִים. אַרְבַּע לֵילוֹת לְבֵית שַׁמַּאי, וּלְבֵית הִלֵּל עַד שֶׁתִּחְיֶה הַמַּכָּה שֶׁל בְּעִילָה, וְהִיא אֵינָהּ צְרִיכָה לְשַׁמֵּשׁ בְּעִדִּים, שֶׁהֲרֵי דָּמָהּ טָהוֹר, וַאֲפִלּוּ נִשְׁתַּנּוּ מַרְאִית דָּמֶיהָ, הוֹאִיל וְשִׁמְּשָׁה, דְּמִשּׁוּם תַּשְׁמִישׁ נִשְׁתַּנָּה. אֲבָל שַׁחֲרִית וְעַרְבִית, צְרִיכָה בְּדִיקָה הֵיכָא דְּלֹא שִׁמְּשָׁה אַחַר בְּעִילָה רִאשׁוֹנָה , שֶׁמָּא נִשְׁתַּנּוּ מַרְאֵה דָמִים שֶׁלָּהּ מִמַּרְאֵה דַם בְּעִילָה רִאשׁוֹנָה. וְאִם נִשְׁתַּנּוּ מַרְאֵה דָמִים שֶׁלָּהּ, טְמֵאָה, שֶׁרַגְלַיִם לַדָּבָר שֶׁאֵין זֶה דַּם בְּתוּלִים: שַׁחֲרִית וּבֵין הַשְּׁמָשׁוֹת. שַׁחֲרִית לְהַכְשִׁיר טָהֳרוֹת שֶׁל לַיְלָה, שֶׁאִם תִּמְצָא טְהוֹרָה תֵּדַע שֶׁלֹּא רָאֲתָה בַּלַּיְלָה, וְלִכְשֶׁתִּמְצָא עַרְבִית טְמֵאָה יָצְאוּ אֵלּוּ מִסָּפֵק. וְכֵן בֵּין הַשְּׁמָשׁוֹת לְהַכְשִׁיר טָהֳרוֹת שֶׁנִּתְעַסְּקָה בָּהֶן בַּיּוֹם: וּבְשָׁעָה שֶׁהִיא עוֹבֶרֶת לְשַׁמֵּשׁ אֶת בֵּיתָהּ. שֶׁמְּכִינָה עַצְמָהּ לְשַׁמֵּשׁ עִם בַּעְלָהּ: רַבִּי יְהוּדָה אוֹמֵר אַף בִּשְׁעַת עֲבָרָתָן מִלֶּאֱכֹל בַּתְּרוּמָה. אַף לְאַחַר שֶׁאָכְלוּ בַּתְּרוּמָה, צְרִיכוֹת לִבְדֹּק עַצְמָן, לְתַקֵּן שְׁיָרֵי תְּרוּמוֹת שֶׁבִּפְנִים. שֶׁאִם תִּמְצָא עַכְשָׁיו טְהוֹרָה, כְּשֶׁתִּבְדֹּק עַרְבִית כְּמִשְׁפָּטָהּ וְתִמְצָא טְמֵאָה, לֹא יִהְיֶה שְׁיָרֵי הַתְּרוּמָה בִּכְלַל טֻמְאָה לְמַפְרֵעַ, שֶׁהֲרֵי בָּדְקָה לְאַחַר הַעֲבָרָתָהּ. וְאֵין הֲלָכָה כְּרַבִּי יְהוּדָה:
