פרק יח

פרק יח - משנה א
הַשִּׁדָּה. נִמְדֶּדֶת מִבִּפְנִים. דִּתְנַן לְעֵיל (פֶּרֶק טו) כְּלֵי עֵץ שֶׁהַמַּחֲזִיק אַרְבָּעִים סְאָה בְלַח שֶׁהֵן כּוֹרַיִם בְּיָבֵשׁ, טָהוֹר, וְהַיְנוּ אַמָּה עַל אַמָּה בְרוּם שָׁלֹשׁ אַמּוֹת. וְהָכָא קָמִפְלְגֵי כֵיצַד מוֹדְדִים, לְבֵית שַׁמַּאי מוֹדְדִין חָלָל שֶׁבִּפְנִים, וְלֹא עֳבִי הַדְּפָנוֹת וְהַשּׁוּלַיִם. וּלְבֵית הִלֵּל הַכֹּל מוֹדְדִין, חוּץ מֵעֳבִי הָרַגְלַיִם וּלְבַזְבְּזִים שֶׁהוּא הַזֵּר שֶׁעַל הַשָּׂפָה. וּלְרַבִּי יוֹסֵי מוֹדְדִים אַף עֳבִי רַגְלֵי הַשִּׁדָּה וְהַלְּבַזְבְּזִים. וּלְרַבִּי שִׁמְעוֹן אִם אֵין רַגְלֶיהָ גְּבוֹהִים טֶפַח מוֹדְדִים אַף הָאֲוִיר שֶׁבֵּין רֶגֶל לְרֶגֶל. וַהֲלָכָה כְּבֵית הִלֵּל אַלִּיבָּא דְתַנָּא קַמָּא:

פרק יח - משנה ב
מוּכְנִי. אוֹפַן כְּעֵין אוֹפַנֵּי הָעֲגָלָה. וּלְפִי שֶׁהַשִּׁדָּה נִשֵּׂאת עָלֶיהָ קְרוּיָה מוּכְנִי. לְשׁוֹן אֶת הַכִּיּוֹר וְאֶת כַּנּוֹ (שְׁמוֹת ל): בִּזְמַן שֶׁהִיא נִשְׁמֶטֶת. שֶׁאָדָם יָכוֹל לְסַלְּקָהּ מִן הָעֲגָלָה: אֵינָהּ חִבּוּר לָהּ. וְהִיא חֲשׁוּבָה כְּלִי לְעַצְמָהּ. וְאִי שִׁדָּה בַּת קִבּוּלֵי טֻמְאָה שֶׁאֵינָהּ מַחְזֶקֶת אַרְבָּעִים סְאָה וְנִטְמֵאת הַשִּׁדָּה, לֹא נִטְמֵאת הַמּוּכְנִי, וְאִי לָאו בַּת קִבּוּלֵי טֻמְאָה הִיא וְנָגְעָה טֻמְאָה בַמּוּכְנִי, הַמּוּכְנִי טְמֵאָה, דְּלָא חֲשִׁיבָא מִן הַשִּׁדָּה, וְהִיא מִטַּלְטֶלֶת מְלֵאָה וְרֵיקָנִית וְיֵשׁ לָהּ בֵּית קִבּוּל: וְאֵינָהּ נִמְדֶּדֶת עִמָּהּ. בְּחֶשְׁבּוֹן אַמָּה עַל אַמָּה בְּרוּם שָׁלֹשׁ אַמּוֹת. דְּלָא חֲשִׁיבָא כָעֳבִי הָרַגְלַיִם, וַאֲפִלּוּ לְרַבִּי יוֹסֵי אֵינָהּ נִמְדֶּדֶת עִמָּהּ: וְאֵינָהּ מַצֶּלֶת עִמָּהּ בְּאֹהֶל הַמֵּת. שֶׁאִם הַשִּׁדָּה בְּבֵית הַקְּבָרוֹת מַצֶּלֶת עַל כֵּלִים שֶׁבְּתוֹכָהּ, כֵּיוָן דְּמַחְזֶקֶת אַרְבָּעִים סְאָה. וְאִם בּוֹלְטִים מִמֶּנָּה כֵּלִים כְּנֶגֶד הַמּוּכְנִי מִלְּמַעְלָה, אֵין הַמּוּכְנִי מַצֶּלֶת עָלֶיהָ, דְּמוּכְנִי מְקַבֶּלֶת טֻמְאָה וְאֵינָהּ חוֹצֶצֶת בִּפְנֵי הַטֻּמְאָה: בִּזְמַן שֶׁבְּתוֹכָהּ מָעוֹת. דִּכְלִי בִּפְנֵי עַצְמָהּ הִיא, וּבָסִיס לְדָבָר הָאָסוּר. אֲבָל אִם אֵינָהּ נִשְׁמֶטֶת שָׁרֵי, מֵאַחַר דְּלֹא הָווּ בְתוֹךְ הַשִּׁדָּה שֶׁהִיא עִקַּר הַכְּלִי: קִמְרוֹן. כְּעֵין קֻבָּא שֶׁעוֹשִׂין עַל גַּבֵּי הַשִּׁדָּה: נִמְדָּד עִמָּהּ. לְהַשְׁלִים אַרְבָּעִים סְאָה: כֵּיצַד מוֹדְדִים אוֹתוֹ. כְּשֶׁהוּא קָבוּעַ: רֹאשׁ תּוֹר. בַּאֲלַכְסוֹן. דְּהַאי קִמְרוֹן אֵינוֹ כְנֶגֶד כָּל הַשִּׁדָּה אֶלָּא כִפָּה כְנֶגֶד מִקְצָת חֲלָלָהּ, וּכְשֶׁבָּא לִמְדֹּד אֲוִירָהּ רוֹאִין כְּאִלּוּ חוּט נָתוּן בַּאֲלַכְסוֹן מִכֹּתֶל הַשִּׁדָּה עַד רֹאשׁ הַכִּפָּה, וְחָלָל שֶׁתַּחְתָּיו נִמְדָּד: אִם אֵינָהּ יְכוֹלָה לַעֲמֹד בִּפְנֵי עַצְמָהּ. הַשִּׁדָּה בְלֹא הַקִּמְרוֹן הַקָּבוּעַ בָּהּ: טְהוֹרָה. אַף עַל פִּי שֶׁאֵינָהּ מַחְזֶקֶת אַרְבָּעִים סְאָה. וְאֵין הֲלָכָה כְּרַבִּי יְהוּדָה:

פרק יח - משנה ג
אַף עַל פִּי שֶׁמְּקַבְּלִים. כְּלוֹמַר אַף עַל פִּי שֶׁלֹּא נִקְּבוּ בִנְטִילַת הָרַגְלַיִם וְיֵשׁ לָהֶן בֵּית קִבּוּל כְּבַתְּחִלָּה, טְהוֹרִין. שֶׁכֵּיוָן שֶׁנִּטַּל אַחַת מִן רַגְלֵיהֶן שׁוּב אֵינָן חֲשׁוּבִין כְּלִי, לְפִי שֶׁאֵינָן מְקַבְּלִין כְּדַרְכָּן אֶלָּא נוֹטִין לְצַד אֶחָד וְכִשְׁבוּרִים דָּמוּ: רַבִּי יוֹסֵי מְטַמֵּא. כֵּיוָן שֶׁלֹּא נִקְּבוּ בִנְטִילַת אַחַת מֵרַגְלֵיהֶן. וְאֵין הֲלָכָה כְּרַבִּי יוֹסֵי. וְרַמְבַּ"ם גָּרֵיס, וְשֶׁאֵינָן מְקַבְּלִים כְּדַרְכָּן רַבִּי יוֹסֵי מְטַמֵּא. וּמִלְּתָא בְאַנְפֵּי נַפְשַׁהּ הִיא, וְהָכִי קָאָמַר, שִׁדָּה תֵבָה וּמִגְדָּל שֶׁאֵינָן מְקַבְּלִין אַרְבָּעִים סְאָה כְּשֶׁהֵן יוֹשְׁבִין כְּדַרְכָּן וְאִם מַטִּין אוֹתָן מְקַבְּלִין אַרְבָּעִים סְאָה, רַבִּי יוֹסֵי מְטַמֵּא כֵּיוָן שֶׁאֵין מְקַבְּלִים כְּדַרְכָּן, וַחֲכָמִים אוֹמְרִים הוֹאִיל וּמְקַבְּלִים מִכָּל מָקוֹם, חֲשִׁיבֵי מַחְזִיקִים אַרְבָּעִים סְאָה וּטְהוֹרִין: וְנַקְלִיטֵי הַמִּטָּה. שְׁנֵי עַמּוּדִים אֶחָד מְרַאֲשׁוֹתֵי הַמִּטָּה וְאֶחָד בְּמַרְגְּלוֹתֶיהָ, וּמְשִׂימִין מַקֵּל מִזֶּה לָזֶה וּמַשְׁלִיכִין עָלָיו סָדִין וְנַעֲשֶׂה אֹהֶל: וַחֲמוֹר. תַּחַת מַלְבֵּן הַמִּטָּה מְשִׂימִין עֵץ חָלוּל שֶׁהַמַּלְבֵּן מֻנָּח עָלָיו וְנִקְרָא חֲמוֹר: וְחִפּוּי. כְּגוֹן צִפּוּי שֶׁמְּצַפִּים הַמִּטָּה וְנַקְלִיטֶיהָ: טְהוֹרִים. דַּהֲוֵי מַתֶּכֶת הַמְשַׁמֵּשׁ לְעֵץ: מַלְבֵּן. לוּחוֹת חֲלוּלִים עֲשׂוּיִים לְהַנִּיחַ תַּחַת כַּרְעֵי הַמִּטָּה שֶׁלֹּא יֵרָקְבוּ מִלַּחְלוּחִית הַקַּרְקַע. וּמִטַּלְטְלִים עִם הַמִּטָּה , הִלְכָּךְ טְמֵאִים, דְּכַמִּטָּה דָמוּ: וּמַלְבְּנֵי בְנֵי לֵוִי טְהוֹרִין. לְפִי שֶׁרְגִילִין לַעֲלוֹת לִירוּשָׁלַיִם וְהוֹלְכִים מֵעִיר לְעִיר, וְאֵין מוֹלִיכִין הַמִּטּוֹת עִמָּהֶן וּמוֹלִיכִין הַמַּלְבְּנִים, הִלְכָּךְ אֲפִלּוּ קְבוּעִים בְּקַרְקְעֵי הַמִּטָּה הוֹאִיל וְסוֹפָן לִנָּטֵל לָאו כַּמִּטָּה חֲשִׁיבֵי:

פרק יח - משנה ד
מַלְבֵּן שֶׁנְּתָנוֹ עַל לְשׁוֹנוֹת. עֵץ הֶעָשׂוּי כְּעֵין לָשׁוֹן שֶׁהַמַּלְבֵּן מֻנָּח עָלָיו: רַבִּי מֵאִיר וְרַבִּי יְהוּדָה מְטַמְּאִין. שֶׁאַף הוּא מִטַּלְטֵל עִם הַמִּטָּה כְּמוֹ מַלְבֵּן: רַבִּי יוֹסֵי וְרַבִּי שִׁמְעוֹן מְטַהֲרִין. דְּלֹא מִטַּלְטֵל בַּהֲדֵי מִטָּה כֻלֵּי הַאי. וַהֲלָכָה כְּרַבִּי יוֹסֵי וְרַבִּי שִׁמְעוֹן: שֶׁמַּלְבְּנֵי בְנֵי לֵוִי טְהוֹרִין. וְאַף עַל גַּב דִּקְבִיעֵי בְכַרְעֵי הַמִּטָּה, הוֹאִיל וְסוֹפָן לִנָּטֵל טְהוֹרִים, הָנֵי לְשׁוֹנוֹת נַמִּי פְעָמִים נִטָּלִים מִמַּלְבְּנֵי הַמִּטָּה וְלָאו כַּמִּטָּה חֲשִׁיבֵי:

פרק יח - משנה ה
מִטָּה שֶׁהָיְתָה טְמֵאָה מִדְרָס. שֶׁשָּׁכַב עָלֶיהָ זָב , אוֹ נִתְלָה אוֹ נִשְׁעַן: קְצָרָה. אוֹתָן עֵצִים שֶׁבְּרֹחַב הַמִּטָּה לְצַד מְרַאֲשֹׁתֶיהָ אוֹ מַרְגְּלוֹתֶיהָ: אֲרֻכָּה. אוֹתָן עֵצִים שֶׁבְּאֹרֶךְ הַמִּטָּה מִיָּמִין אוֹ מִשְּׂמֹאל, וְהֵם נִקְרָאִים אֲרֻכּוֹת הַמִּטָּה: טְהוֹרָה. דְּתוּ לֹא חַזְיָא וְהַוְיָא כְשִׁבְרֵי כֵלִים שֶׁהֵן טְהוֹרִים: רַבִּי נְחֶמְיָה מְטַמֵּא. וְאֵין הֲלָכָה כְּרַבִּי נְחֶמְיָה: גָּדַד. חָתַךְ וְקָטַע. כְּמוֹ גֹדוּ אִילָנָא: שְׁתֵּי לְשׁוֹנוֹת לוֹכְסָן. שְׁנֵי עֵצִים הָעֲשׂוּיִין כְּלָשׁוֹן, שֶׁמַּלְבְּנֵי הַמִּטָּה מֻנָּחִים עֲלֵיהֶן. חֲתָכָן בָּאֲלַכְסוֹן, אֶחָד שֶׁבְּמִקְצוֹעַ מִזְרָחִית דְּרוֹמִית וְאֶחָד שֶׁבְּמִקְצוֹעַ מַעֲרָבִית צְפוֹנִית, טְהוֹרָה, דְּתוּ לֹא חַזְיָא. וְכֵן אִם גָּדַד שְׁתֵּי כְרָעַיִם טֶפַח עַל טֶפַח לוֹכְסָן, מִכֶּרַע שֶׁבְּמִקְצוֹעַ מִזְרָחִית דְּרוֹמִית גָּדַד טֶפַח, וּמִכֶּרַע שֶׁבְּמִקְצוֹעַ מַעֲרָבִית צְפוֹנִית גָּדַד טֶפַח, כִּשְׁבוּרָה דָמְיָא: אוֹ שֶׁמִּעֲטָהּ מִטֶּפַח. שֶׁגָּדַד רַגְלֶיהָ וְאֵין טֶפַח מִן הַמִּטָּה לָאָרֶץ, לָא חֲשִׁיבָא:

פרק יח - משנה ו
וְתִקְּנָהּ טְמֵאָה מִדְרָס. עֲדַיִן עוֹמֶדֶת בְּטֻמְאָתָהּ, כֵּיוָן דְּקַיְמָא שְׁנִיָּה: טְהוֹרָה מִן הַמִּדְרָס. אַף עַל גַּב דְּנִתַּקְּנָה רִאשׁוֹנָה קֹדֶם שֶׁנִּשְׁבְּרָה שְׁנִיָּה, מִכָּל מָקוֹם טְהוֹרָה, דְּפָנִים חֲדָשׁוֹת בָּאוּ לְכָאן, שֶׁלְּאַחַר שֶׁיָּרְדָה לָהּ טֻמְאַת מִדְרָס נִתְחַדְּשׁוּ לָהּ פָּנִים הַלָּלוּ וְאֵין זוֹ הָרִאשׁוֹנָה, מֵאַחַר שֶׁכְּבָר נִתְקַלְקֵל בָּהּ כְּדֵי בִטּוּלָהּ: אֲבָל טְמֵאָה מַגַּע מִדְרָס. מִשּׁוּם דַּאֲרֻכָּה רִאשׁוֹנָה כְשֶׁנִּשְׁבְּרָה וְתִקְּנָהּ שֶׁהֵבִיא אֲרֻכָּה אַחֶרֶת תַּחַת הַשְּׁבוּרָה וְחִבְּרָהּ לַמִּטָּה, נִטְמֵאת מַגַּע מִדְרָס, שֶׁהֲרֵי נָגְעָה בַמִּטָּה הַטְּמֵאָה מִדְרָס , וְכִי נִשְׁבְּרָה שְׁנִיָּה דְפָרַח מִנַּהּ טֻמְאַת מִדְרָס, אִשְׁתַּיַּר בָּהּ טֻמְאַת מַגָּע, דְּכָל הַמִּטָּה מְחֻבֶּרֶת לָרִאשׁוֹנָה שֶׁנִּתַּקְּנָה: עַד שֶׁנִּשְׁבְּרָה שְׁנִיָּה טְהוֹרָה. דְּכֵיוָן דְּנִשְׁבְּרוּ שְׁתֵּי הָאֲרֻכּוֹת, תּוּ לֹא חַזְיָא. וְלֹא דָמֵי לְהֵיכָא דְנִתְפָּרְקָה וְכֻלָּהּ קַיֶּמֶת, דְּהָתָם סוֹפוֹ לְהַחֲזִיר פְּרָקֶיהָ וּלְחַבְּרָהּ:

פרק יח - משנה ז
כֶּרַע. שֶׁל מִטָּה שֶׁהָיְתָה טְמֵאָה מִדְרָס. שֶׁדָּרַס הַזָּב עַל הַכֶּרַע כְּשֶׁהָיָה בִּפְנֵי עַצְמוֹ שֶׁלֹּא הָיָה מְחֻבָּר לַמִּטָּה: כֻּלָּהּ טְמֵאָה מִדְרָס. דְּחִבּוּר עֲבִיד לָהּ כְּאִלּוּ כֻלָּהּ גוּף אֶחָד: הָיְתָה טְמֵאָה טֻמְאַת שִׁבְעָה. כְּגוֹן שֶׁנָּגְעָה בְמֵת : טֻמְאַת עָרֶב. כְּגוֹן שֶׁנָּגְעָה בִטְמֵא מֵת: וְהַמִּטָּה טְהוֹרָה. דְּלֹא נִטְמֵאת בִּנְגִיעַת הַכֶּרַע. דְּרִאשׁוֹן הֲוָה, וְאֵין אָדָם וְכֵלִים מְקַבְּלִין טֻמְאָה אֶלָּא מֵאַב הַטֻּמְאָה: מַעְדֵּר. כְּלִי עֵץ שֶׁיֶּשׁ לוֹ שִׁנַּיִם וּמְהַפְּכִים בּוֹ אֶת הַתֶּבֶן, וְשִׁנָּיו שֶׁל מַתֶּכֶת. וּלְפִי שֶׁהַשִּׁנַּיִם שֶׁל פְּרָקִים וּנְעוּצִים בּוֹ, דִּינוֹ כְדִין מִטָּה שֶׁהִיא שֶׁל פְּרָקִים. וְהַאי וְכֵן, אַטֻּמְאַת שִׁבְעָה קָאֵי, דְּאֵין מַעְדֵּר מִטַּמֵּא מִדְרָס:

פרק יח - משנה ח
תְּפִלָּה. בִּתְפִלִּין שֶׁל רֹאשׁ שֶׁנָּגְעוּ בְמֵת אַיְרֵי, שֶׁיֶּשׁ בָּהֶן אַרְבָּעָה בָתִּים, וְכָל בַּיִת נִקְרֵאת קְצִיצָה: הִתִּיר קְצִיצָה. הֶעֱבִיר בַּיִת אֶחָד וְהֵבִיא אַחֵר תַּחְתָּיו: טְמֵאָה טְמֵא מֵת. דְּחִבּוּרָן מְשַׁוֵּי לְהוּ גוּף אֶחָד: וְכֵן שְׁנִיָּה וְכֵן שְׁלִישִׁית. כָּל זְמַן שֶׁהָרְבִיעִית קַיֶּמֶת, מֵחֲמַת שֶׁמְּחֻבֶּרֶת לָרְבִיעִית. אֲבָל כְּשֶׁהֶעֱבִיר גַּם הָרְבִיעִית וְהֶחְלִיפָהּ בְּאַחֶרֶת, פָּרְחָה מִנַּיְהוּ טֻמְאַת מֵת, וְנִשְׁתַּיֵּר בַּשָּׁלֹשׁ קְצִיצוֹת טֻמְאַת מַגַּע טְמֵא מֵת, שֶׁהֲרֵי נָגְעוּ בָרְבִיעִית כְּשֶׁחִבְּרָן וְהִיא עֲדַיִן קַיֶּמֶת. אֲבָל רְבִיעִית כְּשֶׁהֻחְלְפָה, אֵין בְּגוּפָהּ טֻמְאָה כְלָל, דְּאִי מִשּׁוּם דְּנָגְעָה בַשָּׁלֹשׁ, אֵין מַגָּע עוֹשֶׂה מַגָּע, שֶׁהַשָּׁלֹשׁ רִאשׁוֹנוֹת חֲדָשׁוֹת נִינְהוּ שֶׁנָּגְעוּ בִטְמֵא מֵת, וְאֵין הָרְבִיעִית הַחֲדָשָׁה מְקַבֶּלֶת מֵהֶן טֻמְאָה, וְאֵין טֻמְאָה בָרְבִיעִית הַחֲדָשָׁה אֶלָּא מֵחֲמַת שֶׁמְּחֻבֶּרֶת לַשָּׁלֹשׁ הַטְּמֵאוֹת מַגָּע. הִלְכָּךְ כְּשֶׁחָזַר וְהֶחְלִיף הַשָּׁלֹשׁ פַּעַם שְׁנִיָּה, טָהֲרוּ כֻלָּן וַאֲפִלּוּ מַגָּע לֵית בְּהוּ:

פרק יח - משנה ט
טְהוֹרָה. וַאֲפִלּוּ אִיכָּא אֲרֻכָּה וּשְׁתֵּי כְרָעַיִם, הוֹאִיל וְאֵין בְּדַעְתָּם לְהַחֲזִיר. וְכִי מַהֲנֵי, דַּוְקָא בְמָקוֹם שֶׁהַכֹּל לְאָדָם אֶחָד: מְקַבֶּלֶת טֻמְאָה מִכָּאן וּלְהַבָּא. אֲבָל לְמַפְרֵעַ לֹא, דְּכֵיוָן דְּחִלְּקוּהָ פָּרְחָה טֻמְאָה מִנַּהּ: מִטַּמֵּאת חֲבִילָה. אִם פֵּרְשָׁה הַכֶּרַע מִן הָאֲרֻכָּה וְעוֹמֶדֶת עִם הַקְּצָרָה, הָוֵי חִבּוּר, וְאִם נִטְמֵאת הַכֶּרַע נִטְמֵאת הַקְּצָרָה. כְּשֵׁם שֶׁכְּשֶׁכֻּלָּהּ שְׁלֵמָה אִם נִטְמָא אֶחָד מֵאֲבָרֶיהָ נִטְמֵאת כֻּלָּהּ, כָּךְ נַמִּי כְּשֶׁפֵּרַשׁ הַכֶּרַע מִן הָאֲרֻכָּה וְעוֹמֶדֶת עִם הַקְּצָרָה, נַעֲשֵׂית חֲבִילָה עִם הַקְּצָרָה, וּכְשֶׁנִּטְמֵאת הַכֶּרַע נִטְמֵאת הַקְּצָרָה עִמָּהּ: וּמִטַּהֶרֶת חֲבִילָה. שֶׁאִם הִטְבִּיל הַכֶּרַע עִם הַקְּצָרָה אֵין הַקְּצָרָה חוֹצֶצֶת עָלֶיהָ, כִּי הֵיכִי דְלָא חַיְצָה עָלֶיהָ כְּשֶׁמַּטְבִּילָהּ כְּשֶׁהִיא שְׁלֵמָה: וַחֲכָמִים אוֹמְרִים מִטַּמֵּאת אֵבָרִים. וְלֹא חֲבִילָה. דְּאִם נִטְמֵאת כֶּרַע, הִיא טְמֵאָה וְלֹא קְצָרָה, דְּאֵינָהּ חִבּוּר כִּשְׁלֵמָה: וּמִטַּהֶרֶת אֵבָרִים. כְּשֶׁבָּא לְהַטְבִּילָהּ מַטְבִּיל הַכֶּרַע בִּפְנֵי עַצְמָהּ, וְלֹא חֲבִילָה עִם הַקְּצָרָה, שֶׁהַקְּצָרָה חוֹצֶצֶת עָלֶיהָ, דְּאֵינָהּ חִבּוּר כִּשְׁלֵמָה, דְּמַטְבִּיל מִטָּה שְׁלֵמָה טְהוֹרָה אַף עַל פִּי שֶׁלֹּא פֵרְקָהּ, אֲבָל הַשְׁתָּא דְפֵרְשָׁה הַכֶּרַע מִן הָאֲרֻכָּה לֹא הָוֵי חֲבִילָה עִם הַקְּצָרָה, וְצָרִיךְ לְהַטְבִּילָהּ בִּפְנֵי עַצְמָהּ. וַהֲלָכָה כַּחֲכָמִים. כָּךְ נִרְאֵית פֵּרוּשׁ מִשְׁנָה זוֹ מִן הַתּוֹסֶפְתָּא, וְכָךְ פֵּרְשׁוּהָ רַבּוֹתַי:
