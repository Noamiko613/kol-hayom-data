פרק כד

פרק כד - משנה א
שְׁלֹשָׁה תְרִיסִין הֵן. שְׁלֹשָׁה דִינִים חֲלוּקִין זֶה מִזֶּה יֵשׁ בִּתְרִיסִין, דְּהַיְנוּ מָגִנִּים: תְּרִיס הַכָּפוּף. הַמְּצוּיִין אֶצְלֵנוּ, שֶׁמַּקִּיפִים אֶת הָאָדָם מִשָּׁלֹשׁ רוּחוֹת: טָמֵא מִדְרָס. דְּעָשׂוּי לִשְׁכִיבָה, שֶׁשּׁוֹכְבִים עָלָיו בַּמִּלְחָמָה. וְכָל שֶׁכֵּן שֶׁטָּמֵא טְמֵא מֵת, דְּקַיְמָא לָן כָּל הַטָּמֵא מִדְרָס טָמֵא טְמֵא מֵת: שֶׁמְּשַׂחֲקִים בּוֹ בַּקֻּמְפּוֹן. בַּשָּׂדֶה שֶׁל עֵמֶק הַמֶּלֶךְ, בָּאִים שְׁנַיִם כָּל אֶחָד חַרְבּוֹ בְיָדוֹ וּתְרִיס קָטָן עָגֹל שֶׁאֵינוֹ כָפוּף בְּיָדוֹ הַשְּׂמָאלִית, וּלְמֵדִין לְהָגֵן כָּל אֶחָד בְּמָגִנּוֹ שֶׁלֹּא יַכֵּהוּ חֲבֵרוֹ. וְקוֹרִין לוֹ אשגרימ"ר בְּלַעַ"ז: טָמֵא טְמֵא מֵת. וְהוּא הַדִּין שֶׁטָּמֵא טֻמְאַת שֶׁרֶץ וּנְבֵלָה וּשְׁאָר טֻמְאוֹת כֻּלָּן, חוּץ מִמִּדְרָס, שֶׁאֵינוֹ נַעֲשֶׂה אַב הַטֻּמְאָה אִם שָׁכַב עָלָיו הַזָּב אוֹ יָשַׁב, אֶלָּא רִאשׁוֹן כְּמַגָּעוֹ שֶׁל זָב: וְדִיצַת הָעַרְבִיִּין. תְּרִיס קָטָן בְּיוֹתֵר שֶׁהָעַרְבִיִּים עוֹשִׂים לְדִיצָה וּלְשִׂמְחָה וְלִשְׂחוֹק, וְאֵינוֹ כְלִי שֶׁל תַּשְׁמִישׁ:

פרק כד - משנה ב
כְּקַתֶּדְרָא. שֶׁהִיא קְצָרָה וּמֻקֶּפֶת מִשָּׁלֹשׁ רוּחוֹת: טְמֵאָה מִדְרָס. דִּמְיֻחֶדֶת לִישִׁיבָה: כְּמִטָּה. שֶׁהִיא אֲרֻכָּה וּמְיֻחֶדֶת לְהַנִּיחַ בָּהּ פְּרַקְמַטְיָא, וְהַשּׁוֹכֵב בָּהּ אוֹמְרִים לוֹ עֲמֹד וְנַעֲשֶׂה מְלַאכְתֵּנוּ: וְשֶׁל אֲבָנִים. הָעֲשׂוּיָה לְהוֹלִיךְ בָּהּ אֲבָנִים: טְהוֹרָה מִכְּלוּם. לְפִי שֶׁפְּרוּצָה מִתַּחְתִּיתָהּ נְקָבִים גְּדוֹלִים כְּמוֹצִיא רִמּוֹן:

פרק כד - משנה ג
שֶׁנִּסְדְּקָה טְמֵאָה מִדְרָס. דְּכֵיוָן שֶׁנִּסְדְּקָה וְאֵינָהּ רְאוּיָה לְלִישָׁה, מְיַחֲדִין אוֹתָהּ לִישִׁיבָה: טְמֵאָה טְמֵא מֵת. דִּכְלִי תַשְׁמִישׁ הִיא: הַבָּאָה בַמִּדָּה. שֶׁמַּחְזֶקֶת אַרְבָּעִים סְאָה בְּלַח שֶׁהֵן כּוֹרַיִם בְּיָבֵשׁ: טְהוֹרָה מִכְּלוּם. שֶׁמִּפְּנֵי כָבְדָהּ וְגָדְלָהּ אֵינָהּ מִטַּלְטֶלֶת מְלֵאָה, וַאֲנַן דּוּמְיָא דְשַׂק בָּעִינַן דְּמִטַּלְטֵל מָלֵא וְרֵיקָן:

פרק כד - משנה ד
תֵּבָה שֶׁפִּתְחָהּ מִצִּדָּהּ. מְשַׁמֶּשֶׁת יְשִׁיבָה עִם מְלַאכְתָּהּ, שֶׁיְּכוֹלִים לְהִשְׁתַּמֵּשׁ בָּהּ כְּשֶׁהוּא יוֹשֵׁב, בִּזְמַן שֶׁפִּתְחָהּ מִצִּדָּהּ. מַה שֶּׁאֵין כֵּן כְּשֶׁפִּתְחָהּ מִלְּמַעְלָן. וְהַבָּאָה בְמִדָּה דִתְנִינַן בְּמַתְנִיתִין שֶׁהִיא טְהוֹרָה מִכְּלוּם, אַאֵינָהּ מְיֻחֶדֶת לְמִדְרָס קָאֵי, דְּהַטְּמֵאָה מִדְרָס אֲפִלּוּ בָאָה בְמִדָּה לְעוֹלָם הִיא טְמֵאָה :

פרק כד - משנה ה
תַּרְבּוּסִים. כֵּלִים שֶׁל עוֹר כְּעֵין אַרְגָּזִים: שֶׁל סַפָּרִים. מַקִּיזִים דָּם: טָמֵא מִדְרָס. גָּדוֹל הוּא וַחֲזִי לִישִׁיבָה אוֹ לְמִזְגָּא עֲלֵיהּ : שֶׁאוֹכְלִים עָלָיו. כְּלִי תַשְׁמִישׁ הוּא : וְשֶׁל זֵיתִים. שֶׁסּוֹחֲטִין בּוֹ הַזֵּיתִים , לָא חֲשִׁיב כְּלִי שֶׁל מְשַׁמְּשֵׁי אָדָם:

פרק כד - משנה ו
בְּסִיסָאוֹת. וְאֶת כַּנּוֹ מְתַרְגְּמִינַן וְיַת בְּסִיסֵיהּ: וְשֶׁל דְּלֻפְקִי. כְּלִי עֵץ שֶׁמְּשִׂימִים בּוֹ צְלוֹחִיּוֹת וַאֲשִׁישׁוֹת וָאֳכָלִים וּמַשְׁקִים, וּמִמֶּנּוּ נוֹטְלִים וְנוֹתְנִים עַל הַשֻּׁלְחָן. וְהַבָּסִיס שֶׁלְּפָנָיו אֵינוֹ לִישִׁיבָה , אֲבָל תּוֹרַת כְּלִי עָלָיו: וְשֶׁל מִגְדָּל. אוֹצָר שֶׁל עֵץ. ארמריא"ו בְּלַעַ"ז: טָהוֹר מִכְּלוּם. דְּלָאו כְּלִי הוּא וְצוּרָתוֹ מוֹכַחַת עָלָיו:

פרק כד - משנה ז
שָׁלֹשׁ פִּנְקָסִיּוֹת הֵן. הָעֲשׂוּיִין לִכְתֹּב בָּהֶן. כְּמוֹ חֶנְוָנִי עַל פִּנְקָסוֹ: אֱפִיפוֹרִין. לוּחַ שֶׁמַּנִּיחִים עָלָיו אָבָק שֶׁל עָפָר וְכוֹתְבִים בּוֹ חֶשְׁבּוֹנוֹת. וְגָדוֹל הוּא וַחֲזִי לִישִׁיבָה: שֶׁיֶּשׁ לָהּ בֵּית קִבּוּל שַׁעֲוָה. לוּחַ שֶׁטָּחִין פָּנָיו בְּדוֹנַג וְרוֹשְׁמִין עַל הַדּוֹנַג בְּחֶרֶט: וַחֲלָקָה. שֶׁאֵין בָּהּ שַׁעֲוָה, וְכוֹתְבִים עָלֶיהָ בִּדְיוֹ, וְאֵין לָהּ בֵּית קִבּוּל:

פרק כד - משנה ח
שֶׁל זַגָּגִים. שֶׁמַּנִּיחִים עָלֶיהָ כְּלֵי זְכוּכִית : סָרָגִין. עוֹשֵׂי הַמֶּרְכָּבוֹת. בָּעֲרָבִי קוֹרִין לְמֶרְכָּב סיר"ג. וְיֵשׁ מְפָרְשִׁים, שֶׁמְּסָרְגִים אֶת הַשְּׂבָכוֹת. וּטְהוֹרָה מִכְּלוּם לְפִי שֶׁאֵינָהּ מִמְּשַׁמְּשֵׁי אָדָם:

פרק כד - משנה ט
מַשְׁפֵּלוֹת. קֻפּוֹת עֲשׂוּיוֹת לְהוֹצִיא בָהֶם זְבָלִים לַשָּׂדוֹת: שֶׁל זֶבֶל טְמֵאָה מִדְרָס. לְפִי שֶׁרְאוּיָה לִישִׁיבָה: וְהַפֻּחְלָץ. עָשׂוּי כְּעֵין מִכְבָּר מַעֲשֵׂה רֶשֶׁת , וְהַנְּקָבִים שֶׁלּוֹ רְחָבִים יוֹתֵר מִמַּשְׁפֶּלֶת שֶׁל תֶּבֶן. וְאֵינָהּ רְאוּיָה אֲפִלּוּ לְקַבֵּל תֶּבֶן, שֶׁנְּקָבֶיהָ רְחָבִים בְּיוֹתֵר, וְגַם אֵינָהּ רְאוּיָה לִישִׁיבָה, שֶׁחֲבָלִים שֶׁלָּהּ קָשִׁים וְאֵין רְאוּיִין לֵישֵׁב עֲלֵיהֶן, הִלְכָּךְ טְהוֹרָה מִכְּלוּם:

פרק כד - משנה י
מַפָּצִים. כְּמִין מַחְצְלָאוֹת עֲשׂוּיוֹת מֵחִילָף וּמִסִּיב וְקָנֶה וְגֹמֶא וְגֶמִי וְכַיּוֹצֵא בָהֶן: וְשֶׁל צַבָּעִים. שֶׁהַצַּבָּעִים נוֹתְנִים עֲלֵיהֶם הַבְּגָדִים: טָמֵא טְמֵא מֵת. שֶׁאֵינָם מְיֻחָדִים לִישִׁיבָה, אֲבָל תּוֹרַת כֵּלִים יֵשׁ לָהֶן: וְשֶׁל גִּתּוֹת. הָעֲשׂוּיוֹת לְכַסּוֹת בָּהֶן עֲנָבִים וְזֵיתִים :

פרק כד - משנה יא
חֲמָתוֹת. נֹאדוֹת שֶׁל עוֹר: תּוּרְמְלִין. כִּיסִין גְּדוֹלִים שֶׁל עוֹר, שֶׁהָרוֹעֶה מַנִּיחַ חֲפָצָיו לְתוֹכָן: הַמְקַבְּלִים כַּשִּׁעוּר. הַמְפֹרָשׁ לְמַעְלָה בְּרֵישׁ פֶּרֶק כָּרִים, הַחֵמֶת שֶׁל שִׁבְעַת קַבִּים, וְהַתּוּרְמֵל שֶׁל חֲמֵשֶׁת קַבִּין. וְכָל שֶׁכֵּן אִם מַחֲזִיקִים יוֹתֵר, דְּאָז מְשַׁמְּשִׁים יְשִׁיבָה עִם מְלַאכְתָּן. אֲבָל פָּחוֹת מִכָּאן לֹא: וְשֶׁל עוֹר הַדָּג טְהוֹרִים מִכְּלוּם. דְּכָל הַבָּא מִבְּרִיּוֹת שֶׁבַּיָּם, טָהוֹר:

פרק כד - משנה יב
לְשָׁטִיחַ. לִשְׁטֹחַ בָּאָרֶץ לֵישֵׁב עָלָיו: לְתַכְרִיךְ הַכֵּלִים. לִכְרֹךְ בּוֹ אֶת הַכֵּלִים, כְּגוֹן סַכִּינִים וּמִסְפָּרַיִם וּמְחָטִים כְּדֵי לְשָׁמְרָן: וְשֶׁל רְצוּעוֹת וְשֶׁל סַנְדָּלִים. עוֹר הָעוֹמֵד לַחְתֹּךְ מִמֶּנּוּ רְצוּעוֹת וְסַנְדָּלִים, טָהוֹר מִכְּלוּם, דִּמְחֻסַּר מְלָאכָה הוּא. אֲבָל רְצוּעוֹת הַמְתֻקָּנוֹת כְּבָר, טְמֵאוֹת, כִּדְמוּכָח בְּמַסֶּכֶת נְגָעִים פֶּרֶק י"א:

פרק כד - משנה יג
לְוִילוֹן טָמֵא טְמֵא מֵת. הֶעָשׂוּי לְמָסָךְ לִפְנֵי הַפֶּתַח, טָמֵא, לְפִי שֶׁהַשַּׁמָּשׁ מִתְעַטֵּף בְּשׁוּלָיו לִפְעָמִים וּמִתְחַמֵּם בּוֹ: שֶׁל צוּרוֹת. בֶּגֶד שֶׁהוּא בְיַד רוֹקֵם וּבוֹ מִינֵי צוּרוֹת כְּדֵי לִרְאוֹת בּוֹ לַעֲשׂוֹת כְּמוֹתוֹ בְּבֶגֶד אַחֵר:

פרק כד - משנה יד
שֶׁל יָדַיִם טְמֵאָה מִדְרָס. לְפִי שֶׁפְּעָמִים נוֹתְנָהּ עַל הַכֶּסֶת וְיָשֵׁן עָלֶיהָ: וְשֶׁל תַּכְרִיךְ נִבְלֵי בְנֵי לֵוִי. שֶׁהָיוּ הַלְוִיִּים כּוֹרְכִים כְּלֵי הַשִּׁיר שֶׁלָּהֶן בְּמִטְפַּחַת. וְדֶרֶךְ שְׁאָר בְּנֵי אָדָם לַעֲשׂוֹת לָהֶן תִּיק שֶׁל עוֹר, וְאַף הֵן טְהוֹרִים :

פרק כד - משנה טו
פְּרָקְלִינִין. צוּרַת יָד שֶׁל עוֹר שֶׁנּוֹשְׂאִים צַיָּדֵי עוֹפוֹת כְּשֶׁתּוֹפְסִים בִּידֵיהֶן הָעוֹף שֶׁקּוֹרִין אשטו"ר וְאשפרוי"ר, וּבוֹ יוֹצְאִים לָצוּד חַיָּה אוֹ עוֹף: טָמֵא מִדְרָס. לְפִי שֶׁנִּשְׁעָן עָלָיו: וְשֶׁל חֲגָבִים. הַהוֹלְכִים לָצוּד חֲגָבִים וְנוֹתְנִים אוֹתָם בּוֹ: וְשֶׁל קוֹצִים. לִלְקֹט קוֹצִים. וְאִית דְּגָרְסֵי שֶׁל קַיָּצִים, הַמְיַבְּשִׁים פֵּרוֹת הַקַּיִץ, כְּגוֹן הָעוֹשִׂים גְּרוֹגָרוֹת וְצִמּוּקִים בַּשָּׂדֶה:

סְבָכוֹת. כִּפָּה שֶׁנּוֹשְׂאוֹת הַנָּשִׁים עַל רֹאשָׁן, עֲשׂוּיָה כְּעֵין רֶשֶׁת שֶׁיֶּשׁ בָּהּ נְקָבִים דַּקִּין: שֶׁל יַלְדָּה. רָאוּי לִישִׁיבָה, וּלְכָךְ טְמֵאָה מִדְרָס: שֶׁל זְקֵנָה. מַעֲשֶׂיהָ מוֹכִיחִים עָלֶיהָ שֶׁאֵינָהּ רְאוּיָה לִישִׁיבָה. וּבַתּוֹסֶפְתָּא תַנְיָא, שֶׁל זְקֵנָה טְמֵאָה מִדְרָס, שֶׁאֵינָהּ מַקְפֶּדֶת עָלֶיהָ וּפְעָמִים יוֹשֶׁבֶת עָלֶיהָ . וְשֶׁל יַלְדָּה שֶׁמַּקְפֶּדֶת עַל כֵּלֶיהָ, אֵינָהּ יוֹשֶׁבֶת עָלֶיהָ, וּלְכָךְ טְמֵאָה טְמֵא מֵת: וְשֶׁל יוֹצְאוֹת הַחוּץ. סוּדָר שֶׁנּוֹתְנוֹת הַנָּשִׁים עַל רֹאשָׁן כְּשֶׁיּוֹצְאוֹת לַחוּץ. וְיֵשׁ מְפָרְשִׁים, יוֹצְאוֹת הַחוּץ, תַּרְגּוּם זוֹנָה נַפְקַת בָּרָא, כְּלוֹמַר סְבָכוֹת שֶׁל זוֹנוֹת: טְהוֹרוֹת. לְפִי שֶׁאֵינָן חֲשׁוּבוֹת כְּלִי. פֵּרוּשׁ אַחֵר, סְבָכוֹת שֶׁנִּקְרְעוּ וְרֹב שְׂעַר הָאִשָּׁה יוֹצֵא לַחוּץ, שֶׁאֵינָן מְקַבְּלוֹת רֹב שְׂעַר רֹאשָׁהּ שֶׁל אִשָּׁה:

מְהוּהָה. בֶּגֶד יָשָׁן וּבָלוּי: שֶׁטְּלָיָהּ. כְּמוֹ שֶׁטְּלָאָהּ. מִלְּשׁוֹן טְלַאי עַל גַּבֵּי טְלַאי. כְּלוֹמַר שֶׁשָּׂם הַיְשָׁנָה טְלַאי עַל הַחֲדָשָׁה: הוֹלְכִים אַחַר הַבְּרִיאָה. דָּנִין אוֹתָהּ כְּדִין הַבְּרִיאָה, אִם טְמֵאָה טְמֵאָה, וְאִם טְהוֹרָה טְהוֹרָה : קְטַנָּה עַל הַגְּדוֹלָה. בֵּין שְׁתֵּיהֶן מְהוּהוֹת בֵּין שְׁתֵּיהֶן בְּרִיאוֹת: הוֹלְכִים אַחַר הַגְּדוֹלָה. וְאִם הַגְּדוֹלָה נְקוּבָה כְמוֹצִיא רִמּוֹן שֶׁבִּשְׁבִירָה כָזוֹ נִטְהֲרָה, גַּם הַקְּטַנָּה הַמְחֻבֶּרֶת עִמָּהּ טְהוֹרָה אַף עַל פִּי שֶׁהִיא שְׁלֵמָה. וְאִם הַגְּדוֹלָה שְׁלֵמָה וַהֲרֵי הִיא מְקַבֶּלֶת טֻמְאָה, אַף הַקְּטַנָּה הַמְחֻבֶּרֶת עִמָּהּ טְמֵאָה אַף עַל פִּי שֶׁהִיא נְקוּבָה בְמוֹצִיא רִמּוֹן: הוֹלְכִים אַחַר הַפְּנִימִית. אִם טְמֵאָה הַפְּנִימִית טְמֵאָה הַחִיצוֹנָה, וְאִם הַפְּנִימִית טְהוֹרָה הַחִיצוֹנָה נַמִּי טְהוֹרָה: רַבִּי שִׁמְעוֹן אוֹמֵר כַּף מֹאזְנַיִם. לְפָרוֹשֵׁי מִלְּתָא דְתַנָּא קַמָּא קָאָתֵי. וּבְכַף מֹאזְנַיִם שֶׁל מַתָּכוֹת טְמֵאָה אַיְרֵי , דְּאִם טְלָאָהּ וּכְלֵי אֲבָנִים וּכְלֵי גְלָלִים וּכְלֵי אֲדָמָה, טְהוֹרִים מִכְּלוּם. וְכֵן כָּל דָּבָר הַבָּא מִן הַיָּם, טָהוֹר. וּכְלִי עֵץ הַבָּא בְמִדָּה, שֶׁמַּחֲזִיק אַרְבָּעִים סְאָה בְלַח שֶׁהֵם כּוֹרַיִם
