פרק יב - משנה א
טַבַּעַת אָדָם. כְּגוֹן טַבַּעַת שֶׁל אֶצְבַּע. אֲבָל טַבַּעַת שֶׁהִתְקִינָהּ לַחְגֹּר בּוֹ מָתְנָיו וְלִקְשֹׁר בָּהּ בֵּין כְּתֵפָיו, טְהוֹרָה, וְהַיְנוּ טַבַּעַת כֵּלִים. וְטַעֲמָא, מִשּׁוּם דְּלֹא רַבֵּי קְרָא לְטֻמְאָה אֶלָּא תַכְשִׁיטֵי אָדָם וְלֹא תַכְשִׁיטֵי בְהֵמָה וְכֵלִים : קוֹרַת הַחִצִּים. אִית דִּמְפָרְשֵׁי אַשְׁפָּה שֶׁהַחִצִּים מֻנָּחִים בָּהּ, וְאִית דִּמְפָרְשֵׁי הַמַּטָּרָה לַחֵץ: וְשֶׁל אֲסִירִין. עוֹשִׂין כְּמִין פְּגִימָה בַקּוֹרָה לְהַכְנִיס בָּהּ רֶגֶל אָדָם וְאֵין יָכוֹל לָצֵאת. וּבִלְשׁוֹן מִקְרָא קוֹרֵהוּ סַד, וְתָשֵׂם בַּסַּד רַגְלַי (אִיּוֹב יג). וְקוֹרוֹת שֶׁל חִצִּים וְשֶׁל אֲסִירִין הָאֲמוּרִין כָּאן, דְּמַתֶּכֶת נִינְהוּ וְלֹא שֶׁל עֵץ: קוֹלָר. שֶׁנּוֹתְנִים בְּצַוַּאר הָאֲסִירִים. . תַּרְגּוּם וַיִּתְּנֻהוּ בַסּוּגַר (יְחֶזְקֵאל יט), בְּקוֹלָרִין. וּמִשּׁוּם דְּקוֹלָר עָשׂוּי שֶׁיִּשָּׂאֶנּוּ עִמּוֹ בְכָל מָקוֹם שֶׁהוֹלֵךְ, חֲשׁוּב כְּלִי מַעֲשֶׂה טְפֵי מִקּוֹרַת הָאֲסִירִים שֶׁאֵינוֹ זָז בָּהּ מִמְּקוֹמוֹ, הִלְכָּךְ קוֹלָר טָמֵא: שֶׁיֶּשׁ בָּהּ בֵּית נְעִילָה. עוֹשִׂים לַשַּׁלְשֶׁלֶת בֵּית נְעִילָה וְסוֹגְרִים בְּמַפְתֵּחַ, וַחֲשִׁיבָא כְּלִי: לִכְפִיתָה. שַׁלְשְׁלָאוֹת שֶׁכּוֹפְתִים בָּהֶן הַקּוֹף וְגוּרֵי הַכְּלָבִים הַקְּטַנִּים. וּלְנוֹי הֵן עֲשׂוּיִין וַהֲווּ תַכְשִׁיט דִּבְהֵמָה וְלֹא מִטַּמֵּא: סִיטוֹנוֹת. סוֹחֲרִים גְּדוֹלִים שֶׁמּוֹכְרִים בְּמִדָּה גַסָּה. וְיֵשׁ לָהֵן שַׁלְשֶׁלֶת שֶׁקּוֹבְעִים בְּרֹאשָׁהּ אֶחָד הַמִּדָּה, סְאָה אוֹ תַרְקַב, וְרֹאשָׁהּ הַשֵּׁנִי נוֹתְנִים בְּטַבַּעַת שֶׁבַּדֶּלֶת אוֹ בַקּוֹרָה, וְסוֹגְרִים עָלֶיהָ בְּמַפְתֵּחַ כְּדֵי שֶׁלֹּא תִגָּנֵב הַמִּדָּה: וְשֶׁל בַּעֲלֵי בָתִּים טְהוֹרָה. דִּסְתָמָא עֲשׂוּיָה לְנוֹי וְתַכְשִׁיט לַכְּלִי: מַפְתֵּחַ אֶחָד. מֵחֻלְיָא אַחַת. דְּשַׁלְשֶׁלֶת בְּחֻלְיָא אַחַת לֹא עָבְדֵי אֱינָשֵׁי, וְלָא חֲשִׁיבָא כְלִי: אוֹ שֶׁקָּשַׁר חִלָּזוֹן בְּרֹאשָׁהּ. כִּדְמוּת חִלָּזוֹן שֶׁהוּא דָג שֶׁבַּיָּם שֶׁמִּדָּמוֹ צוֹבְעִים הַתְּכֵלֶת קָשַׁר לָהּ בְּרֹאשָׁהּ עָשׂוּי מִבַּרְזֶל, וְהוּא מְשַׁוֵּי לַהּ מָנָא:

פרק יב - משנה ב
קְנֵה מֹאזְנַיִם שֶׁל סָרוֹקוֹת. סוֹרְקֵי צֶמֶר וּפִשְׁתִּים, וּמוֹכְרִים אוֹתוֹ בְּמִנְיַן לִיטְרָאוֹת, כְּשֶׁרוֹצִין לִשְׁקֹל יֵשׁ לָהֶם מַקֵּל אָרֹךְ שֶׁל עֵץ וְתָלוּי בְּרֹאשׁוֹ כְמִין טַבַּעַת מִלְּמַעְלָה, וְאֻנְקְלָיוֹת שֶׁל בַּרְזֶל תְּלוּיִין מִלְּמַטָּה בְשַׁלְשֶׁלֶת, עֲשׂוּיִין כְּמִין וָוִין וְרָאשֵׁיהֶם כְּפוּפִים מִלְּמַעְלָה, וְיֵשׁ לְאוֹתוֹ מַקֵּל טַבַּעַת אַחֶרֶת שֶׁמְּהַלֶּכֶת עַל פְּנֵי כָל הַמַּקֵּל וְאֵינָהּ יְכוֹלָה לָצֵאת מִמֶּנּוּ, וּכְמִין עֶשֶׁת שֶׁל בַּרְזֶל אוֹ שֶׁל נְחֹשֶׁת עָב וְכָבֵד תָּלוּי בְּאוֹתָהּ טַבַּעַת, וּכְשֶׁבָּא לִשְׁקֹל שַׂק שֶׁל צֶמֶר אוֹ שֶׁל מַטְוֶה, תּוֹחֵב אֶת הָאֻנְקְלִי בַּשַּׂק וְאוֹחֵז בַּטַּבַּעַת שֶׁכְּנֶגְדּוֹ לְמַעְלָה וּמַעֲלֵהוּ בָּאֲוִיר, וְהַטַּבַּעַת שֶׁבּוֹ הָעֶשֶׁת שֶׁל בַּרְזֶל תָּלוּי מַרְחִיקָהּ מִטַּבַּעַת שֶׁאוֹחֵז בָּהּ, וּמַכְבֶּדֶת עַל הַקָּנֶה עַד שֶׁיָּבֹא בְיֹשֶׁר שֶׁלֹּא יַכְרִיעַ לְשׁוּם צַד יוֹתֵר, וּלְפִי הַרְחָקַת הַטַּבַּעַת שֶׁבּוֹ הָעֶשֶׁת שֶׁל בַּרְזֶל יוֹדֵעַ חֶשְׁבּוֹן הַמִּשְׁקָל, שֶׁיֵּשׁ סִימָנִים בַּמַּקֵּל, עַד כָּאן לִיטְרָא אַחַת, עַד כָּאן עֲשָׂרָה, עַד כָּאן מֵאָה. וְזֶה נִקְרָא פֶּלֶס בִּלְשׁוֹן מִקְרָא, כְּדִכְתִיב (יְשַׁעְיָה מ) וְשָׁקַל בַּפֶּלֶס הָרִים. פֶּלֶס וּמֹאזְנֵי מִשְׁפָּט לַה' (מִשְׁלֵי טז): וְשֶׁל בַּעֲלֵי בָתִּים. אֵין רְגִילִים לָתֵת בּוֹ אֻנְקְלָיוֹת שֶׁהֵן אוֹתָן הַוָּוִין שֶׁתּוֹחֲבִים בַּשַּׂק לְהַעֲלוֹתוֹ, אֶלָּא בְלֹא אֻנְקְלָיוֹת קוֹשְׁרִים בַּחֶבֶל וְתוֹלִין בַּקָּנֶה. אֻנְקְלָיוֹת גָּרְסִינַן, וְלֹא גָרְסִינַן אֻנְקְיוֹת : אֻנְקְלִי שֶׁל כַּתָּפִין. עֵץ אָרֹךְ כְּעֵין מוֹט שֶׁהַכַּתָּף מַנִּיחַ עַל כְּתֵפוֹ וְתוֹלֶה בוֹ שְׁנֵי שַׂקִּים אֶחָד לְפָנָיו וְאֶחָד מֵאַחֲרָיו בִּשְׁנֵי אֻנְקְלָיוֹת שֶׁל בַּרְזֶל הַתְּחוּבוֹת בַּמּוֹט עַל שְׁנֵי קְצוֹתָיו: שֶׁל רוֹכְלִים. מוֹכְרֵי בְשָׂמִים וְתַמְרוּקֵי נָשִׁים: טְמֵאָה. מִפְּנֵי שֶׁיֶּשׁ לָהּ בֵּית קִבּוּל: שֶׁלְּפָנָיו טְמֵאָה. לְפִי שֶׁהוּא מִשְׁתַּמֵּשׁ בָּהּ: וְשֶׁלְּאַחֲרָיו טְהוֹרָה. שֶׁאֵינוֹ מִשְׁתַּמֵּשׁ בָּהּ אֶלָּא לִשְׁמִירָה, שֶׁלֹּא יִפֹּל הַמַּשְּׂאוֹי מֵאַחֲרָיו. וְאֵין הֲלָכָה כְּרַבִּי יְהוּדָה: דַּרְגָּשׁ. מִטָּה קְטַנָּה שֶׁל עוֹר וְתוֹלִין רְצוּעוֹת בִּשְׂפַת הָעוֹר סָבִיב וְיֵשׁ אֻנְקְלָיוֹת שֶׁל בַּרְזֶל בַּאֲרֻכּוֹת הַמִּטָּה וּכְשֶׁרוֹצִים לִנְטוֹתָהּ עוֹנְבִין הָרְצוּעוֹת בְּאוֹתָן אֻנְקְלָיוֹת: נַקְלִיטִין. שְׁנֵי עַמּוּדִים שֶׁמַּעֲמִידִין בְּאֶמְצַע דָּפְנֵי הַמִּטָּה, אֶחָד מַרְאֲשׁוֹתֶיהָ וְאֶחָד בְּמַרְגְּלוֹתֶיהָ, וּמְפֻצָּלִין בְּרָאשֵׁיהֶן, וּמַנִּיחִין קָנֶה מִזֶּה לָזֶה וּפוֹרְסִים עָלָיו סָדִין. וּפְעָמִים שֶׁאֵין הָעַמּוּדִים מְפֻצָּלִים, אֶלָּא כְעֵין שְׁנֵי וָוִין מַנִּיחִין בְּרָאשֵׁי הַנַּקְלִיטִין, וְהוּא אֻנְקְלִי: שִׁדָּה. תֵּבָה קְטַנָּה: אָקוֹן. תֵּבָה שֶׁעוֹשִׂין הַצַּיָּדִים לָצוּד בָּהּ דָּגִים : כָּל הַמְחֻבָּר לַטָּמֵא. אֻנְקְלִי הַמְחֻבֶּרֶת לִכְלִי הַמְקַבֵּל טֻמְאָה, טָמֵא: לַטָּהוֹר. כְּגוֹן מְנוֹרַת הָעֵץ שֶׁהִיא טְהוֹרָה, טְהוֹרָה: אַחַת אַחַת בִּפְנֵי עַצְמָן. כָּל אֻנְקְלִי שֶׁפָּרְשָׁה מִכְּלִי: טְהוֹרָה. לְפִי שֶׁאֵין לָהּ שֵׁם בִּפְנֵי עַצְמוֹ:

טֶנִי. הוּא כְלִי שֶׁל מַתֶּכֶת מַחֲזִיק שְׁלֹשָׁה קַבִּים. וְהוּא מִלְּשׁוֹן וְשַׂמְתָּ בַטֶּנֶא. וּבַעֲלֵי בָתִּים מִשְׁתַּמְּשִׁין בּוֹ שֶׁמַּנִּיחִין שָׁם גְּרוּטָאוֹת מִשִּׁבְרֵי כְלֵי מַתָּכוֹת: רַבָּן גַּמְלִיאֵל מְטַמֵּא. דְּסָבַר כִּסּוּי חֲשׁוּב כְּלִי הוֹאִיל וְיֵשׁ לוֹ בֵית קִבּוּל: וַחֲכָמִים מְטַהֲרִין. דְּסָבְרֵי אֵין כִּסּוּי חֲשִׁיב כְּלִי. וַהֲלָכָה כַּחֲכָמִים: וְשֶׁל רוֹפְאִים טָמֵא. לְדִבְרֵי הַכֹּל. מִפְּנֵי שֶׁהָרוֹפְאִים מַנִּיחִין בּוֹ סַמְמָנִים: מִגְדָּל. בֵּית הָאוֹצָר שֶׁל עֵץ. ארמריא"ו בְּלַעַ"ז. וּבַעֲרָבִי אלמנס"ר: וְשֶׁל רוֹפְאִים טְמֵאָה. מִפְּנֵי שֶׁנּוֹתֵן עָלֶיהָ אֶת הָאִסְפְּלָנִית וְתוֹלֶה בָהּ הַמִּסְפָּרַיִם: הַיַּתּוּכִין. מֶלְקָחַיִם שֶׁנּוֹטֵל בָּהֶם הַכְּלִי שֶׁהִתִּיךְ בּוֹ אֶת הַכֶּסֶף, לְהָרִיקוֹ לְתוֹךְ כְּלִי אַחֵר: וְהַפְּרָכִין. בַּתּוֹסֶפְתָּא מְפָרֵשׁ אֵלּוּ הֵן פְּרָכִין, הַקְּבוּעִים בַּכִּירַיִם. וְרַבּוֹתַי לֹא פֵרְשׁוּהָ לִי : עַקְרַב בֵּית הַבַּד. בַּרְזֶל עָקוּם דּוֹמֶה לְעַקְרָב, הַקָּבוּעַ בְּקוֹרַת בֵּית הַבַּד שֶׁדּוֹרְכִים בָּהּ הַזֵּיתִים: אֻנְקְלִי שֶׁבַּכְּתָלִים. וָוִין שֶׁל בַּרְזֶל הַקְּבוּעִים בַּכְּתָלִים לִתְלוֹת בָּהֶם כֵּלִים: טְהוֹרָה. דְּאֵין תַּשְׁמִישָׁן אֶלָּא עִם הַקַּרְקַע:

פרק יב - משנה ד
מַסְמֵר הַגָּרָע. אִזְמֵל שֶׁל אֻמָּן הַמַּקִּיז דָּם. וְנִקְרָא גָרָע עַל שֵׁם שֶׁמְּגָרֵעַ אֶת הַדָּם. וְיֵשׁ מְפָרְשִׁים, מַסְמֵר שֶׁל נַפָּחִים שֶׁקָּבוּעַ לָהֶן מַסְמֵר בַּסַּדָּן, כְּשֶׁמְּסַלְּקִים חֲתִיכָה שֶׁל בַּרְזֶל מִן הָאֵשׁ וְרוֹצֶה לַחְתֹּךְ מִמֶּנָּה מַנִּיחָהּ עַל הַמַּסְמֵר וּמַכֶּה בַקֻּרְנָס וְנִגְרַעַת: אֶבֶן הַשָּׁעוֹת. אֶבֶן שֶׁיֶּשׁ בָּהּ קַוִּים וּרְשׁוּמִים עֲלֵיהֶם שְׁמוֹת הַשָּׁעוֹת, וּמַסְמְרִים תְּחוּבִים בָּהּ שֶׁבָּהֶם מְכַוְּנִים אֶת הַשָּׁעוֹת : מַסְמֵר הַגַּרְדִּים. מַסְמֵר אָרֹךְ שֶׁהָאוֹרֵג מַכְנִיס בְּקָנֶה דַק וּמְחַבֵּר עַל הַקָּנֶה אֶת הַמַּטְוֶה : אָרוֹן שֶׁל גָּרוֹסוֹת. הָעוֹשִׂין גְּרִיסִין שֶׁל פּוֹל בָּרֵחַיִם שֶׁלָּהֶן יֵשׁ לָהֶן כְּמִין אָרוֹן שֶׁל עֵץ : רַבִּי צָדוֹק מְטַמֵּא. אוֹתוֹ אֲרוֹן, דְּתוֹרַת כְּלִי עָלָיו: וַחֲכָמִים מְטַהֲרִין. דִּכְלִי עֵץ הֶעָשׂוּי לְנַחַת הוּא וְאֵינוֹ עָשׂוּי לִטַּלְטֵל. וְרַבִּי צָדוֹק סָבַר, פְּעָמִים מְטַלְטְלִין אוֹתוֹ לְתַקְּנוֹ. וְאֵין הֲלָכָה כְּרַבִּי צָדוֹק: הָיְתָה עֲגָלָה. שֶׁהָאָרוֹן מֻנָּח עָלֶיהָ, שֶׁל מַתֶּכֶת: טְמֵאָה. שֶׁהֲרֵי הִיא עֲשׂוּיָה לְתַשְׁמִישׁ:

פרק יב - משנה ה
שֶׁהִתְקִינוֹ. שֶׁעִקֵּם רֹאשׁוֹ אוֹ חִדְּדוֹ כְּדֵי שֶׁיִּהְיֶה פּוֹתֵחַ וְנוֹעֵל בּוֹ: הֶעָשׂוּי לִשְׁמִירָה. לְסִימָן לִרְאוֹת אִם נִכְנַס שָׁם אָדָם אִם לָאו, שֶׁאִם לֹא יִמְצָאֶנּוּ כְּמוֹ שֶׁהִנִּיחוֹ בְּיָדוּעַ שֶׁאָדָם בָּא לְשָׁם: עַד שֶׁיִּצְרְפֶנּוּ. בָּאֵשׁ לְצֹרֶךְ זֶה . וַהֲלָכָה כַּחֲכָמִים: מַסְמֵר הַשֻּׁלְחָנִי. מַסְמֵר קָבוּעַ בָּעַמּוּד שֶׁלִּפְנֵי הַשֻּׁלְחָנִי לְהַעֲמִיד בּוֹ תְרִיסֵי הַחֲנוּת, וּכְשֶׁמְּסַלֵּק הַתְּרִיסִים נִשְׁאָר שָׁם הַמַּסְמֵר. וּמְטַהֲרִין רַבָּנָן מִשּׁוּם דְּתַשְׁמִישׁוֹ עִם הַקַּרְקַע. וַהֲלָכָה כַּחֲכָמִים:

פרק יב - משנה ו
וּתְלוֹי הַמַּגְרֵדוֹת. בְּבָתֵּי מֶרְחֲצָאוֹת תּוֹלִין מַגְרֵדוֹת שֶׁל מַתָּכוֹת, וְהַנִּכְנָסִים שָׁם כָּל אֶחָד לוֹקֵחַ מִגְרֶדֶת וּמִתְגָּרֵד בָּהּ רַגְלָיו: וְגֹלְמֵי כְלֵי מַתָּכוֹת. כֵּלִים שֶׁצָּרִיךְ לָשׁוּף לְגָרֵד וּלְהָקִישׁ בְּקֻרְנָס, אוֹ שֶׁמְּחֻסָּר אֹגֶן, אוֹ אֹזֶן, אֵינָם טְמֵאִים לְדִבְרֵי חֲכָמִים עַד שֶׁתִּגָּמֵר מְלַאכְתָּן: וְטַבְלָא שֶׁנֶּחְלְקָה לִשְׁנָיִם. טַבְלָא שֶׁל חֶרֶס שֶׁיֶּשׁ בָּהּ לְבִזְבֵּזִים וְנַעֲשֵׂית שְׁתֵּי חֲתִיכוֹת שָׁווֹת, שֶׁאֵין אַחַת מֵהֶן גְּדוֹלָה מֵחֲבֶרְתָּהּ, בְּהָא פְלִיגֵי רַבָּן גַּמְלִיאֵל וַחֲכָמִים. וַהֲלָכָה כַּחֲכָמִים:

פרק יב - משנה ז
דִּינָר שֶׁנִּפְסַל. שֶׁפְּסָלַתּוּ מַלְכוּת אוֹ מְדִינָה, אוֹ שֶׁנֶּחְסַר: וְהִתְקִינוֹ. שֶׁנִּקְּבוֹ כְּדֵי שֶׁיִּתְלֶנּוּ בְּצַוַּאר בְּנוֹ אוֹ בְצַוַּאר בִּתּוֹ: סֶלַע. אַרְבָּעָה דִינָרִים: וְהִתְקִינָהּ לִהְיוֹת שׁוֹקֵל בָּהּ. כֶּסֶף אוֹ זָהָב. דְּאִלּוּ דָבָר הַנִּדְבָּק כְּגוֹן בָּשָׂר וָיַיִן, הָא תַנְיָא בְּפֶרֶק הַמּוֹכֵר אֶת הַסְּפִינָה שֶׁאֵין עוֹשִׂין מִשְׁקָלוֹת שֶׁל מַתֶּכֶת לִשְׁקֹל בָּהֶן דָּבָר הַנִּדְבָּק: עַד שְׁנֵי דִינָרִים. שֶׁהוּא שֶׁקֶל אֶחָד. וְהוּא חֲצִי סֶלַע: פָּחוֹת מִכָּאן יָקוֹץ. שֶׁלֹּא יְרַמּוּ בָהּ בְּנֵי אָדָם שֶׁיַּחְשְׁבוּ שֶׁהוּא שֶׁקֶל. וְכֵן אִם הוּא יוֹתֵר מִשֶּׁקֶל נַמִּי יָקֹץ שֶׁלֹּא יִטְעוּ בוֹ וְיִקְּחוּהוּ בְסֶלַע:

פרק יב - משנה ח
הָאוֹלָר. כְּעֵין מִסְפָּרַיִם שֶׁל מַתֶּכֶת שֶׁמְּתַקְּנִים בּוֹ הַקֻּלְמוֹסִין: וְהַקֻּלְמוֹס. שֶׁל נְחֹשֶׁת וּבַרְזֶל אוֹ שֶׁל כֶּסֶף: מְטוּטֶלֶת. בַּרְזֶל וְעוֹפֶרֶת שֶׁתָּלוּי בְּחֶבֶל שֶׁתּוֹפֵס הַבַּנַּאי בְּיָדוֹ לִרְאוֹת שֶׁלֹּא יָבֹא הַכֹּתֶל עָקוּם: וְהַמִּשְׁקָלוֹת. לִיטְרָא וַחֲצִי לִיטְרָא: הַכִּירוֹן. כְּלִי בַרְזֶל שֶׁבּוֹ תּוֹחֲבִים הַזֵּיתִים בַּמַּעֲצָר שֶׁעוֹצְרִים אוֹתָן בּוֹ : וְהַכַּן וְהַכַּנָּא. הָאַמָּה שֶׁמְּסַרְגְּלִים בָּהּ הַסְּפָרִים, וְהַלּוּחַ שֶׁתַּחְתֶּיהָ . כֵּן פֵּרֵשׁ הֶעָרוּךְ: גֹּלְמֵי כְלֵי עֵץ. שֶׁלֹּא נִשְׁלְמָה מְלַאכְתָּן, כְּגוֹן שֶׁחֲסֵרִים אֹגֶן, אוֹ אֹזֶן, אוֹ הַבָּסִיס שֶׁלָּהֶן שֶׁהֵן יוֹשְׁבִים עָלָיו, אוֹ שֶׁצָּרִיךְ לְהַחְלִיקָן וּלְהַשְׁווֹתָן, נִקְרָאִים גֳּלָמִים: חוּץ מִשֶּׁל אֶשְׁכְּרוֹעַ. גָּרְסִינַן. וְהוּא מִין אֶרֶז. תְּאַשּׁוּר, מְתַרְגְּמִינַן אֶשְׁכְּרוֹעַ. וְגָלְמֵי כֵלִים שֶׁנַּעֲשׂוּ מֵעֵץ זֶה, טְהוֹרִין, לְפִי שֶׁקְּלִפָּתוֹ עָבָה וְקָשָׁה, וְאֵין חֲשׁוּבִין כֵּלִים עַד שֶׁתּוּסַר קְלִפָּתָם וְתִגָּמֵר מְלַאכְתָּם. וְגִרְסַת הַתּוֹסֶפְתָּא, חוּץ מִשֶּׁל פִּירָשָׁע, מִפְּנֵי שֶׁהֵן מְחֻסָּרִים שְׁלִיקָה. וּפִירָשָׁע הוּא עֲצֵי גֹּפֶר. וְכֵלִים הַנַּעֲשִׂים מִמֶּנּוּ אֵין מְלַאכְתָּן נִגְמֶרֶת אֶלָּא בִשְׁלִיקָה, שֶׁמַּכְנִיסִים אוֹתָן בָּאֵשׁ וּמוֹצִיאִין שֶׁלֶק, וְשׁוּב אֵין יְרֵאִים שֶׁמָּא יִסָּדְקוּ: גְּרוֹפִית שֶׁל זַיִת. עָנָף שֶׁל זַיִת קָרוּי גְּרוֹפִית. כְּמוֹ שֶׁל תְּאֵנָה דְנִקְרָא יִחוּר. וְאִם עָשׂוּ כְלִי מִן הַגְּרוֹפִית, גֹּלֶם הוּא וְאֵינוֹ מְקַבֵּל טֻמְאָה: עַד שֶׁיִּשְׁלֹק. שֶׁיַּכְנִיסוּ אוֹתָן בָּאֵשׁ עַד שֶׁיּוֹצִיאוּ הַשֶּׁלֶק שֶׁלָּהֶן. וְאֵין הֲלָכָה כְּרַבִּי יְהוּדָה:
