פרק ג

שִׁעוּר כְּלִי חֶרֶס לִטַּהֵר. דְּתוּ לֹא מְקַבֵּל טֻמְאָה, וְאִם הָיָה טָמֵא טָהוֹר כְּאִלּוּ נִשְׁבַּר כֻּלּוֹ, דְּתוּ לֹא חֲזִי לְמִלְּתֵיהּ וְלֹא חֲשׁוּב תּוֹךְ דִּידֵיהּ. אֲפִלּוּ הָיְתָה חָבִית זוֹ מְיֻחֶדֶת לִגְרוֹגָרוֹת וְלֶאֱגוֹזִים, וְנִקְּבָה בְּמוֹצִיא זַיִת, טְהוֹרָה, כָּל זְמַן שֶׁלֹּא חָזַר וְיִחֲדָהּ אַחַר כָּךְ לְקַבֵּל בָּהּ רִמּוֹנִים וְכַיּוֹצֵא בָּהֶן, אֲבָל חָזַר וְיִחֲדָהּ לְקַבֵּל בָּהּ רִמּוֹנִים, טְמֵאָה עַד שֶׁתִּנָּקֵב בְּמוֹצִיא רִמּוֹן, וּמִשֶּׁנִּקְּבָה בְּמוֹצִיא רִמּוֹן שׁוּב אֵינָהּ מְקַבֶּלֶת טֻמְאָה: שִׁעוּרוֹ בְמַשְׁקִין. בְּנֶקֶב שֶׁכְּשֶׁמְּשִׂימִין הַכְּלִי עַל גַּבֵּי מַשְׁקִין נִכְנָסִין הַמַּשְׁקִין דֶּרֶךְ אוֹתוֹ נֶקֶב לְתוֹךְ הַכְּלִי, וְהוּא נֶקֶב גָּדוֹל מִמּוֹצִיא מַשְׁקֶה: לְכָךְ וּלְכָךְ. כְּגוֹן קְדֵרָה שֶׁמְּבַשְּׁלִים בָּהּ בָּשָׂר, אַף עַל פִּי שֶׁאֵין מַנִּיחִין בָּהּ אֹכֶל בְּלֹא מַשְׁקֶה, וּלְמַשְׁקֶה לֹא חַזְיָא שֶׁהֲרֵי נִקְּבָה בְּכוֹנֵס מַשְׁקֶה, אֲפִלּוּ הָכִי אָזְלִינַן בָּהּ לְחֻמְרָא, וְאֵינָהּ טְהוֹרָה עַד שֶׁתִּנָּקֵב בְּמוֹצִיא זֵיתִים:

חָבִית שִׁעוּרָהּ בִּגְרוֹגָרוֹת. רְגִילִים הָיוּ לְהָנִיחַ גְּרוֹגָרוֹת בֶּחָבִית: רַבִּי מֵאִיר אוֹמֵר בְּזֵיתִים. כִּסְתָם מַתְנִיתִין דִּלְעֵיל. וְהָוֵי סְתָם וְאַחַר כָּךְ מַחֲלֹקֶת וְאֵין הֲלָכָה כִּסְתָם. וַהֲלָכָה כְּרַבִּי יְהוּדָה : הַלְּפָס וְהַקְּדֵרָה שִׁעוּרָן בְּזֵיתִים. לְהַחְמִיר. כִּדְפָרֵישְׁנָא לְעֵיל: הַפַּךְ וְהַטְּפִי. כְּלֵי שֶׁמֶן הֵן. אֶלָּא שֶׁהַטְּפִי פִּיו צַר מְאֹד, לְכָךְ נִקְרָא טְפִי שֶׁאֵין יוֹרֵד מִמֶּנּוּ אֶלָּא טִפִּין טִפִּין. וּטְפִי דִּתְנַן בְּפִרְקִין דִּלְעֵיל שֶׁהוּא טָהוֹר, הַיְנוּ כְּשֶׁהִתְקִינוֹ לְכַסּוֹת בּוֹ סַל שֶׁל עֲנָבִים כִּדְפָרֵישְׁנָא לְעֵיל: בְּשָׁמֶן. הַיְנוּ כּוֹנֵס מַשְׁקֶה שֶׁל שֶׁמֶן. וְאֵין שִׁעוּרוֹ שָׁוֶה לְמַיִם, דְּשֶׁל שֶׁמֶן דַּק, שֶׁהֲרֵי כְלִי שֶׁמַּחֲזִיק מַיִם אוֹ יַיִן אֵינוֹ מַחֲזִיק שֶׁמֶן: הַצַּרְצוּר. כְּלִי עָשׂוּי לִשְׁתּוֹת בּוֹ מַיִם וְעַל פִּיו שְׂבָכָה כְּמַעֲשֵׂה רֶשֶׁת שֶׁלֹּא יִכָּנֵס לְתוֹכוֹ דָבָר רַע וְיִטָּנְפוּ הַמַּשְׁקִין: שְׁלָשְׁתָּן. הַפַּךְ וְהַטְּפִי וְהַצַּרְצוּר, שִׁעוּרָן בְּנֶקֶב מוֹצִיא זֵרְעוֹנִים, כְּגוֹן הָאֲפוּנִים . דְּסָבַר רַבִּי שִׁמְעוֹן דִּלְזֵרְעוֹנִים נַמִּי עֲבִידֵי, וַהֲווּ כְמוֹ הֶעָשׂוּי לְכָךְ וּלְכָךְ דִּלְעֵיל דִּמְטִילִין אוֹתוֹ לְחֻמְרוֹ. וְאֵין הֲלָכָה כְּרַבִּי שִׁמְעוֹן: נֵר שִׁעוּרוֹ בְשָׁמֶן. בְּכוֹנֵס מַשְׁקֶה שֶׁל שֶׁמֶן. דִּסְתָם נֵר לְשֶׁמֶן עֲבִיד: רַבִּי שִׁמְעוֹן אוֹמֵר בִּפְרוּטָה קְטַנָּה. דְּסָבַר עֲבִיד נַמִּי לְאַנּוֹחֵי בֵיהּ פְּרוּטוֹת, וּמְטִילִין אוֹתוֹ לְחֻמְרוֹ. וְאֵין הֲלָכָה כְּרַבִּי שִׁמְעוֹן: נֵר שֶׁנִּטַּל פִּיו. בָּטַל שֵׁם נֵר מִמֶּנּוּ: וְשֶׁל אֲדָמָה. אֵינוֹ מְקַבֵּל טֻמְאָה כָּל זְמַן שֶׁלֹּא נִצְרַף בְּכִבְשָׁן . וְאַף עַל פִּי שֶׁהֻסַּק פִּי הַנֵּר בִּפְתִילָה הַדּוֹלֶקֶת, לָא חֲשִׁיב בְּכָךְ כְּאִלּוּ נִצְרַף בְּכִבְשָׁן, וְטָהוֹר:

חָבִית שֶׁנִּקְּבָה. בְּמוֹצִיא זַיִת , וְטָהֲרָה, כְּשֶׁחָזַר וַעֲשָׂאָהּ בְּזֶפֶת, מְקַבֶּלֶת טֻמְאָה מִכָּאן וּלְהַבָּא: חָזְרָה וְנִשְׁבְּרָה. לִשְׁבָרִים הַרְבֵּה. וְאוֹתוֹ הַשֶּׁבֶר שֶׁיֵּשׁ בּוֹ הַזֶּפֶת מַחֲזִיק רְבִיעִית וְיוֹשֵׁב שֶׁלֹּא מְסֻמָּךְ, טָמֵא. וּבְחָבִית הַמַּחְזֶקֶת מִלֹּג וְעַד סְאָה עָסְקִינַן, כְּדִתְנַן בְּפִרְקִין דִּלְעֵיל מִלֹּג וְעַד סְאָה בִּרְבִיעִית. וְאַף עַל פִּי שֶׁאֵין סְתִימָה מוֹעֶלֶת לְנֶקֶב הַבָּא לְאַחַר שְׁבִירָה, שָׁאנֵי הָךְ סְתִימָה, שֶׁנַּעֲשֵׂית קֹדֶם שֶׁבָּטַל שֵׁם כְּלִי מִמֶּנָּה. דְּחָבִית שֶׁנִּקְּבָה שֵׁם חָבִית עָלֶיהָ אַף עַל פִּי שֶׁטָּהֲרָה, אֲבָל חֶרֶס הַפּוֹרֵשׁ מִן הֶחָבִית בָּטַל שֵׁם כְּלִי מֵעָלָיו אַף עַל פִּי שֶׁלֹּא נִקַּב, לְפִיכָךְ אֵין מוֹעֶלֶת סְתִימָה בַנֶּקֶב שֶׁלּוֹ:

חָבִית שֶׁנִּתְרוֹעֲעָה. כְּמוֹ כֹתֶל רָעוּעַ. שֶׁלֹּא נָפְלוּ חֲרָסֶיהָ אֲבָל הִיא רְעוּעָה כָל כָּךְ שֶׁאִם יְטַלְטְלוּהָ וַחֲצִי קַב גְּרוֹגָרוֹת בְּתוֹכָהּ תִּשָּׁבֵר לְגַמְרֵי. וּטְהוֹרָה הִיא כִּדְאָמְרִינַן לְקַמָּן בְּאִידָךְ פִּרְקִין : וּטְפָלָהּ. שֶׁטַּח פָּנֶיהָ: בִּגְלָלִים. כְּדֵי שֶׁלֹּא יִפְּלוּ הַחֲרָסִים: נִשְׁבְּרָה. שֶׁנָּפְלוּ הַחֲרָסִים וְדִבֵּק הַשְּׁבָרִים בְּקוֹלָן שֶׁל סוֹפְרִים, אוֹ שֶׁהֵבִיא חֲרָסִים מִמָּקוֹם אַחֵר וְדִבְּקָן וְאַחַר כָּךְ טְפָלָן בִּגְלָלִים, טְהוֹרָה. וְהוּא הַדִּין טְפָלָן בְּטִיט, כָּל זְמַן שֶׁלֹּא חָזַר וּצְרָפָהּ בַּכִּבְשָׁן : חֶרֶס מַחֲזִיק רְבִיעִית. וּכְגוֹן שֶׁהַחֶרֶס בָּאָה מֵחָבִית הַמַּחְזֶקֶת מִלֹּג וְעַד סְאָה, דְּשִׁעוּרָהּ בִּרְבִיעִית כִּדְפָרֵישְׁנָא לְעֵיל: כֻּלָּן מִטַּמְּאִין בְּמַגָּע. דְּכֻלָּן נַעֲשִׂים יָד לְאוֹתוֹ חֶרֶס, וּמִטַּמְּאִין מִשּׁוּם יָד. וּכְנֶגְדּוֹ בִּלְבַד מִטַּמֵּא בַּאֲוִיר, דְּאֵין אֲוִיר לְיַד כְּלֵי חֶרֶס:

כְּלִי חֶרֶס הַבָּרִיא. שֶׁאֵינוֹ לֹא רָעוּעַ וְלֹא שָׁבוּר: מְטַמְּאִין. אִם נִטְמָא הַכְּלִי מֵאֲוִירוֹ, טְמֵאִים אֳכָלִים וּמַשְׁקִים שֶׁנָּגְעוּ בַטְּפֵלָה, מִפְּנֵי שֶׁהִיא חֲשׁוּבָה כַכְּלִי עַצְמוֹ: הַטּוֹפֵל אֶת הַבָּרִיא טָהוֹר. דְּדַוְקָא הַכְּלִי רָעוּעַ שֶׁצָּרִיךְ לְאוֹתָהּ טְפֵלָה הוּא דַּחֲשִׁיבָה הַטְּפֵלָה כִכְלִי וְנִטְמְאוּ הָאֳכָלִים הַנּוֹגְעִים בָּהּ כְּשֶׁהַכְּלִי טָמֵא, אֲבָל טְפֵלַת הַבָּרִיא אֵינָהּ חֲשׁוּבָה כַכְּלִי. וַהֲלָכָה כַּחֲכָמִים: וְכֵן בְּחִדּוּק קֵרוּיָה. כְּשֵׁם שֶׁנֶּחְלְקוּ רַבִּי מֵאִיר וְרַבִּי שִׁמְעוֹן עִם חֲכָמִים בִּטְפֵלַת הַבָּרִיא כָּךְ נֶחְלְקוּ בְחִדּוּק קֵרוּיָה, וְהִיא דְּלַעַת יְבֵשָׁה וַחֲלוּלָה שֶׁשּׁוֹאֲבִין בָּהּ מַיִם, וּרְגִילִים לְהַדֵּק סְבִיבוֹתֶיהָ בְּעִגּוּל שֶׁל עֵץ אוֹ שֶׁל בַּרְזֶל כְּדֵי שֶׁלֹּא תִּשָּׁבֵר אִם תִּגֹּף בָּאֶבֶן. דְּרַבִּי מֵאִיר וְרַבִּי שִׁמְעוֹן דִּמְטַמְּאִין בִּטְפֵלַת הַבָּרִיא מְטַמְּאִין אַף בְּזוֹ, וַאֲפִלּוּ הָיְתָה הַקֵּרוּיָה בְרִיאָה. וַחֲכָמִים אוֹמְרִים, אִם הָיְתָה הַקֵּרוּיָה בְרִיאָה וְאֵינָהּ צְרִיכָה לְעֹגֶל זֶה לָא חֲשִׁיב הָעֹגֶל כְּגוּפָהּ שֶׁל קֵרוּיָה, וְאִם נִטְמֵאת הַקֵּרוּיָה לֹא נִטְמָא הָעֹגֶל:

יַבְּלִית. מִין עֵשֶׂב שֶׁשְּׁמוֹ יַבְּלָא. וּרְגִילִין לָטוּחַ בּוֹ הַכַּדּוֹת הַגְּדוֹלוֹת שֶׁלֹּא יִפְלְטוּ הַיַּיִן שֶׁמְּזִיעִין: פִּטָּסִין. כַּדּוֹת גְּדוֹלוֹת. כְּמוֹ הַלָּלוּ בַּעֲלֵי פִטָּסִין דִּבְמַסֶּכֶת בֵּיצָה :

מֵחַם. כְּלִי שֶׁמְּחַמְּמִים בּוֹ הַמַּיִם: בְּחֹמֶר. טִיט עָב: חַרְסִית. דַּק כְּחֶרֶס . וְיֵשׁ מְפָרְשִׁים, כְּתִישַׁת חֲרָסִים: הַנּוֹגֵעַ בַּחֹמֶר טָמֵא. לְפִי שֶׁהוּא נִדְבָּק וּמִתְחַבֵּר בַּמֵּחַם וַחֲשׁוּב כְּעַצְמוֹ שֶׁל כְּלִי, הִלְכָּךְ אִם נִטְמָא הַמֵּחַם נִטְמָא הַחֹמֶר שֶׁסְּבִיבוֹ כְּמוֹתוֹ, וְהָאֳכָלִים הַנּוֹגְעִים בַּחֹמֶר טְמֵאִים: וּבַחַרְסִית טָהוֹר. דְּחַרְסִית אֵינוֹ נִדְבָּק וּמִפְרִיךְ וְנָפִיל, וְלָא חֲשִׁיב כְּעַצְמוֹ שֶׁל כְּלִי: קוּמְקוּם. יוֹתֵר גָּדוֹל מִמֵּחַם . וּבְלַעַ"ז קוֹרְאִין לוֹ קומקומ"ו: שֶׁאֵינוֹ יָכוֹל לְקַבֵּל אֶת הַחַמִּין. שֶׁהַזֶּפֶת נִתָּךְ וְנָמֵס בְּמַיִם חַמִּין. וְאַף עַל פִּי שֶׁיָּכוֹל לְקַבֵּל אֶת הַצּוֹנֵן אֵינוֹ חֲשׁוּב כְּלִי וְטָהוֹר: וְכֵן הָיָה אוֹמֵר בִּכְלֵי זֶפֶת. כֵּלִים הָעֲשׂוּיִין מִן הַזֶּפֶת עַצְמוֹ, שֶׁהֵן טְהוֹרִין, הוֹאִיל וְאֵין יְכוֹלִין לְקַבֵּל אֶת הַחַמִּין כַּצּוֹנֵן. וְרַבּוֹתַי פֵּרְשׁוּ בִּכְלֵי זֶפֶת, בְּכֵלִים הַזְּפוּתִים, וּמְפָרֵשׁ וְאָזֵיל כֵּיצַד, כְּלֵי נְחֹשֶׁת שֶׁזְּפָתָן טְהוֹרִים. אִם הָיוּ הַכֵּלִים טְמֵאִים וְנָגַע אָדָם אוֹ אֳכָלִין וּמַשְׁקִין בַּזֶּפֶת, טְהוֹרִים, דְּאֵין הַזֶּפֶת חִבּוּר וְאֵינוֹ נֶחְשָׁב כְּעַצְמוֹ שֶׁל כְּלִי : וְאִם לְיַיִן. וְאִם זְפָתָן מִתְּחִלָּה לָשׂוּם בְּתוֹכָן יַיִן וְאֵין דַּעְתּוֹ לְהִשְׁתַּמֵּשׁ בָּהֶן בְּחַמִּין כְּלָל : טְמֵאִין. דְּאָז הָוֵי הַכְּלִי צָרִיךְ לַזֶּפֶת וְהָוֵי זֶפֶת חִבּוּר:

יוֹתֵר מִצָּרְכָּהּ. כְּנֶגֶד הַנֶּקֶב הוּא צָרְכָּהּ, וּמַה שֶּׁהוּא שֶׁלֹּא כְּנֶגֶד הַנֶּקֶב הָוֵי יוֹתֵר מִצָּרְכָּהּ, וְלָא חֲשִׁיב חִבּוּר, כֵּיוָן דְּלֹא צָרִיךְ לֶחָבִית. וְרַמְבַּ"ם פֵּרֵשׁ, יוֹתֵר מִצָּרְכָּהּ כְּגוֹן שֶׁעָשָׂה הַסְּתִימָה עָבָה בְיוֹתֵר עַד שֶׁיּוּכַל לְהִנָּטֵל מִמֶּנָּה וְתִשָּׁאֵר סְתוּמָה, וַאֲפִלּוּ הַכֹּל כְּנֶגֶד הַנֶּקֶב: זֶפֶת שֶׁנָּטְפָה עַל הֶחָבִית. שֶׁאֵינָהּ נְקוּבָה: הַנּוֹגֵעַ בָּהּ. בְּטִפַּת הַזֶּפֶת: טָהוֹר. אַף עַל פִּי שֶׁהֶחָבִית טְמֵאָה. שֶׁאֵין הַזֶּפֶת חִבּוּר לֶחָבִית כֵּיוָן שֶׁאֵין הֶחָבִית צְרִיכָה לָהּ: שֶׁפְּקָקוֹ בְזֶפֶת. שֶׁסָּתַם פִּי הַנֶּקֶב שֶׁל הַמַּשְׁפֵּךְ הֶעָשׂוּי לְהַכְנִיס מִמֶּנּוּ יַיִן לֶחָבִית: רַבִּי אֶלְעָזָר בֶּן עֲזַרְיָה מְטַמֵּא. דַּחֲשִׁיב לֵיהּ סְתִימָה, בֵּין בְּשֶׁל עֵץ בֵּין בְּשֶׁל חֶרֶס : וְרַבִּי עֲקִיבָא מְטַמֵּא בְשֶׁל עֵץ. מִפְּנֵי שֶׁהַזֶּפֶת הוּא מִינוֹ, שֶׁהַזֶּפֶת יוֹצֵא מִן הָעֵץ: וּמְטַהֵר בְּשֶׁל חָרֶס. שֶׁאֵינוֹ מִינוֹ וְלָא חֲשִׁיב חִבּוּר. וְלֹא דָּמֵי סְתִימַת פִּי הַמַּשְׁפֵּךְ לִסְתִימַת נֶקֶב הֶחָבִית דַּחֲשִׁיב חִבּוּר, לְפִי שֶׁהַנֶּקֶב שֶׁבְּפִי הַמַּשְׁפֵּךְ שֶׁעֲשָׂאוֹ הַיּוֹצֵר מִתְּחִלָּה חָלָק הוּא וְנִקְלָף הַזֶּפֶת מִמֶּנּוּ וְאֵינוֹ מִתְקַיֵּם, אֲבָל חָבִית שֶׁנִּקְּבָה לְאַחַר שֶׁהֻסְּקָה בַכִּבְשָׁן אִי אֶפְשָׁר לַנֶּקֶב לִהְיוֹת חָלָק וְשָׁוֶה וְאֵין הַזֶּפֶת נִקְלָף מִמֶּנּוּ, אִי נַמִּי, נֶקֶב שֶׁבְּפִי הַמַּשְׁפֵּךְ סוֹפוֹ לְהָסִיר סְתִימָתוֹ כְשֶׁיִּרְצֶה לִשְׁפֹּךְ יַיִן בֶּחָבִית, הִלְכָּךְ לָא חֲשִׁיבָא סְתִימָה אֶלָּא בְמַשְׁפֵּךְ שֶׁל עֵץ לְפִי שֶׁהוּא מִינוֹ, אֲבָל חָבִית שֶׁנִּקְּבָהּ אֵין סוֹפוֹ לְהָסִיר הַסְּתִימָה: רַבִּי יוֹסֵי מְטַהֵר בִּשְׁנֵיהֶם. דִּסְתִימַת פִּי הַמַּשְׁפֵּךְ לָא חֲשִׁיבָא סְתִימָה, וַהֲרֵי הוּא כְאִלּוּ לֹא נִסְתַּם וְאֵין לוֹ בֵּית קִבּוּל. וַהֲלָכָה כְּרַבִּי עֲקִיבָא:
