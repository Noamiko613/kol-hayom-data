פרק יד

פרק יד - משנה א
כְּלֵי מַתָּכוֹת. שֶׁנִּתְרוֹעֲעוּ וְנִשְׁבְּרוּ: כַּמָּה הוּא שִׁעוּרָן. כַּמָּה יִהְיֶה שִׁעוּר הַשֶּׁבֶר הַנִּשְׁאָר מֵהֶן וְיִקָּרֵא עוֹשֶׂה מֵעֵין מְלַאכְתּוֹ וְיִהְיוּ חֲשׁוּבִים כֵּלִים לִטַּמֵּא : מֵחַם. שֶׁמְּחַמְּמִין בּוֹ מַיִם. וְהוּא גָּדוֹל מִקֻּמְקוּם : כְּדֵי לְקַבֵּל סְלָעִים. שֶׁכָּל סֶלַע אַרְבָּעָה דִינָרִין: הַלֶּבֶס. אִית דְּגָרְסֵי הַלֶּפֶס. וְהוּא כְלִי נְחֹשֶׁת גָּדוֹל שֶׁמַּרְתִּיחִים בְּתוֹכוֹ מַיִם וְנוֹתְנִים בּוֹ קִיתוֹנוֹת קְטַנִּים לְהָדִיחָן בְּתוֹכוֹ: כֻּלָּן בִּפְרוּטוֹת. כֻּלָּן מְשַׁעֲרִים אִם יֵשׁ בַּשֶּׁבֶר שֶׁלָּהֶן כְּדֵי לְקַבֵּל פְּרוּטוֹת. וְאֵין הֲלָכָה כְּרַבִּי אֱלִיעֶזֶר: הַמְחֻסָּר הֲגָפָה טָמֵא. רַבִּי עֲקִיבָא מַיְרֵי בִּתְחִלַּת עֲשִׂיָּתָן שֶׁל כֵּלִים, וְקָאָמַר, דְּכֵלִים הַמְחֻסָּרִים הַכִּסּוּי שֶׁלָּהֶן, טְמֵאִין, דְּמִקְּרוּ נִגְמְרָה מְלַאכְתָּן. הֲגָפָה, מִלְּשׁוֹן יָגִיפוּ הַדְּלָתוֹת (נְחֶמְיָה ז). וְהַרְבֵּה יֵשׁ בַּמִּשְׁנָה, כְּדֵי שֶׁיִּפְתַּח וְיִגֹּף וְתִגֹּב. אַף עַל פִּי שֶׁגָּפָן לֹא קָנָה דִּבְמַסֶּכֶת מַעְשֵׂר שֵׁנִי , פֵּרוּשׁ, שֶׁכִּסָּה אוֹתָן בִּמְגוּפָה: וּמְחֻסָּר לְטִישָׁה טָהוֹר. שֶׁהוּא חֲשׁוּב גָּלְמֵי כְלֵי מַתָּכוֹת שֶׁאֵינָן מְקַבְּלִין טֻמְאָה. לְטִישָׁה, לְשׁוֹן לֹטֵשׁ כָּל חֹרֵשׁ נְחֹשֶׁת וּבַרְזֶל (בְּרֵאשִׁית ד):

פרק יד - משנה ב
חַזְיָנָא. נוֹהֲגִים לַעֲשׂוֹת בְּרֹאשׁ הַמַּקֵּל כְּמִין תַּפּוּחַ שֶׁל בַּרְזֶל וְהוּא הַנִּקְרָא חַזְיָנָא. וְכָל עַבְדֵי מֶלֶךְ מִצְרַיִם הַיּוֹם נוֹשְׂאִים בְּיָדָם מַקְלוֹת הַלָּלוּ, וְקוֹרִין לָהֶם אלדבו"ס. וּפְעָמִים שֶׁנּוֹעֲצִים בָּהֶם מַסְמְרִים כְּדֵי שֶׁיַּכּוּ בָּהֶם מַכָּה רַבָּה וְקָשָׁה: שְׁלֹשָׁה סְדָרִים. שֶׁל מַסְמְרִים, וְאָז חֲשׁוּב כְּלִי. וְאֵין הֲלָכָה כְּרַבִּי שִׁמְעוֹן: עֲשָׂאָן לְנוֹי. כְּגוֹן שֶׁעֲשָׂאוֹ בְמַסְמְרִים דַּקִּים מְצֻיָּרִים כְּדֵי לְיַפּוֹת אֶת הַמַּקֵּל: טָהוֹר. דַּהֲוֵי מַתֶּכֶת הַמְשַׁמֵּשׁ אֶת הָעֵץ, דִּתְנַן לְעֵיל שֶׁהוּא טָהוֹר. אֲבָל כְּשֶׁעֲשָׂאוֹ לְהַכּוֹת וְלַחְבֹּט, הֲרֵי עֵץ מְשַׁמֵּשׁ אֶת הַמַּתֶּכֶת וְטָמֵא: מְנִיקָת. כְּמִין שְׁפוֹפֶרֶת שֶׁל מַתֶּכֶת, וְהִכְנִיס בָּהּ אֶת הַמַּקֵּל: וְכֵן בַּדֶּלֶת. תַּחַת הַדֶּלֶת כְּדֵי שֶׁלֹּא תֹאכְלֶנָּה הָאָרֶץ: הָיְתָה כְלִי. הָיְתָה שְׁפוֹפֶרֶת זוֹ כְּלִי בִּפְנֵי עַצְמָהּ וְחִבְּרָהּ לְמַקֵּל וְאַחַר שֶׁחִבְּרָהּ רָאוּי לְהִשְׁתַּמֵּשׁ בָּהּ בִּכְלִי: טְמֵאָה. מְקַבֶּלֶת טֻמְאָה, וּטְמֵאָה כְּמוֹ שֶׁהָיְתָה: מֵאֵימָתַי הִיא טָהֳרָתָהּ. מְנִיקָת שֶׁנִּטְמְאָה מֵאֵימָתַי תִּטְהַר וְתֵצֵא מִידֵי טֻמְאָתָהּ: מִשֶּׁיְּחַבֵּל. מִשֶּׁיַּשְׁחִיתֶנָּה וְיִפְחֹת צוּרָתָהּ: מִשֶּׁיְּחַבֵּר. מִשֶּׁיְּחַבְּרֶנָּה בַדֶּלֶת אוֹ בַמַּקֵּל וְיִקְבְּעֶנָּה בְּמַסְמְרִים שֶׁלֹּא תְּשַׁמֵּשׁ עוֹד בִּכְלִי, כְּבָר בָּטַל שֵׁם כְּלִי מִמֶּנָּה וְיָצְאָה מִידֵי טֻמְאָתָהּ:

פרק יד - משנה ג
הַקַּנְטָר שֶׁל בַּנַּאי. פֵּרְשׁוּ בּוֹ שֶׁהוּא שֵׁבֶט וּמַטֶּה שֶׁל בַּרְזֶל שֶׁהַבַּנָּאִים חוֹתְכִים בּוֹ אֶת הַכֹּתֶל : הַדָּקוֹר. יָתֵד. כְּמוֹ יַחְפֹּר בַּדֶּקֶר וִיכַסֶּה דִּתְנַן בְּרֵישׁ פֶּרֶק קַמָּא דְבֵיצָה: חָרָשׁ. נַגָּר: יִתְדוֹת אֹהָלִים. נוֹעֲצִים יְתֵדוֹת שֶׁל בַּרְזֶל בָּאָרֶץ וּמוֹתְחִים אֹהָלִים וְקוֹשְׁרִים מֵיתָרִים בַּיְתֵדוֹת. וְאֵין זֶה תַשְׁמִישׁוֹ עַל יְדֵי קַרְקַע, שֶׁהֲרֵי לְדָבָר הַמִּטַּלְטֵל הֵן מְשַׁמְּשִׁים: וְיִתְדוֹת הַמָּשׁוֹחוֹת. מוֹדְדֵי הַקַּרְקַע לַחֲלֻקַּת אַחִים, נוֹעֲצִים יְתֵדוֹת בַּקַּרְקַע וְקוֹשְׁרִים בּוֹ חֶבֶל וּמוֹדְדִים: שַׁלְשֶׁלֶת שֶׁל מָשׁוֹחוֹת טְמֵאָה. לְפִי שֶׁבִּמְדִידַת הַחֶבֶל יֵשׁ עָוֶל. פְּעָמִים שֶׁמּוֹתֵחַ לָזֶה יוֹתֵר מִזֶּה, לְפִיכָךְ רְגִילִים לַעֲשׂוֹת שַׁלְשֶׁלֶת לִמְדֹּד בָּהּ: עֲשׂוּיָה לָעֵצִים. לִקְשֹׁר בָּהּ חֲבִילָה שֶׁל עֵצִים, אוֹ לִמְדֹּד בָּהּ עֵצִים, שֶׁיֶּשׁ מְקוֹמוֹת שֶׁמּוֹכְרִים עֵצִים בְּמִדָּה מְלֹא חֶבֶל. וְהַשַּׁלְשֶׁלֶת הַמְיֻחֶדֶת לְכָךְ טְהוֹרָה, דְּהַוְיָא כְמַתֶּכֶת הַמְשַׁמֶּשֶׁת אֶת הָעֵץ: אַרְבָּעָה טְפָחִים. וְיוֹתֵר מִכֵּן לָא חֲשִׁיב יָד: חֲמוֹר שֶׁל נַפָּחִים. הָעֵץ שֶׁהַמַּפּוּחַ שֶׁל נַפָּחִים רוֹכֵב עָלָיו נִקְרָא חֲמוֹר. כָּךְ פֵּרְשׁוּהָ לִי רַבּוֹתַי. וּבְמָקוֹם אַחֵר מָצָאתִי שֶׁהוּא כְלִי בַרְזֶל כְּצוּרַת חֲמוֹר, וְהָאֻמָּן רוֹכֵב עַל הַקָּצֶה הָאֶחָד וּמַכֶּה עַל הַקָּצֶה הַשֵּׁנִי, וְעוֹשִׂים כָּךְ כְּשֶׁרוֹצִים לְהַכּוֹת עַל הַכֵּלִים הַגְּדוֹלִים שֶׁשְּׂפָתָן גְּבוֹהָה וּגְדוֹלָה מְאֹד: מְגֵרָה. הַמַּסָּר שֶׁנּוֹסְרִים בָּהּ הַלּוּחוֹת, שֶׁנִּטְמֵאת, וְעָשָׂה שִׁנֶּיהָ בְּתוֹךְ הַחוֹר שֶׁל דֶּלֶת, לֹא יָצְאָה מִידֵי טֻמְאָה, וַהֲרֵי הִיא טְמֵאָה כְּמוֹ שֶׁהָיְתָה, עַד שֶׁיִּקְבְּעֶנָּה בַדֶּלֶת בְּמַסְמֵר. וְהָכִי מְפֹרָשׁ בַּתּוֹסֶפְתָּא: עֲשָׂאָהּ מִלְּמַטָּה לְמַעְלָה. שֶׁהֲפָכָהּ וּנְתָנָהּ הֲפוּכָה בְחוֹר הַדֶּלֶת: טְהוֹרָה. אַף עַל פִּי שֶׁלֹּא קְבָעָהּ בְּמַסְמֵר: כָּל הַכְּסוּיִן טְהוֹרִין. לְפִי שֶׁאֵין לַכִּסּוּיִין שֵׁם בִּפְנֵי עַצְמָן: חוּץ מִשֶּׁל מֵחָם. וְכָל כַּיּוֹצֵא בוֹ שֶׁיֶּשׁ לוֹ שֵׁם בִּפְנֵי עַצְמוֹ וּמִשְׁתַּמְּשִׁין בּוֹ בִּפְנֵי עַצְמוֹ, כְּמוֹ כִסּוּי טֶנִי שֶׁל רוֹפְאִים דִּתְנַן לְעֵיל פֶּרֶק טַבַּעַת אָדָם דְּטָמֵא לְדִבְרֵי הַכֹּל, לְפִי שֶׁמַּנִּיחִין בוֹ סַמְמָנִים:

פרק יד - משנה ד
הָעֹל שֶׁל מַתֶּכֶת טָמֵא. דִּפְשׁוּטֵי כְלֵי מַתָּכוֹת מְקַבְּלִין טֻמְאָה: קַטְרָב. שְׁנֵי עֵצִים יֵשׁ, מִצַּד זֶה מִן הָעֹל וּמִצַּד זֶה מִן הָעֹל, וְהֵן נְקוּבִים, וּמַכְנִיסִין בְּתוֹךְ אוֹתוֹ נֶקֶב עֵץ וּשְׁמוֹ קַטְרָב, וְקוֹשְׁרִים אוֹתוֹ שֶׁלֹּא יִשָּׁמְטוּ הַבָּקָר. וְרַמְבַּ"ם פֵּרֵשׁ, שֶׁהַקַּטְרָב הוּא הָעֵץ שֶׁנִּמְשָׁךְ עַל שְׁנֵי צְדָדֵי הַבְּהֵמוֹת, וְהָעֹל בְּאֶמְצָעוֹ: וְהַכְּנָפַיִם. הֵם שְׁנֵי קְצָווֹת שֶׁל קַטְרָב. מִלְּשׁוֹן עַל כַּנְפֵי בִגְדֵיהֶם. וְהֵם שֶׁל בַּרְזֶל: הַמְקַבְּלוֹת אֶת הָרְצוּעוֹת. כְּלוֹמַר, אִם נַעֲשׂוּ הַקְּצָווֹת שֶׁל קַטְרָב לְקַבֵּל בָּהֶם אֶת הָרְצוּעוֹת, הֵן טְמֵאִים, שֶׁחֲשׁוּבִין כְּלִי. אֲבָל אִם נַעֲשׂוּ לְנוֹי, טְהוֹרִים, לְפִי שֶׁהִיא מַתֶּכֶת הַמְשַׁמֶּשֶׁת אֶת הָעֵץ: וְהַבַּרְזֶל שֶׁתַּחַת צַוְּארֵי בְהֵמָה. קוֹשְׁרִים עַל צַוְּארֵי בְהֵמָה בַּרְזֶל כְּדֵי שֶׁלֹּא תֵחָנֵק בַּחֶבֶל שֶׁקּוֹשְׁרִים בּוֹ הַקַּטְרָב : הַסּוֹמֵךְ. יֵשׁ בְּאֶמְצַע הָעֹל כְּמִין טַבַּעַת גְּדוֹלָה וּבוֹ נִכְנָס רֹאשׁ הַמַּחְרֵשָׁה וְרֹאשׁ הָעֲגָלָה, וְאוֹתוֹ טַבַּעַת קָרוּי סוֹמֵךְ: מַחְגֵּר. חֶבֶל שֶׁקּוֹשְׁרִים תַּחַת צַוַּאר הַשּׁוֹר . וְאִית דְּגָרְסֵי וְהַמַּסְגֵּר, וְהוּא יָתֵד שֶׁנּוֹתְנִים בִּקְצֵה הָעֹל שֶׁלֹּא תִתְעַוֵּת הָעֲגָלָה מִפְּנֵי אֹרֶךְ הָעֹל: תַּמְחוּיוֹת. כְּמִין קְעָרוֹת קְטַנּוֹת שֶׁל עֹל, וּקְרוּיִין תַּמְחוּיוֹת: עִנְבָּל. הוּא הַבַּרְזֶל הַמְקַשְׁקֵשׁ בְּתוֹךְ הַזּוֹג: צִנּוֹרָא. אֻנְקְלִי שֶׁתּוֹחֲבִים בַּמַּשְּׂאוֹי שֶׁבָּעֲגָלָה שֶׁלֹּא יִפֹּל: וּמַסְמֵר הַמְחַבֵּר אֶת כֻּלָּם. כְּלוֹמַר, כָּל מַסְמֵר שֶׁנַּעֲשָׂה לְחַבֵּר הַפְּרָקִין שֶׁל עֲגָלָה זוֹ בְּזוֹ. לַאֲפוֹקֵי הַמַּסְמְרִים שֶׁאֵינָן עֲשׂוּיִין לְחַבֵּר, אֶלָּא לְנוֹי. שֶׁאֵלּוּ אֵין מְטַמְּאִין:

פרק יד - משנה ה
הָעֹל הַמְצֻפֶּה. בַּרְזֶל , טָהוֹר, לְפִי שֶׁהוּא מַתֶּכֶת שֶׁמְּשַׁמֵּשׁ אֶת הָעֵץ: וּשְׁפוֹפָרוֹת. כְּמִין שְׁפוֹפֶרֶת שֶׁל קָנֶה חָלוּל רְגִילִין לַעֲשׂוֹת מִמַּתָּכוֹת לְהַשְׁמִיעַ קוֹל: וְהָאֲבָר. עוֹפֶרֶת שֶׁנּוֹתְנִין בְּצַד צַוְּארֵי בְּהֵמָה לְנוֹי: וְהַסּוֹבֵב שֶׁל גַּלְגַּל. גַּלְגַּל עֲגָלָה שֶׁל עֵץ עוֹשִׂים סְבִיבוֹתֶיהָ שֶׁל בַּרְזֶל שֶׁלֹּא יִשָּׁבֵר בַּטְּרָשִׁים וּבַסְּלָעִים: טַסִּין. לוּחוֹת דַּקּוֹת. פַּחֵי הַזָּהָב, מְתַרְגְּמִינַן טַסֵּי דְדַהֲבָא: סַנְדְּלֵי בְהֵמָה שֶׁל מַתָּכוֹת טְמֵאִין. לְפִי שֶׁעֲשׂוּיִין לִשְׁתּוֹת בָּהֶן אַנְשֵׁי הַחַיִל בַּמִּלְחָמָה: שַׁעַם. מִין גֶּמִי: הַסַּיִף מֵאֵימָתַי מְקַבֵּל טֻמְאָה. בְּסַיִף שֶׁהֶעֱלָה חֲלֻדָּה מַיְרֵי, שֶׁהוּא טָהוֹר. וּמֵאֵימָתַי מְקַבֵּל טֻמְאָה: מִשֶּׁיְּשׁוּפֶנּוּ. כְּלִי יֵשׁ לְצוֹרְפִין וּלְחָרָשֵׁי בַרְזֶל שֶׁקָּרוּי הַצּוֹרְפִים בִּלְשׁוֹן מִקְרָא, וְתַרְגּוּמוֹ שׁוּפִינָא, וּבוֹ שָׁפִין וּמַחֲלִיקִין אֶת הַכֵּלִים שֶׁהֶעֱלוּ חֲלֻדָּה לְהָסִיר חֲלֻדָּתָן: מִשֶּׁיַּשְׁחִיזֶנּוּ. בַּמַּשְׁחֶזֶת שֶׁלּוֹ לְהָסִיר חֲלֻדָּתוֹ:

פרק יד - משנה ו
כְּסוּי טֶנִי שֶׁל מַתֶּכֶת. שֶׁל בַּעֲלֵי בָתִּים, טָהוֹר, כְּדִבְרֵי חֲכָמִים דִּפְלִיגֵי עֲלֵיהּ דְּרַבָּן גַּמְלִיאֵל לְעֵיל בְּפֶרֶק י"ב . וְאִם מֵרְקוֹ וְלִטְּשׁוֹ וְעָשָׂה מִמֶּנּוּ מַרְאָה: רַבִּי יְהוּדָה מְטַהֵר. דִּסְבִירָא לֵיהּ מַרְאָה לֹא מְשַׁוֵּי לֵיהּ מָנָא:

פרק יד - משנה ז
וַחֲכָמִים מְטַמְּאִין. דְּמַרְאָה מְשַׁוֵּי לֵיהּ מָנָא . וַהֲלָכָה כַּחֲכָמִים: כְּלֵי מַתָּכוֹת מִטַּמְּאִין וּמִטַּהֲרִין שְׁבוּרִים. כְּלֵי מַתָּכוֹת שֶׁנִּטְמְאוּ בְמֵת וְנִשְׁבְּרוּ, וְחָזַר וְעָשָׂה מֵהֶן כֵּלִים בּוֹ בַיּוֹם, שֶׁחָזְרוּ לְטֻמְאָתָן יְשָׁנָה, כֵּיוָן שֶׁהִזָּה עֲלֵיהֶן בְּעוֹדָן שְׁלֵמִים וְנִשְׁבְּרוּ, אֵין צָרִיךְ לְהַמְתִּין לְהַזָּאָה שְׁנִיָּה עַד הַשְּׁבִיעִי, אֶלָּא אֲפִלּוּ בְיוֹם אֶחָד מַזֶּה שְׁתֵּי הַזָּאוֹת, אַחַת קֹדֶם שְׁבִירָה וְאַחַת לְאַחַר שְׁבִירָה, וְהֵן טְהוֹרִין מִיָּד. דְּרַבִּי אֱלִיעֶזֶר כְּרַבָּן שִׁמְעוֹן בֶּן גַּמְלִיאֵל סְבִירָא לֵיהּ, דְּאָמַר לְעֵיל בְּפֶרֶק י"א דְּלֹא אָמְרוּ חָזְרוּ לְטֻמְאָתָן יְשָׁנָה אֶלָּא לְכֵלִים שֶׁנִּטְמְאוּ בְטֻמְאַת נֶפֶשׁ בִּלְבַד, וְטַעַם גְּזֵרָה זוֹ מִשּׁוּם גֶּדֶר מֵי חַטָּאת כִּדְפָרֵישְׁנָא לְעֵיל, הִלְכָּךְ לֹא חָשׁוּ אֶלָּא שֶׁיְּהֵא כָּאן שְׁתֵּי הַזָּאוֹת. וּמִיהוּ בְעוֹדָן שְׁבוּרִים לֹא מַהֲנֵי הַזָּאָה, דְּלֹא שַׁיְכָא הַזָּאָה בִּשְׁבָרִים. וּבָעֵי נַמִּי שֶׁתְּהֵא שְׁבִירָה מַפְסֶקֶת בֵּין הַזָּאָה לְהַזָּאָה, לְפִי שֶׁיֶּשׁ הֶפְסֵק יָמִים בֵּין הַזָּאַת שְׁלִישִׁי לְהַזָּאַת שְׁבִיעִי. וְלֹא רָצוּ לְהַחְמִיר בַּחֲזָרַת טֻמְאָה יְשָׁנָה, שֶׁהִיא דְרַבָּנָן, וְהֵקֵלּוּ שֶׁתּוֹעִיל בָּהּ הַזָּאָה אֲפִלּוּ שְׁתֵּיהֶן הַהַזָּאוֹת בְּיוֹם אֶחָד:

פרק יד - משנה ח
רַבִּי יְהוֹשֻׁעַ אוֹמֵר אֵין הַזָּאָה פָּחוֹת מִשְּׁלִישִׁי וּשְׁבִיעִי. דְּתַקּוּן כְּעֵין דְּאוֹרַיְתָא . וַהֲלָכָה כְּרַבִּי יְהוֹשֻׁעַ: מַפְתֵּחַ שֶׁל אַרְכֻּבָּה. שֶׁמִּתְקַפֵּל כְּאַרְכֻּבָּה עִם הַשּׁוֹק, שֶׁעָשׂוּי כְמִין נוּ"ן כְּפוּפָה שֶׁלָּנוּ . וְרֹב מַפְתְּחוֹת שֶׁבְּאֶרֶץ יִשְׁמָעֵאל הֵם כֵּן: רַבִּי יְהוּדָה מְטַמֵּא. וְאֵין הֲלָכָה כְּרַבִּי יְהוּדָה: וְשֶׁל גַּם. כְּמִין גַּ"ם יְוָנִית שֶׁהִיא כְּמִין נוּ"ן הֲפוּכָה : הָיוּ בּוֹ חָפִין. כְּשֶׁנִּשְׁבַּר מִתּוֹךְ גּוּמוֹ, נִשְׁתַּיְּרוּ בוֹ חָפִין אוֹ נְקָבִין. חָפִין הֵם הַשִּׁנַּיִם. נְקָבִין, שֶׁנִּכְנָסִים בָּהֶן שִׁנֵּי הַמַּנְעוּל, שֶׁפְּעָמִים יֵשׁ בַּמַּפְתֵּחַ שִׁנַּיִם וְהֵן נִכְנָסִים בְּנִקְבֵי הַמַּנְעוּל וּפוֹתְחִים, וּפְעָמִים יֵשׁ נְקָבִים בַּמַּפְתֵּחַ וּמַכְנִיס אוֹתָן נְקָבִים בְּשִׁנֵּי הַמַּנְעוּל וּפוֹתֵחַ: אוֹ שֶׁפָּרְצוּ זֶה לְתוֹךְ זֶה. שֶׁלֹּא נִטְּלוּ הֶחָפִין וְלֹא נִסְתְּמוּ הַנְּקָבִין, אֶלָּא שֶׁפָּרְצוּ, כְּגוֹן שֶׁנֶּעֶקְמוּ הֶחָפִין וְנִתְרַחֲבוּ הַנְּקָבִין, שֶׁנּוֹגְעִים זֶה בָזֶה: מְסַנֶּנֶת שֶׁל חַרְדָּל. כְּמִין נָפָה שֶׁל מַתֶּכֶת שֶׁהַחַרְדָּל נָתוּן בָּהּ וּמִסְתַּנֵּן וְקוֹלֶטֶת אֶת הַפְּסֹלֶת: מִלְּמַטָּן. בְּשׁוּלֵי הַמְסַנֶּנֶת מִשֶּׁנִּפְרְצוּ בָהּ שְׁלֹשָׁה נְקָבִין וְנַעֲשׂוּ שְׁלָשְׁתָּן נֶקֶב אֶחָד, תּוּ לֹא חַזְיָא לְסַנֵּן בָּהּ:
