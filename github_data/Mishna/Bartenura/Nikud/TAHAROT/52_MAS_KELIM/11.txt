פרק יא

כְּלֵי מַתָּכוֹת פְּשׁוּטֵיהֶן טְמֵאִים. דְּלֹא אִתַּקַּשׁ לְשַׂק: חָזְרוּ לְטֻמְאָתָן הַיְשָׁנָה. חֲכָמִים גָּזְרוּ עַל כְּלֵי מַתָּכוֹת שֶׁהִתִּיכָן וְעָשָׂה מֵהֶן כֵּלִים חֲדָשִׁים, שֶׁיַּחְזְרוּ לְטֻמְאָתָן הַיְשָׁנָה. גְּזֵרָה שֶׁמָּא יֹאמְרוּ שְׁבִירָתָן שֶׁל כֵּלִים מְטַהֲרָתָן וּטְבִילָתָן מְטַהֲרָתָן. מַה שְּׁבִירָתָן אֵינָהּ טְעוּנָה הַעֲרֵב שֶׁמֶשׁ, שֶׁהֲרֵי מִשְׁתַּמְשִׁין בָּהֶן בְּאוֹתוֹ הַיּוֹם שֶׁנִּטְמְאוּ, וּשְׁבָרָן וְחָזַר וַעֲשָׂאָן חֲדָשִׁים, אַף טְבִילָתָן אֵינָהּ טְעוּנָה הַעֲרֵב שֶׁמֶשׁ, וְאָתְיָא לְמֶעֱבַד תְּרוּמָה וְקָדָשִׁים בְּכֵלִים שֶׁלֹּא הֶעֱרִיב שִׁמְשָׁן : לֹא לְכָל טֻמְאָה אֶלָּא לְטֻמְאַת הַנָּפֶשׁ. רַבָּן שִׁמְעוֹן בֶּן גַּמְלִיאֵל סָבַר דְּטַעֲמָא דְגָזְרוּ עַל כְּלֵי מַתָּכוֹת שֶׁיַּחְזְרוּ לְטֻמְאָתָן יְשָׁנָה אֵינָהּ אֶלָּא גְזֵרָה שֶׁמָּא תִשְׁתַּכַּח תּוֹרַת מֵי חַטָּאת מִן הַכֵּלִים, שֶׁאֵין לְךָ אָדָם מַמְתִּין שִׁבְעַת יָמִים אֶלָּא שׁוֹבְרָן מִיָּד וְחוֹזֵר וְעוֹשֶׂה אוֹתָן חֲדָשִׁים, הִלְכָּךְ לֹא גָזְרוּ אֶלָּא לְטֻמְאַת מֵת בִּלְבַד שֶׁצְּרִיכִים הַזָּאַת שְׁלִישִׁי וּשְׁבִיעִי. וְאֵין הֲלָכָה כְּרַבָּן שִׁמְעוֹן בֶּן גַּמְלִיאֵל:

שֶׁיֶּשׁ לוֹ שֵׁם בִּפְנֵי עַצְמוֹ. שֶׁהוּא יָדוּעַ וְנִכָּר וּמְיֻחָד בִּשְׁמוֹ: טָמֵא. בֵּין יֵשׁ לוֹ בֵּית קִבּוּל בֵּין אֵין לוֹ בֵּית קִבּוּל: הַנֶּגֶר. יָתֵד שֶׁל בַּרְזֶל שֶׁנּוֹעֵץ רֹאשׁוֹ אֶחָד בַּקַּרְקַע וְרֹאשׁוֹ הַשֵּׁנִי אֲחוֹרֵי הַדֶּלֶת. אִי נַמִּי, כְּמִין עַמּוּד שֶׁסּוֹגֵר וּמַבְרִיחַ הַדֶּלֶת מִן הַקָּצֶה אֶל הַקָּצֶה: מַנְעוּל. כָּל דָּבָר שֶׁנּוֹעֲלִים וְסוֹגְרִים בּוֹ הַדֶּלֶת נִקְרָא מַנְעוּל. וְדֻגְמָתוֹ בַּמִּקְרָא בְּשִׁיר הַשִּׁירִים (פֶּרֶק ה) עַל כַּפּוֹת הַמַּנְעוּל: פּוֹתָה. כְּמִין כּוֹס קָטָן שֶׁל בַּרְזֶל קוֹבְעִים בָּאָרֶץ, וְצִיר הַדֶּלֶת סוֹבֵב עָלָיו כְּדֵי שֶׁיְּהֵא נוֹחַ לְהִפָּתֵחַ וּלְהִסָּגֵר. וְדֻגְמָתוֹ בַמִּקְרָא (מְלָכִים א ז) וְהַפֹּתוֹת לְדַלְתוֹת הַבַּיִת הַפְּנִימִי: וְהַצִּנּוֹר. קנאל"י בְּלַעַ"ז: שֶׁנַּעֲשׂוּ לַקַּרְקָע. לְהִשְׁתַּמֵּשׁ בַּקַּרְקַע. וְכָל הַמְחֻבָּר לַקַּרְקַע הֲרֵי הוּא כַקַּרְקַע וְאֵינוֹ מְקַבֵּל טֻמְאָה:

הָעֶשֶׁת. חֲתִיכָה שֶׁל בַּרְזֶל כְּשֶׁמּוֹצִיאִים אוֹתוֹ מִן הַמַּחְצָב : חֲרָרָה. לְאַחַר שֶׁמַּתִּיכִים הַבַּרְזֶל עוֹשִׂים מִמֶּנּוּ עִגּוּל כְּמִין עוּגָה: סוֹבֵב שֶׁל גַּלְגַּל. גַּלְגַּל הָעֲגָלָה שֶׁל עֵץ, עוֹשִׂים סְבִיבוֹ טַס שֶׁל בַּרְזֶל שֶׁלֹּא יִשָּׁבֵר בַּטְּרָשִׁים וּבַסְּלָעִים: טַסִּין. וַיְרַקְּעוּ אֶת פַּחֵי הַזָּהָב (שְׁמוֹת לט), מְתַרְגְּמִינַן וְרַדִּידוּ יַת טַסֵּי דְּדַהֲבָא: כַּנֵּי כֵלִים. בְּסִיסֵי כֵלִים. לְשׁוֹן וְאֶת כַּנּוֹ (שָׁם), דִּמְתַרְגְּמִינַן וְיַת בְּסִיסֵיהּ: הוֹגְנֵי כֵלִים. כְּמִין עֲנָק סְבִיבוֹת שְׂפַת הַכֵּלִים : אָזְנֵי כֵלִים. בֵּית יָד שֶׁאוֹחֲזִים בּוֹ הַכְּלִי: הַשְּׁחוֹלוֹת. מַה שֶּׁנִּשְׁתַּיֵּר מִן הַכְּלִי לְאַחַר שֶׁנַּעֲשָׂה. לְשׁוֹן כְּמִשְׁחַל בִּינִיתָא מֵחֲלָבָא (בְּרָכוֹת דַּף ח.) כִּי מִן הַמַּיִם מְשִׁיתִהוּ (שְׁמוֹת ב), מְתַרְגְּמִינַן אֲרֵי מִן מַיָּא שְׁחַלְתֵּיהּ: גְּרוּדוֹת. שֶׁנִּגְרַד מִן הַכְּלִי בִּשְׁעַת עֲשִׂיָּתוֹ. לְשׁוֹן לְהִתְגָּרֵד בּוֹ הָאָמוּר בְּאִיּוֹב (פֶּרֶק ב): טְהוֹרִים. דְּלֹא חָיְשִׁינַן בְּכָל הָנֵי שֶׁמָּא מִכְּלִי טָמֵא בָאוּ וּכְשֶׁחָזַר וְעָשָׂה מֵהֶן כְּלִי חָזְרוּ לְטֻמְאָתָן יְשָׁנָה, דְּכָל הָנֵי לֹא נִרְאוּ לְקַבֵּל טֻמְאָה מֵעוֹלָם, מִשּׁוּם דְּגָלְמֵי כְלֵי מַתָּכוֹת נִינְהוּ, וְגָלְמֵי כְלֵי מַתָּכוֹת אֵין מְקַבְּלִין טֻמְאָה. וְהָכִי תַנְיָא בַּתּוֹסֶפְתָּא, גָּלְמֵי כְלֵי מַתָּכוֹת טְהוֹרִים, וְאֵלּוּ הֵן גָּלְמֵי כְלֵי מַתָּכוֹת, כָּל שֶׁעָתִיד לָשׁוּף , לְשַׁבֵּץ, לְגָרֵד , וּלְכַרְכֵּב, וּלְהָקִישׁ בְּקֻרְנָס, מְחֻסָּר אֹגֶן אוֹ אֹזֶן, טָהוֹר: אַף מִן הַקְּצוּצוֹת. אִם קָצַץ כֵּלִים לַחֲתִיכוֹת, כֵּלִים הַנַּעֲשִׂים מֵאוֹתָן חֲתִיכוֹת לֹא חָזְרוּ לְטֻמְאָתָן יְשָׁנָה . וְאֵין הֲלָכָה כְּרַבִּי יוֹחָנָן בֶּן נוּרִי: גְּרוּטָאוֹת. כֵּלִים שֶׁנִּשְׁתַּבְּרוּ מִפְּנֵי תַשְׁמִישָׁן וְנַעֲשָׂה בָּהֶן חֲתִיכוֹת קְטַנּוֹת אַחֲרֵי בְלוֹתָן נִקְרָאִים גְּרוּטוֹת. וְהֵן קְטַנִּים מִשִּׁבְרֵי כֵלִים: וּמִן הַמַּסְמְרוֹת שֶׁיָּדוּעַ שֶׁנַּעֲשׂוּ מִכֵּלִים. אַף עַל גַּב דְּלֹא יָדְעִינַן אִם מִכֵּלִים טְהוֹרִים בָּאוּ אוֹ מִן הַטְּמֵאִין: טְמֵאִין. טְמֵאִים הַכֵּלִים הַנַּעֲשִׂים מֵהֶם. דְּחָיְשִׁינַן שֶׁמָּא מִכֵּלִים טְמֵאִין בָּאוּ וְחָזְרוּ לְטֻמְאָתָן הַיְשָׁנָה: מִן הַמַּסְמְרוֹת. סְתָם. דְּלֹא יָדְעִינַן אִם נַעֲשׂוּ מִכֵּלִים אוֹ מִן הָעֶשֶׁת: בֵּית שַׁמַּאי מְטַמְּאִין. דְּגָזְרוּ הָנֵי אַטּוּ הָנֵי: וּבֵית הִלֵּל מְטַהֲרִין. דְּלֹא גָזְרוּ:

בַּרְזֶל טָמֵא. שֶׁבָּא מִשִּׁבְרֵי כֵלִים: שֶׁבְּלָלוֹ. שֶׁטְּרָפָן זֶה בָזֶה וְהִתִּיכָן וְעָשָׂה מִשְּׁנֵיהֶם כְּלִי: בַּרְזֶל טָהוֹר. שֶׁבָּא מִן הָעֶשֶׁת וּמִן הַחֲרָרָה וְכוּ': אִם רֹב מִן הַטָּמֵא טָמֵא. דְּחוֹזֵר לְטֻמְאָתוֹ הַיְשָׁנָה: הַחֲלָמָא. טִיט הַדָּבֵק כְּלֹבֶן בֵּיצָה : וּמִן הַגְּלָלִים. שֶׁעֵרֵב הַטִּיט עִם גְּלָלֵי הַבָּקָר וְעָשָׂה מֵהֶן כֵּלִים וּצְרָפָן בַּכִּבְשָׁן. אִם רֹב מִן הַחֲלָמָא. מְקַבְּלִים טֻמְאָה כְּדִין כְּלֵי חֶרֶס. וְאִם רֹב מִן הַגְּלָלִים, טְהוֹרִים, דִּכְלֵי גְלָלִים אֵין מְקַבְּלִין טֻמְאָה: קְלוֹסְטְרָא. הַנֶּגֶר שֶׁל הַדֶּלֶת. וְהוּא הַיָּתֵד שֶׁנּוֹעֲלִים בּוֹ הַדֶּלֶת כִּדְפָרֵישְׁנָא לְעֵיל, רְגִילִים לַעֲשׂוֹת בְּרֹאשׁוֹ כְמִין תַּפּוּחַ, וְהוּא נִקְרָא גְלוֹסְטְרָא. וְאִם נַעֲשָׂה מִבַּרְזֶל טָמֵא, חוֹזֵר לְטֻמְאָתוֹ הַיְשָׁנָה, לְפִי שֶׁרָאוּי לָדוּךְ בּוֹ שׁוּמִין בִּמְדוֹכָה: וּמְצֻפָּה טְהוֹרָה. אִם הָיְתָה שֶׁל בַּרְזֶל טָהוֹר וּמְצֻפָּה שֶׁל מַתֶּכֶת טָמֵא, טְהוֹרָה, דְּבָתַר עִקָּרָא אָזְלִינַן וְלֹא בָתַר הַצִּפּוּי שֶׁהוּא שֶׁל מַתֶּכֶת טָמֵא : הַפִּין וְהַפּוּרְנָא. שִׁנַּיִם הַקְּבוּעִים בַּפּוֹתַחַת, וְהַפִּין נִכְנָס בְּתוֹךְ הַפּוּרְנָא : שׁוֹמְטָהּ מִפֶּתַח זֶה וְתוֹלָהּ בַּחֲבֵרוֹ. עַל יְדֵי גְרִירָה בִלְבַד, דְּטִלְטוּל מִן הַצַּד הוּא. אֲבָל טִלְטוּל גָּמוּר, לֹא. וְרַבִּי טַרְפוֹן שָׁרֵי אֲפִלּוּ טִלְטוּל גָּמוּר כִּי אָרְחֵיהּ. וְשֶׁמָּא אֲתָא תַנָּא לְאַשְׁמְעִינַן הָכָא דִקְלוֹסְטְרָא טְמֵאָה דִתְנַן בָּרֵישָׁא לָאו דִּבְרֵי הַכֹּל הִיא אֶלָּא פְּלֻגְתָּא דְרַבִּי יְהוֹשֻׁעַ וְרַבִּי טַרְפוֹן, דִּלְרַבִּי יְהוֹשֻׁעַ טְהוֹרָה, דְּכִי הֵיכִי דְלָא מְשַׁוֵּי לַהּ מָנָא לְעִנְיַן שַׁבָּת הָכִי נַמִּי לֹא מְשַׁוֵּי לַהּ מָנָא לְעִנְיַן טֻמְאָה:

עַקְרָב שֶׁל פְּרֻמְבְּיָא. מֶתֶג וָרֶסֶן וּפְרֻמְבְּיָא, אֶחָד הֵן. וְהַבַּרְזֶל הַנִּכְנָס בְּפִי הַבְּהֵמָה נִקְרָא עַקְרָב: וּלְחָיַיִם. מְחֻבָּרִים לָעַקְרָב כְּמִין שְׁנֵי טַסִּין מְצֻיָּרִים, אֶחָד מִכָּאן וְאֶחָד מִכָּאן עַל לְחָיֵי הַבְּהֵמָה, עֲשׂוּיִין לְנוֹי. וְהֵן טְהוֹרִין כְּשֶׁהֵן בִּפְנֵי עַצְמָן, דְּתַכְשִׁיטֵי בְהֵמָה נִינְהוּ, וְתַכְשִׁיטֵי בְהֵמָה וְכֵלִים אֵין מְקַבְּלִים טֻמְאָה:

פִּיקָה. כְּמִין כַּדּוּר קָטָן נָקוּב בָּאֶמְצַע שֶׁמְּשִׂימוֹת הַנָּשִׁים בְּרֹאשׁ הַפֶּלֶךְ שֶׁטֹּווֹת בּוֹ כְּדֵי שֶׁיִּכְבַּד: רַבִּי עֲקִיבָא מְטַמֵּא. דִּכְלִי מַתָּכוֹת שֶׁיֵּשׁ לוֹ שֵׁם בִּפְנֵי עַצְמוֹ הוּא, וְטָמֵא, כְּדִתְנַן בְּרֵישׁ פִּרְקִין: וַחֲכָמִים מְטַהֲרִין. דְּאֵין לוֹ שֵׁם בִּפְנֵי עַצְמוֹ אֶלָּא שֵׁם לְוַי , פִּיקָא שֶׁל כּוּשׁ, אוֹ פִּיקָה שֶׁל אוֹתוֹ דָבָר שֶׁהִיא מְשַׁמֶּשֶׁת לוֹ. וַהֲלָכָה כַּחֲכָמִים: וּמְצֻפָּה. אִם הִיא שֶׁל עֵץ וּמְצֻפָּה בְמַתֶּכֶת טָמֵא: טְהוֹרָה. דְּבָתַר עִקָּרָא אָזְלִינַן, וְלֹא בָתַר הַצִּפּוּי שֶׁהוּא שֶׁל מַתֶּכֶת טָמֵא: כּוּשׁ. פֶּלֶךְ שֶׁהַנָּשִׁים טֹווֹת בּוֹ: וְהַמַּקֵּל. שֶׁל מַתֶּכֶת טָמֵא. הוֹאִיל וְאָדָם רוֹדֶה בוֹ אֶת הַבְּהֵמָה, כִּכְלִי עָשׂוּי לְתַשְׁמִישׁ אָדָם הוּא חָשׁוּב: סִמְפּוֹנְיָא. מִין כְּלִי זֶמֶר הוּא, כְּדִכְתִיב בְּדָנִיֵּאל (פֶּרֶק ג) פְּסַנְתְּרִין וְסִמְפּוֹנְיָא: וְחָלִיל. אַף הוּא מִין שֶׁל כְּלִי זֶמֶר, וְחָלוּל הוּא וְקוֹלוֹ נִשְׁמָע עַד לְמֵרָחוֹק: וּמְצֻפִּין טְהוֹרִים. אִם שֶׁל עֵץ וּמְצֻפֶּה נְחֹשֶׁת טָמֵא, טָהוֹר הוּא. דְּבָתַר עִקַּר הַכְּלִי אָזְלִינַן שֶׁהוּא שֶׁל עֵץ, וְלָא חֲשִׁיב יֵשׁ לוֹ בֵית קִבּוּל אַף עַל פִּי שֶׁהוּא חָלוּל, כֵּיוָן שֶׁאֵינוֹ עָשׂוּי לְבֵית קִבּוּל: אִם יֶשׁ לָהּ בֵּית קִבּוּל כְּנָפַיִם. כֵּלִים דַּקִּים קְטַנִּים כִּדְמוּת כְּנָפַיִם נוֹתְנִים בְּתוֹךְ הַסִּמְפּוֹנְיָא. שֶׁמְּסַיְּעִים לְהַשְׁמָעַת קוֹל הַנִּגּוּן. וְאִם עָשׂוּ בְתוֹכָהּ מָקוֹם לְבֵית קִבּוּל שֶׁל כְּנָפַיִם הַלָּלוּ, חֲשִׁיב בֵּית קִבּוּל: בֵּין כָּךְ וּבֵין כָּךְ. בֵּין שֶׁהִיא מְצֻפָּה וּבֵין שֶׁאֵינָהּ מְצֻפָּה, טְמֵאָה:

קֶרֶן עֲגֻלָּה וְקֶרֶן פְּשׁוּטָה. חֲצוֹצְרוֹת מִקַּרְנֵי הַבְּהֵמוֹת עֲשׂוּיוֹת פְּרָקִים פְּרָקִים מְחֻבָּרִים , אֶלָּא שֶׁאַחַת מֵהֶן צוּרָתָהּ עֲגֻלָּה וְאַחֶרֶת פְּשׁוּטָה. וְחִבּוּר הַפְּרָקִים שֶׁל עֲגֻלָּה קָשִׁים וּצְרִיכִים אֻמָּן יוֹתֵר מֵחִבּוּר הַפְּרָקִים שֶׁל פְּשׁוּטָה, וּמִפְּנֵי זֶה אָמְרוּ הַמַּחֲזִיר קֶרֶן עֲגֻלָּה בְשַׁבָּת חַיָּב חַטָּאת, מַה שֶּׁאֵין כֵּן בִּפְשׁוּטָה: קֶרֶן עֲגֻלָּה טְמֵאָה. שֶׁיֵּשׁ לָהּ בֵּית קִבּוּל. וְקֶרֶן פְּשׁוּטָה טְהוֹרָה, שֶׁאֵין לָהּ בֵּית קִבּוּל. וְקַרְנַיִם דִּין כְּלֵי עֶצֶם יֵשׁ לָהֶן, שֶׁפְּשׁוּטֵיהֶן טְהוֹרִים כִּפְשׁוּטֵי כְלֵי עֵץ: מְצֻפִּית. מְקוֹם הַנָּחַת הַפֶּה, וְהוּא הַקָּצָר שֶׁבָּהּ . מִלְּשׁוֹן עוֹמְדִים צְפוּפִים: טְמֵאָה. דְּחָשְׁבִינַן לָהּ כִּכְלֵי מַתָּכוֹת, וְאַף עַל פִּי שֶׁהִיא פְּשׁוּטָה מְקַבֶּלֶת טֻמְאָה: הַקַּו שֶׁלָּהּ. הַצַּד הָרָחָב שֶׁבָּהּ, אִם הָיָה שֶׁל מַתָּכוֹת: רַבִּי טַרְפוֹן מְטַמֵּא וַחֲכָמִים מְטַהֲרִים. וַהֲלָכָה כַּחֲכָמִים: קְנֵי מְנוֹרָה טְהוֹרִים. שֶׁאֵין לָהֶם שֵׁם בִּפְנֵי עַצְמָן, וּתְנַן לְעֵיל בְּרֵישׁ פִּרְקִין כָּל כְּלִי מַתָּכוֹת שֶׁיֵּשׁ לוֹ שֵׁם בִּפְנֵי עַצְמוֹ, טָמֵא: הַפֶּרַח. מְקוֹם מוֹשַׁב הַנֵּר שֶׁבְּרֹאשׁ הַמְּנוֹרָה פֶּרַח שְׁמוֹ: בָּסִיס. רַגְלֵי הַמְּנוֹרָה שֶׁהִיא יוֹשֶׁבֶת עֲלֵיהֶן:

קַסְדָּא. כּוֹבַע נְחֹשֶׁת שֶׁנּוֹתְנִים אַנְשֵׁי הַמִּלְחָמָה עַל רֹאשָׁן: וּלְחָיַיִם. טַסִּין שֶׁל בַּרְזֶל שֶׁנּוֹתְנִים עַל לְחָיֵיהֶן: אִם יֵשׁ בָּהֶן בֵּית קִבּוּל מַיִם. שֶׁשּׁוֹתִין בָּהֶם מַיִם בַּמִּלְחָמָה: כִּידוֹן. רֹמַח קָטָן שֶׁבְּיַד הַשָּׂרִים וְהַמְּלָכִים : נִיקוֹן. כְּמוֹ קִינוֹן. לְשׁוֹן וּמִשְׁקַל קֵינוֹ (שְׁמוּאֵל ב כא). וְהוּא הַבַּרְזֶל שֶׁבְּתוֹךְ הָרֹמַח: מַגָּפַיִם. בָּתֵּי שׁוֹקַיִם שֶׁל בַּרְזֶל שֶׁנּוֹשְׂאִין אַנְשֵׁי הַמִּלְחָמָה: תַּכְשִׁיטֵי נָשִׁים. אַף עַל פִּי שֶׁאֵין עֲשׂוּיִין לִמְלָאכָה אֶלָּא לְנוֹי, מְרַבִּינַן לְהוּ מִקְּרָא דִכְתִיב (בְּמִדְבַּר לא) וְכָל אֲשֶׁר לֹא יָבֹא בָּאֵשׁ תַּעֲבִירוּ בַמָּיִם : עִיר שֶׁל זָהָב. כְּמִין עֲטָרָה נוֹתְנוֹת הַנָּשִׁים בְּרֹאשָׁן עֲשׂוּיָה כְצוּרַת יְרוּשָׁלַיִם: קַטְלָיוֹת. חוּטִים שֶׁל זָהָב שֶׁנּוֹתְנוֹת סְבִיב הַצַּוָּאר. וּמִפְּנֵי שֶׁהָאִשָּׁה חוֹנֶקֶת עַצְמָהּ בָּהֶן כְּדֵי שֶׁתֵּרָאֶה בַּעֲלַת בָּשָׂר, לְכָךְ נִקְרָאִים קַטְלָיוֹת: וּנְזָמִים. יֵשׁ מֵהֶן נִזְמֵי הָאַף וְיֵשׁ מֵהֶן נִזְמֵי הָאֹזֶן : בְּחוּט שֶׁל פִּשְׁתָּן. שֶׁהָיוּ הַחֻלְיוֹת נְקוּבוֹת וּמִלְּאוּ מֵהֶן הַחוּט: הַחֻלְיוֹת טְמֵאוֹת. דְּכָל חֲדָא בְאַנְפֵּי נַפְשָׁא חֲשִׁיבָא כְלִי מַעֲשֶׂה: שֶׁל אֲבָנִים. שֶׁהָיוּ מַרְגָּלִיּוֹת וַאֲבָנִים נְקוּבוֹת וּמִלְּאוּ מֵהֶן הַחוּט: טָמֵא. הַחוּט, הוֹאִיל וְשֶׁל מַתֶּכֶת הוּא: כִּמְלֹא צַוַּאר קְטַנָּה. כִּמְלֹא צַוָּאר שֶׁל בַּת קְטַנָּה:

כִּקְדֵרָה מִלְּמַטָּה. רָחָב מִלְּמַטָּה וְיֵשׁ לוֹ בֵית קִבּוּל: וְכָעֲדָשָׁה מִלְמַעְלָה. עַל רֹאשׁ הַנֶּזֶם מִלְּמַעְלָה יֵשׁ לוֹ גַרְעִין שֶׁל זָהָב אוֹ שֶׁל כֶּסֶף עָשׂוּי כָּעֲדָשָׁה: וְנִפְרַק. הָעֲדָשָׁה מִן הַקְּדֵרָה: הַקְּדֵרָה טְמֵאָה. שֶׁהֲרֵי יֵשׁ לָהּ בֵּית קִבּוּל . וְלֹא מִפְּנֵי שֶׁהִיא מִתַּכְשִׁיטֵי נָשִׁים, שֶׁהֲרֵי אֵינָהּ רְאוּיָה לְתַכְשִׁיט כְּמוֹת שֶׁהִיא: וְהָעֲדָשָׁה טְמֵאָה. שֶׁיֵּשׁ לָהּ שֵׁם בִּפְנֵי עַצְמָהּ: צִנּוֹרָא. רֹאשׁ הַנֶּזֶם הַנִּכְנָס בְּנֶקֶב הָאֹזֶן אוֹ בְנֶקֶב הָאַף. וּלְפִי שֶׁהוּא עָשׂוּי כְּמַזְלֵג קָרוּי צִנּוֹרָא. תַּרְגּוּם מִזְלְגוֹתָיו צִנּוֹרְיָתֵהּ: טְהוֹרָה. שֶׁאֵין לָהּ שֵׁם בִּפְנֵי עַצְמָהּ: הֶעָשׂוּי כְּמִין אֶשְׁכּוֹל. נֶזֶם הֶעָשׂוּי אַרְבָּעָה אוֹ חֲמִשָּׁה גַרְגְּרִין זֶה עַל גַּבֵּי זֶה כְּמִין אֶשְׁכּוֹל: וְנִפְרַק טָהוֹר. שֶׁאֵין לְכָל גַּרְגִּיר וְגַרְגִּיר שֵׁם בִּפְנֵי עַצְמוֹ, וְהוּא אֵינוֹ רָאוּי לְתַשְׁמִישׁ מִשֶּׁנִּפְרַק:
