פרק כה

פרק כה - משנה א
כָּל הַכֵּלִים יֵשׁ לָהֶם אֲחוֹרַיִם וָתוֹךְ. כֵּלִים שֶׁנִּטְמְאוּ בְמַשְׁקִין טְמֵאִים, אִם נָגְעוּ הַמַּשְׁקִין מֵאֲחוֹרֵיהֶן שֶׁל כֵּלִים נִטְמְאוּ אֲחוֹרֵיהֶן וְלֹא נִטְמָא תוֹכָן, מִשּׁוּם דְּטֻמְאַת מַשְׁקִים לְטַמֵּא כֵלִים לָאו דְּאוֹרַיְתָא, הֵקֵלּוּ חֲכָמִים בְּטֻמְאָתָן. אֲבָל אִם נִטְמָא תוֹכָן, כְּגוֹן שֶׁנָּגְעוּ מַשְׁקִין טְמֵאִין בְּתוֹכָן, נִטְמְאוּ אֲחוֹרֵיהֶם. וְהַיְנוּ דִתְנַן יֵשׁ לָהֶן אֲחוֹרַיִם וָתוֹךְ, כְּלוֹמַר שֶׁחָלוּק דִּין אֲחוֹרֵיהֶם מִדִּין תּוֹכָן: כְּגוֹן הַכָּרִים וְהַכְּסָתוֹת. וְאַף עַל פִּי שֶׁמִּשְׁתַּמְּשִׁים בָּהֶן בְּתוֹכָן וּבַאֲחוֹרֵיהֶן, שֶׁרָאוּי לְהָפְכָן וְלַעֲשׂוֹת תּוֹכָן אֲחוֹרֵיהֶן וַאֲחוֹרֵיהֶן תּוֹכָן, אַף עַל פִּי כֵן חָלוּק דִּין תּוֹכָן מִדִּין אֲחוֹרֵיהֶן: מַרְצוּפִין. שַׂקִּין גְּדוֹלִים שֶׁל עוֹר שֶׁמּוֹלִיכִים בָּהֶן סְחוֹרוֹת בַּסְּפִינוֹת: כָּל שֶׁיֶּשׁ לוֹ תוֹבְרוֹת. לוּלָאוֹת, כְּמוֹ שֶׁיֶּשׁ לְמִכְנָסַיִם, תְּפוּרוֹת לַאֲחוֹרָיו שֶׁל בֶּגֶד : יֶשׁ לוֹ אֲחוֹרַיִם וָתוֹךְ. דְּכֵיוָן דְּיֵשׁ לוֹ תוֹבְרוֹת לֹא חֲזִי לְהָפְכוֹ, וְתוֹכוֹ חָלוּק מֵאֲחוֹרָיו. אֲבָל כְּשֶׁאֵין לוֹ תוֹבְרוֹת, דַּחֲזִי לְהָפְכוֹ, אֵין תּוֹכוֹ חָלוּק מֵאֲחוֹרָיו, אֶלָּא כְשֶׁנָּגְעוּ מַשְׁקִים טְמֵאִים בֵּין מֵאֲחוֹרָיו בֵּין מִתּוֹכוֹ נִטְמָא כֻלּוֹ. וַהֲלָכָה כְּרַבִּי יְהוּדָה: הַשֻּׁלְחָן וְהַדְּלֻפְקִי. קָסָבַר רַבִּי יְהוּדָה, כֵּיוָן דְּפָנָיו מְשֻׁפִּים וַחֲלָקִין, הַפָּנִים הָהֵם נִדּוֹנִים תּוֹכוֹ. וַהֲלָכָה כְּרַבִּי יְהוּדָה:

פרק כה - משנה ב
מַרְדֵּעַ. הוּא מַקֵּל אָרֹךְ עָגֹל וְעָב שְׁלִישׁ טֶפַח, וּבְרֹאשׁוֹ הָאֶחָד בַּרְזֶל רָחָב מְחֻדָּד לַחְתֹּךְ בּוֹ שָׁרָשִׁים, וּשְׁמוֹ חַרְחוּר, וּבְרֹאשׁוֹ הַשֵּׁנִי בַרְזֶל כְּמוֹ מַרְצֵעַ, וּשְׁמוֹ דָרְבָן, שֶׁבּוֹ מְלַמֵּד הַפָּרָה לִתְלָמֶיהָ: מִשִּׁבְעָה לַחַרְחוּר מֵאַרְבָּעָה לַדָּרְבָן. אִם נָפְלוּ מַשְׁקִין טְמֵאִים עַל הַחַרְחוּר, אֵין טָמֵא מִן הַמַּרְדֵּעַ אֶלָּא שִׁבְעָה טְפָחִים. וְאִם עַל הַדָּרְבָן נָפְלוּ, אֵין טָמֵא אֶלָּא אַרְבָּעָה . וְרַמְבַּ"ם פֵּרֵשׁ, שֶׁאִם נָפְלוּ מַשְׁקִין טְמֵאִין בַּמַּרְדֵּעַ חוּץ לְשִׁבְעָה טְפָחִים מִן הַחַרְחוּר אוֹ חוּץ לְאַרְבָּעָה טְפָחִים מִן הַדָּרְבָן, הָוֵי כִכְלִי שֶׁנִּטְמְאוּ אֲחוֹרָיו וְלֹא נִטְמָא תוֹכוֹ, וְאוֹתָן שִׁבְעָה טְפָחִים הַסְּמוּכִים לַחַרְחוּר וְהָאַרְבָּעָה הַסְּמוּכִים לַדָּרְבָן לֹא נִטְמְאוּ: אֶלָּא לַשְּׁיָרִים. כְּשֶׁנָּתְנוּ חֲכָמִים שִׁעוּר לַחַרְחוּר שִׁבְעָה וְלַדָּרְבָן אַרְבָּעָה, לֹא נָתְנוּ אֶלָּא לְעִנְיָן זֶה שֶׁאִם נִשְׁבַּר הַמַּרְדֵּעַ וְנִשְׁאַר סָמוּךְ לַחַרְחוּר שִׁבְעָה טְפָחִים אוֹ סָמוּךְ לַדָּרְבָן אַרְבָּעָה, טָמֵא, מִפְּנֵי שֶׁרָאוּי לַעֲשׂוֹת מֵעֵין מְלַאכְתּוֹ. פָּחוֹת מִכָּאן, הָוֵי כְשִׁבְרֵי כְלִי וְטָהוֹר. וַהֲלָכָה כְּרַבִּי יְהוּדָה:

פרק כה - משנה ג
זוֹמָא לִסְטְרָא. כְּלִי שֶׁרֹאשׁוֹ אֶחָד כַּף וְרֹאשׁוֹ הָאַחֵר מַזְלֵג. בַּכַּף מַעֲלִין בּוֹ מָרָק מִן הַקְּדֵרָה וּבַמַּזְלֵג מַעֲלִין בּוֹ בָשָׂר: רַבִּי יְהוּדָה אוֹמֵר אֵין לָהֶם. אֲחוֹרַיִם וָתוֹךְ. טַעֲמָא דְרַבִּי יְהוּדָה, מִשּׁוּם דְּכָל הָנֵי כֵלִים יֵשׁ מֵאֲחוֹרֵיהֶן חֲקָק בֵּית מוֹשָׁב כְּעֵין שֶׁעוֹשִׂים לִכְלֵי כֶסֶף, וּפְעָמִים שֶׁכּוֹפֵהוֹ עַל פִּיו וּמִשְׁתַּמֵּשׁ בַּחֲקָק שֶׁבַּאֲחוֹרָיו: רַבִּי שִׁמְעוֹן אוֹמֵר יֶשׁ לָהֶם. לְעִנְיַן מַה שֶּׁבְּתוֹךְ הַכְּלִי יֵשׁ לָהֶן דִּין אֲחוֹרַיִם וָתוֹךְ, שֶׁאִם נִטְמְאוּ אֲחוֹרָיו לֹא נִטְמָא מַה שֶּׁבְּתוֹךְ הַכְּלִי. אֲבָל לְעִנְיַן הַכְּלִי עַצְמוֹ אֵין חִלּוּק דִּין אֲחוֹרָיו מִדִּין תּוֹכוֹ, שֶׁאִם נִטְמְאוּ אֲחוֹרָיו צָרִיךְ לְהַטְבִּיל אֶת כֻּלּוֹ כְאִלּוּ נִטְמָא תּוֹכוֹ. וַהֲלָכָה כְּרַבִּי יְהוּדָה :

פרק כה - משנה ד
הָרֹבַע וַחֲצִי הָרֹבַע. כְּגוֹן חֲתִיכַת עֵץ עָבָה וַאֲרֻכָּה וּבוֹ שְׁתֵּי מִדּוֹת חֲקוּקוֹת זוֹ אֵצֶל זוֹ, אֶחָד מַחֲזִיק רֹבַע וְאֶחָד מַחֲזִיק חֲצִי רֹבַע, וְדֹפֶן מַפְסִיק בֵּין זוֹ לְזוֹ: כְּלִי שֶׁנִּטְמָא תוֹכוֹ. בִּתְמִיהָה. וְכִי כְלִי שֶׁנִּטְמָא תוֹכוֹ לֹא נִטְמְאוּ אֲחוֹרָיו, וְאַמַּאי תְּנַן לֹא נִטְמָא חֲצִי רֹבַע, וַהֲלֹא אֲחוֹרָיו שֶׁל רֹבַע הִיא: קוֹדְמִין הִיא. כָּל אַחַת מֵהֶן לָוְתָה הַקְּדִימָה מֵחֲבֶרְתָּהּ, כְּלוֹמַר שֶׁאֵין לָהֶם קְדִימָה זוֹ עַל זוֹ. וּכְשֵׁם שֶׁאַתָּה אוֹמֵר שֶׁחֲצִי רֹבַע אֲחוֹרַיִם לָרֹבַע, כָּךְ תֹּאמַר שֶׁהָרֹבַע אֲחוֹרַיִם לַחֲצִי רֹבַע, וּכְשֶׁנִּטְמָא הָרֹבַע, אֲחוֹרָיו שֶׁל חֲצִי רֹבַע נִטְמְאוּ וְלֹא תוֹכוֹ. וְגִרְסַת רַבּוֹתַי, שֶׁל כַּת קוֹדְמִים הִיא, לְפִי שֶׁכִּתּוֹת שֶׁל תַּלְמִידִים הָיוּ יוֹשְׁבִים לְפָנָיו שׁוּרוֹת, אָמַר לָהֶם שְׁאֵלָה זוֹ כְּבָר שְׁאָלוּהָ מִמֶּנִּי הַכַּת שֶׁל תַּלְמִידִים יוֹשְׁבִים בַּשּׁוּרָה הַקּוֹדֶמֶת, וְכָךְ הֵשַׁבְתִּי לָהֶם אוֹ שֶׁמָּא וְכוּ':

פרק כה - משנה ה
הָרֹבַע וַאֲחוֹרָיו טְמֵאִים. הָרֹבַע הַיְנוּ תוֹכוֹ, וַאֲחוֹרָיו הַיְנוּ הַדְּפָנוֹת הַמַּקִּיפוֹת אֶת הָרֹבַע מִבַּחוּץ. אֲבָל דֹּפֶן הַמַּפְסִיק בֵּין שְׁנֵיהֶם, בִּכְלַל תּוֹכוֹ שֶׁל חֲצִי רֹבַע , וְטָהוֹר צַד שֶׁכְּלַפֵּי חֲצִי רֹבַע. וְכֵן כְּשֶׁנִּטְמָא חֲצִי הָרֹבַע, חֲצִי רֹבַע וַאֲחוֹרָיו טְמֵאִין. חֲצִי הָרֹבַע הַיְנוּ תוֹכוֹ, וַאֲחוֹרָיו הַיְנוּ דְפָנוֹת הַמַּקִּיפוֹת חֲצִי הָרֹבַע מִבַּחוּץ, וְדֹפֶן הַמַּפְסִיק, צַד שֶׁל כְּלַפֵּי הָרֹבַע בִּכְלַל תּוֹכוֹ שֶׁל רֹבַע: נִטְמְאוּ אֲחוֹרֵי הָרֹבַע. הַיְנוּ דְפָנוֹת הַמַּקִּיפוֹת מִבַּחוּץ: אֵין חוֹלְקִין אֶת הַגַּבָּיִים. הַיְנוּ הָאֲחוֹרַיִם, וְכָל הָאֲחוֹרַיִם טְמֵאִין כְּשֶׁנִּטְמָא אֶחָד מִן הָאֲחוֹרַיִם: מַטְבִּיל אֶת כֻּלּוֹ. רֹבַע וַחֲצִי רֹבַע. וְאֵין הֲלָכָה כְּרַבִּי מֵאִיר. וְדִין דֹּפֶן הָאֶמְצָעִי מְפֹרָשׁ לְעֵיל בְּסוֹף פֶּרֶק ב', דְּרַבִּי יוֹחָנָן בֶּן נוּרִי אוֹמֵר חוֹלְקִין אֶת עָבְיוֹ, הַמְשַׁמֵּשׁ לַטָּמֵא טָמֵא וְהַמְשַׁמֵּשׁ לַטָּהוֹר טָהוֹר:

פרק כה - משנה ו
כַּנֵּי הַכֵּלִים. בְּסִיסֵי הַכֵּלִים. לְשׁוֹן וְאֶת כַּנּוֹ, דִּמְתַרְגְּמִינַן וְיַת בְּסִיסֵיהּ: וְהוֹגְנֵיהֶן. כְּמוֹ וְאָגְנֵיהֶן, כְּלוֹמַר שִׂפְתוֹתֵיהֶן: וְאָזְנֵיהֶן. מָקוֹם שֶׁתּוֹחֲבִין בּוֹ הַבֵּית יָד: הַמְקַבְּלִין. שֶׁיֶּשׁ לָהֶן תּוֹךְ: מְנַגְּבָן וְהֵן טְהוֹרִין. דְּלָא חֲשִׁיבֵי הָנֵי חִבּוּר לְהָבִיא טֻמְאַת מַשְׁקִין לִכְלִי שֶׁיֶּשׁ בּוֹ בֵית קִבּוּל, וְסַגִּי בְנִגּוּב. אֲבָל לִכְלִי שֶׁאֵין לוֹ תוֹךְ בָּעֵי טְבִילָה: וְאָזְנָיו וְיָדָיו טְהוֹרִין. עֲבַדוּ רַבָּנָן בֵּיהּ הֶכֵּרָא, כִּי הֵיכִי דְלָא לִשְׂרֹף עֲלַיְהוּ תְרוּמָה וְקָדָשִׁים. וְדַוְקָא בַתְּרוּמָה עֲבַדוּ הֶכֵּרָא, אֲבָל לֹא בַקָּדָשִׁים, כְּדִתְנַן לְקַמָּן בְּפִרְקִין:

פרק כה - משנה ז
בֵּית הַצְּבִיטָה. גָּרְסִינַן. מָקוֹם שֶׁאוֹחֲזִין הַכְּלִי כְּשֶׁמּוֹשִׁיטִין אוֹתוֹ מִזֶּה לָזֶה. לְשׁוֹן וַיִּצְבֹּט לָהּ קָלִי (רוּת ב) שֶׁפֵּרוּשׁוֹ וַיּוֹשֶׁט. וְאִית דְּגָרְסֵי, בֵּית הַצְּבִיעָה, מָקוֹם שֶׁאוֹחֵז בּוֹ בְאֶצְבָּעוֹ: וְיֵשׁ לָהֶן בֵּית הַצְּבִיטָה. שֶׁאִם נִטְמְאוּ אֲחוֹרָיו, בֵּית הַצְּבִיטָה טָהוֹר. וְאִם נִטְמָא בֵּית הַצְּבִיטָה, אֲחוֹרָיו טְהוֹרִין: לַעֲרֵבָה גְדוֹלָה שֶׁל עֵץ. יֵשׁ לָהּ בֵּית צְבִיטָה. אֲבָל לִשְׁאָר כֵּלִים, לֹא. שֶׁבַּעֲרֵבָה גְדוֹלָה יֵשׁ הֶפְרֵשׁ בֵּין בֵּית הַצְּבִיטָה לָאֲחוֹרַיִם, שֶׁאִם יֵשׁ מַשְׁקִים בַּאֲחוֹרֵי הָעֲרֵבָה וְיָדָיו טְמֵאוֹת, אוֹחֵז בְּבֵית הַצְּבִיטָה וְאֵין חוֹשֵׁשׁ שֶׁמָּא יִגַּע בַּמַּשְׁקִין שֶׁבַּאֲחוֹרָיו וְיַחְזְרוּ וִיטַמְּאוּ אֶת הָעֲרֵבָה : רַבִּי עֲקִיבָא אוֹמֵר לְכוֹסוֹת. אַף לְכוֹסוֹת יֵשׁ לָהֶן בֵּית צְבִיטָה: לְיָדַיִם הַטְּמֵאוֹת וְהַטְּהוֹרוֹת. יֵשׁ הֶפְרֵשׁ בֵּין בֵּית הַצְּבִיטָה לָאֲחוֹרַיִם, לְיָדַיִם הַטְּמֵאוֹת כִּדְפָרֵישִׁית, וְלִטְהוֹרוֹת כְּדִמְפָרֵשׁ לְקַמָּן כֵּיצַד. וְרַבִּי יוֹסֵי סָבַר לְיָדַיִם טְהוֹרוֹת בִּלְבַד לֹא חָיְשִׁינַן, אֲבָל לְיָדַיִם טְמֵאוֹת חָיְשִׁינַן. וַהֲלָכָה כְּתַנָּא קַמָּא וּכְדִמְפָרֵשׁ רַבִּי יוֹסֵי:

פרק כה - משנה ח
הָיוּ יָדָיו טְהוֹרוֹת. וַעֲלֵיהֶם מַשְׁקֶה טוֹפֵחַ: אֵינוֹ חוֹשֵׁשׁ שֶׁמָּא נִטְמְאוּ. מַשְׁקִין שֶׁעַל יָדָיו בַּאֲחוֹרֵי הַכּוֹס וְיַחְזְרוּ וִיטַמְּאוּ אֶת הַיָּדַיִם:

פרק כה - משנה ט
אֵין לָהֶם אֲחוֹרַיִם וָתוֹךְ. דְּהַנּוֹגֵעַ בְּאֶחָד מֵהֶן כְּנוֹגֵעַ בְּכֻלּוֹ. וְיֵשׁ לָהֶן אֲחוֹרַיִם וָתוֹךְ דִּתְנַן לְעֵיל, דַּוְקָא לִתְרוּמָה: וְאֵין מַטְבִּילִין כֵּלִים לְתוֹךְ כֵּלִים. כְּשֶׁשְּׁנֵיהֶם טְמֵאִים, מִפְּנֵי שֶׁכָּבְדוֹ שֶׁל כְּלִי חוֹצֵץ. וְחָשׁוּ לְקֹדֶשׁ יוֹתֵר מִבַּתְּרוּמָה: יוֹרְדִים לִידֵי טֻמְאָתָן בְּמַחֲשָׁבָה. כְּגוֹן טַבַּעַת בְּהֵמָה שֶׁאֵינָהּ מְקַבֶּלֶת טֻמְאָה, חָשַׁב עָלֶיהָ לִתְּנָהּ בְּאֶצְבַּע אָדָם, מְקַבֶּלֶת טֻמְאָה מִכָּאן וָאֵילָךְ בְּמַחֲשָׁבָה זוֹ. אֲבָל טַבַּעַת אָדָם שֶׁחָשַׁב עָלֶיהָ לִתְּנָהּ לִבְהֵמָה, לְעוֹלָם הִיא מְקַבֶּלֶת טֻמְאָה, עַד שֶׁיַּעֲשֶׂה בָהּ שִׁנּוּי מַעֲשֶׂה לְשַׁנּוֹת צוּרָתָהּ: וּמַחֲשָׁבָה אֵינָהּ מְבַטֶּלֶת לֹא מִיַּד מַעֲשֶׂה וְלֹא מִיַּד מַחֲשָׁבָה. אִם לְאַחַר שֶׁחָשַׁב עַל טַבַּעַת בְּהֵמָה לַעֲשׂוֹתָהּ לָאָדָם, שֶׁהִיא מְקַבֶּלֶת טֻמְאָה בְמַחֲשָׁבָה זוֹ, חָזַר וְחָשַׁב עָלֶיהָ לְהַחְזִירָהּ לַבְּהֵמָה, אֵין מַחֲשָׁבָה זוֹ הַשְּׁנִיָּה מַעֲלָה מִידֵי טֻמְאָה שֶׁיָּרְדָה לָהּ בַּמַּחֲשָׁבָה הָרִאשׁוֹנָה:
