פרק טז

פרק טז - משנה א
כָּל כְּלִי עֵץ. שֻׁלְחָן הַכָּפוּל. עָשׂוּי חֻלְיוֹת אוֹ פְרָקִים, וּלְאַחַר שֶׁמְּסִירִים אוֹתוֹ כּוֹפְלִים חֻלְיוֹתָיו אוֹ פְרָקָיו זוֹ עַל זוֹ, וְכָל חֻלְיָא הַוְיָא כְּלִי בִּפְנֵי עַצְמָהּ: וְתַמְחוּי הַמִּזְנוֹן. קְעָרָה גְדוֹלָה מְאֹד וּבְתוֹכָהּ קְעָרוֹת קְטַנּוֹת, וְנוֹתְנִים בְּכָל קְעָרָה מִן הַקְּטַנּוֹת מִין אֶחָד שֶׁל תַּבְשִׁיל. וּלְפִי שֶׁיֶּשׁ בְּתוֹכָהּ מִינִים הַרְבֵּה, נִקְרָא תַמְחוּי הַמִּזְנוֹן. תַּרְגּוּם לְמִינוֹ לִזְנוֹהִי: אֱפִיפוֹרִין. כִּסֵּא שֶׁל פְּרָקִים: הַמָּגֵס. תַּרְגּוּם קְעָרָה מְגִיסְתָּא: וְקוֹד הַבַּבְלִי. כְּמִין קְעָרָה שֶׁל עֵץ. וּלְשׁוֹן חֲכָמִים הוּא, בִּמְקֵדָה בְזוּיָה, פֵּרוּשׁ קְעָרָה בְזוּיָה. וְכָל הַכֵּלִים הַלָּלוּ, אַף עַל פִּי שֶׁנֶּחְלְקוּ, נִשְׁאָר לְכָל חֵלֶק מֵהֶם צוּרַת כְּלִי. וְאֵין הֲלָכָה כְּרַבִּי יְהוּדָה: מִשֶּׁיְּשׁוּפֵם בְּעוֹר הַדָּג. לְהָסִיר קִסְמִין שֶׁבָּהֶם, דְּקֹדֶם לָכֵן לֹא חֲזוּ לִמְלַאכְתָּן, שֶׁמְּשָׂרְטִים אֶת הַבָּשָׂר. אֲבָל אִם אֵינָן מְחֻסָּרִין אֶלָּא שִׁיפָה שֶׁל צִחְצוּחַ וּמְרִיקָה, הָא קַיְמָא לָן דְּגָלְמֵי כְלֵי עֵץ שֶׁעָתִיד לְשַׁבֵּץ וְלָשׁוּף טְמֵאִין: גָּמַר שֶׁלֹּא לָשׁוּף. שֶׁגָּמַר בְּלִבּוֹ לְהִשְׁתַּמֵּשׁ בָּהֶם אַף עַל פִּי שֶׁלֹּא יָשׁוּף עוֹד : שְׁלֹשָׁה בָתִּים. כְּשֶׁמְּסָרְגִים הַמִּטָּה בַחֲבָלִים, בֵּין כָּל חֶבֶל לְחֶבֶל קָרוּי בַּיִת. וְאֵין הֲלָכָה כְּרַבִּי מֵאִיר:

פרק טז - משנה ב
הַסַּלִּים שֶׁל עֵץ. מֵאֵימָתַי גְּמַר מְלַאכְתָּן: מִשֶּׁיַּחְסֹם. כְּשֶׁאָדָם עוֹשֶׂה קֻפָּה אוֹ סַל וְגוֹמֵר אֶת שְׂפָתוֹ זוֹ הִיא חֲסִימָה: וִיקַנֵּב. לְאַחַר שֶׁגּוֹמֵר הַשָּׂפָה, וְנִשְׁתַּיְּרוּ קִסְמִין רָאשֵׁי קְרוּמִיּוֹת הַבּוֹלְטִין, פּוֹסְקָן וְקוֹטְמָן, וְזוֹ הִיא קְנִיבָה: שֶׁל תְּמָרָה. סַלִּים הָעֲשׂוּיִן מֵחֲרִיּוֹת שֶׁל דֶּקֶל: שֶׁכֵּן מְקַיְּמִין. בְּלֹא קְנִיבָה: כַּלְכָּלָה. סַל שֶׁל גֶּמִי: הַתְּלוֹיָה. שֶׁיִּגְמֹר הַחֶבֶל שֶׁתּוֹלֶה בָהּ: בֵּית הַלְּגִינִים. כְּלִי שֶׁמַּנִּיחִין בּוֹ לְגִינִין לִשְׁמֹר: בֵּית הַכּוֹסוֹת. כְּלִי שֶׁמַּצְנִיעִין בּוֹ כּוֹסוֹת:

פרק טז - משנה ג
הַקְּנוֹנִין. קָנוֹן שֶׁבּוֹרְרִים בּוֹ קִטְנִית. וְהוּא נַעֲשֶׂה מֵחֲרִיּוֹת שֶׁל דֶּקֶל: קְלָתוֹת. קֻפּוֹת קְטַנּוֹת: סוּגִים. קֻפּוֹת גְּדוֹלוֹת: שְׁנֵי דוּרִים לָרֹחַב שֶׁלָּהֶן. לְאַחַר שֶׁאָרְגוּ שׁוּלֵי הַקֻּפָּה, אוֹרְגִים סְבִיבוֹתֶיהָ עִגּוּלִים כְּפִי מַה שֶּׁרוֹצֶה שֶׁתִּהְיֶה רְחָבָה בְדָפְנוֹתֶיהָ בְּגָבְהָהּ, וְאוֹתָן עִגּוּלִים שֶׁל דְּפָנוֹת קְרוּיִין דּוּרִים: וְכַן שֶׁל מֹאזְנַיִם. הַבָּסִיס שֶׁנּוֹתְנִים עָלָיו הַמֹּאזְנַיִם : דּוּר אֶחָד. מִשֶּׁיַּגְבִּיהַּ עַל שְׂפָתוֹ עִגּוּל אֶחָד: שְׁתֵּי צְפִירוֹת. שְׁתֵּי סִבּוּבִים : וְהֶעָרָק. שֶׁאֵין שְׂפָתוֹ גְבוֹהָה כִּשְׂפַת הַקֻּפָּה: מִשֶּׁיַּעֲשֶׂה צְפִירָה אַחַת טָמֵא. וְעָרָק עָשׂוּי מִשַּׁעַם וְאָרְכּוֹ יוֹתֵר מֵרָחְבּוֹ, כְּתֵבָה שֶׁל רַצְעָנִים . וְקָרוּי עָרָק. וְעַד שְׂרוֹךְ נַעַל (בְּרֵאשִׁית יד), תַּרְגּוּם וְעַד עַרְקַת מְסָאנָא. וּמִשּׁוּם דִּכְלֵי גֶמִי וְשַׁעַם דִּין כְּלֵי עֵץ יֵשׁ לָהֶם, הִלְכָּךְ תְּנִינְהוּ גַבֵּי הֲדָדֵי עִם כְּלֵי עֵץ:

פרק טז - משנה ד
תּוּרְמֵל. כִּיס שֶׁל עוֹר גָּדוֹל, שֶׁהָרוֹעֶה נוֹתֵן בּוֹ מְזוֹנוֹת וְכָל דָּבָר. וְדוֹמֶה לוֹ, וְלֹא הָרוֹעֶה בְתַרְמִילוֹ: קִיחוֹתָיו. מִסְגְּרוֹתָיו , כְּמִין אָזְנַיִם קְטַנּוֹת סְבִיבוֹת הַתּוּרְמֵל, וּמַכְנִיסִים בְּאוֹתָן אָזְנַיִם חֲבָלִים שֶׁסּוֹגְרִים אוֹתוֹ: סְקֹרְטְיָא. עוֹר שֶׁלּוֹבְשִׁים הָעַבְּדָנִים בִּשְׁעַת מְלַאכְתָּן: צִיצָתָהּ. כְּעֵין צִיצִית שֶׁל טַלִּית, כָּךְ עוֹשִׂין לָהּ פְּתִילִים עַל כְּנָפַיִם לְקָשְׁרָם בָּהֶן. וְרַמְבַּ"ם פֵּרֵשׁ, סְקֹרְטְיָא עוֹר שֶׁאוֹכְלִים עָלָיו, וְצִיצָתָהּ, הוּא אֶמְצַע הָעוֹר שֶׁנּוֹתְנִים בּוֹ טַס שֶׁל מַתֶּכֶת לְנוֹי, לְשׁוֹן צִיץ הַזָּהָב, שֶׁהוּא טַס שֶׁל זָהָב: טַבְּעוֹתֶיהָ. טַבַּעַת נְחֹשֶׁת אוֹ בַּרְזֶל שֶׁנּוֹתְנִים בִּקְצֵה הַסְּקֹרְטְיָא: קָטָבוֹלְיָא. עוֹר שֶׁמַּצִּיעִים עַל גַּבֵּי מִטָּה . וְקוֹרִים לוֹ בַעֲרָבִי נט"ע, וּפְשׁוּטֵי כְלֵי עוֹר מְקַבְּלִים טֻמְאָה מִדְּרַבָּנָן בִּלְבַד, כִּפְשׁוּטֵי כְלֵי עֵץ, וְדִינָן בָּזֶה שָׁוֶה, שֶׁאֵין מְקַבְּלִים טֻמְאָה מִדְּאוֹרַיְתָא אֶלָּא מִדְּרַבָּנָן בְּכָל שְׁאָר טֻמְאוֹת, חוּץ מִמִּשְׁכָּב וּמוֹשָׁב, שֶׁהֵן נַעֲשִׂים מִשְׁכָּב וּמוֹשָׁב מִדְּאוֹרַיְתָא כְּשֶׁהֵן מְיֻחָדִים לְכָךְ: הַכַּר וְהַכֶּסֶת. כַּר קָטָן מִכֶּסֶת , וְעָשׂוּי לְהַנִּיחַ תַּחַת מְרַאֲשׁוֹתָיו. וְכֶסֶת תַּחַת כָּל גּוּפוֹ: פָּחוֹת מֵחֲמִשָּׁה טְפָחִים. מְשַׁיְּרִים מִשּׁוּם פֶּתַח לְהַכְנִיס מִשָּׁם הַמּוֹךְ לְמַלְּאוֹת הַכַּר וְהַכֶּסֶת:

פרק טז - משנה ה
פָּטִילְיָא. קֻפָּה עֲשׂוּיָה מִמִּינִים שֶׁל עֲרָבָה, וְאוֹצְרִים בָּהּ גְּרוֹגָרוֹת: וַחֲסִינָה. כְּמִין קֻפָּה עֲשׂוּיָה מִקּוֹצִים, וּמְשִׂימִין בָּהּ טִיט: סוּגְנִיּוֹת. לְשׁוֹן סוּגָה בַשּׁוֹשַׁנִּים, עוֹשִׂין כְּמִין כְּפִיפָה קְטַנָּה מִן הֶעָלִים וּמְכַסִּים בָּהּ פֵּרוֹת, וְאֵינָהּ אֶלָּא מְלֶאכֶת עֲרַאי: נְצָרִים. מִצּוּרֵי הַדֶּקֶל: חוֹתָל. כְּלִי שֶׁעוֹשִׂים מֵחֲרָיּוֹת שֶׁל דֶּקֶל, וּמַנִּיחִין בּוֹ תְּמָרִים לַחִים וְהֵן מִתְבַּשְּׁלִים בְּתוֹכוֹ : נוֹתֵן לְתוֹכוֹ וְנוֹטֵל מִתּוֹכוֹ. אֲבָל אִם אֵינוֹ יָכוֹל לְהוֹצִיא הַתְּמָרִים אֶלָּא אִם כֵּן יִקְרָעֶנּוּ אוֹ יַתִּיר הַחֲרִיּוֹת שֶׁלּוֹ, טָהוֹר, דְּכֵיוָן דְּקוֹרְעוֹ אוֹ מַתִּירוֹ מַשְׁלִיכוֹ לָאַשְׁפָּה, וְהָוֵי לְתַשְׁמִישׁ עֲרַאי:

פרק טז - משנה ו
קְסָיָה. בֵּית יָד שֶׁל עוֹר: שֶׁל זוֹרֵי גְרָנוֹת וְהוֹלְכֵי דְרָכִים וְשֶׁל עוֹשֵׂי פִשְׁתָּן טְמֵאָה. דְּכֵיוָן דְּהִיא עֲשׂוּיָה כְּדֵי לֶאֱחֹז הַדָּבָר בְּיָדוֹ יָפֶה שֶׁלֹּא יִשָּׁמֵט מִמֶּנּוּ, אוֹ כְדֵי שֶׁלֹּא יִתְחֹב לוֹ קוֹץ בְּיָדוֹ, מִקְּרֵי עָשׂוּי לְקַבָּלָה, כֵּיוָן שֶׁהַדָּבָר שֶׁהוּא אוֹחֵז בּוֹ נִשְׁעָן עַל הַבֵּית יָד וּמְקַבְּלוֹ: שֶׁל גָּרוֹסוֹת. הָעוֹשִׂים גְּרִיסוֹת שֶׁל פּוֹל בָּרֵחַיִם: מִפְּנֵי הַזֵּעָה. לִשְׁאֹב אֶת הַזֵּעָה כְּדֵי לְהָקֵר בְּשָׂרוֹ שֶׁלֹּא יַזִּיעַ. אִי נַמִּי, שֶׁלֹּא תְטַנֵּף זֵיעַת יָדָיו הַדָּבָר שֶׁהוּא אוֹחֵז בּוֹ: טָהוֹר. דְּתַשְׁמִישׁ עֲרַאי חֲשׁוּב:

פרק טז - משנה ז
מַלְקוֹט. עוֹר שֶׁמְּשִׂימִין בְּעֵינֵי הַפָּרָה כְּדֵי שֶׁתְּסַבֵּב וְתָדוּשׁ : הֶחָסוּם. כְּמִין מְצוֹדָה קְטַנָּה שֶׁל חֲבָלִים שֶׁחוֹסְמִים בָּהּ פִּי הַפָּרָה שֶׁלֹּא תֹאכַל: מַדָּף שֶׁל דְּבוֹרִים. כְּלִי שֶׁמְּשִׂימִים בּוֹ אֵשׁ וּגְלָלֵי בָקָר וּמְעַשְּׁנִים כְּדֵי שֶׁיִּבְרְחוּ הַנְּחִילִים שֶׁל דְּבוֹרִים מִן הַכַּוֶּרֶת וְיִקַּח הַדְּבַשׁ. מִלְּשׁוֹן אֲשֶׁר תִּדְפֶנּוּ רוּחַ (תְּהִלִּים א): וְהַמְּנָפָה. כְּלִי שֶׁמְּנִיפִין בּוֹ בִימוֹת הַחַמָּה לְהָבִיא הָרוּחַ: קֻפְסָא. כְּמִין אַרְגָּז קָטָן שֶׁהָאִשָּׁה נוֹתֶנֶת בּוֹ תַכְשִׁיטֶיהָ וְנוֹתֶנֶת עָלָיו כְּסוּת מִלְּמַעְלָה: קַמְטְרָא. תֵּבָה שֶׁשּׁוֹמְרִים בָּהּ הַבְּגָדִים. עַל הַמֶּלְתָּחָה (מְלָכִים ב י), מְתַרְגְּמִינַן עַל קוּמְטְרָה: וְהַמַּכְבֵּשׁ. לְפִי מְלַאכְתּוֹ יֵשׁ לוֹ מָקוֹם שֶׁכּוֹבֵשׁ וְעוֹצֵר הָעֵץ כְּשֶׁהוּא מְעֻוָּת וְעָקֹם עַד שֶׁמְּיַשְּׁרוֹ: וְהַקִּמְרוֹן. הַמִּכְסֶה שֶׁעַל הַתֵּבָה הֶעָשׂוּי כְּעֵין קֻבָּה : אַנְגְּלִין. נַרְתִּיק שֶׁל עוֹר שֶׁעוֹשִׂין לַסֵּפֶר וּמַכְנִיסִים אוֹתוֹ בְתוֹכוֹ: בֵּית הַנֶּגֶר. נַרְתִּיק שֶׁל עוֹר שֶׁנּוֹתְנִים בְּתוֹכוֹ הַנֶּגֶר, הוּא הַיָּתֵד שֶׁמְּשִׂימִים אֲחוֹרֵי הַדֶּלֶת: וּבֵית הַמַּנְעוּל. נַרְתִּיק שֶׁל עוֹר שֶׁנּוֹתְנִים בּוֹ הַמַּנְעוּל שֶׁנּוֹעֲלִים בּוֹ הַדֶּלֶת: וּבֵית הַמְּזוּזָה. קָנֶה שֶׁל מַתָּכוֹת שֶׁנּוֹתְנִים בּוֹ הַמְּזוּזָה: וְהָאֵמוּם שֶׁל גּוֹדְלֵי מִצְנָפוֹת. דְּפוּס כְּעֵין רֹאשׁ שֶׁל אָדָם שֶׁמְּתַקְּנִים עָלָיו הַמִּצְנֶפֶת: וְהַמַּרְכּוֹף. כְּמִין סוּס שֶׁל עֵץ שֶׁהַלֵּיצָנִים מְשַׂחֲקִים בּוֹ. וְאִית דִּמְפָרְשֵׁי, כְּלִי שֶׁל עֵץ אֶרֶז הֶעָשׂוּי לְזַמֵּר: וּרְבִיעִית שֶׁל אַלָּיִת. וּבִמְנַעַנְעִים וּבִמְצַלְצְלִין, מְתַרְגְּמִינַן וּבִרְבִיעִין וּבְצֶלְצֱלִין: שֶׁל אַלָּיִת. שֶׁל הַמְקוֹנֶנֶת. כְּמוֹ (יוֹאֵל א) אֱלִי כִבְתוּלָה חֲגוּרַת שַׂק: גְּנוֹגְנִית הֶעָנִי. תַּרְמִילוֹ שֶׁל עָנִי: וְסָמוֹכוֹת הַמִּטָּה. קוֹרוֹת שֶׁסּוֹמְכִין בָּהֶן הַמִּטָּה שֶׁלֹּא תִפֹּל: טְפוּס שֶׁל תְּפִלִּין. כְּמוֹ דְפוּס שֶׁל תְּפִלִּין: אֵמוּם שֶׁל עוֹשֵׂי סַאגּוֹת. דְּפוּס שֶׁל עוֹשֵׂי שַׁלְשְׁלָאוֹת. שֶׁעוֹשִׂין אוֹתָן עַל גַּבֵּי אֵמוּם הַיְנוּ דְּפוּס. וְרַמְבַּ"ם גָּרֵיס, עוֹשֵׂי סוּתוֹת, לְשׁוֹן וּבְדַם עֲנָבִים סוּתוֹ (בְּרֵאשִׁית מט), וְהוּא הַדְּפוּס שֶׁל עֵץ שֶׁאוֹרְגִין עָלָיו הַבְּגָדִים: מְשַׁמְּשֵׁי מְשַׁמְּשָׁיו שֶׁל אָדָם. כְּגוֹן כִּסּוּיֵי כֵלִים וְנַרְתִּיקֵי כֵלִים, שֶׁהֵם נַעֲשׂוּ לְשַׁמֵּשׁ הַכֵּלִים הַמְשַׁמְּשִׁים אֶת הָאָדָם: בִּשְׁעַת מְלָאכָה וְשֶׁלֹּא בִשְׁעַת מְלָאכָה. בִּזְמַן שֶׁהַכִּסּוּי מְשַׁמֵּשׁ אֶת הַכְּלִי לִשְׁמֹר מַה שֶּׁבְּתוֹכוֹ, וְהַיְנוּ שְׁעַת מְלָאכָה דִידֵיהּ, וּכְשֶׁאֵין כְּלוּם בַּכְּלִי שֶׁאֵינוֹ צָרִיךְ לְכִסּוּי קָרֵי לֵיהּ שֶׁלֹּא בִשְׁעַת מְלָאכָה. וְיֵשׁ כֵּלִים שֶׁמַּנִּיחִים עֲלֵיהֶן הַכִּסּוּי תָּמִיד בֵּין כְּשֶׁהוּא מָלֵא בֵּין כְּשֶׁהוּא רֵיקָן, אַף עַל פִּי שֶׁאֵין הַכִּסּוּי מְחֻבָּר בּוֹ, חֲשִׁיב כַּכְּלִי גוּפֵיהּ לְקַבֵּל טֻמְאָה אֲפִלּוּ בִּזְמַן שֶׁאֵינוֹ מְחֻבָּר בּוֹ, כֵּיוָן שֶׁרָגִיל לִהְיוֹת תָּמִיד עָלָיו. וְהַאי כְּלָלָא דְרַבִּי יוֹסֵי, אַפְּשׁוּטֵי כְלֵי עֵץ הַמְשַׁמְּשִׁין אֶת הַכֵּלִים קָאֵי. אֲבָל כְּלֵי קִבּוּל, מִטַּמְּאִין בְּכָל עִנְיָן. וַהֲלָכָה כְּרַבִּי יְהוּדָה:
