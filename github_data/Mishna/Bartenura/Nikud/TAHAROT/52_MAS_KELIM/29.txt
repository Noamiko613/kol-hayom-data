פרק כט

פרק כט - משנה א
נוֹמֵי הַסָּדִין. נִימִין הַיּוֹצְאִים מִן הַסָּדִין, וְהֵן חוּטִים שֶׁרְגִילִים לָצֵאת מִן הָאָרִיג בֵּין מִן הַצְּדָדִים בֵּין מִלְּמַעְלָה בֵּין מִלְּמַטָּה. אִם נָגַע שֶׁרֶץ בַּנִּימִין, נִטְמָא הַסָּדִין. וְהוּא שֶׁיִּגַּע בְּתוֹךְ הַשִּׁעוּר הַמְפֹרָשׁ בְּמַתְנִיתִין. וְכֵן אִם נָגַע שֶׁרֶץ בַּסָּדִין, נִטְמְאוּ הַנִּימִין עַד מְקוֹם הַשִּׁעוּר הַמְפֹרָשׁ. וּמִשָּׁם וְלַחוּץ טְהוֹרִים, שֶׁאֵין הַנִּימִין חִבּוּר לְכָל בֶּגֶד אֶלָּא עַד מְקוֹם הַשִּׁעוּר הַמְפֹרָשׁ בָּהֶן: טַרְטִין. כְּמִין כּוֹבָעִים: פִּלְיוֹן. סָדִין שֶׁמִּתְעַטֵּף בּוֹ כֻלּוֹ : אַפְקַרְסִין. כְּמוֹ אַפְקַרְסוּתוֹ אֵינָהּ מְעַכֶּבֶת דְּפֶרֶק וְאֵלּוּ מְגַלְּחִים , וְהוּא לְבוּשׁ הַתַּחְתּוֹן וּפָתוּחַ מִכְּתֵפָיו וּכְשֶׁלּוֹבְשׁוֹ קוֹשְׁרוֹ: סָגוֹס. בֶּגֶד שֶׁל צֶמֶר עָב וְכֻלּוֹ צִיצִיּוֹת, וְקוֹרִים לוֹ בַעֲרָבִי אלבורנו"ס: רְדִיד. צָעִיף דַּק שֶׁהָאִשָּׁה מְכַסָּה בוֹ רֹאשָׁהּ. וַתְּכַס בַּצָּעִיף, תַּרְגּוּם יְרוּשַׁלְמִי רְדִידָא: גּוּמְדִין. בֶּגֶד שֶׁהוּא אַמָּה עַל אַמָּה, וּמְכַסִּין הָעַרְבִים חָטְמָן וּפִיהֶם בִּזְמַן הַקֹּר. וּפֵרוּשׁ גֹּמֶד אַמָּה, כְּדִכְתִיב (שׁוֹפְטִים ג) וְלָהּ שְׁנֵי פֵיוֹת גֹּמֶד אָרְכָּהּ: קֻלְקִין. אָרִיג שֶׁל שֵׂעָר וְיֵשׁ לוֹ נִימִין: פֻּנְדָּא. יֵשׁ מְפָרְשִׁים אֵזוֹר חָלוּל וְרָחָב. וְיֵשׁ מְפָרְשִׁים הַבֶּגֶד הַתַּחְתּוֹן שֶׁנּוֹתְנִים עַל הַבָּשָׂר לִשְׁמֹר שְׁאָר הַבְּגָדִים שֶׁלֹּא יִטַּנְּפוּ בַזֵּעָה: מַעֲפֹרֶת. תַּרְגּוּם וַיִּתְחַפֵּשׂ בַּאֲפֵר (מְלָכִים א כ) וְאֶשְׁתַּנֵי בְּמַעֲפָרָא. סוּדָר שֶׁמְּעַטְּפִים בּוֹ הָרֹאשׁ: פַּרְגּוֹד. מָסָךְ שֶׁתּוֹלִין עַל פִּתְחֵי הַמְּלָכִים: כָּל שֶׁהֵן. בֵּין רַב בֵּין מְעַט הָוֵי חִבּוּר. וְכֻלְּהוּ אַחֲרִינֵי שֶׁשִּׁעוּרָן מְפֹרָשׁ, כְּשִׁעוּר הָוֵי חִבּוּר, מַה שֶּׁיָּתֵר מִכְּשִׁעוּר לֹא הָוֵי חִבּוּר, לֹא לַטֻּמְאָה וְלֹא לַהַזָּאָה:

פרק כט - משנה ב
שָׁלֹשׁ כְּסָתוֹת שֶׁל צֶמֶר. תְּפוּרוֹת זוֹ עִם זוֹ כְּשָׁלָל שֶׁל כּוֹבְסִין. אִי נַמִּי, אֲרוּגוֹת יַחַד, כְּעֵין הָאוֹרְגִים מַפּוֹת שֶׁדַּרְכָּן לְהָנִיחַ הֶפְרֵשׁ בֵּין מַפָּה לְמַפָּה וּמְחֻבָּרוֹת בְּחוּטֵי הַשְּׁתִי וְאֵין בּוֹ עֵרֶב. וְכָל הָנֵי דִתְנַן הָכָא, כָּל אֶחָד בְּמִנְיָנוֹ אוֹ פָּחוֹת הָוֵי חִבּוּר, וְאִם נָגְעָה טֻמְאָה בְאֶחָד מֵהֶן נִטְמְאוּ כֻלָּן, וְאִם הָיוּ טְמֵאִין וְהוּזָה מֵאֵפֶר הַפָּרָה בְאֶחָד מֵהֶן נִטְהֲרוּ כֻלָּן. וְיוֹתֵר עַל מִנְיָן זֶה לֹא הָוֵי חִבּוּר לֹא לַטֻּמְאָה וְלֹא לַהַזָּאָה : סוּבְרִיקִין. אִית דְּתָנֵי סַפְרִיקִין. אִית דִּמְפָרְשֵׁי מִכְנָסַיִם, וְאִית דִּמְפָרְשֵׁי כְּמִין בָּתֵּי יָדַיִם מִצֶּמֶר שֶׁמַּלְבִּישִׁים בָּהֶם אֶת הַיָּדַיִם וּמְכַסִּים בָּהֶם הַזְּרוֹעוֹת עַד הָאֲצִילִים: גְּלֻפְקָרִין. כְּסוּת עָבָה שֶׁל צֶמֶר שֶׁלּוֹבְשִׁים בְּעֵת הַקְּרִירוּת: חִבּוּר לַטֻּמְאָה וְאֵינוֹ חִבּוּר לַהַזָּאָה. לְחֻמְרָא. אִם נָגְעָה טֻמְאָה בְּאֶחָד מֵהֶן הָוֵי חִבּוּר וְנִטְמְאוּ כֻלָּן, וְאִם הוּזָה אֶחָד מֵהֶן לֹא הָוֵי חִבּוּר, וְלֹא טָהַר אֶלָּא אוֹתוֹ שֶׁהוּזָה עָלָיו בִּלְבַד: רַבִּי יוֹסֵי אוֹמֵר אַף לֹא לַטֻּמְאָה. וְאֵין הֲלָכָה כְּרַבִּי יוֹסֵי:

פרק כט - משנה ג
חוּט הַמִּשְׁקֹלֶת. הַבַּנָּאִים מַחֲזִיקִים בְּיָדָם חוּט וּבְרֹאשׁוֹ עוֹפֶרֶת וּמוֹרִידִים אוֹתוֹ כְנֶגֶד הַכֹּתֶל לִרְאוֹת שֶׁלֹּא יְהֵא עָקוּם: חָרָשִׁין. נַגָּרִין שֶׁל עֵץ: וְשֶׁל בַּנָּאִים. הַבּוֹנִים הַחוֹמוֹת הַגְּבוֹהוֹת וְהַגְּדוֹלוֹת. וְחוּט הַמִּשְׁקֹלֶת דְּרֵישָׁא דְּלֹא הָוֵי אֶלָּא שְׁנֵים עָשָׂר, מַיְרֵי בִמְיֻחָד לְבִנְיָנִים קְטַנִּים: יָתֵר מִכֵּן. יוֹתֵר מִמִּדּוֹת אֵלּוּ שֶׁמָּנוּ חֲכָמִים בְּכָל אֶחָד וְאֶחָד, אַף עַל פִּי שֶׁהוּא רוֹצֶה בְקִיּוּמָן , טָהוֹר אוֹתוֹ הָעוֹדֵף עַל הַמִּדָּה וְלֹא הָוֵי חִבּוּר לִכְלִי: וְשֶׁל סַיָּדִים. הַטָּחִים פְּנֵי הַכֹּתֶל בְּסִיד וְגַם הֵם צְרִיכִים לְמִשְׁקֹלֶת, וְכֵן הַצַּיָּרִים שֶׁמְּצַיְּרִים צוּרָה בַכְּתָלִים: כָּל שֶׁהֵן. בֵּין רַב בֵּין מְעַט הָוֵי חִבּוּר:

פרק כט - משנה ד
חוּט מֹאזְנַיִם. חוּט שֶׁאוֹחֲזִים בּוֹ הַמֹּאזְנַיִם בְּשָׁעָה שֶׁשּׁוֹקְלִים בָּהֶן: רַבִּי יוֹסֵי אוֹמֵר טֶפַח טָהוֹר. אִם יַד הַקֻּרְדֹּם גָּדוֹל כָּל כָּךְ שֶׁיּוֹצֵא טֶפַח לְאַחַר יָדוֹ, לָא חֲשִׁיב יָד, לְפִי שֶׁמַּזִּיק לוֹ לִמְלַאכְתּוֹ, וְכָל הַיָּד כֻּלּוֹ טָהוֹר. כָּךְ פֵּרֵשׁ מוֹרִי. וְאֵין הֲלָכָה כְּרַבִּי יוֹסֵי:

פרק כט - משנה ה
יַד הַקֻּרְדֹּם מִלְּפָנָיו. דֶּרֶךְ הָאוֹחֵז בְּיַד הַקֻּרְדֹּם כְּשֶׁרוֹצֶה לַחְתֹּךְ בּוֹ אֵינוֹ אוֹחֵז מַמָּשׁ בְּרֹאשׁ הַקַּתָּא שֶׁמָּא יַחְלִיק וְיִפֹּל מִיָּדוֹ אֶלָּא מַנִּיחַ מְעַט לְאַחַר יָדוֹ, וְהַשִּׁעוּר שֶׁיֶּשׁ בֵּין יַד הָאָדָם לַבַּרְזֶל קוֹרֵא מִלְּפָנָיו, וְהַיּוֹצֵא לְאַחַר יָדוֹ קוֹרֵא מֵאַחֲרָיו בְּאִידָךְ בָּבָא. וְהָא דְלָא מְפָרֵשׁ בְּבָבָא זוֹ דִין שֶׁלְּפָנָיו וְשֶׁל אַחֲרָיו, מִשּׁוּם דְּהָכָא חֲשִׁיב הָנָךְ דְּשִׁעוּרָן טֶפַח, וּבְאִידָךְ בָּבָא חֲשִׁיב הַנְהוּ דְשִׁעוּרָן שָׁלֹשׁ אֶצְבָּעוֹת: פַּרְגּוֹל. מְחוֹגָה. קומפש"ו בְּלַעַ"ז. וְהֶחָרָשִׁים צְרִיכִים לַעֲשׂוֹת בּוֹ צוּרוֹת עֲגֻלּוֹת: מַקֶּבֶת. פַּטִּישׁ שֶׁל בַּרְזֶל שֶׁרֹאשׁוֹ אֶחָד חַד וּבוֹ נוֹקְבִים וּפוֹסְלִים אֶת הָאֲבָנִים, וְעַל שֵׁם כָּךְ נִקְרָא מַקֶּבֶת. וּלְשׁוֹן מִקְרָא הוּא, וּמַקָּבוֹת וְהַגַּרְזֶן כָּל כְּלִי בַרְזֶל (מְלָכִים א ו): מְפַתְּחֵי אֲבָנִים. לְשׁוֹן פִּתּוּחֵי חוֹתָם:

פרק כט - משנה ו
שֶׁל צַמָּרִים. מוֹכְרֵי צֶמֶר. וְטַעֲמָא דְהָנָךְ שִׁעוּרָן בִּטְפָחַיִם, מִפְּנֵי שֶׁתּוֹפֵס בָּהֶן בִּשְׁתֵּי יָדָיו. וְהַיְנוּ נַמִּי טַעֲמָא דְכָל הָנָךְ יָדוֹת דִּלְקַמָּן שֶׁשִּׁעוּרָן בְּטִפְחָיִם. וְכָל אוֹתָן שֶׁשִּׁעוּרָן טֶפַח, לְפִי שֶׁדַּרְכּוֹ לֶאֱחֹז בָּהֶן בְּיָדוֹ אַחַת: מַקּוֹר. קֻרְנָס חַד שֶׁמְּנַקְּרִים בּוֹ הָרֵחַיִם. וְעַל שֵׁם שֶׁמְּנַקְּרִים בּוֹ קָרוּי מַקּוֹר: יַד הַמַּעֲצָד שֶׁל לִגְיוֹנוֹת. גָּרְסִינַן. וְהוּא כְלִי מִכְּלֵי הָרוֹצְחִים , רֹאשׁוֹ עָגֹל, וְנוֹשְׂאִים אוֹתוֹ אַנְשֵׁי הַחַיִל בַּמִּלְחָמָה. וּבְעַרְבִי קוֹרִים לוֹ טבריזי"ן: שֶׁל זֶהָבִים. הַמַּכִּים עַל זָהָב לַעֲשׂוֹת טַסִּין דַּקִּין: וְשֶׁל חָרָשִׁים. שֶׁל בַּרְזֶל:

פרק כט - משנה ז
שְׁיָרֵי הַדָּרְבָן מִלְמַעְלָן. הַמַּרְדֵּעַ רֹאשׁוֹ אֶחָד יֵשׁ בּוֹ בַרְזֶל כְּעֵין מַרְצֵעַ שֶׁבּוֹ מְלַמֵּד הַפָּרָה לִתְלָמֶיהָ וְהוּא קָרוּי דָּרְבָן, וּבְרֹאשׁוֹ הַשֵּׁנִי בַּרְזֶל רָחָב מְחֻדָּד לַחְתֹּךְ בּוֹ שָׁרָשִׁים וּשְׁמוֹ חַרְחוּר. וְשִׁעוּר יַד הַמַּרְדֵּעַ, לְמַעְלָה לְצַד הַדָּרְבָן אַרְבָּעָה טְפָחִים חֲשׁוּבִים חִבּוּר, וּלְמַטָּה לְצַד הַחַרְחוּר שִׁבְעָה טְפָחִים הַסְּמוּכִים לוֹ הָווּ חִבּוּר, כְּדִלְקַמָּן. וּמַה שֶּׁהוּא יוֹתֵר מֵאַרְבָּעָה לְצַד הַדָּרְבָן וְשִׁבְעָה לְצַד הַחַרְחוּר אֵינוֹ חִבּוּר, וְאִם נָגַע שָׁם טֻמְאָה לֹא נִטְמָא לֹא הַחַרְחוּר וְלֹא הַדָּרְבָן: בָּדִיד. כְּלִי בַרְזֶל שֶׁבּוֹ חוֹפְרִים גֻּמּוֹת סְבִיב הָאִילָן כְּדֵי שֶׁיַּחֲזִיקוּ מֵימֵיהֶן. וְעַל שֵׁם הַגֻּמּוֹת שֶׁקְּרוּיִין בְּדִידִין נִקְרָא הַכְּלִי הַחוֹפֵר בָּהֶן בָּדִיד: שֶׁל נִכּוּשׁ. פַּטִּישׁ שֶׁמַּקִּישִׁים בּוֹ עַל הָאֲבָנִים וּמְשַׁבְּרִים אוֹתָן : בֶּן פַּטִּישׁ. הַפַּטִּישׁ הַקָּטָן קָרוּי בֶּן פַּטִּישׁ, וְהַגָּדוֹל שְׁמוֹ פַּטִּישׁ: שֶׁל בִּקּוּעַ. קֻרְדֹּם שֶׁמְּבַקְּעִים בּוֹ עֵצִים: שֶׁל עָדִיר. שֶׁחוֹפְרִים בּוֹ הַקַּרְקַע לִזְרִיעָה. כְּמוֹ אֲשֶׁר בַּמַּעְדֵּר יֵעָדֵרוּן (יְשַׁעְיָה ז): שֶׁל סַתָּתִים. חוֹצְבֵי אֲבָנִים וְהַמְקַצְּעִים וּמַשְׁוִין אוֹתָן:

פרק כט - משנה ח
מַגְרֵפָה. שֶׁגּוֹרְפִים בָּהּ תַּנּוּר וְכִירַיִם אוֹ הַטִּיט. תַּרְגּוּם יָעִים, מַגְרוֹפְיָתָא: וְשֶׁל סַיָּדִים. שֶׁגּוֹרְפִים בָּהּ אֶת הַסִּיד: יָתֵר מִכָּאן. יוֹתֵר מִמִּדּוֹת אֵלּוּ שֶׁמָּנוּ חֲכָמִים בְּיַד הַמַּגְרֵפָה, אִם רוֹצֶה בְקִיּוּם זֶה הָאֹרֶךְ הַיָּתֵר: טָמֵא. אִם נִטְמָא הַכְּלִי. וְכֵן אִם נָגְעָה טֻמְאָה בְּכָל זֶה הָאֹרֶךְ שֶׁל יָד, נִטְמָא הַכְּלִי עִמּוֹ, דַּהֲוֵי חִבּוּר לַכְּלִי: יַד מְשַׁמְּשֵׁי הָאוּר. כְּגוֹן יָד שֶׁל שַׁפּוּד וְאַסְכָּלָה וּמַרְחֶשֶׁת וּשְׁאָר כֵּלִים שֶׁתַּשְׁמִישָׁן עַל יְדֵי הָאוּר:
