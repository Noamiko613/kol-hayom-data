פרק א - משנה א
שְׁלֹשָׁה עָשָׂר דָּבָר. צְרִיכָה מַחֲשָׁבָה. שֶׁיַּחֲשֹׁב עָלֶיהָ לַאֲכִילַת אָדָם. שֶׁכָּל הָאֳכָלִים הַמֻּתָּרִים אֵין צְרִיכִים מַחֲשָׁבָה, אֲבָל זוֹ הוֹאִיל וְהִיא אֲסוּרָה בַּאֲכִילָה צְרִיכָה מַחֲשָׁבָה וְאַחַר כָּךְ תְּטַמֵּא. וְדַוְקָא בַּכְּפָרִים דְּלֵיכָּא רֹב עַם, אֲבָל בַּשְּׁוָקִים אֵינָהּ צְרִיכָה מַחֲשָׁבָה, שֶׁכֵּיוָן שֶׁיֵּשׁ רֹב עַם הַבָּאִים לַשּׁוּק אִיכָּא טוּבָא דְּאָכְלֵי נִבְלַת עוֹף טָהוֹר. וְהָכִי מְפֹרָשׁ בְּפֶרֶק בַּתְרָא דְּעֳקָצִין: אֵינָהּ צְרִיכָה הֶכְשֵׁר. אֶלָּא מְטַמְּאָה טֻמְאַת אֳכָלִים כְּכָל שְׁאָר אֳכָלִים שֶׁנָּגְעוּ בְּשֶׁרֶץ שֶׁהֵם נַעֲשִׂים רִאשׁוֹן לַטֻּמְאָה. וְאֵין צָרִיךְ שֶׁיָּבֹאוּ עֲלֵיהֶם מַיִם אוֹ שְׁאָר מַשְׁקִים וְלֹא שֶׁתִּגַּע בְּשֶׁרֶץ. דְּהֶכְשֵׁר בִּזְרָעִים הוּא דִּכְתִיב (וַיִּקְרָא יא) וְכִי יֻתַּן מַיִם עַל זֶרַע וְגוֹ', וְדָרְשִׁינַן מַה זְּרָעִים שֶׁאֵין סוֹפָן לִטָּמֵא טֻמְאָה חֲמוּרָה צָרִיךְ הֶכְשֵׁר, אַף כָּל שֶׁאֵין סוֹפוֹ לִטָּמֵא טֻמְאָה חֲמוּרָה צָרִיךְ הֶכְשֵׁר, יָצְאָה נִבְלַת עוֹף טָהוֹר שֶׁסּוֹפָהּ לִטָּמֵא טֻמְאָה חֲמוּרָה, שֶׁהִיא מְטַמְּאָה אָדָם וּבְגָדִים בְּבֵית הַבְּלִיעָה, שֶׁאֵינָהּ צְרִיכָה הֶכְשֵׁר: וּמְטַמְּאָה טֻמְאַת אֳכָלִין בְּכַבֵּיצָה. דְּכַבֵּיצָה מִמֶּנָּה אִם נָגְעָה בְּאֳכָלִין טְהוֹרִים נִטְמְאוּ : וְכַזַּיִת בְּבֵית הַבְּלִיעָה. אָדָם שֶׁאָכַל כַּזַּיִת מִמֶּנָּה נִטְמָא, וְנִטְמְאוּ בְּגָדָיו כְּשֶׁתַּגִּיעַ לְבֵית הַבְּלִיעָה. דִּכְתִיב (שָׁם יז) וְהַנֶּפֶשׁ אֲשֶׁר תֹּאכַל נְבֵלָה וּטְרֵפָה וְגוֹ', וּבְנִבְלַת עוֹף טָהוֹר הַכָּתוּב מְדַבֵּר, וּכְתִיב תֹּאכַל, וְאֵין אֲכִילָה פָּחוֹת מִכַּזַּיִת: וְטָעוּן הֶעֱרֵב שֶׁמֶשׁ. וְלֹא סַגִּי בִּטְבִילָה לְחוֹדַהּ : וְחַיָּבִים עָלֶיהָ עַל בִּיאַת הַמִּקְדָּשׁ. כִּשְׁאָר כָּל הַטֻּמְאוֹת. דִּכְתִיב בְּטֻמְאַת מִקְדָּשׁ וְקָדָשָׁיו (וַיִּקְרָא ה) לְכֹל טֻמְאָתוֹ אֲשֶׁר יִטְמָא בָּהּ, וְדָרְשִׁינַן לְרַבּוֹת בּוֹלֵעַ נִבְלַת עוֹף טָהוֹר: וְשׂוֹרְפִין עָלֶיהָ אֶת הַתְּרוּמָה. אִם נָגְעָה נִבְלַת עוֹף טָהוֹר בַּתְּרוּמָה, נִשְׂרֶפֶת, שֶׁכְּשֶׁאֵינָהּ בְּבֵית הַבְּלִיעָה דִּין רִאשׁוֹן לַטֻּמְאָה יֵשׁ לָהּ וּמְטַמְּאָה אֶת הַתְּרוּמָה בִּנְגִיעָה לִהְיוֹת שֵׁנִי: וְהָאוֹכֵל אֵבָר מִן הַחַי מִמֶּנָּה סוֹפֵג אֶת הָאַרְבָּעִים. וַאֲפִלּוּ הָיָה הָאֵבָר פָּחוֹת מִכַּזַּיִת . וּסְתָם מַתְנִיתִין דְּלֹא כְּרַבִּי מֵאִיר, דְּאִלּוּ רַבִּי מֵאִיר הָא שָׁמְעִינַן לֵיהּ בְּשִׁלְהֵי גִּיד הַנָּשֶׁה דְּאָמַר אֵבָר מִן הַחַי אֵינוֹ נוֹהֵג אֶלָּא בִּבְהֵמָה טְהוֹרָה בִּלְבַד: שְׁחִיטָתָהּ וּמְלִיקָתָהּ מְטַהֲרוֹת אֶת טְרֵפָתָהּ. וּמוֹצִיאָהּ מִידֵי נְבֵלָה שֶׁלֹּא תְטַמֵּא. לְפִי שֶׁהַמְּלִיקָה מַתֶּרֶת בִּפְנִים לַכֹּהֲנִים כְּדֶרֶךְ שֶׁהַשְּׁחִיטָה מַתֶּרֶת בַּחוּץ, וּכְשֵׁם שֶׁטִּהֲרַתּוֹ שְׁחִיטָתוֹ, כָּךְ תְּטַהֲרֶנּוּ מְלִיקָתוֹ. וּשְׁחִיטָה דִּמְטַהֶרֶת אֶת הַטְּרֵפָה מִידֵי נְבֵלָה נַפְקָא לָן מִקְּרָא, דִּכְתִיב (וַיִּקְרָא יא) וְכִי יָמוּת מִן הַבְּהֵמָה וְגוֹ' כָּל הַנֹּגֵעַ בְּנִבְלָתָהּ יִטְמָא, יֵשׁ מִן הַבְּהֵמָה שֶׁהִיא מְטַמְּאָה, וְיֵשׁ שֶׁאֵינָהּ מְטַמְּאָה, פְּרָט לִטְרֵפָה שֶׁנִּשְׁחֲטָה, וְהוּא הַדִּין לִטְרֵפַת עוֹף: רַבִּי יְהוּדָה אוֹמֵר אֵינָן מְטַהֲרוֹת. דְּלֹא יָלְפִינַן טְרֵפַת עוֹף מִטְּרֵפַת בְּהֵמָה: רַבִּי יוֹסֵי אוֹמֵר שְׁחִיטָתָהּ מְטַהֶרֶת וְלֹא מְלִיקָתָהּ. וַהֲלָכָה כְּרַבִּי יוֹסֵי:

פרק א - משנה ב
מִטַּמְּאוֹת. אִם נָגַע בָּהֶן שֶׁרֶץ נִטְמְאוּ . וְאִם טְמֵאִים הֵן וְנָגְעוּ הַכְּנָפַיִם וְהַנּוֹצָה בְּאֳכָלִין טְהוֹרִים, נִטְמְאוּ, דַּחֲשִׁיבֵי יָד לְהַכְנִיס וּלְהוֹצִיא, וְכָל שֶׁהוּא יָד מִטַּמֵּא וּמְטַמֵּא: וְלֹא מִצְטָרְפוֹת. אִם הָיָה אֹכֶל פָּחוֹת מִכַּבֵּיצָה אֵין הַכְּנָפַיִם וְהַנּוֹצָה מִצְטָרְפוֹת לְהַשְׁלִימוֹ לְכַבֵּיצָה לְטַמֵּא טֻמְאַת אֳכָלִים, דְּלָא חֲשִׁיבֵי שׁוֹמֵר שֶׁמִּצְטָרֵף, כְּדִתְנַן בְּרֵישׁ עֳקָצִין: הַנּוֹצָה מִצְטָרֶפֶת. דַּחֲשִׁיב לַהּ שׁוֹמֵר. וְאֵין הֲלָכָה כְּרַבִּי יִשְׁמָעֵאל: הַחַרְטוֹם. שֶׁהָעוֹף מְנַקֵּר בּוֹ: רָאשֵׁי אֲגַפַּיִם וְרֹאשׁ הַזָּנָב. הַנִּשְׁאָרִים מְחֻבָּרִים לַגּוּף אַחַר שֶׁמָּרְטוּ הַנּוֹצָה, רְגִילִים לְחָתְּכָם וּלְהַשְׁלִיכָם, דְּלֹא חֲזוּ לַאֲכִילָה, אֶלָּא שֶׁבָּעוֹפוֹת הַמְפֻטָּמִים מַנִּיחִים אוֹתָן וְאוֹכְלִים אוֹתָן עִם הָעוֹף, וְהַיְנוּ דְּקָאָמַר שֶׁכֵּן מַנִּיחִים בַּפְּטוּמוֹת. וְאֵין הֲלָכָה כְּרַבִּי יוֹסֵי. וְהָנֵי שְׁלֹשָׁה עָשָׂר דָּבָר דַּחֲשִׁיב בְּמַתְנִיתִין, בְּבָבָא קַמַּיְתָא אִיכָּא תִּשְׁעָה, דִּשְׁחִיטָתָהּ וּמְלִיקָתָהּ חֲדָא חֲשִׁיב לְהוּ, דְּתַרְוַיְהוּ הֲווּ טָהֳרָה לִטְרֵפָתָהּ, וְהַכְּנָפַיִם, וְהַנּוֹצָה, וְהַחַרְטוֹם, וְהַצִּפָּרְנַיִם, הֲרֵי שְׁלֹשָׁה עָשָׂר דָּבָר:

נִבְלַת עוֹף טָמֵא. שֶׁאֵין סוֹפָהּ לְטַמֵּא טֻמְאָה חֲמוּרָה, שֶׁאֵינָהּ מְטַמְּאָה אָדָם וּבְגָדִים: צְרִיכָה מַחֲשָׁבָה. שֶׁיַּחֲשֹׁב עָלֶיהָ לַאֲכִילָה: וְהֶכְשֵׁר. מַיִם אוֹ שְׁאָר מַשְׁקִין, וְאַחַר כָּךְ תִּגַּע בְּשֶׁרֶץ: לִפְסֹל אֶת הַגְּוִיָּה. אִם נִטְמֵאת וְאָכַל מִמֶּנָּה חֲצִי פְּרָס שֶׁהִיא בֵּיצָה וּמֶחֱצָה , נִפְסַל גּוּפוֹ לֶאֱכֹל בַּתְּרוּמָה עַד שֶׁיִּטְבֹּל: וְאֵין בָּהּ כַּזַּיִת בְּבֵית הַבְּלִיעָה. דִּכְתִיב (שָׁם כב) נְבֵלָה וּטְרֵפָה לֹא יֹאכַל לְטָמְאָה בָהּ, מִי שֶׁאִסּוּרוֹ מִשּׁוּם אוֹכֵל נְבֵלָה מְטַמֵּא בַּאֲכִילָה בְּבֵית הַבְּלִיעָה, יָצָא זֶה שֶׁאֵין אִסּוּרוֹ מִשּׁוּם בַּל תֹּאכַל נְבֵלָה אֶלָּא מִשּׁוּם בַּל תֹּאכַל דָּבָר טָמֵא: וְהָאוֹכְלָהּ אֵין טָעוּן הֶעֱרֵב שֶׁמֶשׁ. הָאוֹכֵל חֲצִי פְּרָס מִמֶּנָּה לְאַחַר שֶׁנִּטְמֵאת, שֶׁהוּא פּוֹסֵל אֶת הַגְּוִיָּה לֶאֱכֹל בַּתְּרוּמָה, אִם טָבַל לְאַחַר שֶׁאֲכָלָהּ מֻתָּר בַּתְּרוּמָה מִיָּד וְאֵין טָעוּן הַעֲרֵב שֶׁמֶשׁ: וְאֵין חַיָּבִים עָלֶיהָ עַל בִּיאַת מִקְדָּשׁ. אִם אָכַל חֲצִי פְּרָס נִבְלַת עוֹף טָמֵא וְנִכְנַס לַמִּקְדָּשׁ. דְּטֻמְאָה דְּרַבָּנָן הִיא , אַף עַל גַּב דְּשׂוֹרְפִין עָלֶיהָ תְּרוּמָה: וְהָאוֹכֵל אֵבָר מִן הַחַי מִמֶּנָּה אֵינוֹ סוֹפֵג אֶת הָאַרְבָּעִים. וְאַף עַל גַּב דְּאוֹכֵל כַּזַּיִת בֵּין בָּשָׂר וְגִידִים וַעֲצָמוֹת, דְּהַשְׁתָּא אֲפִלּוּ לְמַאן דְּאָמַר אֵבָר מִן הַחַי צָרִיךְ כַּזַּיִת דַּאֲכִילָה כְּתִיב בֵּיהּ, הָכָא אִיכָּא כַּזַּיִת, אֲפִלּוּ הָכִי אֵינוֹ סוֹפֵג אֶת הָאַרְבָּעִים . וּמִשּׁוּם טָמֵא לֹא מִחַיַּב, דְּלֵיכָּא כַּזַּיִת בָּשָׂר: וְאֵין שְׁחִיטָתָהּ מְטַהַרְתָּהּ. בַּאֲכִילָה, לְבֶן נֹחַ שֶׁמֻּתָּר בְּעוֹף טָמֵא וְאָסוּר בְּאֵבָר מִן הַחַי, וְאִם שְׁחָטָהּ וְהִיא מְפַרְכֶּסֶת לֹא טִהֲרַתָּהּ שְׁחִיטָתָהּ בַּאֲכִילָה עַד שֶׁתָּמוּת, אַף עַל גַּב דְּבִטְהוֹרָה אֲפִלּוּ מְפַרְכֶּסֶת מַתֶּרֶת הַשְּׁחִיטָה לְבֶן נֹחַ מִגּוֹ דְּמִכַּשְּׁרָה לְיִשְׂרָאֵל. וְהָכִי מִפָּרְשָׁה מַתְנִיתִין בְּשִׁלְהֵי פֶּרֶק גִּיד הַנָּשֶׁה:

פרק א - משנה ד
וּבַבְּהֵמָה הָעוֹר וְהָרֹטֶב. כֻּלַּהּ מַתְנִיתִין מִפָּרְשָׁה בְּחֻלִּין פֶּרֶק הָעוֹר וְהָרֹטֶב:

פרק א - משנה ה
מִצְטָרְפִין זֶה עִם זֶה לְטַמֵּא. אִם אֵין בְּאֶחָד מֵהֶן כַּבֵּיצָה, מִצְטָרְפִין לְשִׁעוּר לְטַמֵּא כַּקַּל שֶׁבִּשְׁנֵיהֶם, שֶׁאִם נָגַע בַּתְּרוּמָה פְּסָלָהּ, כְּדִין אֹכֶל שֶׁנִּטְמָא בִּוְלַד הַטֻּמְאָה שֶׁפּוֹסֵל אֶת הַתְּרוּמָה וְאֵינוֹ מְטַמְּאָהּ: כַּחֲצִי בֵיצָה וְכוּ'. שֶׁבְּלָלָן זֶה בָּזֶה: שֵׁנִי. הַתַּעֲרֹבֶת הַזֶּה תּוֹרַת שֵׁנִי עָלָיו וּפוֹסֵל אֶת הַתְּרוּמָה: שְׁלִישִׁי. וּפוֹסֵל אֶת הַקֹּדֶשׁ: רִאשׁוֹן. וְעוֹשֶׂה מַגָּעוֹ שֵׁנִי. דְּהָא אִיכָּא שִׁעוּרָא דְּרִאשׁוֹן בַּגְּוִיָּה לִתֵּן טֻמְאָה: חִלְּקָן זֶה שֵׁנִי וְזֶה שֵׁנִי. שֶׁהֲרֵי בְּכָל אֶחָד חֲצִי שִׁעוּר דְּרִאשׁוֹן וַחֲצִי שִׁעוּר דְּשֵׁנִי, וּתְנַן לְעֵיל רִאשׁוֹן וְשֵׁנִי מִצְטָרְפִין לַקַּל שֶׁבָּהֶן:

פרק א - משנה ו
. וְכָל מָקוֹם שֶׁאֲנִי יָכוֹל לִתְלוֹת שֶׁיֵּשׁ כַּבֵּיצָה מִן הֶחָמוּר בָּזֶה וּבָזֶה, אָזְלִינַן בָּתַר חָמוּר. וְאִם לָאו, אָזְלִינַן בָּתַר הַקַּל:

פרק א - משנה ז
מִקְרָצוֹת. חֲתִיכוֹת שֶׁל בָּצֵק. לְשׁוֹן קֶרֶץ מִצָּפוֹן בָּא (יִרְמְיָה מו): נוֹשְׁכוֹת זוֹ אֶת זוֹ. שֶׁהֵן דְּבוּקוֹת זוֹ בָּזוֹ וְאִי אֶפְשָׁר לְהַפְרִישָׁן אֶלָּא אִם כֵּן נוֹשְׁכוֹת זוֹ מִזּוֹ: כֻּלָּן תְּחִלָּה. דַּחֲשׁוּבוֹת כְּאַחַת וּכְאִלּוּ נָגַע בְּכֻלָּן: פֵּרְשׁוּ. הַחֲתִיכוֹת זוֹ מִזּוֹ: בְּמַשְׁקִין כֻּלָּן שְׁנִיּוֹת. דְּמַשְׁקִין תְּחִלָּה הֲווּ וְעוֹשִׂין שֵׁנִי: בְּיָדַיִם כֻּלָּן שְׁלִישִׁיּוֹת. דְּיָדַיִם שְׁנִיּוֹת הֵן, וְעוֹשׂוֹת שְׁלִישִׁי בַּתְּרוּמָה:

פרק א - משנה ח
מִקְרֶצֶת. חֲתִיכַת בָּצֵק שֶׁהָיְתָה תְּחִלָּה וְחִבֵּר לָהּ חֲתִיכוֹת אֲחֵרוֹת: כֻּלָּן תְּחִלָּה. וְהַנּוֹגֵעַ בָּאֲחֵרוֹת כְּנוֹגֵעַ בָּרִאשׁוֹן לַטֻּמְאָה, דַּחֲשִׁיבָא חִבּוּר. וְכֵן נַמִּי הָיְתָה שְׁנִיָּה, חֲשׁוּבוֹת חִבּוּר, מֵאַחַר דְּאִם פָּרְשׁוּ אִשְׁתְּיַר בְּהוּ טֻמְאָה. אֲבָל הָיְתָה שְׁלִישִׁית, דְּאִם פָּרְשׁוּ לֵית בְּהוּ טֻמְאָה, דְּאֵין שְׁלִישִׁי עוֹשֶׂה רְבִיעִי בַּתְּרוּמָה , אֲפִלּוּ כִּי לֹא פָּרְשׁוּ לָא חֲשִׁיבָן חִבּוּר, וּטְהוֹרוֹת:

כִּכְּרוֹת הַקֹּדֶשׁ. כְּגוֹן שְׁתֵּי הַלֶּחֶם וְלֶחֶם הַפָּנִים וּמְנָחוֹת וְכַיּוֹצֵא בָּהֶן: שֶׁבְּתוֹךְ גֻּמּוֹתֵיהֶן מַיִם הַמְקֻדָּשִׁים. שֶׁהָיוּ בְּתוֹךְ הַגֻּמּוֹת שֶׁעַל גַּבֵּי הַכִּכָּר מַיִם שֶׁנַּעֲשׂוּ עַל טָהֳרַת הַקֹּדֶשׁ, וְנִטְמָא כִּכָּר אֶחָד בְּשֶׁרֶץ, וְנָגַע הַכִּכָּר שֶׁנִּטְמָא בַּשֵּׁנִי, וְהַשֵּׁנִי בַּשְּׁלִישִׁי, וַאֲפִלּוּ הֵן מֵאָה כֻּלָּן טְמֵאִים, וְאַף עַל פִּי שֶׁלֹּא נָגְעוּ בַּמַּיִם שֶׁבַּכִּכָּר אֶלָּא נָגַע כָּל כִּכָּר בַּחֲבֵרוֹ, נִטְמְאוּ כֻּלָּן, שֶׁחִבַּת הַקֹּדֶשׁ עוֹשָׂה כְּאִלּוּ הַמַּשְׁקֶה שֶׁבְּתוֹךְ כָּל כִּכָּר שֶׁהוּא נַעֲשָׂה תְּחִלָּה הוּא שֶׁנָּגַע בַּכִּכָּר וְטִמְּאוֹ. אֲבָל בַּתְּרוּמָה אֵינוֹ כֵּן, אֶלָּא הַכִּכָּר הָרִאשׁוֹן הַנִּטְמָא בַּשֶּׁרֶץ בִּלְבַד מְטַמֵּא אֶת חֲבֵרוֹ לִהְיוֹת שֵׁנִי, וְהַשֵּׁנִי פּוֹסֵל אֶת הַשְּׁלִישִׁי, וְתוּ לֹא, וְלֹא חָיְשִׁינַן לְמַשְׁקֶה שֶׁבְּתוֹךְ הַגֻּמָּא: אִם יֵשׁ בֵּין הַכִּכָּרוֹת מַשְׁקֶה טוֹפֵחַ אַף בַּתְּרוּמָה כֻּלָּן טְמֵאוֹת. הַמַּשְׁקֶה טוֹפֵחַ שֶׁבֵּין כָּל כִּכָּר וְכִכָּר חוֹזֵר לִהְיוֹת תְּחִלָּה, וּמְטַמֵּא הַכִּכָּר שֶׁהוּא נוֹגֵעַ בּוֹ לַעֲשׂוֹתוֹ שֵׁנִי, וְכֵן לְעוֹלָם. פֵּרוּשׁ אַחֵר, כִּכְּרוֹת הַקֹּדֶשׁ הַמֻּנָּחוֹת בְּטַבְלָא הָעֲשׂוּיָה גֻּמּוֹת גֻּמּוֹת, וְכָל כִּכָּר מֻנָּח בְּתוֹךְ גֻּמָּא שֶׁלּוֹ. וְהַמַּיִם הַמְקֻדָּשִׁים גָּרְסִינַן, פֵּרוּשׁ, וְכֵן הַמַּיִם הַמְקֻדָּשִׁים הַמֻּנָּחִים בִּכְלִי הֶעָשׂוּי גֻּמּוֹת גֻּמּוֹת, אִם נִטְמָא אֶחָד מֵהֶן בְּשֶׁרֶץ, נִטְמְאוּ כֻּלָּן, שֶׁהַכְּלִי מְצָרֵף מַה שֶּׁבְּתוֹכוֹ לַקֹּדֶשׁ, אֲבָל לֹא לַתְּרוּמָה, דְּבַתְּרוּמָה אֵינוֹ מְטַמֵּא אֶלָּא רִאשׁוֹן וְשֵׁנִי וּפוֹסֵל שְׁלִישִׁי הֵיכָא דְּנָגְעוּ זֶה בָּזֶה, וְאֵין כְּלִי מְצָרֵף. וְאִם יֵשׁ בֵּין הַגֻּמּוֹת מַשְׁקֶה טוֹפֵחַ הַכֹּל טָמֵא, שֶׁהַכִּכָּר שֶׁהוּא שֵׁנִי עוֹשֶׂה הַמַּשְׁקֶה רִאשׁוֹן וְחוֹזֵר וּמְטַמֵּא כִּכָּר הָאַחֵר, וְכֵן כֻּלָּם:
