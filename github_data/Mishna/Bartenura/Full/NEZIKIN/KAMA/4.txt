פרק ד

פרק ד - משנה א
שׁוֹר שֶׁנָּגַח אַרְבָּעָה וַחֲמִשָּׁה. וּבְכֻלָּם הָיָה תָּם דְּמִשְׁתַּלֵּם מִגּוּפוֹ: יְשַׁלֵּם. חֲצִי נֶזֶק לָאַחֲרוֹן תְּחִלָּה. בַּגְּמָרָא מוֹקֵי לְמַתְנִיתִין כְּגוֹן שֶׁתְּפָסוֹ נִזָּק לַשּׁוֹר הַמַּזִּיק לִגְבּוֹת מִמֶּנּוּ וְנַעֲשָׂה עָלָיו שׁוֹמֵר שָׂכָר . וּכְשֶׁיָּצָא מִתַּחַת יָדוֹ וְהִזִּיק, הַנִּזָּק הָרִאשׁוֹן חַיָּב בִּנְזָקָיו, לְפִיכָךְ הָאַחֲרוֹן מִשְׁתַּלֵּם חֲצִי נִזְקוֹ מָשְׁלַם: וְאִם יֶשׁ בּוֹ מוֹתָר יַחֲזִיר לְשֶׁלְּפָנָיו. הָכִי קָאָמַר, אִם יֵשׁ בּוֹ מוֹתָר בִּנְזָקָיו, יַחֲזִיר לְשֶׁלְּפָנָיו. כְּגוֹן שֶׁחֲצִי נִזְקוֹ שֶׁל הָרִאשׁוֹן הָיָה מָנֶה, וַחֲצִי נֶזֶק הָאַחֲרוֹן הָיָה חֲמִשִּׁים, וְהַשּׁוֹר שָׁוֶה מָאתַיִם. מִתְּחִלָּה כְּשֶׁנָּגַח שׁוֹר זֶה שׁוֹרוֹ שֶׁל הַנִּזָּק הָרִאשׁוֹן שֶׁחֲצִי נִזְקוֹ מָנֶה, הָיָה לַנִּזָּק בְּשׁוֹר זֶה מָנֶה, וְלַבְּעָלִים מָנֶה. וּכְשֶׁתְּפָסוֹ הַנִּזָּק וְנָגַח תַּחַת יָדוֹ, אֵין לַבְּעָלִים לְהַפְסִיד מָנֶה שֶׁהָיָה לָהֶם בּוֹ, שֶׁהֲרֵי לֹא הָיְתָה שְׁמִירָתוֹ עֲלֵיהֶן אֶלָּא עַל הַנִּזָּק שֶׁתְּפָסוֹ. וּכְשֶׁהִזִּיק לַשֵּׁנִי וְהָיָה חֲצִי נִזְקוֹ חֲמִשִּׁים, אִבֵּד הַנִּזָּק הָרִאשׁוֹן מִן הַמָּנֶה שֶׁלּוֹ חֲמִשִּׁים וְנוֹתֵן לָזֶה הַנִּזָּק הַשֵּׁנִי, וְהַמּוֹתָר עַד הַמָּנֶה חוֹזֵר לוֹ. וְהַבְּעָלִים נוֹטְלִים הַמָּנֶה שֶׁלָּהֶם: רַבִּי שִׁמְעוֹן אוֹמֵר וְכוּ'. רַבִּי שִׁמְעוֹן סָבַר שֻׁתְּפֵי נִינְהוּ הַבְּעָלִים וְהַנִּזָּק בַּשּׁוֹר הַמַּזִּיק, וּשְׁנֵיהֶם מִתְחַיְּבִים בִּנְזָקָיו. כֵּיצַד, שׁוֹר שָׁוֶה מָאתַיִם שֶׁנָּגַח וְכוּ': וְשֶׁלְּפָנָיו. זֶהוּ נִזָּק רִאשׁוֹן. נוֹטֵל חֲמִשִּׁים זוּז, וְהַבְּעָלִים חֲמִשִּׁים זוּז. דְּיֵשׁ לְנִזָּק רִאשׁוֹן בּוֹ הַחֵצִי, הִלְכָּךְ מְשַׁלֵּם חֲצִי תַּשְׁלוּמֵי נִזְקוֹ: חָזַר וְנָגַח שׁוֹר שָׁוֶה מָאתַיִם. הָאַחֲרוֹן נוֹטֵל מָנֶה הַחֵצִי מִכָּל מַה שֶּׁהוּא, דְּמִגּוּפוֹ מִשְׁתַּלֵּם. וְנִמְצָא אוֹתוֹ שֶׁלְּפָנָיו שֶׁהָיָה הַחֵצִי שֶׁלּוֹ, מְשַׁלֵּם מֵחֶלְקוֹ חֲצִי מָנֶה שֶׁנּוֹטֵל הָאַחֲרוֹן: וּשְׁנַיִם הָרִאשׁוֹנִים. נִזָּק רִאשׁוֹן וְהַבְּעָלִים שֶׁהָיָה לָהֶם לְכָל אֶחָד רְבִיעַ, מְשַׁלְּמִין כָּל אֶחָד רְבִיעִית שֶׁל נִזְקוֹ: דִּינָר זָהָב. שֶׁהֵם עֶשְׂרִים וַחֲמִשָּׁה דִּינָרִים שֶׁל כֶּסֶף:

פרק ד - משנה ב
מוּעָד לִקְטַנִּים. לַעֲגָלִים : אָמְרוּ לִפְנֵי רַבִּי יְהוּדָה. שָׁאֲלוּ תַּלְמִידָיו מִמֶּנּוּ: מוּעָד לְשַׁבָּתוֹת. מִפְּנֵי שֶׁהוּא בָּטֵל מִמְּלָאכָה וְדַעְתּוֹ זָחָה עָלָיו. אִי נַמִּי לְפִי שֶׁרוֹאֶה בְּנֵי אָדָם בְּמַלְבּוּשִׁים נָאִים שֶׁל שַׁבָּת, חֲשׁוּבִים בְּעֵינָיו נָכְרִים וְאֵינוֹ מַכִּירָם: וְאֵינוֹ מוּעָד לְחֹל. מַה דִּינוֹ: מִשֶּׁיַּחֲזֹר בּוֹ שְׁלֹשָׁה יְמֵי שַׁבָּתוֹת. לְאַחַר שֶׁהוּעַד לְשַׁבָּתוֹת הֶעֱבִירוּ לְפָנָיו שְׁוָרִים בְּשָׁלֹשׁ שַׁבָּתוֹת וְלֹא נָגַח, חָזַר לְתַמּוּתוֹ, וְאִם חָזַר וְנָגַח אֵינוֹ מְשַׁלֵּם אֶלָּא חֲצִי נֶזֶק:

פרק ד - משנה ג
שׁוֹר שֶׁל יִשְׂרָאֵל שֶׁנָּגַח שׁוֹר שֶׁל נָכְרִי פָּטוּר. דִּכְתִיב (חֲבַקּוּק ג) עָמַד וַיְמֹדֶד אֶרֶץ רָאָה וַיַּתֵּר גּוֹיִם, רָאָה שֶׁבַע מִצְוֹת שֶׁנִּצְטַוּוּ בְּנֵי נֹחַ, כֵּיוָן שֶׁלֹּא קִיְּמוּ אוֹתָן, עָמַד וְהִתִּיר מָמוֹנָן לְיִשְׂרָאֵל . וְאוֹמֵר (דְּבָרִים לג) הוֹפִיעַ מֵהַר פָּארָן, גִּלָּה מָמוֹנָן שֶׁל עוֹבְדֵי כּוֹכָבִים וְהִתִּירָן. מֵהַר פָּארָן, מִשָּׁעָה שֶׁסִּבֵּב וְהֶחְזִיר אֶת הַתּוֹרָה עַל הָעוֹבְדֵי כּוֹכָבִים וְלֹא קִבְּלוּהָ:

פרק ד - משנה ד
וְשֶׁל חֵרֵשׁ שׁוֹטֶה וְקָטָן שֶׁנָּגַח שׁוֹר שֶׁל פִּקֵּחַ פָּטוּר. שֶׁאֵין מַעֲמִידִין אַפּוֹטְרוֹפּוֹס לְתָם לִגְבּוֹת מִגּוּפוֹ, דְּמִטַּלְטְלֵי הוּא, וְאָמְרִינַן בְּפֶרֶק קַמָּא שָׁוֶה כֶּסֶף מְלַמֵּד שֶׁאֵין בֵּית דִּין נִזְקָקִים אֶלָּא לִנְכָסִים שֶׁיֵּשׁ לָהֶן אַחֲרָיוּת, וְאוֹקִימְנָא בְּיַתְמֵי : מַעֲמִידִין לָהֶן אַפּוֹטְרוֹפּוֹס וְכוּ'. אִם הֻחְזְקוּ נַגְּחָנִים, מַעֲמִידִין לָהֶם אַפּוֹטְרוֹפּוֹס, וְלֹא לְשַׁלֵּם חֲצִי נֶזֶק אֶלָּא לְשַׁוְּיֵהּ מוּעָד, דְּכִי הָדַר נָגַח מְשַׁלֵּם מִן הָעֲלִיָּה וְיִגְבּוּ הַנֶּזֶק מִקַּרְקַע שֶׁל יְתוֹמִים : חָזַר לְתַמּוּתוֹ. דְּקָסָבַר מוּעָד שֶׁיָּצָא מֵרְשׁוּת בְּעָלָיו וְנִכְנַס לִרְשׁוּת בְּעָלִים אֲחֵרִים חוֹזֵר לְתַמּוּתוֹ. דִּרְשׁוּת מְשֻׁנָּה מְשַׁנָּה אֶת דִּין הַתְרָאָתוֹ : שׁוֹר הָאִצְטָדִין. שֶׁמְּיֻחָד לִנְגִיחוֹת, וּמְלַמְּדִין אוֹתוֹ לְכָךְ:

פרק ד - משנה ה
מוּעָד מְשַׁלֵּם כֹּפֶר. וְאַף עַל גַּב דִּבְקַמָּא דְּנָגַח קָטְלִינַן לֵיהּ, אַשְׁכְּחִינַן מוּעָד, כְּגוֹן שֶׁהָרַג שְׁלֹשָׁה נָכְרִים. אִי נַמִּי, שֶׁהָרַג שְׁלֹשָׁה יִשְׂרָאֵל טְרֵפָה, דְּאַטְּרֵפָה לֹא קָטְלִינַן לֵיהּ, דְּגַבְרָא קְטִילָא קָטֵיל. אִי נַמִּי, דְּקָטֵיל וַעֲרַק לְאַגְמָא לְאַחַר שֶׁהֵעִידוּ בּוֹ: וְכֵן בְּבֵן וְכֵן בְּבַת. תִּינוֹק וְתִינוֹקֶת , חַיָּב עֲלֵיהֶן סְקִילָה וְכֹפֶר כִּגְדוֹלִים:

פרק ד - משנה ו
שׁוֹר שֶׁהָיָה מִתְחַכֵּךְ בְּכֹתֶל וְכוּ'. פָּטוּר מִן הַמִּיתָה. וְאִם הָיָה מוּעָד, כְּגוֹן שֶׁהָיָה מוּעָד לְהִתְחַכֵּךְ בִּכְתָלִים וּלְהַפִּילָם עַל בְּנֵי אָדָם, וְנִתְחַכֵּךְ בְּכֹתֶל לַהֲנָאָתוֹ וְהִפִּילוֹ עַל הָאָדָם וָמֵת , הַשּׁוֹר פָּטוּר מִן הַמִּיתָה, וְהַבְּעָלִים מְשַׁלְּמִים אֶת הַכֹּפֶר. הַשּׁוֹר פָּטוּר מִן הַמִּיתָה, דִּכְתִיב (שְׁמוֹת כא) הַשּׁוֹר יִסָּקֵל וְגַם בְּעָלָיו יוּמַת, כְּמִיתַת הַבְּעָלִים כָּךְ מִיתַת הַשּׁוֹר, מַה בְּעָלִים אֵין חַיָּבִין אִם הָרְגוּ אֶת הַנֶּפֶשׁ עַד שֶׁיַּהַרְגוּ בְּכַוָּנָה, כָּךְ הַשּׁוֹר אֵינוֹ חַיָּב עַד שֶׁיַּהֲרֹג בְּכַוָּנָה. וְהַבְּעָלִים מְשַׁלְּמִים אֶת הַכֹּפֶר, דִּכְתִיב אִם כֹּפֶר, שֶׁהָיָה יָכוֹל לִכְתֹּב כֹּפֶר יוּשַׁת עָלָיו, מַאי אִם כֹּפֶר, לְרַבּוֹת הַהוֹרֵג שֶׁלֹּא בְּכַוָּנָה לְחִיּוּב כֹּפֶר :

פרק ד - משנה ז
שׁוֹר הַיְתוֹמִים. שֶׁאֵין לָהֶם אַפּוֹטְרוֹפּוֹס: וְשׁוֹר הָאַפּוֹטְרוֹפּוֹס. הַשּׁוֹר שֶׁל יְתוֹמִים הוּא, אֶלָּא שֶׁעַל הָאַפּוֹטְרוֹפּוֹס לְשָׁמְרוֹ: הֲרֵי אֵלּוּ חַיָּבִין. דְּשִׁבְעָה שׁוֹר כְּתוּבִים בַּפָּרָשָׁה בְּנוֹגֵחַ אָדָם. חַד לְגוּפֵיהּ, וְשִׁשָּׁה לְשִׁשָּׁה שְׁוָרִים הַלָּלוּ : רַבִּי יְהוּדָה אוֹמֵר שׁוֹר הֶקְדֵּשׁ שׁוֹר הַגֵּר שֶׁמֵּת וְאֵין לוֹ יוֹרְשִׁים פְּטוּרִים מִן הַמִּיתָה. אֲפִלּוּ נָגַח וְאַחַר כָּךְ הִקְדִּישׁ, נָגַח וְאַחַר כָּךְ מֵת הַגֵּר, הָיָה פּוֹטֵר רַבִּי יְהוּדָה . וְאֵין הֲלָכָה כְּרַבִּי יְהוּדָה:

פרק ד - משנה ח
וְאִם שְׁחָטוֹ אָסוּר. בַּאֲכִילָה. דִּכְתִיב (שְׁמוֹת כא) סָקוֹל יִסָּקֵל הַשּׁוֹר וְלֹא יֵאָכֵל אֶת בְּשָׂרוֹ, מִמַּשְׁמַע שֶׁנֶּאֱמַר סָקוֹל יִסָּקֵל הַשּׁוֹר אֵינִי יוֹדֵעַ שֶׁהִיא נְבֵלָה וּנְבֵלָה אֲסוּרָה בַּאֲכִילָה, וּמַה תַּלְמוּד לוֹמַר וְלֹא יֵאָכֵל אֶת בְּשָׂרוֹ, אֶלָּא לוֹמַר לְךָ שֶׁאִם קָדַם וּשְׁחָטוֹ לְאַחַר שֶׁנִּגְמַר דִּינוֹ אָסוּר : הִקְדִּישׁוֹ בְעָלָיו מֻקְדָּשׁ. וְנָפְקָא מִנַּהּ דְּאִי מִתְהַנֵי מִנֵּיהּ מָעַל:

פרק ד - משנה ט
וְנָעַל בְּפָנָיו כָּרָאוּי. בְּדֶלֶת שֶׁיְּכוֹלָה לַעֲמֹד בְּרוּחַ מְצוּיָה. וְהַיְנוּ שְׁמִירָה פְּחוּתָה: וּמוּעָד פָּטוּר. מִצַּד הַעֲדָאָה שֶׁבּוֹ. אֲבָל צַד תַּמּוּת בִּמְקוֹמָהּ עוֹמֶדֶת. וּמְשַׁלֵּם חֲצִי נֶזֶק כְּתָם. דִּכְתִיב וְלֹא יִשְׁמְרֶנּוּ גַּבֵּי מוּעָד, הָא שְׁמָרוֹ כָּל דְּהוּ פָּטוּר מִצַּד הַעֲדָאָה: רַבִּי אֱלִיעֶזֶר אוֹמֵר, אֵין לוֹ שְׁמִירָה. לַמּוּעָד אֶלָּא סַכִּין. עַד שֶׁיִּשְׁחֲטֶנּוּ. וְשָׁלֹשׁ מַחֲלֹקוֹת בַּדָּבָר. לְרַבִּי מֵאִיר בִּשְׁמִירָה פְּחוּתָה חַיָּב וּבִמְעֻלָּה פָּטוּר. וּלְרַבִּי יְהוּדָה בִּשְׁמִירָה פְּחוּתָה נַמִּי פָּטוּר מִצַּד הַעֲדָאָה שֶׁבּוֹ, אֲבָל חַיָּב עַל צַד תַּמּוּת שֶׁבּוֹ, עַד שֶׁיִּשְׁמְרֶנּוּ שְׁמִירָה מְעֻלָּה. וּלְרַבִּי אֱלִיעֶזֶר בִּמְעֻלָּה נַמִּי חַיָּב. וַהֲלָכָה כְּרַבִּי יְהוּדָה. וּמִיהוּ לְכַתְּחִלָּה מִצְוָה לִשְׁחֹט שׁוֹר הַמּוּעָד כְּדֵי לְסַלֵּק הַהֶזֵּק:
