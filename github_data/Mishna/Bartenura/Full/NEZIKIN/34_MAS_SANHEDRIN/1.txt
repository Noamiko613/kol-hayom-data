פרק א

דִּינֵי מָמוֹנוֹת. דְּהַיְנוּ הוֹדָאוֹת וְהַלְוָאוֹת: בִּשְׁלשָׁה. הֶדְיוֹטוֹת. וְלֹא אַצְרְכוּהוּ רַבָּנָן לִשְׁלֹשָׁה מֻמְחִין , שֶׁלֹּא תִנְעֹל דֶּלֶת בִּפְנֵי לֹוִין, שֶׁמָּא יִכְפֹּר הַלֹּוֶה וְלֹא יִמְצָא מֻמְחִין לְכוֹפוֹ לַדִּין אֶלָּא יָחִיד מֻמְחֶה אוֹ שְׁלֹשָׁה הֶדְיוֹטוֹת. אֲבָל גְּזֵלוֹת וַחֲבָלוֹת בִּשְׁלשָׁה מֻמְחִין, דֵּאלֹהִים כְּתִיב בְּפָרָשַׁת שׁוֹמְרִים בְּוְאֵלֶּה הַמִּשְׁפָּטִים שָׁלשׁ פְּעָמִים. וּמִנַּהּ יָלְפִינַן שְׁלֹשָׁה מֻמְחִין: נֶזֶק. אָדָם אוֹ שׁוֹר הַמּוּעָד שֶׁהִזִּיקוּ, דִּמְשַׁלְּמִים נֶזֶק שָׁלֵם: חֲצִי נֶזֶק. שׁוֹר תָּם שֶׁהִזִּיק. וְאַף עַל גַּב דְּהַיְנוּ חֲבָלוֹת , אַיְדֵי דְבָעֵי לְמִתְנֵי תַּשְׁלוּמֵי כֶפֶל וְתַשְׁלוּמֵי אַרְבָּעָה וַחֲמִשָּׁה שֶׁאֵינוֹ מְשַׁלֵּם כְּמוֹ מַה שֶּׁהִזִּיק, שֶׁמְּשַׁלֵּם יוֹתֵר, תָּנָא נַמִּי חֲצִי נֶזֶק שֶׁאֵינוֹ מְשַׁלֵּם כְּמוֹ שֶׁהִזִּיק, שֶׁהֲרֵי מְשַׁלֵּם פָּחוֹת. וְאַיְדֵי דְתָנָא חֲצִי נֶזֶק תָּנָא נַמִּי נֶזֶק: וְהַמּוֹצִיא שֵׁם רַע. לֹא מָצָאתִי לְבִתְּךָ בְּתוּלִים וְעָנְשׁוּ אֹתוֹ מֵאָה כֶסֶף: דִּינֵי נְפָשׁוֹת. דְּאִם אֱמֶת הָיָה הַדָּבָר וְזִנְּתָה תַחְתָּיו וּסְקָלוּהָ. וְדִינֵי נְפָשׁוֹת בְּעֶשְׂרִים וּשְׁלשָׁה כְּדִלְקַמָּן. וַהֲלָכָה כַּחֲכָמִים:

מַכּוֹת. מַלְקוֹת אַרְבָּעִים: בִּשְׁלֹשָׁה. דִּכְתִיב (דְּבָרִים כה) וְנִגְּשׁוּ אֶל הַמִּשְׁפָּט וּשְׁפָטוּם, שְׁנַיִם, וְאֵין בֵּית דִּין שָׁקוּל מוֹסִיפִין עֲלֵיהֶם עוֹד אֶחָד, הֲרֵי כָאן שְׁלֹשָׁה: רַבִּי יִשְׁמָעֵאל אוֹמֵר בְּעֶשְׂרִים וּשְׁלשָׁה. אַתְיָא רָשָׁע רָשָׁע, כְּתִיב הָכָא (שָׁם) וְהִרְשִׁיעוּ אֶת הָרָשָׁע, וּכְתִיב הָתָם (בְּמִדְבָּר לה) אֲשֶׁר הוּא רָשָׁע לָמוּת. מַה לְהַלָּן בְּעֶשְׂרִים וּשְׁלשָׁה, אַף כָּאן בְּעֶשְׂרִים וּשְׁלֹשָׁה: עִבּוּר הַחֹדֶשׁ. קִדּוּשׁ הַחֹדֶשׁ . וְאַיְדֵי דְבָעֵי לְמִתְנֵי עִבּוּר שָׁנָה תָּנָא נַמִּי עִבּוּר הַחֹדֶשׁ: בִּשְׁלשָׁה מַתְחִילִין. לִרְאוֹת אִם יֵשׁ לְהוֹשִׁיב בֵּית דִּין עַל כָּךְ. אִם אֶחָד מֵהַשְּׁלשָׁה אוֹמֵר צָרִיךְ בֵּית דִּין לֵישֵׁב וּלְעַיֵּן אִם צְרִיכָה שָׁנָה לְעַבֵּר, מִפְּנֵי הַתְּקוּפָה וְהָאָבִיב וְהַפֵּרוֹת , וּשְׁנַיִם אוֹמְרִים אֵין צָרִיךְ, שֶׁאֵין כָּאן סָפֵק, דְּוַדַּאי אֵינָהּ צְרִיכָה עִבּוּר, בָּטֵל יָחִיד בְּמִעוּטוֹ. שְׁנַיִם אוֹמְרִים לֵישֵׁב וְאֶחָד אוֹמֵר שֶׁלֹּא לֵישֵׁב, הוֹלְכִין אַחַר הַשְּׁנַיִם, וּמוֹסִיפִין עוֹד שְׁנַיִם אֲחֵרִים שֶׁיִּשְׂאוּ וְיִתְּנוּ בַדָּבָר, וַהֲרֵי כָאן חֲמִשָּׁה. שְׁנַיִם אוֹמְרִים צָרִיךְ לְעַבֵּר וּשְׁלשָׁה אוֹמְרִים אֵין צָרִיךְ, בָּטְלוּ שְׁנַיִם בְּמִעוּטָן. שְׁנַיִם אוֹמְרִים אֵינָהּ צְרִיכָה וּשְׁלשָׁה אוֹמְרִים צְרִיכָה, הוֹלְכִים אַחַר הָרֹב, וּמוֹסִיפִין עוֹד שְׁנַיִם שֶׁיִּהְיוּ שִׁבְעָה וּמְעַבְּרִין אוֹתָהּ. וּבַגְּמָרָא מְפָרֵשׁ דְּהָנֵי שְׁלֹשָׁה חֲמִשָּׁה וְשִׁבְעָה כְּנֶגֶד בִּרְכַּת כֹּהֲנִים , שֶׁיֵּשׁ בְּפָסוּק רִאשׁוֹן שָׁלשׁ תֵּבוֹת, וּבַשֵּׁנִי חָמֵשׁ, וּבַשְּׁלִישִׁי שֶׁבַע. וַהֲלָכָה כְּרַבָּן שִׁמְעוֹן בֶּן גַּמְלִיאֵל:

סְמִיכַת זְקֵנִים. עַל רֹאשׁ פַּר הֶעְלֵם דָּבָר שֶׁל צִבּוּר. וְיֵשׁ בְּמַשְׁמָעוּת הַדְּבָרִים גַּם מִנּוּי הַדַּיָּן, שֶׁהַגָּדוֹל הַסּוֹמֵךְ צָרִיךְ שֶׁיְּצָרֵף עִמּוֹ שְׁנַיִם כְּשֶׁרוֹצֶה לִסְמֹךְ חָכָם כְּדֵי שֶׁיִּקָּרֵא רַבִּי וְיָדוּן דִּינֵי קְנָסוֹת. וּלְשׁוֹן סְמִיכָה, מִשּׁוּם דִּכְתִיב (בְּמִדְבָּר כז) וַיִּסְמֹךְ אֶת יָדָיו עָלָיו. וְלָאו דְּבָעֵי מִסְמָךְ יְדֵיהּ עֲלֵיהּ, אֶלָּא בִשְׁמָא קָרוּ לֵיהּ רַבִּי . וְאֵין סְמִיכָה בְחוּצָה לָאָרֶץ, אֶלָּא צָרִיךְ שֶׁהַסּוֹמֵךְ וְהַנִּסְמָךְ יִהְיוּ כֻלָּן בְּאֶרֶץ יִשְׂרָאֵל, וְאָז יִהְיֶה לוֹ רְשׁוּת לָדוּן דִּינֵי קְנָסוֹת וַאֲפִלּוּ בְחוּץ לָאָרֶץ, לְפִי שֶׁסַּנְהֶדְרִין נוֹהֶגֶת בֵּין בָּאָרֶץ בֵּין בְּחוּץ לָאָרֶץ, אַחַר שֶׁנִּסְמְכוּ בָאָרֶץ. וְכָתַב הָרַמְבַּ"ם שֶׁנִּרְאֶה לוֹ שֶׁבַּזְּמַן הַזֶּה שֶׁאֵין לָנוּ סָמוּךְ אִישׁ מִפִּי אִישׁ עַד מֹשֶׁה רַבֵּנוּ, אִם הָיוּ מַסְכִּימִים כָּל הַחֲכָמִים שֶׁבְּאֶרֶץ יִשְׂרָאֵל לִסְמֹךְ יָחִיד אוֹ רַבִּים, הֲרֵי אֵלּוּ סְמוּכִים וִיכוֹלִים לָדוּן דִּינֵי קְנָסוֹת וְיֵשׁ לָהֶן לִסְמֹךְ לַאֲחֵרִים . וְהַדָּבָר צָרִיךְ הֶכְרֵעַ: וַעֲרִיפַת הָעֶגְלָה בִּשְׁלשָׁה. דִּכְתִיב (וַיִּקְרָא ד) זִקְנֵי הָעֵדָה , מִעוּט זִקְנֵי שְׁנַיִם, וְאֵין בֵּית דִּין שָׁקוּל, הוֹסֵיף עֲלֵיהֶם עוֹד אֶחָד הֲרֵי שְׁלשָׁה: וְרַבִּי יְהוּדָה אוֹמֵר בַּחֲמִשָּׁה. וְסָמְכוּ, שְׁנַיִם. זִקְנֵי, שְׁנַיִם. וְאֵין בֵּית דִּין שָׁקוּל, הוֹסֵיף עֲלֵיהֶם עוֹד אֶחָד, הֲרֵי חֲמִשָּׁה. וַהֲלָכָה כְּרַבִּי יְהוּדָה: מִצְוַת חֲלִיצָה בִּשְׁלשָׁה. דִּכְתִיב (דְּבָרִים כה) וְנִגְּשָׁה יְבִמְתּוֹ אֵלָיו לְעֵינֵי הַזְּקֵנִים, זְקֵנִים, שְׁנַיִם. וְאֵין בֵּית דִּין שָׁקוּל, הוֹסֵיף עֲלֵיהֶם עוֹד אֶחָד, הֲרֵי שְׁלשָׁה. וּשְׁנַיִם אֲחֵרִים שֶׁמּוֹסִיפִים אֵינוֹ אֶלָּא לְפִרְסוּמֵי מִלְּתָא: מֵאוּנִין. קְטַנָּה יְתוֹמָה שֶׁהִשִּׂיאוּהָ אִמָּהּ וְאַחֶיהָ לְדַעְתָּהּ וְיוֹצְאָה בְמֵאוּן שֶׁמְּמָאֶנֶת בְּבַעְלָהּ, צָרִיךְ שֶׁיִּהְיֶה בִשְׁלשָׁה. וּבִיבָמוֹת מוּכָח דְּמֵאוּן בִּפְנֵי שְׁנַיִם מַסְפִּיק: נֶטַע רְבָעִי. אִם בָּא לְחַלְּלוֹ עַל הַמָּעוֹת, וְכֵן מַעֲשֵׂר שֵׁנִי, וְאֵין דְּמֵיהֶן יְדוּעִים. כְּגוֹן פֵּרוֹת וְהִרְקִיבוּ שֶׁאֵין לָהֶם שַׁעַר יָדוּעַ: הַהֶקְדֵּשׁוֹת. הַבָּא לִפְדּוֹתָן צָרִיךְ שְׁלֹשָׁה לְשׁוּמָן: הָעֲרָכִין הַמִּטַּלְטְלִין. הֲרֵי שֶׁאָמַר עֵרֶךְ פְּלוֹנִי עָלַי, וְאֵין לוֹ מָעוֹת לִתֵּן כְּפִי הַדָּמִים הַקְּצוּבִים בַּפָּרָשָׁה, וּבָא לִתֵּן מִטַּלְטְלִין, צָרִיךְ שְׁלֹשָׁה לָשׁוּם אוֹתָן מִטַּלְטְלִים : אֶחָד מֵהֶן כֹּהֵן. דְּבַעֲרָכִין כֹּהֵן כְּתִיב (וַיִּקְרָא כז) כְּעֶרְכְּךָ הַכֹּהֵן: וְהַקַּרְקָעוֹת. וְאִם אֵין לוֹ מִטַּלְטְלִין וּבָא לִתֵּן קַרְקַע , צָרִיךְ שֶׁיָּשׁוּמוּ עֲשָׂרָה אֲנָשִׁים וְאֶחָד מֵהֶן כֹּהֵן אוֹתָהּ קַרְקַע שֶׁיִּהְיֶה כְּפִי הָעֵרֶךְ שֶׁיֵּשׁ עָלָיו לִתֵּן: וְאָדָם כַּיּוֹצֵא בָהֶן. וְאִם אָמַר דְּמֵי פְלוֹנִי עָלַי, שֶׁשָּׁמִין אוֹתוֹ כַּמָּה הוּא יָפֶה לִמָּכֵר בַּשּׁוּק וְנוֹתֵן דָּמָיו, צָרִיךְ גַּם כֵּן שֶׁיִּהְיוּ עֲשָׂרָה וְאֶחָד מֵהֶן כֹּהֵן בְּשׁוּמָא זוֹ:

סַנְהֶדְרִין, שֶׁהָיָה מֻפְלָא שֶׁבְּכֻלָּן וְכֻלָּן נִגְרָרִים אַחֲרָיו: עַל הָעִיר. יְרוּשָׁלַיִם. שֶׁקְּדֻשָּׁתָהּ גְּדוֹלָה מִשְּׁאָר קְדֻשַּׁת אֶרֶץ יִשְׂרָאֵל: וְהָעֲזָרוֹת. קְדֻשָּׁתָן
