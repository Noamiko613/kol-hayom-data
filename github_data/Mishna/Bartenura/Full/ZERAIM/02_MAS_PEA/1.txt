פרק א

אֵלּוּ דְבָרִים שֶׁאֵין לָהֶם שִׁעוּר. מִדְּאוֹרַיְתָא, אֲבָל מִדְּרַבָּנָן יֵשׁ לָהֶם שִׁעוּר, דְּהָא תְּנַן בְּסֵיפָא אֵין פּוֹחֲתִין לַפֵּאָה מִשִּׁשִּׁים: הַפֵּאָה. שֶׁאָדָם חַיָּב לְהַנִּיחַ בְּסוֹף שָׂדֵהוּ לַעֲנִיִּים, כְּדִכְתִיב (וַיִּקְרָא כג) לֹא תְכַלֶּה פְּאַת שָׂדְךָ בְּקֻצְרֶךָ: וְהַבִּכּוּרִים. דִּכְתִיב (שְׁמוֹת כג) רֵאשִׁית בִּכּוּרֵי אַדְמָתְךָ תָּבִיא בֵּית ה' אֱלֹהֶיךָ, וְלֹא נָתְנָה תּוֹרָה בָּהֶם שִׁעוּר : וְהָרֵאָיוֹן. שֶׁחַיָּב אָדָם לְהֵרָאוֹת בָּעֲזָרָה בְּשָׁלֹשׁ רְגָלִים, דִּכְתִיב (שָׁם) שָׁלֹשׁ פְּעָמִים בַּשָּׁנָה יֵרָאֶה כָל זְכוּרְךָ, וְזֹאת הָרְאִיָּה אֵין לָהּ שִׁעוּר, דְּכָל אֵימַת דְּבָעֵי עַיֵּל וּמִתְחָזֵי וְנָפֵיק. פֵּרוּשׁ אַחֵר עוֹלַת רְאִיָּה וְשַׁלְמֵי חֲגִיגָה שֶׁחַיָּב לְהָבִיא כְּדִכְתִיב (שָׁם) וְלֹא יֵרָאוּ פָנַי רֵיקָם, אֵין לָהּ שִׁעוּר מִדְּאוֹרַיְתָא דִּכְתִיב (דְּבָרִים טז) אִישׁ כְּמַתְּנַת יָדוֹ, אֶלָּא שֶׁחֲכָמִים נָתְנוּ בָהּ שִׁעוּר, הָעוֹלָה מָעָה כֶּסֶף וַחֲגִיגָה שְׁתֵּי כֶסֶף: וּגְמִילוּת חֲסָדִים. דִּבְגוּפוֹ, כְּגוֹן בִּקּוּר חוֹלִים וְלִקְבֹּר מֵתִים וְכַיּוֹצֵא בָהֶן, אֲבָל גְּמִילוּת חֲסָדִים דִּבְמָמוֹנוֹ, כְּגוֹן פִּדְיוֹן שְׁבוּיִם וּלְהַלְבִּישׁ עֲרוּמִים וּלְהַאֲכִיל אֶת הָרְעֵבִים וְכַיּוֹצֵא בָהֶן יֵשׁ לָהֶן שִׁעוּר, שֶׁיִּתֵּן בְּכָל פַּעַם שֶׁתָּבֹא מִצְוָה כָּזוֹ לְיָדוֹ חֲמִשִּׁית מִן הָרֶוַח שֶׁבִּנְכָסָיו, וְתוּ לֹא מִחַיֵּב, דְּהָכִי אָמְרִינַן הַמְבַזְבֵּז אַל יְבַזְבֵּז יוֹתֵר מֵחֹמֶשׁ . הִלְכָּךְ מִבָּעֵי לֵיהּ לְאִינָשׁ לְאַפְרוּשֵׁי חֲמִשִּׁית הָרֶוַח בְּכָל עֵת כְּדֵי שֶׁיְּהֵא מָצוּי כָּל זְמַן שֶׁיָּבֹא גְּמִילוּת חֶסֶד לְיָדוֹ שֶׁיְּקַיְּמֶנּוּ, וּבְהָכִי נָפֵיק יְדֵי חוֹבָתוֹ: וְתַלְמוּד תּוֹרָה. אֵין לָהּ שִׁעוּר דִּכְתִיב (יְהוֹשֻׁעַ א) וְהָגִיתָ בּוֹ יוֹמָם וָלַיְלָה: וְתַלְמוּד תּוֹרָה כְּנֶגֶד כֻּלָּם. שָׁקוּל כְּנֶגֶד כֻּלָּם:

אֵין פּוֹחֲתִין לַפֵּאָה מִשִּׁשִּׁים. מִדְּרַבָּנָן: אַף עַל פִּי שֶׁאָמְרוּ אֵין לַפֵּאָה שִׁעוּר. מִדְּאוֹרַיְתָא כִּדְאֲמָרָן: הַכֹּל לְפִי גֹדֶל הַשָּׂדֶה. אִם שָׂדֵהוּ גְדוֹלָה וַעֲנִיִּים מֻעָטִים נוֹתֵן וּמוֹסִיף עַל שִׁשִּׁים לְפִי גֹדֶל שָׂדֵהוּ, וְאִם שָׂדֵהוּ קְטַנָּה וַעֲנִיִּים מְרֻבִּים מוֹסִיף עַל שִׁשִּׁים לְפִי רֹב הָעֲנִיִּים: וּלְפִי הָעֲנָוָה. אִית דְּגָרְסֵי עֲנָוָה בְּוָי"ו, כְּלוֹמַר לְפִי רֹב עַנְוָתוֹ יַרְבֶּה לָעֲנִיִּים, לְשׁוֹן וְעַנְוָתְךָ תַרְבֵּנִי (תְּהִלִּים יח). וְרַמְבַּ"ם פֵּרֵשׁ לְשׁוֹן עֲנִיָּה, כְּלוֹמַר מַה שֶּׁתַּעֲנֶה הָאָרֶץ. וְאִית דְּגָרְסֵי עֲנָבָה בְּבֵי"ת, וּפֵרְשׁוּ גֹדֶל הַגַּרְגְּרִים וְקָטְנָן, שֶׁאִם הָיוּ הַשִּׁבֳּלִים שֶׁל שָׂדֶה בְּמָקוֹם אֶחָד מְלֵאוֹת וְטוֹבוֹת וּבְמָקוֹם אַחֵר צְנוּמוֹת דַּקּוֹת לֹא יַנִּיחַ הַפֵּאָה כֻּלָּהּ מִן הָרָעוֹת אֶלָּא יְשַׁעֵר שִׁשִּׁים לָרַע וְלַיָּפֶה:

מִתְּחִלַּת הַשָּׂדֶה וּמֵאֶמְצָעָהּ. וְאֵין צָרִיךְ לְהַנִּיחַ פֵּאָה בְּסוֹף הַשָּׂדֶה, דִּכְתִיב (וַיִּקְרָא יט) לֹא תְכַלֶּה פְּאַת שָׂדְךָ לִקְצֹר, אָמְרָה תּוֹרָה תֵּן פֵּאָה וְאַף עַל פִּי שֶׁיֵּשׁ לְךָ עֲדַיִן קָמָה לִקְצֹר: וּבִלְבַד שֶׁיִּתֵּן בַּסּוֹף כַּשִּׁעוּר. אַף עַל גַּב דְּרָבִי קְרָא בַּתְחִלָּה וּבָאֶמְצַע שֶׁיֵּשׁ תּוֹרַת פֵּאָה לְמַה שֶּׁנָּתַן, מִכָּל מָקוֹם לֹא מִפְטַר אֶלָּא אִם כֵּן נוֹתֵן בְּסוֹף הַשָּׂדֶה תַּשְׁלוּם שִׁעוּר אֶחָד מִשִּׁשִּׁים עִם מַה שֶּׁנָּתַן בָּאֶמְצַע וּבַתְּחִלָּה. כֵּן מוּכָח בַּיְרוּשַׁלְמִי. וְרַמְבַּ"ם פֵּרֵשׁ וּבִלְבַד שֶׁיִּתֵּן בַּסּוֹף כְּשִׁעוּר פֵּאָה הַצְּרִיכָה לְכָל הַשָּׂדֶה, שֶׁצָּרִיךְ שֶׁיַּנִּיחַ בְּסוֹף הַשָּׂדֶה אֶחָד מִשִּׁשִּׁים חוּץ מִמַּה שֶּׁהִנִּיחַ בַּתְּחִלָּה וּבָאֶמְצַע: אִם שִׁיֵּר קֶלַח אֶחָד. בְּסוֹף הַשָּׂדֶה וְהִנִּיחַ אוֹתוֹ לְפֵאָה סוֹמֵךְ עָלָיו, וְכָל מַה שֶּׁהִנִּיחַ בָּאֶמְצַע וּבַתְּחִלָּה תּוֹרַת פֵּאָה עָלֶיהָ: וְאִם לָאו. שֶׁלֹּא הִנִּיחַ אֲפִלּוּ קֶלַח אֶחָד בְּסוֹף הַשָּׂדֶה לְפֵאָה, הֲרֵי מַה שֶּׁהִנִּיחַ בָּאֶמְצַע וּבַתְּחִלָּה אֵינָהּ חֲשׁוּבָה פֵּאָה אֶלָּא דִּין הֶפְקֵר יֵשׁ לָהּ . וּפָלֵיג אַדְּרַבִּי שִׁמְעוֹן דְּאָמַר דְּמִכָּל מָקוֹם הִיא חֲשׁוּבָה פֵּאָה אֶלָּא דְּלֹא מִפְטַר בַּעַל הַשָּׂדֶה בְּהָכִי . וַהֲלָכָה כְּרַבִּי שִׁמְעוֹן:

כָּל דָּבָר שֶׁהוּא אֹכֶל. לְמִעוּטֵי סְפִיחֵי סִטִּים , שֶׁאַף עַל פִּי שֶׁהֵן אֹכֶל עַל יְדֵי הַדְּחַק לָאו אֹכֶל מִקְרוּ וְאֵין חַיָּבִין בַּפֵּאָה, דִּכְתִיב (וַיִּקְרָא יט) וּבְקֻצְרְכֶם, וְאֵין קְצִירָה אֶלָּא בְּדָבָר הָרָאוּי לַאֲכִילָה: וְנִשְׁמָר. לְמִעוּטֵי הֶפְקֵר שֶׁאֵין לוֹ שׁוֹמְרִים שֶׁאֵינוֹ חַיָּב בַּפֵּאָה, דִּכְתִיב (שָׁם) לֶעָנִי וְלַגֵּר תַּעֲזֹב אוֹתָם, פְּרָט לְהֶפְקֵר שֶׁכְּבָר נֶעֱזַב: וְגִדּוּלָיו מִן הָאָרֶץ. לְמִעוּטֵי כְּמֵהִין וּפִטְרִיּוֹת, שֶׁאֵין לָהֶם שֹׁרֶשׁ בָּאָרֶץ וְגִדּוּלֵיהֶם מִן הָאֲוִיר, שֶׁאֵין חַיָּבִין בַּפֵּאָה דִּכְתִיב (שָׁם) קְצִיר אַרְצְכֶם: וּלְקִיטָתוֹ כְּאֶחָד. לְמִעוּטֵי תְּאֵנָה וְכַיּוֹצֵא בָהּ מִן הָאִילָנוֹת שֶׁנִּלְקָטִים הַפֵּרוֹת שֶׁמִּתְבַּשְּׁלִים בָּהֶם רִאשׁוֹן רִאשׁוֹן , דִּכְתִיב (שָׁם) קְצִיר, מַשְׁמַע דָּבָר הַנִּקְצָר הַרְבֵּה בְּיַחַד : וּמַכְנִיסוֹ לְקִיּוּם. לְמִעוּטֵי יָרָק שֶׁאֵינוֹ מִתְקַיֵּם אֶלָּא מִתְעַפֵּשׁ מִיָּד, דִּכְתִיב (שָׁם כג) פְּאַת שָׂדְךָ בְּקֻצְרֶךָ, וְאֵין הַיָּרָק נִקְצָר : וְהַתְּבוּאָה וְהַקִּטְנִיּוֹת. תְּבוּאָה מֵחֲמֵשֶׁת הַמִּינִים, חִטָּה וּשְׂעוֹרָה וְכוּסֶמֶת שִׁבֹּלֶת שׁוּעָל וְשִׁיפוֹן קִטְנִיּוֹת פּוֹלִין וַאֲפוּנִים וַעֲדָשִׁים וְדוֹמֵיהֶן: בַּכְּלָל הַזֶּה. לְהִתְחַיֵּב בַּפֵּאָה כְּשֶׁיִּשְׁלְמוּ בָהֶם כָּל הַתְּנָאִים הַלָּלוּ :

הָאוֹג. אִילָן שֶׁהַפְּרִי שֶׁלּוֹ אָדוֹם, וְקוֹרִין לוֹ בְּלַעַ"ז קורנ"י אולי"ס. וְרַמְבַּ"ם פֵּרֵשׁ שֶׁהוּא צֶמַח שֶׁמְּעַבְּדִים בּוֹ הָעוֹרוֹת וְקוֹרִין לוֹ בַּעֲרָבִי סומא"ק, וְהָאֶשְׁכֹּלוֹת שֶׁלּוֹ רְאוּיִין לַאֲכִילָה. וְלָאו דַּוְקָא אֵלּוּ שֶׁמָּנָה הַתַּנָּא חַיָּבִים בַּפֵּאָה וְלֹא מִינֵי אִילָנוֹת אֲחֵרִים, אֶלָּא אֵלּוּ וְדוֹמֵיהֶן קָאָמַר:

לְעוֹלָם הוּא נוֹתֵן מִשּׁוּם פֵּאָה. מִי שֶׁלֹּא הִנִּיחַ פֵּאָה בַּשָּׂדֶה בַּמְחֻבָּר חַיָּב לְהַפְרִישׁ מִן הַתָּלוּשׁ, וְהַפֵּאָה שֶׁמַּפְרִישׁ מִן הַתָּלוּשׁ וְנוֹתֵן לָעֲנִיִּים פְּטוּרָה מִן הַמַּעַשְׂרוֹת: עַד שֶׁיְּמָרֵחַ. שֶׁיֵּעָשֶׂה כְּרִי וְצִבּוּר שֶׁל תְּבוּאָה , אֲבָל אִם בָּא לְהַפְרִישׁ פֵּאָה אַחַר הַמֵּרוּחַ צָרִיךְ שֶׁיַּפְרִישׁ הַתְּרוּמָה וְהַמַּעַשְׂרוֹת תְּחִלָּה וְאַחַר כָּךְ יִטֹּל הַפֵּאָה, שֶׁאֵין פָּטוּר מִן הַמַּעַשְׂרוֹת אֶלָּא הַפֵּאָה שֶׁנִּטְּלָה קֹדֶם הַמֵּרוּחַ: וְנוֹתֵן מִשּׁוּם הֶפְקֵר וּפָטוּר מִן הַמַּעַשְׂרוֹת. (מִשּׁוּם הֶפְקֵר) הַמַּפְקִיר תְּבוּאָתוֹ וּבָא אַחֵר וְזָכָה בָהּ, אֵין הַזּוֹכֶה בָהּ חַיָּב בְּמַעַשְׂרוֹת דִּכְתִיב (דְּבָרִים יד) וּבָא הַלֵּוִי כִּי אֵין לוֹ חֵלֶק וְנַחֲלָה עִמָּךְ, יָצָא הֶפְקֵר שֶׁיָּדְךָ וְיָדוֹ שָׁוִין, אֲבָל הַמַּפְקִיר תְּבוּאָתוֹ לְאַחַר הַמֵּרוּחַ אֵין הַהֶפְקֵר הַהוּא פּוֹטֵר מִן הַמַּעֲשֵׂר : וּמַאֲכִיל לַבְּהֵמָה וְכוּ'. וַאֲפִלּוּ הוּא עַצְמוֹ מֻתָּר לֶאֱכֹל אֲכִילַת עֲרַאי קֹדֶם מֵרוּחַ, אֲבָל הַבְּהֵמָה אוֹכֶלֶת אֲפִלּוּ אֲכִילַת קֶבַע: נוֹטֵל מִן הַגֹּרֶן וְזוֹרֵעַ וְכוּ'. דְּמִן הַתּוֹרָה הַזּוֹרֵעַ פָּטוּר מִן הַמַּעֲשֵׂר אֲפִלּוּ לְאַחַר מֵרוּחַ דִּכְתִיב (שָׁם) עַשֵּׂר תְּעַשֵּׂר וְאָכַלְתָּ, וְזוֹרֵעַ לֹא קָרִינַן בֵּיהּ וְאָכַלְתָּ, מִיהוּ מִדְּרַבָּנָן חַיָּב. סְבִירָא לֵיהּ לְרַבִּי עֲקִיבָא דְּקֹדֶם מֵרוּחַ פָּטוּר מִן הַמַּעֲשֵׂר אֲפִלּוּ מִדְּרַבָּנָן. וְאֵין הֲלָכָה כְּרַבִּי עֲקִיבָא: שֶׁלָּקְחוּ אֶת הַגֹּרֶן. שֶׁקָּנוּ תְּבוּאָה מִן הַגֹּרֶן, וְקָנְסוּ חֲכָמִים שֶׁיִּהְיוּ חַיָּבִים לְהַפְרִישׁ תְּרוּמוֹת וּמַעַשְׂרוֹת וְלָתֵת אוֹתָם לְכֹהֲנִים וּלְוִיִּם אֲחֵרִים כְּדֵי שֶׁלֹּא יִהְיוּ קוֹפְצִים לִקְנוֹת תְּבוּאָה אוֹ יַיִן לַגִּתּוֹת וְלַגְּרָנוֹת . וְאִם קָנוּ קֹדֶם מֵרוּחַ, שֶׁעֲדַיִן לֹא הֻקְבְּעָה תְּבוּאָה זוֹ לְמַעֲשֵׂר, הַמַּעַשְׂרוֹת שֶׁלָּהֶם: הַמַּקְדִּישׁ וּפוֹדֶה. הַמַּקְדִּישׁ גָּרְנוֹ וּפְדָאוֹ מִיַּד הַגִּזְבָּר, וְהוּא גָּדִישׁ אוֹ קָמָה שֶׁלֹּא הֻקְבַּע לְמַעֲשֵׂר כְּשֶׁהָיָה בְּיַד הַהֶקְדֵּשׁ, חַיָּב הַפּוֹדֶה לְהוֹצִיא מִמֶּנּוּ הַמַּעַשְׂרוֹת. אֲבָל אִם נִתְמַרֵחַ הַגֹּרֶן בְּיַד הַגִּזְבָּר הוֹאִיל וּבְשָׁעָה שֶׁהָיָה רָאוּי שֶׁיֻּקְבַּע לְמַעֲשֵׂר דְּהַיְנוּ בִּזְמַן הַמֵּרוּחַ הָיְתָה בְּיַד הֶקְדֵּשׁ, אֵין הַפּוֹדֶה חַיָּב לְהוֹצִיא מִמֶּנּוּ מַעַשְׂרוֹת, שֶׁהַהֶקְדֵּשׁ פָּטוּר מִן הַמַּעַשְׂרוֹת :
