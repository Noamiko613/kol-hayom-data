פרק ל

כְּלֵי זְכוּכִית. הַךְ מַתְנִיתִין תְּנִינַן לְעֵיל בְּפֶרֶק ב' וּבְפֶרֶק ט"ו. וְאַיְדֵי דְבָעֵי לְמִתְנֵי דִין הָנָךְ כֵּלִים שֶׁל זְכוּכִית הֲדַר תָּנָא לַהּ הָכָא: טַבְלָא. שֻׁלְחָן: אִסְקוּטְלָא. בִּלְשׁוֹן לע"ז קוֹרִים לִקְעָרָה שקוטיל"א. וְהָכָא מַיְרֵי בִקְעָרָה פְשׁוּטָה שֶׁאֵין לָהּ בֵּית קִבּוּל, אֶלָּא שֶׁמִּתַּחְתֶּיהָ מְחֻבָּר לָהּ בֵּית קִבּוּל קָטָן הַמְשַׁמֵּשׁ כְּשֶׁכּוֹפִין הַכְּלִי עַל פָּנָיו , כְּעֵין שֶׁיֶּשׁ לְכוֹסוֹת שֶׁל כֶּסֶף וְשֶׁל זְכוּכִית שֶׁלָּנוּ: יֶשׁ לָהֶן לְבִזְבֵּז. שָׂפָה סָבִיב, הַלְּבִזְבֵּז עוֹשֵׂהוּ כְלִי שֶׁיֶּשׁ בּוֹ בֵית קִבּוּל: שׁוּלֵי קְעָרָה וְשׁוּלֵי אִסְקוּטְלָא. כְּשֶׁנִּשְׁבָּר מְקוֹם תַּשְׁמִישָׁן נִשְׁאָר הַבֵּית קִבּוּל קָטָן וְהוֹפְכוֹ וּמִשְׁתַּמֵּשׁ בּוֹ, וַאֲפִלּוּ הָכִי טְהוֹרִים, דְּלָא חֲשִׁיבֵי הַשּׁוּלַיִם כְּלִי, מִפְּנֵי מְקוֹם הַשֶּׁבֶר שֶׁמַּזִּיק לַיָּדַיִם כְּשֶׁאוֹחֲזִים בּוֹ: קִרְטְסָן אוֹ שָׁפָן. תִּקְּנָם וַעֲשָׂאָן חֲלָקִים עַד שֶׁרְאוּיִין לְהִשְׁתַּמֵּשׁ בְּאוֹתוֹ בֵית קִבּוּל, טְמֵאִים, דְּלֹא הָווּ תוּ שֶׁבֶר כְּלִי: בְּשׁוּפִין. כְּלִי שֶׁצּוֹרְפִין וְחָרָשֵׁי בַרְזֶל מְסִירִים בּוֹ הַחֲלֻדָּה וּמַחְלִיקִים בּוֹ הַכֵּלִים. תַּרְגּוּם הַצְּרִיפִים, שׁוּפִינָא. לימ"א בְּלַעַ"ז:

אַסְפַּקְלַרְיָא. מַרְאָה שֶׁל זְכוּכִית שֶׁהָאִשָּׁה רוֹאָה בָהּ אֶת פָּנֶיהָ. וּבְלַעַ"ז אספקל"ו . וַאֲפִלּוּ יֵשׁ לָהּ בֵּית קִבּוּל טְהוֹרָה, לְפִי שֶׁאֵינָהּ עֲשׂוּיָה לְקַבֵּל אֶלָּא לִרְאוֹת צוּרַת הַפָּנִים: תַּמְחוּי. כְּמִין קְעָרָה גְדוֹלָה שֶׁל זְכוּכִית: תַּרְוָד. כַּף שֶׁל זְכוּכִית: וְאִם לָאו. אִם כְּשֶׁנּוֹתְנוֹ עַל הַשֻּׁלְחָן אוֹ עַל דָּבָר פָּשׁוּט הוּא מִתְגַּלְגֵּל לְכָאן וּלְכָאן וְנִשְׁפָּךְ מַה שֶּׁבְּתוֹכוֹ, לְפִי שֶׁתַּחְתָּיו חַד וְאֵינוֹ עוֹמֵד: רַבִּי עֲקִיבָא מְטַמֵּא. הוֹאִיל וְיֵשׁ לוֹ בֵית קִבּוּל. וַהֲלָכָה כְּרַבִּי עֲקִיבָא:

כּוֹס. שֶׁל זְכוּכִית: שֶׁנִּפְגַּם רֻבּוֹ טָהוֹר. דַּחֲשִׁיב שֶׁבֶר כְּלִי: נִפְגַּם בּוֹ שָׁלֹשׁ בְּרֻבּוֹ. שֶׁהַשָּׁלֹשׁ פְּגִימוֹת עוֹמְדוֹת בְּרֻבּוֹ שֶׁל כּוֹס וְלֹא בְמִעוּטוֹ: טָהוֹר. אַף עַל פִּי שֶׁאֵינוֹ מְפַזֵּר אֶת רֹב הַמַּיִם: אִם מְפַזֵּר הוּא אֶת רֹב הַמַּיִם טָהוֹר. אֲבָל בְּמִעוּט, טָמֵא. וְאֵין הֲלָכָה כְּרַבִּי שִׁמְעוֹן: בְּבַעַץ. בִּבְדִיל: רַבִּי יוֹסֵי אוֹמֵר בְּבַעַץ טָמֵא. דְּהַכֹּל הוֹלֵךְ אַחַר הַמַּעֲמִיד, וַחֲשִׁיב כִּכְלֵי מַתָּכוֹת. וְאֵין הֲלָכָה כְּרַבִּי יוֹסֵי:

צְלוֹחִית. כְּעֵין צְלוֹחִיּוֹת שֶׁל זְכוּכִית שֶׁלָּנוּ שֶׁיֶּשׁ לָהֶן צַוָּאר אָרֹךְ: שֶׁנִּטַּל פִּיהָ. שֶׁנִּטַּל צַוָּארָהּ: וּבִגְדוֹלָה. שֶׁאֵינָהּ יְכוֹלָה לִנָּטֵל בְּיָדוֹ אַחַת אֶלָּא בִשְׁתֵּי יָדָיו קְרוּיָה גְדוֹלָה, וְשֶׁיְּכוֹלָה לִנָּטֵל בְּיָדוֹ אַחַת קְרוּיָה קְטַנָּה: שֶׁל פַּלְיָטוֹן. שֶׁל שֶׁמֶן אֲפַרְסְמוֹן: טְהוֹרָה. דְּתוּ לֹא חַזְיָא לִמְלַאכְתָּהּ: מִפְּנֵי שֶׁהִיא סוֹרַחַת אֶת הַיָּד. הַבָּא לָתֵת יָדוֹ לְתוֹכָהּ לָקַחַת מִן הַפַּלְיָטוֹן לְהָרִיחַ בּוֹ, יָדוֹ נִסְרֶכֶת וְנִשְׂרֶטֶת וְנִקְרַעַת, סוֹרַחַת, כְּמוֹ סוֹרֶכֶת בְּכָ"ף, לְשׁוֹן סִרְכָא: לְגִינִין גְּדוֹלִים. כְּמוֹ כַדִּין גְּדוֹלִים: מְתַקְּנָן לִכְבָשִׁים. שֶׁכּוֹבְשִׁים בָּהֶן יָרָק בְּמֶלַח וּבְחֹמֶץ וּמַנִּיחִין אוֹתָן שָׁם לְהִתְקַיֵּם: אֲפַרְכֵּס. כְּלִי עָשׂוּי לָרֵחַיִם, שֶׁרָחָב מִלְּמַעְלָה וְקָצָר מִלְּמַטָּה בְשׁוּלָיו, וְנוֹתְנִים בּוֹ חִטִּין הַרְבֵּה בְּפַעַם אַחַת וְהֵן יוֹרְדִים מְעַט מְעַט לְתוֹךְ הָרֵחַיִם וְנִטְחָנִים שָׁם. וְדֻגְמָתוֹ מָצִינוּ בְּרֵישׁ חֲגִיגָה (דַּף ג) עֲשֵׂה אָזְנְךָ כַּאֲפַרְכֶּסֶת, שֶׁמַּכְנִיס הַרְבֵּה וּמוֹצִיא קִמְעָה: אַשְׁרַיִךְ כֵּלִים. מַסֶּכֶת כֵּלִים: שֶׁנִּכְנַסְתְּ בְּטֻמְאָה וְיָצָאת בְּטָהֳרָה. מַתְחֶלֶת בַּאֲבוֹת הַטֻּמְאוֹת וּמְסַיֶּמֶת הָאֲפַרְכֵּס שֶׁל זְכוּכִית טְהוֹרָה:
