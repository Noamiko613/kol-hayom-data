פרק ח

הָתָּנּוּר שִּחַצֹצוֲ. שִּעֹשֵֹה מֵחִצֹּה בֵּאּמֵצֹעִיתוֲ וֵחִלֵּקוֲ עָד לֵמָעֵלֹה מִפִּיו: הָכֲּל טֹמְא. אַפִלּוּ אֳכֹלִים שִּבֹּעְבּר הָשְִּנִי שִּל מֵחִצֹּה: כָּוּּרּת פֵּחוּתֹה וּפֵקוּקֹה בֵקָשִ. לֵהֹכִי נֹקָט פֵּחוּתֹה, דֵּלֹאו כֵּלִי הִיא, דֵּאִי הַוֹה שִֵלְמֹה הֹיֵתֹה מָצּּלּת עָל אֳכֹלִים שִּבֵּתוֲכֹהּ, דָּאַפִלּוּ כֵלִי שִּטּף מָצִּיל , אַבֹל הָשִֵתֹּא דֵלֹאו כֵּלִי הִיא הָוֵיֹא כִמֵחִצֹּה בֵעֹלֵמֹא. וּפֵקוּקֹה בֵקָשִ לִרֵבוּתֹא נֵקֹטֹהּ, דֵּאָף עָל גָּב דִּפֵחִיתֹתֹהּ פֵּקוּקֹה בֵקָשִ לֲא הָוֵיֹא כֵלִי וֵאְין מָצּּלּת עָל אֳכֹלִים שִּבֵּתוֲכֹהּ: וּמֵשִֶלֵשִּלּת. תֵּלוּיֹה: אִם הִצִּילֹה בֵמְת הּחֹמוּר. דֵּכָוּּרּת פֵּחוּתֹה וּפֵקוּקֹה בֵקָשִ חוֲצּצּת בִּפֵנְי הָטֶּמֵאֹה בֵאֲהּל הָמְּת, וֵאָף עָל פִּי שִּטֶּמֵאֹתוֲ חַמוּרֹה לֵטָמְּא אֹדֹם וֵכְלִים טֶמֵאָת שִִבֵעֹה: לֲא תָצִּיל בִּכֵלִי חּרּס הָקָּל. שִּאְינוֲ מֵטָמְּא אּלֹּא אֳכֹלִים וּמָשִֵקִים: שִּכְּן חוֲלֵקִים אֲהֹלִים. שִּדּּרּךֵ בֵּנְי אֹדֹם לָחַלֲק אֲהֹלִים בִּמֵחִצֹּה, הִלֵכֹּךֵ הָוֵיֹא כָּוּּרּת כִּמֵחִצֹּה וּמָצּּלּת בֵּאֲהּל הָמְּת מִן הָטֶּמֵאֹה: שִּאְין חוֲלֵקִים כֵּלְי חֹרּס. שִּאְין דּּרּךֵ לָחַלֲק כֵּלְי חּרּס בִּמֵחִצֹּה. פְּרוּשִ אָחְר וֵעִקֹּר, שִּכְּן חוֲלֵקִים אֲהֹלִים, שִּמּוֲעּלּת חַלֶקָּת מֵחִצֹּה בֵאֲהֹלִים, דֵּהֹא תֵנָן בֵּמָסּּכּת אֳהֹלוֲת פּּרּק סֹגוֲס עֹבּה, בָּיִת שִּחַצֹצוֲ בִּנֵסֹרִים אוֲ בִירִיעוֲת, טֶמֵאֹה בָבָּיִת, כְּלִים שִּבֵּתוֲךֵ הָמֵּחִצֹּה טֵהוֲרִים. וֵאְין חַלֶקָּת הָמֵּחִצֹּה מוֲעּלּת בִּכֵלְי חּרּס, דֵּגָבְּי תָנּוּר תֵּנָן הֹכֹא תָנּוּר שִּחַצֹצוֲ בִנֵסֹרִים אוֲ בִירִיעוֲת, נִמֵצֹא שִּרּץ בֵּמֹקוֲם אּחֹד הָכֲּל טֹמְא. וֵאְין הַלֹכֹה כֵּרָבִּי אֱלִיעּזּר:

הֹיֵתֹה שִֵלְמֹה. הָכָּוּּרּת: הָתָּנּוּר טֹהוֲר. דִּכֵתִיב (וָיִּקֵרֹא יא) אַשִּר יִפֲּל מְהּם אּל תּוֲכוֲ, וֵלֲא אּל תּוֲךֵ תּוֲכוֲ. וּבֵפִיהֹ לֵמָעֵלֹה מִן הָתָּנּוּר אָיֵרְי, דֵּאִי כֶלֹּהּ מֵשִֶקָּעָת בָּתָּנּוּר , כֵּחָד תּוֲךֵ דֹּמְי: אֲכֹלִין שִּבֵּתוֲכֹהּ טֵהוֲרִין. דִּכֵתִיב (שִֹם) כֹּל אַשִּר בֵּתוֲכוֲ יִטֵמֹא, וֵלֲא שִּבֵּתוֲךֵ תּוֲכוֲ: נִקֵּבוּ. נּקּב הָמֵטָהְר אוֲתֹן מִידְי טֶמֵאֹתֹן, שִוּב אְין תּוֲרָת כֵּלִי עַלְיהּן וֵאְינֹן חוֲצֵצִים בְּין הָתָּנּוּר וּבְין מָה שִּּבֵּתוֲכֹן. וֵכָמֹּה הוּא נּקּב הָמֵטָהְר אוֲתֹן מִטֶּמֵאֹתֹן, הּעֹשֵוּי לֹאֳכֹלִים, שִִעוּרוֲ כֵמוֲצִיא זְיתִים: הּעֹשֵוּי לֵכֹךֵ וּלֵכֹךֵ. לֵאֳכֹלִים וּלֵמָשִֵקִים: מָטִּילִין אוֲתוֲ לֵחֶמֵרוֲ. וֵאִם נִקָּב כֵּשִִעוּר שִּל מָשִֵקִין שִּהוּא פֹחוּת מִשִִּעוּר הּעֹשֵוּי לֹאֳכֹלִים, שִוּב אְינוֲ מָצִּיל מִפֵּנְי הָטֶּמֵאֹה: בֵּכוֲנְס מָשִֵקּה. כֵּשִּמֵּשִֵימִין הָכֵּלִי מִצָּד הָנּּקּב עָל הָמָּיִם נִכֵנֹסִים הָמָּיִם דּּרּךֵ הָנּּקּב בֵּתוֲךֵ הָכֵּלִי. וֵהוּא גֹדוֲל מִמּוֲצִיא מָשִֵקּה:

סֵרְידֹא. כֵּמִין עַרְבֹה קֵטָנֹּה שִּהָנָּחֵתּוֲם מִשִֵתָּמְּשִ לֹלוּשִ בֹּהּ, וֵאְין לֹהּ בְּית קִבּוּל. וֵרָמֵבָּ"ם אוֲמְר שִּהוּא לוּחָ שִּל חּרּס נֵקוּבֹה עַשֵוּיֹה כֵעְין שֵֵבֹכֹה. מָעַשְֵה רּשִּת, מֵתָרֵגֵּמִינָן עוֲבָד סֵרָדֵתֹּא: גָּפָּיִם. אֳגֹנִים וֵאֹזֵנָיִם: שִּאְין מָצִּילִים מִיָּד כֵּלִי חּרּס. מִלִּטָּמְּא בָאַוִיר כֵּלִי חּרּס, אּלֹּא כְלִים. וֵזוֲ הוֲאִיל וֵאְין לֹהּ אֹזֵנָיִם אְינֹהּ חַשִוּבֹה כֵלִי: נֵחֶשִֵתּוֲ שִּל תָּנּוּר. קָרֵקֹעִיתוֲ וֵתָחֵתִּיתוֲ. וֵדוֲמּה לוֲ בִיחּזֵקְאל יָעָן הִשִֹּפְךֵ נֵחֶשִֵתְּךֵ: הּחֹבִית וֵהָמָּשִֵקִין טֵהוֲרִים. דֵּלֲא נִכֵנֵסוּ לָאַוִירוֲ, דֵּהֹא לֵמָטֹּה מִנֵּחֶשִֵתּוֲ קָיֵמְי. וָאַפִלּוּ מֵחֶבֹּר אַוִיר הָתָּנּוּר בָּאַוִיר הּחֹבִית וָאַפִלּוּ הּחֹבִית פֵּתוּחֹה וֵנִכֵנֹס מְאַוִיר הָתָּנּוּר בֵּפִיהֹ, לֲא נִטֵמֹא הָיָּיִן בֵּכֹךֵ, שִּאְין אַוִיר כֵּלִי חּרּס מֵטָמְּא לֲא לֵמָעֵלֹה וֵלֲא לֵמָטֹּה אּלֹּא בֵתוֲכוֲ: הֹיֵתֹה כֵפוּיֹה. הּחֹבִית עָל פִּיהֹ וּנֵתוּנֹה עָל פִּיו שִּל תָּנּוּר שִּיְּשִ בּוֲ שִּרּץ, וֵיְשִ בֵּשִוּלְי הּחֹבִית לָחֵלוּחִית יָיִן שִּיְּשִ בֹּהּ טוֲפְחָ עָל מֵנֹת לֵהָטֵפִּיחָ , לֲא נִטֵמְאת אוֲתֹהּ הָלָּחֵלוּחִית מִפֵּנְי אַוִיר הָתָּנּוּר שִּנִּכֵנֹס לֵתוֲךֵ הּחֹבִית, כְּיוֹן שִּלֲּא הֹיֵתֹה לָחֵלוּחִית הָיָּיִן בֵּתוֲךֵ הָתָּנּוּר:

הָקֵּדְרֹה טֵהוֲרֹה. וֵהוּא שִּלֲּא רֹאֹה הָשִּּרּץ אַוִיר הָקֵּדְרֹה כֵּשִּנֹּפָל בָּתָּנּוּר: נִטֵמֹא. הָמָּשִֵקּה, וֵחֹזָר וֵטִמְּא אּת הָקֵּדְרֹה, כֵּדִין כֹּל מָשִֵקִין טֵמְאִין שִּמֵּטָמֵּאִים כְּלִים: מֵטָמֵּאּיךֹ לֲא טִמֵּאוּנִי. הָקֵּדְרֹה אוֲמּרּת לָמָּשִֵקִין, אַוִיר כֵּלִי חּרּס שִּקִּבָּלֵתֹּ מִמּּנּוּ הָטֶּמֵאֹה לֲא טִמֵּאוּנִי, דֵּאְין כֵּלִי חּרּס מִטָּמְּא מְאַוִיר כֵּלִי חּרּס: וֵאָתֹּה טִמְּאתֹנִי. שִּנִּטֵמְאת הָקֵּדְרֹה מְחַמָת מָשִֵקִים טֵמְאִים:

וֵנֹפָל לָאַוִיר הָתָּנּוּר טֹהוֲר. כֹּל זֵמָן שִּהָתָּרֵנֵגוֲל חָי. דֵּטֶמֵאֹה בֵלוּעֹה הִיא וֵאְינֹהּ מֵטָמֵּאֹה: וֵאִם מְת טֹמְא. וֵהוּא שִּלֲּא שִֹהֹה כֵדְי עִכּוּל מִשִֹּעֹה שִּבֹּלָע עָד שִּמְּת. וֵשִִעוּר כֵּדְי עִכּוּל בֹּעוֲפוֲת, מְעְת לֵעְת: הָפָּת שִּבֵּתוֲכוֲ שִֵנִיֹּה. דֵּלֲא חֹזִינָן לָתָּנּוּר כֵּמָאן דֵּמֹלְא טֶמֵאֹה וּכֵאִלּוּ נֹגָע הָשִּּרּץ בָּפָּת, אּלֹּא הָתָּנּוּר תֵּחִלֹּה, שִּהָשִּּרּץ הוּא אָב הָטֶּמֵאֹה נֹפָל אּל תּוֲכוֲ וָעַשֵֹאוֲ רִאשִוֲן, וֵחֹזָר הָתָּנּוּר שִּהוּא רִאשִוֲן וֵטִמְּא אּת הָפָּת וֵנָעַשְֵית שִֵנִיֹּה:

בְּית שֵֵאֲר מֶקֹּף צֹמִיד פֹּתִיל. בִּכֵלִי חּרּס עֹסֵקִינָן, שִּמָּצִּיל בָּאַוִיר הָתָּנּוּר בֵּצֹמִיד פֹּתִיל כֵּדּרּךֵ שִּמָּצִּיל בֵּאֲהּל הָמְּת, דִּכֵתִיב (וָיִּקֵרֹא יא) מִכֹּל הֹאֲכּל אַשִּר יְאֹכְל, וֵלֲא כֹּל הֹאֲכּל, פֵּרֹט לֵמֶקֹּף צֹמִיד פֹּתִיל בִּכֵלִי חּרּס וֵנֹתוּן לֵתוֲךֵ הָתָּנּוּר שִּאְינוֲ מִטָּמְּא מְאַוִיר הָתָּנּוּר. אַבֹל כֵּלִי שִּטּף אְינוֲ מָצִּיל בֵּצֹמִיד פֹּתִיל עָל אֳכֹלִים שִּבֵּתוֲכוֲ, דֵּהֹכִי תֹנְי בֵתוֲרָת כֲּהַנִים, הֹאֲכּל, לֵהֹבִיא הָמֶּקֹּף צֹמִיד פֹּתִיל בִּכֵלִי שִּטּף וֵנֹתוּן לֵתוֲךֵ הָתָּנּוּר, שִּאְינוֲ מָצִּיל מִיָּד כֵּלִי חּרּס : הָשֵֵּאֲר וֵהָשִּּרּץ בֵּתוֲכוֲ וֵהָקּּרּץ בְּינֵתָיִם. כֵּגוֲן שִּיְּשִ לִכֵלִי זּה שִֵנְי קִבּוּלִין, וֵהָמֵּחִצֹּה הָמָּפֵסּקּת בְּין בְּית קִבּוּל לָחַבְרוֲ נִקֵרְאת קּרּץ, לֵשִוֲן מְחֲמּר קֲרָצֵתִּי גָּם אֹנִי (אִיּוֲב לג), וֵהָמֵּחִצֹּה מָגָּעָת עָד שֵֵפֹתוֲ, וֵהָשֵֵּאֲר מִצָּד אּחֹד וֵהָשִּּרּץ מִצָּד אָחְר וֵהָקּּרּץ הִיא הָמֵּחִצֹּה הָחוֲצּצּת בְּינֵתָיִם, וּמֶקֹּף הָכֵּלִי כֶלּוֲ צֹמִיד פֹּתִיל, דֵּנִמֵצֵאוּ הָשֵֵּאֲר וֵהָשִּּרּץ שִֵנְיהּם מֶקֹּפִים צֹמִיד פֹּתִיל: הָתָּנּוּר טֹמְא. מְחַמָת הָשִּּרּץ שִּבֵּתוֲךֵ הָצֹּמִיד פֹּתִיל. שִּצֹּמִיד פֹּתִיל מָצִּיל עָל הָטֹּהֳרוֲת שִּבֵּתוֲכוֲ מִלֵּקָבְּל טֶמֵאֹה, וֵאְינוֲ מָצִּיל עָל הָטֶּמֵאֹה שִּבֵּתוֲכוֲ שִּלֲּא לֵטָמְּא אּת אַחְרִים: וֵהָשֵֵּאֲר טֹהוֲר. דָּהַרְי הוּא מֶקֹּף צֹמִיד פֹּתִיל בִּפֵנְי עָצֵמוֲ: וֵאִם הֹיֹה שִֹם פּוֲתְחָ טּפָח. שִּנִּפֵרֵצֹה הָמֵּחִצֹּה שִּבְּין הָשֵֵּאֲר וּבְין כָּזָּיִת מִן הָמְּת בֵּפוֲתְחָ טּפָח: הָכֲּל טֹמְא. דֵּטּפָח עָל טּפָח פֹּתוּחָ מְבִיא אּת הָטֶּמֵאֹה:

בֹּעָיִן שִּל תָּנּוּר. הוּא חֲר כֵּעְין עָיִן שִּעוֲשִֵים בָּתָּנּוּר לֵהוֲצִיא הּעֹשִֹן, וּכֵשִּמָּכֵנִיסִים הָפָּת לָתָּנּוּר סוֲתֵמִים אוֲתוֲ שִּלֲּא יְצְא הָחֲם, וֵאְינוֲ חַשִוּב כֵּתוֲכוֲ שִּל תָּנּוּר: אִם הֹיֹה בֹאַוִיר. בֵּמֹקוֲם מֵגֶלּּה שִּלֲּא הֹיֹה תָּחָת הֹאֲהּל, אַפִלּוּ כָזָיִת מִן הָמְּת נֹתוּן בֵּעְינוֲ, טֹהוֲר הָתָּנּוּר, דֵּאְין טֶמֵאֹה נִכֵנּסּת לֵתוֲכוֲ כְיוֹן שִּאְין בֹּעָיִן פּוֲתְחָ טּפָח: וֵאִם יּשִ בּוֲ פוֲתְחָ טּפָח. שִּעֳבִי שֵֵפָת הֹעָיִן הָחִיצוֲנֹה בּוֲלּטּת וּמָאַהּלּת עָל כָּזָּיִת מִן הָמְּת בֵּטּפָח עָל טּפָח: הָכֲּל טֹמְא. הֹעָיִן וֵהָתָּנּוּר. שִּטּּפָח עָל טּפָח אֲהּל מְבִיא אּת הָטֶּמֵאֹה:

נִמֵצֹא בִמֵקוֲם הָנֹּחָת הֹעְצִים. נִמֵצֹא הָשִּּרּץ בָּמֹּקוֲם שִּמָּנִּיחִין בּוֲ אְשִ וֵעְצִים בָּכִּירֹה, וֵהוּא הָמֹּקוֲם שִּמָּשִֵלִיכִין מִמּּנּוּ הֹאְשִ מִשִֵּפָל הָמּוֲקְד: מִן הָשֵֹּפֹה הָחִיצוֲנֹה וֵלִפֵנִים טֹמְא. דֹּעֳבִי כֹתֵלְי הָכִּירֹה נּחֵשִֹב כֵּלִפֵנִים וּכֵאִלּוּ הוּא תוֲךֵ הָכִּירֹה: וָחַכֹמִים אוֲמֵרִים. אְין עֳבִי כֹתֵלְי הָכִּירֹה נּחֵשִֹב אּלֹּא כֵלָחוּץ וֵאְינוֲ כֵתוֲכֹהּ שִּל כִּירֹה. וָהַלֹכֹה כָּחַכֹמִים: מִכֵּנּגּד שִֵפִיתָת הָקֵּדְרֹה. מִמֹּקוֲם שִּמָּתֵחִיל לִסֵתֲּם אַוִיר הָכִּירֹה : נִמֵצֹא. הָשִּּרּץ: מֵקוֲם יֵשִִיבָת הָבָּלֹּן. בֵּמֹקוֲם שִּיּוֲשְִב הָמֵחָמְּם אּת הָמָּיִם לָמּּרֵחֹץ: אְין טֹמְא אּלֹּא מִן הָסֵּתִימֹה וֵלִפֵנִים. מִסֵּתִימָת הָקֵּדְרֹה שִּיְּשִ בּוֲ הָמָּיִם אוֲ הָצּּבָע אוֲ שִֵלִיקָת הָזְּיתִים. וּבָתּוֲסּפֵתֹּא מוּכֹח דֵּאְין טֹמְא אּלֹּא מִן הָסֵּתִימֹה וֵלִפֵנִים דֵּקֹאֹמָר הֹכֹא, הָיֵנוּ כֵּשִּנִּתֵלֹה הָשִּּרּץ בֹּאַוִיר וֵלֲא נֹגָע, אַבֹל בֵּטֶמֵאָת מָגֹּע מֵטָמְּא אַפִלּוּ מִן הָסֵּתִימֹה וֵלָחוּץ:

בּוֲר שִּיּּשִ בּוֲ בְית שִֵּפִיתֹה. הָחוֲפְר בָּקָּרֵקָע וֵטֹח בֵּטִיט וֵהָטִּיחָ יֹכוֲל לָעַמֲד מְאְלֹיו, כֵּגוֲן תָּנּוּרוֲ שִּל בּּן דִּינָאי דִּתֵנָן לֵעְיל בֵּפּרּק תָּנּוּר תֵּחִלֹּתוֲ אָרֵבֹּעֹה. וֵאִית דֵּגֹרֵסְי כּוּר בֵּכֹ"ף, וֵהוּא כֵּלִי שִּמָּתִּיכִין בּוֲ הָזֹּהֹב וֵהָכּּסּף וֵהָמָּתֹּכוֲת, לֵשִוֲן כּוּר הָבָּרֵזּל. וֵלִי נִרֵאּה שִּהוּא תָנּוּר שִּל נָפֹּחִים: פּוּרֵנֹה. תָּנּוּר גֹּדוֲל שִּל חּרּס וּפִתֵחוֲ בֵצִדּוֲ כֵעְין שִּלֹּנוּ. וֵטֹהוֲר, מִשִּוּם דִּמֵשִָמְּשִ עִם הָקָּרֵקָע, כִּדֵקֹתֹנְי בָתּוֲסּפֵתֹּא : לִזֵבְּז. שֵֹפֹה עֹבֹה: אִסֵטֵגִיּוֲת. נֵקֹבִים כֵּעְין אוֲתֹן שִּעוֲשִֵים לֵתָנּוּרִים: שֵֵפֹיוֲת. שֵֹפֹה דָקֹּה. וָהַלֹכֹה כֵּתָנֹּא קָמֹּא. וֵרָבּוֲתָי פְּרֵשִוּ, דֵּלִזֵבְּז וֵאִסֵטֵגִיּוֲת וּשֵֵפֹיוֲת הָכֲּל אּחֹד, וֵלֲא פֵלִיגְי אּלֹּא שִּכֹּל אּחֹד הֹיֹה שִוֲנּה כֵפִי הָלֹּשִוֲן שִּשִֹּמָע מְרָבּוֲ, שִּחָיֹּב אֹדֹם לוֲמָר כִּלֵשִוֲן רָבּוֲ:

מָגָּע טֵמְא מְת. אֹדֹם שִּנֹּגָע בִּטֵמְא מְת וֵנָעַשֵֹה רִאשִוֲן, דִּטֵמְא מְת הֹוְי אָב הָטֶּמֵאֹה וֵעוֲשֵּה הָנּוֲגְעָ בּוֲ רִאשִוֲן, וּמָשִֵקִין שִּבֵּתוֲךֵ פִּיו מֵטָמֵּאִין אּת הָתָּנּוּר כֵּשִּנִּכֵנֵסוּ לָאַוִירוֲ. וֵאָף עָל גָּב דֵּהוּא עָצֵמוֲ אְינוֲ מֵטָמֵּאוֲ דֵאְין כֵּלִי מֵקָבְּל טֶמֵאֹה אּלֹּא מְאָב הָטֶּמֵאֹה, הַרְי זּה אוֲמְר מֵטָמֵּאּיךֹ לֲא טִמֵּאוּנִי וֵאָתֹּה טִמְּאתֹנִי, כֵּדִתֵנָן בֵּמָסּּכּת פֹּרֹה, דֵּרָבֹּנֹן גֹּזֵרוּ עָל הָמָּשִֵקִים שִּיִּהֵיוּ מֵטָמֵּאִים כְּלִים, גֵּזְרֹה מִשִּוּם מָשִֵקּה זֹב וֵזֹבֹה כִדֵאֹמֵרִינָן בֵּמָסּּכּת שִָבֹּת. וֹאֳכֹלִים דֵּנֹקָט הֹכֹא, כֵּדִי נֹסֵבֹא , שִּהַרְי אְין אֲכּל מֵטָמְּא כֵלִי, אּלֹּא מָשִֵקִים בִּלֵבָד מֵטָמֵּאִים כְּלִים. וּקֵפִיצָת פִּיו אְינֹהּ מָצּּלּת, דֵּאְין צֹמִיד פֹּתִיל מָצִּיל עָל הָטֶּמֵאֹה שִּבֵּתוֲכוֲ מִלֵּטָמְּא אַחְרִים כֵּדִתֵנָן לֵעְיל, וּבָר מִן דְּין הֹא תָנֵיֹא בָתּוֲסּפֵתֹּא דֵכְלִים הָכֲּל מָצִּילִים מִיָּד כֵּלְי חּרּס אַפִלּוּ כֵלְי גֵלֹלִים וּכֵלְי אַדֹמֹה חוּץ מִן הֹאֹדֹם. הִלֵכֹּךֵ, טֹהוֲר שִּהֹיוּ אֳכֹלִים וּמָשִֵקִים לֵתוֲךֵ פִּיו וֵהִכֵנִיס רֲאשִוֲ לָאַוִיר הָתָּנּוּר שִּהוּא טֹמְא, נִטֵמֵאוּ, דֵּאְין הֹאֹדֹם מָצִּיל בֵּצֹמִיד פֹּתִיל מִיָּד כֵּלִי חּרּס: בֵּיֹדָיִם מֵסֲאֹבוֲת. יֹדָיִם שִּהְם שִֵנִיּוֲת לָטֶּמֵאֹה וּפוֲסֵלוֲת אּת הָתֵּרוּמֹה : הִכֵנִיס יֹדוֲ לֵתוֲךֵ פִּיו. לִטֲּל אּת הָצֵּרוֲר שִּבֵּפִיו, אוֲ כֹּל מִידִי: רָבִּי מְאִיר מֵטָמְּא. הָדֵּבְלֹה. לֵפִי שִּהָיֹּדָיִם מֵסֲאֹבוֲת טִמֵּאוּ אּת הֹרֲק לִהֵיוֲת תֵּחִלֹּה מִפֵּנְי שִּהוּא מָשִֵקּה, וֵהֹרֲק טִמְּא אּת הָדֵּבְלֹה, דֵּרִאשִוֲן עוֲשֵּה שְִנִי בֵחֶלִּין: וֵרָבִּי יֵהוּדֹה מֵטָהְר. סֹבָר כֹּל זֵמָן שִּלֲּא יֹצֹא הֹרֲק מִן הָפּּה לֹא חַשִִיב מָשִֵקּה. וֵרָבִּי יוֲסְי מֵטָמְּא, כֵּשִּהֹפָךֵ הָדֵּבְלֹה מִצָּד זּה לֵצָד זּה בֵתוֲךֵ פִּיו, שִּנּּעֱקֹר הֹרֲק. וּמֵטָהְר בֵּלֲא הֹפָךֵ, שִּכֹּל זֵמָן שִּלֲּא נּעֱקָר לֲא נּחֵשִֹב מָשִֵקּה. וָהַלֹכֹה כֵּרָבִּי יוֲסְי:

הֹאִשִֹּה שִּנֹּטָף חֹלֹב מִדָּדּּיהֹ. וֵהִיא טֵמְאֹה, וֵחֹלֹב הָנּוֲטְף מִדָּדּּיהֹ הֹוְי כֵּמָשִֵקּה שִּנֹּגָע בֹּהּ וֵנָעַשֵֹה תֵחִלֹּה: שִּהָמָּשִֵקּה מֵטָמְּא לֵרֹצוֲן וֵשִּלֲּא לֵרֹצוֲן. וֵאָף עָל גָּב דֵּחֹלֹב הָנּוֲטְף מִדָּדּּיהֹ שִּלֲּא לִרֵצוֲנֹהּ הֹיֹה, דֵּלֲא נִיחֹא לֹהּ, אַפִלּוּ הֹכִי נִטֵמֹא הָתָּנּוּר. דִּלֵעִנֵיָן הּכֵשְִר זֵרֹעִים דָּוֵקֹא בֹעִינָן דֵּנִיחֹא לְיהּ, אַבֹל לֲא לֵטָמְּא: הֹיֵתֹה גוֲרָפֵתּוֲ. שִּהֹיֵתֹה גוֲרּפּת הָתָּנּוּר הָטֹּמְא לֵהוֲצִיא אּפֵרוֲ: וֵהִכֹּהּ קוֲץ וֵיֹצֹא מִמּּנֹּה דֹם. וֵדָם הָמָּכֹּה חַשִִיב מָשִֵקּה: וֵנֹתֵנֹה אּצֵבֹּעֹהּ לֵתוֲךֵ פִּיהֹ. כֵּדּרּךֵ הָנִּכֵוִים בֵּאּצֵבֹּעֹם שִּנּוֲתֵנִים אּצֵבֹּעֹם בֵּפִיהּם: נִטֵמֹא. גֹּרֵסִינָן. כֵּלוֲמָר נִטֵמֹא הָתָּנּוּר מְחַמָת הֹרֲק וֵהָדֹּם :
