פרק ה

תַּנּוּר תְּחִלָּתוֹ אַרְבָּעָה. סְתָם תַּנּוּר עָשׂוּי כְּמִין קְדֵרָה גְדוֹלָה שֶׁאֵין לָהּ שׁוּלַיִם, וּמְחַבְּרִים אוֹתוֹ בְטִיט עַל גַּבֵּי קַרְקַע, וְרִצְפַּת הַקַּרְקַע הִיא תַחְתִּיתוֹ שֶׁל תַּנּוּר . וְאַף עַל פִּי שֶׁמֵּעַצְמוֹ אֵין לוֹ בֵית קִבּוּל, אַף עַל פִּי כֵן טָמֵא, דִּבְתוֹכוֹ תָלָה רַחֲמָנָא וַהֲרֵי יֵשׁ לוֹ תוֹךְ: תְּחִלָּתוֹ אַרְבָּעָה. אֵין גָּבְהוֹ שֶׁל תַּנּוּר פָּחוֹת מֵאַרְבָּעָה טְפָחִים: וּשְׁיָרָיו. אִם הָיָה גָדוֹל וְנִטְמָא וְאַחַר כָּךְ נִשְׁבַּר, לֹא טָהֵר עַד שֶׁלֹּא יִהְיֶה בִשְׁבָרָיו אַרְבָּעָה: בַּמֶּה דְבָרִים אֲמוּרִים. דִּתְחִלָּתוֹ וּשְׁיָרָיו אַרְבָּעָה: בְּגָדוֹל. הֶעָשׂוּי לֶאֱפוֹת וְלִצְלוֹת בּוֹ כְדֶרֶךְ הַתַּנּוּרִים: אֲבָל בְּקָטָן. הֶעָשׂוּי לְתִינוֹקוֹת לְצַחֵק בּוֹ: תְּחִלָּתוֹ כָּל שֶׁהוּא. וְלָאו דַּוְקָא כָל שֶׁהוּא, דִּבְצִיר מִטֶּפַח לֹא הָוֵי : וּשְׁיָרָיו בְּרֻבּוֹ. בַּגְּמָרָא בְּפֶרֶק הָעוֹר וְהָרֹטֶב מַקְשֶׁה עַל מַתְנִיתִין דְּהָכָא, וּבְצִיר מִטֶּפַח לְמַאי חֲזִי. וּמְשָׁנֵי, הַאי וּשְׁיָרָיו בְּרֻבּוֹ אַגָּדוֹל קָאֵי, וְלֹא אַתַּנּוּר קָטָן. וּמוֹתְבִינַן, בְּגָדוֹל, הָא קָאָמַר וּשְׁיָרָיו אַרְבָּעָה, דְּהָא קָתָנֵי בַּמֶּה דְבָרִים אֲמוּרִים בְּגָדוֹל מַשְׁמַע דְּלֹא פְלִיגֵי חֲכָמִים עַל תַּנָּא קַמָּא בְגָדוֹל כְּלָל. וּמְשָׁנֵי, הָא בְתַנּוּר בַּר שִׁבְעָה הָא בְתַנּוּר בַּר תִּשְׁעָה, כְּלוֹמַר בַּר שִׁבְעָה שְׁיָרָיו בְּרֻבּוֹ, וְאַף עַל פִּי דְלָא הָווּ אַרְבָּעָה וְאֵין תַּנּוּר פָּחוֹת מֵאַרְבָּעָה, הוֹאִיל וּמִתְּחִלָּתוֹ הָיָה כְלִי חָשׁוּב מִשְׁתַּמְּשִׁין בִּשְׁיָרָיו אַף עַל פִּי שֶׁאֵינָן רְאוּיִין כָּל כָּךְ. וּבְתַנּוּר שֶׁל תִּשְׁעָה דְרֻבּוֹ הָוֵי שִׁעוּר גָּדוֹל, מְטַמֵּא בִּשְׁיָרָיו בְּאַרְבָּעָה, כִּתְחִלָּתוֹ: מִשֶּׁתִּגָּמֵר מְלַאכְתּוֹ. אַתְּחִלַּת תַּנּוּר קָאֵי. מֵאֵימָתַי הַתַּנּוּר מִטַּמֵּא, מִשֶּׁתִּגָּמֵר מְלַאכְתּוֹ: מִשֶּׁיַּסִּיקֶנּוּ. אַף עַל גַּב דִּשְׁאָר כְּלֵי חֶרֶס סַגִּי לְהוּ בְצֵרוּף כִּבְשָׁן, תַּנּוּר דְּבָעֵי לְדַבֵּק בּוֹ טִיט לְעַבּוֹתוֹ לְהַחֲזִיק חֻמּוֹ לֹא נִגְמְרָה מְלַאכְתּוֹ עַד שֶׁיַּסִּיקֶנּוּ . וּמִיהוּ לֹא בָעֵי הֶסֵּק גָּדוֹל כְּדֵי לֶאֱפוֹת בּוֹ פַת, אֶלָּא כְדֵי לֶאֱפוֹת בּוֹ סֻפְגָּנִים, פַּת שֶׁבְּלִילָתָהּ רַכָּה הָעֲשׂוּיָה כִסְפוֹג, דִּבְהֶסֵּק מֻעָט סַגִּי לַהּ: מִשֶּׁיַּסִּיק אֶת הֶחָדָשׁ כְּדֵי לֶאֱפוֹת בַּיָּשָׁן. דְּחָדָשׁ כְּדֵי לֶאֱפוֹת בּוֹ סֻפְגָּנִים בָּעֵי הֶסֵּק מְרֻבֶּה מִן הַיָּשָׁן. וְאִם הֻסַּק הֶחָדָשׁ הֶסֵּק מֻעָט דִּבְהֶסֵּק כָּזֶה הָיָה יָשָׁן רָאוּי לֶאֱפוֹת בּוֹ סֻפְגָּנִים, הַיְנוּ גְמַר מְלַאכְתּוֹ. וְאֵין הֲלָכָה כְּרַבִּי יְהוּדָה:

כִּירָה. מְקוֹם שְׁפִיתַת שְׁתֵּי קְדֵרוֹת. וּפְעָמִים שֶׁנּוֹתְנִים הַקְּדֵרוֹת לְבַשֵּׁל בְּתוֹכָהּ וּפְעָמִים עַל גַּבָּהּ: תְּחִלָּתָהּ שָׁלֹשׁ. גָּבְהָהּ אֵין פָּחוֹת מִשָּׁלֹשׁ אֶצְבָּעוֹת: מִשֶּׁתִּגָּמֵר מְלַאכְתָּהּ. מֵאֵימָתַי מִטַּמְּאָה תְחִלָּתָהּ, מִשֶּׁתִּגָּמֵר מְלַאכְתָּה: בֵּיצָה קַלָּה. בֵּיצַת תַּרְנְגֹלֶת. וְאַמַּאי קָרֵי לַהּ בֵּיצָה קַלָּה, שֶׁהִיא קַלָּה לְבַשֵּׁל יוֹתֵר מִכָּל הַבֵּיצִים: טְרוּפָה. כְּשֶׁהִיא טְרוּפָה וּשְׁבוּרָה בַקְּעָרָה: וּנְתוּנָה בְאִלְפָּס. שֶׁהוּחַם כְּבָר, שֶׁהִיא מְמַהֶרֶת לְהִתְבַּשֵּׁל: הַכֻּפָּח. שִׁעוּר שְׁפִיתַת קְדֵרָה אַחַת. וְחֻמּוֹ רַב מֵחֹם הַכִּירָה. וּפְעָמִים שֶׁאוֹפִין בּוֹ פַת: שִׁעוּרוֹ כַּתַּנּוּר. כְּדֵי לֶאֱפוֹת סֻפְגָּנִים : שִׁעוּרוֹ כַּכִּירָה. לְבַשֵּׁל בּוֹ בֵיצָה קַלָּה: הָאֶבֶן הַיּוֹצֵא. רְגִילִים לְחַבֵּר אֶבֶן בַּתַּנּוּר שֶׁבּוֹלֶטֶת מִמֶּנּוּ לַחוּץ כְּדֵי שֶׁיִּהְיֶה לוֹ לְבֵית אֲחִיזָה, וּכְתַנּוּר דַּמְיָא. וְאִם נִטְמָא הַתַּנּוּר נִטְמֵאת הָאֶבֶן, וְנִטְמָא הַפַּת שֶׁנָּגַע בָּהּ, מִפְּנֵי שֶׁהוּא יָד לַתַּנּוּר, וּכְתִיב (וַיִּקְרָא יא) תַּנּוּר וְכִירַיִם יֻתָּץ וְגוֹ' טְמֵאִים יִהְיוּ לָכֶם, לְכָל שֶׁבְּצָרְכֵיכֶם, לְרַבּוֹת אֶת הַיָּדוֹת שֶׁהֵן צֹרֶךְ לַכְּלִי: לֹא אָמְרוּ. דְּיוֹתֵר מִטֶּפַח בַּתַּנּוּר וּמִשָּׁלֹשׁ אֶצְבָּעוֹת בַּכִּירָה לֹא הָוֵי חִבּוּר, אֶלָּא כְשֶׁהָאֶבֶן בֵּין הַתַּנּוּר וְלַכֹּתֶל, מִפְּנֵי שֶׁמְּעַכֶּבֶת שֶׁאֵין יְכוֹלִין לְקָרֵב הַתַּנּוּר לַכֹּתֶל וְדוֹחֵק אֶת הַבַּיִת, הִלְכָּךְ יוֹתֵר מִטֶּפַח עוֹמֶדֶת לִנָּטֵל הִיא. אֲבָל לְצַד הַבַּיִת, אֲפִלּוּ גְדוֹלָה הַרְבֵּה, יָד הִיא לַתַּנּוּר. וְאֵין הֲלָכָה כְּרַבִּי יְהוּדָה: וְהַשְּׁאָר טָהוֹר. דְּתַנְיָא בְתוֹרַת כֹּהֲנִים, יָכוֹל שֶׁאֲנִי מְרַבֶּה יוֹתֵר מִכַּשִּׁעוּר, תַּלְמוּד לוֹמַר טְמֵאִים הֵם:

עֲטֶרֶת כִּירָה. בִּנְיָן עָגֹל עָשׂוּי עַל רֹאשׁ הַכִּירָה כְדֵי לִשְׁמֹר חֻמָּהּ: טְהוֹרָה. וְאִם נִטְמֵאת הַכִּירָה לֹא נִטְמֵאת הָעֲטֶרֶת, דְּלָא חֲשִׁיבָא עֲטֶרֶת חִבּוּר לַכִּירָה: טִירַת הַתַּנּוּר. חֲצַר הַתַּנּוּר, לְשׁוֹן בַּחֲצֵרֵיהֶם וּבְטִירוֹתָם. מָקוֹם לִפְנֵי הַתַּנּוּר מֻקָּף מְחִצּוֹת שֶׁנּוֹתְנִים בּוֹ הַלֶּחֶם כְּשֶׁרוֹדִין אוֹתוֹ מִן הַתַּנּוּר: בִּזְמַן שֶׁהִיא גְבוֹהָה אַרְבָּעָה. כְּשִׁעוּר גָּבְהוֹ שֶׁל תַּנּוּר שֶׁתְּחִלָּתוֹ אַרְבָּעָה: אֲפִלּוּ עַל שָׁלֹשׁ אֲבָנִים. שֶׁלֹּא הָיְתָה מְחֻבֶּרֶת חִבּוּר יָפֶה אֶלָּא מֻנַּחַת עַל שָׁלֹשׁ אֲבָנִים בִּלְבַד וּדְבֻקָּה לַתַּנּוּר, חֲשִׁיבָא חִבּוּר: בֵּית הַפַּךְ. מָקוֹם עָשׂוּי בְּצִדֵּי הַכִּירָה שֶׁמַּנִּיחִין בּוֹ פַּךְ שֶׁמֶן לְהַפְשִׁיר: בֵּית הַתְּבָלִין. מָקוֹם שֶׁמַּנִּיחִין בּוֹ אֶת הַתְּבָלִין: בֵּית הַנֵּר. מָקוֹם שֶׁמַּנִּיחִין בּוֹ הַנֵּר: מִטַּמְּאִין בְּמַגָּע. אִם נָגַע שֶׁרֶץ בְּתוֹךְ הַכִּירָה נִטְמְאוּ הַמְּקוֹמוֹת הַלָּלוּ בְטֻמְאַת הַכִּירָה. וְכֵן אִם נִטְמָא אֶחָד מִן הַמְּקוֹמוֹת הַלָּלוּ בְמַגַּע שֶׁרֶץ נִטְמֵאת הַכִּירָה עִמָּהֶם, דַּחֲשִׁיבֵי כְכִירָה וְהֵם דָּבָר אֶחָד לְעִנְיַן טֻמְאַת מַגָּע: וְאֵין מִטַּמְּאִין בַּאֲוִיר. אִם נִתְלָה שֶׁרֶץ בַּאֲוִיר כִּירָה וְלֹא נָגַע בָּהּ, אַף עַל פִּי שֶׁהִיא טְמֵאָה, דִכְלִי חֶרֶס מִטַּמֵּא בַאֲוִיר, הֵן טְהוֹרִין, שֶׁאֵינָם חִבּוּר לָהּ אֶלָּא לְעִנְיַן מַגָּע. מִשּׁוּם דְּמִן הַתּוֹרָה אֵינָם חִבּוּר לָהּ, וְרַבָּנָן הוּא דְגַזּוּר עֲלַיְהוּ טֻמְאָה, וַעֲבַדוּ רַבָּנָן הֶכֵּרָא דְלָא לִטַּמְּאוּ בַאֲוִיר, כִּי הֵיכִי דְלָא לִשְּׂרְפוּ עֲלַיְהוּ תְרוּמָה וְקָדָשִׁים: וְרַבִּי שִׁמְעוֹן מְטַהֵר. אֲפִלּוּ בְמַגָּע, כָּל מָקוֹם שֶׁלֹּא נָגְעָה בוֹ טֻמְאָה, דְּכָל חַד וְחַד כְּלִי בְאַנְפֵּי נַפְשֵׁיהּ הוּא. וַהֲלָכָה כְּרַבִּי מֵאִיר:

תַּנּוּר שֶׁהֻסַּק מֵאֲחוֹרָיו. שֶׁאֵין הַתַּנּוּר מְקַבֵּל טֻמְאָה עַד שֶׁיַּסִּיקֶנּוּ, וְאִם לֹא הֻסַּק מִבִּפְנִים אֶלָּא מִבַּחוּץ, חֲשִׁיב הֶסֵּק: אוֹ שֶׁהֻסַּק שֶׁלֹּא לְדַעְתּוֹ. : אוֹ שֶׁהֻסַּק בְּבֵית הָאֻמָּן. קֹדֶם שֶׁנִּגְמַר מְלֶאכֶת עֲשִׂיָּתוֹ: טָמֵא. גָּרְסִינַן. וְיֵשׁ סְפָרִים שֶׁגּוֹרְסִין טָהוֹר, וְשִׁבּוּשׁ הוּא :

מוּסַף הַתַּנּוּר. תּוֹסַפְתּוֹ שֶׁל תַּנּוּר שֶׁמּוֹסִיפִין בְּגָבְהוֹ כְדֵי לִשְׁמֹר חֻמּוֹ: מִפְּנֵי שֶׁהוּא סוֹמֵךְ עָלָיו אֶת הַשַּׁפּוּד. שֶׁל בָּשָׂר, שֶׁרְגִילִים לִצְלוֹת בּוֹ בָשָׂר, וְהָוֵי דְבַר צֹרֶךְ לַתַּנּוּר. וְרַחֲמָנָא אָמַר וּטְמֵאִים יִהְיוּ לָכֶם, לְכָל שֶׁבְּצָרְכֵיכֶם, כִּדְדָרְשִׁינַן לְעֵיל : מִפְּנֵי שֶׁהוּא אוֹפֶה בוֹ כְּשֶׁהוּא נִדְחָק. כְּשֶׁיֵּשׁ לוֹ אֻכְלוּסִין מְרֻבִּין וְצָרִיךְ לְפַת מְרֻבָּה. וְרַבִּי יוֹחָנָן לֹא סַגִּי לֵיהּ בְּטַעֲמָא דְתַנָּא קַמָּא מִפְּנֵי שֶׁהוּא סוֹמֵךְ עָלָיו אֶת הַשַּׁפּוּד, דִּסְבִירָא לֵיהּ דְּבָעִינַן שֶׁיְּהֵא תַשְׁמִישׁ הַתּוֹסֶפֶת כְּעֵין תַּשְׁמִישׁ גּוּפוֹ שֶׁל תַּנּוּר: מוּסַף הַיּוֹרָה שֶׁל שׁוֹלְקֵי זֵיתִים. דֶּרֶךְ שׁוֹלְקֵי זֵיתִים וְצַבָּעִים שֶׁיֵּשׁ לָהֶם יוֹרוֹת גְּדוֹלוֹת וְעוֹשִׂים לָהֶם תּוֹסֶפֶת טִיט עַל שְׂפָתָם שֶׁיַּעֲלוּ בָהּ הַמַּיִם בְּעֵת רְתִיחָתָם. וְשֶׁל שׁוֹלְקֵי זֵיתִים טָמֵא, לְפִי שֶׁאוֹתָהּ תּוֹסֶפֶת צְרִיכָה לַכְּלִי וּמִשְׁתַּמְּשִׁים בָּהּ: וְשֶׁל צַבָּעִים טָהוֹר. דְּלֹא מִשְׁתַּמֵּשׁ בֵּיהּ כִּי הֵיכִי דְלָא לִפְסִיד צִבְעֵיהּ. וְאֵין הֲלָכָה כְּרַבִּי יוֹחָנָן הַסַּנְדְּלָר:

מֵעָפָר וּלְמַטָּן מִטַּמֵּא בְמַגָּע. אִם נָגַע שֶׁרֶץ בַּתַּנּוּר מִתּוֹכוֹ לְמַעְלָה מִן הֶעָפָר נִטְמָא מַה שֶּׁלְּמַטָּה מִן הֶעָפָר. אֲבָל אִם נִתְלָה הַשֶּׁרֶץ בָּאֲוִיר וְלֹא נָגַע בְּדָפְנֵי הַתַּנּוּר מִתּוֹכוֹ, לֹא נִטְמָא אֶלָּא מַה שֶּׁלְּמַעְלָה מִן הֶעָפָר: בּוֹר. חֲפִירָה לְמַטָּה בָּאָרֶץ: דּוּת. כְּבוֹר הוּא עָשׂוּי, אֶלָּא שֶׁהוּא בִנְיָן לְמַעְלָה מִן הַקַּרְקַע: וְנָתַן שָׁם אֶבֶן. בֵּין הַבּוֹר וּבֵין דֹּפֶן הַתַּנּוּר, לְהַחֲזִיק הַתַּנּוּר שֶׁלֹּא יִפֹּל לְתוֹךְ הַבּוֹר: אִם מַסִּיק מִלְּמַטָּה. אִם כְּשֶׁמַּסִּיק הָאוּר מִלְּמַטָּה בְּתוֹךְ הַבּוֹר, הַתַּנּוּר נִסֹּק מִתּוֹכוֹ לְמַעְלָה דֶרֶךְ שׁוּלָיו , טָמֵא. וְאִם לָאו, שֶׁאֵינוֹ נִסֹּק מֵהֶסֵּק הָאוּר שֶׁלְּמַטָּה בְתוֹךְ הַבּוֹר, טָהוֹר. וּפְלֻגְתָּא דְרַבִּי יְהוּדָה וְרַבָּנָן דַּוְקָא בְהֶסֵּק רִאשׁוֹן שֶׁהוּא גְמַר מְלַאכְתּוֹ שֶׁל תַּנּוּר וְעַל יְדֵי אוֹתוֹ הֶסֵּק הוּא מְקַבֵּל טֻמְאָה, רַבִּי יְהוּדָה סָבַר צָרִיךְ שֶׁיְּהֵא הַתַּנּוּר מְחֻבָּר לָאָרֶץ בְּהֶסֵּק רִאשׁוֹן, דִּכְתִיב תַּנּוּר וְכִירַיִם יֻתָּץ, וְאֵין שַׁיָּךְ לְשׁוֹן נְתִיצָה אֶלָּא בִמְחֻבָּר לַקַּרְקַע, דִּבְמִטַּלְטֵל נוֹפֵל לְשׁוֹן שְׁבִירָה לֹא לְשׁוֹן נְתִיצָה, הִלְכָּךְ אִם הַתַּנּוּר נִסֹּק דֶּרֶךְ שׁוּלָיו מֵהֶסֵּק שֶׁבְּתוֹךְ הַבּוֹר, חֲשִׁיב כִּמְחֻבָּר לַבּוֹר וְקָרִינַן בֵּיהּ יֻתָּץ, וְאִם אֵינוֹ נִסֹּק מֵהֶסֵּק הַבּוֹר הֲרֵי הוּא נִתֹּץ וְעוֹמֵד וְאִי אֶפְשָׁר לְקַיֵּם בּוֹ תוּ יֻתָּץ: וַחֲכָמִים אוֹמְרִים הוֹאִיל וְהוּא נִסֹּק מִכָּל מָקוֹם טָמֵא. דְּאָמַר קְרָא וּטְמֵאִים יִהְיוּ לָכֶם, מִכָּל מָקוֹם, וַאֲפִלּוּ אִם מַסִּיקוֹ כְשֶׁהוּא תָלוּי בְצַוַּאר הַגָּמָל הָוֵי הֶסֵּק וְנִגְמְרָה מְלַאכְתּוֹ בְכָךְ. וְהַכֹּל מוֹדִים בְּהֶסֵּק שֵׁנִי לְאַחַר שֶׁנִּגְמְרָה מְלַאכְתּוֹ שֶׁהוּא מִטַּמֵּא אֲפִלּוּ בְצַוַּאר הַגָּמָל. וַהֲלָכָה כַּחֲכָמִים:

תַּנּוּר שֶׁנִּטְמָא. הַתַּנּוּר עָשׂוּי כִקְדֵרָה גְדוֹלָה שֶׁאֵין לָהּ שׁוּלַיִם וּמִטַּלְטֵל, וּכְשֶׁבָּא לְקָבְעוֹ מְחַבְּרוֹ בָאָרֶץ וּמְדַבֵּק טִיט מִבַּחוּץ וּמְעַבֵּהוּ כְדֵי שֶׁיִּשְׁמֹר חֻמּוֹ, וְקָרֵי לַהּ טְפֵלָה, שֶׁמְּטַפְּלָהּ וּמְחַבְּרָהּ לוֹ: כֵּיצַד מְטַהֲרִים אוֹתוֹ. כַּמָּה תְהֵא שְׁבִירָתוֹ וּתְהֵא מְטַהַרְתּוֹ: חוֹלְקוֹ לִשְׁלֹשָׁה. לַכְּלִי עַצְמוֹ מִבִּפְנִים. דְּאִי לִשְׁנַיִם, הַגָּדוֹל טָמֵא, דְּאִיכָּא רֻבָּא : וְגוֹרֵר אֶת הַטְּפֵלָה. שֶׁדָּבֵק עָלָיו, שֶׁגַּם הִיא מְחַבַּרְתּוֹ: עַד שֶׁיְּהֵא בָאָרֶץ. כְּלוֹמַר מַפִּילָהּ עַד הַקַּרְקַע. אִי נַמִּי, אַחוֹלְקוֹ קָאֵי, דְּלֹא סַגִּי בַחֲלֻקָּה עַד רֻבּוֹ, אֶלָּא חוֹלֵק עַד שֶׁיַּגִּיעַ לַקַּרְקַע : וְלֹא עַד שֶׁיְּהֵא בָאָרֶץ. וְלֹא לְחַלְּקוֹ עַד לְמַטָּה סָמוּךְ לָאָרֶץ: אֶלָּא מְמַעֲטוֹ מִבִּפְנִים. עַד שֶׁלֹּא יִשָּׁאֵר אַרְבָּעָה טְפָחִים שֶׁהוּא שִׁעוּר תְּחִלָּתוֹ שֶׁל תַּנּוּר. וְאֵין הֲלָכָה כְּרַבִּי מֵאִיר: צָרִיךְ לְהַסִּיעוֹ. אַף עַל פִּי שֶׁחִלְּקוֹ, לֹא טָהֵר עַד שֶׁיַּפְרִיד הַחֲלָקִים זֶה מִזֶּה. וְאֵין הֲלָכָה כְּרַבִּי שִׁמְעוֹן: הַגָּדוֹל טָמֵא. מִשּׁוּם דְּאִיכָּא רֻבּוֹ שֶׁל תַּנּוּר:

חֲתָכוֹ חֻלְיוֹת לְרָחְבּוֹ. כְּמוֹ שֶׁחוֹתֵךְ שְׁפוֹפֶרֶת שֶׁל קָנֶה: פָּחוֹת מֵאַרְבָּעָה טְפָחִים. אִם יֵשׁ בְּחֻלְיָא פָּחוֹת מֵאַרְבָּעָה טְפָחִים וּבָאָה מִתַּנּוּר גָּדוֹל דְּלֵיכָּא רֻבָּא, טָהוֹר. אֲבָל אִיכָּא אַרְבָּעָה טְפָחִים, טָמֵא, אַף עַל גַּב דַּהֲוֵי מִעוּט: מֵרְחוֹ בְטִיט. בָּעֵי הֶסֵּק כְּתַנּוּר חָדָשׁ: הִרְחִיק מִמֶּנּוּ אֶת הַטְּפֵלָה. טִפּוּל שֶׁל טִיט שֶׁעָשָׂה, לֹא חִבְּרוֹ לַתַּנּוּר אֶלָּא רָחוֹק מִמֶּנּוּ: וְנָתַן חֹל אוֹ צְרוֹר. בֵּין הַטְּפֵלָה לַתַּנּוּר, וְעַל יְדֵי כָּךְ הַתַּנּוּר מְקַיֵּם חֻמּוֹ וְאוֹפִים בּוֹ: הַנִּדָּה וְהַטְּהוֹרָה אוֹפוֹת בּוֹ. דְּלָא חֲשִׁיבָא הַטְּפֵלָה חִבּוּר לַתַּנּוּר, כֵּיוָן שֶׁהִיא מְרֻחֶקֶת מִמֶּנּוּ: וְהוּא טָהוֹר. שֶׁאֵינוֹ מְקַבֵּל טֻמְאָה:

תַּנּוּר שֶׁבָּא מְחֻתָּךְ. שֶׁעֲשָׂאוֹ הָאֻמָּן דַּפִּים דַּפִּים כְּדֵי לְחַבְּרָן יַחַד עַל יְדֵי מִסְגְּרוֹת, הֵן זִירִים שֶׁקּוֹרִין צירקלא"ש, כְּעֵין שֶׁאָנוּ עוֹשִׂים לֶחָבִיּוֹת שֶׁנּוֹתְנִים בָּהֶן יַיִן. וּלְאוֹתָן זִירִים וּמִסְגְּרוֹת קוֹרֵא הַתַּנָּא לִמּוּדִין : וּנְתָנָן עָלָיו וְהוּא טָהוֹר נִטְמָא. לְאַחַר שֶׁנָּתַן עָלָיו הַלִּמּוּדִים וְהָיָה הַתַּנּוּר טָהוֹר מִתְּחִלָּה לְפִי שֶׁהָיָה דַפִּים דַּפִּים וְלֹא הָיָה מְחֻבָּר, חָזַר לִהְיוֹת מְקַבֵּל טֻמְאָה עַל יְדֵי הַלִּמּוּדִים הַלָּלוּ שֶׁמְּחַבְּרִים אוֹתוֹ: סִלֵּק אֶת לִמּוּדָיו. וְנִתְפָּרְקוּ הַדַּפִּין זֶה מִזֶּה, אֵין לְךָ נְתִיצָה גְדוֹלָה מִזּוֹ, וְנִטְהַר הַתַּנּוּר. וְאַף עַל פִּי שֶׁהֶחֱזִיר הַלִּמּוּדִים פַּעַם שְׁנִיָּה שׁוּב אֵינוֹ מְקַבֵּל טֻמְאָה, שֶׁאֵין הַלִּמּוּדִים מְבִיאִין לוֹ הַטֻּמְאָה אֶלָּא בְּפַעַם רִאשׁוֹנָה בִלְבַד כְּשֶׁבָּא מְחֻתָּךְ מִבֵּית הָאֻמָּן: מֵרְחוֹ בְטִיט. לְתַנּוּר זֶה שֶׁאֵין הַלִּמּוּדִים מְבִיאִין לוֹ הַטֻּמְאָה, טָמֵא עַל יְדֵי מְרִיחַת הַטִּיט. וְאֵין צָרִיךְ לְהַסִּיקוֹ. כִּשְׁאָר כָּל תַּנּוּרִים שֶׁאֵין מְקַבְּלִין טֻמְאָה עַד שֶׁיַּסִּיקֵם, לְפִי שֶׁכְּבָר הֻסַּק בְּפַעַם רִאשׁוֹנָה כְּשֶׁבָּא מְחֻתָּךְ מִבֵּית הָאֻמָּן:

חֲתָכוֹ חֻלְיוֹת. שֶׁחֲתָכוֹ לַתַּנּוּר בְּרָחְבּוֹ לְחֻלְיוֹת , וְהִנִּיחַ חֻלְיָא עַל גַּבֵּי חֻלְיָא: וְנָתַן חֹל בֵּין חֻלְיָא לְחֻלְיָא. וְטָח אוֹתוֹ בְּטִיט וְעָשָׂה לוֹ טְפֵלָה כְדֶרֶךְ שֶׁעוֹשִׂין לְתַנּוּרִים : רַבִּי אֱלִיעֶזֶר מְטַהֵר. דְּכֵיוָן דְּיֵשׁ חֹל בֵּין הַסְּדָקִים, כְּשָׁבוּר דָּמֵי: וַחֲכָמִים מְטַמְּאִין. שֶׁהַטְּפֵלָה עוֹשָׂה אֶת כֻּלּוֹ אֶחָד וּמְחַבֶּרֶת אֶת הַסְּדָקִים, וְאַף עַל פִּי שֶׁיֵּשׁ חֹל בֵּין חֻלְיָא לְחֻלְיָא: וְזֶהוּ תַנּוּרוֹ שֶׁל עַכְנָאי. לְכָךְ נִקְרָא תַנּוּרוֹ שֶׁל עַכְנַאי, שֶׁהִקִּיפוּהוּ הֲלָכוֹת כְּעַכְנָא זוֹ וְטִמְּאוּהוּ. עַכְנָא, נָחָשׁ שֶׁדַּרְכּוֹ לֵעָשׂוֹת כַּעֲגֻלָּה לְהַכְנִיס זְנָבוֹ לְתוֹךְ פִּיו, כָּךְ הִקִּיפוּ בִתְשׁוּבָה וּרְאָיוֹת רַבִּי אֱלִיעֶזֶר וַחֲכָמִים זֶה אֶת זֶה, עַל תַּנּוּר זֶה. וּלְבַסּוֹף מִפְּנֵי שֶׁהָיָה רַבִּי אֱלִיעֶזֶר יָחִיד כְּנֶגֶד רַבִּים, נִפְסְקָה הַהֲלָכָה כְּרַבִּים דְּלֹא כְרַבִּי אֱלִיעֶזֶר. וְהֵבִיאוּ כָל טָהֳרוֹת שֶׁטִּהֵר רַבִּי אֱלִיעֶזֶר וּשְׂרָפוּם בָּאֵשׁ וְנִמְנוּ וּבֵרְכוּהוּ, מִפְּנֵי שֶׁהָיָה מַחֲזִיק בַּמַּחֲלֹקֶת כְּנֶגֶד רַבִּים: יוֹרוֹת הָעַרְבִיִּין. לֶאֱפוֹת פַּת הֵן עֲשׂוּיִין. כְּדִמְפָרֵשׁ וְאָזֵיל: שֶׁל בֶּן דִּינָאי. רַבּוֹתַי פֵּרְשׁוּ שֵׁם אָדָם שֶׁהָיָה עוֹשֶׂה תַנּוּרִים הַלָּלוּ. וַאֲנִי שָׁמַעְתִּי עַל שֵׁם שֶׁדָּנוּ בוֹ דִינִים הַרְבֵּה :

שֶׁל אֶבֶן וְשֶׁל מַתֶּכֶת טָהוֹר. שֶׁל אֶבֶן טָהוֹר לְגַמְרֵי. וְשֶׁל מַתֶּכֶת טָהוֹר מִדִּין כְּלֵי חֶרֶס, שֶׁאֵינוֹ מִטַּמֵּא מֵאֲוִירוֹ, וְיֵשׁ לוֹ טָהֳרָה בְמִקְוֶה: וְטָמֵא מִשּׁוּם כְּלֵי מַתָּכוֹת. שֶׁמְּקַבֵּל טֻמְאָה מִגַּבּוֹ וְנַעֲשֶׂה אַב הַטֻּמְאָה . וְאִם הוּא מְחֻבָּר לַקַּרְקַע אֵינוֹ מִטַּמֵּא, דְּכָל הַמְחֻבָּר לַקַּרְקַע הֲרֵי הוּא כַקַּרְקַע, חוּץ מִתַּנּוּר וְכִירַיִם שֶׁל חֶרֶס דִּכְתִיב בְּהוּ (וַיִּקְרָא יא) תַּנּוּר וְכִירַיִם יֻתָּץ, אֶת שֶׁיֵּשׁ בּוֹ נְתִיצָה: נִקַּב נִפְגַּם נִסְדַּק. וְעָשָׂה טִפּוּל שֶׁל סִיד לִסְתֹּם הַנֶּקֶב וְהַפְּגָם וְהַסֶּדֶק , אוֹ שֶׁעָשָׂה עַל פִּיו תּוֹסֶפֶת שֶׁל טִיט, טָמֵא, כְּדִין תַּנּוּר שֶׁל חֶרֶס: וְכֵן בְּכִירָה. וְכֵן שִׁעוּר נֶקֶב שֶׁל כִּירָה, כְּדֵי שֶׁיֵּצֵא מִמֶּנּוּ הָאוּר, כְּשִׁעוּר נֶקֶב שֶׁל תַּנּוּר: וְעָשָׂה לָהּ פִּטְפּוּטִין. הָכָא לֹא תָנֵי וְעָשָׂה לָהּ טְפֵלָה, שֶׁאֵין עוֹשִׂין טְפֵלָה לְכִירָה שֶׁנִּקְּבָה אוֹ נִפְגְּמָה כְּמוֹ שֶׁעוֹשִׂין לְתַנּוּר, אֶלָּא כְשֶׁנִּקְּבָה אוֹ נִפְגְּמָה הַכִּירָה בִּמְקוֹם מוֹשָׁבָהּ רְגִילִין בִּמְקוֹם הַנֶּקֶב לַעֲשׂוֹת פִּטְפּוּטִין, הֵן רַגְלֵי הַכִּירָה, כְּמוֹ טריפי"ד בְּלַעַ"ז, שֶׁהַכִּירָה יוֹשֶׁבֶת עַל אוֹתָן פִּטְפּוּטִין : מֵרְחָהּ בְּטִיט. טְהוֹרָה. שֶׁאֵין מֵרוּחַ שֶׁל טִיט מוֹעִיל אֶלָּא בְתַנּוּר שֶׁאוֹפִין הַלֶּחֶם בְּתוֹכוֹ, אֲבָל לֹא בְכִירָה, לְפִי שֶׁאֵין אוֹפִין וּמְבַשְּׁלִין בְּתוֹךְ הַכִּירָה עַצְמָהּ אֶלָּא מַנִּיחִין בְּתוֹכָהּ אוֹ עַל גַּבָּהּ הַקְּדֵרָה . וְאֵין הֲלָכָה כְּרַבִּי יְהוּדָה:
