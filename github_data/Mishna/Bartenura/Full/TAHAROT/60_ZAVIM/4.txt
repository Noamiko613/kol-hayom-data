פרק ד

רַבִּי יְהוֹשֻׁעַ אוֹמֵר כִּפָּה שֶׁבְּרֹאשָׁהּ. לִרְבוּתָא נְקָטָהּ, דַּאֲפִלּוּ צָעִיף שֶׁבְּרֹאשׁ הַטְּהוֹרָה נֶחְשָׁב כְּאִלּוּ יָשְׁבָה עָלָיו הַנִּדָּה, אַף עַל פִּי שֶׁאֵינָהּ יוֹשֶׁבֶת עָלָיו. וּפָלֵיג אַסְּתָם מַתְנִיתִין דְּפִרְקִין דִּלְעֵיל דְּלֹא מְטַמֵּא אֶלָּא בִגְדֵיהֶן וְלֹא כִפָּה שֶׁבְּרֹאשָׁן: כֵּלִים שֶׁבְּרֹאשׁ הַנֵּס. הָא נַמִּי לִרְבוּתָא נָקַט, דְּאַף עַל פִּי שֶׁהֵם בַּגֹּבַהּ בְּרֹאשׁ הַנֵּס שֶׁלֹּא בִמְקוֹם דְּרִיסָה, אֲפִלּוּ הָכִי טְמֵאִים מִדְרָס. וּבִסְפִינָה קְטַנָּה וּבְמִטָּה מַחְגֶּרֶת אַיְרֵי, כְּהַהִיא דְרֵישׁ פִּרְקִין דִּלְעֵיל, אֶלָּא דְּתַנָּא דִלְעֵיל לֹא מְטַמֵּא כֵלִים שֶׁבְּרֹאשׁ הַנֵּס. וְאֵין הֲלָכָה כְּרַבִּי יְהוֹשֻׁעַ: בִּזְמַן שֶׁמַּשָּׂאָן כָּבֵד טְמֵאִים מִדְרָס. וְאַף עַל פִּי שֶׁלֹּא נָגְעָה בַבְּגָדִים, דְּחַשְׁבִינָא לַהּ כְּאִלּוּ נִשְׁעֶנֶת עַל הַבְּגָדִים: כְּסוֹסְטְרָא. כְּמוֹ גְּזוּזְטְרָא, כְּמִין נֶסֶר הַבּוֹלֵט מִכֹּתֶל הָעֲלִיָּה לַחוּץ: וְנָפַל כִּכָּר שֶׁל תְּרוּמָה. שֶׁעַל הַכְּסוֹסְטְרָא: טָהוֹר. הַכִּכָּר. וְאַף עַל פִּי שֶׁנָּפַל מֵחֲמַת הֶקֵּשׁ שֶׁל זָב, לָא חֲשִׁיב לֵיהּ כְּהֶסֵּט, מִשּׁוּם דְּכֹחָהּ שֶׁל כְּסוֹסְטְרָא יָפֶה:

מָרִישׁ. קוֹרָה גְדוֹלָה שֶׁבּוֹנִים בַּבִּירָה וְרָאשֵׁי קוֹרוֹת הַגַּג כֻּלָּן סְמוּכוֹת עָלֶיהָ, וְהִיא מֻנַּחַת עַל גַּבֵּי עַמּוּדִים: מַלְבֵּן. כְּמִין רִבּוּעַ שֶׁל קוֹרוֹת קְבוּעוֹת בָּאָרֶץ: צִנּוֹר. מַרְזֵב שֶׁל מַיִם: אַף עַל פִּי שֶׁהוּא עָשׂוּי בַּחֲבָלִים. שֶׁאֵינוֹ קָבוּעַ בָּאָרֶץ אֶלָּא קָשׁוּר בַּחֲבָלִים וְאֵין כֹּחוֹ יָפֶה כָּל כָּךְ: וְעַל הַיָּם. הֶקֵּף שֶׁל עֵץ שֶׁהַקֶּמַח מִתְקַבֵּץ בּוֹ בִּשְׁעַת הַטְּחִינָה: אִצְטְרֻבָּל. עִגּוּל שֶׁל עֵץ שֶׁמַּעֲמִידִין הָרֵחַיִם עָלָיו. וּשְׁנֵיהֶן קְבוּעִים הֵן: חֲמוֹר שֶׁל רֵחַיִם שֶׁל יָד. בִּנְיָן שֶׁל עֵצִים שֶׁהָרֵחַיִם שֶׁל יָד קְבוּעִים בָּהֶן: וּסְאָה שֶׁל רֵחַיִם. מִדָּה גְדוֹלָה וּקְבוּעָה בַקַּרְקַע: קוֹרַת הַבַּלָּנִין. קוֹרָה שֶׁבַּעַל הַמֶּרְחָץ יוֹשֵׁב עָלֶיהָ. וְכָל הָנֵי אִם הִקִּישׁ עֲלֵיהֶן הַזָּב וְנָפַל הַכִּכָּר שֶׁל תְּרוּמָה מִכֹּחַ הַהֶקֵּשׁ, טָהוֹר, דְּכֹחָן יָפֶה וְלָא חֲשִׁיב הֶסֵּט. וּבְקוֹרַת הַבַּלָּנִים נֶחְלְקוּ. וְאֵין הֲלָכָה כְּרַבִּי יוֹסֵי:

הַנֶּגֶר. יָתֵד הַנָּעוּץ אֲחוֹרֵי הַדֶּלֶת בַּנֶּקֶב שֶׁבָּאַסְקֻפָּה: מַנְעוּל. שֶׁסּוֹגְרִים בּוֹ הַפֶּתַח: מָשׁוֹט. שֶׁמַּנְהִיגִין בּוֹ אֶת הַסְּפִינָה. וּלְשׁוֹן מִקְרָא הוּא, כָּל תּוֹפְשֵׂי מָשׁוֹט, בִּיחֶזְקֵאל : קֶלֶת. עִגּוּל שֶׁל עֵץ שֶׁסָּבִיב לָרֵחַיִם הַתַּחְתּוֹנָה לְקַבֵּל הַקֶּמַח הַנִּטְחָן שֶׁלֹּא יִפֹּל לָאָרֶץ, וּמִטַּלְטֵל הוּא: וְעַל סוֹכָה שֶׁכֹּחָהּ רַע. וְעוֹמֶדֶת בְּאִילָן שֶׁכֹּחוֹ יָפֶה: טְמֵאִין. דְּכָל הָנֵי אֵין כֹּחָן יָפֶה, וְכֹחַ הֶקֵּשׁ הַזָּב בְּהוּ חֲשִׁיב הֶסֵּט: מְטַהֲרִים בָּאֵלּוּ. אַשִּׁדָּה תֵּבָה וּמִגְדָּל קָאֵי. לְהָכִי תְנִינְהוּ תַנָּא קַמָּא לְחוֹדַיְהוּ. וְאֵין הֲלָכָה כְּרַבִּי נְחֶמְיָה וְרַבִּי שִׁמְעוֹן:

פֻּנְדִּיּוֹת. כְּעֵין כִּיסִין אֲרֻכִּין : לְאָרְכָּן. שֶׁשּׁוֹכֵב לְאָרְכָּן שֶׁל סַפְסָלִים אוֹ שֶׁל פֻּנְדִּיּוֹת : טְמֵאִים. שֶׁכָּל אֶחָד וְאֶחָד נִשְׁעָן רֻבּוֹ שֶׁל זָב עֲלֵיהֶן, פְּעָמִים עַל זֶה פְּעָמִים עַל זֶה: לְרָחְבָּן. שֶׁשּׁוֹכֵב אָרְכּוֹ לְרָחְבָּן: טְהוֹרִים. דְּאֵין רֻבּוֹ נִשְׁעָן עַל אֶחָד מֵהֶן , וְאֵין מִשְׁכָּב וּמוֹשָׁב לְעוֹלָם, עַד שֶׁיִּשָּׁעֵן רֻבּוֹ עָלָיו, דִּכְתִיב אִם עַל הַמִּשְׁכָּב הוּא , עַד שֶׁיִּנָּשֵׂא רֻבּוֹ עָלָיו: יָשֵׁן. לְרָחְבָּן, וְסָפֵק אִם נִתְהַפֵּךְ לְאָרְכָּן, טְמֵאִין: כִּסְיוֹת. לְשׁוֹן רַבִּים שֶׁל כִּסֵּא : שֶׁתַּחַת הַגּוּף. לְפִי שֶׁרֻבּוֹ נִשָּׂא עָלָיו. וְהָאֲחֵרִים טְמֵאִים מַגַּע הַזָּב. אֲבָל אֵין לָהֶם תּוֹרַת מִשְׁכָּב וּמוֹשָׁב לִהְיוֹת אַב הַטֻּמְאָה לְטַמֵּא אָדָם וְכֵלִים: אִם רְחוֹקִים זֶה מִזֶּה טְהוֹרִים. לְפִי שֶׁאֵין רֻבּוֹ נִשָּׂא לֹא עַל זֶה וְלֹא עַל זֶה. וְאֵין הֲלָכָה כְּרַבִּי שִׁמְעוֹן:

עֶשֶׂר טַלִּיּוֹת. לְשׁוֹן רַבִּים שֶׁל טַלִּית. וְעֶשֶׂר לָאו דַּוְקָא, דַּאֲפִלּוּ אֶלֶף טַלִּיּוֹת וְאֶבֶן גְּדוֹלָה עַל גַּבָּן, וְיָשַׁב הַזָּב עַל הָאֶבֶן, כָּל הַטַּלִּיּוֹת שֶׁתַּחְתֶּיהָ מִטַּמְּאוֹת מוֹשָׁב, דְּמִשְׁכָּב וּמוֹשָׁב מִטַּמְּאִין תַּחַת אֶבֶן מְסָמָא: כָּרַע הַזָּב טְהוֹרִים. מִדִּין מִשְׁכָּב, אֲבָל טְמֵאִין בְּמַשָּׂא הַזָּב: בִּיחִידִי. מִשְׁכָּב אֶחָד: טָמֵא. שֶׁהֲרֵי רֹב הַזָּב נִשָּׂא עָלָיו: וּבִמְרֻבִּים. בְּמִשְׁכָּבוֹת רַבּוֹת בְּכַף מֹאזְנַיִם כְּנֶגֶד הַזָּב: שֶׁאֵין אֶחָד מֵהֶן נוֹשֵׂא אֶת רֻבּוֹ. דְּכִי כָּרְעוּ הֵן, אֵין רֻבּוֹ שֶׁל זָב נִשָּׂא עַל אֶחָד מֵהֶן, אֶלָּא כָּל אֶחָד מֵהַמִּשְׁכָּבוֹת נָשָׂא מִעוּטוֹ שֶׁל זָב:

טְמֵאִין. בֵּין כָּרַע הַזָּב, בֵּין כָּרְעוּ אֳכָלִין וּמַשְׁקִין, נִטְמְאוּ טֻמְאַת הֶסֵּט: וּבַמֵּת הַכֹּל טָהוֹר. בֵּין מוֹשָׁב וּמִשְׁכָּב, בֵּין אֳכָלִים וּמַשְׁקִין, בֵּין כָּרַע הַמֵּת בֵּין כָּרְעוּ הֵם, דְּכָל טֻמְאוֹת הַמַּסִּיטוֹת טְהוֹרוֹת, חוּץ מֵהֶסֵּטוֹ שֶׁל זָב. וְאִם כָּרְעוּ הֵן, נַמִּי טְהוֹרִים, דְּכָל שֶׁהַמֵּת נִשָּׂא עָלָיו כְּגוֹן בְּכַף מֹאזְנַיִם לְבַד מֵאֹהֶל, טָהוֹר, חוּץ מִן הָאָדָם, דְּבִזְמַן שֶׁמַּכְרִיעַ אֶת הַמֵּת, טָמֵא מִפְּנֵי שֶׁהִסִּיטוֹ, כִּדְאָמְרִינַן בְּפֶרֶק בַּתְרָא דְּנִדָּה (דַּף סט), אַטּוּ מֵת מִי לֹא נִטְמָא בְמַשָּׂא: מַדָּף. לְשׁוֹן טֻמְאָה קַלָּה, כְּדִכְתִיב (וַיִּקְרָא כו) קוֹל עָלֶה נִדָּף. וְרַמְבַּ"ם פֵּרֵשׁ, מַדָּף מִלְּשׁוֹן רֵיחוֹ נוֹדֵף, שֶׁרֵיחַ טֻמְאַת הַזָּב הוֹלֶכֶת לְמֵרָחוֹק לְטַמֵּא כָל הַכֵּלִים שֶׁעַל גַּבָּיו אַף עַל פִּי שֶׁלֹּא נָגַע בָּהֶם. אֲבָל לֹא כְטֻמְאַת הַכֵּלִים שֶׁתַּחְתָּיו, שֶׁכֵּלִים עַל גַּבֵּי הַזָּב, אַף עַל פִּי שֶׁיֵּשׁ דְּבָרִים חוֹצְצִים בֵּינָם לְבֵין הַזָּב, כֻּלָּם טְמֵאִים טֻמְאָה קַלָּה לְטַמֵּא אֳכָלִים וּמַשְׁקִין, אֲבָל לֹא לְטַמֵּא אָדָם וְכֵלִים. וְאִלּוּ כֵּלִים שֶׁתַּחְתָּיו אֲפִלּוּ הֵן מֵאָה זֶה עַל גַּב זֶה, הַתַּחְתּוֹן שֶׁבְּכֻלָּם טָמֵא טֻמְאָה חֲמוּרָה לְטַמֵּא אָדָם לְטַמֵּא בְגָדִים כְּמוֹ הָעֶלְיוֹן: וְעַל גַּבָּיו מַדָּף. נְהִי דְּמֵת נַמִּי מְטַמֵּא כֵלִים שֶׁעַל גַּבָּיו טֻמְאַת מַדָּף לְטַמֵּא אֳכָלִים וּמַשְׁקִין, דְּטֻמְאָה בוֹקַעַת וְעוֹלָה, מִכָּל מָקוֹם אִם הַמֵּת בַּבַּיִת סָמוּךְ לִשְׁמֵי קוֹרָה, וְהַמִּשְׁכָּבוֹת בָּעֲלִיָּה, וְנִכְפָּפִים הַנְּסָרִים מִכֹּבֶד הַמִּשְׁכָּבוֹת וּמַכְבִּידִים עַל הַמֵּת, כִּי הַאי גַּוְנָא בַּזָּב טָמֵא מִתַּחְתָּיו לְטַמֵּא אָדָם לְטַמֵּא בְגָדִים, וְעַל גַּבָּיו לְטַמֵּא אֳכָלִים וּמַשְׁקִין, וּבַמֵּת בֵּין עַל גַּבָּיו בֵּין מִתַּחְתָּיו טָהוֹר: מַה שֶּׁאֵין הַמֵּת מְטַמֵּא. דְּמִתַּחְתָּיו אֲפִלּוּ תְהֵא טֻמְאָה בוֹקַעַת וְיוֹרֶדֶת, מִכָּל מָקוֹם אֵין מִשְׁכָּב וּמוֹשָׁב שֶׁתַּחְתָּיו מְטַמֵּא אָדָם לְטַמֵּא בְגָדִים:

שֶׁאֵינָהּ יְכוֹלָה. וְכֵיוָן דְּזֶה אֵינוֹ יָכוֹל וְזֶה אֵינוֹ יָכוֹל, כָּל חַד וְחַד כֻּלֵּיהּ מַעֲשֶׂה עֲבִיד וְאִיכָּא רֻבּוֹ שֶׁל זָב נִשָּׂא עָלָיו: רַבִּי שִׁמְעוֹן מְטַהֵר. דְּבָעֵי שִׁעוּר זִיבָה לְזֶה וְשִׁעוּר זִיבָה לְזֶה, כְּלוֹמַר רֻבּוֹ שֶׁל זָב מַכְבִּיד עַל כָּל אֶחָד וְאֶחָד. וְאָזְדָא לְטַעְמֵיהּ דְּאָמַר לְעֵיל בְּפִרְקִין בִּיחִידִי טָמֵא, וּבִמְרֻבִּים טָהוֹר, שֶׁאֵין אֶחָד נוֹשֵׂא אֶת רֻבּוֹ. וְאֵין הֲלָכָה כְּרַבִּי שִׁמְעוֹן: הָיָה רוֹכֵב עַל גַּבֵּי בְהֵמָה. בְּעוֹמֶדֶת אַיְרֵי, דְּאִי בִמְהַלֶּכֶת, כָּל שָׁעָה הָאַחַת נֶעֱקֶרֶת וְעוֹמֶדֶת עַל שָׁלֹשׁ, וְטָמֵא מַה שֶּׁתַּחְתֵּיהֶן, שֶׁאֵינָהּ יְכוֹלָה לַעֲמֹד עַל שְׁתַּיִם: שֶׁיְּכוֹלָה לַעֲמֹד עַל שָׁלֹשׁ. הִלְכָּךְ כָּל חַד וְחַד הָוֵי לֵיהּ רְבִיעִי , וְאֵינוֹ אֶלָּא מְסַיֵּעַ, וּמְסַיֵּעַ אֵין בּוֹ מַמָּשׁ. אֲבָל טַלִּית אַחַת תַּחַת שְׁתַּיִם, טְמֵאָה, שֶׁאֵינָהּ יְכוֹלָה לַעֲמֹד עַל שְׁתַּיִם: רַבִּי יוֹסֵי אוֹמֵר. אַאַרְבַּע טַלִּיתוֹת תַּחַת אַרְבַּע רַגְלֵי בְהֵמָה קָאֵי: הַסּוּס מְטַמֵּא בְרַגְלָיו. אִם זָב רוֹכֵב עָלָיו וְטַלִּית תַּחַת אַחַת מֵרַגְלָיו שֶׁל סוּס, טְמֵאָה. אֲבָל תַּחַת יָדָיו, טְהוֹרָה, שֶׁאֵין מִשְׁעַנְתּוֹ עַל יָדָיו. וְאֵין הֲלָכָה כְּרַבִּי יוֹסֵי: כֵּלִים שֶׁבָּעֵקֶל טְמֵאִים. דְּמִשְׁעֶנֶת הַקּוֹרָה עַל הָעֵקֶל. וְעֵקֶל הִיא הַכְּפִיפָה שֶׁעוֹצְרִים בָּהּ הַזֵּיתִים וְהִיא עֲשׂוּיָה כְמִין קֻפָּה שֶׁל מַעֲשֵׂה שְׂבָכָה, וְאַחַר שֶׁעוֹצְרִים הַזֵּיתִים וְנִשְׁאָר הַגֶּפֶת שֶׁהוּא הַפְּסֹלֶת שֶׁל זֵיתִים, מַכְנִיסִין אוֹתוֹ בְּאוֹתוֹ עֵקֶל, וְנוֹתְנִים עַל הָעֵקֶל קוֹרָה לְהַכְבִּיד עָלָיו: הַמַּכְבֵּשׁ שֶׁל כּוֹבֵס. לְאַחַר שֶׁכִּבְּסוּ הַבְּגָדִים, מְקַפְּלִין אוֹתָן תַּחַת הַמַּכְבֵּשׁ לְכָבְשָׁן וְלִשְׁמֹר קִפּוּלָן: כֵּלִים שֶׁתַּחְתָּיו טְהוֹרִים. שֶׁאֵין הַיּוֹשֵׁב בְּקָצֶה הָאֶחָד שֶׁל מַכְבֵּשׁ מַכְבִּיד עַל הַכֵּלִים: וְרַבִּי נְחֶמְיָה מְטַמֵּא. סָבַר אִי אֶפְשָׁר שֶׁיֵּשֵׁב בְּקָצֶה הָאֶחָד שֶׁל הַמַּכְבֵּשׁ שֶׁלֹּא יַכְבִּיד עַל הַכֵּלִים . וְאֵין הֲלָכָה כְּרַבִּי נְחֶמְיָה:
