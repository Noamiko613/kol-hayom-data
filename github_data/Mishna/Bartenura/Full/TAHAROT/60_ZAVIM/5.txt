פרק ה - משנה א
הַנּוֹגֵעַ בַּזָּב. וּכְלֵי שֶׁטֶף בְּמַגָּע. דְּכִי הֵיכִי דְּאָדָם הַנּוֹגֵעַ בַּזָּב מְטַמֵּא בְגָדִים בִּשְׁעַת מַגָּעוֹ, כָּךְ מְטַמֵּא כָל כֵּלִים שֶׁנּוֹגֵעַ בָּהֶן בִּשְׁעַת מַגָּעוֹ בַזָּב, חוּץ מֵאָדָם וּכְלֵי חֶרֶס, דְּתַנְיָא בְתוֹרַת כֹּהֲנִים, מִנַּיִן לַעֲשׂוֹת שְׁאָר כֵּלִים כִּבְגָדִים, תַּלְמוּד לוֹמַר (וַיִּקְרָא טו) וְטָמֵא . יָכוֹל יְטַמֵּא אָדָם וּכְלֵי חֶרֶס, תַּלְמוּד לוֹמַר בֶּגֶד , בֶּגֶד הוּא מְטַמֵּא וְאֵינוֹ מְטַמֵּא אָדָם וּכְלֵי חֶרֶס. וְדַוְקָא בִשְׁעַת מַגָּעוֹ בַזָּב. אֲבָל פֵּרַשׁ, לֹא, דִּכְתִיב (וַיִּקְרָא טו) וְאִישׁ אֲשֶׁר יִגַּע בְּמִשְׁכָּבוֹ יְכַבֵּס בְּגָדָיו וְרָחַץ בַּמַּיִם וְטָמֵא, בִּשְׁעַת מַגָּעוֹ מְטַמֵּא בְגָדִים. פֵּרַשׁ, אֵינוֹ מְטַמֵּא בְגָדִים: אֲבָל לֹא בְמַשָּׂא. אִם בִּשְׁעַת מַגָּעוֹ בַזָּב נָשָׂא בְגָדִים וְלֹא נָגַע בָּהֶן, טְהוֹרִים: כֹּל הַמְטַמֵּא בְגָדִים בִּשְׁעַת מַגָּעוֹ. כְּגוֹן הַנּוֹגֵעַ בְּזָב וְזָבָה וְיוֹלֶדֶת, וּבְמַעַיְנוֹתֵיהֶם, בְּמִשְׁכָּבָן, וּבְמוֹשָׁבָן: מְטַמֵּא אֳכָלִים. לִהְיוֹת תְּחִלָּה, מִפְּנֵי שֶׁהוּא נֶחְשָׁב כְּאַב הַטֻּמְאָה בִּשְׁעַת מַגָּעוֹ, כֵּיוָן דְּאָמְרָה תוֹרָה שֶׁהוּא מְטַמֵּא בְגָדִים: וְאֶת הַיָּדַיִם לִהְיוֹת שְׁנִיּוֹת. אֲבָל גּוּפוֹ שֶׁל אָדָם, אֵינוֹ מְטַמֵּא, כִּדְקָתָנֵי סֵיפָא שֶׁאֵינוֹ מְטַמֵּא אָדָם וּכְלֵי חֶרֶס: לְאַחַר פְּרִישָׁתוֹ. מִמְּטַמְּאָיו מְטַמֵּא אֶת הַמַּשְׁקִין לִהְיוֹת תְּחִלָּה. וְאַף עַל פִּי שֶׁאֵינוֹ אֶלָּא רִאשׁוֹן לַטֻּמְאָה, עוֹשֶׂה הַמַּשְׁקִין תְּחִלָּה, דַּאֲפִלּוּ שֵׁנִי לַטֻּמְאָה מְטַמֵּא מַשְׁקִין לִהְיוֹת תְּחִלָּה. לְפִי שֶׁהֵן עֲלוּלִין לְקַבֵּל טֻמְאָה וְאֵינָן צְרִיכִין הֶכְשֵׁר כְּמוֹ הָאֳכָלִים, גָּזְרוּ בָהֶן שֶׁיִּהְיוּ לְעוֹלָם תְּחִלָּה, גְּזֵרָה מִשּׁוּם מַשְׁקִין הַבָּאִין מֵחֲמַת שֶׁרֶץ שֶׁהֵם תְּחִלָּה מִן הַתּוֹרָה: וְהָאֳכָלִים וְהַיָּדַיִם לִהְיוֹת שְׁנִיּוֹת. דְּכֵיוָן דְּפֵרַשׁ מִמְּטַמְּאָיו, אֵינוֹ אֶלָּא רִאשׁוֹן בְּעָלְמָא:

כֹּל הַנִּשָּׂא עַל גַּבֵּי הַזָּב. עֶלְיוֹנוֹ שֶׁל זָב וַאֲפִלּוּ לֹא נָגַע בּוֹ הַזָּב , וַאֲפִלּוּ יֵשׁ דְּבָרִים הַרְבֵּה חוֹצְצִים בֵּינוֹ לְבֵין הַזָּב, הַכֹּל טָמֵא, דִּכְתִיב (שָׁם) וְהַנּוֹגֵעַ בְּכָל אֲשֶׁר יִהְיֶה תַחְתָּיו, מַאי תַחְתָּיו, אִילֵימָא תַּחְתָּיו דְּזָב, הַיְנוּ מִשְׁכָּב, אֶלָּא בְּכָל אֲשֶׁר יִהְיֶה הַזָּב תַּחְתָּיו, דְּהַיְנוּ עֶלְיוֹנוֹ שֶׁל זָב, וְנוֹהֵג בְּכָל דָּבָר, אֲפִלּוּ בְמִידִי דְּאֵין רְגִילוּת לִהְיוֹת עָלָיו, וְלֹא הָווּ כְעֵין תַּחְתָּיו, דְּאֵין נוֹהֵג אֶלָּא בְמִידִי דַּחֲזִי לֵיהּ: חוּץ מִן הָרָאוּי לְמִשְׁכָּב וּלְמוֹשָׁב. וְהוּא הַדִּין לְמֶרְכָּב . אֲבָל שֶׁאֵינוֹ רָאוּי, טָהוֹר, כְּגוֹן אִם כָּפָה סְאָה שֶׁאֵינָהּ עֲשׂוּיָה לִישִׁיבָה וְיָשַׁב עָלֶיהָ, יָכוֹל יְהֵא טָמֵא מוֹשָׁב, תַּלְמוּד לוֹמַר (שָׁם) וְהַיֹּשֵׁב עַל הַכְּלִי אֲשֶׁר יֵשֵׁב עָלָיו הַזָּב, מַה שֶּׁמְּיֻחָד לִישִׁיבָה, יָצָא זֶה שֶׁאוֹמְרִים לוֹ עֲמֹד וְנַעֲשֶׂה מְלַאכְתֵּנוּ: נִדְבָּךְ. שׁוּרַת בִּנְיַן הַחוֹמָה. כְּמוֹ נִדְבָּכִין דִּי אֶבֶן גְּלָל (בְּעֶזְרָא ו): הַטָּהוֹר מִלְּמַעְלָה. עַל הַנִּדְבָּךְ וּמַכְבִּיד עַל אֶצְבָּעוֹ שֶׁל זָב, דְּהַשְׁתָּא הָוֵי נִשָּׂא עַל גַּבֵּי הַזָּב וְנֶחְשָׁב אַב הַטֻּמְאָה: מְטַמֵּא שְׁנַיִם וּפוֹסֵל אֶחָד. מְטַמֵּא רִאשׁוֹן וְשֵׁנִי, וּפוֹסֵל שְׁלִישִׁי בַּתְּרוּמָה: פֵּרַשׁ. שֶׁיָּרַד הַטָּהוֹר מִן הַנִּדְבָּךְ, אוֹ שֶׁסִּלֵּק הַזָּב אֶצְבָּעוֹ מִתַּחַת הַנִּדְבָּךְ. מְטַמֵּא הַטָּהוֹר אֶחָד וּפוֹסֵל אֶחָד, לְפִי שֶׁאֵין הַטָּהוֹר אֶלָּא רִאשׁוֹן מִכֵּיוָן שֶׁפֵּרַשׁ, דְּתוּ לֹא מְטַמֵּא בְגָדִים, וְעוֹשֶׂה שֵׁנִי וּשְׁלִישִׁי בַּתְּרוּמָה: הַטָּמֵא מִלְמַעְלָה. עַל הַנִּדְבָּךְ, וְאֶצְבַּע הַטָּהוֹר תַּחַת הַנִּדְבָּךְ, וּמַכְבִּיד הַטָּמֵא עָלָיו, הֲרֵי הַזָּב נִשָּׂא עָלָיו, לְפִיכָךְ מְטַמֵּא שְׁנַיִם וּפוֹסֵל אֶחָד: וְהַמַּדָּף. כָּל כֵּלִים שֶׁאֵינָם מִשְׁכָּב וּמוֹשָׁב מַדָּף קָרֵי לְהוּ: מִלְמַעְלָן. שֶׁל נִדְבָּךְ, וְאֶצְבַּע הַזָּב מִלְּמַטָּה, אֵינָם אֶלָּא רִאשׁוֹן: מְטַמְּאִין אֶחָד וּפוֹסְלִים אֶחָד. וּלְאַחַר שֶׁפֵּרְשׁוּ נַמִּי כָךְ דִּינָא: פֵּרְשׁוּ מְטַמְּאִין שְׁנַיִם וּפוֹסְלִים אֶחָד. הַמִּשְׁכָּב וּמוֹשָׁב לְעוֹלָם אַב הַטֻּמְאָה אַף עַל פִּי שֶׁפֵּרְשׁוּ: טְהוֹרִים. כִּדְאָמְרִינַן שֶׁכָּל שֶׁהַזָּב נִשָּׂא עָלָיו, טָהוֹר, חוּץ מִמִּשְׁכָּב וּמוֹשָׁב וְהָאָדָם:

מִפְּנֵי שֶׁאָמְרוּ. כְּלוֹמַר מִפְּנֵי מָה אָמְרוּ הָאֳכָלִים וְהַמַּשְׁקִים וְהַמַּדָּף מִלְּמַטָּה תַּחַת מִשְׁכַּב הַזָּב טְהוֹרִים, מִפְּנֵי שֶׁאָמְרוּ כָּל הַנּוֹשֵׂא וְנִשָּׂא כוּ': כֹּל הַנּוֹשֵׂא. כְּגוֹן אֳכָלִין וּמַשְׁקִין וּמַדָּף שֶׁהֵן לְמַטָּה מִמִּשְׁכַּב הַזָּב וְנוֹשְׂאִים אוֹתוֹ וְאֵינָן נוֹגְעִין בּוֹ , שֶׁיֵּשׁ דְּבָרִים חוֹצְצִים בֵּינְתַיִם: נִשָּׂא עַל גַּבֵּי הַמִּשְׁכָּב. שֶׁאֵלּוּ לְמַעְלָה וְהַמִּשְׁכָּב לְמַטָּה וְנִשָּׂא עָלָיו וְאֵין נוֹגְעִין בּוֹ, טְהוֹרִים: חוּץ מִן הָאָדָם. שֶׁהָאָדָם הַנּוֹשֵׂא הַמִּשְׁכָּב, נִטְמָא אַף עַל פִּי שֶׁלֹּא נָגַע בּוֹ וְלֹא הִסִּיטוֹ, מִפְּנֵי שֶׁמִּשְׁכָּב הַזָּב מְטַמֵּא בְמַשָּׂא. וְכֵן אִם נָתְנוּ אֶבֶן מְסָמָא לְמַעְלָה מִמִּשְׁכָּבוֹ שֶׁל זָב וְיָשַׁב אָדָם טָהוֹר עַל הָאֶבֶן, נִטְמָא, כֵּיוָן שֶׁנִּשָּׂא עַל גַּבֵּי מִשְׁכָּב. כְּשֵׁם שֶׁאִם יָשַׁב הַזָּב עַל הָאֶבֶן נִטְמָא הַמִּשְׁכָּב שֶׁלְּמַטָּה מִן הָאֶבֶן, כָּךְ כְּשֶׁיָּשַׁב הַטָּהוֹר עַל הָאֶבֶן, נִטְמָא וְנִשָּׂא מִן הַמִּשְׁכָּב שֶׁתַּחְתֶּיהָ: כֹּל הַנּוֹשֵׂא עַל גַּבֵּי נְבֵלָה. בְּאָדָם קָמַיְרֵי , כְּלוֹמַר וְכָל אָדָם הַנּוֹשֵׂא וְנִתָּן עַל גַּבֵּי נְבֵלָה: טָהוֹר חוּץ מִן הַמַּסִּיט. דְּנוֹשֵׂא אֶת הַנְּבֵלָה וְלֹא זָז מִמְּקוֹמוֹ גַּם לֹא הֱנִיעָהּ מִמְּקוֹמָהּ אֶלָּא שֶׁמֻּנַּחַת עַל גַּבָּיו, כְּגוֹן שֶׁבָּא חֲבֵרוֹ וְהִנִּיחָהּ עָלָיו, טָהוֹר, עַד שֶׁיַּסִּיטֶנָּה : רַבִּי אֱלִיעֶזֶר אוֹמֵר אַף הַנּוֹשֵׂא. בַּגְּמָרָא מְפָרֵשׁ דְּהָכִי קָאָמַר, וְהוּא שֶׁנּוֹשֵׂא, כְּלוֹמַר אֵין הַמַּסִּיט אֶת הַנְּבֵלָה טָמֵא אֶלָּא כְשֶׁנּוֹשֵׂא אוֹתָהּ, שֶׁנְּבֵלָה לְמַעְלָה וְהוּא לְמַטָּה. אֲבָל אִם הַנְּבֵלָה לְמַטָּה וְהוּא לְמַעְלָה, אַף עַל פִּי שֶׁמַּסִּיטָהּ, טָהוֹר, כָּל זְמַן שֶׁלֹּא נָגַע בָּהּ. וְאֵין הֲלָכָה כְּרַבִּי אֱלִיעֶזֶר: חוּץ מִן הַמַּאֲהִיל. שֶׁאֶפְשָׁר לִהְיוֹת נוֹשֵׂא וְנִשָּׂא וְלֹא יַאֲהִיל, כְּגוֹן עֲלִיָּה מַפְסֶקֶת, שֶׁאִם הַמֵּת בָּעֲלִיָּה וָאֳכָלִין וּמַשְׁקִין וְכֵלִים בַּבַּיִת מַגִּיעִין עַד שְׁמֵי קוֹרָה וְהַנְּסָרִים נִכְפָּפִים מִכֹּבֶד הַמֵּת וּמַכְבִּידִין עַל הַכֵּלִים, אוֹ שֶׁהַמֵּת בַּבַּיִת סָמוּךְ לִשְׁמֵי קוֹרָה וְהַכֵּלִים בָּעֲלִיָּה וְנִכְפָּפִים הַנְּסָרִים שֶׁבָּעֲלִיָּה מִכֹּבֶד הַכֵּלִים וּמַכְבִּידִים עַל הַמֵּת , בִּכְהַאי גַּוְנָא הָוֵי נוֹשֵׂא וְנִשָּׂא עַל גַּבֵּי הַמֵּת וְאֵינוֹ מַאֲהִיל, וְטָהוֹר: וְאָדָם בִּזְמַן שֶׁהוּא מַסִּיט. בֵּין נוֹשֵׂא אֶת הַמֵּת, בֵּין נִשָּׂא עַל גַּבֵּי מֵת, טָמֵא אַף עַל פִּי שֶׁאֵינוֹ נוֹגֵעַ בּוֹ. אֲבָל אִם אֵין הַמֵּת זָז מִמְּקוֹמוֹ, שֶׁאֵין כָּאן הֶסֵּט, טָהוֹר :

פרק ה - משנה ד
מִקְצָת טָמֵא עַל הַטָּהוֹר. אֲפִלּוּ אֶצְבָּעוֹ שֶׁל זָב עַל הַטָּהוֹר, אוֹ אֶצְבָּעוֹ שֶׁל טָהוֹר עַל הַזָּב, אוֹ חִבּוּרֵי טָמֵא, כְּגוֹן שְׂעָרוֹ וְצִפָּרְנָיו וְשִׁנָּיו, כָּל אֵלּוּ נִקְרָאִין חִבּוּרֵי, אַף עַל גַּב דְּלֵיכָּא אֶלָּא מִעוּט הַטָּמֵא עַל הַטָּהוֹר אוֹ מִעוּט הַטָּהוֹר עַל הַטָּמֵא, נִטְמָא הַטָּהוֹר, דְּלֹא בָעִינַן שֶׁיִּנָּשֵׂא רֻבּוֹ שֶׁל זָב אֶלָּא עַל הַמִּשְׁכָּב וְלֹא עַל הָאָדָם: וּמִקְצָת טָהוֹר עַל הַטָּמֵא טָהוֹר. דְּבָעִינַן שֶׁיִּנָּשֵׂא רֻבּוֹ. וְאֵין הֲלָכָה כְּרַבִּי שִׁמְעוֹן:

פרק ה - משנה ה
הַטָּמֵא עַל מִקְצָת הַמִּשְׁכָּב. רֻבּוֹ שֶׁל טָמֵא עַל מִעוּטוֹ שֶׁל מִשְׁכָּב, נִטְמָא הַמִּשְׁכָּב. וְכֵן רֻבּוֹ שֶׁל טָהוֹר עַל מִעוּטוֹ שֶׁל הַמִּשְׁכָּב הַטָּמֵא, נִטְמָא הַטָּהוֹר. וְלֹא חָיְשִׁינַן עַל מִעוּט מִשְׁכָּב, כֵּיוָן דְּרֻבָּא דְטָהוֹר וְרֻבָּא דְטָמֵא יוֹשֵׁב עָלָיו, דִּבְמִעוּטוֹ שֶׁל מִשְׁכָּב טֻמְאָה נִכְנֶסֶת, וּבְמִעוּטוֹ יוֹצֵאת, כִּדְקָתָנֵי סֵיפָא נִמְצֵאת טֻמְאָה נִכְנֶסֶת לוֹ וְיוֹצְאָה מִמֶּנּוּ בְּמִעוּטוֹ: מִקְצָת הַטָּמֵא עַל הַמִּשְׁכָּב. כֵּיוָן דְּמִעוּטוֹ שֶׁל טָמֵא הוּא, אַף עַל פִּי שֶׁנִּשְׁעָן עַל כָּל הַמִּשְׁכָּב, טָהוֹר הַמִּשְׁכָּב, עַד שֶׁיִּנָּשֵׂא עָלָיו רֻבּוֹ שֶׁל זָב: וּמִקְצָת טָהוֹר עַל הַמִּשְׁכָּב. כְּלוֹמַר אוֹ מִקְצָת טָהוֹר אֲפִלּוּ עַל כָּל הַמִּשְׁכָּב: טָהוֹר. עַד שֶׁיִּשָּׁעֵן עָלָיו רֻבּוֹ שֶׁל טָהוֹר: וְכֵן כִּכָּר שֶׁל תְּרוּמָה. אַמַּתְנִיתִין דִּלְעֵיל קָאֵי, דִּתְנַן כָּל הַנּוֹשֵׂא וְנִשָּׂא עַל גַּבֵּי הַמִּשְׁכָּב, טָהוֹר. וַאֲפִלּוּ אֳכָלִין שֶׁל תְּרוּמָה, אִם נִשְּׂאוּ עַל גַּבֵּי מִשְׁכָּב וְיֵשׁ בֵּינֵיהֶן דָּבָר חוֹצֵץ אֲפִלּוּ הוּא דַק בְּיוֹתֵר כְּגוֹן נְיָר וְכַיּוֹצֵא בוֹ, אֵינָן מְקַבְּלִין טֻמְאָה מִן הַמִּשְׁכָּב, דְּאֵין מְטַמֵּא אֲחֵרִים וְלֹא מְקַבֵּל טֻמְאָה בְּהֶסֵּט אֶלָּא דָבָר שֶׁיֶּשׁ בּוֹ רוּחַ חַיִּים : וְרַבִּי שִׁמְעוֹן מְטַמֵּא בָזוֹ. הוֹאִיל וּמְקוֹמָהּ טָמֵא, מְטַמֵּא בְהֶסֵּט אֲפִלּוּ דָבָר שֶׁאֵין בּוֹ רוּחַ חַיִּים. וְאֵין הֲלָכָה כְּרַבִּי שִׁמְעוֹן:

פרק ה - משנה ו
הַנּוֹגֵעַ בְּזָב וּבְזָבָה כוּ'. כָּל הָנֵי שָׁוֶה מַגָּעָן לְמַשָּׂאָן דִּמְטַמְּאוּ אָדָם לְטַמֵּא בְגָדִים, הִלְכָּךְ חֲשִׁיבֵי אַב הַטֻּמְאָה כָּל זְמַן שֶׁלֹּא פֵרְשׁוּ, מֵאַחַר דִּמְטַמְּאוּ בְגָדִים, וּלְכָךְ הַנּוֹגֵעַ עוֹשֶׂה רִאשׁוֹן וְשֵׁנִי וּפוֹסֵל שְׁלִישִׁי בַּתְּרוּמָה: פֵּרַשׁ. מִמְּטַמְּאָיו, אֵינוֹ אֶלָּא רִאשׁוֹן בְּעָלְמָא וּמְטַמֵּא אֶחָד לַעֲשׂוֹתוֹ שֵׁנִי, וּפוֹסֵל שְׁלִישִׁי בַּתְּרוּמָה: וְאֶחָד הַמַּסִּיט וְאֶחָד הַנּוֹשֵׂא. וְאַף עַל פִּי שֶׁאֵינָן נוֹגְעִים. זָב וְזָבָה וְנִדָּה בְּהֶדְיָא כְתִיבֵי, דִּכְתִיב זֹאת תּוֹרַת הַזָּב וְגוֹ' וְהַדָּוָה בְּנִדָּתָהּ. וּמְצֹרָע וְיוֹלֶדֶת נַפְקָא לָן מִלַּזָּכָר וְלַנְּקֵבָה, לַזָּכָר לְרַבּוֹת אֶת הַמְצֹרָע, וְלַנְּקֵבָה לְרַבּוֹת אֶת הַיּוֹלֶדֶת:

פרק ה - משנה ז
וְאֶחָד הַמַּסִּיט. הַמֵּזִיז אוֹתָהּ מִמְּקוֹמָהּ אַף עַל פִּי שֶׁאֵינוֹ לֹא נוֹגֵעַ וְלֹא נוֹשֵׂא: רַבִּי אֱלִיעֶזֶר אוֹמֵר אַף הַנּוֹשֵׂא. וְהוּא שֶׁנּוֹשֵׂא קָאָמַר, כְּלוֹמַר אֵין הַמַּסִּיט טָמֵא אֶלָּא אִם כֵּן נוֹשֵׂא , שֶׁהֵן לְמַעְלָה וְהַמַּסִּיט לְמַטָּה. וְאֵין הֲלָכָה כְּרַבִּי אֱלִיעֶזֶר:

פרק ה - משנה ח
הַנּוֹשֵׂא אֶת הַמֶּרְכָּב. בְּנוֹשֵׂא אֶת הַמֶּרְכָּב כְּתִיב כִּבּוּס בְּגָדִים, וְלֹא בְנוֹגֵעַ, וְאֵין נוֹגֵעַ אֶלָּא רִאשׁוֹן בְּעָלְמָא: הַנּוֹשֵׂא אֶת הַנְּבֵלָה. הַיְנוּ מַסִּיט, דְּנוֹשֵׂא בְּלֹא מַסִּיט, טָהוֹר כְּדִפְרִישִׁית לְעֵיל. וְנוֹגֵעַ נַמִּי בִּנְבֵלָה, אַף עַל פִּי שֶׁהוּא טָמֵא, כֵּיוָן דְּלֹא מְטַמֵּא בְגָדִים אֵינוֹ מְטַמֵּא שְׁנַיִם וּפוֹסֵל אֶחָד, דְּאֵינוֹ אֶלָּא רִאשׁוֹן בְּעָלְמָא וּמְטַמֵּא אֶחָד וּפוֹסֵל אֶחָד: מֵי חַטָּאת שֶׁיֵּשׁ בָּהֶן כְּדֵי הַזָּיָה. מְטַמְּאִין בְּגָדִים בְּמַשָּׂא אֲבָל לֹא בְמַגָּע:

פרק ה - משנה ט
וְהִיא בְּבֵית הַבְּלִיעָה. כָּל זְמַן שֶׁהִיא בְּבֵית הַבְּלִיעָה מְטַמֵּא בְגָדִים, וְכֵיוָן דִּמְטַמֵּא בְגָדִים מְטַמֵּא שְׁנַיִם וּפוֹסֵל אֶחָד: הִכְנִיס רֹאשׁוֹ בַּאֲוִיר הַתַּנּוּר. אַף עַל גַּב דְּקַיְמָא נִבְלַת הָעוֹף עִם בֵּית הַבְּלִיעָה שֶׁל הָאוֹכֵל הַכֹּל בַּאֲוִיר הַתַּנּוּר, אֲפִלּוּ הָכִי טָהוֹר, דְּמֵחֲמַת הָאָדָם לֹא מִטַּמֵּא הַתַּנּוּר, דְּכָל הַמְטַמְּאִים בְּגָדִים אֵין מְטַמְּאִין אָדָם וּכְלֵי חֶרֶס . וּמֵחֲמַת נִבְלַת הָעוֹף לֹא מִטַּמֵּא, דְּאֵין לָהּ טֻמְאָה אֶלָּא לָאָדָם בְּבֵית הַבְּלִיעָה: הֱקִיאָהּ אוֹ שֶׁבְּלָעָהּ. הַיְנוּ פֵּרַשׁ. דְּמִשֶּׁיָּרְדָה לַבֶּטֶן אֵין לָהּ טֻמְאָה, וּכְשֶׁהִיא לְתוֹךְ פִּיו אֵין לָהּ טֻמְאָה, דְּלֹא יֹאכַל כְּתִיב (וַיִּקְרָא כב), אֲכִילָה בְּבֵית הַבְּלִיעָה:

פרק ה - משנה י
וּבִנְבֵלָה וּבְמֶרְכָּב. כָּל הָנֵי בְּמַשָּׂא בִּלְבַד מְטַמְּאִין בְּגָדִים וְלֹא בְמַגָּע, הִלְכָּךְ אָדָם הַנּוֹגֵעַ בָּהֶן אַף עַל פִּי שֶׁלֹּא פֵרַשׁ אֵינוֹ אֶלָּא רִאשׁוֹן, וּמְטַמֵּא אֶחָד וּפוֹסֵל אֶחָד: כֹּל הַנּוֹגֵעַ. כְּגוֹן אֳכָלִים וְכֵלִים . דַּאֲפִלּוּ בְאוֹתָן שֶׁאָדָם הַנּוֹגֵעַ בָּהֶן מְטַמֵּא שְׁנַיִם וּפוֹסֵל אֶחָד, כֵּלִים הַנּוֹגְעִים בָּהֶן מְטַמְּאִים אֶחָד וּפוֹסְלִין אֶחָד וַאֲפִלּוּ בִשְׁעַת מַגָּעוֹ: חוּץ מִן הָאָדָם. דְּיֵשׁ מֵהֶן דְּאִם נָגַע בָּהֶן אוֹ נָשָׂא דִּמְטַמֵּא שְׁנַיִם וּפוֹסֵל אֶחָד, כִּי הָנֵי דְַאֲמַרַן לְעֵיל: פֵּרַשׁ. אֲפִלּוּ אָדָם, בְּכֻלְּהוּ מְטַמֵּא אֶחָד וּפוֹסֵל אֶחָד, דְּכֵיוָן דְּפֵרַשׁ אֵינוֹ אֶלָּא רִאשׁוֹן:

פרק ה - משנה יא
בַּעַל קֶרִי כְּמַגַּע שֶׁרֶץ. כְּשֵׁם שֶׁנּוֹגֵעַ בַּשֶּׁרֶץ רִאשׁוֹן לַטֻּמְאָה, כָּךְ בַּעַל קֶרִי רִאשׁוֹן לַטֻּמְאָה, וְטֻמְאָתוֹ שָׁוָה לְטֻמְאַת הַנּוֹגֵעַ בְּשִׁכְבַת זֶרַע : וּבוֹעֵל נִדָּה כִּטְמֵא מֵת. שֶׁהוּא אַב הַטֻּמְאָה וְטָמֵא טֻמְאַת שִׁבְעָה : שֶׁהוּא מְטַמֵּא מִשְׁכָּב וּמוֹשָׁב. וּמֵת עַצְמוֹ וְכָל שֶׁכֵּן טְמֵא מֵת אֵינוֹ מְטַמֵּא מִשְׁכָּב וּמוֹשָׁב, לֹא טֻמְאָה קַלָּה וְלֹא טֻמְאָה חֲמוּרָה, דִּכְתִיב (וַיִּקְרָא טו) וְכָל הַכְּלִי אֲשֶׁר יֵשֵׁב עָלָיו, הַזָּב, וְלֹא טְמֵא מֵת:

אוֹ שֵׁנִי לַטֻּמְאָה, גָּזְרוּ שֶׁיְּהֵא גוּפוֹ שֵׁנִי לַטֻּמְאָה, וּפוֹסֵל אֶת הַתְּרוּמָה בְּמַגָּעוֹ, שֶׁשֵּׁנִי פוֹסֵל בַּתְּרוּמָה: וְהַשּׁוֹתֶה מַשְׁקִים טְמֵאִים. נַעֲשָׂה גַם כֵּן שֵׁנִי לַטֻּמְאָה וּפוֹסֵל אֶת הַתְּרוּמָה. וְטַעֲמָא דְגַזּוּר רַבָּנָן בְּהָנֵי, דְּזִמְנִין דְּאָכִיל אֳכָלִים טְמֵאִים, וְשָׁדֵי מַשְׁקִים דִּתְרוּמָה בְּפֻמֵּיהּ בְּעוֹד שֶׁהָאֳכָלִים טְמֵאִים בְּפִיו, וּפָסֵיל לְהוּ. וְכֵן נַמִּי זִמְנִין דְּשָׁתֵי מַשְׁקִין טְמֵאִין, וְשָׁדֵי אֳכָלִים דִּתְרוּמָה בְּפֻמֵּיהּ בְּעוֹדָן בְּפִיו, וּפָסֵיל לְהוּ: וְהַבָּא רֹאשׁוֹ וְרֻבּוֹ. לְאַחַר שֶׁטָּבַל בּוֹ בַיּוֹם: בְּמַיִם שְׁאוּבִין. לְפִי שֶׁהָיוּ טוֹבְלִים בְּמֵי מְעָרוֹת סְרוּחִים וְהָיוּ נוֹתְנִין עֲלֵיהֶם אַחַר כָּךְ מַיִם שְׁאוּבִין לְהַעֲבִיר סִרְחוֹן הַמַּיִם, הִתְחִילוּ וַעֲשָׂאוּם קֶבַע לוֹמַר לֹא מֵי הַמְּעָרוֹת מְטַהֲרִים אֶלָּא מַיִם שְׁאוּבִים מְטַהֲרִים, עָמְדוּ וְגָזְרוּ עֲלֵיהֶם טֻמְאָה, דִּלְמָא אָתוּ לְבַטּוֹלֵי תּוֹרַת מִקְוֶה וְטָבְלֵי בִשְׁאוּבִים, וּכְדֵי לְקַיֵּם גְּזֵרָה זוֹ לְהַעֲמִידָהּ, הוֹסִיפוּ וְגָזְרוּ עַל טָהוֹר גָּמוּר שֶׁנָּפְלוּ עַל רֹאשׁוֹ שְׁלֹשָׁה לֻגִּין מַיִם שְׁאוּבִין, שֶׁיִּהְיֶה פוֹסֵל אֶת הַתְּרוּמָה עַד שֶׁיִּטְבֹּל: וְהַסֵּפֶר. כָּל חֲכָמִים גָּזְרוּ עַל הַמַּשְׁקִים שֶׁנִּטְמְאוּ בְּשֵׁנִי לַטֻּמְאָה, לִהְיוֹת תְּחִלָּה, וִיטַמְּאוּ אֶת הָאֳכָלִים, גְּזֵרָה מִשּׁוּם מַשְׁקִין הַבָּאִין מֵחֲמַת שֶׁרֶץ שֶׁהֵם תְּחִלָּה מִדְּאוֹרַיְתָא. וְלָמָּה לֹא גָזְרוּ עַל הָאֳכָלִים נַמִּי לִהְיוֹת תְּחִלָּה גְּזֵרָה מִשּׁוּם אֳכָלִים הַבָּאִים מֵחֲמַת שֶׁרֶץ, לְפִי שֶׁהָאֳכָלִים הֵן צְרִיכִין הֶכְשֵׁר וְאֵין מְקַבְּלִין טֻמְאָה אֶלָּא עַל יְדֵי בִּיאַת מַיִם, אֲבָל הַמַּשְׁקִין הֵן עֲלוּלִים לְקַבֵּל טֻמְאָה וְאֵין צְרִיכִין הֶכְשֵׁר וְתִקּוּן, וְטֻמְאָתָן יוֹתֵר מְצוּיָה: וְכֵלִים שֶׁנִּטְמְאוּ בְמַשְׁקִין.
