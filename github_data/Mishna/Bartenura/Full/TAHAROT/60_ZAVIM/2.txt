פרק ב

הַכֹּל מִטַּמְּאִים בְּזִיבָה. הַכֹּל, לַאֲתוֹיֵי תִּינוֹק בֶּן יוֹמוֹ שֶׁהוּא מִטַּמֵּא בְּזִיבָה: אַף הַגֵּרִים. דִּכְתִיב בְּפָרָשַׁת זָב (וַיִּקְרָא טו) דַּבֵּר אֶל בְּנֵי יִשְׂרָאֵל, אֵין לִי אֶלָּא בְּנֵי יִשְׂרָאֵל, מִנַּיִן לְרַבּוֹת גֵּרִים וַעֲבָדִים, תַּלְמוּד לוֹמַר וְאָמַרְתָּ אֲלֵיהֶם אִישׁ, לְרַבּוֹת אֶת הַגֵּרִים וְאֶת הָעֲבָדִים. מִנַּיִן לְרַבּוֹת אֶת הַקְּטַנִּים, תַּלְמוּד לוֹמַר אִישׁ אִישׁ: וּמִטַּמְּאִין בְּדָם כָּאִשָּׁה. שֶׁאֵין הָאִשָּׁה מִטַּמְּאָה בְּלֹבֶן אֶלָּא בְאֹדֶם, דְּדָם כְּתִיב בָּהּ, שֶׁהוּא אָדֹם: וּבְלֹבֶן כָּאִישׁ. דְּזוֹבוֹ כְּתִיב גַּבֵּי אִישׁ . וְהוּא דוֹמֶה לְמֵי בָצֵק שֶׁל שְׂעֹרִים: וְטֻמְאָתָן בְּסָפֵק. אִם רָאוּ לֹבֶן, שֶׁמָּא הֵן אִשָּׁה, וְלֹבֶן טָהוֹר בָּהּ. וְאִם רָאוּ אֹדֶם, שֶׁמָּא הֵן אִישׁ, וְאֹדֶם טָהוֹר גַּבֵּיהּ. אֲבָל אִם רָאָה לֹבֶן וְאֹדֶם כְּאֶחָד, טֻמְאָתָן וַדַּאי מִמַּה נַּפְשָׁךְ, וְשׂוֹרְפִין עֲלֵיהֶם אֶת הַתְּרוּמָה, אֲבָל אֵין חַיָּבִים עֲלֵיהֶם עַל בִּיאַת מִקְדָּשׁ, דְּגַבֵּי שִׁלּוּחַ טְמֵאִים כְּתִיב (בְּמִדְבַּר ה) מִזָּכָר עַד נְקֵבָה תְּשַׁלֵּחוּ, זָכָר וַדַּאי, נְקֵבָה וַדָּאִית , וְלֹא טֻמְטוּם וְאַנְדְּרוֹגִינוֹס :

בְּשִׁבְעָה דְרָכִים בּוֹדְקִין אֶת הַזָּב. מִשּׁוּם דְּאֵינוֹ מְטַמֵּא מֵחֲמַת אֹנֶס, דִּכְתִיב זָב מִבְּשָׂרוֹ, וְלֹא מֵחֲמַת אָנְסוֹ: עַד שֶׁלֹּא נִזְקַק לְטֻמְאָה. דְּהַיְנוּ כְּשֶׁרוֹאֶה רְאִיָּה שְׁנִיָּה שֶׁבָּהּ הוּא נַעֲשֶׂה זָב לְטַמֵּא מִשְׁכָּב וּמוֹשָׁב וּלְהַצְרִיכוֹ לִסְפֹּר שִׁבְעָה נְקִיִּים. אֲבָל רְאִיָּה רִאשׁוֹנָה, מְטַמְּאָה בְאֹנֶס טֻמְאַת עֶרֶב, כְּדִין בַּעַל קֶרִי, וּמִצְטָרֶפֶת עִם הַשְּׁנִיָּה וַאֲפִלּוּ שֶׁהָיְתָה בְאֹנֶס: בְּמַאֲכָל. אִם אָכַל אֲכִילָה גַּסָּה אוֹ אָכַל דְּבָרִים הַמְבִיאִין לִידֵי זִיבָה, וַאֲפִלּוּ כָל שֶׁהֵן, כְּגוֹן בָּשָׂר שָׁמֵן, חָלָב, וּגְבִינָה, בֵּיצִים, וְיַיִן יָשָׁן, וּגְרִיסִין שֶׁל פּוֹל, וְשַׁחֲלִין: וּבְמִשְׁתֶּה. בְּרִבּוּי שְׁתִיָּה: בְּמַשָּׂא. כָּבֵד שֶׁנָּשָׂא: בְּמַרְאֶה. רָאָה אִשָּׁה, אֲפִלּוּ לֹא הִרְהֵר: וּבְהִרְהוּר. אַף עַל פִּי שֶׁלֹּא רָאָה. אִי אֵרַע לוֹ אֶחָד מֵאֵלּוּ הַשִּׁבְעָה קֹדֶם שֶׁרָאָה רְאִיָּה שְׁנִיָּה, אֵינוֹ נַעֲשֶׂה זָב מֵחֲמַת אוֹתָהּ רְאִיָּה, וְהַטִּפָּה אֵינָהּ מְטַמְּאָה בְמַשָּׂא כִּשְׁאָר טִפָּה שֶׁל זוֹב: אֵין אַחֲרָיוּת זָבִין עֲלֵיכֶם. וְאִם לֹא יִהְיוּ זָבִים לְעוֹלָם, מָה אַתֶּם חוֹשְׁשִׁים, הֲלֹא לֹא קִבַּלְתֶּם אַחֲרָיוּת שֶׁל זָבִים עֲלֵיכֶם. וְאֵין הֲלָכָה לֹא כְרַבִּי יְהוּדָה וְלֹא כְרַבִּי עֲקִיבָא: מִשֶּׁנִּזְקַק לְטֻמְאָה. לְאַחַר שֶׁרָאָה רְאִיָּה שְׁנִיָּה שֶׁלֹּא בְאֹנֶס, שֶׁהֲרֵי מִכָּאן וָאֵילָךְ הוּא עוֹשֶׂה מִשְׁכָּב וּמוֹשָׁב: אֵין בּוֹדְקִין אוֹתוֹ. וְאַף אִם רָאָה רְאִיָּה שְׁלִישִׁית בְּאֹנֶס, נַעֲשָׂה זָב לַקָּרְבָּן. וְכֵן אִם רָאָה רְאִיָּה שֶׁל זוֹב בְּשִׁבְעָה יָמִים נְקִיִּים שֶׁהוּא מוֹנֶה, סָתַר הַכֹּל וַאֲפִלּוּ הָיְתָה אוֹתָהּ רְאִיָּה בְּאֹנֶס: סְפֵקוֹ וְשִׁכְבַת זַרְעוֹ טְמֵאִין. סְפֵקוֹ מֵחֲמַת שִׁכְבַת זַרְעוֹ, כְּגוֹן אִם רָאָה שִׁכְבַת זֶרַע תְּחִלָּה, וְטִפָּה זִיבָה הַבָּאָה אַחַר כָּךְ אֵינָהּ מְטַמְּאָה, שֶׁהָרוֹאֶה קֶרִי אֵינוֹ מְטַמֵּא בְזִיבָה מֵעֵת לְעֵת. וְעַד שֶׁלֹּא נִזְקַק לְטֻמְאָה, הַקֶּרִי מְטַהֵר אֶת הַזִּיבָה, מִשּׁוּם דַּהֲוֵי אֹנֶס. וּלְאַחַר שֶׁנִּזְקַק לְטֻמְאָה, אֵין הַקֶּרִי מְטַהֵר אֶת הַזִּיבָה, שֶׁאֵין תּוֹלִין שֶׁמֵּחֲמַת הַקֶּרִי בָּאָה הַזִּיבָה: שֶׁרַגְלַיִם לַדָּבָר. שֶׁאֵין אוֹתָהּ רְאִיָּה מֵחֲמַת אֹנֶס, כֵּיוָן שֶׁכְּבָר נַעֲשָׂה זָב: רָאָה רְאִיָּה רִאשׁוֹנָה בּוֹדְקִין אוֹתוֹ. הַיְנוּ דַּוְקָא לַקָּרְבָּן. דְּאִי מֵחֲמַת אֹנֶס רָאָה, אֵינָהּ מִצְטָרֶפֶת עִם הַשְּׁלִישִׁית לְקָרְבָּן. אֲבָל מִצְטָרֶפֶת עִם הַשְּׁנִיָּה לַטֻּמְאָה. דְּאִם רְאִיָּה שְׁנִיָּה שֶׁלֹּא בְאֹנֶס, אַף עַל פִּי שֶׁהָיְתָה רִאשׁוֹנָה בְּאֹנֶס, עוֹשָׂה מִשְׁכָּב וּמוֹשָׁב וּבָעֵי שִׁבְעָה נְקִיִּים וּבִיאַת מַיִם חַיִּים, דְּכִי כְתִיב מִבְּשָׂרוֹ וְלֹא מֵחֲמַת אָנְסוֹ, בַּשְּׁנִיָּה כְּתִיב . אֲבָל רִאשׁוֹנָה אִתַּקַּשׁ לְשִׁכְבַת זֶרַע, דִּכְתִיב זֹאת תּוֹרַת הַזָּב וַאֲשֶׁר תֵּצֵא מִמֶּנּוּ שִׁכְבַת זֶרַע, מַה שִּׁכְבַת זֶרַע מְטַמְּאָה בְאֹנֶס , אַף רְאִיָּה רִאשׁוֹנָה שֶׁל זוֹב מְטַמְּאָה בְאֹנֶס: בַּשְּׁנִיָּה בּוֹדְקִין אוֹתוֹ. לַטֻּמְאָה וְלַקָּרְבָּן: בַּשְּׁלִישִׁית אֵין בּוֹדְקִין אוֹתוֹ. לֹא לַטֻּמְאָה וְלֹא לַקָּרְבָּן: רַבִּי אֱלִיעֶזֶר אוֹמֵר כוּ'. וְאֵין הֲלָכָה כְּרַבִּי אֱלִיעֶזֶר:

אֵינוֹ מִטַּמֵּא בְזִיבָה מֵעֵת לְעֵת. דְּהַאי זִיבוּת מֵחֲמַת חוּלְשָׁא דְקֶרִי הוּא. וּבִרְאִיָּה שְׁנִיָּה אַיְרֵי, דְּרִאשׁוֹנָה מְטַמְּאָה בְאֹנֶס. וּכְשֵׁם שֶׁתּוֹלִין לְקֶרִי מֵעֵת לְעֵת, כָּךְ לְמַרְאֶה וּלְהִרְהוּר תּוֹלִין מֵעֵת לְעֵת. אֲבָל לְמַאֲכָל וּלְמִשְׁתֶּה לְמַשָּׂא וְלִקְפִיצָה , תּוֹלִין לוֹ כָּל זְמַן שֶׁהוּא מִצְטַעֵר, וְאָמְרִינַן דְּמֵחֲמַת הַצַּעַר הָוֵי הַזִּיבָה וְלֹא מִבְּשָׂרוֹ: רַבִּי יוֹסֵי אוֹמֵר יוֹמוֹ. אֵין תּוֹלִין אֶלָּא אוֹתוֹ הַיּוֹם שֶׁרָאָה קֶרִי בִּלְבַד, וְלֹא מֵעֵת לְעֵת. וְאֵין הֲלָכָה כְּרַבִּי יוֹסֵי: מִיָּד מִטַּמֵּא בְזִיבָה. דְּגֵר שֶׁנִּתְגַּיֵּר כְּקָטָן שֶׁנּוֹלָד דָּמֵי. וְאַף עַל פִּי שֶׁרָאָה זִיבוּת כְּשֶׁהוּא נָכְרִי, מִטַּמֵּא בְזִיבָה בְהַךְ שְׁנִיָּה שֶׁרוֹאֶה בְתוֹךְ מֵעֵת לְעֵת שֶׁל קֶרִי, דְּנַעֲשֵׂית לוֹ כִּרְאִיָּה רִאשׁוֹנָה דִּמְטַמֵּא בְאֹנֶס, וּמִצְטָרֶפֶת עִם הַשְּׁנִיָּה שֶׁיִּרְאֶה אַחַר מֵעֵת לְעֵת שֶׁל קֶרִי. דִּרְאִיַּת זִיבָה שֶׁל נָכְרִי כְּמַאן דְּלֵיתָא, דְּאֵין הַנָּכְרִים מִטַּמְּאִין בְּזִיבָה: הָרוֹאָה דָם. מְטַמְּאָה כָל טָהֳרוֹת שֶׁנִּתְעַסְּקָה בָהֶן לְמַפְרֵעַ מֵעֵת לְעֵת, כִּדְפָרִישְׁנָא בְרֵישׁ נִדָּה. וְאַיְדֵי דְאַיְרֵי בְמֵעֵת לְעֵת, קָחָשֵׁיב כָּל הָנֵי שֶׁשִּׁעוּרָן בְּמֵעֵת לְעֵת: וְכֵן הַמַּקְשָׁה. בִּימֵי זִיבָה. דְּאָמְרִינַן כִּי יָזוּב זוֹב דָּמָהּ, מֵחֲמַת עַצְמָהּ וְלֹא מֵחֲמַת וָלָד. וְאִם שָׁפְתָה מִן הַחֲבָלִין מֵעֵת לְעֵת וְיָלְדָה, לֹא תָלִינַן דַּם קֹשִׁי בְּדַם לֵידָה: הַמַּכֶּה אֶת עַבְדּוֹ. הָאָמוּר בַּתּוֹרָה (שְׁמוֹת כא) אַךְ אִם יוֹם אוֹ יוֹמַיִם יַעֲמֹד, וְדָרְשִׁינַן יוֹם שֶׁהוּא כְיוֹמַיִם, זֶהוּ מֵעֵת לְעֵת: כֶּלֶב שֶׁאָכַל בְּשַׂר הַמֵּת. דְּקַיְמָא לָן דְּאֵין מִתְעַכֵּל בְּמֵעָיו עַד לְאַחַר שְׁלֹשָׁה יָמִים מֵעֵת לְעֵת, דְּהָכִי תְנַן בְּמַסֶּכֶת אֳהָלוֹת פֶּרֶק י"א, וְכַמָּה תִשְׁהֶה בְּמֵעָיו, שְׁלֹשָׁה יָמִים מֵעֵת לְעֵת: וְהוּא שֶׁיְּהֵא כִבְרִיָּתוֹ. שֶׁלֹּא נִשְׁתַּנָּה . וְעוֹד אִיכָּא טוּבָא שֶׁשִּׁעוּרָן מֵעֵת לְעֵת וְלָא חֲשִׁיב, וְתָנָא וְשִׁיֵּר:

מְטַמֵּא אֶת הַמִּשְׁכָּב. וְהוּא הַדִּין לְמוֹשָׁב: עוֹמֵד יוֹשֵׁב שׁוֹכֵב. דִּכְתִיב (וַיִּקְרָא טו) כָּל הַמִּשְׁכָּב אֲשֶׁר יִשְׁכַּב עָלָיו הַזָּב יִטְמָא, וּכְתִיב (שָׁם) עַל הַכְּלִי אֲשֶׁר הִיא יוֹשֶׁבֶת עָלָיו, אֵין לִי אֶלָּא בִזְמַן שֶׁשָּׁכַב עַל הַמִּשְׁכָּב וְיָשַׁב עַל הַמּוֹשָׁב. שָׁכַב עַל הַמּוֹשָׁב וְיָשַׁב עַל הַמִּשְׁכָּב, עָמַד, נִתְלָה, נִשְׁעָן, מִנַּיִן. תַּלְמוּד לוֹמַר יִטְמָא יְטַמֵּא, רִבָּה. וְכֻלָּן נִקְרָאִין מִדְרָס: וְהַמִּשְׁכָּב מְטַמֵּא אֶת הָאָדָם וְכוּ' עוֹמֵד יוֹשֵׁב שׁוֹכֵב נִתְלֶה. שֶׁהַטָּהוֹר בְּכַף מֹאזְנַיִם וְהַמִּשְׁכָּב הַטָּמֵא בְּכַף שְׁנִיָּה, וְכָרַע הַמִּשְׁכָּב הַטָּמֵא, וְנִתְלֶה הַטָּהוֹר. וְנִתְלֶה דְרֵישָׁא, כְּגוֹן שֶׁהַזָּב בְּכַף מֹאזְנַיִם וּמוֹשָׁבוֹת בְּכַף שְׁנִיָּה וְכָרְעוּ הֵן, טְמֵאִין, דְּהַשְׁתָּא הַזָּב נִתְלֶה וּכְאִלּוּ הוּא יוֹשֵׁב עֲלֵיהֶן: לְטַמֵּא בְגָדִים. הוּא הַדִּין כָּל כֵּלִים שֶׁנּוֹגֵעַ בָּהֶן עַד שֶׁלֹּא פֵרַשׁ, חוּץ מֵאָדָם וּכְלֵי חֶרֶס: בְּמַגָּע וּבְמַשָּׂא. אִם נָגַע בַּמִּשְׁכָּב הַטָּמֵא, אוֹ שֶׁנְּשָׂאוֹ:
