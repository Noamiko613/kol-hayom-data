פרק ב - משנה א
זֵיתִים שֶׁכְּבָשָׁן. בְּיַיִן אוֹ בְחֹמֶץ אוֹ בִשְׁאָר מַשְׁקִין : בְּטַרְפֵיהֶן. עִם הֶעָלִים שֶׁלָּהֶם. עֲלֵה זַיִת, מְתַרְגְּמִינַן טְרַף זֵיתָא: טְהוֹרִים. אִם נָגַע שֶׁרֶץ בֶּעָלִין, טְהוֹרִים הַזֵּיתִים וְהֶעָלִים, דְּלָא חֲשִׁיבֵי הֶעָלִים לֹא יָד וְלֹא שׁוֹמֵר: לְפִי שֶׁלֹּא כְבָשָׁן. בְּטַרְפֵיהֶן: אֶלָּא לְמַרְאֶה. כְּדֵי שֶׁיִּהְיוּ נִרְאִים יָפִין יוֹתֵר כְּשֶׁהֶעָלִים עִמָּהֶן, דְּנִרְאִים כְּאִלּוּ מֵחָדָשׁ נִתְלְשׁוּ: כְּשׁוּת שֶׁל קִשּׁוּת. כְּשֶׁהַקִּשּׁוּאִים קְטַנִּים יֵשׁ לָהֶן כְּמִין שֵׂעָר קָטָן, וְנִקְרָא כְּשׁוּת: וְהַנֵּץ שֶׁלָּהּ. בְּפִי הַקִּשּׁוּאִים יֵשׁ כְּמוֹ נֵץ אָרֹךְ מְעַט: טְהוֹרִים. דְּלָא חֲשִׁיבֵי לֹא יָד וְלֹא שׁוֹמֵר: לִפְנֵי הַתַּגָּר טְמֵאָה. דְּנִיחָא לֵיהּ לַתַּגָּר בַּשֵּׂעָר וּבַנֵּץ, שֶׁמִּתּוֹךְ כָּךְ נִרְאִים הַקִּשּׁוּאִים יָפִים וּכְאִלּוּ מֵחָדָשׁ נִתְלְשׁוּ , וְכֵיוָן שֶׁרוֹצָה בְּקִיּוּמָן, חֲשִׁיבֵי שׁוֹמֵר לַפְּרִי. אִי נַמִּי, לְפִי שֶׁהַקִּשּׁוּאִים רַכִּים וְיַד הַכֹּל מְמַשְׁמְשִׁים בָּהֶן לִקְנוֹתָן הֵן מִתְקַלְקְלִים אִם לֹא מִפְּנֵי שֶׁאוֹתוֹ שֵׂעָר שׁוֹמְרָן מִכָּל לִכְלוּךְ וְטִנּוּף:

פרק ב - משנה ב
הַגַּלְעִינִים. כְּמוֹ גַרְעִינִים: וְלֹא מִצְטָרְפוֹת. עִם הָאֹכֶל לְהַשְׁלִימוֹ לְכַבֵּיצָה, דְּיָד חֲשִׁיבָא וְלֹא שׁוֹמֵר: גַּלְעִינָה שֶׁל רֹטֶב. גַּרְעִין שֶׁל תְּמָרָה לַחָה וּרְאוּיָה לָמוֹץ: אַף עַל פִּי יוֹצְאָה. כְּלוֹמַר אַף עַל פִּי שֶׁיּוֹצְאָה מִן הָאֹכֶל, מִצְטָרֶפֶת עִם הָאֹכֶל, לְפִי שֶׁהִיא נֶאֱכֶלֶת, שֶׁמּוֹצֵץ אוֹתָהּ מֵחֲמַת רְטִיבוּת: לְפִיכָךְ חוֹתָם שֶׁל יְבֵשָׁה. גַּרְגִּיר שֶׁל עֲנָבִים וּתְמָרִים יֵשׁ לָהֶם חוֹתָם, שֶׁחוֹתֵם נֶקֶב הַפְּרִי, וּבְרֹאשׁ הַחוֹתָם עֹקֶץ, כָּךְ פֵּרְשׁוּ רַבּוֹתַי. וְקָשֶׁה לְפֵרוּשָׁם, מַאי לְפִיכָךְ . וּבְמִשְׁנָיוֹת דַּוְקָנִיּוֹת מָצָאתִי, לְפִיכָךְ חוֹתָל שֶׁל יְבֵשָׁה, וְכֵן גּוֹרֵס רַמְבַּ"ם, וּמְפָרֵשׁ טַעֲמָא דְּגַרְעִינָה שֶׁל רֹטֶב דְּהַיְנוּ שֶׁל תְּמָרָה לַחָה, מִצְטָרֶפֶת, לְפִי שֶׁהִיא שׁוֹמֵר לַפְּרִי, שֶׁאִם לֹא הַגַּרְעִין הָיָה הַפְּרִי נִפְסָד. וְשֶׁל יְבֵשָׁה אֵינָהּ מִצְטָרֶפֶת, דְּשׁוּב אֵין הַגַּרְעִין שׁוֹמֵר לַפְּרִי, שֶׁכְּבָר כָּלְתָה הַלֵּחָה שֶׁבַּתְּמָרָה וְאֵין הַגַּרְעִין שׁוֹמְרָהּ שֶׁלֹּא תִפָּסֵד. לְפִיכָךְ חוֹתָל שֶׁל יְבֵשָׁה וְהוּא הַקְּרוּם הַמְלֻפָּף וְדָבוּק עַל הַגַּרְעִין שֶׁל תְּמָרָה, מִצְטָרֵף עִם הַתְּמָרָה כְּשֶׁהִיא יְבֵשָׁה, מִפְּנֵי שֶׁהַקְּרוּם הוּא שׁוֹמֵר, לֹא הַגַּרְעִין: וְשֶׁל רֹטֶב. דְּהַיְנוּ תְּמָרָה לַחָה, אֵין הַקְּרוּם מִצְטָרֵף. שֶׁהַגַּרְעִין הוּא שׁוֹמֵר וְלֹא הַקְּרוּם. וְנִקְרָא אוֹתוֹ הַקְּרוּם חוֹתָל, לְפִי שֶׁהוּא מְלֻפָּף וְדָבוּק לַגַּרְעִין . לְשׁוֹן וְהָחְתֵּל לֹא חֻתָּלְתְּ (יְחֶזְקֵאל טז): גַּלְעִינָה שֶׁמִּקְצָתָהּ יוֹצְאָה. כְּגוֹן גַּרְעִינָה שֶׁל תְּמָרָה לַחָה שֶׁנֶּאֱכֶלֶת חֶצְיָהּ וְחֶצְיָהּ נִשְׁאֶרֶת עִם הַגַּרְעִין: מַה שֶּׁכְּנֶגֶד הָאֹכֶל. חָשׁוּב שׁוֹמֵר וּמִצְטָרֵף, אֲבָל אִידָךְ לֹא. וְכֵן עֶצֶם שֶׁיֵּשׁ עָלָיו כַּזַּיִת בָּשָׂר : הָיָה עָלָיו. הָיָה עַל הָעֶצֶם בָּשָׂר בְּצַד אֶחָד שֶׁאֵינוֹ מַקִּיף אֶת הָעֶצֶם, אָמַר רַבִּי שִׁמְעוֹן רוֹאִין אִם יֵשׁ בַּבָּשָׂר שִׁעוּר שֶׁיּוּכַל לְהַקִּיף מִמֶּנּוּ עָבְיוֹ כְּחוּט שֶׁל עֵרֶב כְּטַבַּעַת סְבִיב הָעֶצֶם, מִצְטָרֵף. וְאֵין הֲלָכָה כְּרַבִּי שִׁמְעוֹן: כְּגוֹן הַסִּיאָה וְהָאֵזוֹב וְהַקּוֹרָנִית. שֶׁגַּם אֵלּוּ כָּךְ דִּינָם, מַה שֶּׁכְּנֶגֶד הָאֹכֶל מִצְטָרֵף, וּמַה שֶּׁאֵינוֹ כְנֶגֶד הָאֹכֶל אֵין מִצְטָרֵף. וְסִיאָה וְקוֹרָנִית מִין אֵזוֹב הֵם. סִיאָה תליי"ז בְּלַעַ"ז, וְקוֹרָנִית שדרי"א בְלַעַ"ז:

פרק ב - משנה ג
שֶׁנִּמּוֹק מִקְצָתוֹ אֵין מִצְטָרֵף. דְּמַה שֶּׁנִּמּוֹק אֵינוֹ חָשׁוּב אֹכֶל, כֵּיוָן שֶׁנִּתְמַסְמֵס וְאֵינוֹ רָאוּי לַאֲכִילָה: הַפִּטְמָא שֶׁל רִמּוֹן. כְּמִין דַּד יוֹצֵא בְרֹאשׁ הָרִמּוֹן, וְהוּא הַנִּקְרָא פִטְמָא. וּסְבִיב הַפִּטְמָא כְּמִין פֶּרַח יָרֹק שֶׁיֶּשׁ בּוֹ חוּטִים דַּקִּים, וְהֵן קְרוּיִין נֵץ. וּסְבִיבוֹת אוֹתָן הַחוּטִים כְּעֵין כּוֹס וְרֹאשׁוֹ עָשׂוּי לְתַלְפִּיּוֹת, וְהוּא קָרוּי מַסְרֵק, מִפְּנֵי שֶׁיֵּשׁ לוֹ שִׁנַּיִם כְּמַסְרֵק: אַף הַמַּסְרֵק טָהוֹר. שֶׁאִם יֵחָתֵךְ מִן הַפְּרִי, אֵין מִתְגַּלִּים הַגַּלְעִינִים בְּכָךְ, הִלְכָּךְ לֹא הָוֵי שׁוֹמֵר. אֲבָל הַפִּטְמָא, אִם יַחְתְּכוּהָ עַד שָׁרְשָׁהּ, יִתְגַּלּוּ הַגַּרְעִינִים וְיִפָּסֵד הַפְּרִי. וְאֵין הֲלָכָה כְּרַבִּי אֱלִיעֶזֶר :

פרק ב - משנה ד
כָּל הַקְּלִפִּין. כְּגוֹן קְלִפֵּי אֱגוֹזִים וּשְׁקֵדִים וְרִמּוֹנִים וּקְלִפֵּי בֵיצִים, חֲשִׁיבֵי שׁוֹמֵר לָאֹכֶל, וּמִצְטָרְפִין עִם הָאֹכֶל לְהַשְׁלִימוֹ לְכַשִּׁעוּר לְקַבֵּל טֻמְאָה. וּמִיהוּ קְלִפָּה לַחָה הָעֶלְיוֹנָה שֶׁעַל הָאֱגוֹזִים בִּשְׁעַת לְקִיטָתָן, אֵינָהּ מִצְטָרֶפֶת, דְּלָא חֲשִׁיבָא שׁוֹמֵר, שֶׁאֵין שׁוֹמֵר עַל גַּבֵּי שׁוֹמֵר: קְדוּרָה. נְקוּבָה. לְשׁוֹן מְקַדְּרִים בֶּהָרִים כְּדִתְנַן בְּעֵרוּבִין. וּקְדוּרָה אֵינָהּ מִצְטָרֶפֶת, לְפִי שֶׁאֵינָהּ שׁוֹמֶרֶת כְּשֶׁהִיא נְקוּבָה וַחֲתוּכָה: וְהַחִיצוֹנָה. הִיא הַקְּלִפָּה הַדַּקָּה הַנּוֹפֶלֶת מֵאֵלֶיהָ כְּשֶׁמְּמַשְׁמְשִׁין בַּבָּצָל. וְאֵין הֲלָכָה כְּרַבִּי יְהוּדָה :

הַמְחַתֵּךְ לְבַשֵּׁל. כְּגוֹן מְחַתֵּךְ דְּלוּעִים אוֹ בְצָלִים אוֹ בָשָׂר כְּדֵי לְבַשֵּׁל בַּקְּדֵרָה: שֶׁלֹּא מֵרַק. שֶׁלֹּא גָמַר הַחֲתִיכָה. כְּמוֹ וּמֵרַק אַחֵר שְׁחִיטָה עַל יָדוֹ, בְּמַסֶּכֶת יוֹמָא : אֵינוֹ חִבּוּר. אַף עַל פִּי שֶׁעֲדַיִן הַחֲתִיכוֹת מְעֹרוֹת בְּמִקְצָת: לִכְבֹּשׁ. שֶׁכּוֹבְשִׁים בְּיַיִן וּבְחֹמֶץ כְּדֵי לְהִתְקַיֵּם: וְלִשְׁלֹק. שְׁלִיקָה טְפֵי מִבִּשּׁוּל , וּבְהַאי נִיחָא לֵיהּ בְּחִבּוּרָן, הִלְכָּךְ כָּל זְמַן שֶׁלֹּא מֵרַק הָוֵי חִבּוּר. אֲבָל לְבַשֵּׁל לֹא הָוֵי חִבּוּר, דְּיוֹדֵעַ הוּא שֶׁהַבִּשּׁוּל יַפְרִידֵם וִיחַלְּקֵם זוֹ מִזּוֹ: הִתְחִיל לְפָרֵק. וּלְהַפְרִיד לְגַמְרֵי אֶחָד מֵהֶן, לֹא אָמְרִינַן כֵּיוָן דְּגַלֵּי דַעְתֵּיהּ בְּאֶחָד שֶׁרוֹצֶה לְהַפְרִידָן, כֻּלְּהוּ נַמִּי כְאִלּוּ מֻפְרָדִים זוֹ מִזּוֹ וְלֹא הָוֵי חִבּוּר, אֶלָּא אֹכֶל שֶׁפֵּרֵק בִּלְבָד לֹא הָוֵי חִבּוּר, וּמַה שֶּׁלֹּא פֵרֵק, הָוֵי חִבּוּר: הָאֱגוֹזִים שֶׁאֲמָנָן. שֶׁגִּדְּלָן וְחִבְּרָן זוֹ בָזוֹ. לְשׁוֹן וַיְהִי אֹמֵן אֶת הֲדַסָּה (אֶסְתֵּר ב): שֶׁחֲמָרָן. לְשׁוֹן וַיִּצְבְּרוּ אוֹתָם חֳמָרִם חֳמָרִם (שְׁמוֹת ח). מִדֶּרֶךְ הָרוֹצִים לֶאֱצֹר אֱגוֹזִים אוֹ בְצָלִים, שֶׁחוֹתְכִין אוֹתָן וְקוֹשְׁרִים אוֹתָן בְּחֶבֶל אוֹ בִמְשִׁיחָה וּמְחַבְּרִין אוֹתָן זוֹ עִם זוֹ וְתוֹלִין אוֹתָן כְּדֵי שֶׁיִּתְקַיְּמוּ: לְפַקֵּל. כְּמוֹ לְקַלֵּף. וְדֻגְמָתוֹ שָׁנִינוּ פֶּרֶק ד' דְּמַעַשְׂרוֹת מִשְׁנָה ו', הַבְּצָלִים מִשֶּׁיְּפַקֵּל: הָאֱגוֹזִים וְהַשְּׁקֵדִים. הַשְׁתָּא מַיְרֵי בִמְפַצֵּעַ הָאֱגוֹזִים וְהַשְּׁקֵדִים. שֶׁהַקְּלִפָּה שֶׁל אֱגוֹזִים וְשֶׁל שְׁקֵדִים חֲשִׁיבָא שׁוֹמֵר, וְאַף עַל פִּי שֶׁפְּצָעָן, כָּל זְמַן שֶׁהִיא מְחֻבֶּרֶת בָּאֹכֶל חֲשִׁיבָא שׁוֹמֵר: עַד שֶׁיְּרַסֵּס. לְשׁוֹן וְהִכָּה אֶת הַבַּיִת הַגָּדוֹל רְסִיסִים (עָמוֹס ו). כְּלוֹמַר עַד שֶׁתִּהְיֶה הַקְּלִפָּה רְצוּצָה וּפְתוּתָה לַחֲלָקִים דַּקִּים:

בֵּיצָה מְגֻלְגֶּלֶת. צְלוּיָה קְצָת, עַד שֶׁהִיא מְעֹרֶבֶת וּרְאוּיָה לְגָמְעָהּ: עַד שֶׁיָּגוֹס. הַקְּלִפָּה חֲשִׁיבָא שׁוֹמֵר וּמִצְטָרֶפֶת, עַד שֶׁיַּחְתֹּךְ מִמֶּנּוּ מְעַט , וְאָז לָא חֲשִׁיבָא יוֹתֵר שׁוֹמֵר, שֶׁהַבֵּיצָה שׁוֹתֶתֶת וְיוֹרֶדֶת דֶּרֶךְ אוֹתוֹ שֶׁבֶר וַחֲתָךְ: וּשְׁלוּקָה. אֲבָל אִם נִשְׁלְקָה הַבֵּיצָה, שֶׁהִיא קָשָׁה וְעוֹמֶדֶת בְּתוֹךְ הַקְּלִפָּה וַאֲפִלּוּ שֶׁנִּשְׁבְּרָה, לְעוֹלָם חֲשִׁיבָא הַקְּלִפָּה שׁוֹמֵר, עַד שֶׁתְּהֵא מְרֻצֶּצֶת: וְכֵן עֶצֶם שֶׁיֶּשׁ בּוֹ מֹחַ. חֲשִׁיב הָעֶצֶם שׁוֹמֵר לַמֹּחַ, וְאַף עַל פִּי שֶׁנִּשְׁבַּר הָעֶצֶם, עַד שֶׁיִּהְיֶה מְרֻצָּץ: הָרִמּוֹן שֶׁפְּרָדוֹ. שֶׁחָתַךְ הָרִמּוֹן לִשְׁנַיִם וַעֲדַיִן הַגַּרְגְּרִים מְחֻבָּרִין בַּקְּלִפָּה, הַוְיָא קְלִפָּה חִבּוּר, עַד שֶׁיַּקִּישׁ בְּקָנֶה וְיִתְנַעְנְעוּ הַגַּרְגְּרִים שֶׁבּוֹ: שְׁלַל הַכּוֹבְסִים. דֶּרֶךְ הַכּוֹבְסִים לִשְׁלֹל הַבְּגָדִים הַקָּטָן עִם הַגָּדוֹל כְּדֵי שֶׁלֹּא יֹאבַד, וְסוֹפוֹ לְהַפְרִידָן: וְהַבֶּגֶד שֶׁהוּא תָפוּר בְּכִלְאַיִם. שְׁתֵּי חֲתִיכוֹת שֶׁל בֶּגֶד צֶמֶר שֶׁתְּפָרָן בְּחוּטִין שֶׁל פִּשְׁתָּן, דְּוַדַּאי סוֹפוֹ לְהַפְרִידָן זוֹ מִזּוֹ, מִשּׁוּם אִסּוּר כִּלְאַיִם: חִבּוּר לְטֻמְאָה. וְאִם נִטְמָא הָאֶחָד נִטְמָא חֲבֵרוֹ, עַד שֶׁיַּתְחִיל לְהַתִּיר, וְאַף עַל פִּי שֶׁלֹּא גָמַר לְהַתִּיר, דְּמַעֲשֶׂה מוֹצִיא מִידֵי מַעֲשֶׂה:

יְרֻקִּים מִצְטָרְפִין. דַּחֲזוּ לַאֲכִילָה: לְבָנִים אֵין מִצְטָרְפִין. דְּתוּ לֹא חֲזוּ: וּבַחֲזָרִים. בַּחֲזֶרֶת, אַף עַל גַּב דִּלְבָנִים לֹא חֲזוּ לַאֲכִילָה, מִכָּל מָקוֹם הָווּ שׁוֹמֵר. וְאֵין הֲלָכָה כְּרַבִּי אֶלְעָזָר בַּר צָדוֹק:

פרק ב - משנה ח
בְּנֵי בְצָלִים. עַמּוּד הָאֶמְצָעִי שֶׁהוּא כְמוֹ לֵב (יָרוֹק) וּמִמֶּנּוּ צוֹמְחִים הַבְּצָלִים, לְהַהוּא קָרֵי בְנֵי בְצָלִים: אִם יֵשׁ בָּהֶן רִיר. שֶׁהֵם מְלֵאִים לֵחָה וְאֵין בָּהֶן חָלָל: מִשְׁתַּעֲרִין כְּמוֹ שֶׁהֵן. כְּשֶׁבָּאִים לְשַׁעֵר אִם יֵשׁ בָּהֶן אֹכֶל כַּבֵּיצָה שִׁעוּר טֻמְאָה, מְשַׁעֲרִין אוֹתָן עִם הָרִיר וְאֵין לְמַעֵךְ אֶת חֲלָלָן. וְאִם יֵשׁ בָּהֶן חָלָל, מְמַעֵךְ אֶת חֲלָלָן וְאַחַר כָּךְ מְשַׁעֵר לְכַבֵּיצָה: סְפוֹגָנִית. עֲשׂוּיָה רַכָּה כְּמִין סְפוֹג. וְאֵינָהּ דְּרוּסָה וּבְעוּטָה: מִשְׁתַּעֶרֶת כְּמוֹ שֶׁהִיא. בְּלֹא מִעוּךְ: בְּשַׂר הָעֵגֶל שֶׁנִּתְפַּח. דֶּרֶךְ בְּשַׂר הָעֵגֶל שֶׁהוּא נִתְפָּח וּמִתְרַבֶּה כְּשֶׁהוּא מִתְבַּשֵּׁל, וּבְשַׂר הַשּׁוֹר הַזָּקֵן מִתְמַעֵט: בִּכְמוֹת שֶׁהֵן. אִם בְּשַׂר שׁוֹר מְבֻשָּׁל הוּא שֶׁנִּתְמַעַט וְעָמַד עַל פָּחוֹת מִכַּבֵּיצָה, אֵין אוֹמְרִים כְּשֶׁהָיָה חַי הָיָה בוֹ כַּבֵּיצָה אוֹ יוֹתֵר, וִיטַמֵּא אַף עַל פִּי שֶׁעַכְשָׁיו אֵין בּוֹ כַּשִּׁעוּר. וְכֵן אִם הָיָה בְשַׂר עֵגֶל שֶׁנִּתְפַּח וְנִתְרַבָּה וְיֶשׁ בּוֹ כְּדֵי לְטַמֵּא, אֵין אוֹמְרִים הוֹאִיל וּכְשֶׁהָיָה חַי לֹא הָיָה בוֹ כְּדֵי לְטַמֵּא, אַף עַכְשָׁיו לֹא יְטַמֵּא:

פרק ב - משנה ט
קִשּׁוּת שֶׁנְּטָעָהּ בְּעָצִיץ. שֶׁאֵינוֹ נָקוּב, וְהֻכְשְׁרָה שָׁם לְקַבֵּל טֻמְאָה וְנִטְמֵאת, דְּכִתְלוּשָׁה דָמְיָא: וְהִגְדִּילָה וְיָצְאָה חוּץ לֶעָצִיץ. וְנֶחְשֶׁבֶת כִּמְחֻבָּר, לְפִי שֶׁיּוֹנֶקֶת וְשׁוֹאֶבֶת מִן הַקַּרְקַע שֶׁחוּץ לֶעָצִיץ: טְהוֹרָה. שֶׁפָּרְחָה לָהּ טֻמְאָה מִשֶּׁיָּצְאָה חוּץ לֶעָצִיץ: מַה טִּיבָהּ לִטָּהֵר. דִּכְשֶׁקִּבְּלָה טֻמְאָה הָיְתָה כִתְלוּשָׁה, הִלְכָּךְ מַה שֶּׁבְּתוֹךְ הֶעָצִיץ נִשְׁאָר בְּטֻמְאָתוֹ , וּמַה שֶׁהוֹסִיף אַחֲרֵי כֵן וְיָצָא חוּץ לֶעָצִיץ, טָהוֹר . וְאֵין הֲלָכָה כְּרַבִּי שִׁמְעוֹן:

כְּלֵי גְלָלִים. כֵּלִים הָעֲשׂוּיִין מִגִּלְלֵי הַבָּקָר: שֶׁהַשָּׁרָשִׁים יְכוֹלִין לָצֵאת בָּהֶן. אַף עַל פִּי שֶׁאֵינָן נְקוּבִים, הַשֹּׁרֶשׁ יוֹצֵא מֵעַצְמוֹ וְנוֹקֵב אֶת הַגָּלָל וְאֶת הָאֲדָמָה שֶׁל אוֹתוֹ כְלִי. וּלְהָכִי לֹא נָקַט כְּלֵי אֲבָנִים , לְפִי שֶׁקָּשִׁים הֵן וְאֵין הַשֹּׁרֶשׁ נוֹקֵב אוֹתָן: עָצִיץ נָקוּב אֵינוֹ מַכְשִׁיר אֶת הַזְּרָעִים. שֶׁבְּתוֹכוֹ, דְּלֹא מְקַבְּלֵי הֶכְשֵׁר מִשּׁוּם דַּחֲשִׁיבֵי כִּמְחֻבָּר, וְאֵין הַזְּרָעִים נִכְשָׁרִים לְקַבֵּל טֻמְאָה אֶלָּא כְשֶׁבָּא עֲלֵיהֶן מַיִם לְאַחַר שֶׁנִּתְלְשׁוּ. וַאֲפִלּוּ פֵרוֹת תְּלוּשִׁים אֵין נִכְשָׁרִים בַּמַּיִם שֶׁבְּתוֹכוֹ, לְפִי שֶׁאֵין הַמַּיִם מַכְשִׁירִין כְּשֶׁהֵן בִּמְחֻבָּר, דְּבָעִינַן שֶׁיְּהוּ הַמַּיִם תְּלוּשִׁין, וְהָנֵי מַיִם דְּמַנְּחֵי בְעָצִיץ נָקוּב, כְּמַנְּחֵי בְאַרְעָא דָמוּ: וְשֶׁאֵינוֹ נָקוּב מַכְשִׁיר אֶת הַזְּרָעִים. דְּבֵין מַיִם שֶׁבְּתוֹכוֹ וּבֵין זְרָעִים, תַּרְוַיְהוּ חֲשִׁיבֵי כִתְלוּשִׁים: מִלְאָהוּ. לֶעָצִיץ שֶׁאֵינוֹ נָקוּב: עָפָר עַד שְׂפָתוֹ. הוֹאִיל וְרוֹאֶה אֶת הָאֲוִיר מֵאַרְבַּע רוּחוֹת, הָוֵי כֶעָצִיץ נָקוּב, וַחֲשִׁיבֵי כִמְחֻבָּר, בֵּין הַמַּיִם, וּבֵין הַזְּרָעִים שֶׁבּוֹ: כְּטַבְלָא שֶׁאֵין לָהּ לִזְבֵּז. שֶׁאֵין לָהּ שָׂפָה מִכָּאן וּמִכָּאן אֶלָּא פְשׁוּטָה וַחֲלָקָה, דְּמַיִם שֶׁבָּהּ אֵינָן מַכְשִׁירִין אֶת הַזְּרָעִים וַחֲשִׁיבֵי כִמְחֻבָּר, הוֹאִיל וְרוֹאִין אֶת הָאֲוִיר מֵאַרְבַּע רוּחוֹת:
