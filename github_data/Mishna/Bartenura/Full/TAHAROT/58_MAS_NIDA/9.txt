פרק ט

הָאִשָּׁה שֶׁהִיא עוֹשָׂה צְרָכֶיהָ. מַשְׁתֶּנֶת מַיִם: אִם עוֹמֶדֶת טְמֵאָה. דְּאָמְרִינַן מֵי רַגְלַיִם הֲדוּר לַמָּקוֹר וַאֲתָא דָּם מִן הַמָּקוֹר בַּהֲדַיְהוּ: יוֹשֶׁבֶת טְהוֹרָה. בַּגְּמָרָא מוֹקְמִינַן לַהּ בְּיוֹשֶׁבֶת עַל שְׂפַת הַסֵּפֶל וּמְקַלַּחַת מֵי רַגְלַיִם לְתוֹכוֹ בְּחֹזֶק. וּכְשֶׁהִיא מְקַלַּחַת בְּחֹזֶק, אֵין דֶּרֶךְ דַּם הַמָּקוֹר לָצֵאת עִם מֵי רַגְלַיִם. וְאִי לְבָתַר דְּתַמּוּ מַיָּא בָּא דָּם זֶה מִן הַמָּקוֹר, לֹא הָיָה הַדָּם נִמְצָא בְּתוֹךְ הַסֵּפֶל אֶלָּא עַל שְׂפַת הַסֵּפֶל, שֶׁדָּם הַבָּא מִן הַמָּקוֹר, שׁוֹתֵת וְיוֹרֵד וְאֵינוֹ מְקַלֵּחַ. וְדָם זֶה שֶׁנִּמְצָא בְּתוֹךְ הַסֵּפֶל, עַל כָּרְחָךְ מִמְּקוֹם מֵי רַגְלַיִם בָּא וּמַכָּה יֵשׁ בָּהּ שָׁם, וְלֹא מִן הַמָּקוֹר. וּבְעוֹמֶדֶת עַל שְׂפַת הַסֵּפֶל וּמְקַלַּחַת בְּתוֹךְ הַסֵּפֶל, לֹא הָיָה אֶפְשָׁר לְהַעֲמִידָהּ, דְּעוֹמֶדֶת אִי אֶפְשָׁר לָהּ לְקַלֵּחַ , וּלְהָכִי אִצְטְרִיךְ תַּנָּא לְמִנְקַט יוֹשֶׁבֶת: רַבִּי יוֹסֵי אוֹמֵר בֵּין כָּךְ וּבֵין כָּךְ טְהוֹרָה. וַהֲלָכָה כְּרַבִּי יוֹסֵי:

שֶׁעָשׂוּ צְרָכֵיהֶן. הִטִּילוּ מֵי רַגְלַיִם: רַבִּי יוֹסֵי מְטַהֵר. דְּהָא אֲפִלּוּ בְּאִשָּׁה גְּרֵידְתָא דְּלֵיכָּא אֶלָּא חֲדָא סְפֵקָא, סָפֵק בָּא מִן הַמָּקוֹר, סָפֵק בָּא מִמְּקוֹם מֵי רַגְלַיִם, מְטַהֵר רַבִּי יוֹסֵי לְעֵיל, וְכָל שֶׁכֵּן הָכָא דְּדִלְמָא מֵאִישׁ אָתֵי. וְלֹא הָדַר תַּנָּא הָכָא רַבִּי יוֹסֵי מְטַהֵר, אֶלָּא לְאַשְׁמוֹעִינַן מִמִּשְׁנָה יְתֵירְתָא דַּאֲפִלּוּ לְכַתְּחִלָּה מְטַהֵר רַבִּי יוֹסֵי לְאִשָּׁה שֶׁעוֹשָׂה צְרָכֶיהָ וְרָאֲתָה דָּם, שֶׁתִּתְעַסֵּק בְּטָהֳרוֹת . דְּלֹא תֵימָא דַּוְקָא דִּיעֲבַד מְטַהֵר רַבִּי יוֹסֵי טָהֳרוֹת שֶׁנִּתְעַסְּקָה בָּהֶן, אֲבָל לְכַתְּחִלָּה לֹא אָמְרִינַן לַהּ שֶׁתִּתְעַסֵּק, קָמַשְׁמַע לָן: וְרַבִּי שִׁמְעוֹן מְטַמֵּא. אֲפִלּוּ הָכָא דְּאִיכָּא סְפֵק סְפֵקָא, כִּדְקָתָנֵי טַעֲמָא: שֶׁחֶזְקַת דָּמִים מִן הָאִשָּׁה. וּלְעֵיל פָּסְקִינַן הֲלָכָה כְּרַבִּי יוֹסֵי דִּמְטַהֵר וַאֲפִלּוּ בְּחַד סְפֵקָא, כָּל שֶׁכֵּן הָכָא דְּאִיכָּא תַּרְתֵּי:

הִשְׁאִילָה חֲלוּקָהּ. וְאַחַר כָּךְ לְבָשַׁתָּהּ הִיא וּמָצְאָה עָלָיו כֶּתֶם: לְנָכְרִית אוֹ לְנִדָּה. לְנָכְרִית שֶׁהִיא רוֹאָה דָּם, שֶׁהִיא גְּדוֹלָה וְיָדַעְנוּ בָּהּ שֶׁכְּבָר רָאֲתָה , דּוּמְיָא דְּנִדָּה: הֲרֵי זוֹ תּוֹלָה בָהּ. וּטְהוֹרָה. דְּכֵיוָן דְּנִדָּה הִיא, לֹא מְקַלְקְלָה לַהּ מִידִי. וְנָכְרִית נַמִּי טְמֵאָה וְתָלְיָא בָּהּ. אֲבָל יִשְׂרְאֵלִית טְהוֹרָה שֶׁשָּׁאֲלָה חָלוּק מֵחֲבֶרְתָּהּ טְהוֹרָה, וְאַחַר כָּךְ לְבָשַׁתּוּ חֲבֶרְתָּהּ, שְׁתֵּיהֶן טְמֵאוֹת, כִּדְקָתָנֵי שָׁלֹשׁ נָשִׁים וְכוּ', דְּכֵיוָן דְּאִי תָּלְיָא בָּהּ, מְקַלְקְלָא לַהּ, מַאי חָזֵית דִּמְקַלְקֶלֶת לְהַךְ טְפֵי מֵהַךְ: שֶׁל אֶבֶן. דְּאֵינוֹ מְקַבֵּל טֻמְאָה. וְסַפְסָל דְּרֵישָׁא בִּכְלִי עֵץ דְּרָאוּי לִטָּמֵא בְּמוֹשַׁב הַזָּב וְהַנִּדָּה: אֵינוֹ מְקַבֵּל כְּתָמִים. כְּלוֹמַר לֹא גָּזְרוּ עַל כְּתָמִים הַנִּמְצָאִים בּוֹ. וְטַעֲמָא דְּרַבִּי נֶחֶמְיָה, דִּכְתִיב (יְשַׁעְיָה ג) וְנִקָּתָה לָאָרֶץ תֵּשֵׁב, כֵּיוָן שֶׁיָּשְׁבָה לָאָרֶץ שֶׁאֵינָהּ מְקַבֶּלֶת טֻמְאָה, נִקָּתָה. וְהוּא הַדִּין לְכָל דָּבָר שֶׁאֵינוֹ מְקַבֵּל טֻמְאָה. וְכֵן אִם יָשְׁבָה עַל גַּבֵּי כְּלִי חֶרֶס, שֶׁאֵין מְקַבֵּל טֻמְאָה מִגַּבּוֹ, אוֹ עַל גַּבֵּי מַטְלָנִיּוֹת שֶׁל בֶּגֶד שֶׁאֵין בָּהֶן שָׁלֹשׁ עַל שָׁלֹשׁ שֶׁאֵינָן מְקַבְּלִין טֻמְאָה, בְּכָל אֵלּוּ לֹא גָּזְרוּ עַל כְּתָמִים הַנִּמְצָאִים בָּהֶם. וַהֲלָכָה כְּרַבִּי נֶחֶמְיָה:

וְתוֹלוֹת זוֹ בָזוֹ. שֶׁאִם הָיְתָה הָאַחַת מְעֻבֶּרֶת , תּוֹלָה הַדָּם בְּשֶׁאֵינָהּ מְעֻבֶּרֶת, וְהִיא טְהוֹרָה: וְאִם לֹא הָיוּ רְאוּיוֹת. כְּגוֹן שֶׁהָיוּ כֻּלָּן מְעֻבָּרוֹת אוֹ כֻּלָּן מֵנִיקוֹת, כֻּלָּן טְמֵאוֹת כְּאִלּוּ רְאוּיוֹת לִרְאוֹת, דְּמִכָּל מָקוֹם דָּם מִבֵּינַיְהוּ נְפַק:

תַּחַת הַפְּנִימִית. אוֹתָהּ שֶׁל צַד הַקִּיר קְרוּיָה פְּנִימִית, וְחִיצוֹנָה קְרוּיָה אוֹתָהּ שֶׁל צַד דֶּרֶךְ עֲלִיַּת הַמִּטָּה. וְהָא דִּמְפַלֵּיג תַּנָּא הָכָא בֵּין כְּשֶׁנִּמְצָא תַּחַת הַפְּנִימִית לְנִמְצָא תַּחַת הַחִיצוֹנָה, וּלְעֵיל קָתָנֵי וְנִמְצָא דָּם תַּחַת אַחַת מֵהֶן כֻּלָּן טְמֵאוֹת, וְלֹא מְפַלֵּיג בֵּין נִמְצָא תַּחַת פְּנִימִית לְנִמְצָא תַּחַת חִיצוֹנָה, מְפָרֵשׁ בַּגְּמָרָא דִּלְעֵיל אַיְרֵי כְּשֶׁהֵן תְּכוּפוֹת וּדְבוּקוֹת יַחַד, הִלְכָּךְ אִיכָּא לְסַפּוֹקֵי בַּחִיצוֹנָה אֲפִלּוּ כְּשֶׁנִּמְצָא הַדָּם תַּחַת הַפְּנִימִית: אֵימָתַי, אָמְרוּ נִמְצָא תַּחַת הַחִיצוֹנָה פְּנִימִית טְהוֹרָה: בִּזְמַן שֶׁעָבְרוּ דֶרֶךְ מַרְגְּלוֹת הַמִּטָּה. דְּלֹא עָבְרָה פְּנִימִית עַל מְקוֹם מְצִיאַת הַדָּם: אֲבָל עָבְרוּ שְׁלָשְׁתָּן דֶּרֶךְ עָלֶיהָ. שֶׁהֵן דּוֹרְסוֹת וְעוֹבְרוֹת עַל הַחִיצוֹנָה: כֻּלָּן טְמֵאוֹת. וַאֲפִלּוּ פְּנִימִית, דְּשֶׁמָּא כְּשֶׁעָבְרָה פְּנִימִית דֶּרֶךְ שָׁם נָפַל מִמֶּנָּה: לְגַּל טָמֵא. שֶׁהָיָה שָׁם כַּזַּיִת מִן הַמֵּת: עַד שֶׁמַּגִּיעַ לְסֶלַע אוֹ לִבְתוּלָה. קַרְקַע בְּתוּלָה. וְאִם לֹא מָצָא, הַגַּל טָהוֹר, דְּאֵמוּר בָּא עוֹרֵב וְנָטַל אֶת הַטֻּמְאָה. אֲבָל הָכָא כְּשֶׁבָּדְקוּ שְׁלָשְׁתָּן וּמָצְאוּ טְהוֹרוֹת, מוֹדוּ רַבָּנָן לְרַבִּי מֵאִיר דְּכֻלָּן טְמֵאוֹת, דְּהַאי דָּם מֵהֵיכָא אֲתָא, עַל כָּרְחָךְ מֵחֲדָא מִנַּיְהוּ הוּא. וַהֲלָכָה כַּחֲכָמִים:

שִׁבְעָה סַמְמָנִים מַעֲבִירִין עַל הַכֶּתֶם. שֶׁאֵין הַדָּם בָּטֵל עַד שֶׁיַּעֲבִיר כֻּלָּן עָלָיו: רֹק תָּפֵל וּמֵי גְרִיסִין. מְפָרֵשׁ לְקַמָּן בְּמַתְנִיתִין: נֶתֶר. אלו"ם בְּלַעַ"ז. וּמִין קַרְקַע הוּא וּמַזְהִיר. וּבַעֲרָבִי קוֹרִין לוֹ שי"ב: בּוֹרִית. מִין צֶמַח הוּא שֶׁמְּנַקֶּה וּמְטַהֵר: קְמוּנְיָא. עֵשֶׂב שֶׁמְּיַבְּשִׁים אוֹתוֹ וְטוֹחֲנִים אוֹתוֹ וּמְנַקִּין בַּעֲפָרוֹ אֶת הַיָּדַיִם לְהַעֲבִיר אֶת הַזֻּהֲמָא. וּבִלְשׁוֹן גְּמָרָא קָרוּי שְׁלוֹף דּוֹץ: אֶשְׁלָג. לֹא אִתְפָּרַשׁ לִי: הִטְבִּילוֹ. לַבֶּגֶד שֶׁיֵּשׁ בּוֹ כֶּתֶם קֹדֶם שֶׁהֶעֱבִיר עָלָיו הַסַּמְמָנִים הַלָּלוּ: הֲרֵי זֶה צֶבַע. דְּאִי הֲוָה דָּם הֲוָה עָבַר: שֶׁדֵּהָה. נִשְׁתַּנָּה מֵאַדְמִימוּתוֹ:

שֶׁלֹּא טָעַם כְּלוּם מִבָּעֶרֶב. כָּל אוֹתָהּ הַלַּיְלָה, וְהַיְנוּ תָּפֵל בְּלֹא טַעַם. וְהוּא שֶׁיָּשַׁן מֵחֲצוֹת אַחֲרוֹן שֶׁל לַיְלָה עַד הַבֹּקֶר. וְעַד שָׁלֹשׁ שָׁעוֹת בַּיּוֹם נִקְרָא רֹק תָּפֵל לְעִנְיָן זֶה. וְאִם הִשְׁכִּים בַּבֹּקֶר וְשָׁנָה פִּרְקוֹ, לֹא הָוֵי רֹק תָּפֵל : לְעִסַּת גְּרִיסִין שֶׁל פּוֹל חֲלוּקַת נָפֶשׁ. שֶׁנֶּחֱלַק הָאֹכֶל מִן הַקְּלִפָּה. לוֹעֲסָה לִגְרִיסַת הַפּוֹל וּמַעֲבִירוֹ עַל הַכֶּתֶם: שֶׁהֶחֱמִיצוּ. שֶׁהִסְרִיחוּ. וְכַמָּה חִמּוּצָן, כָּל שֶׁשָּׁהוּ שְׁלֹשָׁה יָמִים: וְצָרִיךְ לְכַסְכֵּס. לְשַׁפְשֵׁף שָׁלֹשׁ פְּעָמִים לְכָל סַם וְסַם, כְּדֶרֶךְ כִּבּוּס בְּגָדִים שֶׁכּוֹפֵל צַד עִם צִדּוֹ וּמְשַׁפְשֵׁף: כְּסִדְרָן. שֶׁשְּׁנוּיִים כָּאן: לֹא עָשָׂה וְלֹא כְלוּם. לֹא לִבְדֹּק וְלֹא לְבַטֵּל :

כָּל אִשָּׁה שֶׁיֶּשׁ לָהּ וֶסֶת דַּיָּהּ שְׁעָתָהּ. לְעֵיל בְּפֶרֶק קַמָּא תְּנֵינָא לַהּ, וְהָתָם מַיְרֵי בִּוְסָתוֹת דְּיוֹמֵי כְּגוֹן מֵחֲמִשָּׁה עָשָׂר יוֹם לַחֲמִשָּׁה עָשָׂר יוֹם, אוֹ מֵחֹדֶשׁ לְחֹדֶשׁ, וְהָכָא מַיְרֵי בִּוְסָתוֹת דִּבְגוּפָהּ, שֶׁכָּל זְמַן שֶׁיֵּרָאוּ בָּהּ סִימָנִים הַלָּלוּ רְגִילָה לִרְאוֹת בָּהֶן, אֲבָל אֵין לָהּ זְמַן קָבוּעַ לִרְאוֹת: מְפַהֶקֶת. יֵשׁ מְפָרְשִׁים, פּוֹתַחַת פִּיהָ בְּכָל כֹּחָהּ. אלחר"י בְּלַעַ"ז. וְיֵשׁ מְפָרְשִׁים, פּוֹשֶׁטֶת זְרוֹעוֹתֶיהָ וְגוּפָהּ. אשטרנ"י קלא"ר בְּלַעַ"ז: בְּפִי כְרֵסָהּ. נֶגֶד טַבּוּרָהּ: שִׁפּוּלֵי מֵעֶיהָ. בֵּית הָרֶחֶם: וְשׁוֹפַעַת. מְפָרֵשׁ בַּגְּמָרָא שֶׁשּׁוֹפַעַת דָּם טָמֵא מִתּוֹךְ דָּם טָהוֹר, כְּגוֹן שֶׁהִיא רְגִילָה לִרְאוֹת דָּם טָהוֹר תְּחִלָּה, דְּהַיְנוּ דָּם שֶׁאֵינוֹ מֵחֲמִשָּׁה דָּמִים טְמֵאִין, וְאַחַר כָּךְ דָּם טָמֵא. וְהַאי נַמִּי הָוֵי וֶסֶת. וְכָל שָׁעָה שֶׁתִּרְאֶה דָּם טָמֵא אַחַר דָּם טָהוֹר, דַּיָּהּ שַׁעְתָּהּ: צְמַרְמוֹרֶת. לְשׁוֹן סָמַר מִפַּחְדְּךָ בְשָׂרִי (תְּהִלִּים קיט). אריצמינ"ט בְּלַעַ"ז: אוֹחֲזִין אוֹתָהּ. כָּךְ לְמוּדָה לִהְיוֹת בְּכָל עֵת שֶׁהִיא רוֹאָה: וְכֵן כַּיּוֹצֵא בָהֶן. מִשְּׁאָר סִימָנִים שֶׁהַנָּשִׁים הָרוֹאוֹת דָּם לְמוּדוֹת לִהְיוֹת בְּכָךְ. כְּגוֹן שֶׁרֹאשָׁהּ כָּבֵד עָלֶיהָ, אוֹ אוֹחֲזָהּ רֶתֶת וָרַעַד, אוֹ כְּאֵב לֵב, וְכַיּוֹצֵא בָּזֶה, אִם קָבְעָה לָהּ שָׁלֹשׁ פְּעָמִים הֲרֵי זֶה וֶסֶת:

לְמוּדָה לִהְיוֹת רוֹאָה בִתְחִלַּת הַוְּסָתוֹת. שֶׁבְּגוּפָהּ, כְּגוֹן בִּתְחִלַּת הָעִטּוּשִׁים וְהַפִּהוּקִין וְכַיּוֹצֵא בָּהֶן: אַף יָמִים וְשָׁעוֹת וְסָתוֹת. שָׁעוֹת נַמִּי וְסָתוֹת נִינְהוּ, וּבָתַר דִּידְהוּ אָזְלִינַן. שֶׁאִם הָיְתָה רְגִילָה לִרְאוֹת מֵעֶשְׂרִים יוֹם לְעֶשְׂרִים יוֹם וּמִשֵּׁשׁ שָׁעוֹת לְשֵׁשׁ שָׁעוֹת, מֻתָּרוֹת לְשַׁמֵּשׁ בְּיוֹם עֶשְׂרִים עַד שֵׁשׁ שָׁעוֹת, וְחוֹשֶׁשֶׁת בְּשָׁעָה שִׁשִּׁית לְבַד . עָבְרָה שָׁעָה שִׁשִּׁית וְלֹא רָאֲתָה, מֻתֶּרֶת לְשַׁמֵּשׁ כָּל הַיּוֹם כֻּלּוֹ: הָיְתָה לְמוּדָה לִהְיוֹת רוֹאָה עִם הָנֵץ הַחַמָּה. סִיּוּמָא דְּמִלְּתָא דְּרַבִּי יוֹסֵי הִיא: אֵינָהּ אֲסוּרָה אֶלָּא עִם הָנֵץ הַחַמָּה. וְכָל הַלַּיְלָה מְשַׁמֶּשֶׁת. וְאִם עָבַר הָנֵץ הַחַמָּה וְלֹא רָאֲתָה, מְשַׁמֶּשֶׁת כָּל הַיּוֹם. דְּדַוְקָא שְׁעַת הַוְּסָתוֹת חוֹשֶׁשֶׁת, וְתוּ לֹא: רַבִּי יְהוּדָה אוֹמֵר כָּל הַיּוֹם שֶׁלָּהּ. כְּלוֹמַר מֻתֶּרֶת לְשַׁמֵּשׁ כָּל הַיּוֹם, מִשֶּׁעָבַר הָנֵץ הַחַמָּה וְלֹא רָאֲתָה. אֲבָל כָּל הַלַּיְלָה שֶׁלִּפְנֵי הָנֵץ הַחַמָּה שֶׁל וֶסְתָּהּ, אֲסוּרָה לְשַׁמֵּשׁ. וְעִם הָנֵץ הַחַמָּה דְּקָתָנֵי, הַיְנוּ דִּרְגִילָה לְמִחְזֵי בְּסוֹף לֵילְיָא מִקַּמֵּי דִּלְמָטֵי תְּחִלַּת יְמָמָא, וַאֲנַן אָסְרִינַן לַהּ כָּל עוֹנַת וֶסְתָּהּ. נִמְצֵאת אֲסוּרָה כָּל הַלַּיְלָה שֶׁהִיא עוֹנַת וֶסְתָּהּ. וַהֲלָכָה כְּרַבִּי יְהוּדָה:

וְאֵינָהּ מִטַּהֶרֶת מִן הַוֶּסֶת. שֶׁפָּסַק מִמֶּנָּה, שֶׁלֹּא לָחֹשׁ לוֹ:

יֵשׁ גֶּפֶן שֶׁיֵּינָהּ אָדֹם וְיֵשׁ גֶּפֶן שֶׁיֵּינָהּ שָׁחֹר. כָּךְ יֵשׁ אִשָּׁה שֶׁדָּם שֶׁלָּהּ אָדֹם, וְיֵשׁ שֶׁדָּם שֶׁלָּהּ שָׁחֹר: כָּל גֶּפֶן יֵשׁ בָּהּ יַיִן. כָּל אִשָּׁה יֵשׁ לָהּ דַּם נִדָּה וְדַם בְּתוּלִים: וְשֶׁאֵין בָּהּ הֲרֵי זֶה דּוֹרְקְטִי. דּוֹר קָטוּעַ. כְּלוֹמַר קְטוּעַ בָּנִים, עֲקָרָה שֶׁאֵינָהּ יוֹלֶדֶת:
