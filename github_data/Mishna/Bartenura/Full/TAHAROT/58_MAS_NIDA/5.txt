פרק ה

יוֹצֵא דֹפֶן. אִשָּׁה שֶׁפָּתְחוּ עַל יְדֵי סַם וְהוֹצִיאוּ הָעֻבָּר לַחוּץ וְנִתְרַפְּאָה: אֵין יוֹשְׁבִין עָלָיו יְמֵי טֻמְאָה. שִׁבְעָה לְזָכָר וּשְׁבוּעַיִם לִנְקֵבָה: וִימֵי טָהֳרָה. וְאֵין לָהּ דַּם טֹהַר: וְאֵין חַיָּבִין עָלָיו קָרְבָּן. שֶׁל יוֹלֶדֶת. דִּכְתִיב (וַיִּקְרָא יב) אִשָּׁה כִּי תַזְרִיעַ וְיָלְדָה, עַד שֶׁתֵּלֵד מִמָּקוֹם שֶׁמַּזְרַעַת: רַבִּי שִׁמְעוֹן אוֹמֵר הֲרֵי זֶה כְיָלוּד. שֶׁנֶּאֱמַר (שָׁם) וְאִם נְקֵבָה תֵלֵד , לְרַבּוֹת יוֹצֵא דֹּפֶן. וְאֵין הֲלָכָה כְּרַבִּי שִׁמְעוֹן: כָּל הַנָּשִׁים מִטַּמְּאוֹת. בְּנִדָּה מִשֶּׁיָּצָא הַדָּם מִן הַמָּקוֹר וְהִגִּיעַ לַבַּיִת הַחִיצוֹן, וְאַף עַל גַּב דְּהֶעֱמִידוּהוּ כָּתְלֵי בֵּית הָרֶחֶם וְלֹא פֵּרַשׁ לַחוּץ. דִּכְתִיב דָּם יִהְיֶה זֹבָהּ בִּבְשָׂרָהּ, מְלַמֵּד שֶׁמְּטַמְּאָה אַף עַל פִּי שֶׁעֲדַיִן הוּא בִּבְשָׂרָהּ. וּבַיִת הַחִיצוֹן מִקְרֵי מָקוֹם שֶׁהָאֵבָר דָּשׁ בּוֹ בִּשְׁעַת בִּיאָה: אֵינָן מִטַּמְּאִין עַד שֶׁתֵּצֵא טֻמְאָתָן לַחוּץ. דִּכְתִיב בַּזָּב (שָׁם) אִישׁ כִּי יִהְיֶה זָב מִבְּשָׂרוֹ, עַד שֶׁתֵּצֵא זוֹבוֹ מִבְּשָׂרוֹ. וּבְבַעַל קֶרִי כְּתִיב (שָׁם כב) אֲשֶׁר תֵּצֵא מִמֶּנּוּ שִׁכְבַת זָרַע:

שֶׁנִּזְדַּעְזְעוּ אֵבָרָיו. שֶׁנֶּעֶקְרָה שִׁכְבַת זֶרַע מִגּוּפוֹ: אוֹחֵז בָּאַמָּה. וְאַף עַל גַּב דְּכָל הָאוֹחֵז בָּאַמָּה וּמַשְׁתִּין כְּאִלּוּ מֵבִיא מַבּוּל לָעוֹלָם, הָכָא כֵּיוָן דְּנֶעֶקְרָה שִׁכְבַת זֶרַע שׁוּב אֵינוֹ חוֹזֵר וּמִתְחַמֵּם לַעֲקֹר עוֹד מִיָּד בְּאוֹתָהּ שָׁעָה: וּמְטַמְּאִין. זוֹב וְשִׁכְבַת זֶרַע: בְּכָל שֶׁהֵן. לָאָדָם הַיּוֹצֵא מִמֶּנּוּ. אֲבָל הַנּוֹגֵעַ, אֵינוֹ טָמֵא עַד שֶׁיִּגַּע בְּכָעֲדָשָׁה:

מִטַּמְּאָה בְנִדָּה. אִם רָאֲתָה דָּם: בַּת עֲשָׂרָה יָמִים מִטַּמְּאָה בְזִיבָה. אִם רָאֲתָה שְׁלֹשָׁה יָמִים רְצוּפִים לְאַחַר שִׁבְעָה. דְּכָל דָּם שֶׁרוֹאָה בְּתוֹךְ שִׁבְעָה דַּם נִדָּה הוּא, הִלְכָּךְ אִי אֶפְשָׁר לָהּ לִטָּמֵא בַּזִּיבָה דְּהַיְנוּ לִהְיוֹת זָבָה גְּדוֹלָה, עַד עֲשָׂרָה יָמִים: תִּינוֹק בֶּן יוֹם אֶחָד מִטַּמֵּא בְזִיבָה. דִּכְתִיב (שָׁם טו) אִישׁ אִישׁ כִּי יִהְיֶה זָב, לִמֵּד עַל תִּינוֹק בֶּן יוֹמוֹ שֶׁמִּטַּמֵּא בְּזִיבָה: וּמִטַּמֵּא בִנְגָעִים. דְּגַבֵּי נְגָעִים כְּתִיב (שָׁם יג) אָדָם כִּי יִהְיֶה בְעוֹר בְּשָׂרוֹ, וְתִינוֹק בֶּן יוֹמוֹ אָדָם הוּא, דְּאָדָם כָּל שֶׁהוּא מַשְׁמַע: וּמִטַּמֵּא בִטְמֵא מֵת. דִּכְתִיב (בְּמִדְבַּר יט) וְעַל הַנְּפָשׁוֹת אֲשֶׁר הָיוּ שָׁם, נֶפֶשׁ כָּל שֶׁהוּא : וְזוֹקֵק לְיִבּוּם. אִם נוֹלַד יוֹם אֶחָד קֹדֶם מִיתַת אָחִיו. אֲבָל נוֹלַד לְאַחַר מִיתַת אָחִיו, אֵינוֹ זוֹקֵק, דִּכְתִיב (דְּבָרִים כח) כִּי יֵשְׁבוּ אַחִים יַחְדָּו, שֶׁהָיְתָה לָהֶם יְשִׁיבָה אַחַת בָּעוֹלָם : וּפוֹטֵר מִן הַיִּבּוּם. אִם נוֹלַד לְאַחַר מִיתַת אָבִיו וְחָיָה שָׁעָה אַחַת וּמֵת . אֲבָל כָּל זְמַן שֶׁמְּעֻבֶּרֶת, אֵינוֹ מַתִּירָהּ לִנָּשֵׂא: וּמַאֲכִיל בַּתְּרוּמָה. בַּת יִשְׂרָאֵל שֶׁנִּסֵּת לְכֹהֵן וּמֵת, וְנוֹלַד בֵּן לְאַחַר מִיתָתוֹ, אוֹכֶלֶת בִּשְׁבִילוֹ בּוֹ בַּיּוֹם שֶׁנּוֹלַד, כְּדִכְתִיב (וַיִּקְרָא כב) וִילִיד בֵּיתוֹ הֵם יֹאכְלוּ בְלַחְמוֹ, וְדָרְשִׁינַן יַאֲכִילוּ אֶת אִמָּם. אֲבָל הִנִּיחָהּ מְעֻבֶּרֶת, אֵינָהּ אוֹכֶלֶת בִּשְׁבִילוֹ עַד שֶׁיִּוָּלֵד: וּפוֹסֵל מִן הַתְּרוּמָה. כְּגוֹן כֹּהֵן שֶׁהָיוּ לוֹ שְׁתֵּי נָשִׁים, אַחַת גְּרוּשָׁה שֶׁעָבַר וּנְשָׂאָהּ כְּשֶׁהִיא גְרוּשָׁה, וְאַחַת שֶׁאֵינָהּ גְּרוּשָׁה, וְיֵשׁ לוֹ בָּנִים מִשֶּׁאֵינָהּ גְּרוּשָׁה, וַעֲבָדִים שֶׁנָּפְלוּ לָהֶן בִּירֻשָּׁה אוֹכְלִים בַּתְּרוּמָה בִּשְׁבִילָן, דִּכְתִיב (שָׁם) וְכֹהֵן כִּי יִקְנֶה נֶפֶשׁ קִנְיַן כַּסְפּוֹ, וְיֵשׁ לוֹ בֶּן יוֹם אֶחָד מִן הַגְּרוּשָׁה וְהוּא חָלָל, וְיֵשׁ לוֹ חֵלֶק בָּעֲבָדִים, וּפוֹסֵל אֶת כֻּלָּן מִן הַתְּרוּמָה , דְּאֵין בְּרֵרָה לָדַעַת מִי מֵהֶן מַגִּיעַ לְחֶלְקוֹ. וְדַוְקָא בֶּן יוֹם אֶחָד דְּרָאוּי לִירֻשָּׁה. אֲבָל כָּל זְמַן שֶׁלֹּא נוֹלַד, אוֹכְלִים כֻּלָּן, דְּאֵין זְכִיָּה לְעֻבָּר : וְנוֹחֵל וּמַנְחִיל. נוֹחֵל אֶת הָאֵם. אִם מֵתָה אִמּוֹ בַּיּוֹם שֶׁנּוֹלַד, הֲרֵי הוּא נוֹחֲלָהּ. וּכְשֶׁמֵּת הוּא, בָּאִין אֶחָיו מֵאָבִיו וְיוֹרְשִׁים הֵימֶנּוּ נִכְסֵי אִמּוֹ. אֲבָל עֻבָּר שֶׁבְּמֵעֶיהָ, אִם מֵתָה מְעֻבֶּרֶת אֵינוֹ נוֹחֵל אֶת אִמּוֹ לְהַנְחִיל לְאֶחָיו מֵאָבִיו, שֶׁהֲרֵי מֵת הוּא קֹדֶם אִמּוֹ, וְאֵין הַבֵּן יוֹרֵשׁ אֶת אִמּוֹ בַּקֶּבֶר לְהַנְחִיל לְאֶחָיו מֵאָבִיו: וְהַהוֹרְגוֹ חַיָּב. דִּכְתִיב (וַיִּקְרָא כד) וְאִישׁ כִּי יַכֶּה כָּל נֶפֶשׁ, מִכָּל מָקוֹם: כְּחָתָן שָׁלֵם. שֶׁחַיָּבִים לְהִתְאַבֵּל עָלָיו. וְהָנֵי מִלֵּי, דְּקִים לְהוּ בְּגַוֵּיהּ שֶׁכָּלוּ לוֹ חֳדָשָׁיו, אֲבָל לֹא קִים לְהוּ שֶׁכָּלוּ לוֹ חֳדָשָׁיו, אֵין חַיָּבִין לְהִתְאַבֵּל עָלָיו עַד שֶׁיִּהְיֶה בֶּן שְׁלֹשִׁים וְאֶחָד יוֹם:

מִתְקַדֶּשֶׁת בְּבִיאָה. אִם מְסָרָהּ אָבִיהָ לְכָךְ: וְאִם בָּא עָלֶיהָ יָבָם קְנָאָהּ. לִהְיוֹת אִשְׁתּוֹ לְכָל דָּבָר, וְיוֹרְשָׁהּ וּמִטַּמֵּא לָהּ וְזָכָה בְּנִכְסֵי אָחִיו: וְחַיָּבִין עָלֶיהָ מִשּׁוּם אֵשֶׁת אִישׁ. אִם קִבֵּל בָּהּ אָבִיהָ קִדּוּשִׁין מֵאֶחָד, וּבָא עָלֶיהָ אַחֵר, נֶהֱרָג עָלֶיהָ , דְּבִיאָתָהּ בִּיאָה: וּמְטַמְּאָה אֶת בּוֹעֲלָהּ. אִם הִיא נִדָּה. אֲבָל פָּחוֹת מִכָּן, אַף עַל פִּי שֶׁהִיא מְטַמְּאָה בְּמַגָּע, אֵינָהּ מְטַמְּאָה אֶת בּוֹעֲלָהּ מִשּׁוּם בּוֹעֵל נִדָּה לְטֻמְאַת שִׁבְעָה, אֶלָּא מִשּׁוּם נוֹגֵעַ וּלְטֻמְאַת עֶרֶב: תַּחְתּוֹן. שֶׁל בּוֹעֵל נִדָּה: כְּעֶלְיוֹנוֹ. שֶׁל זָב, שֶׁמְּטַמֵּא אֳכָלִין וּמַשְׁקִין וְלֹא אָדָם וְכֵלִים: מִן הַפְּסוּלִין. כְּגוֹן נָכְרִי, וְעֶבֶד, וְנָתִין, מַמְזֵר, אוֹ חָלָל: פְּסָלָהּ. מִתְּרוּמַת אָבִיהָ , אִם הִיא בַּת כֹּהֵן: מִכָּל הָעֲרָיוֹת. כְּגוֹן אָבִיהָ אוֹ חָמִיהָ : וְהִיא פְטוּרָה. מִפְּנֵי שֶׁהִיא קְטַנָּה וְאֵינָהּ בַּת עֳנָשִׁים: כְּנוֹתֵן אֶצְבַּע בָּעָיִן. שֶׁדּוֹמֵעַ וְחוֹזֵר לִכְמוֹת שֶׁהָיָה, אַף כָּאן חוֹזֶרֶת בְּתוּלָה כְּמוֹ שֶׁהָיְתָה :

שֶׁבָּא עַל יְבִמְתּוֹ קְנָאָהּ. וְזָכָה בְּנִכְסֵי אָחִיו . וְאַף עַל פִּי שֶׁאֵין קִנְיָן לְקָטָן, שָׁאנֵי הָכָא דְּמִן שְׁמַיָּא אַקְנוֹיַהּ נִיהָלֵיהּ: וְאֵינוֹ נוֹתֵן גֵּט עַד שֶׁיַּגְדִּיל. אִם בָּא לְגָרְשָׁהּ. דְּקִדּוּשֵׁי אָחִיו קִדּוּשִׁין גְּמוּרִים, וְגֵרוּשִׁין שֶׁל זֶה אֵינָן גֵּרוּשִׁין גְּמוּרִין. וְדַוְקָא כְּשֶׁבָּא עָלֶיהָ לְאַחַר שֶׁהִגְדִּיל, דַּי לָהּ בְּגֵט. אֲבָל אִם לֹא בָּא עָלֶיהָ לְאַחַר שֶׁהִגְדִּיל, צְרִיכָה גֵּט וַחֲלִיצָה: וּמִטַּמֵּא בְנִדָּה. אִם בָּא עָלֶיהָ בְּנִדָּתָהּ. דְּבִיאָתוֹ בִּיאָה לְכָל דָּבָר: וּפוֹסֵל. כְּגוֹן נָכְרִי וְעֶבֶד אוֹ חָלָל בֶּן תֵּשַׁע שָׁנִים שֶׁבָּא עַל הַכֹּהֶנֶת, פְּסָלָהּ מִתְּרוּמַת אָבִיהָ: וְאֵינוֹ מַאֲכִיל. אִם הוּא כֹּהֵן וְנָשָׂא יִשְׂרְאֵלִית, אֵינוֹ מַאֲכִילָהּ בַּתְּרוּמָה, לְפִי שֶׁאֵין קִנְיָנוֹ קִנְיָן גָּמוּר. וַאֲפִלּוּ הוּא יָבָם דְּקַנְיַהּ בְּהַךְ בִּיאָה כְּדַאֲמַרַן, אֲפִלּוּ הָכִי אֵינוֹ מַאֲכִילָהּ בַּתְּרוּמָה: וּפוֹסֵל אֶת הַבְּהֵמָה. אִם רְבָעָהּ עַל פִּי עֵד אֶחָד : וְנִסְקֶלֶת עַל יָדוֹ. אִם יֵשׁ שְׁנֵי עֵדִים:

נְדָרֶיהָ נִבְדָּקִים. אִם יָדְעָה לְשֵׁם מִי נָדְרָה, לְשֵׁם מִי הִקְדִּישָׁה, נָדְרָה נֶדֶר: נְדָרֶיהָ קַיָּמִין. וְהוּא שֶׁהֵבִיאָה שְׁתֵּי שְׂעָרוֹת. וְכֵן בֶּן שְׁלֹשׁ עֶשְׂרֵה שָׁנָה וְיוֹם אֶחָד נְדָרָיו קַיָּמִים, דַּוְקָא כְּשֶׁהֵבִיא שְׁתֵּי שְׂעָרוֹת: וּבוֹדְקִין כָּל שְׁתֵּים עֶשְׂרֵה. מַהוּ דְּתֵימָא הוֹאִיל וְאָמַר מַר שְׁלֹשִׁים יוֹם בַּשָּׁנָה חֲשׁוּבִים שָׁנָה, הֵיכָא דִּבְדָקְנוּהָ שְׁלֹשִׁים יוֹם קַמָּאֵי דִּשְׁתֵּים עֶשְׂרֵה וְלֹא יָדְעָה לְפָרֵשׁ לְשֵׁם מִי נָדְרָה, תּוּ לֹא נִבְדֹּק, וּלְחַזְקָהּ כִּקְטַנָּה, קָמַשְׁמַע לָן: קֹדֶם לַזְּמַן הַזֶּה. קֹדֶם רֹאשׁ הַשָּׁנָה שֶׁל שְׁתֵּים עֶשְׂרֵה לְתִינוֹקֶת וְשֶׁל שְׁלֹשׁ עֶשְׂרֵה לְתִינוֹק: אֵין נִדְרֵיהֶן נֶדֶר. דִּקְטַנִּים הֵן:

פַּגָּהּ. סְמָדַר. כְּמוֹ הַתְּאֵנָה חָנְטָה פַגֶּיהָ (שִׁיר הַשִּׁירִים ב). כָּךְ הַתִּינוֹקֶת שֶׁאֵין לָהּ סִימָן לֹא בַּדַּדִּין וְלֹא בַּשְּׂעָרוֹת: בֹּחַל. הַפְּרִי הַקָּרוֹב לְהִתְבַּשֵּׁל. וּלְשׁוֹן חֲכָמִים הוּא, הַתְּאֵנִים מִשֶּׁיַּבְחִילוּ חַיָּבִים בְּמַעֲשֵׂר. כָּךְ כְּשֶׁיֵּשׁ לָהּ סִימַן דַּדִּין קְצָת, בְּיָדוּעַ שֶׁהֵבִיאָה שְׁתֵּי שְׂעָרוֹת וְנַעֲרָה הִיא: צֶמֶל. לְשׁוֹן נוֹטָרִיקוֹן יָצְתָה מְלֵאָה. שֶׁיֶּשׁ לָהּ סִימָן מֻבְהָק בְּדַדֶּיהָ וְהִיא בּוֹגֶרֶת: בָּזוֹ וּבָזוֹ. אֲפִלּוּ כְּשֶׁהִיא נַעֲרָה, אָבִיהָ זַכַּאי בַּכֹּל. דִּכְתִיב (בְּמִדְבַּר ל) בִּנְעֻרֶיהָ בֵּית אָבִיהָ, כָּל שְׁבַח נְעוּרֶיהָ לְאָבִיהָ:

אֵיזוֹ הִיא סִימָנֶיהָ. אַצֶּמֶל קָאֵי: קֶמֶט. שֶׁגָּדְלוּ דַּדֶּיהָ עַד שֶׁכְּפוּלִים מְעַט עַל הֶחָזֶה וְיֵרָאֶה כְּקֶמֶט: מִשֶּׁיַּטּוּ הַדַּדִּין. גְּדוֹלִים יוֹתֵר מִקֶּמֶט: הַפִּטֹּמֶת. רֹאשׁ הַדַּד. כְּמוֹ נִטְּלָה פִּטְמָתוֹ : עַל הָעֹקֶץ. חֻדּוֹ שֶׁל דַּד שֶׁתִּינוֹק יוֹנֵק. וּפִטֹּמֶת הַיְנוּ סְבִיבוֹת הָעֹקֶץ. וְכָל הַסִּימָנִים הַלָּלוּ כְּשֶׁנִּרְאוּ בְּאִשָּׁה הֲרֵי הִיא בּוֹגֶרֶת, לְמַר כִּדְאִית לֵיהּ וּלְמַר כִּדְאִית לֵיהּ, וְשׁוּב אֵין לְאָבִיהָ רְשׁוּת בָּהּ. וַהֲלָכָה כְּדִבְרֵי כֻּלָּם לְהַחְמִיר. הִלְכָּךְ אִם נִרְאָה בָּהּ אֶחָד מִן הַסִּימָנִים הַלָּלוּ, דָּיְנִינַן בָּהּ דִּין בּוֹגֶרֶת לְחֻמְרָא. אֲבָל לֹא לְקֻלָּא , שֶׁאִם קִדְּשָׁה אֶת עַצְמָהּ, צְרִיכָה גֵּט שֶׁמָּא בּוֹגֶרֶת הִיא. וְאִם אָבִיהָ קִדְּשָׁהּ, צְרִיכָה נַמִּי גֵּט, דְּשֶׁמָּא עֲדַיִן נַעֲרָה הִיא, עַד שֶׁיִּהְיוּ בָּהּ כָּל הַסִּימָנִים הַלָּלוּ כֻּלָּם, וְאָז תֵּחָשֵׁב בּוֹגֶרֶת בֵּין לְקֻלָּא בֵּין לְחֻמְרָא:

וְהִיא אַיְלוֹנִית. כְּלוֹמַר וְצָרִיךְ שֶׁיֵּרָאוּ בָּהּ נַמִּי סִימָנֵי אַיְלוֹנִית. שֶׁאִם לֹא נִרְאוּ בָּהּ סִימָנֵי אַיְלוֹנִית, אַף עַל פִּי שֶׁהִיא בַּת עֶשְׂרִים וְלֹא הֵבִיאָה שְׁתֵּי שְׂעָרוֹת, עֲדַיִן קְטַנָּה הִיא, עַד שֶׁתִּהְיֶה בַּת שְׁלֹשִׁים וְחָמֵשׁ שָׁנָה וְיוֹם אֶחָד. וְאִם לֹא הֵבִיאָה שְׁתֵּי שְׂעָרוֹת אָז, אַף עַל פִּי שֶׁלֹּא נִרְאוּ בָּהּ סִימָנֵי אַיְלוֹנִית, הֲרֵי זוֹ אַיְלוֹנִית. וְסִימָנֵי אַיְלוֹנִית פֵּרַשְׁנוּ פֶּרֶק קַמָּא דִּיבָמוֹת: לֹא חוֹלֶצֶת וְלֹא מִתְיַבֶּמֶת. וּמֻתֶּרֶת לִנָּשֵׂא לַשּׁוּק. דְּרַחֲמָנָא פַּטְרַהּ, דִּכְתִיב (דְּבָרִים כה) וְהָיָה הַבְּכוֹר אֲשֶׁר תֵּלֵד, פְּרָט לְאַיְלוֹנִית, שֶׁאֵינָהּ יוֹלֶדֶת: וְהוּא סָרִיס. כְּלוֹמַר שֶׁנִּרְאוּ בּוֹ סִימָנֵי סָרִיס. דְּאִי לֹא נִרְאוּ בּוֹ סִימָנֵי סָרִיס, עֲדַיִן הוּא קָטָן עַד שֶׁיִּהְיֶה בֶּן שְׁלֹשִׁים וְחָמֵשׁ שָׁנָה וְיוֹם אֶחָד, דְּאָז אֲפִלּוּ לֹא נִרְאוּ בּוֹ סִימָנֵי סָרִיס, בְּיָדוּעַ שֶׁהוּא סָרִיס, כֵּיוָן שֶׁלֹּא הֵבִיא שְׁתֵּי שְׂעָרוֹת. וְסָרִיס לָאו בַּר חֲלִיצָה הוּא, דִּכְתִיב (שָׁם) לְהָקִים לְאָחִיו שֵׁם בְּיִשְׂרָאֵל, וְהַאי לָאו בַּר הָכִי הוּא: רַבִּי אֱלִיעֶזֶר אוֹמֵר הַזָּכָר כְּדִבְרֵי בֵית הִלֵּל. בֶּן עֶשְׂרִים, אִם נִרְאוּ בּוֹ סִימָנֵי סָרִיס, הֲרֵי זֶה סָרִיס: וְהַנְּקֵבָה. בַּת שְׁמֹנֶה עֶשְׂרֵה, אִם נִרְאוּ בָּהּ סִימָנֵי אַיְלוֹנִית, הֲרֵי זוֹ אַיְלוֹנִית כְּדִבְרֵי בֵּית שַׁמַּאי. וְאֵין הֲלָכָה כְּרַבִּי אֱלִיעֶזֶר:
