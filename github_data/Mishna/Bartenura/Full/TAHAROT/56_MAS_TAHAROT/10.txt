פרק י

הַנּוֹעֵל בֵּית הַבַּד מִפְּנֵי הַבַּדָּדִין. עַמֵּי הָאָרֶץ הָעוֹשִׂים מְלָאכָה בְּבֵית הַבַּד, וְטִהֲרָן בַּעַל הַזֵּיתִים כְּדֵי שֶׁיַּעֲשׂוּ הַשֶּׁמֶן בְּטָהֳרָה, וְנָעַל בֵּית הַבַּד בִּפְנֵיהֶם שֶׁלֹּא יִכָּנְסוּ עַמֵּי הָאָרֶץ לְשָׁם וְשֶׁלֹּא יֵצְאוּ הֵם וְיִטַּמְּאוּ בְּמַגַּע עַמֵּי הָאָרֶץ חַבְרֵיהֶם: וְהָיוּ שָׁם. בְּתוֹךְ בֵּית הַבַּד: כֵּלִים טְמֵאִין מִדְרָס. שֶׁהֵן אַב הַטֻּמְאָה וּמְטַמְּאִין אָדָם בֵּין בְּמַגָּע בֵּין בְּמַשָּׂא בֵּין בְּהֶסֵּט: רַבִּי מֵאִיר אוֹמֵר בֵּית הַבַּד טָמֵא. דְּחָיְשִׁינַן שֶׁמָּא נִטְמְאוּ בָּהֶן אוֹתָן עַמֵּי הָאָרֶץ הַטְּהוֹרִים וְחָזְרוּ וְטִמְּאוּ אֶת הַכֹּל, דְּאֵין חוֹשְׁשִׁים לִזָּהֵר עַל הַטֻּמְאָה: רַבִּי יְהוּדָה אוֹמֵר בֵּית הַבַּד טָהוֹר. דְּנִזְהָרִין עַל טֻמְאַת וַדַּאי, הוֹאִיל וְחָבֵר זֶה טִהֲרָן לְכָךְ: רַבִּי שִׁמְעוֹן אוֹמֵר אִם טְהוֹרִים לָהֶם. אֵלּוּ הַכֵּלִים. כְּגוֹן שֶׁיָּשְׁבָה עֲלֵיהֶם אֵשֶׁת עַם הָאָרֶץ אוֹ שֶׁהָיוּ שֶׁל עַם הָאָרֶץ, דִּבְעֵינֵיהֶן טְהוֹרִין הֵן: בֵּית הַבַּד טָמֵא. שֶׁאֵין הַבַּדָּדִים נִזְהָרִים מִלִּגַּע בָּהֶן, וְאִיכָּא לְמֵיחַשׁ שֶׁמָּא נִטְמְאוּ: וְאִם טְמֵאִין הֵן. כְּגוֹן שֶׁיָּדְעוּ שֶׁיָּשְׁבָה עֲלֵיהֶן אִשָּׁה נִדָּה אוֹ זָבָה: בֵּית הַבַּד טָהוֹר. דְּנִזְהָרִין הֵן: אָמַר רַבִּי יוֹסֵי. כְּרַבִּי מֵאִיר סְבִירָא לֵיהּ דְּבֵית הַבַּד טָמֵא, אֶלָּא טַעֲמָא דְּרַבִּי מֵאִיר מִשּׁוּם דְּאֵין הַבַּדָּדִין חוֹשְׁשִׁים בַּטֻּמְאָה, וְרַבִּי יוֹסֵי סָבַר דְּחוֹשְׁשִׁין בְּמַגָּע אֲבָל אֵין בְּקִיאִין בְּהֶסֵּט, וְחוֹשְׁבִים שֶׁהֵן טְהוֹרִין כָּל זְמַן שֶׁלֹּא נָגְעוּ, הִלְכָּךְ בֵּית הַבַּד טָמֵא, דְּחָיְשִׁינַן שֶׁמָּא הִסִּיטוּ אַף עַל פִּי שֶׁלֹּא נָגְעוּ בָּהֶם . וַהֲלָכָה כְּרַבִּי יוֹסֵי:

וּמַשְׁקִין טְמֵאִין בְּתוֹךְ בֵּית הַבַּד. וְהֵן נִשְׁפָּכִים עַל גַּבֵּי קַרְקַע וְדוֹרְכִים הַבַּדָּדִים עֲלֵיהֶן וְנִדְבָּקִים בְּרַגְלֵיהֶן, וְחוֹזְרִים אוֹתָן מַשְׁקִין וּמְטַמְּאִים אֶת הַזֵּיתִים כְּשֶׁהַבַּדָּדִין עוֹלִים וְדוֹרְכִים עַל גַּבֵּי הַזֵּיתִים, אֶלָּא אִם כֵּן יֵשׁ בֵּין הַמַּשְׁקִין לַזֵּיתִים אָרֶץ כְּדֵי שֶׁיְּנַגְּבוּ רַגְלֵיהֶן מִן הַמַּשְׁקִין הַטְּמֵאִים עַל יְדֵי דְּרִיסָתָם בָּאָרֶץ. וּכְשֶׁאֵין מִנְעָלִים בְּרַגְלֵיהֶן אַיְרֵי, דְּהָא מַשְׁקֶה מְטַמֵּא כְּלִי וְחוֹזֵר וּמְטַמֵּא מַשְׁקִין שֶׁעִם הַזֵּיתִים. אֲבָל כְּשֶׁאֵין מִנְעָלִים בְּרַגְלֵיהֶן וְנִתְנַגְּבוּ הַמַּשְׁקִין הַטְּמֵאִין שֶׁבְּרַגְלֵיהֶן עַל יְדֵי דְּרִיסָתָן בָּאָרֶץ, שׁוּב אֵין מִטַּמְּאִים הַזֵּיתִים כְּשֶׁדּוֹרְכִין אוֹתָן, דְּמַשְׁקִין טְמֵאִין מְטַמְּאִין כֵּלִים וְאֵין מְטַמְּאִין אָדָם: הַבַּדָּדִין וְהַבּוֹצְרִין. הָא דְּלֹא נָקַט הַמּוֹסְקִין וְהַבּוֹצְרִין, לְפִי שֶׁגָּזְרוּ עַל הַבְּצִירָה שֶׁתֵּעָשֶׂה בְּטָהֳרָה, וְלֹא גָּזְרוּ עַל הַמְּסִיקָה. וְטַעֲמָא מְפֹרָשׁ בְּפֶרֶק קַמָּא דְּשַׁבָּת, שֶׁפְּעָמִים אָדָם הוֹלֵךְ לְכַרְמוֹ לֵידַע אִם הִגִּיעוּ עֲנָבָיו לִבְצֹר אִם לָאו, וְנוֹטֵל אֶשְׁכּוֹל שֶׁל עֲנָבִים וְסוֹחֲטוֹ וּמְזַלֵּף עַל גַּבֵּי עֲנָבִים וּבִשְׁעַת בְּצִירָה עֲדַיִן מַשְׁקֶה טוֹפֵחַ עֲלֵיהֶן וְהֻכְשְׁרוּ לְקַבֵּל טֻמְאָה. וְטַעַם זֶה לֹא שַׁיָּךְ בִּמְסִיקָה: נֶאֱמָנִין לוֹמַר לֹא נָגָעְנוּ. וּכְגוֹן דְּחָבֵר עוֹמֵד עִמָּהֶן בְּבֵית הַבַּד. וְלֹא חָיְשִׁינַן דִּלְמָא נָגְעוּ וְלָאו אַדַּעְתֵּיהּ דְּחָבֵר: הַתִּינוֹקוֹת. בַּדָּדִים שֶׁהֵבִיאוּ בְּנֵיהֶן עִמָּהֶן וְטִהֲרוּם לְכָךְ, וּכְשֶׁרוֹצִין בְּנֵיהֶם לִפָּנוֹת יוֹצְאִים אֲבִיהֶן לַחוּץ עִמָּהֶם וְנִפְנִים אֲחוֹרֵי הַגָּדֵר, וְלֹא חָיְשִׁינַן דִּלְמָא נָגַע בְּטֻמְאָה וְלָאו אַדַּעְתָּא דְּחָבֵר:

כֵּיוָן שֶׁהִכְנִיסָן לִרְשׁוּת הַמְּעָרָה. שֶׁטּוֹבְלִים בָּהּ. כְּמוֹ בּוֹרוֹת שִׁיחִין וּמְעָרוֹת: דַּיּוֹ. וְלֹא חָיְשִׁינַן שֶׁמָּא לֹא יִטְבֹּלוּ אוֹ לֹא יַטְבִּילוּ הַכֵּלִים וְיֹאמְרוּ טָבַלְנוּ וְהִטְבַּלְנוּ: צָרִיךְ לַעֲמֹד עֲלֵיהֶן. לְפִי שֶׁאֵין בְּקִיאִין בְּדִין טְבִילָה וַחֲצִיצָה: טְהוֹרִים לָהֶם. כְּגוֹן כֵּלִים שֶׁל עַם הָאָרֶץ אַחֵר, שֶׁבְּעֵינֵיהֶם הֵן כֵּלִים טְהוֹרִים לְפִי שֶׁאֵין נִזְהָרִים מִמַּגַּע עַם הָאָרֶץ. וְרַבִּי שִׁמְעוֹן לְטַעֲמֵיהּ כִּדְפָרֵישְׁנָא לְעֵיל. וַהֲלָכָה כְּרַבִּי יוֹסֵי:

הַנּוֹתֵן מִן הַסַּלִּים וּמִן הַמַּשְׁטֵחַ שֶׁל אֲדָמָה. עֲנָבִים שֶׁבְּצָרָן מִתְּחִלָּה לַאֲכִילָה, וְלֹא הֻכְשְׁרוּ לְקַבֵּל טֻמְאָה, דְּלֹא גָּזְרוּ אֶלָּא עַל הַבּוֹצֵר מִתְּחִלָּה לַגַּת וְאַחַר כָּךְ נִמְלַךְ עֲלֵיהֶן וּלְקָחָן מִן הַסַּלִּים אוֹ מִן הָאֲדָמָה שֶׁשָּׁטַח הָעֲנָבִים עָלֶיהָ וּנְתָנָן לַגַּת: בֵּית שַׁמַּאי אוֹמְרִים נוֹתֵן בְּיָדַיִם טְהוֹרוֹת. כֵּיוָן שֶׁבָּא לְנָתְנָן לַגַּת הֲרֵי הוּא כְּאִלּוּ בְּצָרָן מִתְּחִלָּה לַגַּת, וְצָרִיךְ שֶׁיִּתֵּן לַגַּת בְּטָהֳרָה: וְאִם נָתַן בְּיָדַיִם טְמֵאוֹת טִמְּאָן. וְאַף עַל גַּב דִּסְתָם יָדַיִם שְׁנִיּוֹת הֵן, וְאֵין שֵׁנִי עוֹשֶׂה שְׁלִישִׁי בְּחֻלִּין, דִּלְמָא קָסָבְרֵי הַטְּבוּלִים לִתְרוּמָה כִּתְרוּמָה דָּמוּ. אִי נַמִּי, בְּיָדַיִם תְּחִלּוֹת עָסְקִינַן, וְרַבִּי שִׁמְעוֹן בֶּן אֶלְעָזָר הִיא כִּדְאִיתָא פֶּרֶק ב' דְּחֻלִּין: וּבֵית הִלֵּל אוֹמְרִים נוֹתֵן לַגַּת בְּיָדַיִם טְמֵאוֹת וּמַפְרִישׁ תְּרוּמָתוֹ בְּטָהֳרָה. לְפִי שֶׁלֹּא הֻכְשְׁרוּ כֵּיוָן דִּבְצָרָן מִתְּחִלָּה לַאֲכִילָה: מִן הָעֲבַט וּמִן הַמַּשְׁטֵחַ שֶׁל עָלִים. עָבִיט הוּא כְּלִי שֶׁנּוֹתְנִים בּוֹ הָעֲנָבִים כְּשֶׁבּוֹצְרִים אוֹתָן לַגַּת. וּמַשְׁטֵחַ שֶׁל עָלִים, שֶׁשָּׁטְחוּ הֶעָלִים עַל פְּנֵי הָאֲדָמָה וְנָתְנוּ הָעֲנָבִים עֲלֵיהֶם. וְהַשְׁתָּא מוֹכְחָא מִלְּתָא דְּהַבּוֹצֵר עֲנָבִים הַלָּלוּ מִתְּחִלָּה, לַגַּת בְּצָרָן וְלֹא לַאֲכִילָה, הִלְכָּךְ שָׁוִים בֵּית שַׁמַּאי וּבֵית הִלֵּל בְּהָא דְּנוֹתֵן בְּיָדַיִם טְהוֹרוֹת, דְּהָא שָׁתִיק לֵיהּ הִלֵּל לְשַׁמַּאי בְּפֶרֶק קַמָּא דְּשַׁבָּת וְהוֹדָה לוֹ שֶׁהַבּוֹצֵר לַגַּת הֻכְשַׁר:

הָאוֹכֵל. עֲנָבִים: מִן הַסַּלִּים וּמִן הַמַּשְׁטֵחַ שֶׁל אֲדָמָה. דִּסְתָמָן לַאֲכִילָה נִבְצְרוּ. אַף עַל פִּי שֶׁיָּדָיו טְמֵאוֹת וְהָעֲנָבִים נוֹטְפִים יַיִן לַגַּת: הֲרֵי הַגַּת טְהוֹרָה. דְּאוֹתָן טִפִּין אֵין לָהֶם תּוֹרַת מַשְׁקֶה, הוֹאִיל וְלַאֲכִילָה נִבְצְרוּ וְלֹא נִיחָא לֵיהּ בְּמַאי דְּטָיֵף. אֲבָל הָאוֹכֵל מִן הֶעָבִיט וּמִן הַמַּשְׁטֵחַ שֶׁל עָלִים, אַף עַל פִּי שֶׁאֵין הָעֲנָבִים מְבֻקָּעוֹת וְאֵינָן מְנַטְּפוֹת, אִם נָפַל מִמֶּנּוּ גַּרְגֵּר יְחִידִי וְאֵין חוֹתָם לְאוֹתוֹ גַּרְגֵּר, דְּהַיְנוּ בַּמָּקוֹם שֶׁהָעֹקֶץ עַל פִּי הַגַּרְגֵּר יֵשׁ כְּמִין חוֹתָם וְאוֹתוֹ חוֹתָם סוֹתֵם פִּי הַגַּרְגֵּר שֶׁאֵין מַשְׁקֶה יוֹצֵא. וּכְשֶׁנֶּעֱקָר הָעֹקֶץ עִם הַחוֹתָם, מַשְׁקֶה יוֹצֵא, וְאַף עַל פִּי שֶׁנְּטָלָם לְאָכְלָם יֵשׁ לוֹ תּוֹרַת מַשְׁקֶה, הוֹאִיל וּמִתְּחִלָּה בְּצָרָם לַגַּת, וְנִטְמָא אוֹתוֹ מַשְׁקֶה בְּיָדָיו טְמֵאוֹת, וּכְשֶׁנָּפַל לַגַּת נִטְמָא. וְהָנֵי מִלֵּי כְּשֶׁנָּגַע בְּיָדָיו בִּמְקוֹם הַחוֹתָם. אֲבָל אִם לֹא נָגַע בִּמְקוֹם הַחוֹתָם, לֹא נִטְמָא הַגַּת, שֶׁהֲרֵי לֹא נָגְעוּ יָדָיו בַּמַּשְׁקִים כְּדֵי שֶׁיַּחֲזֹר הַמַּשְׁקֶה וִיטַמֵּא הַגַּת: נָפְלוּ מִמֶּנּוּ עֲנָבִים. כְּלוֹמַר אֶשְׁכּוֹל שֶׁל עֲנָבִים, שֶׁהַגַּרְגְּרִים מְחֻבָּרִים בָּאֶשְׁכּוֹל וְאֵין מַשְׁקֶה יוֹצֵא, וְנִטְמְאוּ בְּיָדָיו טְמֵאוֹת, דְּהֻכְשְׁרוּ לְפִי שֶׁנִּבְצַר לַגַּת: וּדְרָכָן בְּמָקוֹם הַמֻפְנֶה. מָקוֹם פָּנוּי בַּגַּת שֶׁאֵין שָׁם כְּלוּם לֹא יַיִן וְלֹא עֲנָבִים וּדְרָכוּהוּ בְּטָהֳרָה: כַּבֵּיצָה מְכֻוָּן טָהוֹר. הַגַּת, דְּאֵין אֹכֶל מְטַמֵּא כְּלִי, וּמִשֶׁיָּצָאת טִפָּה רִאשׁוֹנָה חָסֵר לֵיהּ מִכַּבֵּיצָה וְאֵין אֹכֶל מְטַמֵּא אֲחֵרִים בְּפָחוֹת מִכַּבֵּיצָה, נִמְצָא דְּאֵין טֻמְאָה לְטִפָּה זוֹ אֶלָּא מֵחֲמַת טֻמְאָה שֶׁקִּבְּלָהּ כְּשֶׁהָיְתָה אֹכֶל, הִלְכָּךְ הַשְׁתָּא נַמִּי דְּנַעֲשֵׂית מַשְׁקֶה אֵין לָהּ כֹּחַ אֶלָּא כְּאֹכֶל. אֲבָל יוֹתֵר מִכַּבֵּיצָה, טָמֵא, שֶׁהַבֵּיצָה מְטַמְּאָה טִפָּה רִאשׁוֹנָה כְּדִין מַשְׁקֶה, וְחָזְרָה הַטִּפָּה וְטִמְּאָה אֶת הַגַּת, דְּמַשְׁקֶה מְטַמֵּא כְּלִי:

וְנִתְּזָה צִנּוֹרָא. נִתַּז רֹק מִפִּיו שֶׁל עַם הָאָרֶץ הַמְדַבֵּר, וְרֹק הַיּוֹצֵא מִפִּי עַם הָאָרֶץ מְטַמֵּא אָדָם וְכֵלִים מִדְּרַבָּנָן, וְכָל שֶׁכֵּן אֳכָלִים וּמַשְׁקִים: סְפֵקוֹ טָהוֹר. דְּאָמְרִינַן שְׂפַת הַבּוֹר קָלְטָה הַצִּנּוֹרָא וּמְנָעַתָּה מִלָּרֶדֶת לְתוֹךְ הַיַּיִן. וּמַתְנִיתִין אַיְרֵי בְּבוֹר סִיד שֶׁאֵינוֹ מְקַבֵּל טֻמְאָה. וּבְתוֹסֶפְתָּא אָמְרוּ, דִּבְבוֹר שֶׁל שֶׁמֶן טָמֵא , לְפִי שֶׁהַשֶּׁמֶן מַחֲלִיק שְׂפַת הַבּוֹר מַחְלֶקֶת וְיוֹרֶדֶת לְמַטָּה לַשֶּׁמֶן שֶׁבַּבּוֹר:

הַזּוֹלֵף אֶת הַבּוֹר. הַמּוֹצִיא יַיִן וְשֶׁמֶן שֶׁבַּבּוֹר. וּמִפְּנֵי שֶׁאֵינוֹ מוֹצִיאוֹ כֻּלּוֹ בְּבַת אַחַת אֶלָּא מְעַט מְעַט, קָרֵי לֵיהּ זוֹלֵף: וְנִמְצָא שֶׁרֶץ בָּרִאשׁוֹנָה. שֶׁהִכְנִיס חָבִית בַּבּוֹר וּמִלְּאָהּ, וְכֵן שְׁנִיָּה, וְכֵן שְׁלִישִׁית, וְאַחַר כָּךְ נִמְצָא שֶׁרֶץ בֶּחָבִית שֶׁמִּלֵּא רִאשׁוֹנָה: כֻּלָּן טְמֵאוֹת. דְּמֵהָרִאשׁוֹנָה נִטְמָא כָּל מַה שֶּׁבַּבּוֹר, בֵּין שֶׁהָיָה שֶׁרֶץ בַּבּוֹר בֵּין שֶׁהָיָה בֶּחָבִית: בָּאַחֲרוֹנָה הִיא טְמֵאָה וְכֻלָּן טְהוֹרוֹת. דְּמִמֶּנָּה הָיָה הַשֶּׁרֶץ וְלֹא מִן הַבּוֹר. וַאֲפִלּוּ הָיָה מִן הַבּוֹר, שֶׁמָּא נָפַל בַּבּוֹר אַחַר שֶׁנִּתְמַלְּאוּ כָּל הָרִאשׁוֹנוֹת: אֵימָתַי בִּזְמַן שֶׁהוּא זוֹלֵף. כְּשֶׁהוּא מַכְנִיס הֶחָבִיּוֹת עַצְמָן בְּתוֹךְ הַבּוֹר וּמְמַלֵּא אוֹתָן אַחַת אַחַת: אֲבָל הָיָה זוֹלֵף בְּמַחַץ. שֶׁהֶחָבִיּוֹת עוֹמְדוֹת חוּץ לַבּוֹר, וְהוּא מְמַלֵּא אוֹתָן עַל יְדֵי הַמַּחַץ, וְהוּא כְּלִי חֶרֶס הֶעָשׂוּי לְמַלְּאוֹת בּוֹ יַיִן: נִמְצָא שֶׁרֶץ בְּאַחַת מֵהֶן הִיא טְמֵאָה בִּלְבָד. דְּאֵמוּר בְּתוֹךְ הֶחָבִית הָיָה הַשֶּׁרֶץ וְלֹא בַּבּוֹר: אֵימָתַי בִּזְמַן שֶׁהוּא בוֹדֵק. הַמַּחַץ וְהֶחָבִיּוֹת. שֶׁבְּדָקָן שֶׁלֹּא הָיָה בָּהֶן שֶׁרֶץ קֹדֶם שֶׁנָּתַן לְתוֹכוֹ שֶׁמֶן, אֲבָל לֹא הָיָה מְכַסֶּה לֹא הַבּוֹר וְלֹא הֶחָבִיּוֹת כָּל שָׁעָה שֶׁהָיָה מֵבִיא הַמַּחַץ מִזֶּה לָזֶה, דְּאֵמוּר בַּהֲלִיכָתוֹ נָפַל הַשֶּׁרֶץ בֶּחָבִית. וְכֵן מְכַסֶּה וְלֹא בּוֹדֵק נַמִּי שֶׁמָּא הָיָה בֶּחָבִית קֹדֶם לָכֵן: אֲבָל הָיָה בוֹדֵק וּמְכַסֶּה. שֶׁבָּדַק הֶחָבִיּוֹת, וּבְכָל פַּעַם שֶׁהָיָה מוֹלִיךְ הַמַּחַץ מִזֶּה לָזֶה הָיָה מְכַסֶּה בֵּין בּוֹר בֵּין חָבִית, מִמַּה נַּפְשָׁךְ שֶׁרֶץ הָיָה בַּבּוֹר מִתְּחִלָּתוֹ, בֵּין שֶׁנִּמְצָא בֶּחָבִית, בֵּין שֶׁנִּמְצָא בַּבּוֹר, בֵּין שֶׁנִּמְצָא בַּמַּחַץ:

הָעִגּוּלִים לַזַּגִּים. עִגּוּל כְּמִין רֵיחַיִם גָּדוֹל הָיוּ מְשִׂימִין עַל הַזָּגִים לְאַחַר שֶׁדָּרְכוּ הָעֲנָבִים בָּרֶגֶל, כְּדֵי שֶׁיִּתְמַצּוּ וְיֵצֵא כָּל הַיַּיִן הַנִּשְׁאָר בָּהֶן, וּמִתּוֹךְ כָּבְדוֹ וְגָדְלוֹ שֶׁל עִגּוּל זֶה צְרִיכִים בְּנֵי אָדָם הַרְבֵּה לְהַעֲלוֹתוֹ עַל הַצִּבּוּר שֶׁל הַזָּגִין הֶעָשׂוּי כְּמִין תַּפּוּחַ וּלְהוֹרִידוֹ, לְכָךְ נֶחֱשָׁב רְשׁוּת הָרַבִּים לְטַהֵר סְפֵק טֻמְאָה, מִשּׁוּם דְּרַבִּים מְצוּיִין שָׁם: כֶּרֶם שֶׁלִּפְנֵי הַבּוֹצְרִים רְשׁוּת הַיָּחִיד. שֶׁיְּרֵאִין בְּנֵי אָדָם לִכָּנֵס שָׁם, מִפְּנֵי שֶׁנִּתְפָּס עָלָיו כְּגַנָּב: אַחַר הַבּוֹצְרִים רְשׁוּת הָרַבִּים. שֶׁיֵּשׁ רְשׁוּת לְכָל אָדָם לִכָּנֵס: כְּלֵי בֵית הַבַּד שֶׁל גַּת. כְּלוֹמַר וְשֶׁל גַּת. דְּבֵית הַבַּד לְשֶׁמֶן, וְגַת לְיַיִן: הָעֵקֶל. עָשׂוּי כְּמִין קֻפָּה מְסֹרֶגֶת כְּעֵין רֶשֶׁת, וְאַחַר שֶׁסָּחֲטוּ הַזֵּיתִים נוֹתְנִים הַגֶּפֶת בְּאוֹתוֹ עֵקֶל, וּמַנִּיחִים עָלָיו דָּבָר מַכְבִּיד לְהוֹצִיא הַנִּשְׁאָר: מְנַגְּבָן. וְקֹדֶם לָכֵן לֹא מְהַנְיָא טְבִילָה אִם נִטְמְאוּ, עַד שֶׁיֵּצֵא מֵהֶן הַמַּשְׁקֶה הַטָּמֵא שֶׁנִּדְבַּק בָּהֶן: חוֹלְטָן. שׁוֹפֵךְ עֲלֵיהֶן מַיִם רוֹתְחִים וְאַחַר כָּךְ מַטְבִּילָן: בְּשִׁבֹּלֶת הַנָּהָר. מָקוֹם שֶׁהַמַּיִם רוֹדְפִין , כְּדֵי לְהוֹצִיא הַמַּשְׁקֶה הַנִּדְבַּק בָּהֶן. וְאֵין הֲלָכָה כְּרַבִּי יוֹסֵי:
