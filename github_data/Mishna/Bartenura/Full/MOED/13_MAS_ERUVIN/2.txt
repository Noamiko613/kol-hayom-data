פרק ב

עוֹשִׂין פַּסִּין לַבֵּירָאוֹת. שֶׁבִּרְשׁוּת הָרַבִּים. לְפִי שֶׁהַבֵּירָאוֹת רְשׁוּת הַיָּחִיד הֵן שֶׁעֲמֻקִּים עֲשָׂרָה, וְאֵין יְכוֹלִים לְמַלְּאוֹת מֵהֶן וּלְהוֹצִיא לִרְשׁוּת הָרַבִּים, וְעוֹשִׂין פַּסִּין שֶׁיִּהְיוּ הַפַּסִּין עוֹשִׂין סְבִיבוֹת הַבּוֹר רְשׁוּת הַיָּחִיד, וִימַלֵּא וְיוֹצִיא וְיָנִיחַ שָׁם וְתִכָּנֵס בְּהֶמְתּוֹ כֻּלָּהּ אוֹ רֹאשָׁהּ וְרֻבָּהּ וְיַשְׁקֶנָּה: דְּיוּמְדִין. שְׁנֵי עַמּוּדִין. שֶׁכָּל עַמּוּד מֵאֵלּוּ הָאַרְבָּעָה נִרְאֶה כִּשְׁנַיִם. דְּיוּ בִּלְשׁוֹן יָוָן שְׁנַיִם. וּכְשֶׁנּוֹעֵץ הָעַמּוּד הֶעָשׂוּי כִּשְׁנַיִם לִפְאַת דְּרוֹמִית מַעֲרָבִית נִמְצָא נוֹטֶה דָּפְנוֹ אֶחָד לַמִּזְרָח וְדָפְנוֹ אֶחָד לַצָּפוֹן, וְהַשֵּׁנִי לִפְאַת מַעֲרָבִית צְפוֹנִית נוֹטֶה צִדּוֹ אֶחָד לַמִּזְרָח וְצִדּוֹ אֶחָד לַדָּרוֹם, וּכְשֶׁנּוֹתֵן אַרְבַּעְתָּן לְאַרְבַּע הַפֵּאוֹת נִמְצָא לְכָל רוּחַ שְׁתֵּי אַמּוֹת דֹּפֶן אַמָּה כְּנֶגֶד אַמָּה וְהָרֶוַח בֵּינְתַיִם: וְאַרְבָּעָה פְשׁוּטִין. בָּאֶמְצַע נוֹתֵן לְכָל רוּחַ לוּחַ רְחָבָה אַמָּה. וּבִזְמַן שֶׁאֵין בֵּין פַּס לְפַס מֵאַרְבָּעָה דְּיוּמְדִין אֶלָּא עֶשֶׂר אַמּוֹת אוֹ פָּחוֹת מוֹדֶה רַבִּי מֵאִיר דְּאֵין צָרִיךְ לְפַסִּין פְּשׁוּטִין, וְאִם יֵשׁ בֵּינֵיהֶן יוֹתֵר מִשְּׁלֹשׁ עֶשְׂרֵה אַמָּה וּשְׁלִישׁ מוֹדֶה רַבִּי יְהוּדָה דְּצָרִיךְ פְּשׁוּטִין, לֹא נֶחְלְקוּ אֶלָּא מֵעֶשֶׂר עַד שְׁלֹשׁ עֶשְׂרֵה אַמָּה וּשְׁלִישׁ, דְּרַבִּי מֵאִיר אוֹמֵר צָרִיךְ פְּשׁוּטִים וְרַבִּי יְהוּדָה אוֹמֵר אֵין צָרִיךְ. וַהֲלָכָה כְּרַבִּי יְהוּדָה: שְׁתֵּי רְבָקוֹת שֶׁל שָׁלֹשׁ שָׁלֹשׁ בָּקָר. שֶׁרֹחַב כָּל בָּקָר וּבָקָר אַמָּתַיִם פָּחוֹת שְׁלִישׁ, נִמְצָא רֹחַב שִׁשָּׁה בָּקָר עֶשֶׂר אַמּוֹת, וְזֶה הָרֶוַח שֶׁיָּכוֹל לִתֵּן בֵּין פַּס לְפַס לְדִבְרֵי רַבִּי מֵאִיר, וְאִם יֵשׁ רֹחַב יוֹתֵר מִזֶּה צָרִיךְ שֶׁיַּרְבֶּה בְּפַסִּין פְּשׁוּטִין: רְבָקוֹת. לְשׁוֹן עֵגֶל מַרְבֵּק (שְׁמוּאֵל א כח): רַבִּי יְהוּדָה אוֹמֵר שֶׁל אַרְבַּע אַרְבַּע. הֵן שְׁלֹשׁ עֶשְׂרֵה אַמָּה וּשְׁלִישׁ: קְשׁוּרוֹת אֲבָל לֹא מֻתָּרוֹת. דְּלֹא תֵּימָא כְּעֵין קְשׁוּרוֹת וְלֹא קְשׁוּרוֹת מַמָּשׁ, לְהָכִי תָּנָא אֲבָל לֹא מֻתָּרוֹת, לְהַחְמִיר וּלְמַעֵט רֶוַח שֶׁבֵּינְתַיִם: אַחַת נִכְנֶסֶת וְאַחַת יוֹצֵאת. רְבָקָה אַחַת שֶׁל בָּקָר נִכְנֶסֶת וּרְבָקָה אַחַת יוֹצֵאת, וְיֵשׁ יוֹתֵר רֹחַב מִשֶּׁאִלּוּ הָיוּ שְׁתֵּיהֶן יוֹצְאוֹת אוֹ שְׁתֵּיהֶן נִכְנָסוֹת. וְזוֹ קֻלָּא הִיא, דְּלֹא בָּעִינַן מְצֻמְצָמוֹת כָּל כָּךְ :

מֻתָּר לְהַקְרִיב לַבְּאֵר. הַפַּסִּין, וְלַעֲשׂוֹת הֶקֵּף קָצָר. וּבִלְבַד שֶׁיְּהֵא מִשְּׂפַת הַבְּאֵר עַד בֵּין הַפַּסִּין כְּדֵי רֹאשָׁהּ וְרֻבָּהּ שֶׁל פָּרָה שֶׁהֵן שְׁתֵּי אַמּוֹת, אֲבָל בָּצִיר מֵהָכִי לֹא, שֶׁמָּא יִמָּשֵׁךְ אַחַר פָּרָתוֹ וְיוֹצִיא הַדְּלִי חוּץ לַמְּחִצָּה: מֻתָּר לְהַרְחִיק כָּל שֶׁהוּא. לַעֲשׂוֹת הֶקֵּף גָּדוֹל כְּמוֹ שֶׁלִּבּוֹ חָפֵץ: וּבִלְבַד שֶׁיַּרְבֶּה בְּפַסִּין. דְּכָל מַה שֶּׁמַּרְחִיק מִן הַבּוֹר מַרְבֶּה בָּרֶוַח שֶׁבֵּין הַפַּסִּין, וְצָרִיךְ לְהַרְבּוֹת בְּפַסִּין עַד כְּדֵי שֶׁלֹּא יְהֵא בֵּין פָּשׁוּט לְפָשׁוּט יוֹתֵר מִשְּׁלֹשׁ עֶשְׂרֵה אַמָּה וּשְׁלִישׁ, כְּדִבְרֵי רַבִּי יְהוּדָה שֶׁהֲלָכָה כְּמוֹתוֹ:

רַבִּי יְהוּדָה אוֹמֵר עַד בֵּית סָאתָיִם. לֹא יַעֲשֶׂה הֶקֵּף גָּדוֹל לַבְּאֵר אֶלָּא עַד בֵּית סָאתַיִם: לְגִנָּה וּלְקַרְפֵּף. שֶׁאֵין הֶקֵּפָן לְדִירָה: קַרְפֵּף. הֶקֵּף גָּדוֹל חוּץ לָעִיר לְהַכְנִיס שָׁם עֵצִים לְאוֹצָר: דִּיר. שֶׁל בְּהֵמוֹת שֶׁעוֹשִׂין בַּשָּׂדוֹת הַיּוֹם כָּאן וּלְמָחָר כָּאן כְּדֵי לְזַבְּלָהּ בִּגְלָלֵי בְּהֵמָה: סַחַר. לַבְּהֵמוֹת בָּעִיר. וְאִית דְּגָרְסֵי סַהַר, בֵּית הָאֲסוּרִים: חָצֵר. מָקוֹם מְגֻלֶּה שֶׁלִּפְנֵי הַבָּתִּים: מֻקְצֶה. רְחָבָה שֶׁאֲחוֹרֵי הַבָּתִּים , דְּכָל הָנֵי הֶקֵּפָן לְדִירָה הוּא . וְהָנֵי פַּסֵּי בֵּירָאוֹת נַמִּי הוֹאִיל וּמֵימֵיהֶן רְאוּיוֹת לִשְׁתִיַּת אָדָם תַּשְׁמִישׁ מְעַלְּיוּתָא הוּא: וּמֻתָּר לְהַרְחִיק כָּל שֶׁהוּא. כְּלוֹמַר כְּשֵׁם שֶׁכָּל אֵלּוּ שֶׁהֻקְּפוּ לְדִירָה אֲפִלּוּ גְּדוֹלִים עַד עֶשֶׂר כּוֹרִים מֻתָּר לְטַלְטֵל בָּהֶם, כָּךְ מֻתָּר לְהַרְחִיק הַהֶקֵּף שֶׁל פַּסֵּי בֵּירָאוֹת כָּל מַה שֶּׁלִּבּוֹ חָפֵץ, וְכֵן הֲלָכָה:

אִם הָיְתָה דֶרֶךְ הָרַבִּים. דֶּרֶךְ שֶׁרַבִּים בּוֹקְעִים בָּהּ מַפְסֶקֶת בֵּין הַפַּסִּים, יְסַלְקֶנָּה לַצְּדָדִים מִחוּץ לַפַּסִּים, שֶׁלֹּא יִהְיוּ הָרַבִּים מְהַלְּכִים בֵּין הַפַּסִּים, דְּמַשְׁווּ לַהּ כִּרְשׁוּת הָרַבִּים וּמְבַטְּלִים הַמְּחִצָּה. וְאֵין הֲלָכָה כְּרַבִּי יְהוּדָה: בּוֹר הָרַבִּים. אִי פָּסְקֵי מַיָּא מַדְכְּרֵי אַהֲדָדֵי, שֶׁלֹּא הִתִּירוּ פַּסֵּי בֵּירָאוֹת אֶלָּא שֶׁיִּהְיוּ מַיִם מְצוּיִין לִבְהֵמוֹת עוֹלֵי רְגָלִים, וְכָל זְמַן שֶׁאֵין שָׁם מַיִם אֵין לַפַּסִּין תּוֹרַת מְחִצָּה: בְּאֵר הַיָּחִיד. נַמִּי שָׁרֵי, דִּבְאֵר מַיִם חַיִּים הֵן וְהָא לֹא פָּסְקֵי מַיָּא: לִבְאֵר הָרַבִּים. דְּאִיכָּא תַּרְתֵּי לִמְעַלִּיּוּתָא. וְכֵן הֲלָכָה. וְאֵין מֻתָּר לְמַלְאוֹת מַיִם וּלְהוֹצִיא מִן הַבְּאֵר עַל יְדֵי פַּסֵּי בֵּירָאוֹת אֶלָּא כְּדֵי לְהַשְׁקוֹת בְּהֶמְתָּן שֶׁל עוֹלֵי רְגָלִים וּבְאֶרֶץ יִשְׂרָאֵל בִּלְבַד. אֲבָל לְהוֹצִיא מַיִם כְּדֵי לִשְׁתּוֹת הָאָדָם עַצְמוֹ אָסוּר , אֶלָּא יוֹרֵד לַבּוֹר וְשׁוֹתֶה , אוֹ עוֹשֶׂה מְחִצָּה גְּבוֹהָה עֲשָׂרָה טְפָחִים:

וְעוֹד אָמַר רַבִּי יְהוּדָה בֶן בָּבָא. מִשּׁוּם דְּאָמַר חֲדָא לְחֻמְרָא גַּבֵּי פַּסִּין, דְּאָמַר אֵין עוֹשִׂין אוֹתָן אֶלָּא לִבְאֵר הָרַבִּים, וְהַשְׁתָּא קָאָמַר חֻמְרָא אַחֲרִיתָא דְּאַף בְּמֻקָּף לְדִירָה לֹא שָׁרֵי טְפֵי מִבֵּית סָאתַיִם, מִשּׁוּם הָכִי תָּנָא וְעוֹד: שׁוֹמֵירָה. סֻכַּת שׁוֹמְרִים. דְּאַף עַל גַּב דְּהֻקַּף לְדִירָה, בֵּית סָאתַיִם הוּא דְּשָׁרֵי, טְפֵי לֹא: סְמוּכָה לָעִיר. דְּהוֹאִיל וּקְרוֹבָה לְבֵיתוֹ דַּעְתּוֹ לְהִשְׁתַּמֵּשׁ בָּהּ תָּמִיד וּכְמֻקָּף לְדִירָה דָּמִי: וּבִלְבַד שֶׁתְּהֵא שִׁבְעִים אַמָּה וְשִׁירַיִם עַל שִׁבְעִים אַמָּה וְשִׁירָיִם. וְלֹא יוֹתֵר. וְתַנָּא קַמָּא נַמִּי דִּלְעֵיל דְּאָמְרוּ לוֹ לֹא אָמְרוּ בֵּית סָאתַיִם אֶלָּא לַקַּרְפֵּף וְלַגִּנָּה אֲבָל אִם הָיָה דִּיר אוֹ סַהַר אוֹ חָצֵר וְכוּ', כְּרַבִּי עֲקִיבָא קָאֲמַר, דְּהֵיכָא דְּאִיכָּא בֵּית דִּירָה אֲפִלּוּ טוּבָא נַמִּי, וּכְשֶׁלֹּא הֻקַּף לְדִירָה בֵּית סָאתַיִם. וּמַאי בֵּינַיְהוּ, אָמְרִינַן בַּגְּמָרָא דְּאִיכָּא בֵּינַיְהוּ דָּבָר מֻעָט, שֶׁבֵּית סָאתַיִם יוֹתֵר עַל שִׁבְעִים אַמָּה וְשִׁירַיִם רְבוּעִים, וְתַנָּא קַמָּא שָׁרֵי בְּבֵית סָאתַיִם שְׁלֵמִים, וּלְרַבִּי עֲקִיבָא שִׁבְעִים אַמָּה וְשִׁירַיִם עַל שִׁבְעִים אַמָּה וְשִׁירַיִם וְתוּ לֹא. וְכֵיצַד יָדַעְנוּ דְּבֵית סָאתַיִם יוֹתֵר עַל שִׁבְעִים אַמָּה וְשִׁירַיִם רְבוּעִים, דְּהָא אָמְרִינַן בַּגְּמָרָא וְכַמָּה הֵן סָאתַיִם, כַּחֲצַר הַמִּשְׁכָּן, וַחֲצַר הַמִּשְׁכָּן כְּתִיב בֵּיהּ (שְׁמוֹת כז) אֹרֶךְ הֶחָצֵר מֵאָה בָּאַמָּה וְרֹחַב חֲמִשִּׁים בַּחֲמִשִּׁים, וְדָרְשִׁינַן מַאי חֲמִשִּׁים בַּחֲמִשִּׁים, אָמְרָה תּוֹרָה טֹל חֲמִשִּׁים שֶׁאָרְכּוֹ יָתֵר עַל רָחְבּוֹ וְסַבֵּב חֲמִשִּׁים הַנּוֹתָרִים לְשִׁעוּר שַׁבָּת, דַּהֲווּ לְהוּ שִׁבְעִים אַמָּה וְאַרְבָּעָה טְפָחִים מְרֻבָּעִים. כֵּיצַד, עֲשֵׂה מֵהֶם חָמֵשׁ רְצוּעוֹת שֶׁל עֶשֶׂר אַמָּה רֹחַב וְאָרְכָּן חֲמִשִּׁים, תֵּן אַחַת לְמִזְרָחָהּ וְאַחַת לְמַעֲרָבָהּ, הֲרֵי רָחְבָּהּ שִׁבְעִים וְאָרְכָּהּ חֲמִשִּׁים, שִׂים אַחַת לַדָּרוֹם וְאַחַת לַצָּפוֹן, הֲרֵי שִׁבְעִים עַל שִׁבְעִים, אֶלָּא שֶׁהַקְּרָנוֹת פְּגוּמִים לְכָל קֶרֶן וְקֶרֶן עֶשֶׂר עַל עֶשֶׂר מִפְּנֵי הַתּוֹסֶפֶת שֶׁהוֹסַפְתָּ. טֹל מִן הָרְצוּעָה הַחֲמִשִּׁית אַרְבַּע חֲתִיכוֹת שֶׁל עֶשֶׂר וְשִׂים לְאַרְבַּע קְרָנוֹת, וְנִתְמַלְּאוּ. טֹל עֶשֶׂר עַל עֶשֶׂר הַנּוֹתָרִים שֶׁהֵם שִׁשִּׁים טְפָחִים וַעֲשֵׂה אוֹתָם רְצוּעוֹת שֶׁל שְׁנֵי טְפָחִים, הֲרֵי שְׁלֹשִׁים רְצוּעוֹת אֹרֶךְ כָּל אַחַת עֶשֶׂר אַמּוֹת, שֶׁהֵן שְׁלֹשׁ מְאוֹת אַמָּה. תֵּן שִׁבְעִים לְכָל רוּחַ, הֲרֵי שִׁבְעִים וְאַרְבָּעָה טְפָחִים עַל שִׁבְעִים אַמּוֹת וְאַרְבָּעָה טְפָחִים, אֶלָּא שֶׁהַקְּרָנוֹת פְּגוּמִים טְפָחַיִם עַל טְפָחַיִם, וְנִשְׁאֲרוּ בְּיָדְךָ עֶשְׂרִים אַמָּה, טֹל מֵהֶם שְׁמֹנָה טְפָחִים וְשִׂים לַקְּרָנוֹת וְנִתְמַלְּאוּ, וְנִשְׁאֲרוּ בְּיָדְךָ שְׁמֹנֶה עֶשְׂרֵה אַמּוֹת וְאַרְבָּעָה טְפָחִים אֹרֶךְ בְּרֹחַב טְפָחַיִם, וְהַיְנוּ דָּבָר מֻעָט, שֶׁאִם בָּאתָ לְחַלְּקָן וּלְהַקִּיף אֵין מַגִּיעַ תּוֹסֶפֶת לְרֹחַב שְׁנֵי שְׁלִישֵׁי אֶצְבַּע, דְּהָא בָּעֵי לְמֶעְבַּד מִנָּהּ רְצוּעוֹת שֶׁל רפ"ג אֹרֶךְ לְהַקִּיף הָאַרְבַּע רוּחוֹת. כָּךְ מָצָאתִי פֵּרוּשׁ שִׁיטָה זוֹ בְּפֵרוּשֵׁי רַשִׁ"י, וְנָכוֹן הוּא. וְהָרַמְבַּ"ם בִּקֵּשׁ חֶשְׁבּוֹנוֹת רַבִּים וְלֹא יָרַדְתִּי לְסוֹף דַּעְתּוֹ . וּבַמֶּה שֶׁאָמַר רַבִּי עֲקִיבָא אֲפִלּוּ אֵין בָּהּ אֶחָד מֵאֵלּוּ מְטַלְטְלִים בְּתוֹכָהּ, הִלְכְתָא כְּוָתֵיהּ. אֲבָל בְּמַאי דְּפָלֵיג אַחֲכָמִים דְּאָמְרוּ לוֹ דִּלְעֵיל וְסָבַר דְּשִׁבְעִים אַמָּה וְשִׁירַיִם עַל שִׁבְעִים אַמָּה וְשִׁירַיִם וְתוּ לֹא, בְּהָא הֲלָכָה כַּחֲכָמִים. דִּבְבֵית סָאתַיִם כַּחֲצַר הַמִּשְׁכָּן שָׁרֵי, שֶׁהוּא דָּבָר מֻעָט יוֹתֵר עַל שִׁבְעִים אַמָּה וְשִׁירַיִם רְבוּעִים: אִם הָיְתָה אָרְכָּהּ יוֹתֵר עַל רָחְבָּהּ. וְאַף עַל גַּב דְּמִעֵט רָחְבָּהּ וְהוֹסִיף עַל אָרְכָּהּ וְאֵין בְּכֻלָּהּ אֶלָּא בֵּית סָאתַיִם, אֵין מְטַלְטְלִין, דִּמְרֻבָּע הוּא דְּשָׁרוּ רַבָּנָן הֵיכָא דְּלֹא הֻקַּף לְדִירָה. וַהֲלָכָה כְּרַבִּי יוֹסֵי דְּפָלֵיג אַדְּרַבִּי אֱלִיעֶזֶר וְלֹא בָּעֵי מְרֻבָּעוֹת:

אֲפִלּוּ הִיא כְּבֵית כּוֹר. אַפְּלֻגְתָּא דִּלְעֵיל קָאֵי אַגִּנָּה וְקַרְפֵּף: אַנְשֵׁי חָצֵר שֶׁשָּׁכַח אַחַד מֵהֶם וְלֹא עֵרֵב. וּלְמָחָר בִּטֵּל רְשׁוּת חֲצֵרוֹ לַחֲבֵרָיו, שֶׁהָיָה אוֹסֵר עֲלֵיהֶן מִלְּהוֹצִיא מִבָּתֵּיהֶן לֶחָצֵר, שֶׁהֶחָצֵר בִּרְשׁוּת כֻּלָּן וְחֶלְקוֹ אוֹסֵר עָלֶיהָ. הוֹאִיל וּבִטֵּל רְשׁוּת חֲצֵרוֹ בִּטֵּל אַף רְשׁוּת בֵּיתוֹ, אַף עַל פִּי שֶׁלֹּא פֵּרֵשׁ, וְנַעֲשָׂה אוֹרֵחַ אֶצְלָם, לְפִיכָךְ אַף בֵּיתוֹ מֻתָּר לָהֶם. אֲבָל אָסוּר לוֹ לְהַכְנִיס וּלְהוֹצִיא מִבֵּיתוֹ לֶחָצֵר, אַף עַל פִּי שֶׁהוּא מֻתָּר לְהַכְנִיס וּלְהוֹצִיא מִבָּתֵּיהֶם לֶחָצֵר כִּשְׁאָר כָּל אָדָם הַנִּכְנָס לְבֵית חֲבֵרוֹ שֶׁמֻּתָּר לוֹ לְהוֹצִיא מִבֵּית חֲבֵרוֹ לֶחָצֵר, דְּהָא רְשׁוּתָא דְּחַד הוּא, מִכָּל מָקוֹם מִבֵּיתוֹ אָסוּר הוּא לְהוֹצִיא. וְלֹא אָמְרִינַן כֵּיוָן דְּבִטֵּל לְהוּ רְשׁוּת בֵּיתוֹ הֲרֵי בֵּיתוֹ כְּבָתֵּיהֶם, דִּבְשָׁעָה שֶׁמּוֹצִיא מִבֵּיתוֹ לֶחָצֵר חוֹזֵר וּמַחֲזִיק בִּרְשׁוּתוֹ הוּא וְאוֹסֵר עֲלֵיהֶן, כְּדִתְנַן בְּהַדָּר עִם הָעוֹבֵד כּוֹכָבִים, מִי שֶׁנָּתַן רְשׁוּתוֹ וְחָזַר וְהוֹצִיא בֵּין בְּשׁוֹגֵג בֵּין בְּמֵזִיד הֲרֵי זֶה אוֹסֵר: שֶׁיּוֹצְאִין בְּעַקְרַבְנִין בְּפֶסַח. מִשּׁוּם מָרוֹר. וְעַקְרַבְנִין יֶרֶק שֶׁעָלִין שֶׁלּוֹ דּוֹמוֹת לְעַקְרָב. וַאֲנִי שָׁמַעְתִּי סִיב הַגָּדֵל סְבִיבוֹת הַדֶּקֶל: וּבִקַּשְׁתִּי לִי חָבֵר. לִשְׁלֹשָׁה דְּבָרִים הַלָּלוּ אִם שָׁמַע מִפִּיו, וְלֹא מָצָאתִי. וְאֵין הֲלָכָה כְּאֶחָד מֵהֶן. דְּגַבֵּי גִּנָּה וְקַרְפֵּף לֹא שָׁרֵי טְפֵי מִבֵּית סָאתַיִם. וְאַנְשֵׁי חָצֵר שֶׁשָּׁכַח אֶחָד מֵהֶן וְלֹא עֵרֵב וּבִטֵּל רְשׁוּתוֹ וְלֹא בִּטֵּל בֵּיתוֹ, אַף לָהֶם אָסוּר בֵּיתוֹ, אֲבָל מִבָּתֵּיהֶם לֶחָצֵר מֻתָּר. וְעַקְרַבְנִין לָאו מָרוֹר הוּא וְאֵין אָדָם יוֹצֵא בּוֹ יְדֵי חוֹבָתוֹ בְּפֶסַח:
