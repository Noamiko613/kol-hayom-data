פרק ב - משנה א
הַר הַבַּיִת הָיָה חֲמֵשׁ מֵאוֹת אַמָּה. מֻקָּף חוֹמָה סָבִיב: רֻבּוֹ מִן הַדָּרוֹם שֵׁנִי לוֹ מִן הַמִּזְרָח. כְּלוֹמַר, הָרֹחַק שֶׁיֵּשׁ מֵחוֹמַת הַר הַבַּיִת לְחוֹמַת הָעֲזָרָה לְצַד דָּרוֹם, יוֹתֵר מִן הָרֹחַק שֶׁיֵּשׁ בֵּינֵיהֶן מִצַּד מִזְרָח. וְהָרֹחַק שֶׁיֵּשׁ בֵּינֵיהֶן מִצַּד מִזְרָח, יוֹתֵר מִן הָרֹחַק שֶׁיֵּשׁ בֵּינֵיהֶן לְצַד צָפוֹן, וְהַצְּפוֹנִי יוֹתֵר מִן הַמַּעֲרָב:

פרק ב - משנה ב
נִכְנָסִים דֶּרֶךְ יָמִין וּמַקִּיפִין דֶּרֶךְ שְׂמֹאל. כְּגוֹן הַנִּכְנָסִים דֶּרֶךְ שַׁעֲרֵי חֻלְדָּה שֶׁהֵן מִן הַיָּמִין, מַקִּיפִין דֶּרֶךְ שַׁעַר הַטָּדִי : שֶׁאֲנִי אָבֵל. נִשְׁאָלִין אוֹתוֹ מַה לְךָ לְהַקִּיף לִשְׂמֹאל, וְאָמַר שֶׁאֲנִי אָבֵל, אוֹמְרִים לוֹ הַשּׁוֹכֵן בַּבַּיִת הַזֶּה יְנַחֶמְךָ וְכוּ': עֲשִׂיתָן כְּאִלּוּ עָבְרוּ עָלָיו אֶת הַדִּין. אִם הֵן אוֹמְרִים לוֹ כֵּן, נִרְאֶה כְּאִלּוּ חֲבֵרָיו עִוְּתוּ הַדִּין וְנִדּוּהוּ שֶׁלֹּא כַּהֲלָכָה: יִתֵּן בְּלִבְּךָ שֶׁתִּשְׁמַע לְדִבְרֵי חֲבֵרֶיךָ. הַשְׁתָּא מַשְׁמַע שֶׁהוּא עָשָׂה שֶׁלֹּא כַּדִּין וְצָרִיךְ תְּשׁוּבָה. וַהֲלָכָה כְּרַבִּי יוֹסֵי:

פרק ב - משנה ג
לִפְנִים. מֵחוֹמַת הַר הַבַּיִת: סוֹרֵג. מְחִצָּה הָעֲשׂוּיָה נְקָבִים נְקָבִים כְּמִטָּה מְסֹרֶגֶת בַּחֲבָלִים, וְהִיא עֲשׂוּיָה מַדְּפֵי עֵץ אֲרֻכִּים וּקְצָרִים שֶׁמַּרְכִּיבִין אוֹתָן זוֹ עַל זוֹ בַּאֲלַכְסוֹן: וְגָזְרוּ כְנֶגְדָּם שְׁלֹשׁ עֶשְׂרֵה הִשְׁתַּחֲוָיוֹת. כְּשֶׁמַּגִּיעַ כְּנֶגֶד כָּל פִּרְצָה וּפִרְצָה, מִשְׁתַּחֲוֶה וּמוֹדֶה עַל אָבְדַּן מַלְכוּת יָוָן: לִפְנִים. מֵאוֹתוֹ סוֹרֵג הָיָה מָקוֹם פָּנוּי עֶשֶׂר אַמּוֹת, וְהוּא נִקְרָא חֵיל: וּשְׁתֵּים עֶשְׂרֵה מַעֲלוֹת הָיוּ שָׁם. כְּדֵי לַעֲלוֹת מִשָּׁם לְעֶזְרַת נָשִׁים: רוּם מַעֲלָה. כָּל מַעֲלָה מֵהֶן גְּבוֹהָה מֵחֲבֶרְתָּהּ חֲצִי אַמָּה. וְכֵן הַמַּעֲלָה הָרִאשׁוֹנָה גְּבוֹהָה מִן הַקַּרְקַע חֲצִי אַמָּה: וְשִׁלְחָהּ. מֶשֶׁךְ רֹחַב הַמַּעֲלָה, דְּהַיְנוּ מְקוֹם דְּרִיסַת הָרֶגֶל, חֲצִי אַמָּה: חוּץ מִשֶּׁל אוּלָם. חוּץ מִן הַמַּעֲלוֹת שֶׁבֵּין הָאוּלָם וְלַמִּזְבֵּחַ שֶׁלֹּא הָיוּ כֻּלָּם כֵּן, כְּדִתְנַן בְּפֶרֶק ג': חוּץ מִפֶּתַח שֶׁל אוּלָם. דִּתְנַן לְקַמָּן בְּאִידָךְ פִּרְקִין גָּבְהוֹ אַרְבָּעִים אַמָּה וְרָחְבּוֹ עֶשְׂרִים: הָיוּ לָהֶן שְׁקוֹפוֹת. אֶבֶן מֻנַּחַת עַל שְׁתֵּי הַמְּזוּזוֹת שֶׁהַדֶּלֶת שׁוֹקֵף עָלָיו. וּשְׁקוֹפוֹת לְשׁוֹן מַשְׁקוֹף: מִפְּנֵי שֶׁנַּעֲשָׂה בָהֶן נֵס. כְּמוֹ שֶׁמְּפֹרָשׁ בְּיוֹמָא פֶּרֶק ג' : מַצְהִיבוֹת. כְּמוֹ מַזְהִיבוֹת , שֶׁמַּרְאִיתָן דּוֹמָה לְזָהָב לְפִיכָךְ לֹא הֻצְרְכוּ לַעֲשׂוֹתוֹ שֶׁל זָהָב:

פרק ב - משנה ד
כָּל הַכְּתָלִים שֶׁהָיוּ שָׁם. בְּכָל בִּנְיְנֵי הַר הַבַּיִת: הָיוּ גְבוֹהִים. הַרְבֵּה, עַד שֶׁהָיוּ כָּל הַפְּתָחִים שֶׁבָּהֶן גָּבְהָן עֶשְׂרִים אַמָּה, לְבַד מִמַּה שֶּׁעַל הַפְּתָחִים: חוּץ מִכֹּתֶל מִזְרָחִי. הוּא הַתַּחְתּוֹן לְרַגְלֵי הַר הַבַּיִת: שֶׁהַכֹּהֵן הַשּׂוֹרֵף אֶת הַפָּרָה עוֹמֵד בְּהַר הַמִּשְׁחָה. הוּא הַר הַזֵּיתִים אֲשֶׁר עַל פְּנֵי יְרוּשָׁלַיִם מִקֶּדֶם, וּפְנֵי הַכֹּהֵן לַמַּעֲרָב, וּמִתְכַּוֵּן וְרוֹאֶה מֵעַל גֹּבַהּ רֹאשׁ הַכֹּתֶל דֶּרֶךְ הַשְּׁעָרִים שֶׁלִּפְנִים מִמֶּנּוּ אֶת פִּתְחוֹ שֶׁל הֵיכָל, כְּשֶׁהוּא מַזֶּה בְּדָם, כְּדִכְתִיב (בְּמִדְבַּר יט) וְהִזָּה אֶל נֹכַח פְּנֵי אֹהֶל מוֹעֵד. וְאִם הָיָה הַכֹּתֶל גָּבוֹהַּ, אַף עַל פִּי שֶׁהַשְּׁעָרִים מְכֻוָּנוֹת כֻּלָּם זֶה כְּנֶגֶד זֶה, שַׁעַר הַר הַבַּיִת כְּנֶגֶד שַׁעַר עֶזְרַת נָשִׁים, וְשַׁעַר עֶזְרַת נָשִׁים כְּנֶגֶד שַׁעַר הָעֲזָרָה הַגְּדוֹלָה, וְשַׁעַר הָעֲזָרָה הַגְּדוֹלָה כְּנֶגֶד פֶּתַח הַהֵיכָל, לֹא הָיָה יָכוֹל לִרְאוֹת אֶת פֶּתַח הַהֵיכָל דֶּרֶךְ הַפְּתָחִים, לְפִי שֶׁהָהָר הוֹלֵךְ וּמַגְבִּיהַּ וְעוֹלֶה עַד שֶׁקַּרְקַע פֶּתַח הַהֵיכָל גָּבוֹהַּ (שְׁתֵּי) יוֹתֵר מִקַּרְקַע רַגְלֵי הַר הַבַּיִת, וְנִמְצָא שֶׁאַסְקֻפַּת הַהֵיכָל גְּבוֹהָה מִמַּשְׁקוֹף פֶּתַח הַר הַבַּיִת אַמּוֹת, שֶׁהֲרֵי לֹא הָיָה פֶּתַח הַר הַבַּיִת גָּבוֹהַּ אֶלָּא עֶשְׂרִים כְּדִתְנַן לְעֵיל, וְנִמְצָא שֶׁאֵין הַכֹּהֵן הַשּׁוֹחֵט אֶת הַפָּרָה יָכוֹל לִרְאוֹת חֲלָלוֹ שֶׁל פֶּתַח הַהֵיכָל דֶּרֶךְ אוֹתוֹ פֶּתַח:

פרק ב - משנה ה
אֹרֶךְ. מִן הַמִּזְרָח לַמַּעֲרָב: עַל רֹחַב. מִן הַצָּפוֹן לַדָּרוֹם: קְטֻרוֹת אֶלָּא שֶׁאֵינָן מְקוֹרוֹת. מִלְּשׁוֹן וְהִנֵּה עָלָה קִיטוֹר הָאָרֶץ (בְּרֵאשִׁית יט), כְּלוֹמַר מַעֲלוֹת עָשָׁן, לְפִי שֶׁאֵין לָהֶן תִּקְרָה: וּמְשַׁלְּחִים תַּחַת הַדּוּד. לְשָׂרְפָן בָּאֵשׁ שֶׁתַּחַת הַיּוֹרָה שֶׁמְּבַשְּׁלִים בָּהּ הַשְּׁלָמִים, דִּכְתִיב (בְּמִדְבַּר ו) וְנָתַן עַל הָאֵשׁ אֲשֶׁר תַּחַת זֶבַח הַשְּׁלָמִים: מַתְלִיעִין בָּעֵצִים. מְסִירִין הָעֵצִים שֶׁנִּמְצָא בָּהֶן תּוֹלַעַת , לְפִי שֶׁהֵן פְּסוּלִים לַמַּעֲרָכָה: לִשְׁכַּת מְצֹרָעִים. שֶׁשָּׁם הַמְּצֹרָעִים טוֹבְלִים בַּשְּׁמִינִי לְטָהֳרָתָן כְּשֶׁבָּא לְהַכְנִיס יָדוֹ בִּפְנִים לְמַתַּן בְּהוֹנוֹת וְאַף עַל פִּי שֶׁטָּבַל מִבָּעֶרֶב : אָמַר רַבִּי אֱלִיעֶזֶר בֶּן יַעֲקֹב שָׁכַחְתִּי מֶה הָיְתָה מְשַׁמֶּשֶׁת. מִכְּלַל דְּרֵישָׁא כֻּלַּהּ רַבִּי אֱלִיעֶזֶר בֶּן יַעֲקֹב קָאָמַר לָהּ, וְהָכִי מוּכָח בַּגְּמָרָא דְּיוֹמָא דִּסְתָם מַתְנִיתִין דְּמִדּוֹת רַבִּי אֱלִיעֶזֶר בֶּן יַעֲקֹב הִיא: וַחֲלָקָה הָיְתָה בָּרִאשׁוֹנָה. פֵּרֵשׁ רַמְבַּ"ם, פְּרוּצָה, שֶׁלֹּא הָיְתָה מֻקֶּפֶת מְחִצָּה : כְּצוֹצְטְרָה. כְּמוֹ גְּזוּזְטְרָא הִקִּיפוּ סָבִיב לְעֶזְרַת נָשִׁים, שֶׁיִּהְיוּ הַנָּשִׁים עוֹמְדוֹת לְמַעְלָה עַל הַגְּזוּזְטְרָא וְהָאֲנָשִׁים לְמַטָּה לִרְאוֹת בְּשִׂמְחַת בֵּית הַשּׁוֹאֵבָה, כְּדֵי שֶׁלֹּא יָבֹאוּ לִידֵי קַלּוּת רֹאשׁ: חֲמֵשׁ עֶשְׂרֵה מַעֲלוֹת. גֹּבַהּ קַרְקַע עֶזְרַת יִשְׂרָאֵל מֵעֶזְרַת נָשִׁים: לֹא הָיוּ טְרוּטוֹת. אֲרֻכּוֹת וּבַעֲלֵי זָוִיּוֹת כְּדֶרֶךְ כָּל הַמַּעֲלוֹת, אֶלָּא עֲגֻלּוֹת כַּחֲצִי גֹּרֶן עֲגֻלָּה:

וְרָאשֵׁי פִסְפָּסִין. רָאשֵׁי קוֹרוֹת בּוֹלְטִים וְיוֹצְאִים מִן הַכֹּתֶל לְהַבְדִּיל בֵּין עֶזְרַת יִשְׂרָאֵל לְעֶזְרַת כֹּהֲנִים: מַעֲלָה הָיְתָה שָׁם. בְּעֶזְרַת יִשְׂרָאֵל: וּגְבוֹהָה אַמָּה. וְאָרְכָּהּ כְּאֹרֶךְ כָּל הָעֲזָרָה: וְדוּכָן. שֶׁל לְוִיִּם בָּנוּי עָלֶיהָ וְעָשׂוּי כְּמִין אִצְטַבָּא, וְגֹבַהּ הַדּוּכָן אַמָּה וָחֵצִי: וּבוֹ שָׁלֹשׁ מַעֲלוֹת שֶׁל חֲצִי חֲצִי אַמָּה. שֶׁעוֹלִין בָּהֶן לַדּוּכָן : כָּל הָעֲזָרָה. מִתְּחִלַּת עֶזְרַת יִשְׂרָאֵל עַד אַחַת עֶשְׂרֵה אַמָּה מָקוֹם פָּנוּי שֶׁהָיָה אֲחוֹרֵי בֵּית הַכַּפֹּרֶת: עַל רֹחַב. מִצָּפוֹן לְדָרוֹם: שְׁלֹשָׁה עָשָׂר שְׁעָרִים. כִּדְקָא חֲשִׁיב לְהוּ וְאָזִיל. וְהָאוֹמֵר שִׁבְעָה שְׁעָרִים הָיוּ לָעֲזָרָה נוֹתֵן טַעַם לִשְׁלֹשָׁה עָשָׂר הִשְׁתַּחֲוָיוֹת, כְּנֶגֶד שְׁלֹשָׁה עָשָׂר פְּרָצוֹת שֶׁפָּרְצוּ מַלְכֵי יָוָן בַּסּוֹרֵג, כִּדְאָמְרִינַן לְעֵיל בְּפִרְקִין. וְכֻלָּהּ מַתְנִיתִין מְפֹרֶשֶׁת בְּפֶרֶק קַמָּא:
