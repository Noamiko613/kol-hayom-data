פרק ד

פרק ד - משנה א
הֶשֵּׂג יָד בַּנּוֹדֵר. תּוֹרַת דִּין הַשָּׂגַת יָד, שֶׁעָנִי נִדּוֹן בַּעֲרָכִין כְּפִי הֶשֵּׂג יָדוֹ, בָּתַר נוֹדֵר אָזְלִינַן, וְלֹא בָתַר נִדָּר, כְּדִמְפָרֵשׁ לְקַמָּן: וְהַשָּׁנִים בַּנִּדָּר. יֶלֶד שֶׁהֶעֱרִיךְ עֵרֶךְ זָקֵן, נוֹתֵן עֵרֶךְ זָקֵן, דְּלֹא אָזְלִינַן בָּתַר שָׁנִים דְּנוֹדֵר. וְתַנָּא דְּקָרֵי לְמַעֲרִיךְ נוֹדֵר, לִישְׁנָא דִּקְרָא נָקַט, דִּכְתִיב (וַיִּקְרָא כז) עַל פִּי אֲשֶׁר תַּשִּׂיג יַד הַנּוֹדֵר יַעֲרִיכֶנּוּ הַכֹּהֵן. וְאַיְדֵי דְאָמַר הֶשֵּׂג יָד בַּנּוֹדֵר אָמַר נַמִּי הַשָּׁנִים בַּנִּדָּר: וְהָעֲרָכִים בַּנֶּעֱרָךְ. קְצַב הָעֵרֶךְ דְּזָכָר וּנְקֵבָה, בָּתַר נֶעֱרָךְ אָזֵיל, וְלֹא בָּתַר מַעֲרִיךְ, כְּדִמְפָרֵשׁ לְקַמָּן דְּאִישׁ שֶׁאָמַר עֵרֶךְ אִשָּׁה פְלוֹנִית עָלָי נוֹתֵן עֵרֶךְ אִשָּׁה: וְהָעֵרֶךְ בִּזְמַן הָעֵרֶךְ. כְּדִמְפָרֵשׁ לְקַמָּן, שֶׁאִם הֶעֱרִיךְ עַצְמוֹ כְשֶׁהָיָה פָּחוֹת מִבֶּן עֶשְׂרִים דַּהֲוֵי עֵרֶךְ קָטָן, וְקֹדֶם נְתִינָתוֹ הָיָה בֶּן עֶשְׂרִים, אֵינוֹ נוֹתֵן אֶלָּא כְּשָׁעָה שֶׁהֶעֱרִיךְ:

פרק ד - משנה ב
אוֹמֵר אֲנִי אַף בַּעֲרָכִין כֵּן. אִלּוּ מִתְרְמֵי אַף בַּעֲרָכִין דּוּמְיָא דְקָרְבָּנוֹת, הָוֵי כְקָרְבָּנוֹת, דְּהָא דְאָמַרְתְּ דַּעֲרָכִין אֵינָן כְּקָרְבָּנוֹת, מִשּׁוּם דְּלֹא דָמוּ אַהֲדָדֵי, דְּמִפְּנֵי מָה עָנִי שֶׁהֶעֱרִיךְ עָשִׁיר נוֹתֵן עֵרֶךְ עָנִי לְפִי הֶשֵּׂג יָד, לְפִי שֶׁאֵין הֶעָשִׁיר חַיָּב כְּלוּם, וְלֹא דָמִי לִמְצֹרָע. וְזֶה שֶׁאָמַר עַל הֶעָשִׁיר, לֹא נִתְכַּוֵּן אֶלָּא לְפִי מִדַּת שְׁנוֹתָיו שֶׁל עָשִׁיר שֶׁפְּחוּתִים אוֹ יְתֵרִים עַל שְׁנוֹתָיו שֶׁלּוֹ, הִלְכָּךְ נִדּוֹן בְּהֶשֵּׂג יָד. אֲבָל עָשִׁיר שֶׁאָמַר עֶרְכִּי עָלָי דַּהֲוֵי חַיָּב עֵרֶךְ שָׁלֵם, דּוּמְיָא דִמְצֹרָע , וְשָׁמַע הֶעָנִי וְאָמַר מַה שֶּׁאָמַר זֶה עָלָי, נוֹתֵן עֵרֶךְ עָשִׁיר: הָכִי גָּרְסִינַן הָיָה עָנִי וְהֶעֱשִׁיר אוֹ עָשִׁיר וְהֶעֱנִי מְשַׁלֵּם עֵרֶךְ עָשִׁיר. הָיָה עָנִי וְהֶעֱשִׁיר קֹדֶם נְתִינָה , מְשַׁלֵּם עֵרֶךְ עָשִׁיר, אֲשֶׁר תַּשִּׂיג יַד הַנּוֹדֵר אָמַר רַחֲמָנָא, וַהֲרֵי יָדוֹ מַשֶּׂגֶת. עָשִׁיר וְהֶעֱנִי נַמִּי תַּשִּׂיג יַד הַנּוֹדֵר כְּתִיב, וַהֲרֵי הָיְתָה מַשֶּׂגֶת בְּעֵת שֶׁנָּדַר: רַבִּי יְהוּדָה אוֹמֵר אֲפִלּוּ עָנִי וְהֶעֱשִׁיר וְחָזַר וְהֶעֱנִי נוֹתֵן עֵרֶךְ עָשִׁיר. דִּכְתִיב (וַיִּקְרָא כז) וְאִם מָךְ הוּא מֵעֶרְכְּךָ, עַד שֶׁיְּהֵא בְמָכוּתוֹ מִתְּחִלָּתוֹ וְעַד סוֹפוֹ. וְאֵין הֲלָכָה כְּרַבִּי יְהוּדָה:

פרק ד - משנה ג
אֲפִלּוּ אָבִיו מֵת. בְּשָׁעָה שֶׁנִּתְחַיֵּב זֶה בְּקָרְבָּן הָיָה אָבִיו גּוֹסֵס וְנוֹטֶה לָמוּת, וּמֵת וְיָרַשׁ מִמֶּנּוּ רִבּוֹא קֹדֶם שֶׁהֵבִיא קָרְבָּנוֹ, אֵינוֹ מֵבִיא אֶלָּא קָרְבַּן עָנִי, כְּמוֹ שֶׁהָיָה בְּשָׁעָה שֶׁנִּתְחַיֵּב בַּקָּרְבָּן: סְפִינָתוֹ בַיָּם. לֹא שֶׁהָיְתָה סְפִינָתוֹ טְעוּנָה מִסְּחוֹרוֹת שֶׁלּוֹ בְּרִבּוֹא, דְּאִם כֵּן עָשִׁיר הוּא. אֶלָּא שֶׁהָיְתָה סְפִינָתוֹ מֻשְׂכֶּרֶת לַאֲחֵרִים בְּרִבּוֹא שָׂכָר, וְהוּא אֵין בְּיָדוֹ אֶלָּא אוֹתָהּ סְפִינָה. וּמִשּׁוּם שְׂכָרָהּ לָאו עָשִׁיר הוּא, דְּאֵין שְׂכִירוּת מִשְׁתַּלֶּמֶת אֶלָּא בְּסוֹפָהּ, וְנִמְצָא שֶׁעַכְשָׁיו עָנִי הוּא. וּמִשּׁוּם הַסְּפִינָה עַצְמָהּ לָאו עָשִׁיר הוּא, דְּהַאי תַּנָּא סָבַר כְּמַאן דְּאָמַר לְקַמָּן שֶׁאִם הָיָה הַמַּעֲרִיךְ חַמָּר נוֹתֵן לוֹ הַכֹּהֵן חֲמוֹרוֹ וְאֵינוֹ נוֹטְלוֹ לַהֶקְדֵּשׁ, וְאִם הָיָה אִכָּר מַנִּיחַ לוֹ צֶמֶד בְּקָרוֹ דְּהִיא פַּרְנָסָתוֹ, וְהָכִי נַמִּי מַנִּיחַ לוֹ סְפִינָתוֹ:

פרק ד - משנה ד
יֶלֶד. מִבֶּן עֶשְׂרִים וְעַד בֶּן שִׁשִּׁים, דַּהֲוֵי עֶרְכּוֹ חֲמִשִּׁים שְׁקָלִים: נוֹתֵן כִּזְמַן הָעֵרֶךְ. דִּכְתִיב (וַיִּקְרָא כז) כְּעֶרְכְּךָ יָקוּם , אֵינוֹ נוֹתֵן אֶלָּא כִּזְמַן הָעֵרֶךְ: יוֹם שְׁלֹשִׁים כִּלְמַטָּה. אִם אָמַר עֵרֶךְ פְּלוֹנִי קָטָן עָלָי, וְאוֹתוֹ קָטָן הָיָה בְּאוֹתוֹ יוֹם בֶּן שְׁלֹשִׁים, הָוֵי כִלְמַטָּה וְלֹא אָמַר כְּלוּם. דְּאֵין עֵרֶךְ לְפָחוּת מִבֶּן חֹדֶשׁ. דְּהָכִי כְּתִיב וְאִם מִבֶּן חֹדֶשׁ: מִבֶּן שִׁשִּׁים שָׁנָה וָמַעְלָה. מַשְׁמַע שֶׁשָּׁלְמָה שְׁנַת שִׁשִּׁים וְאָז הוּא נִדּוֹן כְּשִׁשִּׁים, אֲבָל בִּשְׁנַת שִׁשִּׁים נִדּוֹן כְּיֶלֶד: הֵן אִם עָשִׂינוּ. בִּתְמִיהָה. כְּלוֹמַר, וְכִי כֵן הוּא, אִם עָשִׂינוּ שִׁשִּׁים כִּלְמַטָּה לְהַחֲמִיר דְּעֵרֶךְ גָּדוֹל יֵשׁ לְפָחוּת מִבֶּן שִׁשִּׁים חֲמִשִּׁים סֶלַע, וּלְיוֹתֵר מִבֶּן שִׁשִּׁים אֵינוֹ אֶלָּא חֲמִשָּׁה עָשָׂר: כִּלְמַטָּה מִמֶּנָּה לְהָקֵל. דְּעֵרֶך יוֹתֵר עַל בֶּן עֶשְׂרִים, גָּדוֹל הוּא מִפָּחוּת מִבֶּן עֶשְׂרִים. וְכֵן בְּבֶן חָמֵשׁ. כְּדִכְתִיבֵי בִקְרָאֵי: רַבִּי אֱלִיעֶזֶר אוֹמֵר. לְעוֹלָם שְׁנַת חָמֵשׁ וּשְׁנַת עֶשְׂרִים וּשְׁנַת שִׁשִּׁים כִּלְמַטָּה, עַד שֶׁתִּשְׁלַם כָּל הַשָּׁנָה וְחֹדֶשׁ וְיוֹם אֶחָד מִשָּׁנָה הָאַחֶרֶת. דְּיָלֵיף גְּזֵרָה שָׁוָה לְמַעְלָה לְמַעְלָה מִבְּכוֹר, נֶאֱמַר כָּאן מִבֶּן שִׁשִּׁים שָׁנָה וָמַעְלָה, וְנֶאֱמַר בִּבְכוֹר אָדָם (בְּמִדְבַּר יח) וּפְדוּיָו מִבֶּן חֹדֶשׁ וָמַעְלָה , מַה לְּהַלָּן חֹדֶשׁ וְיוֹם אֶחָד, שֶׁאֵין בְּכוֹר נִפְדֶּה פָּחוּת מִבֶּן חֹדֶשׁ וְיוֹם אֶחָד, אַף כָּאן עַד שֶׁיּוֹסִיף עַל שִׁשִּׁים חֹדֶשׁ וְיוֹם אֶחָד . וּשְׁנַת חָמֵשׁ וּשְׁנַת עֶשְׂרִים יָלְפִינַן בִּגְזֵרָה שָׁוָה שָׁנָה שָׁנָה מִשְּׁנַת שִׁשִּׁים, כִּדְיָלֵיף תַּנָּא קַמָּא לְעֵיל. וְאֵין הֲלָכָה כְּרַבִּי אֶלְעָזָר:
