פרק א

הַמְּקַנֵּא לְאִשְׁתּוֹ. אַף עַל גַּב דְּהַמְּקַנֵּא מַשְׁמַע דִּיעֲבַד אִין לְכַתְּחִלָּה לֹא , הַהֲלָכָה הִיא שֶׁחַיָּב אָדָם לְקַנְאוֹת לְאִשְׁתּוֹ אִם הִיא צְרִיכָה לְכָךְ. וּלְקַמָּן מְפָרֵשׁ כֵּיצַד מְקַנֵּא: מְקַנֵּא לָהּ עַל פִּי שְׁנַיִם. אִם בָּא לְהַשְׁקוֹתָהּ מֵי הַמָּרִים, צָרִיךְ לְהָבִיא שְׁנֵי עֵדִים וְלוֹמַר לָהּ בִּפְנֵיהֶם אַל תִּסָּתְרִי עִם אִישׁ פְּלוֹנִי. וְאִם לֹא קִנֵּא לָהּ בִּפְנֵי שְׁנַיִם, אֵינָהּ נֶאֱסֶרֶת עָלָיו בִּסְתִירָה זוֹ וְאֵינוֹ מַשְׁקֶה אוֹתָהּ: עַל פִּי עֵד אֶחָד אוֹ עַל פִּי עַצְמוֹ. אֲפִלּוּ אֵין עֵדִים שֶׁנִּסְתְּרָה אֶלָּא עֵד אֶחָד, אוֹ הוּא עַצְמוֹ אוֹמֵר רְאִיתִיהָ שֶׁנִּסְתְּרָה אַחַר שֶׁקִּנֵּאתִי לָהּ, נֶאֶסְרָה בִּסְתִירָה זוֹ עַד שֶׁתִּשְׁתֶּה, שֶׁסְּתִירָתָהּ אוֹסַרְתָּהּ עָלָיו מִסָּפֵק : רַבִּי יְהוֹשֻׁעַ אוֹמֵר כוּ'. צָרִיךְ שְׁנֵי עֵדִים אַף לִסְתִירָה. וַהֲלָכָה כְּרַבִּי יְהוֹשֻׁעַ. וְאַף עַל פִּי כֵן אָמְרִינַן בַּגְּמָרָא דְּבַזְּמַן הַזֶּה לֹא לֵימָא אֱינִישׁ לְבֵיתֵיהּ אֲפִלּוּ בֵּינוֹ לְבֵינָהּ אַל תִּסָּתְרִי עִם אִישׁ פְּלוֹנִי, שֶׁיֵּשׁ לָחוּשׁ לְדִבְרֵי רַבִּי יוֹסֵי בְּרַבִּי יְהוּדָה דְּאָמַר מְקַנֵּא אָדָם לְאִשְׁתּוֹ עַל פִּי עֵד אֶחָד אוֹ עַל פִּי עַצְמוֹ, וְאִם נִסְתְּרָה אַחַר שֶׁקִּנֵּא לָהּ בֵּינוֹ לְבֵינָהּ, נֶאֶסְרָה עָלָיו אִסּוּר עוֹלָם בַּזְּמַן הַזֶּה שֶׁאֵין שָׁם מֵי הַמָּרִים:

כֵּיצַד הוּא מְקַנֵּא לָהּ אָמַר לָהּ בִּפְנֵי שְׁנַיִם אַל תְּדַבְּרִי עִם אִישׁ פְּלוֹנִי כוּ'. בַּגְּמָרָא מְפָרֵשׁ לְמַתְנִיתִין הָכִי, אָמַר לָהּ בִּפְנֵי שְׁנַיִם אַל תְּדַבְּרִי וְדִבְּרָה, אַל תְּדַבְּרִי וְנִסְתְּרָה, וְלֹא כְּלוּם הוּא וְאֵינָהּ אֲסוּרָה לֹא לְבַעְלָהּ וְלֹא לַתְּרוּמָה. אָמַר לָהּ אַל תִּסָּתְרִי וְדִבְּרָה עִמּוֹ שֶׁלֹּא בַּסֵּתֶר, עֲדַיִן מֻתֶּרֶת לְבַעְלָהּ וּמֻתֶּרֶת לֶאֱכֹל בַּתְּרוּמָה אִם אֵשֶׁת כֹּהֵן הִיא: נִכְנְסָה עִמּוֹ לְבֵית הַסֵּתֶר וְשָׁהֲתָה עִמּוֹ כְדֵי טֻמְאָה אֲסוּרָה לְבֵיתָהּ וַאֲסוּרָה לֶאֱכֹל בַּתְּרוּמָה. וְשִׁעוּר כְּדֵי טֻמְאָה, כְּדֵי לִצְלוֹת בֵּיצַת תַּרְנְגֹלֶת וּלְגָמְעָהּ: וְאִם מֵת חוֹלֶצֶת וְלֹא מִתְיַבָּמֶת. וְאִם מֵת בַּעְלָהּ קֹדֶם שֶׁיַּשְׁקֶנָּה, חוֹלֶצֶת וְלֹא מִתְיַבֶּמֶת. דִּכְתִיב בְּאִשָּׁה שֶׁמָּצָא בָּהּ עֶרְוַת דָּבָר (דְּבָרִים כד) וְיָצְאָה מִבֵּיתוֹ וְהָלְכָה וְהָיְתָה לְאִישׁ אַחֵר, וְדָרְשִׁינַן לְאִישׁ אַחֵר וְלֹא לַיָּבָם. וְאַף עַל פִּי כֵן אֵינָהּ מֻתֶּרֶת לְזָר בְּלֹא חֲלִיצָה, שֶׁכְּשֵׁם שֶׁאִם הָיָה בַּעְלָהּ חַי לֹא הָיְתָה מֻתֶּרֶת בְּלֹא גֵּט, כָּךְ אֵינָהּ מֻתֶּרֶת מִן הַיָּבָם בְּלֹא חֲלִיצָה:

וְאֵלּוּ אֲסוּרוֹת לֶאֱכֹל בַּתְּרוּמָה. לְעוֹלָם. וַאֲפִלּוּ הִיא בַּת כֹּהֵן. וְאִלּוּ שְׁאָר סוֹטוֹת אֵינָן אֲסוּרוֹת אֶלָּא עַד שֶׁיִּשְׁתּוּ וְיִמָּצְאוּ טְהוֹרוֹת: וְשֶׁבָּאוּ עֵדִים שֶׁהִיא טְמֵאָה. וַאֲפִלּוּ לְאַחַר שֶׁשָּׁתְתָה וְלֹא בָּדְקוּהָ הַמַּיִם, אֲסוּרָה אִסּוּר עוֹלָם. לְפִי שֶׁאֵין הַמַּיִם בּוֹדְקִים בְּאִשָּׁה שֶׁיֵּשׁ בָּהּ עֵדִים שֶׁנִּטְמֵאת, שֶׁנֶּאֱמַר (בְּמִדְבָּר ה) וְנִסְתְּרָה וְהִיא נִטְמָאָה וְעֵד אֵין בָּהּ , דְּלֵיכָּא דְּיָדַע בָּהּ הוּא שֶׁהַמַּיִם בּוֹדְקִים, אֲבָל אִיכָּא דְּיָדַע בָּהּ, אֲפִלּוּ הָעֵדִים הֵם בִּמְדִינַת הַיָּם, אֵין הַמַּיִם בּוֹדְקִים: וְשֶׁבַּעְלָהּ בָּא עָלֶיהָ בַדֶּרֶךְ. דְּשׁוּב אֵין הַמַּיִם בּוֹדְקִים אוֹתָהּ, דִּכְתִיב (בְּמִדְבָּר ה) וְנִקָּה הָאִישׁ מֵעָוֹן, בִּזְמַן שֶׁהָאִישׁ מְנֻקֶּה מֵעָוֹן, שֶׁלֹּא בָּא עָלֶיהָ לְאַחַר סְתִירָה, אָז וְהָאִשָּׁה הַהִיא תִּשָּׂא אֶת עֲוֹנָהּ. אֵין הָאִישׁ מְנֻקֶּה מֵעָוֹן, אֵין הַמַּיִם בּוֹדְקִים אֶת אִשְׁתּוֹ: שְׁנֵי תַלְמִידֵי חֲכָמִים. שֶׁיֵּדְעוּ לְהַתְרוֹת בּוֹ: רַבִּי יְהוּדָה אוֹמֵר בַּעְלָהּ נֶאֱמָן עָלֶיהָ. אִם הַבַּעַל נֶאֱמָן עַל אִשְׁתּוֹ בְּאִסּוּר חָמוּר שֶׁל נִדָּה, לֹא יְהֵא נֶאֱמָן בְּאִסּוּר קַל שֶׁל לָאו גְּרֵידָא. וְטַעֲמַיְהוּ דְּרַבָּנָן, מִתּוֹךְ שֶׁאִסּוּר קַל הוּא, תַּקִּיף עֲלֵיהּ יִצְרֵיהּ . וְאֵין הֲלָכָה כְּרַבִּי יְהוּדָה:

הָיוּ מַעֲלִין אוֹתָהּ לְבֵית דִּין הַגָּדוֹל. דִּכְתִיב בְּסוֹטָה (שָׁם) וְעָשָׂה לָהּ הַכֹּהֵן אֶת כָּל הַתּוֹרָה הַזֹּאת, וּכְתִיב הָתָם (דְּבָרִים יז) עַל פִּי הַתּוֹרָה אֲשֶׁר יוֹרוּךָ, מַה לְּהַלָּן בְּבֵית דִּין הַגָּדוֹל אַף כָּאן בְּבֵית דִּין הַגָּדוֹל: וּמְאַיְּמִין עָלֶיהָ. כְּדֵי שֶׁתּוֹדֶה: הַרְבֵּה יַלְדוּת עוֹשָׂה. וְגוֹרֵם לְיַלְדָּה שֶׁתָּבֹא לִידֵי קִלְקוּל: דְּבָרִים שֶׁאֵינָן כְּדַאי לְשׁוֹמְעָן. שֶׁאֵינָהּ רְאוּיָה לִשְׁמֹעַ אוֹתָן הַדְּבָרִים שֶׁאוֹמְרִים לְפָנֶיהָ מִמַּעֲשֵׂיהֶן שֶׁל צַדִּיקִים, כְּגוֹן יְהוּדָה הוֹדָה וְלֹא בּוֹשׁ בְּמַעֲשֵׂה תָּמָר, רְאוּבֵן הוֹדָה וְלֹא בּוֹשׁ בְּמַעֲשֵׂה בִּלְהָה :

שׁוֹבֶרֶת כְּתֻבָּתָהּ. כּוֹתֶבֶת שׁוֹבֵר עַל כְּתֻבָּתָהּ, זָנִיתִי וְאִבַּדְתִּי כְּתֻבָּתִי, כְּדֵי שֶׁלֹּא תוּכַל לַחְזֹר וְלִתְבֹּעַ כְּתֻבָּתָהּ מִמֶּנּוּ. וּבְמָקוֹם שֶׁאֵין כּוֹתְבִין כְּתֻבָּה אֶלָּא סוֹמְכִין עַל תְּנַאי בֵּית דִּין הוּא שֶׁכּוֹתֶבֶת שׁוֹבֵר, אֲבָל בְּמָקוֹם שֶׁכּוֹתְבִים כְּתֻבָּה, מְקָרַעַת כְּתֻבָּתָהּ וְאֵינָהּ כּוֹתֶבֶת שׁוֹבֵר: מַעֲלִין אוֹתָהּ לְשַׁעַר הַמִּזְרָח. מַעֲלִין וּמוֹרִידִין אוֹתָהּ כְּדֵי לְיַגְּעָהּ, אוּלַי תִטָּרֵף דַּעְתָּהּ וְתוֹדֶה: שֶׁעַל פֶּתַח שַׁעַר נִקָּנוֹר. אָדָם שֶׁשְּׁמוֹ נִקָּנוֹר שֶׁהֵבִיא דְּלָתוֹת לְאוֹתוֹ פֶּתַח מֵאֶלֶכְּסַנְדְרִיָּא שֶׁל מִצְרַיִם וְנַעֲשׂוּ לוֹ נִסִּים וְנִקְרָא שַׁעַר נִקָּנוֹר עַל שְׁמוֹ כִּדְאָמְרִינַן בְּיוֹמָא : שֶׁשָּׁם מַשְׁקִין אֶת הַסּוֹטוֹת. דְּבָעִינַן לִפְנֵי ה': וּמְטַהֲרִין אֶת הַיּוֹלְדוֹת. כְּדֵי שֶׁיַּעַמְדוּ עַל גַּבֵּי קָרְבָּנָן . וְהוּא הַדִּין זָבִין וְזָבוֹת הַטְּעוּנִים קָרְבָּן: וּמְטַהֲרִין אֶת הַמְּצֹרָעִים. כְּדִכְתִיב (וַיִּקְרָא יד) וְהֶעֱמִיד הַכֹּהֵן הַמְטַהֵר וְכוּ' לִפְנֵי ה': אוֹחֵז בִּבְגָדֶיהָ. בְּבֵית הַצַּוָּאר שֶׁלָּהֶן: אִם נִקְרְעוּ. אֵינוֹ חוֹשֵׁשׁ, וְאִם נִפְרְמוּ אֵינוֹ חוֹשֵׁשׁ. פְּרִימָה גְּדוֹלָה מִקְּרִיעָה, שֶׁנִּקְרַע לִקְרָעִים הַרְבֵּה. לָשׁוֹן אַחֵר, פְּרִימָה מִן הַצְּדָדִין קְרִיעָה בָּאֹרֶךְ: עַד שֶׁמְּגַלֶּה אֶת לִבָּהּ. דִּכְתִיב (בְּמִדְבָּר ה) וּפָרַע אֶת רֹאשׁ הָאִשָּׁה, אֵין לִי אֶלָּא רֹאשָׁהּ, גּוּפָה מִנַּיִן, תַּלְמוּד לוֹמַר הָאִשָּׁה, אִם כֵּן מַה תַּלְמוּד לוֹמַר אֶת רֹאשׁ, מְלַמֵּד שֶׁסּוֹתֵר אֶת שְׂעָרָהּ: רַבִּי יְהוּדָה אוֹמֵר כוּ'. שֶׁמָּא תֵּצֵא זַכָּאָה וְיִתְגָּרוּ בָּהּ פִּרְחֵי כְהֻנָּה שֶּׁרָאוּ לִבָּהּ נָאֶה וְשַׂעֲרָהּ נָאֶה. וְאֵין הֲלָכָה כְּרַבִּי יְהוּדָה:

הָיְתָה מִתְכַּסָּה בִלְבָנִים מְכַסָּהּ בִשְׁחוֹרִים. וְאִם הָיוּ שְׁחוֹרִים יָפִין לָהּ, מְכַסָּהּ בִּבְגָדִים מְכֹעָרִים: קַטְלָיוֹת. עֲנָקִים שֶׁמְּשִׂימִים סְבִיב הַצַּוָּאר. וּמִפְּנֵי שֶׁהָאִשָּׁה חוֹנֶקֶת עַצְמָהּ בָּהֶם כְּדֵי שֶׁתֵּרָאֶה בַּעֲלַת בָּשָׂר נִקְרָאִים קַטְלָיוֹת: נְזָמִים וְטַבָּעוֹת מַעֲבִירִים מִמֶּנָּה. מְפָרֵשׁ בַּגְּמָרָא דְּלֹא תֵימָא שֶׁיָּנִיחוּ לָהּ הַנְּזָמִים וְהַטַּבָּעוֹת אַחַר שֶׁקָּרְעוּ בְּגָדֶיהָ וְגִלּוּ אֶת לִבָּהּ, שֶׁזֶּה יוֹתֵר גְּנַאי לָהּ, כְּמוֹ שֶׁגְּנַאי יוֹתֵר לְאָדָם עָרוּם שֶׁיֵּלֵךְ בְּמִנְעָלִים בְּרַגְלָיו מִלֵּילֵךְ יָחֵף, קָא מַשְׁמַע לָן דְּלֹא: חֶבֶל מִצְרִי. מְפָרֵשׁ בַּיְרוּשַׁלְמִי, לְפִי שֶׁעָשְׂתָה מַעֲשֵׂה מִצְרַיִם . וְאִם אֵין חֶבֶל מִצְרִי מֵבִיא שְׁאָר חֲבָלִים: וְקוֹשְׁרוֹ לְמַעְלָה מִדַּדֶּיהָ. שֶׁלֹּא יִפְּלוּ בְּגָדֶיהָ וְתִשָּׁאֵר עֲרוּמָה: וְכָל הָרוֹצֶה לִרְאוֹת בָּא לִרְאוֹת. וְדַוְקָא בַּאֲנָשִׁים רְשׁוּת וְאֵינוֹ חוֹבָה. אֲבָל בְּנָשִׁים, חוֹבָה עַל כָּל הַנָּשִׁים הַמְּצוּיוֹת שָׁם לָבֹא וְלִרְאוֹת, כְּדִכְתִיב וְנִוַּסְּרוּ כָל הַנָּשִׁים: וְכָל הַנָּשִׁים מֻתָּרוֹת לִרְאוֹתָהּ. חַיָּבִין לִרְאוֹתָהּ. הָכִי מְפָרֵשׁ לַהּ בַּגְּמָרָא:

הַמָּקוֹם נִוְּלָהּ. שֶׁמְּגַלֶּה כֹּהֵן אֶת רֹאשָׁהּ וְסוֹתֵר אֶת שַׂעֲרָה וְקוֹרֵעַ אֶת בְּגָדֶיהָ: הִיא גִלְּתָה אֶת עַצְמָהּ. יָשְׁבָה בְּפָרָשַׁת דְּרָכִים וְגִלְּתָה עַצְמָהּ לַנּוֹאֵף: הַמָּקוֹם גִּלָּה עָלֶיהָ. שֶׁכֹּהֵן מַעֲמִידָהּ עַל שַׁעַר נִקָּנוֹר וּמַרְאֶה קְלוֹנָהּ לַכֹּל: בַּיָּרֵךְ הִתְחִילָה בָעֲבֵרָה תְחִלָּה. בְּדֶרֶךְ תַּשְׁמִישׁ הַיָּרֵךְ נֶהֱנֶה תְּחִלָּה בְּקֵרוּב בָּשָׂר: וּשְׁאָר כָּל הַגּוּף לֹא פָלֵט. שְׁאָר אֶבְרֵי גּוּפָהּ אֵינָם פְּלֵטִים מִן הַחֹלִי, וְאַף עַל פִּי שֶׁלֹּא הֻזְכְּרוּ בַּמִּקְרָא אֶלָּא בֶּטֶן וְיָרֵךְ:

לוֹנְבִיּוֹת. חֲנִיתוֹת: לֵב אָבִיו. אָמַר לוֹ אַבְשָׁלוֹם לְאָבִיו, כְּתֹב לִי שֶׁיֵּלְכוּ עִמִּי שְׁנַיִם אִישׁ שֶׁאֶבְחַר מִכָּל יִשְׂרָאֵל, כָּתַב לוֹ. וְהָיָה מַרְאֶה חוֹתָמוֹ שֶׁל אָבִיו לִשְׁנַיִם, וְהוֹלְכִים אַחֲרָיו, לִשְׁנַיִם אֲחֵרִים וְהוֹלְכִים אַחֲרָיו, עַד שֶׁכִּנֵּס מָאתַיִם אִישׁ, כֻּלָּן רָאשֵׁי סַנְהֶדְרָאוֹת. הֲרֵי שֶׁגָּנַּב לֵב אָבִיו וְלֵב בֵּית דִּין. וְלֵב כָּל יִשְׂרָאֵל, דִּכְתִיב וַיְּגַנֵּב אַבְשָׁלוֹם אֶת לֵב אַנְשֵׁי יִשְׂרָאֵל:

וְכֵן לְעִנְיַן הַטּוֹבָה. נוֹתְנִים לוֹ לָאָדָם שָׂכָר מֵעֵין הַטּוֹבָה שֶׁעָשָׂה. וּמִיהוּ יוֹתֵר מִמַּה שֶּׁעָשָׂה מִשְׁתַּלֵּם, דְּמִדָּה טוֹבָה מְרֻבָּה בְּתַשְׁלוּמִין מִמִּדַּת פֻּרְעָנוּת :
