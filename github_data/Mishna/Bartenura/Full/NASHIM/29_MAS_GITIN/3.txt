פרק ג

פרק ג - משנה א
כָּל גֵּט. קוֹל סוֹפְרִים מַקְרִין. הַגְּדוֹלִים מַקְרִין לְתַלְמִידֵיהֶם, כְּשֶׁיָּבֹא גֵּט לְפָנֶיךָ כָּתְבֵהוּ כָּךְ. וְהִזְכִּירוּ שֵׁם אֱינַשׁ בְּעָלְמָא: יָתֵר מִכֵּן. וְלֹא זֶה בִּלְבַד שֶׁנִּכְתַּב שֶׁלֹּא לְשֵׁם גֵּרוּשִׁין אֶלָּא לְהִתְלַמֵּד, אֶלָּא אַף זֶה שֶׁנִּכְתַּב לְשֵׁם גֵּרוּשִׁין גְּמוּרִים וְנִמְלַךְ, פָּסוּל: יָתֵר מִכֵּן הָיוּ לוֹ שְׁתֵּי נָשִׁים כוּ'. וְלֹא זֶה בִּלְבַד שֶׁנִּכְתַּב שֶׁלֹּא לְשֵׁם גֵּרוּשִׁין שֶׁל אָדָם זֶה, פָּסוּל, אֶלָּא אַף זֶה שֶׁהָיוּ לוֹ שְׁתֵּי נָשִׁים שֶׁנִּכְתַּב לְשֵׁם גֵּרוּשִׁין דְּהַאי גַּבְרָא, פָּסוּל, כֵּיוָן שֶׁנִּכְתַּב שֶׁלֹּא לְשֵׁם גֵּרוּשִׁין שֶׁל זוֹ: גְּדוֹלָה וּקְטַנָּה. לָאו דַוְקָא: יָתֵר מִכֵּן אָמַר לַלַּבְלָר וְכוּ'. הָא קָא מַשְׁמַע לָן דְּלֹא אָמְרִינַן הֻבְרַר הַדָּבָר לְמַפְרֵעַ דְּבִשְׁעַת כְּתִיבָה נַמִּי דַּעְתֵּיהּ אַהָא וְאִיכָּא גֵּרוּשִׁין דִּידֵיהּ וְדִידָהּ :

פרק ג - משנה ב
הַכּוֹתֵב טָפְסֵי גִטִּין. סוֹפֵר שֶׁרוֹצֶה שֶׁיִּהְיוּ מְזֻמָּנִין אֶצְלוֹ, שֶׁפְּעָמִים אָדָם בָּא לְשָׂכְרוֹ וְהוּא טָרוּד בִּשְׁטָרוֹת אֲחֵרִים: צָרִיךְ שֶׁיָּנִיחַ מְקוֹם הָאִישׁ וּמְקוֹם הָאִשָּׁה וּמְקוֹם הַזְּמַן. וּבַגְּמָרָא מוֹסִיף אַף מְקוֹם הֲרֵי אַתְּ מֻתֶּרֶת לְכָל אָדָם : מִפְּנֵי הַתַּקָּנָה. הִתִּירוּ לִכְתֹּב טָפְסֵי גִּטִּין וּשְׁטָרוֹת שֶׁלֹּא לִשְׁמָן מִפְּנֵי תַּקָּנַת הַסּוֹפֵר שֶׁיִּהְיוּ מְזֻמָּנִים לוֹ, וּבִלְבַד שֶׁיְּשַׁיֵּר תּוֹרֵף לְכָתְבוֹ לִשְׁמָהּ. וְגָזְרִינַן תּוֹרֵף שְׁאָר שְׁטָרוֹת אַטּוּ תּוֹרֵף דְּגִטִּין: רַבִּי יְהוּדָה פוֹסֵל בְּכֻלָּן. דְּגָזַר טֹפֶס אַטּוּ תּוֹרֵף, וּשְׁאָר שְׁטָרוֹת אַטּוּ גִּטִּין: רַבִּי אֶלְעָזָר מַכְשִׁיר בְּכֻלָּן. דְּלֹא גָּזַר שְׁאָר שְׁטָרוֹת אַטּוּ גִּטִּין: חוּץ מִגִּטֵּי נָשִׁים שֶׁנֶּאֱמַר וְכָתַב לָהּ לִשְׁמָהּ. וְגָזְרִינַן טֹפֶס אַטּוּ תּוֹרֵף. וַהֲלָכָה כְּרַבִּי אֶלְעָזָר :

פרק ג - משנה ג
לְאַלְתַּר. מִיָּד . כָּשֵׁר: וְאִם לָאו פָּסוּל. וְדַוְקָא שֶׁאָבַד בְּמָקוֹם שֶׁהַשַּׁיָּרוֹת מְצוּיוֹת, דְּאִיכָּא לְמֵימַר מֵעוֹבְרִים וְשָׁבִים נָפַל . אֲבָל אִם אָבַד בְּמָקוֹם שֶׁאֵין הַשַּׁיָּרוֹת מְצוּיוֹת, אֲפִלּוּ לְאַחַר זְמַן מְרֻבֶּה כָּשֵׁר. וַאֲפִלּוּ אָבַד בְּמָקוֹם שֶׁהַשַּׁיָּרוֹת מְצוּיוֹת, אִם יֵשׁ לָעֵדִים בּוֹ סִימָן מֻבְהָק , כְּגוֹן שֶׁיֹּאמְרוּ נֶקֶב יֵשׁ בּוֹ בְּצַד אוֹת פְּלוֹנִית, אוֹ שֶׁיֹּאמְרוּ מֵעוֹלָם לֹא חָתַמְנוּ בְּגֵט בְּאֵלּוּ הַשֵּׁמוֹת אֶלָּא בָזֶה בִּלְבַד, כָּשֵׁר, וַאֲפִלּוּ לְאַחַר זְמַן מְרֻבֶּה: בַּחֲפִיסָה אוֹ בִדְלֻסְקְמָא. אַמְתָּחוֹת שֶׁרְגִילִין לָתֵת בָּהֶם שְׁטָרוֹת, וְיֵשׁ בָּהֶן סִימָן שֶׁהֵם שֶׁלּוֹ : אִם מַכִּירוֹ כָּשֵׁר. מִלְּתָא בְּאַנְפֵּי נַפְשָׁהּ הִיא. וְהָכִי קָאָמַר, מְצָאוֹ בַחֲפִיסָה אוֹ בִדְלֻסְקְמָא אַף עַל פִּי שֶׁאֵינוֹ מַכִּירוֹ לַגֵּט, אוֹ שֶׁמַּכִּירוֹ לַגֵּט אַף עַל פִּי שֶׁמְּצָאוֹ בְּכָל מָקוֹם, כָּשֵׁר: נוֹתְנוֹ לָהּ בְּחֶזְקַת שֶׁהוּא קַיָּם. וְלֹא חָיְשִׁינַן שֶׁמָּא מֵת וּבָטֵל שְׁלִיחוּתוֹ, דְּאָמְרִינַן הַעֲמֵד דָּבָר עַל חֶזְקָתוֹ. אֲבָל אִם נוֹדַע שֶׁמֵּת קֹדֶם שֶׁהִגִּיעַ גֵּט לְיָדָהּ, הַגֵּט בָּטֵל, דְּאֵין גֵּט לְאַחַר מִיתָה: מַקְרִיבִין אוֹתָהּ. וְלֹא חָיְשִׁינַן שֶׁמָּא מֵתוּ בְּעָלֶיהָ, וְחַטָּאת שֶׁמֵּתוּ בְּעָלֶיהָ לְמִיתָה עוֹמֶדֶת:

פרק ג - משנה ד
כַּרְקוֹם. תַּרְגּוּם מָצוֹר כַּרְכּוֹמִין: הַמִּטָּרֶפֶת. וַעֲדַיִן לֹא טָבְעָה : לִדּוֹן. בְּדִינֵי נְפָשׁוֹת: בַּת יִשְׂרָאֵל לְכֹהֵן. חֻמְרֵי מֵתִים: בַּת כֹּהֵן לְיִשְׂרָאֵל. חֻמְרֵי חַיִּים:

פרק ג - משנה ה
הַמֵּבִיא גֵט בְּאֶרֶץ יִשְׂרָאֵל. שֶׁאֵין צָרִיךְ לוֹמַר בְּפָנַי נִכְתַּב וּבְפָנַי נֶחְתָּם: מְשַׁלְּחוֹ בְיַד אַחֵר. וְעוֹשֶׂה שָׁלִיחַ מֵאֵלָיו וְלֹא בְּבֵית דִּין. וְדַוְקָא חָלָה: וְאִם אָמַר לוֹ. בַּעַל לַשָּׁלִיחַ: טֹל לִי הֵימֶנָּה חֵפֶץ פְּלוֹנִי. כְּשֶׁתִּתֵּן לָהּ אֶת הַגֵּט . לֹא יְשַׁלְּחֵנוּ בְּיַד אַחֵר:

פרק ג - משנה ו
עוֹשֶׂה בֵית דִּין. שָׁלִיחַ, וּמְשַׁלְּחוֹ: שְׁלִיחַ בֵּית דִּין אָנִי. וּמִסְתָּמָא בֵּית דִּין עָשׂוּ הַדָּבָר בְּהֶכְשֵׁרוֹ. וְשָׁלִיחַ שֵׁנִי מְמַנֶּה שָׁלִיחַ שְׁלִישִׁי עַד מֵאָה, וְכֻלָּן בְּבֵית דִּין, מִדִּתְנַן בְּמַתְנִיתִין אֵין הַשָּׁלִיחַ הָאַחֲרוֹן כוּ' אֶלָּא אוֹמֵר שְׁלִיחַ בֵּית דִּין אָנִי:

פרק ג - משנה ז
לִהְיוֹת מַפְרִישׁ עֲלֵיהֶן מֵחֶלְקָן. כְּשֶׁיַּפְרִישׁ הַתְּרוּמָה יִמְכְּרֶנָּהּ וִיעַכֵּב הַדָּמִים לְעַצְמוֹ בִּשְׁבִיל חוֹבוֹ שֶׁיֵּשׁ לוֹ עַל הַכֹּהֵן. וּמַעֲשֵׂר רִאשׁוֹן וּמַעֲשַׂר עָנִי יְעַכֵּב וְיֹאכַל בִּשְׁבִיל הַחוֹב שֶׁיֵּשׁ לוֹ עַל הַלֵּוִי וְעַל הֶעָנִי, אֶלָּא שֶׁמַּפְרִישׁ מִמַּעֲשֵׂר רִאשׁוֹן תְּרוּמַת מַעֲשֵׂר לַכֹּהֵן. וְאִם הוּא רָגִיל לִתֵּן תְּרוּמוֹתָיו וּמַעַשְׂרוֹתָיו לָזֶה הַכֹּהֵן אוֹ הַלֵּוִי אוֹ הֶעָנִי שֶׁהִלְוָה לָהֶן, אֵין צָרִיךְ לְזַכּוֹת לָהֶם מַעַשְׂרוֹתָיו וּתְרוּמוֹתָיו עַל יַד אַחֵר, אֶלָּא לוֹקְחָן לְעַצְמוֹ מִיַּד אַחַר שֶׁיַּפְרִישֵׁם . אֲבָל אִם רָגִיל לִתֵּן תְּרוּמוֹתָיו וּמַעַשְׂרוֹתָיו לַאֲחֵרִים, אֵינוֹ יָכוֹל לְעַכְּבָן בְּחוֹבוֹ עַד שֶׁיְּזַכֶּה לָהֶם עַל יְדֵי אַחֵר תְּחִלָּה, וְאַחַר כָּךְ יַחֲזֹר וְיִטְּלֵם בְּחוֹבוֹ: צָרִיךְ לִטֹּל רְשׁוּת מִן הַיּוֹרְשִׁים. שֶׁיָּרְשׁוּ מֵהֶם קַרְקַע מְשֻׁעְבֶּדֶת לְבַעַל חוֹב , צָרִיךְ לִטֹּל מֵהֶם רְשׁוּת אִם רוֹצִים לִפְרֹעַ חוֹב זֶה מִתְּרוּמוֹת וּמַעַשְׂרוֹת הַלָּלוּ, דְּשֶׁמָּא רוֹצִים הֵם לִקַּח מַתְּנוֹתֵיהֶן וְלִפְרֹעַ חוֹב מוֹרִישָׁם מִמָּקוֹם אַחֵר:

פרק ג - משנה ח
הַמַּנִּיחַ פֵּרוֹת לִהְיוֹת מַפְרִישׁ עֲלֵיהֶן כוּ'. סוֹמֵךְ עַל אֵלּוּ וְאוֹכֵל טְבָלִים אֲחֵרִים שֶׁיֵּשׁ לוֹ, וְאוֹמֵר, הֲרֵי תְרוּמָתָן בְּאוֹתָן פֵּרוֹת שֶׁהִקְצֵיתִי לְכָךְ: וְאִם אָבְדוּ. הָלַךְ לְבָדְקָן וּמְצָאָן שֶׁאָבְדוּ: הֲרֵי זֶה חוֹשֵׁשׁ. לְאוֹתָן טְבָלִים שֶׁתִּקֵּן בְּהַבְטָחָתָן שֶׁל אֵלּוּ. וְאִם לֹא אֲכָלָן צָרִיךְ לְהַפְרִישׁ מֵהֶם, שֶׁמָּא כְּשֶׁאָמַר הֲרֵי תְרוּמָתָן בַּפֵּרוֹת שֶׁהִקְצֵיתִי כְּבָר הָיוּ אֲבוּדִין: מֵעֵת לְעֵת. שֶׁל בְּדִיקָה. כְּשֶׁבְּדָקָן וּמְצָאָן אֲבוּדִין חוֹשֵׁשׁ שֶׁמָּא מֵאֶתְמוֹל בָּעֵת הַזֹּאת אָבְדוּ, וְאִם עֲשָׂאָן מַעֲשֵׂר תּוֹךְ מֵעֵת לְעֵת עַל פֵּרוֹת אֲחֵרִים צָרִיךְ לְהַפְרִישׁ עֲלֵיהֶם מִסָּפֵק. וּטְפֵי מֵהָכִי לֹא אַחְמוּר רַבָּנָן לְמֵיחַשׁ, אֶלָּא סוֹמְכִין אַחֲזָקָה: בּוֹדְקִין אֶת הַיַּיִן. שֶׁהִנִּיחוֹ לִהְיוֹת מַפְרִישׁ עָלָיו, צָרִיךְ לְבָדְקוֹ שֶׁמָּא הֶחְמִיץ, וְאֵין תּוֹרְמִין מִן הַחֹמֶץ עַל הַיַּיִן: בְּקָדִים שֶׁל מוֹצָאֵי הֶחָג. כְּשֶׁמְּנַשֶּׁבֶת רוּחַ קָדִים בְּמוֹצָאֵי הֶחָג: וּבִשְׁעַת כְּנִיסַת מַיִם. כְּשֶׁהֵן כְּפוֹל הַלָּבָן נִקְרָאִים בֹּסֶר . וּכְשֶׁהַלַּחְלוּחִית נִכְנֶסֶת וּגְדֵלָה בְּתוֹכוֹ שֶׁיָּכוֹל לַעֲצֹר מֵהֶן כָּל שֶׁהוּא, הַיְנוּ כְּנִיסַת מַיִם. פֵּרוּשׁ אַחֵר, הָיוּ כּוֹתְשִׁים הָעֲנָבִים כְּשֶׁהֵן בֹּסֶר וְנוֹתְנִים לְתוֹכוֹ מַיִם וְעוֹשִׂים חֹמֶץ לְטַבֵּל. וַהֲלָכָה כְּרַבִּי יְהוּדָה:
