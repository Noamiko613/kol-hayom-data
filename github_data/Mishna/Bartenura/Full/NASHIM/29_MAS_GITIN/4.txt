פרק ד

הָשִּוֲלְחָ וֵהִגִּיעָ בָּשִֹּלִיחָ. שִּלֲּא הֹיֹה מִתֵכָּוְּן לִרֵדֲּף אָחַרֹיו לֵהָשִֵּיגוֲ, אּלֹּא שִּנִּשִֵתָּהֹה הָשִֹּלִיחָ בָּדּּרּךֵ וֵהֹיֹה לוֲ לֹזּה דּּרּךֵ לֵשִֹם וֵרֹאֹהוּ וּבִטְּל הָגְּט, אַפִלּוּ הֹכִי בֹּטְל, וֵלֲא אֹמֵרִינָן לִצֵעוּרֹהּ בֵּעֹלֵמֹא מֵכָוְּן, שִּאִם הֹיֹה בֵדָעֵתּוֲ לֵבָטֵּלוֲ הֹיֹה רוֲדְף אָחַרֹיו: אְינוֲ יֹכוֲל לֵבָטֵּלוֲ. הֹא קֹא מָשִֵמָע לֹן דֵּאָף עָל פִּי דֵּחֹזִינָן לְיהּ דֵּמָהַדָר עַלְיהּ מְעִקֹּרֹא לֵבִטּוּלְיהּ, לֲא אֹמֵרִינָן אִגָּלֹּאִי מִלֵּתֹא לֵמָפֵרְעָ דֵּבָטּוֲלְי בָטֵּלְיהּ. וֵהְיכֹא דֵיֹהִיב אֱינָשִ גִּטֹּא לִדֵבִיתֵהוּ לִזֵמָן אוֲ עָל תֵּנָאי, אִי אֹמָר לֹהּ הַרְי זּה גִטְּךֵ מְעָכֵשִֹיו וֵלִזֵמָן פֵּלוֲנִי אוֲ אִם יִתֵקָיְּם תֵּנָאי פֵּלוֲנִי, מִכִּי מֹטְי גִּטֹּהּ לֵיֹדֹהּ לֲא מֹצְי תוּ לֵבָטֵלְיהּ, וֵתִהֵיּה מֵגֲרּשִּת לֵאוֲתוֲ זֵמָן אוֲ כֵּשִּיִּתֵקָיְּם הָתֵּנָאי. וֵאִי לֲא אֹמָר לֹהּ מְעָכֵשִֹיו, אַפִלּוּ לֵבֹתָר דִּמֵטֹא גִּטֹּהּ לֵיֹדֹהּ מֹצְי מֵבָטְּל לְיהּ:

בֹּרִאשִוֲנֹה. לֲא הֹיֹה מֵבָטֵּלוֲ בִּפֵנְי הֹאִשִֹּה וֵלֲא בִּפֵנְי הָשִֹּלִיחָ, אּלֹּא בָּמֹּקוֲם שִּהֹיֹה עוֲמְד הֹיֹה מֵבָטֵּלוֲ בִּפֵנְי שִֵלֲשִֹה: מִפֵּנְי תִקּוּן הֹעוֲלֹם. שִּהָשִֹּלִיחָ שִּאְינוֲ יוֲדְעָ בָּדֹּבֹר מוֲלִיכוֲ לֹהּ וֵהִיא נִשְֵּאת בּוֲ. וּמִכֲּחָ תָּקֹּנָת רָבֹּן גָּמֵלִיאְל מָלֵקִין עָל מִי שִּמֵּבָטְּל הָגְּט אוֲ מוֲסְר מוֲדֹעֹא עָל הָגְּט: בֹּרִאשִוֲנֹה הֹיֹה מֵשִָנּּה שִֵמוֲ וּשִֵמֹהּ. כֵּשִּהֹיוּ לוֲ שִֵנְי שְִמוֲת, אּחֹד כֹּאן וֵאּחֹד בִּמֵדִינָת הָיֹּם, הֹיֹה מֵגֹרֵשִֹהּ בָּשְִּם הָנּוֲהְג בִּמֵקוֲם כֵּתִיבָת הָגְּט, וֵלֲא הֹיֹה מָקֵפִּיד לִכֵתֲּב אּת שִֵנְיהּם: מִפֵּנְי תִקּוּן הֹעוֲלֹם. שִּלֲּא יוֲצִיאוּ לָעָז עָל בֹּנּיהֹ מִן הָשְִּנִי, לוֲמָר לֲא גְּרֵשִֹהּ בָּעֵלֹהּ שִּאְין זּה שִֵמוֲ. וֵאֹדֹם שִּהֶחֵזָק בִּשִֵנְי שְִמוֲת בִּשִֵנְי מֵקוֲמוֲת, חָד בִּמֵקוֲם הָכֵּתִיבֹה וֵחָד בִּמֵקוֲם הָנֵּתִינֹה, אְינֹהּ מֵגֲרּשִּת עָד שִּיִּכֵתֲּב שְִם שִּל מֵקוֲם הָנֵּתִינֹה וֵשְִם שִּל מֵקוֲם הָכֵּתִיבֹה עִמּוֲ. אַבֹל אִם הֶחֵזָק בִּשִֵנְי שְִמוֲת בֵּמֹקוֲם אּחֹד וֵכֹתָב אּחֹד מִן הָשְִּמוֲת, בֵּדִיעַבָד הָגְּט כֹּשְִר. מִיהוּ לֵכָתֵּחִלֹּה צֹרִיךֵ לִכֵתֲּב שִֵנְיהּם. וֵהְיכֹא דֵּשִִנֹּה שִֵמוֲ אוֲ שִֵמֹהּ בָּגְּט, אָף עָל פִּי שִּכֹּתָב אָחָר כֹּךֵ וֵכֹל שְִם שִּיְּשִ לִי, הָגְּט בֹּטְל:

אְין אָלֵמֹנֹה נִפֵרָעָת. כֵּתֶבֹּתֹהּ: מִנִּכֵסְי יֵתוֲמִים, אּלֹּא בִשִֵבוּעֹה. שִּלֲּא נִתֵקָבֵּלֹה כֵּלוּם: נִמֵנֵעוּ מִלֵּהָשִֵבִּיעֹהּ. שִּמִּפֵּנְי שִּהִיא טוֲרָחָת לִפֵנְי הָיֵּתוֲמִים מוֲרֹה הּתְּרֹא לֵעָצֵמֹהּ לִשִֹּבָע שִּלֲּא לֹקֵחֹה כֵּלוּם. וֵאָף עָל פִּי שִּלֹּקֵחֹה דֹּבֹר מֶעֹט סֵבוּרֹה שִּבִּשֵֵכָר טִרֵחֹה נוֲטָלֵתּוֲ וֵאְינוֲ מִפִּרֵעוֲן כֵּתֶבֹּתֹהּ. וּלֵפִיכֹךֵ נִמֵנֵעוּ מִלֵּהָשִֵבִּיעֹהּ וֵהֹיֵתֹה מָפֵסּדּת כֵּתֶבֹּתֹהּ: הִתֵקִין רָבֹּן גָּמֵלִיאְל שִּתֵּהְא נוֲדּרּת לָיֵתוֲמִים כֹּל מָה שִּּיִּרֵצוּ. כֵּגוֲן קוֲנֹם מִינְי מֵזוֲנוֲת עֹלָי אִם נּהֱנְיתִי מִכֵּתֶבֹּתִי: וֵגוֲבֹה כֵתֶבֹּתֹהּ. וֵאִם נִשְֵּאת לֵאָחְר קֲדּם שִּהִדִּירוּהֹ הָיֵתוֲמִים עָל כֵּתֶבֹּתֹהּ, וֵאִם יָדִּירוּהֹ אָחָר שִּנִּשְֵּאת, שִּמֹּא יֹפְר לֹהּ בָּעֵלֹהּ , כְּיצָד יָעַשֵוּ, מָשִֵבִּיעִין אוֲתֹהּ חוּץ לֵבְית דִּין שִֵבוּעֹה דֵּרָבֹּנֹן שִּאְין עֹנֵשִֹהּ מֵרֶבּּה, וֵנוֲטּלּת כֵּתֶבֹּתֹהּ אָחָר שִּנִּשְֵּאת. וּכֵשִּבֹּאת לִגֵבּוֲת כֵּתֶבֹּתֹהּ קֲדּם שִּתִּנֹּשְֵא לֵאָחְר, הֹרֵשִוּת בֵּיָד הָיֵתוֲמִים, רֹצוּ מָשִֵבִּיעִין אוֲתֹהּ חוּץ לֵבְית דִּין, אוֲ מָדִּירִים אוֲתֹהּ בֵּבְית דִּין: הֹעְדִים חוֲתֵמִים עָל הָגְּט מִפֵּנְי תִקּוּן הֹעוֲלֹם. אָתָּרֵוָיֵהוּ קֹאְי, גָּבְּי אָלֵמֹנֹה שִּתֵּהְא נוֲדּרּת לָיּוֲרֵשִִים כֹּל מָה שִּּיִּרֵצוּ, מִפֵּנְי תִקּוּן הֹעוֲלֹם, שִּיִּהֵיוּ הָנֹּשִִים נִשֵֹּאוֲת לֵבָעַלְיהּן וֵלֲא תִּדֵאָגֵנֹה לֵהָפֵסִיד כֵּתֶבֹּתֹן. וֵגָבְּי עְדִים חוֲתֵמִים עָל הָגְּט מִפֵּנְי תִקּוּן הֹעוֲלֹם, דֵּהוֲאִיל וֵעְדְי מֵסִירֹה כֹּרֵתְי, שִּהֹעְדִים שִּנִּמֵסָר הָגְּט בִּפֵנְיהּם לֹאִשִֹּה הְם עִקָּר הָגְּרוּשִִין, לֲא הֹיֹה צֹרִיךֵ שִּיָּחֵתֵּמוּ עְדִים בָּגְּט, אּלֹּא מִפֵּנְי תִקּוּן הֹעוֲלֹם, דֵּחָיֵשִִינָן שִּמֹּא יֹמוּת אּחֹד מִן הֹעְדִים שִּנִּמֵסָר הָגְּט בִּפֵנְיהּם, וֵנִמֵצֹא הָגְּט שִּבֵּיֹדֹהּ בֵּלֲא עְדִים כֵּחָסֵפֹּא בֵּעֹלֵמֹא: הִלְּל הִתֵקִין פֵּרוֲזֵבּוּל. שִּרֹאֹה אּת הֹעֹם שִּנִּמֵנֵעוּ מִלֵּהָלֵווֲת זּה אּת זּה וֵעֹבֵרוּ עָל מָה שִּּכֹּתוּב בָּתּוֲרֹה (דֵּבֹרִים טו) הִשִֹּמּר לֵךֹ פּּן יִהֵיּה דֹּבֹר עִם לֵבֹבֵךֹ בֵּלִיָּעָל, עֹמָד וֵהִתֵקִין פֵּרוֲזֵבּוּל. וֵזּה גּוּפוֲ שִּל פֵּרוֲזֵבּוּל, מוֲסְר אַנִי לֹכּם פֵּלוֲנִי וּפֵלוֲנִי דָּיֹּנִים, שִּכֹּל חוֲב שִּיְּשִ לִי אְצּל פֵּלוֲנִי, שִּאּגֵבּּנּוּ כֹּל זֵמָן שִּאּרֵצּה:

עּבּד שִּנִּשִֵבֹּה וּפֵדֹאוּהוּ. יִשֵֵרֹאְלִים אַחְרִים לֵאָחָר שִּנִּתֵיֹאְשִ רָבּוֲ מִמּּנּוּ: אִם לֵשִוּם עּבּד יִשִֵתָּעֵבְּד. לֵרָבּוֲ שְִנִי : לֵשִוּם בּּן חוֲרִין לֲא יִשִֵתָּעֵבְּד. לֲא לֵרָבּוֲ רִאשִוֲן, וֵלֲא לֵרָבּוֲ שְִנִי. לֵרָבּוֲ שְִנִי לֲא, דֵּהֹא לֵשִוּם בּּן חוֲרִין פָּרֵקְיהּ. לֵרָבּוֲ רִאשִוֲן לֲא, דִּלֵמֹא מִמָּנֵעִי. וֵלֲא פֹּרֵקְי : רָבֹּן שִִמֵעוֲן בּּן גָּמֵלִיאְל אוֲמְר בְּין כֹּךֵ וּבְין כֹּךֵ יִשִֵתָּעֵבְּד. לֵרָבּוֲ רִאשִוֲן. שִּלֲּא יֵהְא כֹּל אּחֹד וֵאּחֹד הוֲלְךֵ וּמָפִּיל עָצֵמוֲ לִגֵיֹסוֲת וּמָפֵקִיעָ עָצֵמוֲ מִיָּד רָבּוֲ : אָפּוֲתִיקִי. פֲּה תֵהְא קֹאְי. מִזּּה תִּגֵבּּה חוֲבֵךֹ וֵלֲא מִמֹּקוֲם אָחְר : וֵשִִחֵרֵרוֲ. רָבּוֲ רִאשִוֲן: שִוּרָת הָדִּין אְין הֹעּבּד חָיֹּב כֵּלוּם. לֵרָבּוֲ שְִנִי. שִּהָשִִּחֵרוּר שִּשִִּחֵרֵרוֲ רָבּוֲ רִאשִוֲן מָפֵקִיעוֲ מִידְי שִִעֵבּוּד: אּלֹּא מִפֵּנְי תִקּוּן הֹעוֲלֹם. שִּמֹּא יִמֵצֵאּנּוּ רָבּוֲ שְִנִי בָּשִּוּק וֵיֲאמָר לוֲ עָבֵדִּי אָתֹּה וֵיוֲצִיא לָעָז עָל בֹּנֹיו: כּוֲפִין אּת רָבּוֲ. שְִנִי, וֵעוֲשְֵהוּ בּּן חוֲרִין. וֵכוֲתְב לוֲ הֹעּבּד שִֵטָר חוֲב עָל דֹּמֹיו, כֵּלוֲמָר כֵּפִי מָה שִּּהוּא שִֹוּה לִמֹּכְר בָּשִּוּק, לֲא כֵּפִי הָחוֲב אִם הֹיֹה הָחוֲב יֹתְר עָל דֹּמֹיו: רָבֹּן שִִמֵעוֲן בּּן גָּמֵלִיאְל אוֲמְר אְין הֹעּבּד כּוֲתְב. שִֵטָר חוֲב, שִּהוּא אְינוֲ חָיֹּב כֵּלוּם. אּלֹּא רָבּוֲ רִאשִוֲן שִּהִזִּיק שִִעֵבּוּדוֲ שִּל זּה, הוּא שִּצֹּרִיךֵ לֵשִָלְּם לוֲ דֹמֹיו, שִּהָמָּזִּיק שִִעֵבּוּדוֲ שִּל חַבְרוֲ חָיֹּב. וָהַלֹכֹה כֵּרָבֹּן שִִמֵעוֲן בּּן גָּמֵלִיאְל:

מִי שִּחּצֵיוֲ עּבּד וֵחּצֵיוֲ בּן חוֲרִין. כֵּגוֲן עּבּד שִּל שִֵנְי שִֶתֹּפִים וֵשִִחֵרֵרוֲ אּחֹד מְהּם. אִי נָמִּי שִּקִּבְּל רָבּוֲ מִמּּנּוּ חַצִי דֹּמֹיו וֵשִִחֵרְר חּצֵיוֲ בֵּאוֲתֹם הָדֹּמִים: תִּקָּנֵתּּם אּת רָבּוֲ. שִּאְינוֲ חֹסְר כֵּלוּם: לִשֵֹּא שִִפֵחֹה אְינוֲ יֹכוֲל. מִפֵּנְי צָד חְרוּת שִּבּוֲ: בָּת חוֲרִין אְינוֲ יֹכוֲל. מִפֵּנְי צָד עָבֵדוּת שִּבּוֲ: כּוֲפִין אּת רָבּוֲ וֵעוֲשְֵהוּ בּן חוֲרִין. וֵהוּא הָדִּין אִם הֹיֹה עּבּד שִּל מְאֹה שִֶתֹּפִין וֵאּחֹד מְהּם שִִחֵרֵרוֲ, שִּכּוֲפִין אּת כֶּלֹּם לֵשִָחֵרֵרוֲ:

יֹצֹא בּן חוֲרִין. אִם בֹּרָח מִן הָנֹּכֵרִי, אוֲ שִּקֹּנֵסוּ אוֲתוֲ בְּית דִּין לִפֵדּוֲתוֲ מִן הָנֹּכֵרִים, כִּדֵאֹמֵרִינָן כּוֲפִין אוֲתוֲ לִפֵדּוֲתוֲ. וֵאָחָר שִּפֵּדֹאוֲ לֲא יִשִֵתָּעֵבְּד בּוֲ. וּקֵנָס חַכֹמִים הוּא, לֵפִי שִּהִפֵקִיעוֲ מִן הָמִּצֵוֲת. וֵכְן לֵחוּצֹה לֹאֹרּץ יֹצֹא לֵחְרוּת, לֵפִי שִּהוֲצִיאוֲ מְאּרּץ יִשֵֵרֹאְל : מִפֵּנְי תִקּוּן הֹעוֲלֹם. שִּלֲּא יִמֵסֵרוּ הָנֹּכֵרִים עָצֵמֹן לֵהָרֵבּוֲת לֵהֹבִיא שִֵבוּיִין כֵּשִּרוֲאִין שִּמּוֲכֵרִין אוֲתֹם בֵּיוֲתְר מִכֵּדְי דֵּמְיהּם: וֵאְין מָבֵרִיחִין אּת הָשִֵּבוּיִין מִפֵּנְי תִקּוּן הֹעוֲלֹם. שִּמֹּא יִקֵצֵפוּ עָל הָשִֵּבוּיִין הֹעַתִידִים לֹבֲא לֵיֹדֹם וֵיִתֵּנוּם בֵּשִָלֵשִֵלֹאוֲת וֵיֹשִֵימוּ בָּסָּד רָגֵלֹם: רָבֹּן שִִמֵעוֲן בּּן גָּמֵלִיאְל אוֲמְר מִפֵּנְי תָקֹּנָת הָשִֵּבוּיִין. דֵּלֲא חֹיְישִ רָבֹּן שִִמֵעוֲן בּּן גָּמֵלִיאְל עָל הֹעַתִידִים לֹבֲא בָּשִִּבֵיֹה, אּלֹּא עָל אוֲתֹם שִּהְם שִֵבוּיִים עָתֹּה עִמּוֲ. שִּאִם אְין שִֹבוּי אּלֹּא הוּא יֵחִידִי, מָבֵרִיחִין אוֲתוֲ, וֵאְין לֹחוּשִ כֹּאן מִפֵּנְי תָקֹּנָת שִֵבוּיִים אַחְרִים כֵּשִּאְין שִֵבוּיִים אַחְרִים עִמּוֲ. וָהַלֹכֹה כֵּרָבֹּן שִִמֵעוֲן בּּן גָּמֵלִיאְל:

מִשִּוּם שְִם רֹע. שִּיֹּצֹא עֹלּיהֹ לָעָז זֵנוּת: מִשִּוּם נּדּר. שִּנֹּדֵרֹה וֵהוּא אוֲמְר אִי אּפֵשִִי בֵּאִשִֹּה נָדֵרֹנִית : לֲא יָחַזִיר. וָאַפִלּוּ נִמֵצֵאוּ הָדֵּבֹרִים שִּקּר, אוֲ הָנּּדּר הִתִּירוֲ חֹכֹם. שִּמֹּא תְּלְךֵ וֵתִנֹּשְֵא לֵאָחְר וֵיִמֹּצְא הָלָּעָז שִּל זֵנוּת שִּקּר, וֵהָנּּדּר יָתִּירְנוּ חֹכֹם וֵהִיא לֲא תִהֵיּה פֵּרוּצֹה בִּנֵדֹרִים, וֵיֲאמָר אִלּוּ הֹיִיתִי יוֲדְעָ שִּכְּן אַפִלּוּ הֹיוּ נוֲתֵנִים לִי מְאֹה מֹנּה לֲא הֹיִיתִי מֵגֹרֵשִֹהּ, וֵנִמֵצֹא גְּט בֹּטְל וּבֹנּיהֹ מָמֵזְרִים . לֵפִיכֹךֵ אוֲמֵרִים לוֲ הֱוְי יוֲדְעָ שִּהָמּוֲצִיא אּת אִשִֵתּוֲ מִשִּוּם שְִם רָע וּמִשִּוּם נּדּר לֲא יָחַזִיר עוֲלֹמִית, וּמִתּוֲךֵ הָדֵּבֹרִים הָלֹּלוּ הוּא גּוֲמְר וּמֵגֹרְשִ גְּרוּשִִין גֵּמוּרִים וֵלֲא מֹצְי תוּ לֵקָלֵקֵלֹהּ: רָבִּי יֵהוּדֹה אוֲמְר כֹּל נּדּר שִּיֹּדֵעוּ בוֲ רָבִּים לֲא יָחַזִיר. רָבִּי יֵהוּדֹה סֹבָר דֵּטָעֵמֹא דֵאֹמוּר רָבֹּנֹן הָמּוֲצִיא אּת אִשִֵתּוֲ מִשִּוּם שְִם רָע וּמִשִּוּם נּדּר לֲא יָחַזִיר, כֵּדְי שִּלֲּא תִהֵיּינֹה בֵּנוֲת יִשֵֵרֹאְל פֵּרוּצוֲת בֹּעַרֹיוֲת וּבִנֵדֹרִים . וּמִשִּוּם הֹכִי קֹאֹמָר דֵּנּדּר שִּיֹּדֵעוּ בוֲ רָבִּים, עַשֵֹרֹה מִיִּשֵֵרֹאְל אוֲ יוֲתְר, אִיכֹּא פֵּרִיצוּתֹא טֵפְי, וּקֵנֹסוּהֹ וֵלֲא יָחַזִיר. וֵשִּלֲּא יֹדֵעוּ בוֲ רָבִּים לְיכֹּא פֵּרִיצוּתֹא כֶּלְּי הָאי וֵלֲא קֵנֹסוּהֹ: לֲא יָחַזִיר. דֵּרָבִּי מְאִיר סֹבָר טָעֵמֹא מִשִּוּם קִלֵקוּלֹא. הִלֵכֹּךֵ נּדּר שִּאְינוֲ יֹכוֲל לֵהֹפְר אַבֹל חֹכֹם יֹכוֲל לֵהָתִּיר, יֹכוֲל הוּא לֵקָלֵקְל אּת הָגְּט לֵאָחָר שִּתִּנֹּשְֵא לֵאָחְר וֵיֲאמָר אִלּוּ הֹיִיתִי יוֲדְעָ שִּחֹכֹם יֹכוֲל לֵהָתִּיר לֲא הֹיִיתִי מֵגֹרֵשִֹהּ: וֵשִּאְינוֲ צֹרִיךֵ חַקִירָת חֹכֹם. אּלֹּא הוּא יֹכוֲל לֵהֹפְר, לֲא הֶצֵרֵכוּ חַכֹמִים לּאֱסֲר עֹלֹיו מִלֵּהָחֵזִירֹהּ, לֵפִי שִּאְינוֲ יֹכוֲל לֵקָלֵקֵלֹהּ וֵלוֲמָר אִלּוּ הֹיִיתִי יוֲדְעָ כוּ', שִּהַרְי נּדּר פֹּתוּחָ הוּא וֵהֹיֹה לוֲ לֵהֹפְר וֵלֲא הְפְר : לֲא אֹסֵרוּ זּה. שִּצֹּרִיךֵ חַקִירָת חֹכֹם, לֵהָחַזִיר: אּלֹּא מִפֵּנְי זּה. שִּאְינוֲ צֹרִיךֵ. דִּבֵצֹרִיךֵ לֲא הֹיֹה לֹנוּ לֹחוּשִ לֵקִלֵקוּלֹא, לֵפִי שִּאְינוֲ יֹכוֲל לוֲמָר אִלּוּ הֹיִיתִי יוֲדְעָ שִּחֹכֹם יֹכוֲל לֵהָתִּירוֲ לֲא הֹיִיתִי מֵגֹרְשִ, דָּאַנָן סָהַדְי שִּאָף עָל פִּי כְן הֹיֹה מֵגֹרְשִ, שִּאְין אֹדֹם רוֲצּה שִּתִּתֵבָּזּּה אִשִֵתּוֲ בֵּבְית דִּין בִּפֵנְי חֹכֹם לְילְךֵ לֵבְית דִּינוֲ וֵלִשִֵאֲל עָל נִדֵרֹהּ. אּלֹּא מִפֵּנְי נּדּר שִּאְינוֲ צֹרִיךֵ חֹכֹם וּבָעָל יֹכוֲל לֵהֹפְר, אֹסֵרוּ אּת כֶּלֹּן, שִּלֲּא יֲאמָר אִלּוּ הֹיִיתִי יוֲדְעָ שִּהֹיִיתִי יֹכוֲל לֵהֹפְר לֲא הֹיִיתִי מֵגֹרֵשִֹהּ: אֹמָר רָבִּי יוֲסְי בֵרָבִּי יֵהוּדֹה כוּ'. בָּגֵּמֹרֹא מֵפֹרְשִ דָּחַסוּרְי מִחָסֵּרֹא וֵהֹכִי קֹתֹנְי, בָּמּּה דֵּבֹרִים אַמוּרִים, כֵּשִּנֹּדֵרֹה הִיא, אַבֹל נֹדָר הוּא שִּיֵּגֹרֵשִּנֹהּ וֵגְרֵשִֹהּ, יָחַזִיר וֵלֲא חֹיֵשִִינָן לֵקִלֵקוּלֹא : וֵאֹמָר רָבִּי יוֲסְי בֵרָבִּי יֵהוּדֹה מָעַשֵּה נָמִּי בֵּצָיֵּדֹן בֵּאּחֹד שִּאֹמָר לֵאִשִֵתּוֲ קוֲנֹם אִם אְינִי מֵגֹרֵשְִךֵ. כֵּלוֲמָר יְאֹסֵרוּ כֹּל פְּרוֲת שִּבֹּעוֲלֹם עֹלָי אִם אְינִי מֵגֹרֵשְִךֵ, וֵגְרֵשִֹהּ, וֵהִתִּירוּ לוֲ חַכֹמִים שִּיָּחַזִירּנֹּה: מִפֵּנְי תִקּוּן הֹעוֲלֹם. כֵּלוֲמָר שִּלֲּא אֹמֵרוּ חַכֹמִים הָמֵּגֹרְשִ אּת אִשִֵתּוֲ מִשִּוּם נּדּר לֲא יָחַזִירּנֹּה אּלֹּא מִפֵּנְי תִקּוּן הֹעוֲלֹם דֵּחֹיֵשִִינָן לֵקִלֵקוּלֹא, וֵהֹא לֲא שִָיֹּךֵ אּלֹּא מִשִּוּם נּדּר דִּידֹהּ, אַבֹל כֵּשִּנֹּדָר הוּא שִּאְין כֹּאן מִשִּוּם תִּקּוּן הֹעוֲלֹם, הִתִּירוּ לֵהָחַזִיר. וָהַלֹכֹה כֵּרָבִּי יֵהוּדֹה:

רָבִּי יֵהוּדֹה אוֲמְר לֲא יָחַזִיר. שִּמֹּא תִנֹּשְֵא וֵתְלְד וֵיֲאמָר אִלּוּ הֹיִיתִי יוֲדְעָ שִּכְּן, אַפִלּוּ הֹיוּ נוֲתֵנִים לִי מְאֹה מֹנּה לֲא הֹיִיתִי מֵגֹרֵשְִךֵ : וָחַכֹמִים אוֲמֵרִים יָחַזִיר. שִּלֲּא חֹיֵשִִינָן לֵקִלֵקוּלֹא. וּבָגֵּמֹרֹא מֵפֹרְשִ מָאן חַכֹמִים, רָבִּי מְאִיר, דֵּאֹמָר בֹּעִינָן תֵּנָאי כֹּפוּל. וֵהֹכֹא בֵּמָאי עֹסֵקִינָן דֵּלֲא כָּפֵלְיהּ לִתֵנֹאְיהּ , שִּלֲּא אֹמָר לֹהּ הֹוְי יוֲדָעָת שִּמִּשִּוּם אָיֵלוֲנִית אַנִי מוֲצִיאְךֵ, וֵאִם אְין אָתֵּ אָיֵלוֲנִית לֲא יֵהְא גְּט , דֵּהָשִֵתֹּא הֹוְי גְּט אַפִלּוּ אְינֹהּ אָיֵלוֲנִית: וֵהִיא תוֲבָעָת כֵּתֶבֹּתֹהּ. שִּהֹאָיֵלוֲנִית אְין לֹהּ כֵּתֶבֹּה. וֵעָכֵשִֹיו שִּנִּמֵצֹא שִּאְינֹהּ אָיֵלוֲנִית, תּוֲבָעָת כֵּתֶבֹּתֹהּ: שִֵתִיקוּתִיךֵ יֹפּה לִיךֵ מִדִּבּוּרִיךֹ. שִּיֲּאמָר אִלּוּ יֹדָעֵתִּי שִּסּוֲפִי לִתְּן לֹךֵ כֵּתֶבֹּה לֲא גְּרָשִֵתִּיךֵ , נִמֵצֹא גְּט בֹּטְל וּבֹנּיהֹ מָמֵזְרִים:

אְין פּוֲדִין אוֲתוֲ. וֵהוּא שִּרֹגִיל בֵּכֹךֵ. כֵּגוֲן שִּמֹּכָר וֵשִֹנֹה וֵשִִלְּשִ: לוֲקְחָ וּמְבִיא בִכּוּרִים. בֵּכֹל שִֹנֹה צֹרִיךֵ לִקָּח מְהָנֹּכֵרִי בִּכּוּרְי פְרוֲתּיהֹ בֵּדֹמִים, וּמֵבִיאֹן לִירוּשִֹלָיִם : מִפֵּנְי תִקּוּן הֹעוֲלֹם. שִּלֲּא יֵהְא רֹגִיל לִמֵכֲּר קָרֵקָע בֵּאּרּץ יִשֵֵרֹאְל לֵנֹכֵרִים. וֵאִם מֹכָר יִטֵרָח לָחֵזֲר וֵלִפֵדּוֲת:
