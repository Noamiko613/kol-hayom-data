פרק ד

עַשֵֹרֹה יוֲחַסִין עֹלוּ מִבֹּבּל. שִּהִפֵרִישִ עּזֵרֹא כֹּל הָפֵּסוּלִים שִּהֹיוּ בֵּבֹבּל וֵהוֲלִיכֹן עִמּוֲ, כֵּדְי שִּלֲּא יִתֵעֹרֵבוּ עִם הָמֵיֶחֹסִין, מְאָחָר שִּאְין שִֹם בְּית דִּין : חַלֹלְי. כֲּהַנִים שִּנּוֲלֵדוּ מִן הָפֵּסוּלוֲת לָכֵּהֶנֹּה, כֵּגוֲן אָלֵמֹנֹה לֵכֲהְן גֹּדוֲל, גֵּרוּשִֹה חַלֹלֹהּ זוֲנֹה לֵכֲהְן הּדֵיוֲט: חַרוּרְי. עַבֹדִים מֵשִֶחֵרֹרִים: נֵתִינְי. גִּבֵעוֲנִים שִּמֹּלוּ בִּימְי יֵהוֲשִֶעָ, וֵנּאּסֵרוּ לֹבֲא בָקֹּהֹל : שִֵתוּקְי וָאַסוּפְי. מֵפֹרְשִ לֵהוּ וֵאֹזְיל: גְּרְי וָחַרוּרְי מָמֵזְרִין כוּ' כֶּלֹּם מֶתֹּרִין לֹבֲא זּה בֹזּה. דִּקֵהָל גְּרִים לֲא אִקֵּרְי קֹהֹל, וֵלֲא הֶזֵהַרוּ מָמֵזְרִים לֹבֲא בִּקֵהָל גְּרִים. אַבֹל לֵוִיְּי וֵיִשֵֵרֵאְלְי בֵּמָמֵזְרְי לֲא. וּשִֵתוּקְי וָאַסוּפְי סֹפְק מָמֵזְרִים הְם וּמֶתֹּרִין לֵהִתֵעֹרְב בֵּמָמֵזְר וָדָּאי, דֵּאֹמֵרִינָן בֵּקֹהֹל וָדָּאי הוּא דֵּלֲא יֹבֲא, הֹא בֵקֹהֹל סֹפְק יֹבֲא :

שִֵתוּקְי. כֲּל שִּהוּא מָכִּיר אּת אִמּוֲ. שִּקּוֲרְא אָבֹּא, וֵאִמּוֲ מָשִֵתָּקֵתּוֲ: בֵּדוּקְי. שִּבּוֲדֵקִין אּת אִמּוֲ. אִם אֹמֵרֹה לֵכֹשְִר נִבֵעָלֵתִּי, הָוֹּלֹד כֹּשְִר. וָהַלֹכֹה כֵאָבֹּא שִֹאוּל:

כֹּל הֹאַסוּרִים לֹבֲא בָקֹּהֹל. אָף עָל גָּב דִּתֵנִינָן לָהּ בֹּרְישִֹא מָמֵזְרְי וּנֵתִינְי וֵכוּ' מֶתֹּרִים לֹבֲא זּה בֹזּה, הֹא קֹמָשִֵמָע לֹן כֵּגוֲן גְּר עָמּוֲנִי וּמוֲאֹבִי עִם מָמֵזְר שִֵתוּקִי וָאַסוּפִי: רָבִּי יֵהוּדֹה אוֲסְר. מֵפֹרְשִ בָּגֵּמֹרֹא דֵהֹכִי קֹאֹמָר, אַפִלּוּ רָבִּי יֵהוּדֹה שִּאוֲסְר גְּר בֵּמָמֵזּרּת, הֹנְי מִלְּי גְר שִּל שִֵאֹר נֹכֵרִים שִּמֶּתֹּר לֹבֲא בָקֹּהֹל הוּא דֵאֹסוּר בֵּמָמֵזּרּת לֵרָבִּי יֵהוּדֹה, דִּסֵבִירֹא לְיהּ קֵהָל גְּרִים אִקֵּרְי קֹהֹל, אַבֹל גְּר עָמּוֲנִי וּמוֲאֹבִי שִּאֹסוּר לֹבֲא בָּקֹּהֹל מוֲדּה רָבִּי יֵהוּדֹה שִּמֶּתֹּר בֵּמָמֵזּרּת: וָדֹּאֹן בֵּוָדֹּאֹן מֶתֹּר. כֵּגוֲן מָמֵזְרְי בִּנֵתִינְי: וָדֹּאֹן בִּסֵפְקֹן. מָמֵזְרְי וּנֵתִינְי בִּשִֵתוּקְי וָאַסוּפְי: סֵפְקֹן בִּסֵפְקֹן. שִֵתוּקִי בִשִֵתוּקִית, וָאַסוּפִי בָּאַסוּפִית, וָאַסוּפִי בִשִֵתוּקִית. אָף עָל פִּי שִּזּּה סֹפְק וֵזּה סֹפְק, אֹסוּר, שִּמֹּא זּה כֹשְִר וֵזּה פֹסוּל. וָהַלֹכֹה כֵּרָבִּי אֱלִיעּזּר: כּוּתִי. הַוְי סֵפְקָיֵהוּ לֵפִי שִּאְין בֵּקִיאִין בֵּתוֲרָת גִּטִּין וֵקִדּוּשִִין. וֵהֹאִידֵנֹא עַשֵֹאוּם כֵּנֹכֵרִים גֵּמוּרִים לֵכֹל דִּבֵרְיהּם:

אָרֵבָּע אִמֹּהוֲת. שִֵתָּיִם מִצָּד הֹאֹב וּשִֵתָּיִם מִצָּד הֹאְם: שִּהְם שִֵמֲנּה. אָרֵבָּע מִן הֹאֹב. וֵאָרֵבָּע מִן הֹאְם. וֵאְלּוּ הְן, אִמֹּהּ, אְם אִמֹּהּ, אְם אַבִי אִמֹּהּ, אְם אְם אַבִי אִמֹּהּ, אְם אֹבִיהֹ, אְם אְם אֹבִיהֹ, אְם אַבִי אֹבִיהֹ, אְם אְם אַבִי אֹבִיהֹ. בּוֲדֵקִים בֹּהּם שִּלֲּא הֹיֹה בֵאָחָת מְהּן מִן הָפֵּסוּלוֲת : וֵאִם הֹיֵתֹה לֵוִיֹּה וֵיִשֵֵרֵאְלִית. וֵכֲהְן בֹּא לֵכֹנֵסֹהּ: מוֲסִיפִין עַלְיהּן עוֲד אּחֹת. אְם אָחָת בֵּכֹל זוּג וֵזוּג. כֵּגוֲן אִמֹּהּ וֵאְם אִמֹּהּ וֵאִם אְם אִמֹּהּ, וֵכְן כֶּלֹּם. וֵהֹא דֵבֹדֵקִינָן בֹּאִמֹּהוֲת וֵלֲא בֹדֵקִינָן בֹּאֹבוֲת שִּמֹּא יְשִ פֵּסוּל בֵּאֹבִיהֹ אוֲ בָּאַבִי אֹבִיהֹ, מִשִּוּם דֵּדּרּךֵ הֹאַנֹשִִים כֵּשִּמֵּגָנִּים זּה אּת זּה מִתּוֲךֵ קֵטֹטֹה, מֵגָנִּין בֵּיוֲחַסִין, וֵאִם הֹיֹה שִוּם פֵּסוּל מִצָּד הֹאֹבוֲת הֹיֹה יֹדוּעָ. אַבֹל הָנֹּשִִים אְין זוֲ מֵגָנֹּה אּת זוֲ בִּפֵסוּל יִחוּס אּלֹּא בִזֵנוּת, וֵכִי אִיכֹּא בֵהוּ מִלֵּתֹא בִפֵסוּל יוֲחַסִין לְית לְיהּ קֹלֹא. וֵלֲא נּאּמֵרֹה חִיּוּב בֵּדִיקֹה זוֲ אּלֹּא בֵּמִשִֵפֹּחֹה שִּקֹּרֹא עֹלּיהֹ עָרֵעֹר , אַבֹל מִשִֵפֹּחֹה שִּאְין בֹּהּ חַשִֹד אְינוֲ צֹרִיךֵ לִבֵדֲּק, שִּכֹּל הָמִּשִֵפֹּחוֲת בֵּחּזֵקָת כָּשִֵרוּת הְן עוֲמֵדוֲת. וֵהֹאִישִ בִּלֵבָד הוּא שִּצֹּרִיךֵ לִבֵדֲּק כֵּשִּבֹּא לִשֵֹּא אִשִֹּה מִמִּשִֵפֹּחֹה שִּיְּשִ בֹּהּ חַשִֹד, אַבֹל כֲּהּנּת שִּבֹּאת לִנֹּשְֵא אְינֹהּ צֵרִיכֹה לִבֵדֲּק אָחָר הֹאִישִ, שִּלֲּא הֶזֵהַרוּ כֵּשְִרוֲת לִנֹּשְֵא לָפֵּסוּלִים, וֵכֲהּנּת מֶתּּרּת לֵכָתֵּחִלֹּה לֵהִנֹּשְֵא לֵגְר וּלֵחֹלֹל, וֵכֹל שִּכְּן לֵוִיֹּה וֵיִשֵֵרֵאְלִית :

אְין בּוֲדֵקִין מִן הָמִּזֵבְּחָ וּלֵמָעֵלֹה. הִתֵחִיל לִבֵדֲּק בֹּאִמֹּהוֲת וּמֹצֹא שִּאַבִי אֹבִיהֹ שִִמְּשִ עָל גָּבְּי הָמִּזֵבְּחָ , אְין צֹרִיךֵ לִבֵדֲּק אָחָר אְם אַבִי אֹבִיהֹ, דֵּכְיוֹן דִּבֵנֹהּ שִִמְּשִ עָל גָּבְּי הָמִּזֵבְּחָ בֵּיֹדוּעָ שִּמֵּיֶחֹס הֹיֹה: וֵלֲא מִן הָדּוּכֹן וּלֵמָעֵלֹה. אִם מֹצֹא שִּהֹיֹה מֵשִוֲרְר עָל הָדּוּכֹן: וֵלֲא מִן הָסָּנֵהּדֵרִין וּלֵמֹעֵלֹה. וֵדָוֵקֹא מִן הָסָּנֵהּדֵרִין שִּבִּירוּשִֹלָיִם , וָאַפִלּוּ מְאוֲתֹן שִּדֹּנִין דִּינְי מֹמוֲנוֲת בִּלֵבָד, שִּלֲּא הֹיוּ מָעַמִידִין מִן הָסָּנֵהּדֵרִין בִּירוּשִֹלָיִם אּלֹּא כֲהְן וֵלְוִי וֵיִשֵֵרֹאְל מֵיֶחֹס, שִּנּּאֱמָר (בֵּמִדֵבֹּר יא) וֵהִתֵיָצֵּבוּ שִֹם עִמֹּךֵ, בֵּדוֲמִין לֵךֹ בֵּיָחָס וּבֵחֹכֵמֹה: וֵגָבֹּאְי צֵדֹקֹה. כְּיוֹן דֵּמִנֵּצוּ בָּהַדְי אֱנֹשְִי, שִּמֵּמָשִֵכֵּנִין עָל הָצֵּדֹקֹה וָאַפִלּוּ בֵּעּרּב שִָבֹּת, אִי הַוֹה בֵהוּ מִלֵּתֹא, קֹלֹא הַוֹה לֵהוּ: בֹּעָרֵכִי יֵשִֹנֹה שִּל צִפּוֲרִי. בֵּסָנֵהּדֵרֹאוֲת שִּל עִיר שִּשִֵּמֹהּ יֵשִֹנֹה, הָסֵּמוּכֹה לֵצִפּוֲרִי: עֹרֵכִי. סָנֵהּדֵרֹאוֲת שִּהֹיוּ עוֲרֵכִין וּמֵסָדֵּרִין הָיּוּחַסִין : בֵּאִסֵטֵרָטֵיֹא שִּל מּלּךֵ. בָּחַלֶקָּת חֲדּשִ בֵּחֲדּשִ לֹצְאת לֵמִלֵחּמּת בְּית דֹּוִד מִשִֵפֹּחֹה פֵלוֲנִית בֵּחֲדּשִ פֵּלוֲנִי. וֵלֲא הֹיוּ יוֲצֵאִים לָמִּלֵחֹמֹה אּלֹּא מֵיֶחֹסִין , שִּתֵּהְא זֵכוּתֹן וּזֵכוּת אַבוֲתֹן מֵסָיֵּעֹתָן:

בָּת חֹלֹל זֹכֹר פֵּסוּלֹה מִן הָכֵּהֶנֹּה לֵעוֲלֹם. בָּת בֵּנוֲ אוֲ בָּת בּּן בֵּנוֲ, עָד סוֲף כֹּל הָדּוֲרוֲת. אַבֹל בָּת בִּתּוֲ מִיִּשֵֵרֹאְל כֵּשְִרֹה לָכֵּהֶנֹּה, דֵּבָת חֹלֹל עָצֵמֹהּ מִיִּשֵֵרֹאְל כֵּשְִרֹה לָכֵּהֶנֹּה: בָּת גְּר זֹכֹר. אַפִלּוּ מִיִּשֵֵרֵאְלִית, פֵּסוּלֹה לָכֵּהֶנֹּה כֵבָת חֹלֹל זֹכֹר:

רָבִּי אֱלִיעּזּר בּּן יָעַקֲב אוֲמְר כוּ'. מָסֵּקִינָן בָּגֵּמֹרֹא שִּהָבֹּא לִמֹּלְךֵ מוֲרִין לוֲ כֵּרָבִּי אֱלִיעּזּר בּּן יָעַקֲב שִּלֲּא יִשֵֹּא כֲהְן בָּת גְּר וֵגִיּוֲרּת. אַבֹל אִם נֹשֵֹא, הַלֹכֹה כֵּרָבִּי יוֲסְי, וֵאְין מוֲצִיאִין אוֲתֹהּ מִיֹּדוֲ, וֵזָרֵעוֲ מִמּּנֹּה כֹּשְִר:

הֹאוֲמְר בֵּנִי זּה מָמֵזְר הוּא, אְינוֲ נּאֱמֹן. דֵּקֹרוֲב הוּא אּצֵלוֲ, וֵאְין קֹרוֲב כֹּשְִר לֵהֹעִיד : וָאַפִלּוּ שִֵנְיהּם. הָבָּעָל וֵאִשִֵתּוֲ. וֵלֲא מִבָּעֵיֹא כֵּשִּהֹאֹב לֵבָדּוֲ מְעִיד שִּהוּא מָמֵזְר, דֵּלֲא מֵהְימֹן, דֵּלֲא קִים לְיהּ בֵּגָוְּיהּ. אּלֹּא אַפִלּוּ אִמּוֲ דֵּקִים לֹהּ בֵּגָוְּיהּ, אְינֹהּ נּאֱמּנּת: אַפִלּוּ עָל עֶבֹּר שִּבֵּמְעּיהֹ. שִּלֲּא הֹיֹה לֵעוֲלֹם בֵּחּזֵקָת כָּשִֵרוּת: רָבִּי יֵהוּדֹה אוֲמְר נּאֱמֹנִים. טָעַמֹא דֵרָבִּי יֵהוּדֹה דִכֵתִיב יָכִּיר, יָכִּירּנּוּ לָאַחְרִים. מִכֹּאן שִּהֹאֹב נּאֱמֹן לִפֵסֲל אּת בֵּנוֲ, וֵאְין הֹאְם נּאֱמּנּת עָל בֵּנֹהּ . וֵדָוֵקֹא עָל בֵּנוֲ הוּא נּאֱמֹן, אַבֹל לֲא עָל בּּן בֵּנוֲ, שִּאִם הֹיוּ לֵבְן זּה בֹנִים אְינוֲ נּאֱמֹן לֵפֹסֵלֹם . וָהַלֹכֹה כֵּרָבִּי יֵהוּדֹה:

וֵכְן הֹאִשִֹּה שִּנֹּתֵנֹה רֵשִוּת לִשִֵלוּחֹהּ. אִצֵטֵרִיךֵ תָּנֹּא לֵאָשִֵמוֲעִינָן בֵּאֹב שִּעֹשֵֹה שִֹלִיחָ לֵקָדְּשִ אּת בִּתּוֲ וּבֵאִשִֹּה שִּעֹשֵֵתֹה שִֹלִיחָ לֵקָדְּשִ אּת עָצֵמֹהּ. דֵּאִי אָשִֵמוֲעִינָן בֵּאֹב, הַוֹה אַמִינֹא אֹב דֵּקִים לְיהּ בֵּיוֲחַסִין, וּכֵשִּמֹּצֹא מֵיֶחֹס זּה קִדֵּשִֹהּ לוֲ, הוּא דֵאֹמֵרִינָן דֵּבִטְּל אּת הָשִֹּלִיחָ, אַבֹל אִתֵּתֹא דֵלֹא קִים לֹהּ בֵּיוֲחַסִין אָף עָל גָּב דֵּקִדֵּשִֹה אּת עָצֵמֹהּ לֲא סֹמֵכֹה עָל קִדּוּשִּיהֹ וֵלֲא בִטֵּלֹה אּת הָשִֹּלִיחָ, דֵּסֹבֵרֹה דִּלֵמֹא מָשִֵכָּח שִֹלִיחָ אֹדֹם מֵיֶחֹס מִזּּה. וֵאִי אָשִֵמוֲעִינָן בֵּדִידֹהּ, הַוֹה אַמִינֹא מִשִּוּם דֵּאִתֵּתֹא דֹיֵקֹא וּמִנֹּסֵבֹא, כִּי קִדֵּשִֹה עָצֵמֹהּ בִּטֵּלֹה אּת הָשִֹּלִיחָ, אַבֹל אֹב דֵּלֲא קֹפְיד עָל בִּתּוֲ אִם תִּנֹּשְֵא לֵבָעָל כֹּל דֵּהוּ, לֲא בָטֵּלְיהּ לִשִֵלִיחוּת דֵּשִֹלִיחָ, וֵהָאי דֵּקֹדִים וֵקִדֵּשִֹהּ, סֹבָר דִּלֵמֹא לֲא מָשִֵכָּח, צֵרִיכֹא:

אְינוֲ צֹרִיךֵ לֵהֹבִיא רֵאֹיֹה לֲא עָל הֹאִשִֹּה. שִּהִיא מֵיֶחּסּת, שִּכֵּבֹר בֵּדֹקֹהּ כֵּשִּנֵּשֵֹאֹהּ : וֵלֲא עָל הָבֹּנִים. בֵּבֹנִים קֵטָנִּים שִּכֵּרוּכִים אָחַרְי אִמֹּן:

לֲא יִתֵיָחְד אִישִ אּחֹד עִם שִֵתְּי נֹשִִים. מִפֵּנְי שִּדָּעֵתֹּן קָלֹּה וּשִֵתְּיהּן נוֲחוֲת לֵהִתֵפָּתּוֲת: אַבֹל אִשִֹּה אָחָת מִתֵיָחּדּת עִם שִֵנְי אַנֹשִִים. שִּהֹאּחֹד בּוֲשִ מְחַבְרוֲ. וּפֵסָק הָהַלֹכֹה, לֲא תִתֵיָחְד אִשִֹּה אָחָת עִם שִֵנְי אַנֹשִִים, וֵכֹל שִּכְּן אִישִ אּחֹד עִם שִֵתְּי נֹשִִים, אּלֹּא אִם כְּן הֹיוּ שִֵתְּי הָנֹּשִִים צֹרוֲת אוֲ יֵבֹמוֲת, אוֲ אִשִֹּה וּבָת בָּעֵלֹהּ, אוֲ אִשִֹּה וָחַמוֲתֹהּ , אוֲ אִשִֹּה עִם תִּינוֲקּת שִּיּוֲדָעָת טָעָם בִּיאֹה וֵאְינֹהּ מוֲסּרּת עָצֵמֹהּ לֵבִיאֹה, לֵפִי שִּאְלּוּ שֵוֲנֵאוֲת זוֲ אּת זוֲ, וּמִתֵיֹרֵאוֲת זוֲ מִזּוֲ, וֵכְן הִיא מִתֵיֹרְאת מִן הָקֵּטָנֹּה שִּמֹּא תִרֵאּה וֵתָגִּיד. וּמָלֵקִים עָל יִחוּד הָפֵּנוּיֹה, וֵעָל יִחוּד עִם הֹעַרֹיוֲת , חוּץ מְאְשִּת אִישִ שִּאְין מָלֵקִין עֹלּיהֹ, שִּלֲּא לֵהוֲצִיא לָעָז עָל בֹּנּיהֹ. וּמֶתֹּר לֵהִתֵיָחְד עִם הָבֵּהְמֹה וֵעִם הָזֹּכֹר, שִּלֲּא נּחֵשִֵדוּ יִשֵֵרֹאְל עָל הָזֹּכוּר וֵעָל הָבֵּהְמֹה: הִגֵדִּילוּ זוֲ יֵשְִנֹה בִכֵסוּתֹהּ וֵכוּ'. וֵהוּא שִּתִּהֵיּה הָבָּת מִשִֵּתְּים עּשֵֵרְה שִֹנֹה וֵיוֲם אּחֹד, וֵהָבְּן מִשִֵּלֲשִ עּשֵֵרְה שִֹנֹה וֵיוֲם אּחֹד. וּבִזֵמָן שִּהִיא בּוֲשִֹה לָעַמֲד לֵפֹנֹיו עַרֶמֹּה , אַפִלּוּ בֵּפֹחוֲת מִכְּן יֵשְִנִים הוּא בֵּבִגֵדּוֲ וֵהִיא בֵבִגֵדֹּהּ:

רָוֹּק. פֹּנוּי בֵּלֲא אִשִֹּה: לֲא יִלֵמָד סוֲפֵרִים. לֲא יָרֵגִּיל עָצֵמוֲ לִהֵיוֲת מִן הָסּוֲפֵרִים, כֵּלוֲמָר מֵלָמְּד תִּינוֲקוֲת , מִפֵּנְי אִמּוֲתְיהּן שִּל תִּינוֲקוֲת שִּמֵּצוּיוֲת אּצֵלוֲ לֵהֹבִיא אּת בֵּנְיהּן אּל בְּית הָסְּפּר: אָף מִי שִּאְין לוֲ אִשִֹּה. אָף עָל פִּי שִּאְינוֲ פֹנוּי אּלֹּא יּשִ לוֲ אִשִֹּה וֵאְינֹהּ שִֵרוּיֹה עִמּוֲ , לֲא יִלֵמָד סוֲפֵרִים. וֵאְין הַלֹכֹה כֵּרָבִּי אֱלִיעּזּר:

וָחַכֹמִים מָתִּירִין. וָהַלֹכֹה כָּחַכֹמִים. שִּלֲּא נּחֵשִֵדוּ יִשֵֵרֹאְל עָל הָזֹּכוּר : כֹּל שִּאֶמֹּנוּתוֲ בְּין הָנֹּשִִים. שִּמֵּלּאכּת אֶמֹּנוּתוֲ נָעַשְֵית לֵנֹשִִים, וֵהָנֹּשִִים צֵרִיכוֲת לוֲ: לֲא יִתֵיָחְד עִם הָנֹּשִִים. וָאַפִלּוּ עִם הָרֵבְּה נֹשִִים. לֵפִי שִּלִּבֹּן גָּס בּוֲ וּמֵחָפּוֲת עֹלֹיו. וֵאִלּוּ אֱנֹשִ אָחַרִינֹא עִם שִֵתְּי נֹשִִים תֵּנָן דֵּלֲא, שִֹלֲשִ וֵאָרֵבָּע שִָפִּיר דֹּמְי. וֵרָמֵבָּ"ם פְּרְשִ דֵּלֲא שִֹרִינָן אַפִלּוּ מִשִּוּם כֵּדְי חָיֹּיו לֵהִתֵיָחְד עִם הָנֹּשִִים, הוֲאִיל וּפָרֵנֹסֹתוֲ מְהּן: חָמֹּר גָּמֹּל וֵסָפֹּן. כֹּל אְלּוּ אֶמֹּנוּת לִסֵטוּת, כֵּשִּלֹּנִין בָּדֵּרֹכִים נִכֵנֹסִים וֵלוֲקֵטִים עְצִים וּפְרוֲת מִן הָכֵּרֹמִים. וֵעוֲד שִּנִּשֵֵכֹּרִים לִבֵנְי אֹדֹם וּמָעַבִירִין עָל תֵּנֹאֹם: רוֲעּה. שִּמָּעַבִיר הָבֵּהְמוֲת לִרֵעוֲת בִּשֵֵדְה אַחְרִים: חּנֵוֹנִי. מֵלֶמֹּד בֵּאוֲנֹאֹה, לֵהֹטִיל מָיִם בֵּיָיִן, וּצֵרוֲרוֲת בֵּחִטִּין. שִּחָיֹּב אֹדֹם לֵלָמְּד אּת בֵּנוֲ אֶמֹּנוּת נֵקִיֹּה:
