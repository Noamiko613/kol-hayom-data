פרק ד

פרק ד - משנה א

נְתָנוּהוּ. לְפִי שֶׁדֶּרֶךְ הוּא כְּשֶׁמּוֹלִיכִין הָאָדָם מוֹלִיכִין אוֹתוֹ אֶל עִיר, הִלְכָּךְ קָתָנֵי הוֹלִיכוּהוּ גַּבֵּי עִיר. אֶלָּא שֶׁאִם אֵרַע לִפְעָמִים שֶׁלֹּא יוּכְלוּ לָבֹא אֶל עִיר אוֹ מִפְּנֵי סִבָּה וּמִקְרֶה שֶׁיִּתְּנוּהוּ לְמָקוֹם אַחֵר הַמִּשְׁתַּמֵּר כְּמוֹ בְּדִיר אוֹ סַהַר, לְכָךְ תָּנֵי גַּבַּיְהוּ נְתָנוּהוּ: וְאֵין הָכִי נַמִּי דְּבָעִיר נַמִּי גָּזְרוּ דְּהָא וַדַּאי מַשְׁמַע דְּאַכָּל מַה שֶּׁהֻזְכַּר בְּדִבְרֵי רַבָּן גַּמְלִיאֵל וְרַבִּי אֶלְעָזָר בֶּן עֲזַרְיָה קָמִפְלְגֵי, וְנָקְטֵי חֲדָא מִתַּרְתֵּי בָּבֵי דְּתַנָּא קַמָּא, וְהוּא הַדִּין לְאִידָךְ. תּוֹסְפוֹת יוֹם טוֹב: לְפִי שֶׁאֵינוֹ דּוֹמֶה לְדִיר וְסַהַר אֶלָּא כְּשֶׁהִיא קְבוּעָה בַּנָּמָל כְּמוֹ שֶׁכָּתַב הָרַמְבַּ"ם. אִי נַמִּי כְּפֵרוּשׁ רַשִׁ"י דִּכְשֶׁהַסְּפִינָה מְהַלֶּכֶת כְּשֶׁעוֹקֵר רַגְלוֹ קֹדֶם שֶׁיַּנִּיחֵהוּ מוֹצִיאָתוֹ הַסְּפִינָה מֵאַרְבַּע אַמּוֹת שֶׁלּוֹ וְנִכְנָס בְּאַרְבַּע אַמּוֹת אֲחֵרוֹת עַל כָּרְחוֹ, וְהָוֵי כְּמוֹ שֶׁהוֹצִיאוּהוּ לִסְטִים וְנוֹתְנִין אוֹתוֹ תָדִיר בְּאַרְבַּע אַמּוֹת אֲחֵרוֹת, שֶׁנּוֹתְנִין לוֹ אַרְבַּע אַמּוֹת, וְכֵן לְעוֹלָם: כִּדְאִיתָא בַּגְּמָרָא דִּלְמָא עָמְדָה פִּתְאוֹם וְלָאו אַדַּעְתַּיְהוּ. וּלְטַעֲמָא דְּשָׁבְתוּ בַּאֲוִיר כוּ' נִרְאֶה דְּהוֹאִיל וְיָצְאוּ חוּץ לַתְּחוּם רָצוּ לְהַחְמִיר, שֶׁלֹּא תִטְעֶה וְלֹא תְחַלֵּק בֵּין יוֹצֵא לְחוּץ לַתְּחוּם וְשָׁבַת בַּמְּחִצּוֹת הָהֵן מִבְּעוֹד יוֹם כְּמוֹ בִּסְפִינָה לְהֵיכָא דְּלֹא שָׁבַת כְּמוֹ בְּדִיר וְסַהַר וְתַתִּיר הַכֹּל:

פרק ד - משנה ג

גְּמָרָא. וְטַעֲמָא שֶׁמָּא יַרְגִּישׁוּ בָּהֶם הָאוֹיְבִים שֶׁיָּצְאוּ וְיִרְדְּפוּ אַחֲרֵיהֶן וְאֵין כְּלֵי זַיִן בְּיָדָם לְהַצִּיל. וְלִמְקוֹמָן הַיְנוּ לַמָּקוֹם שֶׁנָּתְנוּ לָהֶם חֲכָמִים, שֶׁהוּא עַד אַלְפַּיִם אַמָּה. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב:

פרק ד - משנה ד

וְהַמְּדִידָה הוּא מַהֲלַךְ אַלְפַּיִם פְּסִיעוֹת בֵּינוֹנִית שֶׁלֹּא יַרְחִיק רַגְלָיו, וְזֶה שִׁעוּר אַמָּה אַחַת בֵּין רֶגֶל לְרֶגֶל. הָרִי"כ: יִכָּנֵס. כֵּיוָן שֶׁאִלּוּ הָיָה יוֹדֵעַ שֶׁהָעִיר תּוֹךְ תְּחוּמוֹ לֹא הָיָה קוֹנֶה שְׁבִיתָה אֶלָּא עִם בְּנֵי הָעִיר, הֲרֵי הוּא כְּמִי שֶׁקָּנָה שְׁבִיתָה עִמָּהֶם. בֵּית יוֹסֵף:

פרק ד - משנה ה

וּמִטַּעֲמָא אַחֲרִינָא, הוֹאִיל וְנֵעוֹר קוֹנֶה יָשֵׁן נַמִּי קוֹנֶה, כִּדְפֵרֵשׁ רַשִׁ"י: הַיְנוּ בְּלֹא רֹאשׁוֹ, וּבְפִשּׁוּט יָדַיִם (וְרַגְלַיִם) נִכְנָס הָרֹאשׁ. תּוֹסָפוֹת: וְהוּא בְאֶמְצָעָן. נָטַר עַד הָכָא וְהוּא הַדִּין לְבִתְרֵי בָּבֵי דִּבְמִשְׁנָה קַמַּיְתָא: וּמוֹדֶה רַבִּי יְהוּדָה. וְהַיְנוּ דְּרַבָּנָן סָבְרֵי אַרְבַּע לְכָל צְדָדָיו הַיְנוּ שְׁמֹנֶה עַל שְׁמֹנֶה, וְרַבִּי יְהוּדָה סָבַר לְחַד רוּחַ וְתוּ לֹא:

פרק ד - משנה ו

דַּחֲפָצָיו כְּרַגְלָיו כְּדִתְנַן הַבְּהֵמָה וְהַכֵּלִים כְּרַגְלֵי הַבְּעָלִים:

פרק ד - משנה ז

וְחָשְׁכָה לוֹ הַיְנוּ אִם יִרְצֶה לֵילֵךְ לְבֵיתוֹ, אֲבָל לְעִקָּרוֹ שֶׁל אִילָן אָזֵיל. וְאִיכָּא דְּאָמְרֵי בַּגְּמָרָא דְּחָשְׁכָה לוֹ אִם יֵלֵךְ מְעַט מְעַט אֲבָל כִּי רָהֵיט מָטֵי לְהָתָם, וְכֵן הַסְכָּמַת הַפּוֹסְקִים: וְהָא דִּבְפֶרֶק ג' מִשְׁנָה ד' אֲפִלּוּ לְמַאן דְּאָמַר דִּסְפֵק עֵרוּב אֵינוֹ עֵרוּב מִכָּל מָקוֹם יֵשׁ לוֹ אַלְפַּיִם שֶׁמִּבֵּיתוֹ כְּמוֹ שֶׁכָּתַב הָרַ"ב שָׁם, וְלֹא אָמְרִינַן שֶׁלֹּא יָזוּז מִמְּקוֹמוֹ, הָתָם כְּעוֹמֵד בְּעִירוֹ לֹא עָקַר דַּעְתּוֹ מֵעִירוֹ, וְדַעְתּוֹ שֶׁאִם לֹא יִקְנֶה לוֹ שָׁם שְׁבִיתָה שֶׁיִּהְיֶה לוֹ תְּחוּם בֵּיתוֹ. תּוֹסָפוֹת: אַלְפַּיִם אַמָּה. דְּהָא עִקַּר עֵרוּב בָּרֶגֶל כִּדְמַסְקִינַן בְּמִשְׁנָה ט', וְהָכָא דְּדָחוּק הוּא דְּחָשְׁכָה לוֹ שָׁרִינַן לֵיהּ בְּאוֹמֵר שְׁבִיתָתִי כוּ':

פרק ד - משנה ח

זָכָה לוֹ. וְקָא מַשְׁמַע לָן שֶׁאֵין נִקְרָא זֶה שְׁבִיתָה בְּטָעוּת כִּדְאָמַר רַבִּי יְהוּדָה בְּמִשְׁנָה ד', דְּלֹא דָּמִי מִי שֶׁהוּא סָמוּךְ לָעִיר וְלֹא יָדַע, דְּנִכָּר לַכֹּל וּמְפֻרְסָם שֶׁשְּׁבִיתָתוֹ בְּטָעוּת, מַה שֶּׁאֵינוֹ נִכָּר דְּאִלּוּ הָיָה יוֹדֵעַ הָאִילָן אוֹ הַגָּדֵר לֹא הָיָה אוֹמֵר תְּהֵא שְׁבִיתָתִי בִּמְקוֹמִי. הָרִי"כ: וְלֹא קַשְׁיָא אַמַּה שֶּׁכָּתַב בְּפֶרֶק ג' מִשְׁנָה ה' דִּתְחוּמִין דְּרַבָּנָן. שֶׁהֲרֵי כָּתַב הַטּוּר סִימָן שצ"ז דִּקְרָאֵי אַסְמַכְתָּא בְּעָלְמָא:

פרק ד - משנה ט

כְּלוֹמַר וּפְלוּגְתַּיְהוּ בְּאוֹמֵר שְׁבִיתָתִי בִּמְקוֹמִי עָנִי, שֶׁיֵּשׁ לוֹ פַּת: הָכִי קָאָמַר, אַף עַל פִּי שֶׁיֵּשׁ עִמּוֹ פַּת הָוֵי כְּמִי שֶׁאֵין עִמּוֹ פַּת, הוֹאִיל וּבַדֶּרֶךְ הוּא וְחָשְׁכָה לוֹ, כִּדְמַסִּיק שֶׁהָיָה יָכוֹל לָרוּץ. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: הֶעָנִי כוּ'. אֲתָאָן לְרַבִּי מֵאִיר, וּכְלוֹמַר בְּזוֹ אָמְרוּ הֶעָנִי כוּ', אֲבָל עִקַּר עֵרוּב בְּפַת גְּמִירָא וְטַעֲמֵיהּ דְּכֵיוָן שֶׁמֵּבִיא פַּת גִּלָּה דַּעְתּוֹ וְנִכָּר שֶׁרְצוֹנוֹ לִשְׁבֹּת שָׁם. אֲבָל כְּשֶׁמְּעָרֵב בְּרַגְלָיו אָמְרוּ אֱינָשֵׁי אִם תַּלְמִיד חָכָם הוּא שְׁמַעְתֵּיהּ מְשַׁכְתֵּיהּ, וְאִם עַם הָאָרֶץ חֲמוֹרוֹ אִתְאַבִיד לֵיהּ וְהוֹלֵךְ לְבַקְּרוֹ שָׁם. וְרַבִּי יְהוּדָה סָבַר כֵּיוָן שֶׁטּוֹרֵחַ הוּא בְּעַצְמוֹ יֵשׁ הֶכֵּירָא יוֹתֵר מִשֶּׁשּׁוֹלֵחַ שָׁם וּבְיָדוֹ פַּת:

פרק ד - משנה י

וְחָזַר מִלָּלֶכֶת הַיּוֹם, אֲבָל דַּעְתּוֹ לֵילֵךְ לְמָחָר: כְּלוֹמַר דְּלֹא אָמְרִינַן נִמְלָךְ הוּא שֶׁלֹּא לִקְנוֹת שְׁבִיתָה, אֶלָּא אָמְרִינַן דִּבְדַעְתּוֹ עוֹמֵד. הָרַמְבַּ"ם:

פרק ד - משנה יא

וְדִכְוָתֵיהּ שָׁלֹשׁ לֹא יִכָּנֵס, אִם עוֹמֵד בָּאַמָּה הַשְּׁלִישִׁית. גְּמָרָא: וְזֶה לְשׁוֹן הָרִי"כ, לֹא יִכָּנֵס, אִם אָמַר שְׁבִיתָתִי בָּעִיר לֹא אָמַר כְּלוּם שֶׁהֲרֵי הוּא חוּץ שֶׁסִּיֵּם יוֹתֵר מֵאַלְפַּיִם אַמָּה, וּבִמְקוֹמוֹ נַמִּי לֹא קָנֵי שְׁבִיתָה. וְזוֹ הִיא כַּוָּנַת הָרַ"ב בַּמֶּה שֶׁכָּתַב שֶׁהֲרֵי גִּלָּה דַּעְתּוֹ, רְצוֹנוֹ לוֹמַר שֶׁאָמַר. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: אֶלָּא כְּלוֹמַר לְפִי סְבָרַת הָאָדָם שֶׁיּוּכַל לְשַׁעֵר עַד חָמֵשׁ עֶשְׂרֵה אַמּוֹת וְלֹא יוֹתֵר, אֲבָל אֶפְשָׁר לִפְחוֹת וּלְשַׁעֵר, שֶׁאִי אֶפְשָׁר שֶׁיִּטְעוּ כָּל כָּךְ: לֹא רָאָה דִּבְרֵי רַשִׁ"י וְהָרִי"כ שֶׁכָּתְבוּ וְזֶה לְשׁוֹנָם, אָמְרִינַן לְקַמָּן בְּפֶרֶק ה' אֵין מוֹדְדִין אֶלָּא בְּחֶבֶל שֶׁל חֲמִשִּׁים אַמָּה . וְכָל חֶבֶל וְחֶבֶל מִתְמַעֵט שְׁתֵּי אֲחִיזוֹת, זֶה תּוֹפֵס מִכָּאן וְזֶה מִכָּאן. וְהָאֲחִיזָה טֶפַח וַחֲצִי אֶצְבַּע. הֲרֵי חֲמֵשׁ עֶשְׂרֵה אַמּוֹת: כְּדִמְתַרְגֵּם וּמַדֹּתֶם חוּץ לָעִיר, וּמְשַׁחְתּוּן מִבָּרָא לְקַרְתָּא:
