פרק ט - משנה א

דַּהֲוֵי רְשׁוּת הַיָּחִיד כִּדְפֵּרֵשׁ רֵישׁ פֶּרֶק י"א דְּשַׁבָּת: גָּבוֹהַּ עֲשָׂרָה אוֹ נָמוֹךְ. אַף עַל גַּב דְּהַיְנוּ הָךְ, דְּאִם הָאֶחָד גָּבוֹהַּ עֲשָׂרָה בְּהֶכְרֵחַ הַשֵּׁנִי נָמוּךְ מִמֶּנּוּ עֲשָׂרָה, נָקַט הָכִי מִשּׁוּם דְּלִפְעָמִים אֵין בַּיִת אֶחָד גָּבוֹהַּ יוֹתֵר מֵחֲבֵרוֹ, אֶלָּא שֶׁהָאֶחָד נִבְנָה בָּעֵמֶק שֶׁהַקַּרְקַע נָמוּךְ, וְאָז לֹא שַׁיָּךְ לוֹמַר שֶׁזֶּה גָּבוֹהַּ אֶלָּא שֶׁזֶּה נָמוּךְ: וּבְלֹא הֻקַּף לְדִירָה, וּכְרַבִּי עֲקִיבָא בְּפֶרֶק ב' מִשְׁנָה ה' דַּהֲלָכָה כְּמוֹתוֹ. וְאִי הֻקַּף לְדִירָה אֲפִלּוּ בְּיָתֵר נַמִּי, טוּר. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: וּבִלְבַד שֶׁלֹּא יִכָּנֵס בִּכְלִי זֶה לְבַיִת מִן הַבָּתִּים, אֶלָּא אִם כֵּן עֵרְבוּ. הָרַמְבַּ"ם: וְאַף עַל גַּב דְּהַשְׁתָּא שְׁכִיחָא מָאנֵי דְּבָתִּים בֶּחָצֵר, אֲפִלּוּ הָכִי כֵּלִים שֶׁשָּׁבְתוּ לְתוֹכָן שָׁרֵי לְטַלְטוֹלֵי, וְלֹא גָּזְרִינַן דִּלְמָא אָתֵי לַאֲפוֹקֵי מָאנֵי דְּבָתִּים לֶחָצֵר:

פרק ט - משנה ב

גַּג גָּדוֹל. פֵּרֵשׁ הָרַמְבַּ"ם כְּשֶׁיֵּשׁ כֹּתֶל מַקִּיף בַּגַּגּוֹת, וְכֵן נִרְאֶה מִדִּבְרֵי הָרַ"ב (אֶלָּא שֶׁלֹּא פֵּרֵשׁ בְּהֶדְיָא לְפִי שֶׁאֵין צָרִיךְ כְּתָלִים מַמָּשׁ אֶלָּא כְּשֶׁאֵין הַגַּגִּין בּוֹלְטוֹת מֵהַמְּחִצּוֹת אָמְרִינַן גּוּד אַסִּיק), וְכִדְרַב בַּגְּמָרָא. וּמְפָרֵשׁ בַּגְּמָרָא דִּלְהָכִי תָּנֵי גַּג וְחָצֵר, דְּגַג דּוּמְיָא דְּחָצֵר, מֶה חָצֵר מִנְכְּרָא מְחִצּוֹתֶיהָ, אַף גַּג מִינְכְּרָא מְחִצּוֹתֶיהָ. דְּאִי לֹא, לֹא אָמְרִינַן גּוּד אַסִּיק, וְגָדוֹל נַמִּי אָסוּר: דְּאִלּוּ כֵּלִים שֶׁשָּׁבְתוּ בְּתוֹכָן מֻתָּרִין בִּשְׁתֵּיהֶן כִּדְקָיְמָא לָן לְרַבִּי שִׁמְעוֹן: כְּלוֹמַר בִּשְׁבִיל אוֹתָן גִּפּוּפִין שָׁרֵי בְּכָל הַגַּג וּכְלִישָׁנֵיהּ גַּבֵּי חָצֵר: גְּדוֹלָה מֻתֶּרֶת. אִם עֵרְבָה לְעַצְמָהּ. רַשִׁ"י: לִרְשׁוּת הָרַבִּים. אֲפִלּוּ בְּשַׁבָּת. טוּר סִימָן שע"ד: מִתּוֹכָהּ לִרְשׁוּת הָרַבִּים. וְכָתַב רַשִׁ"י וְהוּא הַדִּין מִתּוֹכָהּ לִרְשׁוּת הַיָּחִיד, וְקָשֶׁה דְהֲווּ לֵיהּ לְמֵימַר בְּלִישְׁנֵיהּ דְּרַבִּי אֱלִיעֶזֶר מִתּוֹכָהּ לִרְשׁוּת הַיָּחִיד. וּלְמַסְקָנָא דִּגְמָרָא נִיחָא, דִּבְתוֹכָהּ לְכֻלֵּי עָלְמָא כַּרְמְלִית הוּא. כִּי פְלִיגֵי בִּמְקוֹם מְחִצָּה, דְּרַבִּי אֱלִיעֶזֶר סָבַר צִדֵּי רְשׁוּת הָרַבִּים כִּרְשׁוּת הָרַבִּים, וּמְקוֹם מְחִצָּה זוֹ צִדֵּי רְשׁוּת הָרַבִּים הִיא, וְרַבָּנָן סָבְרֵי דְּצִדֵּי רְשׁוּת הָרַבִּים לָאו כִּרְשׁוּת הָרַבִּים. וּמִתּוֹכָהּ דְּרַבִּי אֱלִיעֶזֶר מִתּוֹךְ מְקוֹם מְחִצָּתָהּ קָאָמַר, וּמִשּׁוּם דְּרַבָּנָן נָקְטֵי מִתּוֹכָהּ נָקַט אִיהוּ נַמִּי מִתּוֹכָהּ. וְרַבָּנָן הָכִי קָאָמְרֵי מִי לֹא קָא מוֹדֵית לָן דְּמִתּוֹכָהּ לִרְשׁוּת הָרַבִּים פָּטוּר דְּכַרְמְלִית הִיא, צִדֵּי נַמִּי לֹא שְׁנָא. וְרַבִּי אֱלִיעֶזֶר, הָתָם לֹא קָא דָּרְסוּ לַהּ רַבִּים, וְהָכָא דָּרְסוּ לַהּ רַבִּים. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב:

פרק ט - משנה ג

אַף עַל גַּב דְּלִפְנִים בִּמְקוֹם שֶׁכָּלְתָה הַפִּרְצָה מֵהַכְּתָלִים מִשָּׁם וָאֵילָךְ יֵשׁ תִּקְרָה, לֹא אָמְרִינַן יוֹרֵד וְסוֹתֵם הוֹאִיל וּבָאֲלַכְסוֹן הוּא. תּוֹסָפוֹת. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: כְּלוֹמַר דִּלְעֵיל בִּקְטַנָּה שֶׁנִּפְרֶצֶת לִגְדוֹלָה אָמְרִינַן דְּהֻתְּרָה כוּ', וְעַל זֶה קָאָמַר וְלֹא דָּמִי לְדִלְעֵיל. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב:

פרק ט - משנה ד

וַחֲכָמִים אוֹסְרִין. הוֹאִיל וּמְפֻלָּשׁ לֹא אָמְרִינַן פִּי תִּקְרָה יוֹרֵד וְסוֹתֵם. וּבְמַתְנִיתִין דִּלְעֵיל הֲוָה מָצֵי לְאוֹקְמַהּ בִּשְׁתֵּי רוּחוֹת זוֹ כְּנֶגֶד זוֹ וּכְרַבָּנָן, כָּתְבוּ הַתּוֹסָפוֹת דְּנִיחָא לֵיהּ לְאוֹקְמֵיהּ בְּכָל עִנְיָן: וְעוֹד אָמַר רַבִּי יְהוּדָה. כְּלוֹמַר וְעוֹד זוֹ מַתִּיר אַף עַל גַּב דְּלֵיכָּא פִּי תִּקְרָה. גְּמָרָא. מִכְּלַל הָא כָּל שֶׁכֵּן הַךְ לְעֵיל דְּאִיכָּא נַמִּי פִּי תִּקְרָה, אֶלָּא דְּאָמַר רַבִּי יְהוּדָה לְהָא דִּלְעֵיל דְּקָסָבַר דְּרַבָּנָן יוֹדוּ לוֹ בְּהַהוּא: מְעָרְבִין. מִשְׁתַּתְּפִין. הָרִי"כ:
