פרק ח

פרק ח - משנה א

מֵנִיחַ. וְאִי מִשֶּׁלּוֹ כִּדְמַשְׁמַע צָרִיךְ לְזַכּוֹת עַל יְדֵי אַחֵר כְּדִתְנַן פֶּרֶק ז' מִשְׁנָה ו': וּלְבֵית הַמִּשְׁתֶּה הַיְנוּ מִשְׁתֶּה דְּחָתָן וְכַלָּה: לִישְׁנָא דִּגְמָרָא. וּמִדְּקָאֲמַר אֵין מְעָרְבִין שְׁמַע מִנָּהּ דְּקָפְדִינַן שֶׁלֹּא לְעָרֵב. וְהַטַּעַם דִּכְשֶׁמְּעָרֵב חַיָּב לְבָרֵךְ כוּ' וְהִלְכָּךְ אֵין מְעָרְבִין כְּדֵי שֶׁלֹּא לְבָרֵךְ בְּרָכָה לְבַטָּלָה. וּלְפֵרוּשׁ רַשִׁ"י לֹא הִתִּירוּ חֲכָמִים לָצֵאת חוּץ לַתְּחוּם עַל יְדֵי עֵרוּב אֶלָּא לִדְבַר מִצְוָה. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: הָכִי מִפָּרְשָׁה בַּגְּמָרָא, מִשּׁוּם דְּקָיְמָא לָן יֵשׁ בְּרֵרָה בִּתְחוּמִין לְפִי שֶׁהֵן דְּרַבָּנָן וּכִדְאִיתָא בְּמִשְׁנָה ה' פֶּרֶק ג', וְהִלְכָּךְ כְּשֶׁהוֹדִיעוּהוּ כְּשֶׁקִּבֵּל מִשֶּׁתֶּחְשַׁךְ אָמְרִינַן הֻבְרַר לְמַפְרֵעַ, אֲבָל אִם לֹא הוֹדִיעוּהוּ מַאי בְּרֵרָה אִיכָּא, הָא לֹא יָדַע דְּאִיכָּא עֵרוּב דְּדַעְתֵּיהּ עִלָּוֵיהּ:

פרק ח - משנה ב

וְאִם תֹּאמַר דִּבְמִשְׁנָה ב' פֶּרֶק ט"ז דְּשַׁבָּת מַשְׁמַע בְּהֶדְיָא דְּחַיָּב בְּשָׁלֹשׁ סְעוּדוֹת, וּמְסַיֵּעַ לְהַפּוֹסְקִים דְּיוֹצְאִים בַּסְּעוּדָה הַשְּׁלִישִׁית בְּמִינֵי תַּרְגִּימָא. וּלְהַסּוֹבְרִים דְּבָעִינַן נַמִּי פַּת יֵשׁ לוֹמַר מִכָּל מָקוֹם אֲכִילַת אָדָם דְּבַר יוֹם בְּיוֹמוֹ אֵינָהּ אֶלָּא שְׁתֵּי סְעוּדוֹת, וְלֹא אָמְרוּ לְהַחְמִיר אֶלָּא לְהָקֵל כִּדְאִיתָא לְקַמָּן: וְאַף עַל גַּב דְּבִשְׁאָר יוֹמֵי כִּי אָכֵיל בַּיּוֹם סְעוּדַת שַׁחֲרִית אוֹכֵל לָעֶרֶב סְעוּדַת עַרְבִית. בְּשַׁבָּת שָׁאנֵי שֶׁצָּרִיךְ שֶׁיֹּאכַל הַשְּׁלִישִׁית מִבְּעוֹד יוֹם, וּלְפִיכָךְ מְמַעֵט בְּשֶׁל שַׁחֲרִית. וְהָרִי"כ מְפָרֵשׁ לְפִי שֶׁבְּשַׁבָּת מְמַלֵּא בִּטְנוֹ מִתַּבְשִׁיל וּמִינֵי מְתִיקָה וְיַמְעִיט הָאֲכִילָה: מִכִּכָּר. אַף עַל פִּי שֶׁעֵרוּבֵי תְּחוּמִין בְּיַיִן כִּדְקָתָנֵי מַנִּיחַ אֶת הֶחָבִית וְסִיֵּם בְּפַת, וְנִרְאֶה לִי דְּלֹא תֵּימָא בְּפַת שֶׁהוּא חָשׁוּב לֹא הֻצְרְכוּ שִׁעוּר כָּל כָּךְ, קָא מַשְׁמַע לָן. וְשִׁעוּר שְׁאָר דְּבָרִים אִיתָא בַּגְּמָרָא בְּפֶרֶק בַּכֹּל מְעָרְבִין:

פרק ח - משנה ג

וַהֲדַר הִתְנָה, בְּדִבּוּר כָּל שֶׁגָּבוֹהַּ כוּ', שֶׁלֹּא תְּהֵא הַמִּרְפֶּסֶת גָּבוֹהַּ עֲשָׂרָה מִן הָעַמּוּד. וִיפָרֵשׁ בְּאֶחָד מִשְׁתֵּי פָנִים. אוֹ שֶׁרְצוֹנוֹ לוֹמַר שֶׁהָעֲלִיּוֹת הַפְּתוּחוֹת לַמִּרְפֶּסֶת אֵינָן גְּבוֹהִים עֲשָׂרָה וְהָוֵי שַׁרְיוּתָא דִּלְאוֹתוֹ גֹּבַהּ לְכָל בְּנֵי הָעֲלִיּוֹת, אוֹ שֶׁזֶּה שֶׁכָּתַב שֶׁפִּתְחֵי הָעֲלִיּוֹת כוּ' לָאו לְמֵימְרָא שׁוּם שַׁרְיוּתָא לַעֲלִיּוֹת אֶלָּא פֵּרוּשָׁא קָא מְפָרֵשׁ לַמִּרְפֶּסֶת מַה הִיא, וּלְעוֹלָם הָעֲלִיּוֹת גְּבוֹהוֹת הַרְבֵּה יוֹתֵר מֵעֲשָׂרָה טְפָחִים מִן הָעַמּוּד כוּ', וְשַׁרְיוּתָא דְּמַתְנִיתִין לִבְנֵי מִרְפֶּסֶת עַצְמָן רְצוֹנוֹ לוֹמַר לְאוֹתָן הַדָּרִים בַּמִּרְפֶּסֶת, כְּגוֹן שֶׁהַמִּרְפֶּסֶת עַצְמָהּ מֻקֶּפֶת בְּבִנְיָן וְדָרִין בָּהּ. וּפֵרוּשׁ הַשֵּׁנִי עוֹלֶה יוֹתֵר עַל פִּי שִׁיטַת הַגְּמָרָא: שֶׁשָּׁכְחוּ. מֵהַלָּשׁוֹן מַשְׁמַע שֶׁעָסְקוּ בָּעֵרוּב, שֶׁעֵרְבוּ כָּל אֶחָד בִּפְנֵי עַצְמוֹ, אֶלָּא שֶׁשָּׁכְחוּ לְעָרֵב בְּיַחַד. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: לֶחָצֵר. וְהוּא הַדִּין אִם הַמִּרְפֶּסֶת גָּבוֹהַּ עֲשָׂרָה טְפָחִים מֵהָעַמּוּד, דְּהַשְׁתָּא נַמִּי לֹא נִיחָא תַּשְׁמִישְׁתָּם דִּבְנֵי מִרְפֶּסֶת: בַּגְּמָרָא. וְטַעֲמָא כֵּיוָן דְּלִשְׁנֵיהֶם תַּשְׁמִישָׁן קָשֶׁה וְאֵין לָזֶה נַחַת מִזֶּה הִלְכָּךְ שְׁנֵיהֶם אֲסוּרִים. וְהָא דַּאֲסוּרִים פֵּרֵשׁ הָרַמְבַּ"ם דְּהַיְנוּ לְעִנְיַן כֵּלִים שֶׁשָּׁבְתוּ בַּבַּיִת, דִּלְעִנְיַן כֵּלִים שֶׁשָּׁבְתוּ בֶּחָצֵר אֲפִלּוּ לֹא עֵרְבוּ כְּלָל מֻתָּר לְטַלְטֵל, וַהֲרֵי הֵם בִּכְלַל גַּגּוֹת וַחֲצֵרוֹת דְּרֵישׁ פִּרְקִין דִּלְקַמָּן, דְּהָא הַכֹּל רְשׁוּת אֶחָד: דְּאִי לָאו הָכִי אֶלָּא בְּבוֹר שֶׁדּוֹלִין מַיִם אֶפְשָׁר דְּחָסַר בְּשַׁבָּת. וּבַגְּמָרָא צְרִיכָא, דְּאִי אַשְׁמְעִינַן סֶלַע דְּלֵיכָּא לְמִגְזַר, אֲבָל בּוֹר לִגְזוּר זִמְנִין דְּמַלְּיֵהּ פֵּרוֹת הָרְאוּיִן לְטַלְטֵל בְּשַׁבָּת, צְרִיכָא. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: בַּמֶּה דְבָרִים אֲמוּרִים. אָרֵישָׁא נַמִּי קָאֵי כָּל שֶׁגָּבוֹהַּ כוּ', דְּמַאי שְׁנָא. וּקְצָת קָשֶׁה לָמָּה לֹא תָּנֵי כָּל שֶׁגָּבוֹהַּ עֲשָׂרָה טְפָחִים כְּגוֹן חֻלְיַת כוּ':

פרק ח - משנה ד

בְּבֵית שַׁעַר. כָּל בֵּית שַׁעַר בֵּין דְּיָחִיד בֵּין דְּרַבִּים. בֵּית יוֹסֵף: וּמִרְפֶּסֶת. צְרִיכִין לְחַלֵּק בֵּין מִרְפֶּסֶת דְּהָכָא לְדִלְעֵיל לְפִי פֵּרוּשׁ הַשֵּׁנִי שָׁם. דַּהֲרֵי לְעֵיל בָּעִינַן שֶׁבְּנֵי הַמִּרְפֶּסֶת עֵרְבוּ. אֶלָּא דִּלְעֵיל מַיְרֵי בַּעֲשׂוּיָה לְדִירָה, וְהָכָא שֶׁאֵינָהּ עֲשׂוּיָה לְדִירָה אֶלָּא דִּבְדֶרֶךְ עֲרַאי אֵרַע שֶׁדָּר שָׁם: הַיְנוּ כְּשֶׁמּוֹלִיכִין עֵרוּבָן אֵצֶל אֲחֵרִים וּכְדִתְנַן פֶּרֶק ו' מִשְׁנָה ז'. רִי"כ: וְצָרִיךְ לוֹמַר דְּמֵחֲמַת אִסּוּר שַׁבָּת שֶׁאֵין כָּאן הִלְכָּךְ אִי בָּעֵי כוּ', אֲבָל לֹא שֶׁיְּהֵא רַשַּׁאי מִן הַדִּין לְהַשְׁלִיכָם לַחוּץ, שֶׁהֲרֵי מָקוֹם יֵשׁ לוֹ לַמַּשְׂכִּיר לְהַנִּיחַ שָׁם חֲפָצָיו. וְכֵן כָּתַב הַבֵּית יוֹסֵף דְּתַרְתֵּי בָּעִינַן, שֶׁיִּהְיֶה כֵּלָיו שָׁם וְשֶׁיִּהְיֶה לוֹ פִּנָּה מְיֻחֶדֶת. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב:

פרק ח - משנה ז

כֵּיוָן שֶׁבָּאָה מִכֹּחַ כַּרְמְלִית דְּזֶהוּ הַנָּהָר שֶׁנִּמְשֶׁכֶת מִמֶּנּוּ. הָרִי"כ: מַעֲשֶׂה. אֵין זֶה מַעֲשֶׂה לִסְתֹּר, דְּאַמִּלְּתֵיהּ דִּלְעֵיל קָאֵי דְּלֹא בָּעֵי מְחִצָּה לְשֵׁם מַיִם. תּוֹסָפוֹת: שֶׁאִם עוֹבֶרֶת בִּרְשׁוּת הָרַבִּים הָוְיָן רְשׁוּת הָרַבִּים, כְּדִתְנַן בְּפֶרֶק י"א דְּשַׁבָּת מִשְׁנָה ד' רְקָק מַיִם וּרְשׁוּת הָרַבִּים מְהַלֶּכֶת בּוֹ חַיָּב וְכַמָּה הוּא רְקָק פָּחוֹת מֵעֲשָׂרָה טְפָחִים, אַלְמָא מַיִם שֶׁאֵינָן עֲמֻקִּים עֲשָׂרָה אֵינָן רְשׁוּת לְעַצְמָן, וְכִי הָווּ נַמִּי בֶּחָצֵר הָווּ לְהוּ רְשׁוּת הַיָּחִיד. רַשִׁ"י, וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב:
