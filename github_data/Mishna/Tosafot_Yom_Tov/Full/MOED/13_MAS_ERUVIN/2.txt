פרק ב - משנה א

גָּבְהָן עֲשָׂרָה כוּ'. גָּבְהָן עֲשָׂרָה שֶׁאֵין נִקְרֵאת מְחִצָּה בְּפָחוֹת מֵעֲשָׂרָה, וְרָחְבָּן שִׁשָּׁה טְפָחִים דְּזֶהוּ אַמָּה, שֶׁאַף עַל פִּי שֶׁבִּמְחִצַּת חָצֵר סַגִּי בְּפַס אַרְבָּעָה, לֹא דָּמִי, דְּהָתָם בְּחָצֵר אִיכָּא שָׁלֹשׁ מְחִצּוֹת שְׁלֵמוֹת אֲבָל בְּפַסֵּי בֵּירָאוֹת שֶׁרֻבּוֹ פָּרוּץ מִכָּל צַד מַצְרִיכִין פַּס שֶׁל שִׁשָּׁה טְפָחִים. רִי"כ: קְשׁוּרוֹת. וְאַף עַל פִּי שֶׁאֵין הַפָּרוֹת מוֹשְׁכוֹת בָּעֹל בְּשַׁבָּת אֲפִלּוּ הָכִי מְשַׁעֲרִין בְּהוּ, לְפִי שֶׁגַּם בִּימוֹת הַחוֹל נִכְנָסוֹת שָׁם לִשְׁתּוֹת, וּפְעָמִים שֶׁכְּשֶׁהַצֶּמֶד הָאֶחָד נִכְנָס לִשְׁתּוֹת מוֹצֵא הַשֵּׁנִי שֶׁשָּׁתָה מִשָּׁם וְרוֹצֶה לָצֵאת, וְאִם יִהְיוּ בֵּין הַדְּיוּמְדִין פָּחוֹת מִזֶּה הַשִּׁעוּר יָבֹאוּ הַפָּרוֹת וְיַפִּילוּ הַדְּיוּמְדִין לָאָרֶץ. הָרִי"כ: הָכִי תָּנוּ רַבָּנָן. וּפֵרֵשׁ רַשִׁ"י וְלֹא פָרָה נִכְנֶסֶת וּפָרָה יוֹצֵאת, דְּאִם כֵּן לֹא הֲוָה קְשׁוּרוֹת: רְצוֹנוֹ לוֹמַר דְּאִלּוּ עֶשֶׂר אַמּוֹת לְרַבִּי מֵאִיר וְכֵן הַשְּׁלֹשׁ עֶשְׂרֵה וּשְׁלִישׁ לְרַבִּי יְהוּדָה לֹא בָּעִינַן מְצֻמְצָמוֹת. וְכֵן הוּא בְּרַשִׁ"י וְרַמְבַּ"ם:

פרק ב - משנה ג

סַחַר. בְּפֶרֶק ג' דִּשְׁבִיעִית מִשְׁנָה ד' גָּרַס סַהַר, וַאֲפִלּוּ הָכִי מְפָרֵשׁ כְּמוֹ סַחַר, אֲבָל הָתָם אִי אֶפְשָׁר לְפָרֵשׁ בֵּית הָאֲסוּרִים: וְנִקְרֵאת מֻקְצֶה לְפִי שֶׁאֵין רֹב תַּשְׁמִישׁ שָׁם: פֵּרוּשׁ לְדִירַת אָדָם לִכְנִיסָה וִיצִיאָה תָּמִיד. רַשִׁ"י:

פרק ב - משנה ד

בּוֹר. הוּא הַגֶּבֶא שֶׁיִּתְכַּנְּסוּ שָׁם הַמַּיִם. הָרַמְבַּ"ם: וְהִקְשׁוּ בַּתּוֹסָפוֹת דְּאִם כֵּן אַמַּאי בָּעִינַן שֶׁיִּהְיוּ רְאוּיִין לְאָדָם. וְיֵשׁ לוֹמַר שֶׁאִם דָּלָה לְצֹרֶךְ בְּהֵמָה וְנִשְׁתַּיְּרוּ שָׁם שָׁרֵי לָאָדָם. אִי נַמִּי דִּתְחִלַּת הַתִּקּוּן לֹא הִתִּירוּ אֶלָּא לְצֹרֶךְ בְּהֵמָה, אֲבָל בְּשֶׁכְּבָר נִתְקַן לְצֹרֶךְ בְּהֵמָה שָׁרֵי אַף לְצֹרֶךְ אָדָם: הַיְנוּ כְּשֶׁהַבּוֹר קְצָרָה, אֲבָל אִם הִיא רְחָבָה כָּל כָּךְ שֶׁלֹּא יוּכַל לְהַרְחִיב רַגְלָיו בְּצִדֵּי הַבּוֹר וְיֵרֵד, מֻתָּר לְכַתְּחִלָּה לְכָל אֶחָד מֵהַנֵּי פֵּרוּשִׁים, אוֹ לִדְלוֹת אוֹ לְתַקֵּן. תּוֹסָפוֹת: חֲגוֹרָה. מְחִצָּה, וְכֵן הוּא בַּתּוֹסָפוֹת, שֶׁמִּפְּנֵי שֶׁמַּצְרִיכִין מְחִצָּה מֵאַרְבַּע רוּחוֹתֶיהָ קוֹרֵא לָהּ חֲגוֹרָה. הָרִי"כ:

פרק ב - משנה ה

וְאַף עַל גַּב דְּרַבִּי יְהוּדָה נַמִּי תָּנָא חֲדָא לְחֻמְרָא בְּמִשְׁנָה ג' וְקָתָנֵי אַחֲרִיתָא בְּמִשְׁנָה ד' וְלֹא קָתָנֵי וְעוֹד, הָתָם אַפְסְקוּהוּ רַבָּנָן בְּמִלְּתָא אַחֲרִיתָא דְּאִיהוּ לֹא אָיְרֵי אֶלָּא בְּפַסֵּי בֵּירָאוֹת כוּ'. גְּמָרָא: שׁוֹמֵירָה אוֹ בֵּית דִּירָה כוּ'. שׁוֹמֵירָה, כְּעֵין סֻכָּה שֶׁל שׁוֹמְרֵי הַצֹּאן אוֹ הַשָּׂדוֹת אוֹ הַכְּרָמִים בִּקְבִיעוּת יוֹמָם וְלַיְלָה כָּל הַשָּׁנָה, אוֹ בֵּית דִּירָה לְבַעַל הַבַּיִת אַף עַל פִּי שֶׁאֵינָהּ תְּדִירָה אֶלָּא לִפְרָקִים אוֹ שֶׁתְּהֵא סְמוּכָה לָעִיר בְּתוֹךְ אַלְפַּיִם אַמָּה דְּיָכוֹל בַּעַל הַבַּיִת לָלֶכֶת וּלְטַיֵּל בָּהּ. הָרִי"כ: אֶלָּא בוֹר. רְצוֹנוֹ לוֹמַר אֶחָד מֵהֶם אוֹ בּוֹר אוֹ שִׁיחַ אוֹ מְעָרָה. הָרַמְבַּ"ם: עַנְוְתָנוּתוֹ תַּרְבֵּנוּ וּמִתּוֹרָתוֹ יְלַמְּדֵנוּ וּבְמִצְווֹתָיו יְצַוֵּנוּ וִיעוֹרְרֵנוּ לְהָבִין וּלְהַשְׂכִּיל וְלֵירֵד לְסוֹף דַּעַת הָרַמְבַּ"ם. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב:

פרק ב - משנה ו

כְּבֵית כּוֹר. וְאַף עַל גַּב דְּלֹא הֻקַּף לְדִירָה, דְּלֹא דָּמִי לְבִקְעָה, דְּבִקְעָה לֹא מְסַיְּמֵי מְחִצּוֹתֶיהָ אֶלָּא בְּדֶרֶךְ מִדְרוֹן נִכְנָס לָהּ וְיוֹצֵא מִמֶּנָּה, אֲבָל בְּגִנָּה וְקַרְפֵּף יֵשׁ לָהֶם מְחִצּוֹת מַעֲלְיוּתָא אַךְ לֹא נַעֲשׂוּ עַל דַּעַת שֶׁיָּדוּרוּ בָּהֶם. הָרִי"כ: רֵישָׁא דְּמַתְנִיתִין נָקַט אַף עַל גַּב שֶׁהוּא דְּלֹא כְּהִלְכְתָא. וּלְעִנְיַן פְּסַק הֲלָכָה דְּמִשְׁנָה זוֹ כָּתַב הָרַמְבַּ"ם שֶׁכֻּלָּן דְּחוּיוֹת: בֵּיתוֹ אָסוּר כוּ'. כָּתַב הָרַמְבַּ"ם רְצוֹנוֹ לוֹמַר רְשׁוּתוֹ בֶּחָצֵר. וּמוּסָב עַל בִּטֵּל רְשׁוּתוֹ סְתָם:
