פרק ג

וְאַף עַל גַּב דִּתְחוּמִין חֲמִירֵי, שֶׁיֵּשׁ לָהֶם סֶמֶךְ מִן הַמִּקְרָא, הֵקֵלוּ בִּתְחוּמִין שֶׁאֵינָהּ אֶלָּא לִדְבַר מִצְוָה. תּוֹסָפוֹת. וְהָא דְּקָרֵי לֵיהּ עֵרוּב, לְפִי שֶׁהוּא מְעָרֵב הַתְּחוּמִין שֶׁלֹּא הָיָה יָכוֹל לֵילֵךְ בּוֹ תְּחִלָּה עִם הַתְּחוּם שֶׁהָיָה יָכוֹל לֵילֵךְ בּוֹ. הַמַּגִּיד מִשְׁנֶה: חוּץ כוּ'. דְּטַעֲמָא דְּעֵרוּבִין מִשּׁוּם דְּדַעְתּוֹ וְדִירָתוֹ בִּמְקוֹם מְזוֹנוֹתָיו הוּא, וְנַעֲשֶׂה הַמְעָרֵב כְּאִלּוּ אוֹכֵל וְשׁוֹבֵת שָׁם בַּמָּקוֹם שֶׁהִנִּיחַ הָעֵרוּב, וּמִשָּׁם יֵשׁ לוֹ אַלְפַּיִם אַמָּה לְכָל רוּחַ, וְהִלְכָּךְ דָּבָר בָּעִינַן. רַשִׁ"י: דְּכֻלָּהּ סְפֵקָא מִשּׁוּם עֲצָמוֹת הוּא, שֶׁהַבָּשָׂר כְּבָר נִרְקַב כַמָּה שָׁנִים, וַעֲצָמוֹת בְּלֹא בָּשָׂר אֵין מְטַמְּאִין בְּאֹהֶל אֶלָּא בְּמַגָּע, וּכְשֶׁמְּנַפֵּחַ לִפְנֵי רַגְלָיו אִם יֵשׁ עֶצֶם גָּדוֹל נִרְאֶה הוּא וְיִזָּהֵר שֶׁלֹּא יִגַּע בּוֹ רַגְלָיו, וְאִם עֶצֶם קָטָן הוּא נִדְחֶה בַּנְּפִיחָה וְאֵינוֹ מְסִיטוֹ בְּרַגְלָיו, וְאִם מְכֻסֶּה בְּעָפָר אֵינוֹ מְטַמֵּא, דְּהָא עֶצֶם כִּשְׂעוֹרָה אֵינוֹ מְטַמֵּא אֶלָּא בְּמַגָּע. רִי"כ: בַּעֲגָלוֹת, מִבְּעוֹד יוֹם (רַשִׁ"י). וְעַיֵּן רֵישׁ פֶּרֶק ח' דְּאָהֳלוֹת שֶׁצָּרִיךְ שֶׁמַּחְזִיקִים אַרְבָּעִים סְאָה כוּ', וּפָרֵיךְ בַּגְּמָרָא הֵיכִי מַיְתֵי לַהּ כוּ', וּמִדְּנָקַט רַבִּי יְהוּדָה וְלֶאֱכֹל שְׁמַע מִנָּהּ דְּרָאוּי לְאָכְלוֹ בָּעֵי, וּמְשָׁנֵי בִּפְשׁוּטֵי כְּלֵי עֵץ שֶׁאֵין בְּרָחְבָּן טֶפַח, דְּלֹא מְקַבְּלֵי טֻמְאָה וְאֵין מְבִיאִין אֶת הַטֻּמְאָה: וְכָל תַּשְׁמִישֵׁי הַמֵּת אֲסוּרִין בַּהֲנָאָה מִוַּתָּמָת שָׁם מִרְיָם, וְגָמַר שָׁם שָׁם מֵעֶגְלָה עֲרוּפָה. רַשִׁ"י:

וּבִתְרוּמָה. כֵּן הַגִּרְסָא בִּסְפָרִים מְדֻיָּקִים וּבָרִי"ף. וְהָכִי מִסְתַּבְּרָא, דְּמַאי שְׁנָא תְּרוּמָה מֵחַלָּה. וּמַתְנִיתִין כְּסוּמְכוֹס, וְאֵינָהּ הֲלָכָה. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: הַטַּעַם דְּעֵרוּבֵי תְּחוּמִין הוֹאִיל וְאַסְמְכוּהוּ אַקְּרָא הֶחְמִירוּ וְלֹא הֵימְנוּהוּ רַבָּנָן לְקָטָן. תּוֹסָפוֹת. וְהָרִי"כ פֵּרֵשׁ דְּעֵרוּבֵי חֲצֵרוֹת דְּקָנֵי מִמֵּילָא כְּדִתְנַן בְּפֶרֶק ו' מִשְׁנָה ה' בַּעַל הַבַּיִת שֶׁהָיָה שֻׁתָּף עִם שְׁכֵנָיו אֵין צָרִיךְ לְעָרֵב, וּלְפִיכָךְ קָטָן גּוֹבֶה אוֹתוֹ וְלֹא אִכְפַּת לָן שֶׁהֲרֵי הוּא אֵין צָרִיךְ שֶׁיֹּאמַר בּוֹ כְּלוּם, אֲבָל בְּעֵרוּבֵי תְּחוּמִין צָרִיךְ הוּא שֶׁיֹּאמַר בְּמָקוֹם עֵרוּבִי זֶה תְּהֵא שְׁבִיתָתוֹ שֶׁל פְּלוֹנִי, וְהוּא לָאו בַּר דַּעַת הוּא:

הַיְנוּ בִּמְקוֹם מִצְוָה, כְּמוֹ שֶׁכָּתַב הָרַמְבַּ"ם: הָכִי מוֹקִים בַּגְּמָרָא. וְקָשֶׁה, לִפְלוּג וְלִתְנֵי בְּאִילָן גּוּפֵיהּ בֵּין הָעוֹמֵד בִּרְשׁוּת הָרַבִּים לְעוֹמֵד בְּכַרְמְלִית:

חוּץ לַתְּחוּם. אֲפִלּוּ תְּחוּמִים דְּרַבָּנָן, וַאֲפִלּוּ לְרַבִּי דְּאָמַר כָּל דָּבָר שֶׁהוּא מִשּׁוּם שְׁבוּת לֹא גָּזְרוּ בּוֹ כוּ', מִכָּל מָקוֹם לֹא הָוֵי עֵרוּב אַף עַל גַּב דְּבֵין הַשְּׁמָשׁוֹת מָצֵי לְמִשְׁקְלֵיהּ וְאָתוֹיֵי, כֵּיוָן דְּאִם קָנָה שְׁבִיתָה בַּמָּקוֹם שֶׁנִּתְגַּלְגֵּל הָעֵרוּב הָיָה עוֹמֵד עַתָּה חוּץ לַתְּחוּם וְאֵין לוֹ אֶלָּא אַרְבַּע אַמּוֹת, הִלְכָּךְ תְּקִינוּ רַבָּנָן שֶׁלֹּא יְהֵא עֵרוּב אֶלָּא כְּשֶׁהוּא בְּתוֹךְ הַתְּחוּם. תּוֹסָפוֹת: וְכָתַב הָרִי"כ שֶׁיֵּשׁ לוֹ אַרְבַּע אַמּוֹת כְּנֶגֶד כָּל הָעִיר, שֶׁהוּא מַפְסִיד, וְלֹא דָּמִי לְמִי שֶׁיָּצָא חוּץ לַתְּחוּם דְּאָמְרִינַן בְּמַתְנִיתִין סוֹף פֶּרֶק ד' דַּאֲפִלּוּ אַמָּה אַחַת לֹא יִכְנֹס, דְּהָתָם אֵין רָאוּי לָתֵת לוֹ אַרְבַּע אַמּוֹת יוֹתֵר שֶׁהֲרֵי נִתְכַּוֵּן לִשְׁבֹּת בָּעִיר שֶׁהִיא עוֹלָה לוֹ כְּאַרְבַּע אַמּוֹת, וְאִם יֵצֵא חוּץ לַתְּחוּם כְּבָר הוּא גַּם חוּץ לְאַרְבַּע אַמּוֹת, אֲבָל הָכָא שֶׁהָיָה רוֹצֶה לַעֲקֹר אֶת עַצְמוֹ מִן הָעִיר רָאוּי לָתֵת לוֹ בִּמְקוֹם שֶׁמֵּנִיחַ שָׁם עֵרוּבוֹ אַרְבַּע אַמּוֹת, שֶׁשָּׁם הוּא בֵּיתוֹ: בַּגְּמָרָא. וְנִרְאֶה לְפָרֵשׁ דְּחוֹפֵר בִּכְלִי, וְהוּא תּוֹלָדָה דְּחוֹרֵשׁ. תּוֹסְפוֹת יוֹם טוֹב: חַמָּר גַּמָּל. מִשּׁוּם דִּסְבִירָא לֵיהּ תְּחוּמִין דְּאוֹרַיְתָא. וְהָא דִּמְטַהֵר בִּסְפֵק טֻמְאָה דְּאוֹרַיְתָא בְּמִשְׁנָה ז' פֶּרֶק ה' דְּטָהֳרוֹת, הָתָם תְּרֵי חֲזָקוֹת לְקֻלָּא, דְּהַעֲמֵד מֵת עַל חֶזְקָתוֹ שֶׁהָיָה חַי וְהַעֲמֵד הַנּוֹגֵעַ עַל חֶזְקָתוֹ שֶׁהָיָה טָהוֹר, וְהָכָא חֲדָא חֲזָקָה, דְּהַעֲמֵד תְּרוּמָה עַל חֶזְקָתָהּ, אֲבָל לֵיכָּא עוֹד חֲזָקָה וְלוֹמַר הַעֲמֵד גַּבְרָא אַחֶזְקָתוֹ שֶׁעֵרֵב. גְּמָרָא:

וּבַיְּרוּשַׁלְמִי, כַּנֵּי מַתְנִיתִין מַתְנֶה אָדָם עַל : בִּדְרַבָּנָן. וּתְחוּמִין דְּרַבָּנָן כְּדִסְתָם פֶּרֶק ה' מִשְׁנָה ה': וְאִם כוּ'. אִצְטְרִיךְ, אֲפִלּוּ אֶחָד רַבּוֹ מֻבְהָק. תּוֹסָפוֹת:

שְׁנֵי עֵרוּבִין. כְּגוֹן דְּאַנַּח לֵיהּ בְּסוֹף אֶלֶף אַמָּה לְכָאן אֶלֶף אַמָּה לְכָאן. דְּאִי לֹא תֵּימָא הָכִי אֶלָּא דְּמָנַח לֵיהּ לְסוֹף אַלְפַּיִם לְכָאן נִמְצָא שֶׁמַּפְסִיד כָּל הָאַלְפַּיִם שֶׁל צַד אַחֶרֶת וַהֲרֵי אֵינוֹ יָכוֹל לֵילֵךְ בַּיּוֹם הָאֶחָד אֶל מְקוֹם הָעֵרוּב שֶׁל יוֹם הַשֵּׁנִי, וַאֲנַן בָּעִינַן שֶׁיְּהֵא הוּא וְעֵרוּבוֹ בְּמָקוֹם אֶחָד בְּשָׁעָה שֶׁיִּקְנֶה הָעֵרוּב, וּכְדִתְנַן בְּמִשְׁנָה ד': מִשּׁוּם יוֹם טוֹב שֶׁחָל לִהְיוֹת אַחַר הַשַּׁבָּת נָקֵיט לֵיהּ, שֶׁאִי אֶפְשָׁר לוֹ בְּיוֹם הַשַּׁבָּת לְהוֹלִיךְ הָעֵרוּב. אֲבָל בְּיוֹם טוֹב שֶׁלִּפְנֵי הַשַּׁבָּת יָכוֹל לִטֹּל אוֹתוֹ הַפַּת עַצְמוֹ, שֶׁכְּבָר נִקְרָא עָלָיו שֵׁם עֵרוּב, וּלְהוֹלִיכוֹ לְרוּחַ אַחֶרֶת. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: כְּלוֹמַר לְאֵיזֶה רוּחַ שֶׁעֵרֵב אֵינוֹ רוֹצֶה לְהַפְסִיד הָאַלְפַּיִם שֶׁכְּנֶגְדוֹ כוּ', אֲבָל לְעוֹלָם אֵין כָּאן הֶפְסֵד אֶלָּא אַלְפַּיִם מִצַּד אֶחָד, וְהַיְנוּ אוֹתָן אַלְפַּיִם שֶׁהִיא כְּנֶגֶד אוֹתָהּ שֶׁעֵרֵב בָּהּ: טַעֲמוֹ דְּאִלּוּ עַל יְדֵי עַצְמוֹ הָא קָיְמָא לָן דְּעִקַּר עֵרוּב בָּרֶגֶל וְאֵינוֹ צָרִיךְ פַּת: מִשְׂתַּכֵּר. וְכָל שֶׁכֵּן לְרַבִּי אֱלִיעֶזֶר בָּעִינַן שֶׁיִּהְיֶה קַיָּם בֵּין הַשְּׁמָשׁוֹת דְּשֵׁנִי, כְּמוֹ שֶׁכָּתַב הָרַמְבַּ"ם, שֶׁאִם אֵינוֹ קַיָּם אֵין כָּאן עֵרוּב לְיוֹם שֵׁנִי דְּהָא קְדֻשָּׁה אַחֲרִיתָא הִיא, הִלְכָּךְ הַאי מִשְׂתַּכֵּר כוּ' לֹא אָמְרוּ חֲכָמִים לַאֲפוֹקֵי מִדְּרַבִּי אֱלִיעֶזֶר, אֶלָּא מִלְּתָא אַגַּב אוֹרְחָא קָאָמְרֵי:

הָאֱמֶת בָּא לְהוֹדִיעֵנוּ בְּהֶדְיָא בַּגְּמָרָא כוּ'. אֲבָל לְעִנְיַן פֵּרוּשׁ הַמִּשְׁנָה לֹא אִכְפַּת לָן וְלֹא מִידֵי, דְּהָא אֶפְשָׁר לִהְיוֹת דַּחֲכָמִים דִּלְעֵיל דִּמְסַפְּקָא לְהוּ בְּיוֹם טוֹב וְשַׁבָּת דְּאִינְהוּ גּוּפַיְהוּ סְבִירָא לְהוּ בִּשְׁנֵי יָמִים דְּרֹאשׁ הַשָּׁנָה דְּוַדַּאי כְּיוֹמָא אֲרִיכָא הֵן. תּוֹסְפוֹת יוֹם טוֹב:

וְעוֹד אָמַר רַבִּי יְהוּדָה. וּצְרִיכָא, דְּאִי אַשְׁמְעִינַן בְּעֵרוּב בְּהָא קָאָמַר רַבִּי יְהוּדָה מִשּׁוּם דְּלֹא קָעֲבִיד מִידֵי, אֲבָל כַּלְכָּלָה דְּמֵחֲזֵי כִּמְתַקֵּן טִבְלָא אֵימָא מוֹדֶה לְהוּ לְרַבָּנָן. וְאִי אַשְׁמְעִינַן הָנֵי תַּרְתֵּי מִשּׁוּם דְּלֵיכָּא לְמִגְזַר עֲלַיְהוּ, אֲבָל בֵּיצָה דְּאִיכָּא לְמִגְזַר בַּהּ מִשּׁוּם פֵּרוֹת הַנּוֹשְׁרִין וּמִשּׁוּם מַשְׁקִין שֶׁזָּבוּ אֵימָא מוֹדֶה לְהוּ לְרַבָּנָן, קָא מַשְׁמַע לָן. גְּמָרָא: וְלֹא הֻצְרְכוּ לוֹמַר בְּיוֹם שֵׁנִי אִם הַיּוֹם קֹדֶשׁ כוּ' אֶלָּא כְּדֵי שֶׁיִּתֵּן אֶל לִבּוֹ שֶׁהַיּוֹם קֹדֶשׁ וְלֹא יְזַלְזֵל בִּמְלָאכָה, דְּדַי הָיָה לוֹ שֶׁיֹּאמַר אִם הַיּוֹם חֹל תִּהְיֶה זוֹ תְּרוּמָה עַל זוֹ. הָרִי"כ: וְלֵיכָּא לְמֵימַר מִגּוֹ דְּאִתְקְצֵי לְבֵין הַשְּׁמָשׁוֹת אִתְקְצֵי לְכֻלֵּיהּ יוֹמָא כִּדְאִיתָא בְּסוֹף פֶּרֶק ג' דְּשַׁבָּת, דְּהָתָם הַאי יוֹמָא דְּשַׁבָּתָא דְּאָסְרִינַן בֵּיהּ וַדַּאי קֹדֶשׁ וּבֵין הַשְּׁמָשׁוֹת הוּא הַסָּפֵק אָמְרִינַן מִגּוֹ כוּ', אֲבָל הָכָא יוֹם הַשֵּׁנִי שֶׁבּוֹ נָקוּם וְנֶאֱסֹר הוּא עַצְמוֹ הַסָּפֵק, הִלְכָּךְ לֵיכָּא לְמֵימַר מִגּוֹ כוּ':

אִם לְמָחָר. כְּדֵי שֶׁלֹּא יֵרָאֶה בִּתְפִלָּתוֹ כְּשַׁקְרָן. הָרִי"כ: בַּגְּמָרָא הַטַּעַם דְּזִכָּרוֹן אֶחָד עוֹלֶה לְכָאן וּלְכָאן. וּפֵרְשׁוּ הַתּוֹסְפוֹת דִּבְרֹאשׁ הַשָּׁנָה כְּתִיב זִכְרוֹן תְּרוּעָה, וּבְרֹאשׁ חֹדֶשׁ כְּתִיב וְהָיָה לָכֶם לְזִכָּרוֹן וְגוֹ' וּבְרָאשֵׁי חָדְשֵׁיכֶם. הִלְכָּךְ כְּשֶׁיֹּאמַר בַּתְּפִלָּה יוֹם הַזִּכָּרוֹן הַזֶּה עוֹלֶה לִשְׁנֵיהֶם: דְּאָתֵי לְזִלְזוּלֵי בֵּיהּ בַּשֵּׁנִי אִי מַחְזִיקִינַן לֵיהּ בְּסָפֵק, אֲבָל כִּי אָמְרִינַן סְתָמָא מַחְזִיקִינַן לְהוּ בִּקְדֻשָּׁה אַחַת. גְּמָרָא:
