פרק א - משנה א

דְּיָלְפִינַן מִדִּכְתִיב תַּשְׁבִּיתוּ וְלֹא כְתִיב תְּבַעֲרוּ, וְהַשְׁבָּתָה דְלֵב הִיא הַשְׁבָּתָה. רַשִׁ"י: כְּלוֹמַר דִּכְמוֹ שֶׁבְּבִטּוּל בְּעָלְמָא דְּהַיְנוּ שֶׁיַּחְשֹׁב בְּלִבּוֹ כְּאִלּוּ הוּא עָפָר סַגִּי בְכָךְ אַף עַל פִּי שֶׁלֹּא הוֹצִיא בִשְׂפָתָיו כְּלוּם, הָכִי נַמִּי בְּמַחֲשָׁבָה חוֹזֵר מִבִּטּוּלוֹ וְזוֹכֶה בָהּ, דְּמַחֲשָׁבָה מְבַטֶּלֶת מַחֲשָׁבָה: שֶׁאֵין. בַּגְּמָרָא מְפָרֵשׁ לַאֲתוֹיֵי כוּ' וְאוֹצְרוֹת יַיִן וְשֶׁמֶן דְּאֵין צְרִיכִים בְּדִיקָה. וְאַהָא מַקְשֶׁה וְלָמָה אָמְרוּ כוּ' דְּהָא אֵין צְרִיכִים בְּדִיקָה, וּמְשָׁנֵי בְּמַרְתֵּף כוּ': פֵּרֵשׁ רַשִׁ"י עַד הַתִּקְרָה, וְהוּא כְמוֹ שְׁמֵי הַשָּׁמַיִם: אָרְכּוֹ הוּא שֶׁאָמַר שֶׁבּוֹדֵק שׁוּרָה הַחִיצוֹנָה מִן הָאָרֶץ, וְרָחְבּוֹ הוּא שֶׁאָמַר שֶׁחוֹזֵר וּבוֹדֵק הָעֶלְיוֹנוֹת. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: וַעֲשׂוּיָה כְּמוֹ כַּ"ף פְּתוּחָה שֶׁלָּנוּ, שׁוּרָה בִזְקִיפָה וְשׁוּרָה עֶלְיוֹנָה בִשְׁכִיבָה, כְּאָדָם הַבּוֹדֵק כֹּתֶל הַבַּיִת מִבַּחוּץ וְהַגַּג הָרוֹאֶה אֶת הַקּוֹרָה, דְּהַיְנוּ כָּל גַּג הַשּׁוּרוֹת. רַשִׁ"י: וְהַאי פֶּתַח לָאו כְּנֶגְדּוֹ מַמָּשׁ קָאָמַר, אֶלָּא הַכֹּתֶל שֶׁהַפֶּתַח בּוֹ קָרֵי הָכִי. רַשִׁ"י:

פרק א - משנה ג

וְאִיכָּא לְמֵידַק מִשֶּׁתֶּחְשַׁךְ וְאֵילָךְ לָמָּה לֹא יִבְדֹּק, וְאֶפְשָׁר דְּהַיְנוּ טַעֲמָא מִשּׁוּם דַּאֲפִלּוּ רַבָּנָן מוֹדוּ דְּכָל הֵיכָא דְעָנוּשׁ כָּרֵת אֲפִלּוּ מְחַזֵּר עָלָיו לְשָׂרְפוֹ חָיְשִׁינַן דִּלְמָא אָתֵי לְמֵיכַל מִנֵּיהּ. וְלֹא סָמְכִינַן אַהַאי טַעֲמָא אֶלָּא עַד שֶׁתֶּחְשַׁךְ, דְּלֵיכָּא כָרֵת. הָרַ"ן: שֶׁמְּשַׁיֵּר. בַּלַּיְלָה לְצָרְכּוֹ שֶׁיֹּאכַל בַּלַּיְלָה גַּם בַּבֹּקֶר. הָרַ"ן: שֶׁלֹּא. דְּהָא דִתְנַן לְעֵיל דְּאֵין חוֹשְׁשִׁין הַיְנוּ מִסָּפֵק. אֲבָל הָכָא אִיכָּא לְמֵיחַשׁ שֶׁמָּא תִטֹּל חֻלְדָּה בְּפָנֵינוּ וְיִהְיֶה צָרִיךְ בְּדִיקָה. גְּמָרָא:

פרק א - משנה ד

שֶׁנֶּאֱמַר אַךְ בַּיּוֹם הָרִאשׁוֹן תַּשְׁבִּיתוּ שְׂאֹר מִבָּתֵּיכֶם. וּמִפִּי הַשְּׁמוּעָה לָמְדוּ שֶׁהָרִאשׁוֹן זֶהוּ אַרְבָּעָה עָשָׂר, רְאָיָה לְדָבָר זֶה שֶׁכָּתוּב בַּתּוֹרָה לֹא תֹאכַל עָלָיו חָמֵץ, כְּלוֹמַר עַל קָרְבַּן הַפֶּסַח, כָּךְ לָמְדוּ בְּפֵרוּשׁ דָּבָר זֶה, לֹא תֹאכַל חָמֵץ מִשָּׁעָה שֶׁרְאוּיָה לִשְׁחִיטַת הַפֶּסַח, שֶׁהוּא בֵין הָעַרְבַּיִם וְהוּא חֲצִי יוֹם. הָרַמְבַּ"ם:

פרק א - משנה ה

דִּלְשַׁהוֹיֵי עַד אַחַר הַפֶּסַח עוֹבֵר בַּעֲשֵׂה דְּוּבָאתָ שָּׁמָה וַהֲבֵאתֶם שָׁמָּה (כְּמוֹ שֶׁכָּתַב הָרַ"ב רֵישׁ רֹאשׁ הַשָּׁנָה):

פרק א - משנה ו

מִדְּקָתָנֵי לְהַדְלִיק בְּנֵר. דְּאִלּוּ שֶׁמֶן שֶׁל קֹדֶשׁ כְּשֶׁנִּפְסַל נִשְׂרָף בַּקֹּדֶשׁ בְּבֵית הַדֶּשֶׁן וְאֵין לְהַכֹּהֲנִים הֶתֵּר בָּהֶם. אֲבָל בִּתְרוּמָה כְתִיב מִשְׁמֶרֶת תְּרוּמֹתָי, בִּשְׁתֵּי תְרוּמוֹת הַכָּתוּב מְדַבֵּר, אַחַת טְהוֹרָה וְאַחַת טְמֵאָה, וְאָמַר רַחֲמָנָא שֶׁלְּךָ תִהְיֶה לְהַסִּיק תַּחַת תַּבְשִׁילְךָ. רַשִׁ"י: בִּטְמֵא מֵת. דִּיֵּק לוֹמַר שֶׁנִּטְמָא בִטְמֵא מֵת, אַף עַל גַּב דְּמַשְׁקִין לְעוֹלָם תְּחִלָּה הָווּ מִגְּזֵרַת שְׁמֹנָה עָשָׂר דָּבָר , מִשּׁוּם דְּאִיהוּ סְבִירָא לֵיהּ דְּמִן הַתּוֹרָה מַשְׁקִין מְטַמְּאִין אֳכָלִין, וְאַשְׁמְעִינַן הַשְׁתָּא דִּכְשֶׁנִּטְמְאוּ בְנֵר שֶׁנִּטְמָא בִטְמֵא מֵת הוֹסִיפוּ עַל טֻמְאָתָן, שֶׁנַּעֲשֶׂה רִאשׁוֹן וּמְטַמְּאִין אֳכָלִין מִדְּאוֹרָיְתָא. גְּמָרָא:

פרק א - משנה ז

כֵּן כָּתַב רַשִׁ"י. וְהִקְשָׁה דְּסוֹף סוֹף כֵּיוָן דְּאֵין אֹכֶל מְטַמֵּא אֹכֶל אֶלָּא מִדְּרַבָּנָן מַה קָּא מְהַנֵּי לֵיהּ הַאי בָּשָׂר לְבָשָׂר שְׁלִישִׁי, אִי לְשַׁוְּיֵהּ שֵׁנִי מִדְּרַבָּנָן, מֵעִקָּרָא נַמִּי שֵׁנִי מִדְּרַבָּנָן הָוֵי, דַּהֲרֵי הַמַּשְׁקִין שֶׁנָּגְעוּ בָהֶן לְעוֹלָם תְּחִלָּה הָוְיָן, אֶלָּא עַל כָּרְחֲךָ לְהַאי שִׁנּוּיָא סְבִירָא לֵיהּ אֹכֶל מְטַמֵּא אֹכֶל בְּקָדָשִׁים מִדְּאוֹרָיְתָא. וּלְמָאן דְּאָמַר דְּאֵין מְטַמֵּא אֵין צָרִיךְ לְאוֹקְמָא בְמַשְׁקִין, וּמִדִּבְרֵיהֶן מֵרַבִּי אֱלִיעֶזֶר וְרַבִּי יְהוֹשֻׁעַ. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: וְאִם תֹּאמַר אִם כֵּן רַבִּי חֲנִינָא נַמִּי מוֹסִיף אַדְּרַבִּי עֲקִיבָא, דְּמִדְּרַבִּי עֲקִיבָא דְּאָמַר דְּפָסוּל מִדְּאוֹרָיְתָא מֻתָּר לַעֲשׂוֹת רִאשׁוֹן לֹא שָׁמְעִינַן שֶׁיְּהֵא שְׁלִישִׁי דְרַבָּנָן מֻתָּר לַעֲשׂוֹת שֵׁנִי, וּבַיְּרוּשַׁלְמִי פָּרֵיךְ לַהּ דְּלֹא בָא רַבִּי עֲקִיבָא אֶלָּא לִפְחֹת, וּמְשָׁנֵי בִּטְבוּל יוֹם מִדִּבְרֵיהֶם. תּוֹסָפוֹת. וְהַשְׁתָּא אָתֵי שַׁפִּיר לִישְׁנָא דְּמִדִּבְרֵיהֶם, דְּמִדְּבָרָיו הֲוָה לֵיהּ לְמֵימַר:
