פרק ז

שֶׁהַפְּנִימִי מִתְחַמֵּם עַל יְדֵי קְצָתוֹ הַחִיצוֹן. רַשִׁ"י: תּוֹחֲבוֹ מִתּוֹךְ פִּיו. לְפִי שֶׁנּוֹתֵן הַגַּס שֶׁבַּשַּׁפּוּד לְמַטָּה כְּדֵי שֶׁלֹּא יִפֹּל הַפֶּסַח וְאָז הֲוָה בֵית הַשְּׁחִיטָה לְמַטָּה וְקָדָיֵיב דָּמָא דֶּרֶךְ שָׁם. תוֹסָפוֹת: וְצָרִיךְ לוֹמַר שֶׁהֵם מְסֻבָּבִים וּכְרוּכִים בְּעִנְיָן שֶׁלֹּא יִפְּלוּ מֵעַל הָרֹאשׁ, שֶׁהֲרֵי פִי הַטָּלֶה לְמַטָּה:

וּפֵרֵשׁ רַשִׁ"י עֲשׂוּיָה כִבְרִיחִים כְּעֵין שֶׁלָּנוּ, וְיֵשׁ חָלָל גָּדוֹל בֵּין בְּרִיחַ לִבְרִיחַ, וְנוֹתֵן הַשַּׁפּוּד (רְצוֹנוֹ לוֹמַר שֶׁל רִמּוֹן) לְרָחְבּוֹ, וְכָל הַטָּלֶה נִצְלֶה בָאֲוִיר שֶׁבֵּין שְׁנֵי בְרִיחִים, שֶׁאֵין בְּשָׂרוֹ נוֹגֵעַ בַּבַּרְזֶל: וְאִם תֹּאמַר נָגַע בְּחַרְסוֹ כוּ' אַמַּאי סַגִּי לֵיהּ בִּקְלִיפָה, הֲלֹא הָרֹטֶב שֶׁיֵּשׁ בְּאוֹתוֹ קְלִיפָה נִבְלָע בַּפֶּסַח עַד כְּדֵי שֶׁיִּטֹּל , וְאִיכָּא לְאוֹקְמֵי בְּשֶׁאֵין רֹטֶב כִּי אִם מְעַט, וּמְעַט רֹטֶב אֵינוֹ אוֹסֵר כָּל כָּךְ כְּדִלְקַמָּן גַּבֵּי סִיכָה. תּוֹסָפוֹת: רַשִׁ"י. וְאֵין צָרִיךְ לוֹמַר גַּבֵּי קְלִיפָה וּנְטִילָה שֶׁאֵין בָּהֶם תַּעֲרֹבוֹת חֻלִּין:

יְדִיחֶנּוּ. וִינַגֵּב. הָרַמְבַּ"ם: יִקְלֹף. וְאִם תֹּאמַר וּמַאי שְׁנָא מֵרֹטֶב דִּלְעֵיל דְּיִטֹּל מְקוֹמוֹ, וּבַגְּמָרָא שָׁאנֵי סִיכָה דְּמַשֶּׁהוּ בְעָלְמָא הוּא דְעָבִידֵי: דִּכְתִיב וְנָתַתָּ כוּ', כֵּן כָּתַב רַשִׁ"י. וְהִקְשׁוּ בַתּוֹסָפוֹת דְּהָא קְרָא בְּפִדְיוֹן מַיְרֵי שֶׁמּוֹצִיאוֹ מִקְּדֻשָּׁתוֹ וּמְחַלְּלוֹ עַל דָּבָר אַחֵר, אֲבָל הָכָא אֲפִלּוּ לְאָכְלוֹ בִקְדֻשָּׁתוֹ כִּדְפֵּרֵשׁ בַּקּוּנְטְרֵס. וְנִרְאֶה לְרִ"י דְּבִירוּשָׁלַיִם גָּזְרוּ מִדְּרַבָּנָן מְכִירָה אַטּוּ פְדִיָּה דְּאָסוּר מִן הַתּוֹרָה. וּלְרַשְׁבָּ"א נִרְאֶה דְּלֹא אָסְרוּ מְכִירָה אֶלָּא מִשּׁוּם בִּזְיוֹן מִצְוָה, כְּדִתְנַן בְּרֵישׁ מַסֶּכֶת מַעֲשֵׂר שֵׁנִי שֶׁאֵין מוֹכְרִין אוֹתוֹ כוּ'. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב:

וּבַבְּרַיְתָא, מִנַּיִן לָעֹמֶר וּשְׁתֵּי הַלֶּחֶם הָאֲמוּרִים בְּתוֹרַת כֹּהֲנִים וְאֵינָם אֲמוּרִים בְּפִנְחָס, תַּלְמוּד לוֹמַר (וַיִּקְרָא כג) וַיְּדַבֵּר מֹשֶׁה אֶת מֹעֲדֵי ה' וְגוֹ': כִּדְנָפְקָא לָן בַּגְּמָרָא דַּף ע' דְּאָמַר קְרָא כוּ': וְאַף עַל גַּב דִּבְפָרָשַׁת פִּנְחָס כְּתִיב נַמִּי רֹאשׁ חֹדֶשׁ, הֲוָה אֲמֵינָא דְּאֵלֶּה תַּעֲשׂוּ לַה' בְּמוֹעֲדֵיכֶם דִּכְתִיב בַּסּוֹף לֹא קָאֵי אַרֹאשׁ חֹדֶשׁ אֶלָּא אַשְּׁאָר יָמִים טוֹבִים דִּכְתִיבֵי בַתְרֵיהּ, דְּאִקְּרוּ רְגָלִים כְּמוֹ דְמַצְרְכִינַן קְרָאֵי אַשַּׁבָּת וְתָמִיד. תּוֹסָפוֹת: קָא מַשְׁמַע לָן דְּרֹאשׁ חֹדֶשׁ אִקְּרֵי מוֹעֵד כִּדְאַבַּיֵי, דְּאָמַר אַבַּיֵּי קָרָא עָלַי מוֹעֵד (אֵיכָה א), וּפֵרֵשׁ רַשִׁ"י תַּמוּז דְּשִׁלּוּחַ מְרַגְּלִים מַלּוּיֵי מַלְּיוּהַ, וְכָלוּ אַרְבָּעִים יוֹם דִּידְהוּ בְתִשְׁעָה בְאָב, וְיוֹם עִבּוּר הַחֹדֶשׁ קָרוּי מוֹעֵד. וְכָתְבוּ הַתּוֹסָפוֹת וְתֵימַהּ וְלִתְנֵי שְׂעִיר רֹאשׁ חֹדֶשׁ וְלֹא לִתְנֵי אִידָךְ. תּוֹסָפוֹת:

שֶׁהַדָּם מַתִּיר אֵמוּרִים לַמִּזְבֵּחַ. רַשִׁ"י. וְכֵן צָרִיךְ לוֹמַר בְּהָרַ"ב:

עַיֵּן מִשְׁנָה ה' פֶּרֶק ח', וּפֶרֶק ט' מִשְׁנָה ד'. וּלְשׁוֹן הָרַמְבַּ"ם, כִּי טֻמְאַת מֵת בִּלְבַד הִיא שֶׁתִּדְחֶה בְצִבּוּר: הַכֹּהֲנִים טְמֵאִים. דְּהַיְנוּ צִבּוּר טְמֵאִים. דְּמַה לִּי אִם הָאֳכָלִין טְמֵאִים אוֹ אִם הַמַּקְרִיבִים טְמֵאִים, שֶׁאִי אֶפְשָׁר בְּלֹא כֹהֲנִים, וְכָתְבוּ הַתּוֹסָפוֹת אַף עַל פִּי שֶׁיְּכוֹלִין לִזָּהֵר שֶׁלֹּא יִגְּעוּ בַבָּשָׂר, כֵּיוָן דְּאִשְׁתְּרַאי טֻמְאַת אֵמוּרִים אִשְׁתְּרַאי נַמִּי טֻמְאַת בָּשָׂר:

וַאֲפִלּוּ לְרַבִּי יהוֹשֻׁעַ דִּסְבִירָא לֵיהּ דִּבְיָחִיד אֵין הַצִּיץ מְרַצֶּה עַל הַטֻּמְאָה שֶׁנָּגַע בְּדָבָר הַנֶּאֱכָל כְּגוֹן בָּשָׂר, הָנֵי מִלֵּי לְכַתְּחִלָּה, אֲבָל בְּדִיעֲבַד מְרַצֶּה כוּ'. וְהָא דִמְרַצֶּה, לֹא שֶׁיְּהֵא בָשָׂר טָמֵא נֶאֱכָל, דְּאֵין צִיץ מְטַהֵר וְדוֹחֶה לֹא תַעֲשֶׂה כוּ', אֶלָּא דִמְרַצֶּה לְהַתִּיר דָּם לִזְרִיקָה. רַשִׁ"י: וּמַשְׁמַע אֲפִלּוּ קֹדֶם הַגִּלּוּחַ אֵינוֹ סוֹתֵר. וּבְמִשְׁנָה ב' פֶּרֶק ט' דְּנָזִיר כָּתַב דְּנוֹדַע קֹדֶם הַגִּלּוּחַ סוֹתֵר. וְעַיֵּן מַה שֶּׁכָּתַבְנוּ שָׁם:

מֵעֲצֵי הַמַּעֲרָכָה. הַטַּעַם מִפְּנֵי שֶׁלֵּב בֵּית דִּין מַתְנֶה עֲלֵיהֶם, לְפִי שֶׁהִזְקִיקוּהוּ לְשָׂרְפוֹ לִפְנֵי הַבִּירָה כְּדֵי לְבַיְּשׁוֹ, לֹא הִטְרִיחוּהוּ לְהָבִיא עֵצִים מִשֶּׁלּוֹ, וְיָכוֹל לֵהָנוֹת מִשֶּׁל הֶקְדֵּשׁ. וְכֵן בַּצַּיְקָנִין הִתְנוּ כְּמוֹ כֵן, דְּמִתּוֹךְ צַיְקָנוּתָן הָיוּ נִמְנָעִים מִלְּשָׂרְפוֹ וַאֲתוֹ לִידֵי עֲבֵרָה: בַּגְּמָרָא, שֶׁאִם בָּא לְשָׂרְפוֹ לִפְנֵי הַבִּירָה מֵעֲצֵי עַצְמוֹ אֵין שׁוֹמְעִין לוֹ:

כְּדִין קָדָשִׁים קַלִּים שֶׁמְּחִצָּתָן חוֹמַת יְרוּשָׁלַיִם:

הָכִי קָאָמַר רַב אַשִׁי בַגְּמָרָא. וּבְפֶרֶק ב' דְּשַׁבָּת מְפָרֵשׁ הָרַ"ב כְּתָנָא דְבֵי חִזְקִיָּה: בְּשַּׁבָּת. דְּלֹא תֵימָא דְּלֹא לִדְחוּ לִשְׂרֵפָתָם כֻּלֵּי הַאי, קָא מַשְׁמַע לָן:

דְּבָצִיר מֵהָכִי לֹא סַגִּי, דִּכְתִיב וּמִיּוֹם הַשְּׁמִינִי וָהָלְאָה יֵרָצֶה: בַּטָּהוֹר. אֲפִלּוּ בַטָּהוֹר. רַשִׁ"י. כְּלוֹמַר וְאֵין צָרִיךְ לוֹמַר בְּפֶסַח הַבָּא בְטֻמְאָה. וְכֵן הַשּׁוֹבֵר בַּטָּמֵא פֵּרֵשׁ הָרַמְבַּ"ם בְּפֶסַח שֶׁבָּא בְטֻמְאָה:

אֵבָר שֶׁיָּצָא. בְּאַרְבָּעָה עָשָׂר מְחִצָּתוֹ חוֹמַת יְרוּשָׁלַיִם, וּבְלֵיל חֲמִשָּׁה עָשָׂר מְחִצָּתוֹ הַבַּיִת שֶׁנֶּאֱכָל בּוֹ: וְחוֹתֵךְ. שֶׁהָעֶצֶם שֶׁהָיָה עָלָיו כַּזַּיִת בָּשָׂר וּשְׁבָרוֹ שֶׁלֹּא בִמְקוֹם הַבָּשָׂר חַיָּב אַף עַל פִּי שֶׁהַמָּקוֹם שֶׁשָּׁבַר פָּנוּי מִבְּשָׂרוֹ: בְּאֵין בּוֹ מֹחַ עָסְקִינַן, דְּאִי בְיֵשׁ בּוֹ מֹחַ הָא מְפֹרָשׁ בְּמִשְׁנָה י' דְּטָעוּן שְׂרֵפָה. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: וְלִפְנִים כְּלִפְנִים. הָא הָאֲגַף עַצְמוֹ נַעֲשֶׂה כְמִן הָאֲגַף וְלַחוּץ. הָרַמְבַּ"ם: וְהוּא הַדִּין לְקָדְשֵׁי קָדָשִׁים בָּעֲזָרָה וְלַאֲכִילַת הַפֶּסַח בַּבַּיִת. הָרַמְבַּ"ם:

וַהֲוָה מָצֵי לְאַשְׁמוֹעִינַן בְּמַתְנִיתִין רְבוּתָא דְּאוֹכְלִין בִּשְׁנֵי בָתִּים. אֶלָּא דְבִכְהַאי גַּוְנָא נִיחָא לֵיהּ לְמִנְקַט עֲלַהּ דִּינָא דְשַׁמָּשׁ וּדְכַלָּה. וְהָרַמְבַּ"ם פֵּרֵשׁ אֵלּוּ הוֹפְכִין, צְרִיכִין לַהֲפֹךְ כְּדֵי שֶׁלֹּא יִהְיוּ נִרְאִין כַּחֲבוּרָה אַחַת, דְּאִיהוּ תָנֵי בַבְּרַיְתָא מִן הַבָּשָׂר חוּצָה , תֵּן חוּצָה. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: וְצָרִיךְ לוֹמַר דְּאַשְׁמְעִינַן דְּלֹא חָשְׁבִינַן כִּשְׁנֵי מְקוֹמוֹת אַף עַל פִּי שֶׁאָכְלָה מִקְצָת סְעוּדָה בְּלֹא הֲפִיכַת פָּנִים. אִי נַמִּי קָא מַשְׁמַע לָן (דְּחַיֶּבֶת) לַהֲפֹךְ פָּנֶיהָ מִשּׁוּם צְנִיעוּת. תּוֹסָפוֹת:
