פרק ג - משנה א

יְרוּשַׁלְמִי. וְנִרְאֶה דְלָאו דַּוְקָא, וְהוּא הַדִּין סְמָדַר דְּפָחוּת מִבֹּסֶר, אֶלָּא נָקֵיט בֹּסֶר לְאַשְׁמְעִינַן שֶׁעֲדַיִן לֹא בָא לִכְלַל עוֹנַת מַעֲשֵׂר. תּוֹסְפוֹת יוֹם טוֹב: בְּגֶמִי. וְהַכַּוָּנָה בָזֶה שֶׁיִּרְשֹׁם אוֹתָם בְּאֵיזֶה דָבָר שֶׁיִּזְדַּמֵּן כְּדֵי שֶׁיַּכִּיר אוֹתָם. וּמַה שֶּׁיִּחֵד גֶּמִי, כְּפִי הַנָּהוּג. הָרַ"מ: וְאֵינִי יוֹדֵעַ הֵיכִי מַפִּיק לַהּ מִפְּרִי, דְּהָא אַף בִּמְחֻבָּר פְּרִי הוּא. תּוֹסְפוֹת יוֹם טוֹב:

פרק ג - משנה ג

שָׁלְחוּ לִפְנֵיהֶם. שְׁלוּחִים לְהוֹדִיעַ לְאַנְשֵׁי יְרוּשָׁלַיִם: וְהוּא הַדִּין לִשְׁאָר פֵּרוֹת. וְזֶה לְשׁוֹן הָרַ"מ, אִם הָיָה הַכֹּל רְטֻבִּים הָיָה מַרְאֶה הַטּוֹב וְהַיָּפֶה שֶׁבְּכֻלָּם: וּלְפִי זֶה הֵם עַל מִנּוּי אֶחָד הֻנְּחוּ. אֲבָל לְשׁוֹן הָרַ"מ וְהַסְּגָנִים, וּלְפִי זֶה נִרְאֶה דְהַסְּגָנִים סְגָנֵי לְוִיָּה הֵם, וּסְגָנֵי כְהֻנָּה קוֹרֵא בְּשֵׁם יוֹתֵר נִכְבָּד, וְהוּא פַּחוֹת, וּמֵעֵין שֵׁם מֶמְשָׁלָה הוּא כְּמוֹ זְרֻבָּבֶל פַּחַת יְהוּדָה (נְחֶמְיָה ה):

פרק ג - משנה ד

אַגְרִיפַּס. הַמֶּלֶךְ, מִמַּלְכֵי בַיִת שֵׁנִי, בַּעַל נֶפֶשׁ גְּדוֹלָה וּשְׂרָרָה רַבָּה. הָרַ"מ:

פרק ג - משנה ו

רַבִּי יְהוּדָה אוֹמֵר. הַתּוֹסְפוֹת יוֹם טוֹב הֶאֱרִיךְ לְפָרֵשׁ בְּמַאי פְּלִיגֵי תַּנָּא קַמָּא וְרַבִּי יְהוּדָה. וּמַסִּיק דְּצָרִיךְ לוֹמַר דְּרַבִּי יְהוּדָה לֹא בָא לַחֲלֹק אֶלָּא לְפָרֵשׁ דִּבְרֵי תַנָּא קַמָּא, דְּזֶהוּ שֶׁאָמַר עוֹדֵהוּ הַסַּל עַל כְּתֵפוֹ קוֹרֵא מֵהִגַּדְתִּי עַד כוּ' דְּלָאו לְמֵימְרָא דְכָל זֶה בְּעוֹד הַסַּל עַל כְּתֵפוֹ, שֶׁזֶּה אֵינוֹ אֶלָּא עַד אֲרַמִּי כוּ', וּמִשֶּׁהִגִּיעַ לַאֲרַמִּי כוּ' מוֹרִיד כוּ', וְתַנָּא קַמָּא לֹא נָחֵית לְפֵרוּשָׁא מַאי דְעָבֵיד בֵּינֵי וּבֵינֵי, אֶלָּא הַתְחָלַת הָעִנְיָן הוּא אוֹמֵר, שֶׁבְּעוֹד עַל כְּתֵפוֹ מַתְחִיל לִקְרֹא הִגַּדְתִּי. וּמַה שֶּׁאוֹמֵר וְגוֹמֵר כָּל הַפָּרָשָׁה אֵינוֹ מוּסָב עַל עוֹדֵהוּ עַל כְּתֵפוֹ, אֶלָּא אוֹמֵר שֶׁגּוֹמֵר כָּל הַפָּרָשָׁה כֻלָּהּ. וְלֹא בֵאֵר בְּאֵיזֶה עִנְיָן, וְאָתָא רַבִּי יְהוּדָה וּמְבָאֲרוֹ וְהוֹלֵךְ. וְעַיֵּן עוֹד: וְהַתּוֹסָפוֹת בִּמְנָחוֹת דַּף ס"א כָּתְבוּ דְּתַחַת יַד לָאו דַּוְקָא, אֶלָּא הַבְּעָלִים אוֹחֲזוֹ בְשִׂפְתוֹתָיו. וּמִיהוּ הֵבִיאוּ בְּשֵׁם הַיְרוּשַׁלְמִי דְּמוּכָח שֶׁתַּחַת יָד מַמָּשׁ: יְרוּשַׁלְמִי, דִּכְתִיב לִפְנֵי ה', יָכוֹל בַּמַּעֲרָב שֶׁהַהֵיכָל בְּמַעֲרַב הַמִּזְבֵּחַ, תַּלְמוּד לוֹמַר אֶל פְּנֵי הַמִּזְבֵּחַ. אִי (לִפְנֵי) הַמִּזְבֵּחַ, יָכוֹל בַּדָּרוֹם שֶׁשָּׁם הַכֶּבֶשׁ וְהוּא פְּנֵי הַמִּזְבֵּחַ, תַּלְמוּד לוֹמַר לִפְנֵי ה'. הָא כֵּיצַד, מַגִּישִׁין עַל קֶרֶן דְּרוֹמִית מַעֲרָבִית. וְכֵן הוּא בַגְּמָרָא מְנָחוֹת דַּף י"ט:

פרק ג - משנה ז

שֶׁאֵינוֹ יוֹדֵעַ. שֶׁמִּקְרָא בִכּוּרִים בִּלְשׁוֹן הַקֹּדֶשׁ, כְּדִתְנַן בְּמִשְׁנָה ג' פֶּרֶק ז' דְּסוֹטָה:

פרק ג - משנה ח

בְּסַלֵּי נְצָרִים כוּ'. נְצָרִים הֵם עֲנָפִים אוֹ שְׁבָטִים, כְּמוֹ וְנֵצֶר מִשָּׁרָשָׁיו יִפְרֶה (יְשַׁעְיָה יא). וּפֵרוּשׁ קְלוּפָה, שֶׁמְּסִירִין קְלִפָּתָהּ, תַּרְגּוּם מַחְשׂוֹף הַלָּבָן קִלוּף. הָרַ"מ: וְקַשְׁיָא לִי, דְּאִי הָכִי עֲנִיִּים מִתְבַּיְּשִׁים, וְלָמָּה לֹא תִקְּנוּ כְּדֵי שֶׁלֹּא יִתְבַּיְּשׁוּ הָעֲנִיִּים שֶׁאַף הָעֲשִׁירִים יָבִיאוּ בִּנְצָרִים כְּמוֹ שֶׁמָּצִינוּ בְּכַמָּה מְקוֹמוֹת. וְיֵשׁ לוֹמַר מִשּׁוּם כְּבוֹד בֵּית אֱלֹהֵינוּ וּמְשָׁרְתָיו לֹא חָשׁוּ עַל שֶׁיִּתְבַּיְּשׁוּ הָעֲנִיִּים. תּוֹסְפוֹת יוֹם טוֹב:

פרק ג - משנה י

חַיָּב בַּדְּמַאי. וְצָרִיךְ עִיּוּן אִי נֶאֱכַל אַף בְּטֻמְאָה, הוֹאִיל וְלֹא תָנֵי בְּהֶדְיָא. תּוֹסְפוֹת יוֹם טוֹב: יְרוּשַׁלְמִי. אֲבָל הָא דְבָעֵי לְמֵילָף מִנֵּיהּ דְּהוּא הַדִּין לַתּוֹסֶפֶת, מִלְּתָא דִתְמִיהָה הוּא, דְּהָא חָמוּר בַּתּוֹסֶפֶת מִבָּעִטּוּר כְּדִלְעֵיל, וּבִכְלַל מִין בְּשֶׁאֵינוֹ מִינוֹ הוּא. תּוֹסְפוֹת יוֹם טוֹב:

פרק ג - משנה יב

וְאִשָּׁה בִּכְתֻבָּתָהּ. לִגְבּוֹת מֵחַיִּים אִם נִתְגָּרְשָׁה. אִי נַמִּי דְּתָפְסָה מֵחַיִּים, וַאֲפִלּוּ לְרַבָּנָן דְּרַבִּי מֵאִיר דְּמִשְׁנָה ז' פֶּרֶק ח' דִּכְתֻבּוֹת: וּפֵרוּשׁוֹ דָּחוּק. וְרַבִּי יִצְחָק בְּרַבִּי מַלְכִּי צֶדֶק פֵּרֵשׁ דִּכְמוֹ דְסֵפֶר תּוֹרָה אַף עַל גַּב דִּמְכָרָהּ קְדֻשָּׁתֵיהּ עֲלֵיהּ, דְּסֵפֶר תּוֹרָה הוּא, הָכִי נַמִּי הַבִּכּוּרִים בִּכּוּרִים נִינְהוּ וְלֹא נָפְקֵי מִקְּדֻשַּׁתַיְהוּ לְהֵאָכֵל לְזָרִים, אֶלָּא לְעוֹלָם צָרִיךְ לְזַבְּנִינְהוּ לְכֹהֲנִים טְהוֹרִים לְמֵכְלִינְהוּ בְּטָהֳרָה, וְהַיְנוּ דְקָתָנֵי כְּסֵפֶר תּוֹרָה: פֵּרוּשׁ, הַכֹּהֵן לְאַחַר שֶׁבָּאוּ לְיָדוֹ. וּבְטוֹבָה פָּלֵיג עִם תַּנָּא קַמָּא, וּבְאֵין נוֹתְנִין אֶלָּא לְחָבֵר דְּקָאָמַר רַבִּי יְהוּדָה מְהַדְרֵי לֵיהּ חֲכָמִים, נוֹתְנִין אוֹתָהּ לְאַנְשֵׁי מִשְׁמָר כוּ'. וְאֵין נוֹתְנִין דְּרַבִּי יְהוּדָה בַּכֹּהֲנִים, וְנוֹתְנִים דַּחֲכָמִים בַּבְּעָלִים:
