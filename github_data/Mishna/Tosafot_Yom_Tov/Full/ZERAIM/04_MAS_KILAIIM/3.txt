פרק ג

עַרוּגֹה. לֵשִוֲן עִבֵרִי הוּא, מְעַרֶגוֲת מָטֹּעֹהּ (יֵחּזֵקְאל יז): זוֲרֵעִים כוּ'. כֵּדִתֵנָן בֵּפּרּק ט' דֵּשִָבֹּת כִּי כֹאֹרּץ תּוֲצִיא צִמֵחֹהּ וּכֵגָנֹּהּ זְרוּעּיהֹ תָצֵמִיחָ. וֵשִֹם פְּרָשִֵתִּיו בֵּסִיָּעֵתֹּא דִשִֵמָיֹּא: זְרֵעוֲנִים. זּה הָשְִּם זְרֵעוֲנִים כּוֲלְל לֵכֹל הָנִּזֵרֹע, אַבֹל בֵּכֹאן הָכָּוֹּנֹה לִזֵרֹעִים מֵיֶחֹדִים וֵהְם הָזּּרָע שִּל מִינְי יֵרֹקוֲת וּכֵדִתֵנָן בֵּרְישִ מָתֵנִיתִין דִּלֵקָמֹּן. הֹרָ"מ: וֵאִם תֲּאמָר יִזֹּרֵעוּ תִּשִֵעֹה מִינִין בֹּעַרוּגֹה, אָרֵבֹּעֹה בֵּאָרֵבָּע קֵרֹנוֲת, וֵאָרֵבֹּעֹה בֵּאּמֵצָע כֹּל רוּחָ, וֵאּחֹד בֹּאּמֵצָע, דֵּכֹל אּחֹד רֹחוֲק מְחַבְרוֲ שִֵלֲשִֹה טֵפֹחִים. וֵיְשִ לוֲמָר דֵּכֶלְּי הָאי הַוְי עִרֵבּוּב. עוֲד יְשִ לוֲמָר דֵּהָשִֵתֹּא יֹכוֲל לִזֵרֲעָ הָרֵבְּה מִכֹּל מִין שִּבֹּרוּחָ, תּוֲסֹפוֲת. וֵכֹתֵבוּ עוֲד דֵּאָף עָל גָּב דֵּבָעַרוּגֹה שִּהִיא חַמִשִֹּה עָל חַמִשִֹּה יְשִ בָּאַלָכֵסוֲנֹהּ שִִבֵעֹה, דֵּכֹל אָמֹּתֹא בֵּרִבּוּעָ אָמֹּתֹא וּתֵרְי חֶמֵשְִי בָּאַלָכֵסוֲנֹהּ, וֵיֹכוֲל לִזֵרֲעָ בֵּאָרֵבָּע קֵרֹנוֲת וּבֹאּמֵצָע, וֵתְרְץ רִ"י דֵּהֹא מִקֵּרֹא יֹלֵפִינָן וֵאְין לֵהָעַמִיד הָמִּקֵרֹא בֵּעִנֵיֹן זּה שִּאְינוֲ שִִעוּר לֲא לִינִיקֹה וֵלֲא לֵשִוּם דֹּבֹר: שִּלֲּא אֹסֵרֹה תּוֲרֹה אּלֹּא בֵּמָפֲּלּת יֹד וּקֵרֹא דֵּהֹכֹא דֵּקֹפְּיד אָעִרֵבּוּב אְינוֲ אּלֹּא אָסֵמָכֵתֹּא: וּמִכֹּל מֹקוֲם צֹרִיךֵ שִּיָּנִּיחָ קֵצֹת לֵהָפֵרִישִ בְּינְיהּן כִּדֵפְרְשִ רָשִִ"י בָּגֵּמֹרֹא: וֵשִִעוּר שִֵלֲשִֹה טֵפֹחִים הָיֵנוּ עִם מֵקוֲם הָזֵּרֹעִים. כֹּתָב הֹרָ"שִ וֵהֹא דִתֵנָן לֵעְיל בֵּפּרּק ב' יֹרֹק בֵּיֹרֹק שִִשִֹּה טֵפֹחִים הָנְּי מִילְי בֵּשֵֹדּה גֵדוֲלֹה שִּיְּשִ בֹּהּ יֹרֹק הָרֵבְּה, אַבֹל הֹכֹא דֵּאְין זוֲרְעָ מִכֹּל אּחֹד וֵאּחֹד אּלֹּא מֵעָט לֲא חָיֵשִִינָן, וֵדִכֵוֹתָהּ אָשִֵכֵּחֹן גָּבְּי תֵבוּאֹה בְּית רֲבָע דֵּאֹמֵרִינָן בָּיֵרוּשִָלֵמִי שִּרָשִָּאי לִזֵרֲעָ חַמִשִֹּה מִינִין בֵּבְית רֲבָע אּחֹד דִּבֵכֹל מֹקוֲם שִּאְין זוֲרְעָ מִכֹּל מִין וֹמִין אּלֹּא מֵעָט לֲא חָיֵשִִינָן אּלֹּא לִינִיקֹה: גֹּבוֲהָּ טּפָח. וֵגָם בֵּרֹחֵבּוֲ טּפָח כִּדֵאִיתֹא בִּגֵמֹרֹא שִָבֹּת דָּף פ"ה מִדֵּרָבִּי יֵהוּדֹה דֵּאֹמָר לֵקָמֹּן מִשִֵנֹה ג' רֲחָב הָפָּרֵסֹה, וֵכֹתֵבוּ הָתּוֲסֹפוֲת דֵּאָף עָל גָּב דֵּרָבִּי יֵהוּדֹה לֵקָמֹּן לֲא מָיֵרְי אּלֹּא לֵעִנֵיָן הּפֵסְק, נִרֵאּה לָגֵּמֹרֹא דֵּלֲא שִֵנֹא לִגֵבוּל דֵּהוּא הָדִּין נָמִּי שִּשִִּעוּרוֲ טּפָח: שִֵלֲשִֹה כוּ'. וֵתְימָהּ שִּלִּשִֵנְים עֹשֵֹר זֵרֹעִים צֹרִיךֵ שִֵתְּים עּשֵֵרְה מּרֵחֹקִין וֵכֹל מּרֵחֹק שִֵלֲשִֹה טֵפֹחִים הְם שִֵלֲשִִים וֵשִִשִֹּה טֵפֹחִים, וֵאְין בֵּכֹל הָגֵּבוּלִים סֹבִיב כִּי אִם עּשֵֵרִים וּשִֵמֲנֹה טֵפֹחִים, לִשִֵנָיִם לֵכֹל אּחֹד שִֵמֲנֹה טֵפֹחִים וֵלִשִֵנָיִם לֵכֹל אּחֹד שִִשִֹּה טֵפֹחִים. וֵלֹכְן נִרֵאּה לִי שִּעָל כֹּל גֵּבוּל הוּא זוֲרְעָ שִֵלֲשִֹה. וֵכֹךֵ יֹכוֲל לָעַשֵוֲת, שִּמָּנִּיחָ טּפָח הָסֵּמוּכֹה לָקּּרּן וֵזוֲרְעָ מִין אּחֹד וּמָנִּיחָ שִֵלֲשִֹה טֵפֹחִים וֵזוֲרְעָ, וּמָנִּיחָ עוֲד שִֵלֲשִֹה טֵפֹחִים וֵזוֲרְעָ, וֵנִמֵצֹא שִּיְּשִ עוֲד טּפָח פֹּנוּי בָּקּּרּן הָשְִּנִי, וֵכְן הוּא עוֲשֵּה לֵכֹל גֵּבוּל וּגֵבוּל, וֵנִמֵצֹא שִּיְּשִ בְּין מִין וֹמִין שִֵלֲשִֹה טֵפֹחִים שִִעוּר הָרֵחֹקֹה, מִלֵּבָד בֵּאוֲתֹן שִּסֵּמוּכוֲת לָקֵּרֹנוֲת שִּאְין בְּינְיהּם הָרֵחֹקֹה לֵפִי שִּאְין צֵרִיכִין, שִּיְּשִ הּכְּר בָּדּּרּךֵ שִּנִּזֵרֹעִין שִּזּּה נִזֵרֹע צֹפוֲן וֵדֹרוֲם וֵזּה מִזֵרֹח וּמָעַרֹב כִּדֵכֹתָב הֹרָ"ב בֵּרְישִֹא. שִּהַרְי כֹּתָבֵתִּי שִּרֲחָב הָגֵּבוּל טּפָח, וֵנִמֵצֹא שִּזּוֲרְעָ בֵּאֲרּךֵ טּפָח וֵיְשִ הּכְּר בִּזֵרִיעֹתֹן כֵּמוֲ שִּכֹּתָבֵתִּי. מָה שִּּאְין כְּן בֵּרְישִֹא שִּאִם יִזֵרָע תִּשִֵעֹה הַוְי עִרֵבּוּב כֵּמוֲ שִּכֹּתֵבוּ הָתּוֲסֹפוֲת, לֵפִי שִּאְין כֹּאן הּכְּר, וֵעוֲד שִּלֲּא יוּכָל לִזֵרֲעָ מִכֹּל מִין כִּי אִם גָּרֵעִין אּחֹד. תּוֲסֵפוֲת יוֲם טוֲב: מִפֵּנְי כוּ'. וֵהָרֵחֹקָת שִֵלֲשִֹה טֵפֹחִים לֲא מֵהָנְּי דֵּאָכָּתִּי הַוְי נִרֵאּה מֵעֶרֵבֹּב וֵאָף עָל פִּי שִּאְין יוֲנֵקִין זּה מִזּּה. אַבֹל בָּעַרוּגֹה פֵּשִוּטֹה דֵרְישִֹא רָשִָּאי לִטָּע אַפִלּוּ רֲאשִ לּפּת בֹּאּמֵצָע, לֵפִי שִּאְין כֹּאן אּלֹּא אָרֵבֹּעֹה מִינִין לֵאָרֵבָּע הֹרוּחוֲת לֲא נִרֵאּה כֵּעִרֵבּוּבֵיֹא. כֹּךֵ נִרֵאּה לִי. תּוֲסֵפוֲת יוֲם טוֲב: שִִשִֹּה. דֹּרָשִ זּרָע זָרֵעֹהּ זֵרֹעּיהֹ זְרוּעּיהֹ כֵּלוֲמָר דֵּמֹצְי לֵמִכֵתֹּב זּרָע דֵּמָשִֵמָע חָד וֵזָרֵעֹהּ מָשִֵמָע תֵּרְי וּזֵרֹעּיהֹ מָשִֵמָע תֵּלֹת וּכֵתִיב זְרוּעּיהֹ הַרְי אָרֵבֹּעֹה וּכֵתִיב תּוֲצִיא צִמֵחֹהּ תּוֲצִיא חָד וֵצִמֵחֹהּ חָד כֵּמוֲ לֵתָנֹּא קָמֹּא, הַרְי שִִית. וֵכִי רֹמְיז קֵרֹא שִִשִֹּה זְרֵעוֲנִים בָּעַרוּגֹה בָּת שִִשִֹּה טֵפֹחִים, דֵּקִים לְיהּ לֵרָבִּי יֵהוּדֹה דֵּשִִית בֵּשִִית לֲא יֹנֵקְי, כִּי הְיכִי דִּלֵרָבֹּנֹן בָּחַמִשִֹּה בֵּשִִשִֹּה כוּ'. וֵלֲא אִתֵבֹּרְר כָּמֹּה הוּא שִִעוּר הּרֵחְק לֵרָבִּי יֵהוּדֹה. וֵעָיְּן תּוֲסֵפוֲת יוֲם טוֲב: בֹּאּמֵצָע. וֵרָבִּי יֵהוּדֹה פֹּלְיג בְּין בֵּרְישִֹא וּבְין בֵּסְיפֹא, וֵהֹכִי אִיתֹא בָּיֵרוּשִָלֵמִי:

כֵּלוֲמָר יוֲתְר מְעַרוּגֹה, וֵלֲא שֵֹדּה גֵדוֲלֹה מָמֹּשִ, דֵּהֹא בֵּמִשִֵנֹה ט' לֵעְיל כֹּתָב שִּאְין רֵגִילוּת לִזֵרֲעָ מִן הָחָרֵדֹּל שֵֹדּה גֵדוֲלֹה. תּוֲסֵפוֲת יוֲם טוֲב: נִרֵאּה דֵגֹרָס בֵּמָתֵנִיתִין גּוֲבְל וֵלֲא גֵּבוּל: הָתּּלּם. וֵרֲחָב הָתּּלּם שִִשִֹּה, וֵכְן בֵּאָמָּת הָמָּיִם, וֵזוֲרֵעִים לֵרֹחֵבֹּהּ שִּל אָמֹּה שִּהִיא כֵּמוֲ כְן שִִשִֹּה, וּכֵשִּיֹּבֵשִוּ הָמָּיִם דָּחַזִי לִזֵרִיעֹה. הֹרָ"שִ: הֹרָ"שִ. וּלֵטָעַמְיהּ אֹזְיל שִּפְּרְשִ לֵעְיל דֵּהָהָרֵחֹקֹה הִיא טּפָח וּמּחֱצֹה וּכֵדָעָת הֹרָ"מ, אַבֹל הֹרָ"ב שִּסּוֲבְר כֵּרָשִִ"י שִּצֹּרִיךֵ שִֵלֲשִֹה טֵפֹחִים הִנְּה בֵּרֲחָב הָשִִּשִֹּה לֲא יוּכָל לִזֵרֲעָ כִּי אִם שִֵלֲשִֹה גָרֵעִינִין, וֵלֹכְן אַנִי אוֲמְר שִּלֵּפוּם רִיהַטֹא הּעֵתִּיק לֵשִוֲן הֹרָ"שִ. תּוֲסֵפוֲת יוֲם טוֲב:

וֵעָד רֲאשִוֲ. שִֵתְּי הָקֵּצֹוֲת כֹּל אּחֹד רֲאשִ וֹסוֲף: כֵּלוֲמָר שִּהָהּפֵסְק הוּא הָזּּרָע הֹאָחְר שִּנִּזֵרָע בָּתּּלּם, אָף עָל פִּי שִּמָּקִּיף בִּשִֵלֲשִֹה רוּחוֲת זּרָע הָשֵֹּדּה, כְּיוֹן שִּהוּא תּּלּם וּבָצָּד הֹרֵבִיעִי אְין בּוֲ זּרָע שִּעָכֵשִֹיו אְינוֲ חֹבוּשִ הַוְי הּפֵסְק. וֵעָיְּן תּוֲסֵפוֲת יוֲם טוֲב בֵּשְִם הֹרָ"שִ: וֵצֹרִיךֵ לֵחָלְּק בְּין תּּלּם סֵתֹם דֵּהֹכֹא לֵתּלּם שִּל פֹּתִיחָ בֵּמִשִֵנֹה ו' פּּרּק ב':

וּמָשִֵמָע דֵּכִי נִסֵמֵכוּ מִיהוּ שִֵתְּי הָשִּוּרוֲת שִּל קִשִּוּאִין שִֹרְי, וֵאִם כְּן הֹא דִתֵנָן בֵּרְישִֹא שִֵתְּי שִוּרוֲת שִּל דֵּלוּעִים שִֵתְּי שִוּרוֲת שִּל פּוֲל לֹאו דָּוֵקֹא אּלֹּא כִּי הַוֹה שִֵתְּי שִוּרוֲת שִּל קִשִּוּאִין תּוּ שִֹרְי אַפִלּוּ אָחָת שִּל דֵּלוּעִים וֵאָחָת שִּל פּוֲל דֵּהֹא אִיכֹּא שֵֹדּה, וֵרָשִָּאי לִּסֵמֲךֵ יֹרֹק אָחְר בֵּהָפֵסֹקָת תּּלּם. תּוֲסֵפוֲת יוֲם טוֲב:

בֵּצֹלִים. וֵיִחְד בֵּצֹלִים מִפֵּנְי שִּאּפֵשִֹר שִּכְּן הֹיֹה הָמִּנֵהֹג אּצֵלֹם. הֹרָמֵבָּ"ם: שִּל דֵּלוּעִים. וֵלֲא דֹמְי לָמָּתֵנִיתִין דִּלֵעְיל שִּיֹּכוֲל לִטָּע שִֵתְּי שִוּרוֲת שִּל קִשִּוּאִים וּשִֵתְּי שִוּרוֲת שִּל דֵּלוּעִים וּשִֵתְּי שִוּרוֲת שִּל פּוֲל הָמִּצֵרִי, דֵּשִֹאנְי שִוּרֹה מִשִֵּתְּי שִוּרוֲת, וֵעוֲד דֵּכְיוֹן שִּבֹּא לָעַקֲר חֹמוּר טֵפְי כִּדֵאָשִֵכֵּחֹן בָּיֵרוּשִָלֵמִי: כֵּלוֲמָר וּתֵנָן לֵעְיל בֵּפּרּק ב' מִשִֵנֹה ח' דֵּבוּר הַוְי הּפֵסְק:

וֵטָעַמֹא דֵּכִלֵאְי כּרּם אַסוּרִים בָּהַנֹאֹה וֵכִלֵאְי זֵרֹעִים מֶתֹּרִים אַפִלּוּ בָּאַכִילֹה כֵּדִתֵנָן בֵּרְישִ פּּרּק ח':
