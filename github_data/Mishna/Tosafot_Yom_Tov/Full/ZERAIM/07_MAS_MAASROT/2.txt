פרק ב - משנה א

דְּאִלּוּ רָאוּ פְּנֵי הַבַּיִת עַל יְדֵי בַעַל הַבַּיִת וְהֻקְבְּעוּ לְמַעֲשֵׂר טָבֵיל לְהוּ וַאֲסוּרִים לְכָל אָדָם: לְבָתֵּיכֶם. דַּוְקָא לְבָתֵּיכֶם, דְּאִלּוּ לְבֵיתִי אֵין בַּיִת אַחֵר טוֹבֵל כְּדִלְקַמָּן. תּוֹסְפוֹת יוֹם טוֹב:

פרק ב - משנה ב

וְצָרִיךְ לוֹמַר שֶׁהַפֵּרוֹת מֻנָּחִים בַּשּׁוּק, דְּאִלּוּ הַפֵּרוֹת הֵמָּה גַם כֵּן בַּשַּׁעַר אוֹ בַחֲנוּת, אִם כֵּן טְבוּלִים וְהֻקְבְּעוּ לַמַּעֲשֵׂר וַאֲסוּרִין גַּם לַאֲחֵרִים. וְהוּא הַדִּין דַּהֲוָה מָצֵי לְמִתְנֵי הָיוּ יוֹשְׁבִים בַּבַּיִת, אֶלָּא שֶׁדֶּרֶךְ הַמּוֹכֵר פֵּרוֹת לֵישֵׁב בַּשַּׁעַר אוֹ בַחֲנוּת: דִּבְרֵי רַבִּי נְחֶמְיָה בְּמִשְׁנָה ה' פֶּרֶק ג'. וּבַיְרוּשַׁלְמִי, רַבִּי יְהוּדָה וְרַבִּי נְחֶמְיָה אָמְרוּ דָּבָר אֶחָד:

פרק ב - משנה ג

דְּאִלּוּ הִגִּיעַ הַשַּׁבָּת טָבְלוּ אַף עַל פִּי שֶׁלֹּא יִחֲדָם לְצֹרֶךְ הַשַּׁבָּת. יְרוּשַׁלְמִי: הָרִאשׁוֹן. כְּלוֹמַר כֵּיוָן שֶׁהִגִּיעַ לַבַּיִת הָרִאשׁוֹן שֶׁבָּעִיר נִטְבְּלוּ הַפֵּרוֹת אַף עַל פִּי שֶׁלֹּא לָן שָׁם. הָרַ"שׁ:

פרק ב - משנה ד

וַחֲכָמִים אוֹסְרִים. טַעֲמַיְהוּ, דְּסָבְרֵי דִּכְשֶׁכּוֹנְסִין תְּאֵנִים בַּכַּלְכָּלָה יֵשׁ שֶׁמַּנִּיחִין אוֹתָן כָּךְ וַהֲוָה גְמַר מְלָאכָה, אַף עַל גַּב דְּלָאו כֻּלֵּי עַלְמָא עָבְדֵי הָכִי אֶלָּא מַחֲלִיקִים אוֹתָם כִּדְאִיתָא סוֹף פֶּרֶק דִּלְעֵיל. תּוֹסְפוֹת יוֹם טוֹב:

פרק ב - משנה ה

בַּגְּמָרָא פֶּרֶק ז' דְּבָבָא מְצִיעָא אַסְמְכִינַן אַקְּרָא כִּי קִבְּצָם כֶּעָמִיר גֹּרְנָה (מִיכָה ד), וּפֵרֵשׁ רַשִׁ"י אֵין גֹּרֶן בְּלֹא קִבּוּץ, וְאֵין קִבּוּץ פָּחוֹת מִשְּׁתַּיִם:

פרק ב - משנה ו

וְאוֹכֵל. פֵּרוּשׁ בְּעוֹדוֹ בִמְחֻבָּר כְּמוֹ שֶׁכָּתַב הָרַמְבַּ"ם, שֶׁלֹּא יִכְרֹת הָאֶשְׁכּוֹל כֻּלּוֹ כְּדֵי שֶׁלֹּא יִתְחַיֵּב בַּמַּעַשְׂרוֹת, שֶׁהֲרֵי לֹא קָנָה אֶלָּא הַנִּתְלַשׁ. מַה שֶּׁאֵין כֵּן בְּסֵיפָא דְּאָמְרִינַן אֵלּוּ בִמְחֻבָּר קָנָה, וּתְנַן בְּרֵישׁ פֶּרֶק ה' דְּהַקּוֹנֶה בִמְחֻבָּר לֹא נִקְבַּע לַמַּעֲשֵׂר: בִּמְחֻבָּר. וְנַעֲשָׂה כְּבַעַל הַגִּנָּה דִּתְנַן לְעֵיל דְּאוֹכֵל מֵהֶן עֲרַאי, וְעוֹד תְּנַן בִּמְחֻבָּר לַקַּרְקַע פָּטוּר. הָרַ"שׁ:

פרק ב - משנה ז

בְּנִי בִּשְׂכָרִי. פֵּרֵשׁ רַשִׁ"י בְּמַסֶּכֶת בָּבָא מְצִיעָא דַּף צ"ב בַּשָּׂכָר שֶׁקָּצַצְתָּ לִי. וְהַתּוֹסָפוֹת מְפָרְשִׁים בִּשְׂכָרִי הַיְנוּ תַּחְתַּי. וְחַיָּב אֲפִלּוּ עַל מַה שֶּׁהָיָה אוֹכֵל הָאָב, לֹא בִלְבָד מַה שֶּׁיֹּאכַל הוּא יוֹתֵר מִן הָאָב, מִשּׁוּם דְּמֵחֲזֵי כְּמִקָּח: אוֹכֵל וְחַיָּב. סְתָמָא דְּלֹא כְרַבִּי יְהוּדָה דְּסוֹף פִּרְקִין שֶׁסּוֹבֵר דְּכָל מִקָּח דְּלִקְצוֹת פָּטוּר (וּבְסֵפֶר תּוֹסָפוֹת חֲדָשִׁים מוֹכִיחַ מִלְּשׁוֹן הָרַמְבַּ"ם דְּלִקְצוֹת דְּרַבִּי יְהוּדָה אֵינוֹ דָמִי לִדְהָכָא דְּרוֹצֶה לֶאֱכֹל מִיָּד וַהֲוָה אֶצְלוֹ דָּבָר הַגָּמוּר) וְעַיֵּן עוֹד אַחֲרֵי זֶה:

פרק ב - משנה ח

בִּלְבָסִים. מַתְנִיתִין כְּשֶׁלְּבָסִים וּתְאֵנִים מֻרְכָּבִים בְּיִחוּר אֶחָד, וַאֲפִלּוּ הָכִי לֹא יֹאכַל, דְּאִי לֹא תֵימָא הָכִי הָא תָּנֵינָא בָּבָא מְצִיעָא (דַּף צא) הָיָה עוֹשֶׂה בְּיִחוּר זֶה לֹא יֹאכַל בְּיִחוּר אַחֵר. הָרַ"שׁ בְּשֵׁם הַיְרוּשַׁלְמִי: וְטַעֲמָא דְמַתְנִיתִין דִּלְעֵיל סְתָמָא וְאַחַר כָּךְ מַחֲלוֹקֶת דְּהָכָא, וּבְכִי הָא לֵית הִלְכְתָא כִּסְתָם כוּ'. וּמִיהוּ נִרְאֶה לִי דְּאַף לְרַבִּי יְהוּדָה עַד כָּאן לֹא פָטַר בְּלִקְצוֹת אֶלָּא כְּשֶׁאוֹכֵל אַחַת אַחַת וְלֹא צֵרַף שְׁתַּיִם, דְּצֵרוּף קוֹבֵעַ לְמִקָּח כְּדִלְעֵיל מִשְׁנָה ה'. תּוֹסְפוֹת יוֹם טוֹב:
