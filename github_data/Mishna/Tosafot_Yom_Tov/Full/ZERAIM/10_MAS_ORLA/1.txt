פרק א - משנה א

בַּיְרוּשַׁלְמִי כָּתַב לִסְיָג לְקוֹרוֹת וּלְעֵצִים פָּטוּר, וְלֹא קַשְׁיָא, דְּקוֹרוֹת הַיְנוּ לְבִנְיָן כְּפֵרוּשׁ הָרַ"ב וְעֵצִים הַיְנוּ לַהֲסָקָה: וְהַחִיצוֹן פָּטוּר. בַּיְרוּשַׁלְמִי מְפָרֵשׁ טַעֲמֵיהּ, מִמַּשְׁמַע שֶׁנֶּאֱמַר לֹא יֵאָכֵל אָנוּ יוֹדְעִין שֶׁבְּעֵץ מַאֲכָל הַכָּתוּב מְדַבֵּר, מַה תַּלְמוּד לוֹמַר וּנְטַעְתֶּם כָּל עֵץ מַאֲכָל, לוֹמַר לְךָ אֶת שֶׁהוּא לְמַאֲכָל כְּגוֹן הַפְּנִימִי חַיָּב, וְאֶת שֶׁהוּא לִסְיָג כוּ' כְּגוֹן הַחִיצוֹן פָּטוּר:

פרק א - משנה ב

וְכֵן כָּתַב הָרַ"מ, עִנְיָנוֹ שֶׁנָּטַע לְיִשְׂרָאֵל בִּשְׂכִירוּת אוֹ בְחִנָּם, אֲבָל אִם נָטַע הַנָּכְרִי לְעַצְמוֹ וְקָנָה מִמֶּנּוּ הַיִּשְׂרָאֵל אַחַר כָּךְ, אֵינוֹ חַיָּב בָּעָרְלָה. עַד כָּאן. וְאִם כֵּן מַה שֶּׁכָּתַב הָרַ"ב בִּתְחִלַּת הַמִּשְׁנָה בֵּין נְטָעוֹ נָכְרִי הַיְנוּ נַמִּי דַּוְקָא כְּשֶׁנְּטָעוֹ לְיִשְׂרָאֵל. מִיהוּ בְּסוֹף פֶּרֶק ג' מְפָרֵשׁ דְּנָכְרִי הַנּוֹטֵעַ לְעַצְמוֹ נַמִּי נוֹהֵג עָרְלָה, וְכֵן בְּפֵרוּשׁ הָרַ"שׁ, וְאַף הָרַ"מ בְּחִבּוּרוֹ חָזַר בּוֹ וְכָתַב דְּנָכְרִי הַנּוֹטֵעַ לְעַצְמוֹ חַיָּב בָּעָרְלָה. וְעַיֵּן עוֹד: כְּלוֹמַר אַף עַל פִּי שֶׁבִּפְנֵי הַזְּרָעִים עוֹמֵד וּבָעֵי נְקִיבָה, וְהַיְנוּ דִּבְמִשְׁנָה ב' פֶּרֶק ב' דְּחַלָּה מְפָרֵשׁ בִּסְפִינָה נְקוּבָה, וְלֹא מְחַלֵּק בֵּין עֵץ לְחֶרֶס:

פרק א - משנה ג

מִצִּדּוֹ. פֵּרוּשׁ עַל יְדֵי רוּחַ אוֹ נָהָר דּוּמְיָא דְרֵישָׁא: כְּעָפָר. שֶׁיְּנַעְנַע הָאִילָן עַצְמוֹ וְיַעֲלֶה אוֹתוֹ מִמְּקוֹמוֹ, אֲבָל לֹא יַפְרִישֶׁנּוּ מִן הָאָרֶץ וְאַחַר כָּךְ יִתֵּן הֶעָפָר סְבִיבָיו. הָרַ"מ. וְלָכֵן הַגִּרְסָא בְּעָפָר בְּבֵי"ת:

פרק א - משנה ד

אִילָן שֶׁנֶּעֱקַר. מֵהֶעָפָר שֶׁסְּבִיבָיו: וְזֶה לְשׁוֹן הֶעָרוּךְ. הָאוֹרְגִים יֵשׁ לָהֶם עֵץ אָרֹךְ בְּרוֹחַב הַיְרִיעָה, וּבְרָאשָׁיו שְׁנֵי מְחָטִין שְׁבוּרִין, וְנוֹעֲצִים בְּרוֹחַב הַיְרִיעָה מִכָּאן וּמִכָּאן כְּדֵי לְמָשְׁכָהּ וּלְמָתְחָהּ, וּמִכְּלֵי הָאוֹרְגִים הוּא:

פרק א - משנה ה

הָרֹא"שׁ בְּשֵׁם הָרַ"שׁ בְּסוֹף פֶּרֶק קַמָּא דְקִדּוּשִׁין כָּתַב, וְיוֹצֵא רֹאשׁ הַיִּחוּר: שָׁנָה אַחַר שָׁנָה. פֵּרוּשָׁא דְרֵישָׁא הוּא. הָרַ"מ: כִּלְשׁוֹן הַרְכָּבָה בְּאִילָן כוּ', וְשַׁיָּךְ לְשׁוֹן סִפּוּק בְּדָבָר הַקָּצָר כְּשֶׁמְּחַבֵּר בּוֹ דָּבָר אַחֵר לְהַאֲרִיכוֹ, כְּדִתְנַן בְּפֶרֶק לוּלָב הַגָּזוּל סִפְּקוֹ בְחוּט. הָרַ"שׁ. וְסִפְּקָהּ בְּחֶבֶל דְּפֶרֶק ו' דְּכִלְאַיִם: אַף עַל פִּי כוּ'. שֶׁהִבְרִיךְ הַסִּפּוּק הַזֶּה בָּאָרֶץ וּטְמָנוֹ בְעָפָר, דְּאִיכָּא לְמֵיחַשׁ שֶׁמָּא הִשְׁרִישׁ בָּאָרֶץ קֹדֶם שֶׁנֶּאֱחַז בַּזְּקֵנָה, אֲפִלּוּ הָכִי מֻתָּר, כְּדִמְפָרֵשׁ בַּיְרוּשַׁלְמִי דְּשָׁרָשִׁין אֵין בָּהֶם מַמָּשׁ, כְּלוֹמַר וַאֲפִלּוּ אִם תִּמְצֵי לוֹמַר בְּעַלְמָא דְּאֵין זְקֵנָה מַתֶּרֶת יַלְדָּה הַנְּטוּעָה מִקֹּדֶם שֶׁסִּפְּקָהּ אַחַר כָּךְ בַּזְּקֵנָה (כְּמַאן דְּאָמַר אֶחָד בַּיְרוּשַׁלְמִי) הָכָא שָׁרֵי אֲפִלּוּ הִשְׁרִישָׁה קֹדֶם, דְּאֵין מַמָּשׁ בִּגְרוֹפִית זוֹ שֶׁסִּפֵּק. הָרַ"שׁ: וּמְפָרֵשׁ בַּיְרוּשַׁלְמִי דְּתַנָּא קַמָּא נַמִּי סְבִירָא לֵיהּ דְּאָזְלִינַן בָּתַר יְנִיקָתוֹ וְאִם אֵינוֹ יוֹנֵק מִכֹּחַ הַזְּקֵנָה דְּאָסוּר. וְנוּכַל לַעֲמֹד עַל זֶה בֶּעָלִין, שֶׁכְּשֶׁהִיא יוֹנֶקֶת מִן הַזְּקֵנָה הוֹפֶכֶת עָלֶיהָ שֶׁלֹּא יַעַמְדוּ כְּנֶגֶד הֶעָלִין שֶׁל הָאָב, וְאִם לָאו הוֹפֶכֶת עָלֶיהָ כְּלַפֵּי עֲלֵי הַזְּקֵנָה. וְסִימָן לַדָּבָר דְּאָכֵיל מֵחַבְרֵיהּ מִתְבַּיֵּשׁ לְהִסְתַּכֵּל בּוֹ. וְכִי פְּלִיגֵי בְּנָשְׁרוּ הֶעָלִין, וְלֵיכָּא לְמֵיקָם עָלֶיהָ דְּמִילְתָא מִכֹּחַ הֶעָלִין. הָרַ"שׁ:

פרק א - משנה ו

וּבְסוֹף פֶּרֶק ז' דְּכִלְאַיִם פֵּרַשְׁתִּי דְּלֹא נֶאֶסְרוּ הַגְּפָנִים אֶלָּא אִם כֵּן הִנִּיחַ הֶעָצִיץ בַּקַּרְקַע: גְּמָרָא גִּטִּין דַּף נ"ד, וּפֵרֵשׁ רַשִׁ"י אֵין לְךָ אָדָם נוֹטֵעַ עָרְלָה בֵּין שְׁאָר נְטִיעוֹת בְּלֹא סִימָן וְאוֹסֵר כַּרְמוֹ בִּשְׁבִיל נְטִיעָה אַחַת, וְכֵיוָן דְּלֹא שְׁכִיחָא לֹא הֶחְמִירוּ בָהּ רַבָּנָן: כְּלוֹמַר וְטַעֲמָא דְּשֶׁמָּא לְכַתְּחִלָּה יְעָרֵב אִסּוּר בְּהֶתֵּר בִּכְדֵי שֶׁיִּתְבַּטֵּל, וְהָכָא נַמִּי נֵיחוֹשׁ:

פרק א - משנה ז

וְהַלּוּלָבִים. עֲנָפִים הָרַכִּים הַיּוֹצְאִים בְּרָאשֵׁי הַבַּדִּים הַקָּשִׁים, וְהֵם רַכִּים מְאֹד. הָרַ"מ: וְלֹא קָשֶׁה עַל מַה שֶּׁכָּתַבְתִּי בְּמִשְׁנָה ה' פֶּרֶק ה' דְּמַעֲשֵׂר שֵׁנִי דְּבֹסֶר נַמִּי נִקְנֶה בְּמַתָּנָה, וְאַף עַל גַּב דְּאֵין נֶטַע רְבָעִי נִתָּן בְּמַתָּנָה כִּדְאִיתָא הָתָם, דְּמִכָּל מָקוֹם דִּין נֶטַע רְבָעִי יֵשׁ לוֹ, אֶלָּא מִשּׁוּם שֶׁלֹּא הִגִּיעַ עֲדַיִן לְעוֹנָתוֹ נִתָּן בְּמַתָּנָה (וְעַיֵּן שָׁם בְּתוֹסְפוֹת יוֹם טוֹב): וְעַיֵּן עוֹד מִשְׁנָה ז' פֶּרֶק ג' דַּעֲבוֹדָה זָרָה, וְעַיֵּן רֵישׁ פֶּרֶק ג' דְּסֻכָּה: מַעֲמִיד. פֵּרוּשׁוֹ שֶׁמַּקְפִּיא הֶחָלָב וְעוֹשֶׂה מִמֶּנּוּ גְּבִינָה, כְּמוֹ וְכַגְּבִנָה תַּקְפִּיאֵנִי . הָרַ"מ: מֻתָּר. לִכְאוֹרָה נִרְאֶה דְּלֹא אָמַר רַבִּי יְהוֹשֻׁעַ דִּשְׂרָף עָלִים וְעִקָּרִים מֻתָּר אֶלָּא בְּאִילָן הָעוֹשֶׂה פֵּרוֹת, דְּבֵיהּ אַיְרֵי כְּאָמְרוֹ בִּשְׂרָף פַּגִּים. אֲבָל בְּאֵינוֹ עוֹשֶׂה פֵּרוֹת מוֹדֶה דִּקְטָפוֹ זֶהוּ פִּרְיוֹ, וְכֵן מוֹרֶה בְּפֵרוּשׁ הָרַ"מ מִשְׁנָה ו' פֶּרֶק ז' דִּשְׁבִיעִית. אָמְנָם בְּחִבּוּרוֹ מַשְׁמַע שֶׁאֵין לְחַלֵּק. וְעַיֵּן עוֹד תּוֹסְפוֹת יוֹם טוֹב:

פרק א - משנה ח

וּבָאֲשֵׁרָה וּבַנָּזִיר. בָּאֲשֵׁרָה מִדִּכְתִיב (דְּבָרִים יג) וְלֹא יִדְבַּק בְּיָדְךָ מְאוּמָה מִן הַחֵרֶם. וּבַנָּזִיר מִדִּכְתִיב (בְּמִדְבָּר ו) מֵחַרְצַנִּים וְעַד זָג לֹא יֹאכֵל, וְאוֹמֵר (שָׁם) חֹמֶץ יַיִן וְחֹמֶץ שֵׁכָר וְכָל מִשְׁרַת עֲנָבִים כוּ': וּרְבָעִי אִיתְקַשׁ לְמַעֲשֵׂר שֵׁנִי. וְכֵן כָּתַב הָרַ"מ. וּבַמַּתְנִיתִין דִּלְעֵיל כָּתְבוּ דְּהַנֵּי לָאו פְּרִי נִינְהוּ, הַיְנוּ טַעֲמָא מִשּׁוּם דִּלְעֵיל בָּעָרְלָה נַמִּי מֻתָּרִים מֵהַאי טַעֲמָא דְּלָאו פְּרִי נִינְהוּ, וּרְבָעִי מֵהָעָרְלָה הוּא בָא שֶׁכֵּן בְּמִשְׁנָה ג' פֶּרֶק ב' קָרֵי לַהּ תַּנָּא עָרְלָה לְנֶטַע רְבָעִי מִשּׁוּם דְּמֵעָרְלָה הוּא בָא, הִלְכָּךְ נִיחָא לְהוּ לְפֵרוּשֵׁי לְעֵיל טַעֲמָא דְּשַׁיָּךְ גַּבֵּי עָרְלָה, אַף עַל גַּב דְּטַעֲמָא דִּמְפָרֵשׁ גַּבֵּי נָזִיר דְּלָאו בְּנֵי אֲכִילָה נִינְהוּ שַׁיָּךְ נַמִּי גַּבֵּי נֶטַע רְבָעִי כְּדִמְפָרֵשׁ הָכָא. אֲבָל הָכָא דִּמְרַבִּינַן לְהַנֵּי לְעִנְיַן עָרְלָה מֵאֶת פִּרְיוֹ, הַטָּפֵל לְפִרְיוֹ, וַהֲוָה אֲמֵינָא דְּהוּא הַדִּין לִרְבָעִי כֵּיוָן שֶׁמֵּעָרְלָה הוּא בָא, הִלְכָּךְ אִיצְטְרִיכוּ לְהֶקֵּשָׁא דְמַעֲשֵׂר שֵׁנִי:

יִחוּר. דְּלֹא גָזְרִינַן יִחוּר אָטוּ אֱגוֹז. תּוֹסָפוֹת: גְּמָרָא דַעֲבוֹדָה זָרָה דַּף מ"ט. וּפֵרֵשׁ רַשִׁ"י בִּלְשׁוֹן אַחֲרוֹן, אִם נָטַע הָאֱגוֹז שֶׁל עָרְלָה שֶׁהוּא אִסּוּר הֲנָאָה, וְלִכְשֶׁיִּגְדַּל וְנַעֲשָׂה נְטִיעָה הִבְרִיכָהּ וְהִרְכִּיבָהּ בְּאִילָן דְּהֶתֵּר, מֻתָּר, דְּאוֹתָן פֵּרוֹת שֶׁטּוֹעֵן אַחַר זְמָן בָּאִין נַמִּי עַל יְדֵי הָאִילָן שֶׁל הֶתֵּר. וְקַשְׁיָא לִי דִּבְלֹא הִבְרִיךְ וְהִרְכִּיב נַמִּי אִיכָּא זֶה וְזֶה גּוֹרֵם, אֱגוֹז דְּאִסּוּר וְקַרְקַע דְּהֶתֵּר, עַד כָּאן לְשׁוֹנוֹ. וְכָתְבוּ הַתּוֹסָפוֹת דְּלֹא דָק, דְּאֵין נִקְרָא זֶה וְזֶה גּוֹרֵם אֶלָּא כְּשֶׁשְּׁנֵיהֶם מֵעִנְיָן אֶחָד, כְּמוֹ הַרְכָּבַת אִילָן אִסּוּר בְּאִילָן הֶתֵּר שֶׁשְּׁנֵיהֶם אִילָן, וּכְמוֹ זֶבֶל נְבִיָּה וְקַרְקַע. אֲבָל אֱגוֹז וְקַרְקַע שְׁנֵי עִנְיָנִים הֵם:
