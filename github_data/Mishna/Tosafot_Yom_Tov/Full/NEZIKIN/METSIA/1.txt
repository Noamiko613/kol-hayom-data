פרק א

וְיֵשׁ לְכָל אֶחָד לְגַלְגֵּל עַל חֲבֵרוֹ שֶׁכָּל מַה שֶּׁנָּטַל כַּדִּין נָטַל. הָרַמְבַּ"ם. וְנִרְאֶה לִי דְּטַעֲמוֹ שֶׁעַל מַה שֶּׁבְּיָדוֹ מַמָּשׁ לֹא שַׁיְכָא תַּקָּנָה זוֹ, דְּאִם לֹא כֵן בְּכוֹפֵר הַכֹּל נַמִּי הֲוָה לְהוּ לְתַקְנָהּ: בַּגְּמָרָא, בְּמִקָּח וּמִמְכָּר. פֵּרֵשׁ רַשִׁ"י דְּאִיכָּא לְמֵימַר שְׁנֵיהֶם קְנָאוּהָ וְלִשְׁנֵיהֶם נִתְרַצָּה הַמּוֹכֵר. אֲבָל זֶה אוֹמֵר אֲנִי אֲרַגְתִּיהָ וְזֶה אוֹמֵר אֲנִי כוּ' לֹא יַחֲלֹקוּ, דְּוַדַּאי חַד מִנַּיְהוּ רַמַּאי הוּא, וּתְהֵא מֻנַּחַת עַד שֶׁיָּבֹא אֵלִיָּהוּ. וְדַעַת הַתּוֹסָפוֹת דְּגַם בִּכְהַאי גַוְנָא נַמִּי יַחֲלֹקוּ, וְלֹא אָמְרוּ יְהֵא מֻנָּח אֶלָּא בִּשְׁנַיִם שֶׁהִפְקִידוּ כֵּיוָן שֶׁיָּדוּעַ לַנִּפְקָד שֶׁהוּא וַדַּאי שֶׁל אֶחָד מֵהֶן: דְּאִי מֵחַד אֵינוֹ נֶאֱמָן לוֹמַר שֶׁמָּכַר לְאַחֵר. וְיֵשׁ שִׁטּוֹת רַבּוֹת בָּזֶה: שֶׁאֵינוֹ נוֹגֵעַ בָּעֵדוּת כְּלָל. דְּאַף עַל גַּב דְּקִבֵּל דָּמִים אֵינוֹ עוֹמֵד אֶלָּא בְּמִי שֶׁפָּרַע. וְאֵינוֹ דּוֹמֶה לְעֵד הַצָּרִיךְ שְׁבוּעָה דְּאֵינוֹ עֵד, דְּהָתָם רַחֲמָנָא אָמַר עַל פִּי שְׁנַיִם יָקוּם דָּבָר, דְּמַשְׁמַע הַנֶּאֱמָנִים בְּדִבּוּרָם בִּלְבַד וְלֹא שֶׁצָּרִיךְ שְׁבוּעָה לְהַאֲמִין דְּבָרָיו. וּמִי שֶׁפָּרַע לֵיכָּא אֶלָּא דַּאֲנַן לָיְטִינַן. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: וְכֵן פּוֹטֵר מִשְּׁבוּעָה לְאוֹתוֹ הַמְסַיְּעוֹ. הַמַּגִּיד מִשְׁנֶה: מֵאַחַר שֶׁזֶּה תָּפוּס וְעוֹמֵד וְזֶה תָּפוּס וְעוֹמֵד. גְּמָרָא. דְּהֵיכָא דְּלֵיכָּא חֲזָקָה לְשׁוּם אֶחָד וּתְפוּסִים בּוֹ יֵשׁ לְהַחְזִיק שְׁנֵיהֶם בְּשָׁוֶה: מַסְּקִינַן דְּאַף עַל גַּב דְּחָשׁוּד אַמָּמוֹנָא, לֹא חָשׁוּד אַשְּׁבוּעָתָא. וְהַיְנוּ טַעֲמָא מִשּׁוּם דִּשְׁבוּעָה חֲמוּרָה, שֶׁהָעוֹלָם נִזְדַּעְזַע עַל לֹא תִשָּׂא. תּוֹסָפוֹת: שֶׁאִם יוֹדֵעַ שֶׁחֲבֵרוֹ קְנָאָהּ לֵיהּ לֹא הָוֵי תָּפִיס, וּמִדְּתָפַס בָּהּ סָבַר שֶׁלּוֹ הִיא וְנִתְרַצָּה הַמּוֹכֵר, דְּהָא נָקַט זוּזֵי מִתַּרְוַיְהוּ, אֵימָא לֹא כוּ', דְּמִשּׁוּם שְׁבוּעָה לֹא יִפְרֹשׁ שֶׁהֲרֵי סָבוּר הוּא לוֹמַר אֱמֶת, קָא מַשְׁמַע לָן דַּאֲפִלּוּ הָכִי יִשָּׁבַע. תּוֹסָפוֹת: חֲבֵרִי יֵלֵךְ וְיִמְצָא אַחֶרֶת. רַשִׁ"י: וְזֶה כוּ'. אֲפִלּוּ לְבֶן נַנָּס דְּמִשְׁנָה ה' פֶּרֶק ז' דִּשְׁבוּעוֹת דְּאָמַר כֵּיצַד אֵלּוּ וְאֵלּוּ בָּאִין לִידֵי שְׁבוּעַת שָׁוְא, הָתָם וַדַּאי אִיכָּא שְׁבוּעַת שָׁוְא, (דְּחַד וַדַּאי מִשְׁתָּבַע לְשִׁקְרָא), הָכָא אִיכָּא לְמֵימַר דְּלֵיכָּא שְׁבוּעַת שָׁוְא, דְּתַרְוַיְהוּ בַּהֲדֵי הֲדָדֵי אַגְּבוּהָ. גְּמָרָא. (וּלְעִנְיָן מִקָּח וּמִמְכָּר נַמִּי שֶׁמָּא לִשְׁנֵיהֶם נִתְרַצָּה. רַשִׁ"י): לְרַבָּנָן דְּבֶן נַנָּס אִצְטְרִיךְ, דְּלֹא חָיְשִׁינַן לִשְׁבוּעַת שָׁוְא. תּוֹסָפוֹת: וּכְלוֹמַר דִּלְשׁוֹן הַשְּׁבוּעָה אֵינָהּ אֶלָּא שֶׁאֵין לוֹ בָּהּ פָּחוֹת מֵחֶצְיָהּ וּכְלִישְׁנָא דְּמַתְנִיתִין. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: וְיַחֲלֹקוּ. הַיְנוּ לִדְּמֵי אִי הֲוָה פְּסֵדָא בַּחֲלֻקַּת הַטַּלִּית עַצְמָהּ. גְּמָרָא: זֶה כוּ'. הָא תוּ לָמָּה לִי, סַלְקָא דַּעְתָּךְ אֲמֵינָא הַאי דְּקָאָמַר חֶצְיָהּ שֶׁלִּי לֶהֱוֵי כְּמֵשִׁיב אֲבֵדָה וְלִפָּטֵר, קָא מַשְׁמַע לָן דְּהַאי אַעֲרוּמֵי קָא מַעֲרִים, סָבַר אִי אֲמֵינָא כֻּלָּהּ שֶׁלִּי בָּעִינָא אִשְׁתְּבוֹעֵי, אֵימָא הָכִי דְּאֶהֱוֵי כְּמֵשִׁיב אֲבֵדָה וְאִפָּטֵר. גְּמָרָא: מִשְּׁלֹשָׁה. וְאִם תֹּאמַר עַל הַחֵצִי שֶׁמּוֹדֶה חֲבֵרוֹ לָמָּה צָרִיךְ לִשָּׁבַע. וְיֵשׁ לוֹמַר שֶׁאִלּוּ לֹא הָיָה נִשְׁבַּע אֶלָּא עַל הָרְבִיעַ בִּלְבַד, שֶׁמָּא הָיָה מִתְכַּוֵּן לְאוֹתוֹ חֵלֶק שֶׁמּוֹדֶה לוֹ חֲבֵרוֹ כוּ'. נִמּוּקֵי יוֹסֵף:

דִּלְגוּפֵיהּ לֹא אִצְטְרִיךְ, אֶלָּא לְאַשְׁמוֹעִינַן דְּרוֹכֵב קָנֵי אֲפִלּוּ שֶׁלֹּא מַנְהִיג כוּ'. רַשִׁ"י: תֵּימָהּ אַמַּאי כֵּיוָן שֶׁהַקִּנְיָן דִּרְכִיבָה לְחוּד הָוֵי קִנְיָן לָמָּה יִגָּרַע כְּנֶגֶד אַחֵר, כֵּיוָן דְּקִנְיָן הוּא לִקְנֵי. וּבַגְּמָרָא לֹא אָמְרוּ כוּ' אֶלָּא לִשְׁמוּאֵל דִּסְבִירָא לֵיהּ רוֹכֵב לְחוּדָא לֹא קָנֵי, מוֹקֵי לְסֵיפָא דְּקָנוּ שְׁנֵיהֶם כְּשֶׁרוֹכֵב וּמַנְהִיג בְּרַגְלָיו (וּמַתְנִיתִין אַשְׁמְעִינַן דִּשְׁנַיִם רוֹכְבִים קָנוּ שְׁנֵיהֶם, וְלֹא אָמְרִינַן קַמָּא עִקָּר, אִי נַמִּי רוֹכֵב וּמַנְהִיג כְּאֶחָד קָנוּ שְׁנֵיהֶם. הָרֹא"שׁ). אֲבָל אִי הֲוָה סָבוּר דְּרוֹכֵב לְחוּדָא קָנֵי לֹא אִצְטְרִיךְ לְאוֹקֵי מַתְנִיתִין שֶׁמַּנְהִיג בְּרַגְלָיו גַּם כֵּן. תּוֹסְפוֹת יוֹם טוֹב: וְתֵימָהּ דַּאֲפִלּוּ הָכִי מַאי רְבוּתָא הִיא. וּבַגְּמָרָא אִי בְּמִקָּח וּמִמְכָּר צְרִיכָא לְמֵימַר. אֶלָּא לָאו בִּמְצִיאָה. שְׁמַע מִנַּהּ דְּהַמַּגְבִּיהַּ מְצִיאָה לַחֲבֵרוֹ אַף עַל פִּי שֶׁלֹּא עֲשָׂאוֹ שָׁלִיחַ לְכָךְ. קָנָה חֲבֵרוֹ. דְּאִי סַלְקָא דַּעְתָּךְ לֹא קָנָה חֲבֵרוֹ, תֵּעָשֶׂה זוֹ כְּמִי שֶׁמֻּנַּחַת עַל גַּבֵּי קַרְקַע וְלֹא יִקְנֶה לֹא זֶה וְלֹא זֶה, וְכָל הָרוֹצֶה יַחְטְפֶנָּה מִיָּדָם. רַשִׁ"י:

רוֹכֵב. אוֹרְחָא דְמִלְּתָא נָקַט, דְּאִי מְהַלֵּךְ וְרָאָה יִטְּלֶנּוּ בְּעַצְמוֹ: נְטָלָהּ כוּ'. מַסְּקִינַן בַּגְּמָרָא דַּאֲפִלּוּ כִּי לֹא בָּעֵי לְמִזְכֵּי לְנַפְשֵׁיהּ מָצֵי זָכֵי לַחֲבְרֵיהּ בְּמִגּוֹ דְּאִי בָּעֵי זָכֵי לְנַפְשֵׁיהּ כוּ'. וְאִלּוּ מַתְנִיתִין דִּתְנַן וְאָמַר אֲנִי זָכִיתִי בָהּ זָכָה מִשּׁוּם דְּאָמַר לֵיהּ תְּחִלָּה, פֵּרֵשׁ רַשִׁ"י הֲרֵי אֲנִי זָכִיתִי בָהּ דְּקָאָמַר, תְּחִלָּה קָאָמַר לֵיהּ. מִתְּחִלָּה הִגְבַּהְתִּיהָ לְצָרְכִּי וְלֹא לְצָרְכְּךָ, וּלְעוֹלָם הַמַּגְבִּיהַּ מְצִיאָה לַחֲבֵרוֹ קָנָה חֲבֵרוֹ. עַד כָּאן. וְרִיהֲטָא דִּשְׁמַעְתְּתָא דְּאִי מִתְּחִלָּה הָיְתָה דַעְתּוֹ לְצֹרֶךְ חֲבֵרוֹ תּוּ לֹא מָצֵי מִהֲדַר. אֲבָל הַטּוּר לֹא סְבִירָא לֵיהּ. וְהַסְּמַ"ע הִסְכִּים לִדְבָרָיו. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: תְּחִלָּה. מִשּׁוּם רֵישָׁא דְּמַיְרֵי דַּוְקָא שֶׁאָמַר תְּחִלָּה, הִלְכָּךְ דִּיֵּק בַּסֵּיפָא לְמִתְנֵי תְּחִלָּה, כְּלוֹמַר וְאִם מִשֶּׁנְּתָנָהּ לוֹ טָעַן אוֹתָהּ טַעֲנָה דְּרֵישָׁא. אֲבָל מִשּׁוּם סֵיפָא לֹא אִצְטְרִיךְ לְמֵידַק שֶׁיִּטְעֹן תְּחִלָּה, דַּאֲפִלּוּ אִם לֹא טָעַן תְּחִלָּה מִסְּתָמָא תְּחִלָּה קָאָמַר. דְּמִי מָצֵי לְמֵימַר אֲנִי זוֹכֶה בָּהּ עַכְשָׁיו, וְהַלֹּא אֵינוֹ בְּיָדוֹ. רַשִׁ"י. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב:

כָּאן חָזַר מִפֵּרוּשׁ הַשֵּׁנִי שֶׁכָּתַב בְּמִשְׁנָה ג' פֶּרֶק ד' דְּפֵאָה דְּטַעְמָא מִשּׁוּם דְּנָפַל עָלֵיהָ כוּ', וְטַעְמָא שֶׁאֵין אַרְבַּע אַמּוֹת כוּ' כָּתַב רַשִׁ"י שֶׁהֲרֵי אֵין לָאָדָם יוֹם טוֹב: וְזֶה לְשׁוֹן הָרַמְבַּ"ם, שֶׁכָּתְבוּ לָהֶם הַדַּיָּנִין הוֹרָדָה בַּקַּרְקַע שֶׁיִּמְכְּרוּ אוֹתָהּ לִמְזוֹנוֹת: וְכָל כוּ'. מְפָרֵשׁ בַּגְּמָרָא כְּגוֹן שְׁטָרֵי חַלְטָאתָא, שֶׁמַּחְלִיטִין נִכְסֵי לֹוֶה לַמַּלְוֶה כְּשֶׁאֵין הַלֹּוֶה בְּפָנֵינוּ. דְּאִי הֲוָה בְּפָנֵינוּ הַיְנוּ אִגְּרוֹת שׁוּם דְּרֵישָׁא. סְמַ"ע: כְּלוֹמַר שֶׁכּוֹרֵךְ שְׁטָר וְקוֹפְלוֹ בְּעַצְמוֹ. וְאַחַר כָּךְ כּוֹרְכוֹ בְּתוֹךְ שְׁטָר אַחֵר. וְכֵן הַשְּׁנַיִם בַּשְּׁלִישִׁי: כְּלוֹמַר שֶׁהִשְׁכִּיב שְׁלֹשָׁה שְׁטָרוֹת זֶה עַל זֶה וּכְרָכָן בְּיַחַד, וּלְפִיכָךְ כְּשֶׁכְּרָכָן בְּאֹפֶן זֶה אִי אֶפְשָׁר שֶׁיִּשָּׁמֵט אֶחָד מֵחֲבֵרוֹ, לְכָךְ קוֹרֵא אוֹתָהּ אֲגֻדָּה. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: רַשִׁ"י. וּלְדִבְרֵיהֶם אֵין דֶּרֶךְ לְהַנִּיחַ שְׁטָרוֹת בַּחֲפִיסָה וְכוּ' וּלְפִיכָךְ הָוֵי כְלִי סִימָן. וְלֹא דָמֵי לְמִשְׁנָה ב' פֶּרֶק ב' דְּפֵרְשׁוּ דִּכְלִי יֵשׁ בּוֹ סִימָן. אֲבָל לְדִבְרֵי הַתּוֹסָפוֹת דְּדֶרֶךְ לְהַנִּיחַ שְׁטָרוֹת בַּחֲפִיסָה מַיְרֵי הָכָא שֶׁנּוֹתֵן בַּכְּלִי סִימָן. וְנִמּוּקֵי יוֹסֵף כָּתַב אֲבָל בְּסִימָן דְּגוּפָן דִּשְׁטָרוֹת לֹא מְהַדְּרִינַן לְחַד מִנַּיְהוּ, שֶׁכָּל אֶחָד מֵהֶן בָּקִי בָּהֶם: צָרִיךְ עִיּוּן, דְּאִי הָכִי שְׁלֹשָׁה לָמָּה לִי. וְהַנִמּוּקֵי יוֹסֵף כָּתַב, סִימָן הַכֶּרֶךְ עִם סִימָן הַמִּנְיָן כוּ', וְכֵן פָּסַק הָרַ"ב עַצְמוֹ פֶּרֶק ב' מִשְׁנָה ב' כוּ'. וּבְצִבּוּר פֵּרוֹת (שָׁם) כָּתַב מִנְיָן הֲוָה סִימָן, וְאִם כֵּן כְּרִיכָה לָמָּה לִי. וְהָרֹא"שׁ כָּתַב דְּהַיְנוּ כְּדֵי שֶׁיּוּכַל לִתֵּן סִימָן בַּמִּנְיָן וְלֹא מִתְיָאֵשׁ לוֹמַר שֶׁמָּא לֹא נָפִיל אַהֲדָדֵי. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: וְכַמָּה אֲגֻדָּה. מְפָרֵשׁ לְמַאי דִּמְסַיֵּם בֵּיהּ, וְהוּא הַדִּין לְתַכְרִיךְ, וּכְמוֹ שֶׁכָּתַב הָרַ"ב: וְיִתֵּן לְמִי שֶׁנּוֹתֵן בָּהֶם סִימָן. הָרַמְבַּ"ם: כְּלוֹמַר אֶלָּא לְמִי שֶׁנּוֹתֵן בָּהֶם סִימָן: לַמַּלְוֶה. דְּלֹוֶה לֹא מְקַיֵּם שְׁטָרֵיהּ, הִלְכָּךְ לֵיכָּא לְמֵיחַשׁ דִּלְמָא לְקִיּוּמִינְהוּ אָזֵיל. גְּמָרָא: כָּל דָּבָר הַמְּבַטֵּל דָּבָר קָרֵי סִמְפּוֹן שֶׁלּוֹ. כִּדְאָמַר בִּכְתֻבּוֹת דַּף נ"ז גַּבֵּי קִדּוּשֵׁי אִשָּׁה שֶׁהַמּוּם קָרוּי סִמְפּוֹן מִפְּנֵי שֶׁמְּבַטֵּל הַמִּקָּח. רַשִׁ"י. וְכָתְבוּ הַתּוֹסָפוֹת דִּבְיַד הַמַּלְוֶה קָרוּי סִמְפּוֹן וּבְיַד הַלֹּוֶה שׁוֹבֵר:
