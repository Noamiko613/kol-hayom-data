פרק ו

הָאֻמָּנִין. מַשְׁמַע לֵיהּ בֵּין בִּשְׂכִיר יוֹם בֵּין בְּקַבְּלָנוּת. אֲבָל בְּפִּרְקִין דְּבָתַר הָכִי נָקַט פּוֹעֲלִים, מִשּׁוּם דְּפוֹעֲלִים לֹא מַשְׁמַע אֶלָּא שְׂכִירֵי יוֹם, דִּבְהָכִי מַיְרֵי כִּדְאָמַר הָתָם, בִּזְמַן שֶׁאָמַר לְהַשְׁכִּים וּלְהַעֲרִיב. תּוֹסָפוֹת: וּמַתְנִיתִין אוֹרְחָא דְמִלְּתָא נָקַט, שֶׁכֵּן דֶּרֶךְ הָעוֹלָם שֶׁשּׁוֹלֵחַ אֶחָד מִן הָאֻמָּנִין, וְהוּא הַדִּין לְאַחֵר: וְהַאי תַּנָּא חָזְרוּ נַמִּי הִטְעוּ קָרֵי לֵיהּ. גְּמָרָא: בִּשְׁנֵי רֵישִׁי"ן. וְלַסְּפָרִים דְּגָרְסֵי קַדָּר יֵשׁ לוֹמַר דִּלְהָכִי נָקַט קַדָּר לְפִי שֶׁקַּדָּרִין רְגִילִים לִהְיוֹת לָהֶם קְרוֹנוֹת. תּוֹסָפוֹת: פִּרְיָפְרִין. דָּבָר הָאָבֵד חָשֵׁיב לֵיהּ, שֶׁעַתָּה זְמַנּוֹ הוּא לַעֲשׂוֹת. תּוֹסָפוֹת: נִרְאֶה דְּלֹא גָּרַס לֵיהּ בַּמִּשְׁנָה. וּלְשׁוֹן רַשִׁ"י, חֲלִילִין לַכַּלָּה לְשַׂמֵּחַ חָתָן וְכַלָּה, אוֹ לַמֵּת לְקוֹנֵן: כְּלוֹמַר עַכְשָׁיו אֵינוֹ מוֹצֵא פּוֹעֲלִים, אֲבָל וַדַּאי כְּשֶׁשָּׂכַר אֵלּוּ הָיָה מוֹצֵא. דְּאִי לָא תֵימָא הָכִי, מַה הִפְסִידוּ עָלָיו. רַשְׁבָּ"א: וְחָזְרוּ כוּ'. בְּכָאן לֹא פֵרֵשׁ הָרַ"ב לְאַחַר שֶׁעָשׂוּ מִקְצָת כִּדְפֵרֵשׁ לְקַמָּן. דְּהָכָא אֲפִלּוּ קֹדֶם שֶׁהִתְחִילוּ מַיְרֵי דּוּמְיָא דְּרֵישָׁא דְּאוֹקְמִינַן בַּגְּמָרָא בְּהֶדְיָא בְּשֶׁלֹּא הָלְכוּ. הר"א: עַד כְּדֵי הַכֶּפֶל מִשְּׂכַר הָרִאשׁוֹנִים, דִּבְהָכִי סָמְכֵי דַּעְתַּיְהוּ. וּפֵרֵשׁ הַסְּמַ"ע זַ"ל כִּי הֵיכִי דְּמָצִינוּ בְּעֵרְבוֹנִי יִקּוֹן דִּמְהַדֵּר לֵיהּ אִם אֶחְזֹר בּוֹ אֶכְפֹּל עֲרֵבוּתְךָ:

דְּאִלּוּ פּוֹעֵל, דְּהַיְנוּ שְׂכִיר יוֹם, יָכוֹל לַחֲזֹר בּוֹ אֲפִלּוּ בַּחֲצִי הַיּוֹם, דְּאָמַר רַחֲמָנָא עֲבָדַי הֵם, וְלֹא עֲבָדִים לָעֲבָדִים. וְהַאי טַעֲמָא לֹא שַׁיָּךְ אֶלָּא בִּשְׂכִירוּת, אֲבָל בְּקַבְּלָנוּת אֵין זֶה עֶבֶד אֶלָּא לְעַצְמוֹ. אֲבָל בְּדָבָר הָאָבוּד אֲפִלּוּ פּוֹעֵל יֵשׁ לוֹ דִּין הַשָּׁנוּי בְּמַתְנִיתִין דִּלְעֵיל. גְּמָרָא: רַשִׁ"י. וְלֹא רָצָה לְפָרֵשׁ כְּשֶׁהוּקְרוּ פּוֹעֲלִים וְיִתֵּן לָהֶם סֶלַע וָחֵצִי עַל חֵצִי שֶׁעָשׂוּ כְּמוֹ שֶׁשָּׁוֶה עַכְשָׁיו. שֶׁאֵין זֶה סְבָרָא, כִּי לָמָּה יִתֵּן לָהֶם בַּעַל הַבַּיִת יוֹתֵר מִמַּה שֶּׁהִתְנָה עִמָּהֶם, כֵּיוָן שֶׁאֵין מַפְסִידִים בַּחֲזָרַת בַּעַל הַבַּיִת, וְאַדְּרַבָּה יוֹתֵר מַרְוִיחִים שֶׁעַתָּה יִמְצְאוּ שֶׁיִּתְּנוּ לָהֶם סֶלַע וָחֵצִי לִקְצֹר חֲצִי קָמָה, וְהוּא לֹא הָיָה נוֹתֵן לָהֶם כִּי אִם סֶלַע. תּוֹסָפוֹת: דְּסָתַם לָן תַּנָּא כְּרַבִּי יְהוּדָה דְּפֶרֶק ט' דְּבָבָא קַמָּא מִשְׁנָה ד'. גְּמָרָא:

וְזוֹ כוּ'. וְהוּא הַדִּין אֲפִלּוּ זוֹ שֶׁהָלַךְ בָּהּ קְצָרָה יוֹתֵר, אֶלָּא שֶׁלֹּא חָשׁ לְהַאֲרִיךְ. תּוֹסָפוֹת: וְכֵן כָּתַב רַשִׁ"י. וְדוֹחֵק לוֹמַר כֵּן שֶׁדֶּרֶךְ הַבְּהֵמוֹת לְלַמְּדָם לָלֶכֶת בָּהָר וּבַבִּקְעָה. וְהַתּוֹסָפוֹת פֵּרְשׁוּ שֶׁיָּדוּעַ שֶׁאוֹתוֹ הַיּוֹם הָיָה אֲוִיר מְשֻׁנֶּה בֶּהָרִים יוֹתֵר מִבַּבִּקְעָה, אוֹ אִיפְּכָא, מִשֶּׁלֶג וּמָטָר, וְהָוֵי תְּחִלָּתוֹ וְסוֹפוֹ בִּפְשִׁיעָה. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: גִּרְסָתוֹ וְהִבְדִּיקָה בְּדַלֶ"ת. אֲבָל בַּגְּמָרָא בְּרֵי"שׁ, וְאָמְרוּ בְּפֵרוּשׁוֹ נְהוֹרִיתָא, עַל דֶּרֶךְ סַגִּי נָהוֹר: אַנְגַּרְיָא. שֶׁכְּשֶׁפּוֹגְעִים בַּדֶּרֶךְ לוֹקְחִים וְאֵין מְחַפְּשִׂים בַּבָּתִּים, וּלְפִיכָךְ אוֹמֵר לוֹ הֲרֵי כוּ' דְּמַזָּלְךָ גָּרַם, שֶׁאִלּוּ הָיְתָה בַּבַּיִת לֹא הָיְתָה נִטֶּלֶת. וּלְפִיכָךְ אֲפִלּוּ אֵינָהּ חוֹזֶרֶת, דְּלֹא דָּמֵי לְמֵתָה, דְּהָכָא מוּכָח דְּמַזָּלָא דִּידֵיהּ גָּרַם. תּוֹסָפוֹת: לֹא קָאֵי אַאַנְגַּרְיָא. הַמַּגִּיד: רַשִׁ"י. וּלְשׁוֹנוֹ מְגֻמְגָּם. דְּהָכָא אַיְרֵי בַּחֲמוֹר סְתָם, וּלְהָכִי כָּתַב בְּרֵישׁ דְּבָרָיו שֶׁצָּרִיךְ לְהוֹסִיף מָעוֹת כוּ', וְכֵן מִדְּתָנֵי בְּמַתְנִיתִין חַיָּב לְהַעֲמִיד לוֹ חֲמוֹר אַחֵר. הָרֹא"שׁ. וְטַעְמָא דְּבִמְשִׁיכַת הַחֲמוֹר מִשְׁתַּעְבְּדִין נִכְסֵי הַמַּשְׂכִּיר לְהַעֲמִיד לוֹ חֲמוֹר אַחֵר: פָּטוּר. וְלֹא הָוֵי תְּחִלָּתוֹ בִּפְשִׁיעָה וְסוֹפוֹ בְּאֹנֶס דְּחַיָּב, דְּלֹא חִיְּבוּהוּ אֶלָּא כְּשֶׁבָּא הָאֹנֶס מֵחֲמַת הַפְּשִׁיעָה, וְהָכָא אִלּוּ לֹא פָּשַׁע וְהוֹלִיכָהּ בָּהָר כָּל שֶׁכֵּן שֶׁהֻחְלְקָה. תּוֹסָפוֹת: הַמַּעֲלָה. הוּחַמָּה בַּעֲלוֹתָהּ מֵרַגְלֵי הָהָר לְרֹאשׁוֹ. רַשִׁ"י. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב:

וְזֶה אוֹחֵז הַדָּרְבָן לְכַוֵּן פָּרָה לִתְלָמֶיהּ, וְזֶה הוֹלֵךְ אַחַר הַמַּחֲרֵשָׁה וּמַכְבִּיד הַיָּתֵד בַּקַּרְקַע, שֶׁבּוֹ הַבַּרְזֶל, וְהוּא קַנְקַן דְּמַתְנִיתִין: פָּטוּר. רְצוֹנוֹ לוֹמַר שֶׁלֹּא יְשַׁלֵּם הַשּׂוֹכֵר לְבַעַל הַפָּרָה, אֶלָּא הָאֻמָּן שֶׁחָרַשׁ וְהָיָה תּוֹפֵס הַמַּחֲרֵשָׁה וּשְׁבָרָהּ, הוּא יְשַׁלֵּם לְבַעַל הַפָּרָה. הָרַמְבַּ"ם. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב:

כְּלוֹמַר דְּהָכִי מַשְׁמַע כְּשֶׁאוֹמֵר טֹל אֶת שֶׁלְּךָ וְהָבֵא מָעוֹת, וְלֹא אָמַר הָבֵא מָעוֹת וְטֹל שֶׁלְּךָ, דַּהֲוֵי מַשְׁמַע שֶׁעֲדַיִן תּוֹפְסוֹ כוּ'. אֲבָל כְּשֶׁאוֹמֵר גְּמַרְתִּיו פְּשִׁיטָא דְּשׁוֹמֵר חִנָּם הוּא, כֵּיוָן דְּיָדְעוּ הַבְּעָלִים שֶׁכָּלְתָה מְלַאכְתּוֹ הָוֵי כְּשׁוֹאֵל שֶׁכָּלְתָה יְמֵי שְׁאִילָתוֹ, שֶׁפָּטוּר מֵאֳנָסִים, וּכְמוֹ שֶׁכָּתַב הָרַ"ב פֶּרֶק ח' מִשְׁנָה ג'. גְּמָרָא: וּפֵרֵשׁ רַשִׁ"י בְּעָלָיו שֶׁל חֵפֶץ זֶה שֶׁנִּגְנַב בִּמְלַאכְתּוֹ שֶׁל שׁוֹמֵר הָיָה, שֶׁאַף הוּא מְשַׁמֵּר הָיָה לוֹ, וּכְתִיב אִם בְּעָלָיו עִמּוֹ לֹא יְשַׁלֵּם, וְדָרְשִׁינַן עִמּוֹ בִּמְלַאכְתּוֹ. וְאַף עַל גַּב דִּבְשׁוֹאֵל כְּתִיב, לְקַמָּן (דַּף צה) דָּרְשִׁינַן לֵיהּ אַכֻּלְּהוּ שׁוֹמְרִין: הַנַּח כוּ'. מַיְרֵי דְּמֻנַּח בְּמָקוֹם שֶׁקּוֹנֶה. נִמוּקֵי יוֹסֵף:

בְּהַהִיא הֲנָאָה דְּלֹא בָּעֵי לְמֵיתַב לֵיהּ רִפְתָּא לְעַנְיָא בְּשָׁעָה שֶׁמַּלְוֶה עָלָיו, דְּהָעוֹסֵק בְּמִצְוָה פָטוּר מִן הַמִּצְוָה. וּבְהַהִיא הֲנָאָה נַעֲשֶׂה שׁוֹמֵר שָׂכָר כָּל זְמַן שֶׁהַמַּשְׁכּוֹן אֶצְלוֹ. גְּמָרָא. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: רַשִׁ"י. וּכְלוֹמַר, אֲבָל הוּא לֹא יִשְׁתַּמֵּשׁ בָּהֶן וִינַכֶּה לוֹ שְׂכָרוֹ בְּחוֹבוֹ. וְטַעְמָא, כָּתַב הַטּוּר מִשּׁוּם חֲשָׁדָא. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב:

כְּלוֹמַר רַבִּי אֱלִיעֶזֶר דְּתָמַהּ, מִשּׁוּם דְּקָא סַלְקָא דַּעְתֵּיה כוּ': דִּסְתָם מַתְנִיתִין רַבִּי מֵאִיר הִיא: לְטַעְמֵיהּ דִּסְבִירָא לֵיהּ נִתְקָל פּוֹשֵׁעַ הוּא כוּ' וּלְפִיכָךְ מַשְׁבִּיעִין דְּשֶׁלֹּא בְּכַוָּנָה שְׁבַרְתִּיהָ. אֲבָל לְרַבִּי יְהוּדָה דְּהָתָם וְהִלְכָתָא כְּוָתֵיהּ דִּסְבִירָא לֵיהּ נִתְקָל לָאו פּוֹשֵׁעַ הוּא, מַשְׁבִּיעִין לֵיהּ שֶׁלֹּא בִּפְשִׁיעָה נִשְׁבְּרָה, בֵּין שׁוֹמֵר חִנָּם בֵּין שׁוֹמֵר שָׂכָר. הָרִי"ף וְהָרֹא"שׁ. וְנִרְאֶה מִלְּשׁוֹנָם דְּלָאו פּוֹשֵׁעַ הוּא אֲפִלּוּ שֶׁלֹּא בִּמְקוֹם מִדְרוֹן. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב:
