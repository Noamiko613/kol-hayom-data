פרק ג

בְּהֵמָה אוֹ כֵּלִים. אָמְרִינַן בַּגְּמָרָא דְּאִצְטְרִיךְ לְמִתְנֵי בְּהֵמָה, לְאַשְׁמְעִינַן דִּכְשֶׁרוֹצֶה לְשַׁלֵּם וְאֵינוֹ רוֹצֶה לִשָּׁבַע שֶׁמַּקְנֶה לוֹ כֶּפֶל זֶה אַף עַל פִּי שֶׁהוּא מְרֻבֶּה, דְּהָא כְּשֶׁטָּבַח וּמָכַר הַגַּנָּב מְשַׁלֵּם אַרְבָּעָה וַחֲמִשָּׁה. וְאִצְטְרִיךְ לְמִתְנֵי כֵּלִים לְאַשְׁמְעִינַן דְּאַף עַל גַּב דְּאֵין טֹרַח בִּשְׁמִירָתָן כְּמוֹ דְּאִיכָּא בַּבְּהֵמָה, אֲפִלּוּ הָכִי מַקְנֵי לֵיהּ כְּפֵלָהּ כְּשֶׁרוֹצֶה לְשַׁלֵּם: שֶׁאָבְדוּ. נָקַט אָבְדוּ אַף עַל פִּי שֶׁאֵין כֶּפֶל אֶלָּא בְּדָבָר הַנִּגְנָב, מִשּׁוּם יֻקְרָא, שֶׁאִם נִתְיַקֵּר אַחַר אֲבֵדָה וְזֶה רָצָה לְשַׁלֵּם וְלֹא לִשָּׁבַע, אִם נִמְצָא אַחַר כָּךְ וְנִתְיַקֵּר, הַכֹּל שֶׁל נִפְקָד, כֵּיוָן דְּלֹא אַטְרְחֵיהּ לְשַׁלֵּם בְּבֵי דִינָא. נִמּוּקֵי יוֹסֵף: לְמִי כוּ'. דְּקִים לְהוּ לְרַבָּנָן דְּנִיחָא לְהוּ לַבְּעָלִים שֶׁיְּהֵא בָּטוּחַ בַּקֶּרֶן עַל מְנָת שֶׁיְּהֵא סְפֵק כֶּפֶל הֶעָתִיד לָבֹא שֶׁל שׁוֹמֵר. וַהֲרֵי הִיא כִּמְסוּרָה לוֹ עַל מְנָת כֵּן, שֶׁאִם תִּגָּנֵב וִישַׁלֵּם לוֹ קֶרֶן, שֶׁתְּהֵא פָּרָה קְנוּיָה לוֹ מִשָּׁעָה שֶׁמְּסָרָהּ. גְּמָרָא: נִשְׁבַּע כוּ'. לִכְאוֹרָה קָשֶׁה הַךְ סֵיפָא לָמָּה לִי, פְּשִׁיטָא. מִיהוּ בְּבָבָא קַמָּא דִּיֵּק רָבָא מִנַּהּ דַּאֲפִלּוּ נִשְׁבַּע, אִם שִׁלֵּם אַחַר כָּךְ, הַכֶּפֶל לַשּׁוֹמֵר. וְאַבַּיֵי פָּלֵיג בָּזֶה. וַהֲלָכָה כְּרָבָא. נִמּוּקֵי יוֹסֵף:

וְהִשְׁאִילָהּ. לַעֲשׂוֹת בָּהּ יְמֵי שְׂכִירוּתוֹ. רַשִׁ"י: יְשַׁלֵּם לַשּׂוֹכֵר. וּבַגְּמָרָא, וְנֵימָא מַשְׂכִּיר לַשּׂוֹכֵר, דַּל אַנְתְּ וְדַל שְׁבוּעֲתָךְ, וַאֲנָא מִשְׁתָּעִינָא דִינָא בַּהֲדֵי שׁוֹאֵל. וּמְשָׁנֵי, מִי סָבַרְתָּ שׂוֹכֵר בִּשְׁבוּעָה הוּא דְּקָנֵי, מִשְּׁעַת מִיתָה הוּא דְּקָנֵי, וּשְׁבוּעָה לְהָפִיס דַּעְתּוֹ שֶׁל בַּעַל הַבַּיִת. וּפֵרֵשׁ רַשִׁ"י דְּמַה שֶּׁהוּא נִשְׁבַּע לַמַּשְׂכִּיר, לְהָפִיס דַּעְתּוֹ שֶׁלֹּא יֹאמַר פָּשַׁעְתָּ בָּהּ. וּמַה שֶּׁכָּתַב הָרַ"ב בִּשְׁבוּעָה כוּ', הָעְתַּק מִנֻּסְחָא מֻטְעֵית בְּרַשִׁ"י. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: כֵּיצַד כוּ'. טַעֲמָא דְּרַבִּי יוֹסֵי מִשּׁוּם דְּחָשׁוּב שׂוֹכֵר כְּאִלּוּ הִשְׁאִילוֹ בִּשְׁלִיחוּת הַמַּשְׂכִּיר, הִלְכָּךְ דִּין הַמַּשְׂכִּיר עִם הַשּׁוֹאֵל. הָרֹא"שׁ:

אָבִיו כוּ'. אֲבָל אֶחָד מִכֶּם הִפְקִיד כוּ' פָּטוּר, דְּהַמַּפְקִיד עַצְמוֹ הֲוָה לֵיהּ לְמֵידַק. הָרֹא"שׁ: שֶׁכְּלָל גָּדוֹל הוּא לְכָל הַנּוֹטְלִים שֶׁלֹּא כַדִּין שֶׁאֵין נוֹטְלִין אֶלָּא בִּשְׁבוּעָה, כִּדְאִיתָא בִּשְׁבוּעוֹת דַּף מ"ח. נִמּוּקֵי יוֹסֵף: מִשּׁוּם דְּמַתְנִיתִין תְּנַן אָבִיו נָקַט נַמִּי בְּתָבְעוּ אָבִי כוּ'. אֲבָל הוּא הַדִּין אִי תְּבָעוֹ כָּל אֶחָד וְאוֹמֵר אֲנִי הִפְקַדְתִּי. וְכֵן מַשְׁמַע בְּהָרַ"ב בְּמַתְנִיתִין דִּלְקַמָּן. וְלֹא דָּמֵי לְמִקָּח דְּפֶרֶק ט"ו דִּיבָמוֹת אוֹת כ"א, דְּהָתָם עוֹמֵד לְהִתְפָּרַע מִיָּד וְלֹא הֲוָה לֵיהּ לְמֵידַק. נִמּוּקֵי יוֹסֵף:

וּמִשּׁוּם הָכִי לֹא דָּמֵי לְמַה שֶּׁכָּתַב הָרַ"ב בְּמִשְׁנָה ז' פֶּרֶק י' דְּבָבָא קַמָּא, דִּבְטוֹעֲנוֹ שֶׁמָּא הִלְוִיתִיךָ חַיָּב לָצֵאת יְדֵי שָׁמַיִם. נִמּוּקֵי יוֹסֵף: וְהַשְּׁאָר כוּ'. וְאֵין לְהַקְשׁוֹת כֵּיוָן שֶׁכָּל אֶחָד תּוֹבֵעַ מָאתַיִם וְהוּא מוֹדֶה וַדַּאי מָנֶה, הֲוָה לֵיהּ כַּחֲמִשִּׁין יָדַעְנָא וַחֲמִשִּׁין לֹא יָדַעְנָא, דְּאָמְרִינַן מִשּׁוּם דְּאֵינוֹ יָכוֹל לִשָּׁבַע מְשַׁלֵּם. דְּהָתָם רְמָא רַחֲמָנָא שְׁבוּעָה עֲלֵיהּ שֶׁלֹּא יְעַכֵּב מָמוֹן חֲבֵרוֹ בִּרְשׁוּתוֹ. אֲבָל הָכָא הֲרֵי מוֹדֶה הַשְּׁלֹשׁ מֵאוֹת שֶׁתּוֹבְעִים בֵּין שְׁנֵיהֶם, וְאֵין הַכְּפִירָה וְהַהוֹדָאָה אֶלָּא בֵּין הַמַּפְקִידִין. נִמּוּקֵי יוֹסֵף:

גְּמָרָא. וְאָמְרִינַן דְּגֻזְמָא הִיא, דְּהָא בְּיוֹתֵר מִכְּדֵי חֶסְרוֹנָן מוֹכְרָן כִּדְמַסִּיק הָרַ"ב. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: וְהוּא שִׁעוּר שֶׁמְחַסְּרִים לְשָׁנָה, וְנֶחְסְרוּ בְּפָחוֹת מִשָּׁנָה. וְהַיְנוּ טַעֲמָא דְּפָלֵיג רַבָּן שִׁמְעוֹן בֶּן גַּמְלִיאֵל. נִמּוּקֵי יוֹסֵף:

וְכִי כוּ'. וְטַעֲמַיְהוּ דְּרַבָּנָן דְּהָנֵי עַכְבְּרֵי רְשִׁיעֵי נִינְהוּ. כַּד חָמְיָן עֲבוּר לֹא מִיסָּתַיְהוּ דְּאָכְלֵי אֶלָּא קָרְיָן לְחַבְרֵיהוֹן דְּאָכְלוּ עִמְּהוֹן. יְרוּשַׁלְמִי: כְּלוֹמַר מִדָּה מְרֻבָּה הַיְנוּ עֲשָׂרָה כוֹרִין:

וַחֲדָא מִתַּרְתֵּי דִּלְעֵיל נָקַט. דְּהָא בָּעִינַן נַמִּי שֶׁהָיָה מִסְתַּפֵּק מֵהֶן וְאֵינוֹ יוֹדֵעַ כַּמָּה כְּדִלְעֵיל. דְּאִי לָאו הָכִי הָיָה מְחַסֵּר לְפִי הָעֶרֶךְ: לֹג כוּ'. פֵּרוּשָׁא דִּשְׁלֹשָׁה לֻגִּין הוּא:

כְּגוֹן לַעֲלוֹת עָלֶיהָ כְּדֶרֶךְ סֻלָּם לִטֹּל גּוֹזָלוֹת מִקֵּן גָּבוֹהַּ, וַאֲפִלּוּ הָכִי הָוֵי גְזֵלָה לְמֵיקָם בִּרְשׁוּתֵיהּ עַד דַּעֲבֵיד הֲשָׁבָה, דְּקָסָבַר שׁוֹאֵל שֶׁלֹּא מִדַּעַת גַּזְלָן הוּא. וּלְהִשְׁתַּמֵּשׁ הַיְנוּ וְנִשְׁתַּמֵּשׁ. גְּמָרָא: לָאו הַיְנוּ דְּמִשְׁנָה ח' פֶּרֶק י' דְּבָבָא קַמָּא, דְּהָתָם בְּבַעֲלֵי חַיִּים כוּ' וּצְרִיכָה שְׁמִירָה יוֹתֵר, וְהָכָא בְּרוֹעֶה שֶׁשּׁוֹמֵר בִּרְשׁוּתוֹ וְגָנַב מֵרְשׁוּת עַצְמוֹ הוּא. וּסְבִירָא לֵיהּ לְרַבִּי יִשְׁמָעֵאל דְּלֹא בָּעִינַן דַּעַת בְּעָלִים, דַּעֲדַיִן שׁוֹמֵר הוּא עָלָיו, וְסַגִּי בְּדַעְתֵּיהּ. גְּמָרָא: דְּכֵיוָן שֶׁנִּשְׁתַּמֵּשׁ בָּהּ לְצָרְכּוֹ כָּלְתָה לוֹ שְׁמִירָתוֹ, שֶׁלֹּא הָיוּ רוֹצִין הַבְּעָלִים שֶׁיִּהְיֶה עוֹד שׁוֹמֵר עֲלֵיהֶן, וְלֹא הַוְיָא חֲזָרָה כְּלָל. רַשִׁ"י:

וּתְמִיהָנִי דְּהָא תְּנַן וְנָעַל בִּפְנֵיהֶם שֶׁלֹּא כָרָאוּי חַיָּב, מַשְׁמַע הָא כָּרָאוּי פָטוּר. וְהַנִמּוּקֵי יוֹסֵף כָּתַב דְּהָא דְּאָמְרִינַן דְּדַוְקָא בַּקַּרְקַע. דַּוְקָא לְהָנֵי דָּרֵי וּבְהַנְהוּ אַתְרֵי דִּשְׁכִיחֵי גַּנָּבֵי. וְכֵן מַשְׁמַע בַּגְּמָרָא. וְכֵן הוּא בְּהֶדְיָא בַּיְרוּשַׁלְמִי, נָעַל כָּרָאוּי כוּ' פָּטוּר: אַחַר שֶׁיִּשָּׁבַע הַנִּפְקָד שְׁבוּעַת הַתּוֹרָה שֶׁנָּתַן אוֹתוֹ הַדָּבָר לָהֶם. הָרַמְבַּ"ם: וְנָעַל. לֹא קָאֵי אַמְּסָרָן כוּ' אֶלָּא בְּהִנִּיחָן בְּבֵיתוֹ וּבְתֵבָתוֹ אַיְרֵי:

דְּהַמַּפְקִיד אֶצְלוֹ יוֹדֵעַ שֶׁהוּא צָרִיךְ מִיָּד לַמָּעוֹת. וּלְפִיכָךְ כְּשֶׁהִפְקִידָן צְרוּרִין נַמִּי יִשְׁתַּמֵּשׁ בָּהֶן, דְּדֶרֶךְ כָּל אָדָם לָצוּר מְעוֹתָיו. אֲבָל חֲתוּמִים, גָּלֵי דַּעְתֵּיהּ שֶׁאֵינוֹ חָפֵץ שֶׁיִּשְׁתַּמֵּשׁ בָּהֶם. רַשִׁ"י: וְאַף עַל גַּב דִּבְפֶרֶק דִּלְעֵיל מִשְׁנָה ז' פָּרְשִׁינַן דְּבִדְמֵי אֲבֵדָה הָוֵי שׁוֹאֵל עֲלַיְהוּ אֲפִלּוּ קֹדֶם שֶׁנִּשְׁתַּמֵּשׁ, הֲנָאַת שִׁמּוּשׁ דְּהָכָא גְּרִיעָא מִדְּהָתָם, לְפִי שֶׁיָּרֵא לִשְׁלֹחַ בָּהֶן יָד לִקְנוֹת בָּהֶן סְחוֹרָה, שֶׁמָּא יָבֹא הַמַּפְקִיד לִתְבֹּעַ פִּקְדוֹנוֹ, אֲבָל דְּמֵי אֲבֵדָה יוֹדֵעַ שֶׁיִּשְׁהֶה אֶצְלוֹ יָמִים רַבִּים. הָרַאֲבַ"ד וְהָרֹא"שׁ: לְפִיכָךְ כוּ'. וְהוּא שֶׁיַּטְמִין בַּקַּרְקַע. הָרַמְבַּ"ם:

הַשּׁוֹלֵחַ כוּ'. שֶׁחִיְּבוֹ הַכָּתוּב אֲפִלּוּ נֶאֶנְסוּ. דִּכְתִיב אוֹ מֵת אוֹ נִשְׁבָּר אוֹ נִשְׁבָּה שְׁבוּעַת ה' תִּהְיֶה בֵּין שְׁנֵיהֶם אִם לֹא שָׁלַח, דְּמַשְׁמַע הָא שָׁלַח נִתְחַיֵּב שׁוּב בְּמִיתָתָהּ וּשְׁבִירָתָהּ. רַשִׁ"י: גְּמָרָא. וּפֵרֵשׁ רַשִׁ"י, כִּשְׁעַת הַגְּזֵלָה: רַבִּי עֲקִיבָא אוֹמֵר כוּ'. פֵּרְשׁוּ הַתּוֹסָפוֹת דִּפְלוּגְתַּיְהוּ, דְּבֵית שַׁמַּאי סָבְרֵי שִׁנּוּי אֵינוֹ קוֹנֶה וּבֵית הִלֵּל סָבְרֵי דְּקוֹנֶה, וְהַיְנוּ בְּחָסֵר וְיָתֵר. אֲבָל בְּיוֹקְרָא וְזוּלָא מוֹדוּ בֵּית הִלֵּל לְבֵית שַׁמַּאי דְּאִם הוּקְרָה מְשַׁלֵּם כִּשְׁעַת הוֹצָאָה מִן הָעוֹלָם וּבְהוּזְלָה מְשַׁלֵּם כִּשְׁעַת הַגְּזֵלָה. וְרַבִּי עֲקִיבָא פָּלֵיג אַזּוֹל וְיֹקֶר, דִּמְשַׁלֵּם כִּשְׁעַת הַתְּבִיעָה בֵּין בְּזוֹל בֵּין בְּיֹקֶר. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: קְרָא אַפְּקֵיהּ מִכְּלָלָא דְּאֵין שָׁלִיחַ לִדְבַר עֲבֵרָה, כְּמוֹ דְּאַפֵּיק מְעִילָה וְכֵן טְבִיחָה. הָכִי אִיתָא בְּהֶדְיָא בַּגְּמָרָא. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: גְּמָרָא. וּמִיהוּ עַל מְנָת לִטֹּל בָּעִינַן. דִּשְׁלִיחוּת יָד אַף עַל פִּי שֶׁאֵין צָרִיךְ חִסָּרוֹן, עַל מְנָת לְחַסְּרוֹ בָּעִינַן. וְלֹא דָּמֵי לְמִשְׁנָה ט' דְּלֹא בָּעִינַן עַל מְנָת לְחַסְּרוֹ אֶלָּא בְּתַשְׁמִישׁ בְּעָלְמָא מְחָיְבִינַן לֵיהּ. דְּהָתָם מִשּׁוּם שׁוֹאֵל שֶׁלֹּא מִדַּעַת מְחָיְבִינַן לֵיהּ. וּלְכָךְ כָּתַב הַטּוּר דְּלֹא מִחַיַּב אֶלָּא מִשְּׁעַת תַּשְׁמִישׁ וְלֹא מִשְּׁעַת הַגְבָּהָה, שֶׁהֲרֵי אֵינוֹ שׁוֹלֵחַ בּוֹ יָד כֵּיוָן שֶׁאֵינוֹ רוֹצֶה לְחַסְּרוֹ: שֶׁכֵּן דֶּרֶךְ הַיַּיִן לְהַחְמִיץ בִּכְלִי חָסֵר. רַשִׁ"י:
