פרק ז

לְהַשְׁכִּים כוּ'. כְּשֶׁשְּׂכָרָן סְתָמָא וְאָמַר לָהֶם אַחַר שֶׁהִשְׂכִּירָן כְּבָר לְהַשְׁכִּים כוּ'. אֲבָל אִם הִתְנָה מֵעִקָּרָא, הַכֹּל לְפִי תְּנָאוֹ. תּוֹסָפוֹת: כְּלוֹמַר שֶׁזּוֹ הַמְּלָאכָה הִיא עֲבִידְתָא שַׁפִּירְתָּא. וְאַדַּעְתָּא דְּהָכִי טְפַאת לָן: וְאִם אֵין הָאֹכֶל מְזֻמָּן נַמְתִּין בְּבֵיתוֹ וְלֹא נֵצֵא הַשָּׂדֶה לִמְלַאכְתֵּנוּ. תּוֹסָפוֹת: דִּכְתִיב וַיִּקַּח בֶּן בָּקָר רַךְ וְטוֹב. בֶּן בָּקָר אֶחָד, רַךְ שְׁנַיִם, וָטוֹב שְׁלֹשָׁה. גְּמָרָא. וְנָקַט נַמִּי יִצְחָק וְיַעֲקֹב מִשּׁוּם כִּי בְּיִצְחָק יִקָּרֵא לְךָ זֶרַע. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: כְּמִנְהַג כוּ'. הַכֹּל לַאֲתוֹיֵי הַשּׂוֹכֵר אֶת הַפּוֹעֵל וְאָמַר לֵיהּ כְּאֶחָד וְכִשְׁנַיִם מִבְּנֵי הָעִיר. מְשַׁמְּנִין בֵּינֵיהֶן. פֵּרֵשׁ רַשִׁ"י שֶׁנּוֹתְנִין לָהֶם כְּמוֹ שֶׁנּוֹתְנִין לְבֵינוֹנִי. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב:

מִן הַתּוֹרָה. מִשּׁוּם דְּאַיְרֵי מֵעִקָּרָא בְּפוֹעֲלִים שֶׁאוֹכְלִים עַל יְדֵי שֶׁפָּסְקוּ לָהֶם מְזוֹנוֹת אוֹ עַל יְדֵי מִנְהַג הַמְּדִינָה, לְהָכִי נָקַט וְאֵלּוּ אוֹכְלִין מִן הַתּוֹרָה בְּלֹא פְּסַק וּבְלֹא מִנְהָג. תּוֹסָפוֹת: כְּלוֹמַר שֶׁנִּגְמַר מְלֶאכֶת בִּשּׁוּלָן, שֶׁאֵינָן צְרִיכִין עוֹד לַקַּרְקַע: וּבְפוֹעֵל הַכָּתוּב מְדַבֵּר, נֶאֱמַר כָּאן בִּיאָה וְנֶאֱמָר לְהַלָּן לֹא תָבֹא עָלָיו הַשָּׁמֶשׁ. גְּמָרָא: כְּלוֹמַר וְאֵין שְׁעַת גְּמַר מְלַאכְתּוֹ לְמַעֲשֵׂר פוֹטַרְתָּן: דְּהָא שׁוֹר לָאו דַּוְקָא, דְּכָל מִלֵּי אִיתְנְהוּ בַּחֲסִימָה כִּדְיָלְפִינַן בְּסוֹף פֶּרֶק ה' דְּבָבָא קַמָּא. גְּמָרָא: אוֹכֵל בִּמְחֻבָּר. אַף נֶחְסָם אוֹכֵל בִּמְחֻבָּר. גְּמָרָא: דְּעִקַּר פְּעֻלָּה מִשּׁוּם תִּקּוּן גְּדוֹלִים הֵם. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: שֶׁאֵין אוֹכְלִין. אֲפִלּוּ מֵהִלְכוֹת מְדִינָה. גְּמָרָא:

בִּכְתֵפוֹ. דְּאֵינוֹ מֵזִיז יָדָיו וְרַגְלָיו. רַשִׁ"י: אוֹכֵל. מַאי טַעֲמָא, כִּי תָבֹא בְּכֶרֶם רֵעֲךָ, בְּכָל מַה דַּעֲבֵיד. גְּמָרָא. וְנִרְאֶה לִי דְּמַפֵּיק לֵיהּ מִדְּלֹא כְתִיב כִּי תָבֹא לִבְצֹר בְּכֶרֶם רֵעֶךָ:

בִּתְאֵנִים כוּ'. שֶׁנֶּאֱמַר בְּכֶרֶם וְאָכַלְתָּ עֲנָבִים, עֲנָבִים וְלֹא תְאֵנִים. סִפְרִי (וִירוּשַׁלְמִי). וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: לֹא כוּ'. וַאֲפִלּוּ שְׂכָרוֹ לַעֲשׂוֹת בִּשְׁנֵיהֶן. טוּר: לֹא אָמְרוּ. שֶׁיֹּאכַל אֶלָּא בְּשָׁעָה שֶׁהוּא עוֹסֵק בַּמְּלָאכָה, וְלֹא שֶׁיֵּשֵׁב לוֹ וְיֹאכַל לוֹמַר הֲרֵי מָנַעְתִּי אֶת עַצְמִי עַד כָּאן וְלֹא נָטַלְתִּי לֶאֱכֹל וְעַכְשָׁיו אֵשֵׁב לִי וְאוֹכַל: דְבַגְּמָרָא מַקְּשׁוּ בָּהּ. כְּשֶׁהִיא פוֹרֶקֶת מֵהֵיכָא אָכְלָה (כָּל מַשָּׂאָהּ פּוֹרְקִין בְּבַת אַחַת וְהוֹלְכִין), אֵימָא עַד שֶׁתְּהֵא פוֹרֶקֶת. וּפֵרְשׁוּ הַתּוֹסָפוֹת דְּסַלְקָא דַּעְתָּךְ אֲמֵינָא שֶׁאֵינָהּ אוֹכֶלֶת אֶלָּא כְּשֶׁהִיא רוֹאָה בְּפָנֶיהָ כוּ', קָא מַשְׁמַע לָן. וְלֹא קָאֵי אַמִּפְּנֵי הֲשָׁבַת אֲבֵדָה, דְּמִן הַדִּין אוֹכֶלֶת. אֶלָּא מִלְּתָא בְּאַפֵּי נַפְשַׁהּ הִיא. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב:

מַתִּירִין. וְהָא דִּכְתִיב כְּנַפְשְׁךָ שָׂבְעֶךָ מְפָרֵשׁ בַּגְּמָרָא דְּדָרְשִׁינַן לֵיהּ מַה נַּפְשְׁךָ אִם חָסַמְתָּ פָּטוּר, אַף פּוֹעֵל אִם חָסַמְתָּ פָּטוּר כוּ'. וּבַסִּפְרִי, וַחֲכָמִים אוֹמְרִים שָׂבְעֶךָ, מְלַמֵּד שֶׁאוֹכֵל יָתֵר עַל שְׂכָרוֹ. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב:

קוֹצֵץ כוּ'. וּמִדַּעְתָּן. טוּר. וּמִשּׁוּם הָכִי לְכֻלֵּי עָלְמָא גְּדוֹלִים מַמָּשׁ. וְלֹא פְּלִיגֵי אָמוֹרָאֵי כְּמוֹ בְּמִשְׁנָה ה' דְּפֶרֶק קַמָּא: עַבְדּוֹ כוּ'. כְּנַעֲנִים. הַמַּגִּיד: שֶׁאֵין כוּ'. שֶׁהַפּוֹעֵל מִשֶּׁל שָׁמַיִם הוּא אוֹכֵל (וְלֵית לְהוּ לְמָרַיְהוּ זְכִיָּה בַּאֲכִילָתוֹ עַד שֶׁיִּתְּנֵהוּ לְתוֹךְ פִּיו. רַשִׁ"י). גְּמָרָא:

לֹא כוּ'. דְּלֹא אָמְרִינַן שֶׁהָיוּ סְבוּרִים שֶׁפָּדָה אוֹתָם כְּדֵי שֶׁיֹּאכְלוּ. אֶלָּא מִדְּלֹא אִתְנוּ, אַדַּעְתָּא דְּהָכִי נַחְתֵי כְּאִלּוּ קָצַץ כֵּן עִמָּהֶם. נִמּוּקֵי יוֹסֵף: אִם לֹא כוּ'. לֹא שֶׁשְּׂכָרָן סְתָם לַעֲשׂוֹת לוֹ מְלָאכָה, דְּבִכְהַאי גַוְנָא אַמַּאי פּוֹדֶה וּמַאֲכִילָן, אֶלָּא כְּגוֹן שֶׁשְּׂכָרָן לִבְצֹר כֶּרֶם וְלֹא הוֹדִיעָן שֶׁהוּא נֶטַע רְבָעִי כוּ' וְהָוֵי כְמִקַּח טָעוּת. הַמַּגִּיד: נִתְפָּרְסוּ כוּ'. רֵישָׁא בִּמְחֻבָּר וְסֵיפָא בְּתָלוּשׁ: אִם לֹא כוּ'. כְּגוֹן שֶׁאָמַר לָהֶם תְּחִלָּה לִדְרֹס בְּעִגּוּלִין, סְבוּרִין שֶׁזּוֹ הִיא תְּחִלַּת דְּרִיסָתָן וַעֲדַיִן לֹא נִגְמְרָה מְלַאכְתָּן כוּ'. הַמַּגִּיד מִשְׁנֶה:

אַרְבָּעָה שׁוֹמְרִין. וְדִינֵיהֶן שְׁלֹשָׁה. גְּמָרָא. דְּשׂוֹכֵר כְּנוֹשֵׂא שָׂכָר: נִשְׁבָּע. פְּשִׁיעָה לֹא חָשֵׁיב, דִּבְהָא כֻּלְּהוּ חַיָּבִים. תּוֹסָפוֹת: הַכֹּל. מַה שֶּׁהֻזְכַּר בַּפָּרָשָׁה. אֲבָל מֵתָה מֵחֲמַת מְלָאכָה לֹא. תּוֹסָפוֹת. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: וְאַף עַל גַּב דְּהַשְׁתָּא מִחַיַּב בִּכְפֵלָא מִשֶּׁטָּעַן טַעֲנַת גַּנָּב וּבָאוּ עֵדִים שֶׁהוּא גְּנָבוֹ כְּדִתְנַן פֶּרֶק ט' דְּבָבָא קַמָּא, וּפָרָשָׁה שְׁנִיָּה דִּמְחֻיָּב בִּגְנֵבָה לֹא אָתֵי לִידֵי כְּפֵלָא, שֶׁכְּשֶׁטָּעַן שֶׁנִּגְנַב מְחֻיָּב בְּקֶרֶן, אֲפִלּוּ הָכִי קַרְנָא בְּלֹא כְּפֵלָא עֲדִיפָא, דְּהָא שׁוֹאֵל חָמוּר וְאֵינוֹ מְשַׁלֵּם אֶלָּא קֶרֶן. גְּמָרָא: לַאֲפוֹקֵי שׁוֹאֵל דְּמַשְׁכַּחַת לַהּ דְּכָל הֲנָאָה שֶׁלּוֹ, בְּקַיְמָא בָּאֲגַם, דְּלֹא בָּעֵי מְזוֹנֵי, וּבְנָטַר מָתָא דְּלֹא בָּעֵי נְטִירוּתָא, אִי נַמִּי בִּשְׁאִילַת כֵּלִים. גְּמָרָא:

וּבַגְּמָרָא, אַמַּאי, לֵיקוּם גַּבְרָא כְּלַפֵּי גַּבְרָא. וּמוֹקֵי לָהּ בְּלִסְטִים מְזֻיָּן. וַאֲפִלּוּ הָרוֹעֶה מְזֻיָּן, פָּטוּר, דְּהַאי מָסַר נַפְשֵׁיהּ שֶׁעַל מְנָת כֵּן בָּא, לַהֲרֹג אוֹ לֵהָרֵג אוֹ לִקַּח מָמוֹן, אֲבָל הָרוֹעֶה אֵין לוֹ לִמְסֹר נַפְשׁוֹ עַל כָּךְ:

אֵינוֹ כוּ'. תֵּימַהּ דִּפְשִׁיטָא דִּפְשִׁיעָה גְמוּרָה הִיא. וְיֵשׁ לוֹמַר דְּקָא מַשְׁמַע לָן דַּאֲפִלּוּ יֵשׁ לִתְלוֹת וְלוֹמַר שֶׁמֵּתָה גַּם בִּשְׁבִיל דָּבָר אַחֵר, אֲפִלּוּ הָכִי חַיָּב, דְּלֹא הֲוָה לֵיהּ לְסַגְּפָהּ. וְהָרֹא"שׁ כָּתַב וַאֲפִלּוּ לֹא מֵתָה מִיָּד, דְּתָלִינַן שֶׁמִּשֶּׁסִּגְּפָהּ הִתְחִילָה לְהִתְקַלְקֵל. וְכֵן כָּתַב הַמַּגִּיד. וְכָתַב עוֹד, דְּהַאי אֹנֶס לָאו דַּוְקָא, דְּהוּא הַדִּין דַּהֲוָה פְּשִׁיעָה לְעִנְיָן שׁוֹמֵר חִנָּם. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: פֵּרֵשׁ רַשִׁ"י שֶׁתְּקָפַתּוּ עַל כָּרְחוֹ וְעָלְתָה, וְלֹא יָכוֹל לְהַחְזִיק בָּהּ, שֶׁחֲזָקָה הָיְתָה מִמֶּנּוּ. וְכָל שֶׁכֵּן דִּנְפִילָה הָיְתָה שֶׁתּוֹקְפַתּוֹ כוּ'. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: הֶעֱלָהּ. אוֹ שֶׁעָלְתָה מֵאֵלֶיהָ וְהוּא יָכוֹל לְמוֹנְעָהּ וְלֹא מְנָעָהּ. הָרַמְבַּ"ם: וְנָפְלָה. אַף עַל פִּי שֶׁתְּקָפַתּוּ וְנָפְלָה. הָרַמְבַּ"ם: דְּכֻלָּהּ מַתְנִיתִין רַבִּי מֵאִיר הִיא, כִּדְאִיתָא בַּגְּמָרָא: דִּכְדֶרֶךְ שֶׁתִּקְּנוּ מְשִׁיכָה בְּלָקוֹחוֹת כָּךְ תִּקְּנוּ מְשִׁיכָה בְּשׁוֹמְרִין. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב:

וְלֹא מָצֵי לְפָרוֹשֵׁי דְּמַתְנִיתִין בְּדָבָר שֶׁאֵינוֹ שֶׁל מָמוֹן. מִשּׁוּם דְּכָל תְּנַאי שֶׁיֵּשׁ בּוֹ מַעֲשֶׂה וְכוּ' שָׁמְעִינַן בְּעָלְמָא דְּרַבִּי מֵאִיר אָמְרָה. גְּמָרָא. וַאֲפִלּוּ הָכִי הָרַמְבַּ"ם מְפָרֵשׁ בְּאֵינוֹ שֶׁל מָמוֹן: וּבֵין עָשָׂה מַעֲשֶׂה שֶׁנָּתַן לוֹ קֹדֶם הַתְּנַאי אוֹ לֹא נָתַן לוֹ עַד אַחַר הַתְּנַאי, כָּל שֶׁהִקְדִּים אֲמִירַת הַמַּעֲשֶׂה הֲרֵי זֶה וְכוּ' קֹדֶם הַתְּנַאי, הַתְּנַאי בָּטֵל. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב:
