וְנָעַל. פֵּרֵשׁ הָרַ"ב לְקַמָּן, שֶׁהִיא בְּדֶלֶת שֶׁיְּכוֹלָה לַעֲמֹד בְּרוּחַ מְצוּיָה. וְטַעֲמָא מְפָרֵשׁ בַּגְּמָרָא, דְּהַתּוֹרָה מִעֲטָה בִּשְׁמִירַת שֵׁן וָרֶגֶל, דִּכְתִיב בְּהוּ וְשִׁלַּח וּבִעֵר, עַד דַּעֲבֵיד כְּעֵין וְשִׁלַּח, עַד דַּעֲבֵיד כְּעֵין וּבִעֵר, שֶׁיַּאֲכִילֶנָּה בַּיָּדַיִם, כְּלוֹמַר בִּפְשִׁיעָה: בַּלַּיְלָה. הָא בַּיּוֹם חַיָּב, דְּקָלָא אִית לַהּ לְמִלְּתָא וּמִסְתָּמָא יָדַע שֶׁנִּפְרְצָה. אִי נַמִּי בַּלַּיְלָה אֲפִלּוּ נוֹדַע לוֹ וְיָצְאָה הַבְּהֵמָה אֵין לוֹ לִטְרֹחַ יוֹתֵר מִדַּאי לַחֲזֹר אַחֲרֶיהָ בָּאֲפֵלָה. תּוֹסָפוֹת: לִסְטִים. וְאִצְטְרִיךְ לְאַשְׁמוֹעִינַן דַּאֲפִלּוּ לִסְטִים פְּטוּרִים כְּשֶׁלֹּא הוֹצִיאוּהָ. תּוֹסָפוֹת: חַיָּבִים. וְאַף עַל גַּב דִּמְשִׁיכָה לֹא קַנְיָא אֶלָּא בְּמִצְוַת הַמּוֹכֵר, וּמַאן אָמַר לֵיהּ לִמְשֹׁךְ כוּ'. וְנִרְאֶה לִי דִּסְבִירָא לֵיהּ דִּקְנָס הוּא שֶׁקָּנְסוּ חֲכָמִים לַגַּזְלָן. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב:

דְּמִדְּקָתָנֵי לְרוֹעֶה וְלֹא קָתָנֵי לְאַחֵר, שְׁמַע מִנֵּיהּ מַאי לְרוֹעֶה, לְרוֹעֶה שֶׁתַּחְתָּיו, כִּדְאִיתָא בַּגְּמָרָא: לְגִנָּה. בְּאֹנֶס, מִגַּג בְּעָלֶיהָ הַסָּמוּךְ לְגִנַּת חֲבֵרוֹ, אוֹ שֶׁהָיְתָה רְשׁוּת הָרַבִּים גָּבוֹהַּ מִן הַגִּנָּה וְנָפְלָה. רַשִׁ"י: וְכָךְ פָּסְקוּ הָרִי"ף וְהָרַמְבַּ"ם. אֶלָּא דִּלְפִי זֶה סֵיפָא דִּתְנַן יָרְדָה כוּ', הֲוָה לֵיהּ לְמִתְנֵי דַּאֲפִלּוּ הִפִּילוּהָ חַבְרוֹתֶיהָ מְשַׁלֶּמֶת, וְסִיַּעְתָּא לְהָרֹא"שׁ וְסִיעָתוֹ דִּדְחָפוּהָ כוּ' פָּטוּר: בַּגְּמָרָא, דְּמִדְּנָקַט בִּשְׂדֵה אַחֵר וְלֹא נָקַט בִּשְׂדֵה חֲבֵרוֹ, אַשְׁמוֹעִינַן עַל גַּב שָׂדֶה אַחֶרֶת. וּמִדְּנָקַט לִקְרָא גַּבֵּי וּבִעֵר וְלֹא נָקַט גַּבֵּי תַּשְׁלוּמִין, וְלֵימָא הָכִי מֵיטַב שָׂדֵהוּ וּמֵיטַב כַּרְמוֹ יְשַׁלֵּם בִּשְׂדֵה אַחֵר, דְּשָׁמִין עַל גַּב שָׂדֶה אַחֶרֶת, דָּרְשִׁינַן נַמִּי דַּוְקָא בִּשְׂדֵה אַחֵר וְלֹא בִּרְשׁוּת הָרַבִּים: וּלְהָכִי אֵין שָׁמִין עַל גַּב שָׂדֶה שְׁלֵמָה, דְּאִם כֵּן יַפְסִיד הַנִּזָּק, דְּלֹא יְזַלְזֵל בָּהּ בִּשְׁבִיל הֶפְסֵד עֲרוּגָה אַחַת כִּי אִם דָּבָר מֻעָט. רַשִׁ"י. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: גְּמוּרִים. דְּכֵיוָן דְּלֹא צְרִיכֵי לַשָּׂדֶה, לֹא עֲלֵיהֶם אָמְרָה תּוֹרָה וּבִעֵר בִּשְׂדֵה אַחֵר אַגַּב שְׂדֵה אַחֵר:

הֻזְּקָה. וְהָנֵי מִלֵּי שֶׁהֻחְלְקָה, כִּדְפֵרֵשׁ הָרַ"ב בְּמִשְׁנָה ב' פֶּרֶק דִּלְעֵיל: וַאֲפִלּוּ רַבִּי דְּמִשְׁנָה ג' פֶּרֶק דִּלְעֵיל מוֹדֶה בְּהָא:

אָדָם. לְפִי שֶׁאֵינָן בְּנֵי שִׁלּוּחִים. תּוֹסָפוֹת: חַיָּב. וְאַף עַל גַּב דְּיָלְפִינַן דִּשְׁלוּחוֹ שֶׁל אָדָם כְּמוֹתוֹ בְּכָל הַתּוֹרָה כֻּלָּהּ, אָמְרִינַן בַּגְּמָרָא דְּשָׁאנֵי הָכָא, דְּאֵין שָׁלִיחַ לִדְבַר עֲבֵרָה, דְּדִבְרֵי הָרַב וְדִבְרֵי הַתַּלְמִיד דִּבְרֵי מִי שׁוֹמְעִין. אֲבָל בְּחֵרֵשׁ שׁוֹטֶה וְקָטָן כֵּיוָן דְּלָאו בְּנֵי חִיּוּבָא נִינְהוּ לֹא שַׁיָּךְ בְּהוּ דִּבְרֵי הָרַב כוּ', אֶלָּא טַעֲמָא כְּמוֹ שֶׁכָּתוּב בְּסָמוּךְ: הָעֵצִים כוּ'. דְּאִי לָאו אַחֲרוֹן, קַמָּא לֹא עֲבֵיד מִידֵי. רַשִׁ"י: הָרוּחַ. וּפֵרֵשׁ בַּיְרוּשַׁלְמִי דַּוְקָא בְּרוּחַ שֶׁאֵינָהּ מְצוּיָה, וְהִיא שֶׁאֵין הָעוֹלָם מִתְנַהֵג בָּהּ תָּמִיד אֶלָּא לִפְעָמִים, אַף עַל פִּי שֶׁהוּא בָּא בְּעִתִּים הַרְבֵּה. וְאֵין צָרִיךְ לוֹמַר רוּחַ סְעָרָה שֶׁאֵינָהּ נוֹשֶׁבֶת אֶלָּא לְעִתִּים רְחוֹקִים. הַמַּגִּיד: עֵצִים כוּ'. מִדִּכְתִיב הַקָּמָה אַיְּתַר שָׂדֶה לְגוּפוֹ שֶׁל קַרְקַע, דְּהַיְנוּ לִחֲכָה נִירוֹ וְסִכְסְכָה אֲבָנָיו. גְּמָרָא: גָּבוֹהַּ אַרְבַּע אַמּוֹת. מֵעֲצֵי הַדְּלֵקָה. גְּמָרָא: אוֹ נָהָר. יֵשׁ אוֹמְרִים דְּבָעִינַן רָחָב שְׁמֹנֶה אַמּוֹת וְיֵשׁ בּוֹ מַיִם. וְיֵשׁ אוֹמְרִים אִם הוּא רָחָב שְׁמֹנֶה אֲפִלּוּ אֵין בּוֹ מַיִם, וְאִם יֵשׁ בּוֹ מַיִם אֲפִלּוּ כָּל שֶׁהוּא: בְּתוֹךְ שֶׁלּוֹ. אַף עַל גַּב דְּעַד הַשְׁתָּא נַמִּי בְּתוֹךְ שֶׁלּוֹ אָיְרִינַן (דִּבְשֶׁל חֲבֵרוֹ אֲפִלּוּ עָבְרָה גָּדֵר חַיָּב וְאֵין לוֹ שִׁעוּר), מִכָּל מָקוֹם הוֹאִיל וְטַעַם הַפְּטוּר דְּהָכָא מִשּׁוּם דִּבְתוֹךְ שֶׁלּוֹ הוּא, נָקַט מִלְּתָא בְּטַעֲמָא. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: בֵּית כּוֹר. אִם יֵשׁ לוֹ חֲצִי בֵּית כּוֹר לְכָל רוּחַ, פָּטוּר. רַשִׁ"י: שֵׁשׁ עֶשְׂרֵה אַמּוֹת. לֹא בְּאֶמְצַע שֵׁשׁ עֶשְׂרֵה קָאָמַר, דְּהָא מוֹקְמִינַן לָרֵישָׁא דִּתְנַן דֶּרֶךְ הָרַבִּים כְּוָתֵיהּ, וְהָתָם אֶמְצַע מַאן דָּכַר שְׁמֵיהּ. וּמִדְּרַבִּי אֱלִיעֶזֶר נִשְׁמַע נַמִּי לְרַבִּי עֲקִיבָא דְּבַתְרֵיהּ:
