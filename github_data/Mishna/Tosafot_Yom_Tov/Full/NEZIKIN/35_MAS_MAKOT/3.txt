פרק ג

פֵּרוּשׁ אִם הִתְרוּ בָהֶן מִשּׁוּם מַלְקוֹת. שֶׁאִם עָשׂוּ תְשׁוּבָה, בֵּית דִּין שֶׁל מַעְלָה מוֹחֲלִין לָהֶם. אֲבָל חַיָּבֵי מִיתוֹת בֵּית דִּין אֵינָן בִּכְלַל מַלְקוּת אַרְבָּעִים, שֶׁאִם עָשׂוּ תְשׁוּבָה אֵין בֵּית דִּין שֶׁל מַטָּה מוֹחֲלִין לָהֶם אֶת הַמִּיתָה, וּכְתִיב כְּדֵי רִשְׁעָתוֹ, מִשּׁוּם רִשְׁעָה אַחַת אַתָּה מְחַיְּבוֹ וְאִי אַתָּה מְחַיְּבוֹ מִשּׁוּם שְׁתֵּי רִשְׁעָיוֹת. אֲבָל חַיָּבֵי כְרֵתוֹת כֵּיוָן דְּאִי עֲבִיד תְּשׁוּבָה מוֹחֲלִין לוֹ, הִלְכָּךְ לָקוּ. וְאַף עַל גַּב דְּהַשְׁתָּא מִיהַת לֹא עֲבִיד תְּשׁוּבָה וְאַכַּתִּי עֹנֶשׁ כָּרֵת עָלָיו, לֹא פְסִיקָא מִלְּתָא לְכָרֵת הוֹאִיל וְהוּא תָלוּי בִּתְשׁוּבָה. גְּמָרָא: זוֹ הִיא שִׁיטַת הַתּוֹסָפוֹת שֶׁכָּתַבְתִּי בְּפֶרֶק קַמָּא דְשַׁבָּת (אוֹת כד). וְהָא דְלָא תָנֵי לָאו דִּבְגֵיוּתָן מִלֹּא תִקַּח, דְּתָנָא וְשִׁיֵּר. וּלְשִׁיטַת הָרַמְבַּ"ם, תָּנֵי נְתִינָה אַגַּב גְּרָרָא דְּמַמְזֶרֶת. כֵּן כָּתַב הַמַּגִּיד. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: רַשִׁ"י. וּמַשְׁמַע דְּבַחֲלוּצָה לְחוֹדָא אִיכָּא מַלְקוֹת מִדְּאוֹרַיְתָא. וּכְבָר כָּתַבְתִּי בְּפֶרֶק ט' דִּיבָמוֹת דַּחֲלוּצָה לֹא מִתַּסְּרָא כְלָל מִדְּאוֹרַיְתָא:

הַטָּמֵא כוּ'. כֻּלְּהוּ לָאוֵי דְמַתְנִיתִין מַיְתֵי לְהוּ קְרָאֵי לְאַזְהָרָתָן. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: וְהַמְפַטֵּם. מַעֲשֵׂה רוֹקֵחַ, תַּרְגּוּם יְרוּשַׁלְמִי מְפַטֵּם. אִישׁ בָּרִיא, גְּבַר פָּטִים. עָרוּךְ: הָכִי יָלְפִינַן בַּגְּמָרָא מִדִּכְתִיב וַאֲשֶׁר יִתֵּן מִמֶּנּוּ, מֵאוֹתוֹ שֶׁל מֹשֶׁה. וְעוֹד תַּנְיָא הָתָם שֶׁמֵּעוֹלָם לֹא נַעֲשָׂה שֶׁמֶן אַחֵר, וְלֹא עוֹד אֶלָּא שֶׁכֻּלּוֹ קַיָּם לֶעָתִיד לָבֹא, שֶׁנֶּאֱמַר שֶׁמֶן מִשְׁחַת קֹדֶשׁ יִהְיֶה זֶה לִי לְדֹרֹתֵיכֶם: שְׁקָצִים. נִרְאֶה לִי דְּבִשְׁקָצִים נִכְלָל דָּג טָמֵא וְכֵן שֶׁרֶץ הַמַּיִם, וְהוּא הַבְּרִיּוֹת שֶׁבַּמַּיִם שֶׁלֹּא בְצוּרַת דָּג טָהוֹר וְלֹא דָג טָמֵא, כְּגוֹן כֶּלֶב הַמַּיִם וְהַצְּפַרְדְּעִים. וְעוֹף טָמֵא וְשֶׁרֶץ הָעוֹף, וְהַיְנוּ כְּגוֹן זְבוּב אוֹ יַתּוּשׁ אוֹ דְבוֹרָה וְכַיּוֹצֵא בָהֶן. וְשֶׁרֶץ הָאָרֶץ, הַיְנוּ כְּגוֹן הַחֹלֶד וְהַצָּב נְחָשִׁים וְעַקְרַבִּים וְחַפָשִׁים וְנַדָּל וְכַיּוֹצֵא בָהֶן כוּ' דִּבְכֻלָּן נֶאֱמַר בָּהֶן שֶׁקֶץ, דּוֹק בַּפָּסוּק. וְיִתָּכֵן שֶׁהַחֲכָמִים כּוֹלְלִים גַּם כֵּן בִּלְשׁוֹן שְׁקָצִים בְּהֵמָה טְמֵאָה וְחַיָּה טְמֵאָה אַף עַל פִּי שֶׁאֵין כֵּן בִּלְשׁוֹן תּוֹרָה, דִּלְשׁוֹן חֲכָמִים לְחוּד. וְגַם בִּלְשׁוֹן חֲכָמִים לָאו כְּלָלָא הוּא כוּ': וּרְמָשִׂים. וְהֵן הַמִּינִין הַנִּבְרָאִים בָּאַשְׁפָּה וּבְגוּפֵי הַנְּבֵלוֹת כְּגוֹן רִמָּה וְתוֹלֵעָה וְכַיּוֹצֵא בָהֶן, שֶׁאֵינָם נִבְרָאִים מִזָּכָר וּנְקֵבָה אֶלָּא מִן הַגְּלָלִים שֶׁהִסְרִיחוּ וְכַיּוֹצֵא, וּבִכְלַל זֶה הַמִּינִין הַנִּבְרָאִים בָּעוֹפוֹת וּבְמַאֲכָלָם, אִם פָּרְשׁוּ וְיָצְאוּ לָאָרֶץ, אַף עַל פִּי שֶׁחָזְרוּ לְתוֹךְ הָאֹכֶל, שֶׁנֶּאֱמַר בְּכָל הַשֶּׁרֶץ: וּבִתְרוּמַת מַעֲשֵׂר כְּתִיב, בְּסוֹף פָּרָשַׁת קֹרַח: שֶׁאֵינוֹ חַיָּב מַלְקוֹת עַד שֶׁיֹּאכְלֶנּוּ בִירוּשָׁלַיִם. דְּאָמַר רַחֲמָנָא בִּבְכוֹר שֶׁנָּפַל בּוֹ מוּם, בִּשְׁעָרֶיךָ תֹּאכְלֶנּוּ, וְאָמְרוֹ בְכָאן לֹא תוּכַל לֶאֱכֹל בִּשְׁעָרֶיךָ, וּבָאָה הַקַּבָּלָה גַּם כֵּן בָּזֶה לֹא תוּכַל לֶאֱכֹל בִּירוּשָׁלַיִם כְּבִשְׁעָרֶיךָ, כְּלוֹמַר לֹא תוּכַל לֶאֱכֹל מַעֲשֵׂר בְּטֻמְאָה בִירוּשָׁלַיִם כְּמוֹ שֶׁתֹּאכַל בְּכוֹר בְּטֻמְאָה בִּשְׁעָרֶיךָ. הָרַמְבַּ"ם: וְכֵן כָּתַב הָרַמְבַּ"ם. וְתֵימַהּ, דְּמַאי אַזְהָרָה שַׁיָּךְ בְּמַה שֶּׁהוּא עַצְמוֹ אוֹמֵר וּמִתְוַדֶּה שֶׁלֹּא עָשָׂה כֵן, וְדִלְמָא לֹא הֻזְהַר עָלָיו בְּלָאו אֶלָּא מִכְּלַל עֲשֵׂה וְהָוֵי עֲשֵׂה. וּבַגְּמָרָא, וְהֵיכָן מֻזְהָר עַל אֲכִילָה, לֹא תוּכַל לֶאֱכֹל בִּשְׁעָרֶיךָ, וּלְהַלָּן הוּא אוֹמֵר בִּשְׁעָרֶיךָ תֹּאכְלֶנּוּ הַטָּמֵא וְהַטָּהוֹר בִּבְכוֹר בַּעַל מוּם כוּ'. וְקָאָמַר רַחֲמָנָא הַאי טָמֵא דְשָׁרֵי לָךְ גַּבֵּי בְכוֹר, הָכָא לֹא תֵיכוּל: וְהַתּוֹסָפוֹת הִקְשׁוּ עַל זֶה מִגִּיד הַנָּשֶׁה. וּמַסְּקֵי דְּרַבָּנָן לְדִבְרֵי רַבִּי שִׁמְעוֹן קָאָמְרֵי, לְדִידָן כָּל דְּפָרֵישׁ הַכָּתוּב לֹא תֹאכַל גִּיד לֹא תֹאכַל עוֹף טָמֵא, כְּאִלּוּ פֵּרֵשׁ בֵּין גָּדוֹל בֵּין קָטָן וּבִלְבַד שֶׁיִּהְיוּ שְׁלֵמִים, אֲבָל בְּטֶבֶל לֹא כָתַב חִטָּה. אֶלָּא לְדִידָךְ דְּחָשַׁבְתְּ חִטָּה בְּרִיָּה אַלְמָא לֵית לָךְ טַעֲמָא דִמְפָרֵשׁ, מִכָּל מָקוֹם לֹא דָמֵי לִנְמָלָה דְּאִיכָּא נְשָׁמָה:

כִּדְיָלְפִינַן מִדִּכְתִיב קָרְחָה יְתֵרָה, דַּהֲוָה מָצֵי לְמִכְתַּב לֹא יִקְרְחוּ בְרֹאשָׁם. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: וְדִתְנַן בְּמִשְׁנָה ב' וְהָאוֹכֵל חֵלֶב, טַעֲמָא מִשּׁוּם דְּאוֹכֵל חַיָּב כָּרֵת, וּתְנִינְהוּ לְחַיָּבֵי כְּרֵתוֹת מִטַּעַם שֶׁכָּתַב הָרַ"ב בְּרֵישׁ פִּרְקִין: אֲחוֹרֵי אָזְנוֹ אֵין שֵׂעָר כְּלוּם, וְכָךְ בְּמִצְחוֹ אֵין שֵׂעָר כְּלוּם, אֲבָל בִּצְדָעָיו שֶׁבָּאֶמְצַע יֵשׁ שֵׂעָר, וְאִם הוּא מַשְׁוֶה וְנוֹטֵל כָּל הַשֵּׂעָר שֶׁבִּצְדָעָיו לְמִדַּת אֲחוֹרֵי אָזְנָיו וּפַדַּחְתּוֹ זֶהוּ מַקִּיף פְּאַת הָרֹאשׁ. לְשׁוֹן רַבִּי יְהוּדָה בְרַבִּי נָתָן תַּלְמִיד רַשִׁ"י זַ"ל: אַחַת. הָכָא לָאו דַּוְקָא נָקַט שְׂרִיטָה אַחַת, אֶלָּא מִשּׁוּם דְּבָעֵי לְפַלּוֹגֵי לְקַמֵּיהּ בֵּין שְׂרִיטָה אַחַת לִשְׁתֵּי שְׂרִיטוֹת, נָקֵיט לֵיהּ נַמִּי הָכָא. וְכִמְדֻמֶּה דְּלֹא גָרְסִינַן לֵיהּ. רִיבָ"ן הַנִּזְכָּר לְעֵיל: דְּמָצֵי לְמִכְתַּב לְנֶפֶשׁ לֹא תִשְׂרְטוּ. לִישָׁנָא אַחֲרִינָא מֵרִבּוּיָא דְּוָי"ו. רִיבָ"ן. וְמִדְרַבִּינַן מִלָּנֶפֶשׁ לְחַיֵּב עַל כָּל נֶפֶשׁ וְנֶפֶשׁ, נִרְאֶה לִי מִדְּלֹא כְתִיב לָמֵת כְּדִכְתִיב גַּבֵּי קָרְחָה, וּכְתִיב לָנֶפֶשׁ, אַף עַל גַּב דְּעִנְיָנוֹ לְנֶפֶשׁ הַמֵּת, אֶלָּא מִשּׁוּם הָכִי אַפְּקֵיהּ בִּלְשׁוֹן נֶפֶשׁ, לְחַיְּבוֹ עַל כָּל נֶפֶשׁ וְנֶפֶשׁ אֲפִלּוּ לֹא שָׂרַט עַל כֻּלָּן אֶלָּא שְׂרִיטָה אַחַת: שְׁתַּיִם. דִּשְׁתֵּי פֵאוֹת יֵשׁ לוֹ לָרֹאשׁ, שֶׁהָרֹאשׁ בִּשְׁתֵּי חֲתִיכוֹת, מְקוֹם שֵׂעָר חֲתִיכָה אַחַת, וּמְקוֹם הַפָּנִים וְהַזָּקָן חֲתִיכָה אַחַת, וּמִתְחַבְּרוֹת זוֹ עִם זוֹ בְּצַד הָאֹזֶן מִלְּפָנָיו מָקוֹם שֶׁלּוֹעֲזִין טנפל"א (בִּלְשׁוֹן אַשְׁכְּנַז שלא"ף), וְשָׁם נִקְרָא פֵאָה, שֶׁשָּׁם סוֹף הָרֹאשׁ מְקוֹם חִבּוּר הַפְּרָקִים. וְנִמְצָא שֶׁיֵּשׁ לוֹ שְׁתֵּי פֵאוֹת, צִדְעָה מִכָּאן וְצִדְעָה מִכָּאן, וְחַיָּב עֲלֵיהֶן שְׁתַּיִם אֲפִלּוּ נְטָלָן בְּבַת אַחַת בִּשְׁתֵּי יָדָיו. וְאַף עַל גַּב דְּחַד הַתְרָאָה קָא מַתְרֶה לֵיהּ, אַל תַּקִּיף, דְּזִיל הָכָא הָאִיכָּא פֵאָה וְזִיל הָכָא פֵּאָה אִיכָּא. רִיבָ"ן. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: כְּדֵי לְהַסְבִּיר הָעִנְיָן יָפֶה הֶעְתִּיק שְׁנֵי לְשׁוֹנוֹת, וּשְׁנֵיהֶם לְדָבָר אֶחָד נִתְכַּוְּנוּ. הַלָּשׁוֹן הָרִאשׁוֹן עַד הֲרֵי חָמֵשׁ, הוּא מִדִּבְרֵי הָרַ"ן בְּשֵׁם רַשִׁ"י זַ"ל. וְהַלָּשׁוֹן דְּמִלֶּחִי הָעֶלְיוֹן עַד וְסוֹף הַזָּקָן, הוּא מֵהָרַמְבַּ"ם. וְסַנְטֵר פֵּרֵשׁ הַתִּשְׁבִּי בִּלְשׁוֹן אַשְׁכְּנַז קי"ן: אֲבָל בְּמִסְפָּרַיִם מֻתָּר, דְּלֵית בְּהוּ הַשְׁחָתָה, שֶׁהֲרֵי נִכָּר שָׁם הַשֵּׂעָר. הָרַ"ן: וּבְרֵישׁ פֶּרֶק י"ב דְּשַׁבָּת פֵּרֵשׁ דְּמַעֲצָד הוּא קוֹפִיץ קָטָן. וְזֶהוּ לֹא יִתָּכֵן לְפָרֵשׁ בְּכָאן. וְעַיֵּן בְּפֶרֶק י"ג דְּכֵלִים מִשְׁנָה ד' כוּ'. וְהָרַ"ן פֵּרֵשׁ, רְהִיטְנִי הוּא דֻּגְמַת הַמְּלַקֵט. וְעַל שֵׁם שֶׁהוּא רָץ בְּגִלּוּחוֹ וְקַל, נִקְרָא כָךְ, כְּדִמְתַרְגֵּם וַיָּרָץ וּרְהַט: חַיָּב. דְּהָנֵי נַמִּי עוֹשִׂין הַשְׁחָתָה וְדֶרֶךְ לְגַלֵּחַ בָּהֶן וַעֲבִיד בְּהוּ הַשְׁחָתָה וְגִלּוּחַ. וְתַנָּא קַמָּא סְבִירָא לֵיהּ אַף עַל גַּב דְּמַשְׁחִיתִים אֵין דֶּרֶךְ לְגַלֵּחַ בָּהֶן. גְּמָרָא:

שָׁם כוּ'. כְּלוֹמַר, עַל אוֹתָהּ שְׂרִיטָה יִכְתֹּב אֶת הַשֵּׁם. הָרַ"ן. וּקְרִיאַת מִלַּת שָׁם, בְּקָמָץ:

כִּדְאַשְׁכְּחַן בְּהַתְרָאַת מִיתַת בֵּית דִּין כוּ' שֶׁצָּרִיךְ שֶׁיַּעֲבֹר וְיַעֲשֶׂה תֵכֶף לַהַתְרָאָה בְּתוֹךְ כְּדֵי דִבּוּר, אֲבָל אַחַר תּוֹךְ כְּדֵי דִבּוּר צָרִיךְ הַתְרָאָה אַחֶרֶת כוּ'. וְהֻצְרַךְ הָרַ"ב לְפָרֵשׁ זֶה בְכָאן שֶׁלֹּא נִטְעֶה בִלְשׁוֹן הַמִּשְׁנָה דְּאַחַת מִיהָא מִחַיַּב עַל אֵיזֶה שְׁתִיָּה שֶׁתִּהְיֶה בְּכָל אוֹתוֹ הַיּוֹם, קָא מַשְׁמַע לָן דְּלֹא מִחַיַּב אֶלָּא אוֹתָהּ שֶׁהִיא תוֹךְ כְּדֵי דִבּוּר. וּמַתְנִיתִין הָא קָא מַשְׁמַע לָן, דַּאֲפִלּוּ שָׁתָה כָל הַיּוֹם בְּלֹא הֶפְסֵק אֵינוֹ חַיָּב אֶלָּא אַחַת. אֲבָל הַתְחָלַת הַשְּׁתִיָּה בְּתוֹךְ כְּדֵי דִבּוּר בָּעִינַן. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב:

עַל כָּל אַחַת. כְּשֶׁפֵּרַשׁ וְחָזַר וְנָגַע אוֹ נָשָׂא וְהֶאֱהִיל. וְאִיכָּא מַאן דְּאָמַר אֲפִלּוּ כְּשֶׁפֵּרַשׁ אֵינוֹ חַיָּב אֶלָּא בְּטֻמְאַת וּבִיאַת אֹהֶל שֶׁהֵם שְׁנֵי שֵׁמוֹת. גְּמָרָא. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: בְּכִלְאַיִם. אֶחָד מִן הַשּׁוּק וְלֹא נָזִיר. רִיבָ"ן: פֵּרוּשׁ הָרִאשׁוֹן נַמִּי אִיתָא בַגְּמָרָא עַיּוֹלֵי וְאַפּוֹקֵי. אֶלָּא מִשּׁוּם דְּפֵרוּשׁ הָרִאשׁוֹן קָרוֹב יוֹתֵר לְמַשְׁמָעוּת הַמִּשְׁנָה, כְּתָבוֹ בִסְתָם, וַהֲדַר קָאָמַר דְּבַגְּמָרָא מְפָרֵשׁ יוֹתֵר. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב:

יֵשׁ כוּ'. בַּגְּמָרָא פָרֵיךְ טוּבָא דְּמָצֵי לְמִחְשְׁבִינְהוּ נַמִּי, וְאִפְּרִיךְ דְּלֹא מְשָׁנֵי מִידֵי, וְצָרִיךְ לוֹמַר דְּתָנָא וְשִׁיֵּר. וְכָתַב הָרַמְבַּ"ם, אֵין בְּכָל אֵלּוּ אִסּוּר חָל עַל אִסּוּר. אֲבָל הֵם אִסּוּרִים הַרְבֵּה נִכְפְּלוּ בִזְמַן אֶחָד, וּלְפִיכָךְ אֵינוֹ מִן הַנִּמְנָע שֶׁיִּצְטָרְפוּ אֶל אֵלּוּ הַלָּאוִין זוּלָתָן הַרְבֵּה בְּאוֹתוֹ הַזְּמַן בְּעַצְמוֹ: וְלָמַדְנוּ עַל כָּל הַקָּדָשִׁים שֶׁהֵם אֲסוּרִים בַּעֲבוֹדָה. הָרַמְבַּ"ם. וְתֵימַהּ כוּ' דְּתֵיפּוּק לֵיהּ בְּשׁוֹר נַמִּי מִשּׁוּם מְעִילָה כְּגוֹן שֶׁהוּא קָדְשֵׁי קָדָשִׁים כוּ'. וּלְכָךְ מַסְּקֵי הַתּוֹסָפוֹת כְּפֵרוּשׁ הָרַמְבַּ"ם. וְלֹא חָשֵׁיב נַמִּי מְעִילָה, מִשּׁוּם דְּלֹא לָקֵי בֵיהּ עַד שֶׁיַּחֲרֹשׁ בְּשָׁוֶה פְרוּטָה, וְלֹא קָחָשֵׁיב אֶלָּא הָנָךְ דְּלָקֵי אַחֲרִישָׁה כָל שֶׁהוּא: וּמֻקְדָּשִׁין מַתְנִיתִין לָאו דַּוְקָא דִשְׁנֵיהֶם מֻקְדָּשִׁין. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: וְחוֹרֵשׁ תּוֹלָדָה דְזוֹרֵעַ, כְּגוֹן חוֹרֵשׁ לְכַסּוֹת הַזְּרִיעָה. רַשִׁ"י: וְיוֹם טוֹב. וְשַׁבָּת לֹא תָנָא, מִשּׁוּם דְּלָאו שֶׁנִּתָּן לְאַזְהָרַת מִיתַת בֵּית דִּין הוּא, דְּאֵין לוֹקִין עָלָיו. רַשִׁ"י. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: וּלְשׁוֹן הָרִיבָ"ן, אַף נָזִיר. נָזִיר וְכֹהֵן קָאָמַר, וְלֹא דָיֵיק:

סָמוּךְ. הָרַמְבַּ"ם יָהֵיב טַעֲמָא, שֶׁהַתּוֹרָה אָמְרָה יַכֶּנּוּ בְמִסְפָּר, כְּלוֹמַר שֶׁלֹּא יַכֶּנּוּ בְלֹא שִׁעוּר אֶלָּא בְּמִנְיַן מְשֹׁעָר כְּפִי כֹחַ שֶׁיּוּכַל לִסְבֹּל, וּמַה שֶּׁאָמְרָה אַרְבָּעִים, הוּא שֶׁאֵין מוֹסִיפִין עַל אַרְבָּעִים אֲפִלּוּ הָיָה חָזָק וּבָרִיא כְּשִׁמְשׁוֹן, וּלְפִיכָךְ אֵין מְשַׁעֲרִין לְהַכּוֹת יוֹתֵר מִשְּׁלֹשִׁים וָתֵשַׁע. שֶׁאִם יֶאֱרַע שֶׁהוֹסִיף וְהִכָּה עוֹד אַחַת, נִמְצָא שֶׁלֹּא הִכָּהוּ אֶלָּא אַרְבָּעִים הָרְאוּיוֹת לוֹ, וְלֹא עָבַר עַל פֶּן יוֹסִיף: כְּתֵפָיו. דִּכְתִיב (זְכַרְיָה יג) מָה הַמַּכּוֹת הָאֵלֶּה בֵּין יָדֶיךָ וְאָמַר אֲשֶׁר הֻכֵּיתִי בֵּית מְאַהֲבָי. וְרַבָּנָן הַהוּא בְתִינוֹקוֹת שֶׁל בֵּית רַבָּן הוּא דִכְתִיב. גְּמָרָא:

שְׁמֹנֶה עֶשְׂרֵה. כְּלוֹמַר שֶׁאֲמָדוּהוּ לְקַבֵּל חֲצִי אַרְבָּעִים שֶׁהוּא עֶשְׂרִים, אֵין אוֹמְרִים יִלְקֶה עֶשְׂרִים וְאַחַת כְּדֵי שֶׁיִּהְיוּ יְכוֹלִין לְהִשְׁתַּלֵּשׁ, אֶלָּא יִלְקֶה שְׁמֹנֶה עֶשְׂרֵה. הָרַמְבַּ"ם: גְּמָרָא. וּפֵרֵשׁ הָרִיבָ"ן, דַּחֲשׁוּבִין אוֹתָן שָׁלֹשׁ אֹמֶד לַלָּאו הָאַחֵר. אֲבָל אֲמָדוּהוּ אַרְבָּעִים וְאַחַת, שֶׁלֹּא הוֹסִיפוּ אֶלָּא שְׁתַּיִם, לוֹקֶה וּמִתְרַפֵּא כוּ', דְּמַכּוֹת שֶׁאֵינָן רְאוּיִין לְהִשְׁתַּלֵּשׁ לֹא מַלְקִין לֵיהּ:

וְזֶה לְשׁוֹן הָרִיבָ"ן, נָעוּץ בַּקַּרְקַע וְגָבוֹהַּ כְּנֶגֶד שְׁתֵּי אַמּוֹת אוֹ אַמָּה וָחֵצִי, וְהוּא כָפוּף וּמֻטֶּה עַל אוֹתוֹ הָעַמּוּד כְּאָדָם הַסּוֹמֵךְ עַל בְּרִיחַ דֶּלֶת וְתוֹלֶה יָדָיו לְמַטָּה וְכוֹפְתִין לוֹ יָדָיו בְּצִדֵּי הָעַמּוּד: מְגַלֶּה כוּ'. שֶׁאֵינוֹ מַכֵּהוּ עַל כְּסוּתוֹ, שֶׁנֶּאֱמַר וְהִכָּהוּ, וְלֹא לִכְסוּתוֹ. הָרַמְבַּ"ם: וְאֵינוֹ בְמַשְׁמַע לְשׁוֹן הַמִּשְׁנָה. וְזֶה לְשׁוֹן הָרִיבָ"ן, כְּפוּלָה אַחַת לִשְׁתַּיִם, וְעוֹד רְצוּעָה אַחֶרֶת כְּפוּלָה לִשְׁתַּיִם, הַיְנוּ שְׁתֵּי רְצוּעוֹת שֶׁהֵן אַרְבַּע. וּבַגְּמָרָא יָלֵיף לֵיהּ. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: וְזֶה לְשׁוֹן הָרִיבָ"ן, כֵּן דֶּרֶךְ תְּפִירָה שֶׁל רְצוּעָה, תּוֹחֵב רֹאשׁ רְצוּעָה בַּנֶּקֶב כְּנֶגֶד מַטָּה, וְחוֹזֵר וְתוֹחֵב בְּנֶקֶב אַחֵר דֶּרֶךְ מַעְלָה, וְנִרְאֶה כְּמַעֲלֶה וּמוֹרִיד:

וּלְשׁוֹן הָרִיבָ"ן, מַקֵּל שֶׁתְּלוּיָה בָהּ:

וְהַקּוֹרֵא כוּ'. הָא פְשִׁיטָא לָן דְּבָעֵי קְרִיָּה כִּדְאָמְרִינַן בִּקֹּרֶת תִּהְיֶה, בִּקְרִיאָה תְהֵא. וּלְהָכִי קָתָנֵי הָכָא וְהַקּוֹרֵא. רִיבָ"ן: לְשׁוֹן הַבְּרַיְתָא, וְהַשְּׁלִישִׁי אוֹמֵר הַכֵּהוּ. וּמְסַיֵּם הָרַמְבַּ"ם, שֶׁכָּל זְמַן שֶׁמַּכֶּה, עַל פִּיו הוּא מַכֶּה: וּתְנַן בְּמִשְׁנָה ב' פֶּרֶק ב', יָצָא הָאָב כוּ' וּשְׁלִיחַ בֵּית דִּין. רִיבָ"ן: דְּאִי לֹא טָעָה אֶלָּא שֶׁהִכָּהוּ בְמֵזִיד, אֵין מֵזִיד גּוֹלֶה. וּרְצוּעָה אַחַת דִּתְנַן הוּא כִנּוּי לְהַלְקָאָה, וּכְלוֹמַר, שֶׁהִלְקָהוּ עוֹד אַחַת. וְלֹא שֶׁהוֹסִיף רְצוּעָה עַל הָאַרְבַּע רְצוּעוֹת:

דִּבְרֵי כוּ'. אַף עַל פִּי שֶׁבַּמִּשְׁנָה אֵין מַחֲלֹקֶת, הָא אָמְרִינַן בַּגְּמָרָא חֲלוּקִין עָלָיו חֲבֵרָיו עַל רַבִּי חֲנַנְיָא בֶן גַּמְלִיאֵל. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: כְּמוֹ שֶׁאִיתָא פֶּרֶק קַמָּא מִשְׁנָה ז' דִּבְמִדַּת פֻּרְעָנוּת כוּ' וּבְמִדָּה טוֹבָה כְתִיב וְעֹשֶׂה חֶסֶד לַאֲלָפִים, וּמְתַרְגְּמִינַן לְאַלְפֵי דָרֵי, אַרְבָּעָה דוֹרוֹת כְּנֶגֶד אַלְפַּיִם דּוֹרוֹת הַיְנוּ אֶחָד מֵחֲמֵשׁ מֵאוֹת. מִעוּט לַאֲלָפִים אֵינוֹ פָחוֹת מִשְּׁנֵי אֲלָפִים. וְכָל שֶׁכֵּן אִם כִּפְשׁוּטוֹ דְּמַשְׁמַע אֲלָפִים עַד סוֹף כָּל הַדּוֹרוֹת: שָׂכָר. שֶׁנֶּאֱמַר לְמַעַן יִיטַב לְךָ וּלְבָנֶיךָ אַחֲרֶיךָ עַד עוֹלָם כִּי תַעֲשֶׂה הַטּוֹב וְהַיָּשָׁר בְּעֵינֵי ה' אֱלֹהֶיךָ. הָרַמְבַּ"ם:
