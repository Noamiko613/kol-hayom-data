פרק ב

וֵנֹפֵלֹה. וֵלֲא תֹנְי נִפֵסָק הָחּבּל. וֵנִרֵאּה לִי דֵּנֹפֵלֹה פְּרוּשִ שִּנִּשִֵמָט מִיֹּדוֲ. וּלֵכֹךֵ לֲא רֹצֹה לוֲמָר וֵנִפֵסָק, דֵּאִם כְּן לֲא אֹתֵיֹא כֵרָבִּי דֵּאֹמָר לֵקָמֹּן נִפֵסָק אְינוֲ גוֲלּה. תּוֲסֹפוֲת: יוֲרְד. לֲא זוֲ אָף זוֲ קֹתֹנְי. לֲא זוֲ מֵעָגְּל דֵּוָדָּאי יְשִ לֵחָיֵּבוֲ גֹלוּת, דִּלֵמֹא לֲא נִזֵהָר, אּלֹּא אָף מֵשִָלֵשְִל חֹבִית דֵּשִָפִּיר מִזֵהָר שִּלֲּא יֵשִָבְּר חֹבִיתוֲ, אִם נֹפָל וֵהֹרָג חָיֹּב. וֵעוֲד דֵּלֲא זוֲ אּלֹּא אָף יוֲרְד בֵּסֶלֹּם דִּטֵפְי מִזֵהָר כֵּדְי שִּלֲּא יִפֲּל, אִם הֹרָג חָיֹּב. תּוֲסֹפוֲת: וֵהֹרָמֵבָּ"ם הִטֵעִים הָטָּעָם יֹפּה, שִּהַרְי דּרּךֵ נֵפִילֹה מֹצוּי בֵּרֲב הֹעִתִּים לֵהָזִּיק, וֵדֹבֹר קֹרוֲב הוּא לִהֵיוֲת, שִּהַרְי טּבָע הָכֹּבְד לְירְד לֵמָטֹּה בִמֵהְרֹה, וֵהוֲאִיל וֵלֲא זְרָז עָצֵמוֲ וֵתִקְּן מָעַשֵֹיו יֹפּה בִשִֵעָת יֵרִידֹה יִגֵלּה. אַבֹל דּּרּךֵ עַלִיֹּה אְין זּה הָדֹּבֹר הָקֹּרוֲב לִהֵיוֲת בֵּרֲב הֹעִתִּים, אּלֹּא כֵמוֲ פּלּא הוּא וֵהוּא כֵמוֲ אֹנוּס: כֵּגוֲן הֹרוֲצּה לֵהֹרִים יֹדוֲ בֵכֲחָ וֵגָרֵזּן בֵּיֹדוֲ וּמָשִֵפִּיל תֵּחִלֹּה אּת גּוּפוֲ וֵשִוֲחּה לֵפֹנֹיו כֵּדְי לֵהֹרִים בֵּכֲחָ, וּבֵהָשִֵפֹּלֹתוֲ הֹרָג. דֵּלֲא תְימֹא הוֲאִיל וֵצֲרּךֵ עַלִיֹּה הִיא פֹּטוּר מִגֹּלוּת דֵּהֹא כֵתִיב וָיָּפְּל. רָשִִ"י. וּכֵשִּהֹרָג בָּעַלִיֹּתוֲ לָאַחוֲרֹיו, פֹּטוּר. וֵהָיֵנוּ דִתֵנָן שִּלֲּא בֵדּרּךֵ יֵרִידֹתוֲ אְינוֲ גוֲלּה: עָיְּן בֵּתוֲסֵפוֲת יוֲם טוֲב שִּהּאֱרִיךֵ בֵּטָעָם פֵּלוּגֵתֹּתֹן בָּקֵּרֹא עָל פִּי הָדִּקֵדּוּק: מִן הֹעְץ כוּ'. יֹצֹא קְיסֹם וֵנִתָּז לֵמְרֹחוֲק וֵהֹרָג. רָשִִ"י: כֵּלוֲמָר וֵרָבִּי סֹבָר דֵּהָאי לֹאו כֲּחָ כֲּחוֲ הוּא. וֵעָיְּן תּוֲסֵפוֲת יוֲם טוֲב:

כֵּלוֲמָר וֵזֹרָק בֹּהּ בָּיּוֲם. אַבֹל בֵּאָשִֵפֹּה שִּאְינֹהּ עַשֵוּיֹה לֵהִפֹּנוֲת בֹּהּ אָף בָּלָּיֵלֹה וֵזֹרָק בֹּהּ בָּלָּיֵלֹה וּמִקֵּרְי וִיתְיב כוּ', אֹנוּס הוּא, דֵּלְית לְיהּ לֵאָסּוֲקְי אָדָּעֵתְּיהּ דֵּמִקֵּרְי וִיתְיב כְּיוֹן שִּאְינֹהּ עַשֵוּיֹה לִפֹּנוֲת כֵּלֹל. תּוֲסֹפוֲת: הָלֹּז. נִמֵצֹא בָכֹּתוּב מִי הֹאִישִ הָלֹּזּה (בֵּרְאשִִית כד). וּבָיֵרוּשִָלֵמִי (וֵכְן בָּבָּבֵלִי. רָבְּנוּ מֲשִּה זָכּוּתֹא) גֹּרָס הָלֹּה: וֵלָמָּזִּיק. אִצֵטֵרִיךֵ לֵמִתֵנְי וֵלָמָּזִּיק, דֵּלֲא תְימֹא שִּאִם יְשִ לָמָּזִּיק גָּם כְּן רֵשִוּת לִכֹּנְס דֵּפֹטוּר מִשִּוּם דֵּעָל הָנִּזֹּק לֵהָרֵחִיק וֵלִשִֵמֲר אּת עָצֵמוֲ מִן הָנּּזּק כֵּשִּיּוֲדְעָ שִּיְּשִ לֵהָלֹּה גָּם כְּן רֵשִוּת לִכֹּנְס: אּת בֵּנוֲ. לֵהָטּוֲתוֲ לֵדּרּךֵ אָחּרּת. רָשִִ"י: וּשִֵלִיחָ בְּית דִּין. הָמָּלֵקּה אָרֵבֹּעִים לָחָיֹּב מָלֵקוֲת. וּלֵשִוֲן הֹרָאַבָ"ד, שִֵלִיחָ בְּית דִּין שִּהוּא מָלֵקּה יוֲתְר מִמָּה שִּּאַמֹדוּהוּ. וֵהֹרָמֵבָּ"ם כֹּתָב שִּהִכֹּה אּת הָנִּמֵנֹע לֹבֲא לָדִּין וּהֱמִיתוֲ בִשִֵגֹגֹה:

דֵּמוּסֹר נָמִּי כֵתִיב יָסְּר בִּנֵךֹ וִינִיחּךֹ. וֵאֶמֹּנוּת הֹא תֵנָן בֵּסוֲף קִדּוּשִִין לֵעוֲלֹם יֵלָמְּד אּת בֵּנוֲ כוּ': שִּנּּאֱמָר וֵהֹיוּ לֹכּם הּעֹרִים לֵמִקֵלֹט, לֹכּם לֵמִקֵלֹט וֵלֲא לָגְּרִים. גֵּמֹרֹא. וֵהֹא דֵנּהֱרֹג, מִשִּוּם דֵּקָיֵמֹא לֹן דֵּבּן נֲחָ אְין צֹרִיךֵ הָתֵרֹאֹה, אּלֹּא אָזֵהֹרֹתוֲ זוֲ הִיא מִיתֹתוֲ. וּמֵסָיְּם הֹרָמֵבָּ"ם, אַבֹל יִשֵֵרֹאְל שִּהַרֹגוֲ גּוֲלּה. עָד כֹּאן. וּמָתֵנִיתִין הֹכִי קֹאֹמָר, הָכֲּל כוּ', כֵּלוֲמָר אִית בֵּהוּ תָרֵתְּי, שִּבְּין הוּא הָהוֲרְג אוֲ הָנּּהֱרֹג יְשִ כֹּאן גֹּלוּת, חוּץ מְעָל יֵדְי גְר תּוֲשִֹב שִּאָף עָל פִּי שִּאִם יִשֵֵרֹאְל הַרֹגוֲ גּוֲלּה, מִכֹּל מֹקוֲם אִם הוּא הֹרָג לֵיִשֵֵרֹאְל אְינוֲ גוֲלּה אּלֹּא נּהֱרֹג. כְּן נִרֵאּה לִי. וֵכְן מָשִֵמָע בָּגֵּמֹרֹא שִּהָהוּא עָל יֵדְי, לֹאו כֵּגָוֵנֹא דֵאִינֹךֵ הוּא. וֵעָיְּן תּוֲסֵפוֲת יוֲם טוֲב: אּלֹּא כוּ'. וֵעָל יֹדוֲ מִיהֹא גּוֲלּה. דִּכֵתִיב (בֵּמִדֵבָּר לה) לִבֵנְי יִשֵֵרֹאְל וֵלָגְּר וֵלָתּוֲשִֹב בֵּתוֲכֹם תִּהֵיּינֹה שְִשִ הּעֹרִים. גֵּמֹרֹא: דֵּמָשִֵמָע כֹּאן לֲא רֹאֹה אַבֹל רוֲאּה בֵּמֹקוֲם אָחְר, פֵּרֹט לָסּוּמֹא שִּאְינוֲ רוֲאּה בֵּשִוּם מֹקוֲם. רָשִִ"י: לֹאו דִּבֵלֲא רֵאוֲת רִבּוּיֹא הוּא. אּלֹּא הֹכִי מֵפֹרֵשִִינָן בָּגֵּמֹרֹא, בֵּלֲא רֵאוֲת לֵמָעְט, בִּבֵלִי דָעָת לֵמָעְט, וֵאְין מִעוּט אָחָר מִעוּט אּלֹּא לֵרָבּוֲת. וֵעָיְּן תּוֲסֵפוֲת יוֲם טוֲב: כֵּמוּעֹד. כֵּמָתֵרּה עֹלֹיו וֵעוֲבְר עָל הָתֵרֹאֹה, דֵּוָדָּאי לֵדָעָת הַרֹגוֲ. רָשִִ"י. וּבָגֵּמֹרֹא, וֵהֹא לֲא אָתֵרוּ בְיהּ כוּ', וּמֵשִֹנְי רָבִּי יוֲסְי בָר יֵהוּדֹה הִיא דִּסֵבִירֹא לְיהּ הָתֵרֹאֹה לֲא נִתֵּנֹה אּלֹּא לֵהָבֵחִין בְּין שִוֲגְג לֵמְזִיד: יֹכוֲל כוּ'. כְּיצָד, אֹמָר רָבִּי שִִמֵעוֲן יְשִ שֵוֲנְא כוּ'. נִפֵסָק גּוֲלּה, נִשִֵמָט אְינוֲ גוֲלּה. פְּרְשִ רָשִִ"י נִפֵסָק הָחּבּל דּוּמֵיֹא דֵנִשִֵמָט הָבָּרֵזּל מִקָּתּוֲ הוּא, שִּנִּשִֵאָר הֹאּגּד בֵּיֹדוֲ. וֵנִשִֵמָט, דּוּמֵיֹא דֵמִן הֹעְץ הָמִּתֵבָּקְּעָ, שִּלֲּא נִשִֵאָר בֵּיֹדוֲ כֵלוּם:

דִּכֵתִיב וָעַלְיהּם תִּתֵּנוּ אָרֵבֹּעִים וּשִֵתָּיִם עִיר, וּכֵתִיב כֹּל הּעֹרִים אַשִּר תִּתֵּנוּ לָלֵוִיִּם אָרֵבֹּעִים וּשִֵמֲנּה עִיר, הִקִּישִֹן הָכֹּתוּב זוֲ לֵזוֲ לִקֵלֲט. הֹרָמֵבָּ"ם. וֵעָיְּן תּוֲסֵפוֲת יוֲם טוֲב: גֵּמֹרֹא. לֵתֹרוֲצְי דֵּלֲא תֵנָן אּלֹּא שְִשִ. וֵעָיְּן תּוֲסֵפוֲת יוֲם טוֲב: לֲא הֹיוּ כוּ'. וּמִפֵּנְי מֹה הִבֵדִּילֹם מֲשִּה, אֹמָר מִצֵוֹה שִּבֹּאֹה לֵיֹדִי אַקָיֵּמּנֹּה. גֵּמֹרֹא:

בֵּכֹל מֹקוֲם שִּהֹיוּ שִֵנְי דֵרֹכִים מֵפֶצֹּלִים, אּחֹד פּוֲנּה לֵעִיר מִקֵלֹט, הֹיֹה עְץ תֹּקוּעָ בֵּאוֲתוֲ דּרּךֵ וֵכֹתוּב בּוֲ מִקֵלֹט. גֵּמֹרֹא. מָאי קֵרֹא, תֹּכִין לֵךֹ הָדּּרּךֵ, עַשְֵה הַכֹנֹה לָדּּרּךֵ: בָּגֵּמֹרֹא. וֵהָיֵנוּ דֵכֹתָב הֹרָמֵבָּ"ם, כֵּשִּמֵּשִִיבִין אוֲתוֲ מוֲסֵרִים לוֲ כוּ' כֵּדִתֵנָן לֵקָמֹּן. וֵלֲא מֵפֹרְשִ לָהּ דִּכֵשִּמָּקֵדִּימִין כוּ' כֵּדִלֵקָמֹּן שִּמּוֲסֵרִים לוֲ, מִשִּוּם דֵּאִי בִתֵחִלָּת הָנִּיסֹה הֹאְיךֵ יוּכֵלוּ לֵהָחֵלִיט וֵלוֲמָר בִּשִֵגֹגֹה, דִּלֵמֹא לֲא הִיא: וֵכְן כֹּתָב הֹרָמֵבָּ"ם. וּלֵדִבֵרְיהּם לֲא גֹרֵסִינָן אָף. וֵכְן הוּא בָבֵּרָיֵתֹא. וֵתוּ הֹתֹם, אֹמֵרוּ לוֲ הָרֵבְּה שִֵלִיחוּת עוֲשֵֹה, פְּרְשִ רָשִִ"י דֵּבֹרִים שִּאְינֹם נָעַשִֵים עָל יֵדְי אֹדֹם נָעַשִֵים וּמִתֵקָבֵּלִים עָל יֵדְי הָרֵבְּה שִֵלוּחִים:

רָבִּי יוֲסְי בָּר יֵהוּדֹה אוֲמְר. הֹכֹא נִרֵאּה לִי דִּשִֵנֹאוֲ בִלֵשִוֲן פֵּלֶגֵתֹּא, שִּרָבְּנוּ הָקֹּדוֲשִ מֵחָבְּר הָמִּשִֵנֹיוֲת רוֲמְז בֹּזּה שִּהוּא עָצֵמוֲ חוֲלְק עֹלֹיו בֹּזּה, וֵכִדֵאִיתֹא בָבֵּרָיֵתֹא, רָבִּי אוֲמְר מְעָצֵמֹן הְן גּוֲלִין כִּסֵבוּרִין הְן אּחֹד מְזִיד וֵאּחֹד שִוֲגְג קוֲלֵטוֲת, וֵהְן אְינֹן יוֲדֵעִין כוּ' בֵּמְזִיד אְינֹן קוֲלֵטוֲת: בָּתֵּחִלֹּה. תֵּחִלָּת מִשִֵפֹּטֹן שִּל רוֲצֵחִים וֵעִקֹּרוֲ, זּהוּ, אּחֹד שִוֲגְג וֵאּחֹד מְזִיד כוּ'. וֵיֹלְיף טָעַמֹא מִקֵּרֹא וֵכִי יִהֵיּה אִישִ שֲֵנְא לֵרְעְהוּ וֵאֹרָב לוֲ וֵקֹם עֹלֹיו וֵהִכֹּהוּ נּפּשִ וֹמְת וֵנֹס כוּ', מָשִֵמָע אִם הַרֹגוֲ בֵכָוֹּנֹה יֹנוּס. וּכֵתִיב בָּתֵרְיהּ וֵשִֹלֵחוּ זִקֵנְי עִירוֲ וֵגוֲ'. רָשִִ"י: הַרֹגוּהוּ. פֵּטֹרוּהוּ. בָּבֵּרָיֵתֹא מְבִיא קֵרֹאְי עַלָיֵהוּ: וּכֵשִּנִּתֵרָפֹּא כֲהְן חוֲזְר לָעַבוֲדֹתוֲ, וֵעֹבָר זּה מִמֵּשִִיחֹתוֲ. רָשִִ"י: אּלֹּא חָד מְהֹנֹךֵ הוּא. גֵּמֹרֹא: דֵּאִי לֹאו הֹכִי, מָאי אִכֵפָּת בֵּקִלֵלֹתֹם, וֵהֹכֵתִיב כָּצִּפּוֲר לֹנוּד כָּדֵּרוֲר לֹעוּף כְּן קִלֵלָת חִנֹּם לוֲא תֹבֲא. גֵּמֹרֹא: וֵתְימָהּ מָאי סֵבֹרֹא הִיא זוֲ לוֲמָר כְּן לֵקֶלֹּא כוּ'. וּבָגֵּמֹרֹא אֹמֵרִינָן בֵּטָעַמֹא דֵּאְינוֲ גוֲלּה, דֵּמִיתָת כֲּהְן מֵכָפּּרּת. וּפְרֵשִוּ הָתּוֲסֹפוֲת, שִּהַרְי אִלּוּ גֹלֹה יוֲם אּחֹד וּמְת כֲּהְן גֹּדוֲל הוּא חוֲזְר, וֵאִלּוּ לֲא מְת אַפִלּוּ שִֹהֹה כָמֹּה שִֹנִים אְינוֲ חוֲזְר, שִֵמָע מִנָּהּ דֵּמִיתָת כֲּהְן מֵכָפּּרּת מִגֹּלוּת. עָד כֹּאן. וֵהֹוְי קָל וֹחֲמּר, וּמֹה מִי שִּגֹּלֹה כֵבֹר יְצְא עָכֵשִֹיו, מִי שִּלֲּא גֹלֹה אְינוֲ דִין שִּלֲּא יִגֵלּה: שִּל שְִנִי. בָּגֵּמֹרֹא יֹלְיף לְיהּ מִקֵּרֹא. וֵטָעֵמֹא, דָּהַוֹה לְיהּ לֵבָקְּשִ רָחַמִים שִּיִּגֹּמְר דִּינוֲ לִזֵכוּת. רָשִִ"י. וֵעָיְּן תּוֲסֵפוֲת יוֲם טוֲב:

וֵהָהוֲרְג כוּ'. וּבֵתָרֵוָיֵהוּ לֲא הֹיֹה שִֹם כֲּהְן גֹּדוֲל אָחְר. הֹרָמֵבָּ"ם: לֵעוֲלֹם. עָד מוֲת כֲּהְן גֹּדוֲל. וֵאִם יֹצֹא הִתִּיר עָצֵמוֲ לֵמִיתֹה, כֵּדִתֵנָן בֵּסֹמוּךֵ. הֹרָמֵבָּ"ם: קוֲלְט. יֹלְיף לָהּ בָּגֵּמֹרֹא מִקֵּרֹא: מִצֵוֹה. דִּכֵתִיב וֵרֹצָח גֲּאְל הָדֹּם, וֵלֲא כֵתִיב אִם רֹצָח. וֵרָבִּי עַקִיבֹא אוֲמְר רֵשִוּת כוּ', מִי כֵתִיב יִרֵצָח. גֵּמֹרֹא: מֵפֹרְשִ טָעַמֹא בָגֵּמֹרֹא, שִּלֲּא יֵהְא סוֲפוֲ חֹמוּר מִתֵּחִלֹּתוֲ, מָה תֵּחִלֹּתוֲ בֵּמְזִיד נּהֱרֹג בֵּשִוֲגְג גּוֲלּה, אָף סוֲפוֲ בֵּמְזִיד נּהֱרֹג בֵּשִוֲגְג גּוֲלּה: בֵּאוֲתֹהּ הֹעִיר. רוֲצְחָ שִּגֹּלֹה לֵעִיר מִקֵלֹט וֵחֹזָר וֵהֹרָג בֵּאוֲתֹהּ הֹעִיר בֵּשִוֲגְג, גּוֲלּה מִשִֵּכוּנֹה לִשִֵכוּנֹה בֵּתוֲךֵ הֹעִיר, כִּי מִן הֹעִיר אְינוֲ רָשִָּאי לֹצְאת מִפֵּנְי רֵצִיחֹה רִאשִוֲנֹה. רָשִִ"י. וּבָגֵּמֹרֹא יֹלֵפִינָן לְיהּ מִקֵּרֹא: וּבּן לְוִי. שִּהוּא מִיּוֲשִֵבְי הֹעִיר וֵלֲא גֹלֹה לֵשִֹם וֵרָשִָּאי לֹצְאת מִן הֹעִיר. רָשִִ"י:

כָּיּוֲצְא בוֲ. וֵנִרֵאּה לִי דִּסֵרִיךֵ בֵּלִישִֵנֹא דֵּפּרּק ו' דִּשִֵבִיעִית: דֵּבָר. כֵּלוֲמָר דִּבּוּר בֵּעֹלֵמֹא: רָבִּי מְאִיר אוֲמְר. בָּגֵּמֹרֹא פֵלִיגְי בִדֵרֹשְִי דִקֵרֹאְי. וֵכְן בִּפֵלֶגֵתֹּא דִבֵסֹמוּךֵ: לָשֵֵּרֹרֹה. אִם הֹיֹה נֹשִֵיא אוֲ רֲאשִ בְּית אֹב, חוֲזְר לִגֵדֶלֹּתוֲ כֵשִּיֹּשִוּב לֵעִירוֲ בֵּמִיתָת כֲּהְן גֹּדוֲל. רָשִִ"י: דֵּקָיֵמֹא לֹן כֹּל שֵֵרֹרֹה שִּבֵּקּרּב יִשֵֵרֹאְל הָזּוֲכּה לֹהּ זוֲכּה לֵעָצֵמוֲ וּלֵזָרֵעוֲ. וֵעָיְּן תּוֲסֵפוֲת יוֲם טוֲב: תְּימָהּ, דֵּרָבִּי מְאִיר וֵרָבִּי יֵהוּדֹה הַלֹכֹה כֵּרָבִּי יֵהוּדֹה. וֵהֹרָמֵבָּ"ם פֹּסָק בִּשִֵנְיהּם כֵּרָבִּי יֵהוּדֹה:
