פרק ו

הָדָּיֹּנִין. הֹא דֵלֹא נֹקָט נוֲהּגּת בָּאַנֹשִִים וּבֵנֹשִִים כֵּדִתֵנָן לֵעְיל גָּבְּי פִקֹּדוֲן, נִרֵאּה לִי מִשִּוּם דִּשִֵבוּעָת הָדָּיֹּנִין הִיא עָצֵמֹהּ שִֵבוּעָת הָפִּקֹּדוֲן. וֵזּה לֵשִוֲן הֹרָמֵבָּ"ם, שִּדֹּבֹר בֹּרוּר וֵגֹלוּי שִּכֹּל הָנִּשִֵבֹּע שִֵבוּעָת הָדָּיֹּנִים אוֲ הּסְּת בֵּשִּקּר, שִּיֵּהְא חָיֹּב מִשִּוּם שִֵבוּעָת הָפִּקֹּדוֲן: רָשִִ"י. פְּרוּשִ לֵפְרוּשִוֲ, שִּאָף עָל פִּי שִּשִֵּבוּעָת הָפִּקֹּדוֲן הוּא בֵטוֲעְן נּאּנֵסוּ אוֲ נּאּבֵדוּ אוֲ נִגֵנֵבוּ, וֵכְן שִֵבוּעָת עְד אּחֹד שִֵבוּעָת הָדָּיֹּנִין נָמִּי הִיא, הֹכִי קֹתֹנְי, שִֵבוּעָת הָדָּיֹּנִין שִּל מוֲדּה בֵמִקֵצֹת צֹרִיךֵ שִּתֵּהְא הָטָּעַנֹה שִֵתְּי כּסּף וֵהָהוֲדֹאֹה שִֹוּה פֵרוּטֹה. וֵלֹאו כֹּל שִּשִֵּבוּעָת הָדָּיֹּנִין קֹתֹנְי. אִי נָמִּי, דִּשִֵבוּעָת הָפִּקֹּדוֲן לֹאו דֵּדָיֹּנִין הִיא, דֵּשְִם אָחְר יְשִ לֹהּ. וֵכְן שִֵבוּעָת עְד אּחֹד כְּיוֹן דֵּלֲא נִכֵתֵּבֹה בֵהּדֵיֹא לְיתֹא בִכֵלָל זּה. הֹרָ"ן. וֵטָעַמֹא דֵעְד אּחֹד לֲא בֹעְי שִֵתְּי כּסּף, מִשִּוּם דֵּכֹל מֹקוֲם שִּשִֵּנָיִם מֵחָיֵּבִין מֹמוֲן עְד אּחֹד מֵחָיֵּבוֲ שִֵבוּעֹה: וֵטָעַמֹא מֵפֲרֹשִ בָּיֵרוּשִָלֵמִי, שִּאֹמֵרוּ סוֲף מָטֵבְּעָ כּּסּף, מֹעֹה, כֵּלוֲמָר כְּיוֹן שִּאֹמֵרֹה תוֲרֹה כּּסּף, כֵּדִכֵתִיב כִּי יִתְּן כוּ' כּּסּף כוּ', וֵאִי אָתֹּה מוֲצְא מָטֵבְּעָ כּּסּף מֶזֵכֹּר בָּתּוֲרֹה פֹּחוֲת מִמֹּעֹה, כֵּדִמֵתָרֵגֵּמִינָן עּשֵֵרִים גְּרֹה, עּשֵֵרִים מֹעִין, אְין לֵךֹ לֵחָיֵּבוֲ בֵּפֹחוֲת מִמָּטֵבְּעָ זוֲ. וּמָקֵשִּה, וּתֵהְא מֹעֹה, כֵּלוֲמָר מִנָּיִן לֵךֹ שִּיִּהֵיוּ שִֵתָּיִם. וּמֵפֹרְק, אוֲ כְלִים, מָה כְּלִים שִֵנָיִם, אָף כּּסּף שִֵנָיִם. הֹרָ"ן: וֵרָב הוּא דֵּאֹמָר הֹכִי בָגֵּמֹרֹא. וּפְרְשִ רָשִִ"י טָעַנֹה דִתֵנָן בֵּמָתֵנִיתִין לֹאו אָתֵּחִלָּת טָעַנֹה קֹאְי, אּלֹּא אָכֵּפִירֹה שִּהוּא כוֲפְר בֵּטָעַנָת חַבְרוֲ לֵבָד הוֲדֹאָת הָפֵּרוּטֹה, דֵּבֹעִינָן שִּתֵּהְא תֵחִלָּת הָטָּעַנֹה שִֵתְּי כּסּף וּפֵרוּטֹה. וֵיֹלְיף לְיהּ בָּגֵּמֹרֹא, דֵּכּסּף לֹמֹּה לִּי (וָהַוֹה דֹרֵשִִינָן מָה כְּלִים שִֵנָיִם אָף כֹּל שִֵנָיִם), אִם אְינוֲ עִנֵיֹן לֵטָעַנֹה תֵּנְהוּ עִנֵיֹן לִכֵפִירֹה כוּ'. וֵנִמֵצֹא שִּמָּה שִּּכֹּתָב הֹרָ"ב בִּתֵחִלָּת הָמִּשִֵנֹה שִּלֲּא תֵהְא הָטָּעַנֹה, לֲא בֹא לֵדָקֵדְּק בֵּפְרוּשִ הָטָּעַנֹה מָמֹּשִ. וֵעָיְּן תּוֲסֵפוֲת יוֲם טוֲב: כִּדֵאָשִֵכֵּחָן בֵּאָבֵרֹהֹם דֵּאָתֵפֵּסְיהּ לּאֱלִיעּזּר בִּבֵרִית מִילֹה, שִּנּּאֱמָר (בֵּרְאשִִית כד) שִֵים נֹא יֹדֵךֹ תָּחָת יֵרְכִי וֵאָשִֵבִּיעַךֹ. גֵּמֹרֹא. וֵעָיְּן תּוֲסֵפוֲת יוֲם טוֲב: לֵשִוֲן שִֵימֹה. שִּשֵֹּמוּ חַכֹמִים עֹלֹיו שִֵבוּעֹה. כֵּמוֲ אִם ה' הֱסִיתֵךֹ (שִֵמוּאְל א כו): אָף עָל פִּי שִּהִיא נִזֵכּּרּת בָּמִּשִֵנֹה. הֹרָמֵבָּ"ם: בֵּשִֹוּה פֵרוּטֹה. דֵּמִשִֵנֹתְנוּ בִּשִֵבוּעָת הָדָּיֹּנִים עָל יֵדְי הוֲדֹאָת מִקֵצֹת הָטָּעַנֹה. כִּדֵפְרְשִ הֹרָ"ב. וּמֵסָיְּם רָשִִ"י, שִּלֵּמֹדוּהֹ מְאַשִּר יֲאמָר כִּי הוּא זּה. (דֵּעְרוּב פֹּרֹשִִיּוֲת כֵּתִיב כֹּאן וֵכִי הוּא זּה דִּכֵתִיב בֵּפֹרֹשִָת שִוֲמֵרִים שִֹדְי אָפֹּרֹשִָת אִם כּּסּף תָּלֵוּה) וּכֵתִיב עַלָהּ וֵנִקֵרָב בָּעָל הָבָּיִת אּל הֹאֱלֲהִים אִם לֲא שִֹלָח יֹדוֲ, דֵּהָיֵנוּ שִֵבוּעֹה. וֵעָיְּן תּוֲסֵפוֲת יוֲם טוֲב: פֹּטוּר. דֵּכִי הוּא זּה מָשִֵמָע מִמִּין הָטָּעַנֹה: וֵאָלִּיבֹּא דִשִֵמוּאְל מֵפֹרֵשִִינָן הֹכִי בָגֵּמֹרֹא, וּסֵבִירֹא לְיהּ דֵּטָעַנֹה דֵמָתֵנִיתִין בִּתֵחִלָּת טָעַנֹה, וֵלֲא בֹעִינָן שִֵתְּי כּסּף וּפֵרוּטֹה, וֵהֹכֹא שִֵתְּי כּסּף דָּוֵקֹא אֹמָר וֵלֲא שִֹוּה, וּלֵכֹךֵ פֹּטוּר, דֵּמָה שִּּטֵּעֹנוֲ לֲא הוֲדֹה לוֲ. וֵכְיצָד אָמָּאי דִּסֵמִיךֵ לְיהּ קֹאְי, דֵּאִם אְין הָהוֲדֹאֹה כוּ'. אַבֹל לֵרָב, שִֹוּה קֹאֹמָר. וֵכְיצָד אֹרְישִֹא קֹאְי. וֵקָשִֵיֹא לֵהֹרָ"ב דִּמֵפֹרְשִ לֹרְישִֹא אָלִּיבֹּא דֵרָב, וֵהֹכֹא (וּבֵבֹבֹא מֵצִיעֹא פּ"ד מ"ז) מֵפֹרְשִ אָלִּיבֹּא דִשִֵמוּאְל. וֵעָיְּן תּוֲסֵפוֲת יוֲם טוֲב: כֵּמְשִִיב אַבְדֹה. שִּהֹיֹה יֹכוֲל לֵהֹעִיז פֹּנֹיו וֵלִכֵפֲּר, וּמִדֵּלֲא הְעִיז הַרְי הוּא כֵמְשִִיב אַבְדֹה. רָשִִ"י:

שֵוֲחְק הֹיִיתִי בֵךֹ, שִּהֹיִיתֹ שִוֲאַלְנִי מָה שִּּלֲּא הֹיֹה. רָשִִ"י: לֵשִוֲן רָשִִ"י, כֵּגוֲן שִּאוֲמְר לֲא הֹיוּ דֵבֹרִים מְעוֲלֹם. רֵצוֲנוֲ לוֲמָר, שִּפְּרְשִ דֵּבֹרֹיו וֵאֹמָר אְין לֵךֹ בֵיֹדִי מְעוֲלֹם. אַבֹל אִם אֹמָר סֵתֹם אְין לֵךֹ בֵיֹדִי לֲא הֶחֵזָק כָּפֵרֹן, דֵּמֹצְי לֵמְימָר אְין לֵךֹ בֵיֹדִי עָתֹּה קֹאַמִינֹא, שִּהַרְי פֵרָעֵתִּיו לֵךֹ. וּמֵתֹרְץ דִּבּוּרְיהּ הֹוְי, וֵאְין סוֲתְר דֵּבֹרֹיו הֹרִאשִוֲנִים. הֹרֲא"שִ. וֵעָיְּן תּוֲסֵפוֲת יוֲם טוֲב שִּאְין כְּן דָּעָת רָב הָאי: רָשִִ"י. וּפְרוּשִ לֵפְרוּשִוֲ, שִּכֵּשִּחוֲזְר אָחָר כֹּךֵ וֵטוֲעְן פֹּרָעֵתִּי אְינוֲ נּאֱמֹן בִּשִֵבוּעֹה כוּ', וֵאָהֹא מֵסָיְּם נָמִּי דִּבֵשִּלֲּא בֵבְית דִּין, כֵּלוֲמָר, שִּזּּה שִּאֹמָר אְין לֵךֹ בֵיֹדִי הֹיֹה שִּלֲּא בֵבְית דִּין, אִם בֹּא לֵבְית דִּין וֵטוֲעְן פֹּרָעֵתִּי לֲא הֶחֵזָק כָּפֵרֹן, וֵנּאֱמֹן. וֵהָטָּעָם פְּרְשִ הֹרֲא"שִ, דָּעַבִיד אֱינִישִ דֵּלֲא מֵגָלּּה טָעַנוֲתֹיו אּלֹּא בֵבְית דִּין: שִּצֹּרִיךֵ כוּ'. כְּיוֹן שִּאֹמָר לוֲ אָל תִּתֵנְהוּ לִי אּלֹּא בֵעְדִים. מְהֹא שִֹמֵעִינָן דָּאַפִלּוּ לֵאָחָר הָלֵוֹאֹה יֹכוֲל הָמָּלֵוּה לוֲמָר לָלֲּוּה אָל תִּפֵרֹעְנִי אּלֹּא בֵעְדִים. דֵּהֹא מָתֵנִיתִין לֵאָחָר הָלֵוֹאֹה הִיא, וֵקֹתֹנְי בֵהּדֵיֹא דֵּחָיֹּב מִפֵּנְי כוּ'. וֵהָמָּגִּיד כֹּתָב הָטָּעָם, מִשִּוּם דֵּעּבּד לֲוּה לֵאִישִ מָלֵוּה. וֵעָיְּן תּוֲסֵפוֲת יוֲם טוֲב:

וָאַפִלּוּ לֵרָב דִּסֵבִירֹא לְיהּ בִּסֵתֹמֹא שִֹוּה טוֲעְן, לִיטֵרֹא שִֹאנְי, דֵּכְיוֹן דֵּטוֲעַנוֲ מִשִֵקֹל, וָדָּאי דָּוֵקֹא קֹתֹנְי. גֵּמֹרֹא. וֵרָשִִ"י: וּטֵרִיסִית כוּ'. אוֲ אוֲ קֹתֹנְי. רָשִִ"י: שִּהָכֲּל כוּ'. בָּגֵּמֹרֹא פֹּרֵכִינָן לִשִֵמוּאְל דֵּאֹמָר דָּוֵקֹא טוֲעַנוֲ, אָמָּאי חָיֹּב. וּמוֲקֵמִינָן בֵּשִּטּוֲעַנוֲ בֵּדִינֹר מָטֵבֵּעוֲת. פְּרְשִ רָבְּנוּ חַנָנֵאְל, כֵּלוֲמָר, טוֲעְן דִּינָר זֹהֹב נֹתָתִּי לֵךֹ לֹתְת לִי בוֲ מָטֵבֵּעוֲת, וֵהוּא אוֲמְר לֲא נֹתָתֹּ לִּי אּלֹּא דִּינָר כּּסּף לֹתְת לֵךֹ מָטֵבֵּעוֲת וֵיְשִ לֵךֹ בֵשִֹוֵיוֲ בֵיֹדִי טֵרִיסִית כוּ'. וֵקֹא מָשִֵמָע לֹן דִּפֵרוּטוֲת אִיתֹא בִכֵלָל מָטֵבֵּעוֲת שִּאֹמָר לוֲ לִקָּח. וֵרָב מֵפֹרְשִ דֵּהֹכִי קֹאֹמָר, שִּכֹּל דִּין מָטֵבְּעָ אָחָת הִיא, כֵּלוֲמָר, דִּבֵכֶלֵּהוּ שִֹוּה קֹאֹמָר. וֵשִּאְינוֲ מָטֵבְּעָ כֵּגוֲן לִיטֵרֹא, לֵעוֲלֹם דָּוֵקֹא קֹאֹמָר. הֹרָמֵבָּ"ן וֵהֹרָ"ן: פֹּטוּר. בֵּפּרּק ח' דֵּבֹבֹא מֵצִיעֹא מִשִֵנֹה ב' כֹּתָב הֹרָ"ב, אָף מִדֵּמְי שֵֵעֲרִים. וֵטָעַמֹא, כֹּתָב הָטּוּר דֵּחֹשְִיב כֵּאִלּוּ הוֲדֹה הָתּוֲבְעָ שִּאְינוֲ חָיֹּב לוֲ שֵֵעֲרִים. וֵכֹתָב אֹבִי אַדוֲנִי הֹרֲא"שִ ז"ל, דָּוֵקֹא כֵּשִּאֹמָר לוֲ הִלֵוִיתִיךֹ חִטִּין בֵּאוֲתוֲ יוֲם וּבֵאוֲתֹהּ שִֹעֹה, וֵהָלֹּה אוֲמְר שֵֵעֲרִים הֹיוּ כוּ'. דֵּאִי לֲא תְימֹא הֹכִי, פֵּשִִיטֹא שִּהוּא חָיֹּב. שִּאְין אֹדֹם מוֲחְל עָל שִֵאֹר תֵּבִיעוֲתֹיו אִם לֲא תֹבָע אּלֹּא אָחָת: מֵחָיְּב. דֵּלֲא בֹעְי הוֲדֹאֹה מִמִּין הָטָּעַנֹה. רָשִִ"י. וּכֵלוֲמָר דִּשִֵבוּעֹה הוּא דִמֵחָיֵּבוֲ, אַבֹל מִדֵּמְי שֵֵעֲרִין אָף הוּא פוֲטֵרוֲ, וֵלֲא פֹלְיג בֵּהֹא אָרָבֹּנֹן. תּוֲסֹפוֲת. וֵעָיְּן תּוֲסֵפוֲת יוֲם טוֲב: רָשִִ"י. וּבְין אָדֵמוֲן וּבְין חַכֹמִים סֵבִירֹא לֵהוּ דֵּטוֲעַנוֲ בָקָּנֵקָנִּים גָּם כְּן. וּבֵהֹא פֵלִיגְי, אָדֵמוֲן סוֲבְר טוֲעַנוֲ חִטִּין וּשֵֵעֲרִין וֵהוֲדֹה בֵאָחָת מְהּן חָיֹּב. וָחַכֹמִים אוֲמֵרִים פֹּטוּר, דֵּבֹעִינָן טָעַנֹה וֵהוֲדֹאֹה בֵּדֹבֹר אּחֹד. גֵּמֹרֹא בָּסּוּגֵיֹא דֵהֹכֹא. וּבָסּוּגֵיֹא דִכֵתֶבּוֲת מֵפֹרְשִ לְיהּ בֵּעִנֵיֹן אָחְר כֵּמוֲ שִּכֹּתָב שִֹם הֹרָ"ב: כֵּדִלֵקָמֹּן בֵּמִשִֵנֹה ה'. וּמִשִּוּם הֹכִי נָמִּי כֵּשִּהוֲדֹה בָכְּלִים פֹּטוּר מִשִֵּבוּעֹה: פֹּטוּר. אִצֵטֵרִיךֵ, דֵּלֲא תְימֹא טָעַמֹא דֵפֹטוּר בֹּרְישִֹא מִשִּוּם דָּהַוְי כִטֵעֹנוֲ חִטִּין וֵהוֲדֹה לוֲ בִשֵֵעֲרִין. תּוֲסֹפוֲת. וּפְרְשִ בֵּחֹכֵמָת שִֵלֲמֲה דֵּלֲא תְימֹא דִּסֵבִירֹא לְיהּ דָּאַפִלּוּ טֵעֹנוֲ בִשִֵנְי מִינִין וֵהוֲדֹה בֵאּחֹד הֹוְי כֵמוֲ טֵעֹנוֲ חִטִּין וֵהוֲדֹה בִשֵֵעֲרִין דֵּפֹטוּר:

שִּלֲּא יֵהְא זּה נוֲטְל מֹמוֲנוֲ כֵשִּהוּא קֹטֹן וֵיְלְךֵ לוֲ בֵחִנֹּם כוּ'. הָחְרְשִ וֵהָשִּוֲטּה אְין נִזֵקֹקִין לֹהּם כוּ' לֲא לִשִֵבוּעֹה קָלֹּה כוּ'. הֹרָמֵבָּ"ם: תֵּנְינֹא כוּ'. וּמֵשִֹנְי, חַדֹא לִקֵטָנִּים וָחַדֹא לִגֵדוֲלִים. גֵּמֹרֹא: תֵּנְינֹא כוּ'. אִצֵטֵרִיךֵ, סֹלֵקֹא דָעֵתֹּךֵ אַמְינֹא הּדֵיוֲט הוּא דֵּאֹדֹם עוֲשֵּה קֵנוּנֵיֹא עָל הּדֵיוֲט, אַבֹל הּקֵדְּשִ דֵּאְין אֹדֹם עוֲשֵּה קֵנוּנֵיֹא עָל הּקֵדְּשִ, קֹא מָשִֵמָע לֹן:

דָּהַוֹה מֹצְי לֵמְימָר שִוֲאְל אְינוֲ מֵשִָלְּם אֳנֹסִין וּגֵנְבֹה וָאַבְדֹה, דֵּאִתָּקָּשִ לֵשִוֲמְר שֵֹכֹר. כִּדֵאִיתֹא בָגֵּמֹרֹא:
וֵקֹשִּה, דִּבֵקָרֵקֹעוֲת נָמִּי שִָיֹּךֵ שִֵאְלֹה, וֵכִדֵאִיתֹא סוֲף פּּרּק ח' דֵּבֹבֹא מֵצִיעֹא, עָיְּן שִֹם. וֵהָנִּמּוּקְי יוֲסְף כֹּתָב, מִשִּוּם דֵּסְיפֹא אָרֵבֹּעֹה וָחַמִשִֹּה לֲא קֹאְי אּלֹּא אָהּקֵדְּשִוֲת, וּבֵהּקֵדְּשִ לֲא שִָיֹּךֵ שִֵאְלֹה:

וֵאִם תֲּאמָר אָמָּאי בֹּעִינָן הוֲדֹאֹה בֵמִקֵצֹת, וֵהֹא כִּי הוּא זּה דֵּמִינְיהּ יֹלֵפִינָן דֵּבֹעִינָן הוֲדֹאֹה בֵמִקֵצֹת שִֹדִינָן לְיהּ אָפֹּרֹשִָת אִם כּּסּף תָּלֵוּה. וֵיְשִ לוֲמָר דֵּלֹאו דֵּעֹקֵרִינָן לֵגָמֵרְי מִפֹּרֹשִֹה שִּכֹּתוּב בֹּהּ, אּלֹּא עְרוּב פֹּרֹשִִיּוֲת אֹמֵרִינָן, וֵקֹאְי נָמִּי אָפֹּרֹשִָת אִם כּּסּף. אֹמֵנֹם לֵדָעָת הֹרָמֵבָּ"ם שִּאְין אּחֹד מְהָשִּוֲמֵרִים צֹרִיךֵ לֵהוֲדוֲת בֵּמִקֵצֹת, וֵכְן פְּרְשִ רָשִִ"י דֵּעֹקֵרִינָן לָהּ לֵגָמֵרְי מִפֹּרֹשִָת שִוֲמֵרִים וֵשִֹדִינָן אָפֹּרֹשִָת אִם כּּסּף, צֹרִיךֵ עִיּוּן. וֵעָיְּן תּוֲסֵפוֲת יוֲם טוֲב:
דֵּכְיוֹן דֵּאָדָּעֵתֹּא לֵמִשִֵקָל זָבֵּנִינֵהוּ נִיהֹלְיהּ, כִּבֵצוּרוֲת דֹּמֵיֹן, וֵדֹנִינָן בֵּהוּ דִּין מִטָּלֵטֵלִין בֵּכֹל מִלְּי, כֵּגוֲן דִּין אוֲנֹאֹה וּשִֵבוּעֹה וֵכָיּוֲצְא בֹּהּן כוּ'. וֵגָבְּי שִוֲמְר הָיֵנוּ טָעַמֹא דֵּכְיוֹן דֵּלִשִֵמִירֹה הְן כֵּשִּהְן מֵחֶבֹּרִין לָקָּרֵקָע וֵאָדָּעֵתֹּא דֵהֹכִי מָסֵרִינֵהוּ נִיהֹלְיהּ, הֹוְי לְיהּ כֵּקָרֵקָע, דֵּהֹא לֹאו לֵמִתֵלֵשִִינֵהו מָסֵרִינֵהוּ נִיהֹלְיהּ. טוּר בֵּשְִם הֹרִ"י הָלְּוִי. וֵנִרֵאּה לִי לֵדָעָת הֹרָמֵבָּ"ם אַפִלּוּ בֵאְין צֹרִיךֵ לָקָּרֵקָע. וֵעָיְּן תּוֲסֵפוֲת יוֲם טוֲב:
מֵחָיְּב כוּ'. וֵאִם תֲּאמָר וֵהֹא הְילֹךֵ הוּא. יְשִ לוֲמָר מָשִֵכָּחָת לָהּ כֵּשִּבֵּצֹרֹן. הֹרָ"ן. וֵעָיְּן תּוֲסֵפוֲת יוֲם טוֲב: שִּבָּמִּדֹּה כוּ'. נֹפֵקֹא לֹן מִמָּשִֵמֹעוּת דֵּכִי הוּא זּה. תּוֲסֹפוֲת. וֵהֹרָ"ן כֹּתָב, דֵּבִמֵכִילֵתֹּא מָפִּיק לְיהּ מִכּּסּף אוֲ כְלִים, מָה כּּסּף שִּדָּרֵכּוֲ לִמֹּנוֲת אָף כְּלִים שִּדָּרֵכֹּן לֵהִמֹּנוֲת. וֵעָיְּן תּוֲסֵפוֲת יוֲם טוֲב: לִישִֵנֹא דִּבֵרָיֵתֹא. וֵלָאַתוֲיְי טוֲעַנוֲ בָּיִת זּה מֹלְא. דֵּלֲא תְימֹא דֵּכְיוֹן דֵּאֹמָר בָּיִת זּה מֹלְא, וֵזּה מָחַזִיר לוֲ חֹסְר, נִרֵאּה הָחִסֹּרוֲן, וָהַרְי הוּא כֵּזּה אוֲמְר עָד הָזִּיז וֵזּה אוֲמְר עָד הָחָלּוֲן, קֹא מָשִֵמָע לֹן דֵּלֲא, עָד שִּיּוֲדּה בֵּדֹבֹר שִּבָּמִּדֹּה. וֵעָיְּן תּוֲסֵפוֲת יוֲם טוֲב: חָיֹּב. לֵדִידֹן דֵּקָיֵמֹא לֹן הְילֹךֵ פֹּטוּר, עָל כֹּרֵחַךֹ כֵּשִּהִרֵקִיבוּ פְרוֲת מָיֵרְי כוּ'. וֵרְישִֹא דֵּמָה שִּּהִנָּחֵתֹּ כוּ' פֹּטוּר וֵאָף עָל גָּב דָּהַרְי אֹנוּ רוֲאִין הָפְּרוֲת וֵהוֲדֹאֹתוֲ יֵדוּעֹה, נִרֵאּה לִי דֵּהָיֵנוּ טָעַמֹא מִשִּוּם דִּבֵשִֹעֹה שִּהוֲדֹאֹה יוֲצֵאֹה מִפִּיו בֹּעִינָן שִּתֵּהְא דֹּבֹר שִּבָּמִּדֹּה:

וֵלֲא אֹמֵרִינָן שִּלֲּא הוֲדֹה לוֲ שִּיִּתֵחָיְּב לֵשִָלְּם לוֲ אּלֹּא אִם כְּן שִּיָּחַזִיר לוֲ מָשִֵכּוֲנוֲ וֵכֹל עוֲד שִּלֲּא יוּכָל לָחַזֲר לוֲ מָשִֵכּוֲנוֲ נִמֵצֹא שִּלֲּא הוֲדֹה לוֲ בֵּשִוּם חִיּוּב, מִשִּוּם דֵּמָשִֵכּוֲן לֹאו פְּרֹעוֲן הוּא וֵיֹכוֲל הָמָּלֵוּה לֹכוֲף לָלֲּוּה שִּיֵּשִָלְּם לוֲ חוֲבוֲ אָף עָל פִּי שִּרוֲצּה לֵעָכְּב הָמָּשִֵכּוֲן אּצֵלוֲ בִּדֵמְי שִֹוֵיוֲ שִּאוֲמְר, וֵעוֲבְר בֵּלֲא תָחֵמֲד כוּ', וֵאְין הָלֲּוּה נִפֵטֹר מְחִיּוּבוֲ בֵּכֹךֵ, וֵחָיֹּב בֵּבֹא לֹצְאת יֵדְי שִֹמָיִם לֵפֹרֵעוֲ אּת הֹעֲדּף. הֹרָ"ן: וֵהִקֵשִוּ הָתּוֲסֹפוֲת, וֵכִי עֹבֵדִינָן תָּקֹּנֹה לִרֵשִִיעְי. אּלֹּא טָעַמֹא שִּמֹּא יוֲצִיא הָלֹּה הָפִּקֹּדוֲן וֵנִמֵצֹא שִֵבוּעָת לֲוּה לֵבָטֹּלֹה: פֹּטוּר. אוֲקִימֵנֹא בָגֵּמֹרֹא בֵּמָאַמִינוֲ לֲוּה לָמָּלֵוּה שִּאְינוֲ בִרֵשִוּתוֲ. שִּאִם לֲא כְן מִתּוֲךֵ שִּצֹּרִיךֵ לִשִֹּבָע שִּאְינוֲ בִרֵשִוּתוֲ נִשִֵבֹּע נָמִּי כָמֹּה הֹיֹה שִֹוּה עָל יֵדְי גִלֵגּוּל, אָף עָל פִּי שִּבָּשִִּוּוּי אְינוֲ מָאַמִינוֲ, הָיֵנוּ מִשִּוּם דֵּלֲא קִים לְיהּ לָמָּלֵוּה בֵּגָוְּיהּ. וּמָלֵוּה אְינוֲ מָאַמִין לָלֲּוּה, דִּמֵקָיְּם בְּיהּ וֵסּלּף בֲּגֵדִים יֵשִֹדְּם (מִשִֵלְי יא), שִּאִם הֹיֹה נּאֱמֹן וֵיֹשִֹר לֲא הֹיוּ מָצֵרִיכִין לוֲ מִן הָשִֹּמָיִם שִּיֵּהְא לֲוּה. אַבֹל לֲוּה מֵקָיְּם בָּמָּלֵוּה תֶּמָּת יֵשִֹרִים תָּנֵחְם, שִּאִם לֲא שִּנּּאֱמֹן וֵיֹשִֹר הוּא לֲא הֹיוּ מָעַשִִירִין אוֲתוֲ. וֵכֹתָב הֹרָ"ן, וּבָסְּיפֹא דֵרְישִֹא מֵחָיֵּבִינָן לָמָּלֵוּה לִשִֹּבָע שִּאְינוֲ בִרֵשִוּתוֲ וֵאָף עָל פִּי שִּמָּאַמִינוֲ, שִּמֹּא יוֲצִיא כוּ'. וֵעָיְּן תּוֲסֵפוֲת יוֲם טוֲב:
שִּמֹּא כוּ'. וֵאִם תֲּאמָר, וֵלֹמֹּה לֲא שִֹנֹה נָמִּי הָטָּעָם דֵּעִקָּר שִֵבוּעֹה לֹמֹה הוּא דֵּשִּמֹּא נֹתָן עְינֹיו בֹּהּ. תְּרְץ הֹרָמֵבָּ"ם לֵפִי שִּזּּה הָטָּעָם כּוֲלְל כֹּל מָשִֵכּוֲן. אַבֹל שִּמֹּא עְינֹיו כוּ' אְינוֲ כוֲלְל כֹּל מָשִֵכּוֲן, שִּאִם הֹיֹה הָמָּשִֵכּוֲן לֵשִוֲן זֹהֹב וֵכָדּוֲמּה, אְין כֹּאן טָעָם שִּמֹּא עְינֹיו נֹתָן בֹּהּ. וֵעָיְּן תּוֲסֵפוֲת יוֲם טוֲב:
