פרק ב - משנה א

רַשִׁ"י. מַשְׁמַע שֶׁפֵּרֵשׁ כֻּלַּהּ מַתְנִיתִין בִּידִיעָה גְמוּרָה. שֶׁפֵּרֵשׁ וְיָדַע שֶׁנִּטְמָא כוּ'. וְהַמְדַקְדֵּק יָכוֹל לְפָרֵשׁ כֻּלַּהּ מַתְנִיתִין בִּידִיעַת בֵּית רַבּוֹ. תּוֹסָפוֹת. וְעַיֵּן פֶּרֶק קַמָּא מִשְׁנָה ג'. וְעוֹד דַּע, שֶׁעִם זֶה שֶׁצָּרִיךְ שֶׁתְּהֵא לוֹ יְדִיעָה שֶׁנִּטְמָא צָרִיךְ גַּם כֵּן שֶׁתְּהֵא לוֹ יְדִיעָה שֶׁזֶּה מִקְדָּשׁ וְשֶׁזֶּה קֹדֶשׁ:
וְהַיְנוּ טַעֲמָא נַמִּי דְּלֹא תָנֵי אַרְבַּע שֶׁהֵם שְׁמֹנֶה, וְהַיְנוּ עוֹד אַרְבַּע יְדִיעוֹת בַּסּוֹף, דְּלֹא חָשֵׁיב הָנֵי יְדִיעוֹת דְּאִיתְנְהוּ בְכָל הַתּוֹרָה כֻּלָּהּ, שֶׁאֵין אָדָם מֵבִיא חַטָּאת אֶלָּא אִם כֵּן נוֹדַע לוֹ שֶׁחָטָא, כִּדְאִיתָא בַגְּמָרָא: בְּהֶדְיָא. אֶלָּא מֵרִבּוּיָא דִּוְנֶעְלַם דָּרְשִׁינַן לַהּ בְּסוֹף פִּרְקִין:
וְלֹא קַשְׁיָא אַמַּאי לֹא חָשֵׁיב לְהוּ וְלֶהֱוֵי שֵׁשׁ, דְּמִכֵּיוָן דְּחָשְׁבִינַן יְדִיעַת הַטֻּמְאָה דְלַקֹּדֶשׁ וְכֵן דְּלַמִּקְדָּשׁ, וְכֵן יְדִיעַת הַמִּקְדָּשׁ גַּם יְדִיעַת הַקֹּדֶשׁ, אֵין לְהוֹסִיף בְּמִנְיַן יְדִיעוֹת הֶעְלֵם שְׁנֵיהֶם יַחַד, כֵּיוָן שֶׁכְּבָר נִמְנוּ כָּל אַחַת בִּפְרָט:

פרק ב - משנה ב

לָעֲזָרָה. דְּבָעֲזָרָה נֶאֱמַר כָּרֵת לְטָמֵא הַנִּכְנָס, כִּדְאִיתָא בְּמִשְׁנָה ח' פֶּרֶק קַמָּא דְכֵלִים:
שֶׁעָלָיו נֶאֱמַר וַיְהִי בִישֻׁרוּן מֶלֶךְ. הָרַמְבַּ"ם: כְּלוֹמַר וְכֵיוָן שֶׁאָחִיו כֹּהֵן גָּדוֹל הָיָה אִם כֵּן הָיָה שָׁם אוּרִים וְתֻמִּים. וְהִקְשׁוּ הַתּוֹסָפוֹת אַף עַל גַּב דְּאַהֲרֹן הָיָה בְבִנְיַן הַמִּשְׁכָּן וְכִי נִשְׁאֲלוּ בְאוּרִים וְתֻמִּים לַעֲשׂוֹת מִשְׁכָּן, הֲלֹא קֹדֶם שֶׁנִּתְכַּהֵן אַהֲרֹן עַל יְדֵי הַמִּלּוּאִים נַעֲשָׂה הַמִּשְׁכָּן: רַשִׁ"י וְהָרַמְבַּ"ם. וּבְסַנְהֶדְרִין פֵּרֵשׁ הָרַ"ב מִשּׁוּם דְּמֹשֶׁה בִּמְקוֹם סַנְהֶדְרִין הֲוָה קָאֵי. וְטַעֲמָא דְמִסְתַּבֵּר הוּא, לְפִי שֶׁלֹּא נִמְצָא מִנּוּי שִׁבְעִים לַסַּנְהֶדְרִין כִּי אִם בְּפָרָשַׁת מִתְאוֹנְנִים, וְהָיָה זֶה אַחַר שֶׁנַּעֲשָׂה הַמִּשְׁכָּן כוּ'. וּלְפֵרוּשָׁם דְּהָכָא, צָרִיךְ לוֹמַר דְּשִׁבְעִים זְקֵנִים שֶׁנֶּאֶמְרוּ בְסוֹף פָּרָשַׁת מִשְׁפָּטִים הֵם הָיוּ מְיֻחָדִים שֶׁבָּעֵדָה לְכָל דָּבָר שֶׁבִּקְדֻשָּׁה, כְּעֵין סַנְהֶדְרִין: כְּלוֹמַר שֶׁאֵין טַעַם בַּדָּבָר לָמָּה זוֹ נֶאֱכֶלֶת וְלָמָּה זוֹ נִשְׂרֶפֶת, אֶלָּא עַל פִּי נָבִיא בְרוּחַ הַקֹּדֶשׁ עָשׂוּ כֵן. דְּלֵיכָּא לְמֵימַר דְּחִיצוֹנָה כִי אָתְיָא קִדְּשָׁה הַמָּקוֹם וְהַפְּנִימִית הִיא לֹא נִפְסֶלֶת בְּיוֹצֵא וְשֶׁלְּכָךְ נֶאֱכֶלֶת, דְּהָא בְּלֹא הָא לֹא סַגִּי: בַּגְּמָרָא, מַאי גְּדוֹלוֹת, אִילֵימָא כוּ', אֶלָּא גְדוֹלוֹת שֶׁבַּתּוֹדָה, וּמַאי נִיהוּ חָמֵץ. דְּחַלּוֹת הֶחָמֵץ הָיוּ גְדוֹלוֹת מִשֶּׁל מַצָּה (כְּדִתְנַן רֵישׁ פּ"ז דִּמְנָחוֹת):
פֵּרֵשׁ רַשִׁ"י, הוּא מִזְמוֹר שֶׁל חֲנֻכַּת הַבַּיִת, לְכָךְ אוֹמֵר אוֹתוֹ. וּבַגְּמָרָא מְסַיֵּם וְשִׁיר שֶׁל פְּגָעִים עַד כִּי אַתָּה ה' מַחְסִי עֶלְיוֹן שַׂמְתָּ מְעוֹנֶךָ, וְחוֹזֵר וְאוֹמֵר מִזְמוֹר לְדָוִד בְּבָרְחוֹ ה' מָה רַבּוּ צָרָי כוּ'. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: וּבֵית דִּין כוּ'. לְמֵימְרָא דְּבֵית דִּין קַמֵּי תוֹדָה אָזְלֵי, וְהַכְּתִיב (נְחֶמְיָה יב) וַיֵּלֵךְ אַחֲרֵיהֶם הוֹשַׁעְיָה וַחֲצִי שָׂרֵי יְהוּדָה. הָכִי קָאָמַר, בֵּית דִּין מְהַלְּכִין וּשְׁתֵּי תוֹדוֹת מְהַלְּכוֹת וּבֵית דִּין אַחֲרֵיהֶם. גְּמָרָא. כְּלוֹמַר דְּתָנֵי בֵית דִּין כוּ' וַהֲדַר מְפָרֵשׁ כֵּיצַד. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: הַפְּנִימִית. פְּלִיגֵי תַנָּאֵי בַגְּמָרָא, חַד אָמַר זוֹ כְנֶגֶד זוֹ הָיוּ מְהַלְּכוֹת, כְּלוֹמַר שְׁנֵי כֹהֲנִים נוֹשְׂאִים אוֹתָם זֶה אֵצֶל זֶה, וְהַוְיָא הַפְּנִימִית הַךְ דְּמִקָּרְבָא אֵצֶל חוֹמָה. וְחַד אָמַר זוֹ אַחַר זוֹ, וְהָוֵי הַפְּנִימִית הַךְ דְּמִקָּרְבָא לְבֵית דִּין: אֵין חַיָּבִין. הָכִי קָאָמַר, הַנִּכְנָס לְמָקוֹם פְּלוֹנִי וְהַנִּכְנָס לְמָקוֹם פְּלוֹנִי וְהַנִּכְנָס כוּ' כֻּלָּם אֵין חַיָּבִין עַל הַמָּקוֹם שֶׁנִּכְנָס כָּל אֶחָד וְאֶחָד:

פרק ב - משנה ג

בָּעֲזָרָה. בַּגְּמָרָא יָלֵיף לַהּ אַף לַנִּטְמָא בִּפְנִים. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: וְזוֹ כְרִיעָה בְעָלְמָא. גְּמָרָא: גְּמָרָא. וּפֵרֵשׁ רַשִׁ"י, מִחַיַּב מִשּׁוּם שְׁהִיָּה. וְהַדִּין עִמּוֹ כוּ'. וְהָא קָא מַשְׁמַע לָן דְּהִשְׁתַּחֲוָיָה דִכְלַפֵּי חוּץ בָּעֵי שְׁהִיָּה, וְאִי לֹא, לֹא. אֲבָל לֹא דְבָעֵי הִשְׁתַּחֲוָיָה עַל כָּל פָּנִים. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: לֹא בִמְהִירוּת וְלֹא בַאֲרִיכוּת. הָרַמְבַּ"ם: וְהִשְׁתַּחֲוָה כוּ'. שִׁעוּרִין הַלָּלוּ הֲלָכָה לְמֹשֶׁה מִסִּינַי הֵם בְּמִי שֶׁנִּטְמָא בְתוֹךְ הָעֲזָרָה. רַשִׁ"י:

פרק ב - משנה ד

שֶׁבַּנִּדָּה. דְּאָמַר קְרָא (וַיִּקְרָא טו) וּתְהִי נִדָּתָהּ עָלָיו, אֲפִלּוּ בִשְׁעַת נִדָּתָהּ תְּהֵא עָלָיו. גְּמָרָא. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב:

פרק ב - משנה ה

דְּאִלּוּ אַשְּׁאָר מְחֻיָּבֵי קָרְבָּנוֹת, מְחַיֵּב רַבִּי אֱלִיעֶזֶר אֲפִלּוּ לֹא נוֹדַע לוֹ בְּאֵיזֶה חֵטְא שֶׁחָטָא. כְּדִתְנַן בְּמִשְׁנָה ב' פֶּרֶק ד' דִּכְרֵתוֹת. גְּמָרָא: רַבִּי עֲקִיבָא אוֹמֵר כוּ'. וְאַיְדֵי דְבָעֵי לְמִכְתַּב בְּהֵמָה וְחַיָּה לִדְרָשָׁא כוּ', כָּתַב נַמִּי שֶׁרֶץ, כִּדְתָנָא דְבֵי רַבִּי יִשְׁמָעֵאל כָּל פָּרָשָׁה שֶׁנֶּאֶמְרָה וְנִשְׁנֵית לֹא נִשְׁנֵית אֶלָּא בִשְׁבִיל דָּבָר שֶׁנִּתְחַדֵּשׁ בָּהּ:
