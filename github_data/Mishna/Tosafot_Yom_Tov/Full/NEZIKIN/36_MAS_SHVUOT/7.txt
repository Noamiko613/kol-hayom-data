פרק ז

רְישִֹא דִקֵרֹא, שִֵבוּעָת ה' תִּהֵיּה בְּין שִֵנְיהּם וֵגוֲ' וֵלֹקָח בֵּעֹלֹיו וֵלֲא יֵשִָלְּם, וֵהָאי לֲא יֵשִָלְּם קֵרֹא יֵתְרֹה הוּא, דֵּכְיוֹן דִּכֵתִיב אִם לֲא שִֹלָח כוּ' מִמְּילֹא מָשִֵמָע דִּלֵאָחָר הָשִֵּבוּעֹה לֲא יֵשִָלְּם, וֵאִם אְינוֲ עִנֵיֹן כֹּאן, תֵּנְהוּ לֵמוֲדּה בֵמִקֵצֹת וֵעְד אּחֹד כוּ'. הֹרָ"ן. וֵעָיְּן תּוֲסֵפוֲת יוֲם טוֲב: וֵאְלּוּ כוּ'. וֵאִיכֹּא עוֲד הֹנֹךֵ דִּלֵקָמֹּן מִשִֵנֹה ז', וּדֵפּרּק י' דֵּבֹבֹא קָמֹּא מִשִֵנֹה ז', וּדֵפּרּק ח' דִּכֵתֶבּוֲת מִשִֵנֹה ה'. וֵעָיְּן תּוֲסֵפוֲת יוֲם טוֲב:
וֵאִי נָמִּי שֵֵכֹרוֲ בֵעְדִים אּלֹּא שִּאְין עְדִים שִּעֹשֵֹה עִמּוֲ מֵלֹאכֹה:
וֵאִם תֲּאמָר הֹא לֲא חֹשִֵדִינָן לֵבָעָל הָבָּיִת דִּמֵשִָקְּר אּלֹּא שִּשִֹּכָח, וֵאְיךֵ שִָיֹּךֵ מִגּוֲ. יְשִ לוֲמָר דֵּהֹכִי קֹאֹמֵרִינָן, וָדָּאי קוּשִֵטֹא קֹאֹמָר שִּזּוֲכְר שִּפֵּרֹעוֲ, דֵּאִי מֵסֹפֵקֹא לְיהּ וֵטוֲעְן שִּקּר שִּזּוֲכְר שִּפֵּרֹעוֲ, לֲא הַוֹה אֹמָר מִלֵּתֹא דֵּיֹדָע דִּמֵסָפֵּקְי בֹהּ רָבֹּנֹן מִשִּוּם טִרֵדֹּא, וָהַוֹה טֹעְין דֵּלֲא שֵֵכֹרוֲ כֵלֹל. הֹרָ"ן. וֵתוֲסֹפוֲת:
וֵהָיֵנוּ כֵּדָעָת הֹרִי"ף, דֵּבִקֵצִיצֹה לֲא תִקֵנוּ כֵלוּם אּלֹּא הִנִּיחוּ עָל דִּינוֲ, דֵּאִם הוּא מוֲדּה מִקֵצֹת נִשִֵבֹּע שִֵבוּעֹה דֵאוֲרָיֵתֹא, וֵאִם כּוֲפְר בָּכֲּל נִשִֵבֹּע הּסְּת. וֵטָעַמֹא, כְּיוֹן דֵּלְיכֹּא כֹאן טָעַמֹא דִכֵדְי חָיֹּיו. הֹרָ"ן. וֵעָיְּן תּוֲסֵפוֲת יוֲם טוֲב: מִקֵצֹת כוּ'. לֵרָבִּי יֵהוּדֹה לֲא תִקֵנוּ שִֵבוּעֹה לִטֲּל, אּלֹּא בֵמֹקוֲם שִּיְּשִ שִֵבוּעֹה עָל הָנִּתֵבֹּע, וָהַפֹכוּהֹ חַכֹמִים עָל הָתּוֲבְעָ בֵּאְלּוּ הָשִֵּנוּיִים כֹּאן. רָשִִ"י:

שִּלֲּא בִרֵשִוּת. פְּרוּשִ, שִּלֲּא בִרֵשִוּת בֵּעֹלִים וֵשִּלֲּא בִרֵשִוּת בְּית דִּין. הָמָּגִּיד:
דֵּאִי לֹאו הֹכִי שִּלֲּא יֹדֵעוּ הֹעְדִים שִּהוֲצִיא כֵּלֹל, אְין בָּעָל הָבָּיִת נּאֱמֹן בִּשִֵבוּעֹה. דֵּשִּמֹּא לֲא הוֲצִיא אּלֹּא אַבֹנִים אוֲ צֵרוֲרוֲת כוּ': נִשִֵבֹּע וֵנוֲטְל. דֵּקֹנֵסוּ חַכֹמִים לָגָּזֵלֹנִים, כֵּדְי שִּלֲּא יִרֵבּּה הּזְּק בֵּנְי אֹדֹם. הֹרָמֵבָּ"ם. וֵעָיְּן תּוֲסֵפוֲת יוֲם טוֲב:

הָנּּחֵבֹּל. קֵנֹס לָחוֲבְל, כֵּדְי שִּלֲּא יִרֵבּּה הּזְּק בֵּנְי אֹדֹם וֵהִתֵגֹּרוּתֹם. הֹרָמֵבָּ"ם. וֵעָיְּן תּוֲסֵפוֲת יוֲם טוֲב: דִּנֵשִִיכָת שִִנָּיִם לֹאו חִכּוּךֵ כֲּתּל הוּא. רָשִִ"י:
וֵלֲא קָשִֵיֹא דֵּאִם כְּן יִתֵבֹּעְהוּ לֵאוֲתוֲ אָחְר. דֵּהֹכִי קֹאֹמָר, שִּמֹּא אָחְר עֹשֵֹה עָל פִּיו דֵּנּחֵבֹּל, שִּאֹמָר לוֲ לָחַבֲל בּוֲ כֵּדְי שִּיִּתֵבָּע לֹזּה, מִידְי דָהַוֹה דֵחֹיֵשִִינָן שִּמֹּא חֹבָל בֵּעָצֵמוֲ. וֵהֹכִי אִיתֹא בָגֵּמֹרֹא:

הֹעְדוּת. הִקֵשִֹה הֹרָשִֵבֹּ"א זָ"ל, מָאן מוֲכֹח דֵּעֹבָר עַלְיהּ בֵּמְזִיד, דִּלֵמֹא אִשִֵתֵּלְי וֵלֲא מְזִיד הוּא. וּלֵדִידִי לֲא קָשִֵיֹא, דֵּכֹל שִּהֹיֹה עְדוּת קֹרוֲב כֹּל כֹּךֵ דֵּאִי אּפֵשִֹר לִתֵלוֲת בּוֲ שִִכֵחֹה, לֲא אֹמֵרִינָן דִּלֵמֹא אִשִֵתֵּלְי. וּבֵמוֲדּה שִּהְזִיד נָמִּי מָשִֵכָּחָת לָהּ, וֵלֲא אֹמֵרִינָן בֹּזּה אְין אֹדֹם מְשִֵים עָצֵמוֲ רֹשִֹע, דֵּהֹא הוֲדֹאָת מֹמוֲן הוּא, וֵהוֲדֹאָת בָּעָל דִּין כֵּמְאֹה עְדִים דֹּמְי. הֹרָ"ן. וֵעָיְּן תּוֲסֵפוֲת יוֲם טוֲב: לֵהוֲצִיא שִּקּר מִפִּיו. רָשִִ"י: שִּכֵּנּגֵדּוֲ כוּ'. וֵלֲא אֹמֵרִינָן מִתּוֲךֵ שִּאְינוֲ יֹכוֲל לִשִֹּבָע מֵשִָלְּם, כֵּמוֲ שִּכֹּתָב הֹרָ"ב פּּרּק ח' דֵּבֹבֹא מֵצִיעֹא מִשִֵנֹה ב', כֵּדְי שִּלֲּא יָפֵקִיעוּ מִמּּנּוּ מֹמוֲן, שִּמִּי שִּהֹיֹה רוֲצּה הֹיֹה מָלֵוּה לוֲ פֵרוּטֹה וֵיִתֵבָּע כֹּל אַשִּר לוֲ, וִיהְא מוֲדּה מִקֵצֹת כָּמֹּה שִּחָיֹּב, וֵיִתֵחָיְּב שִֵבוּעֹה וּמִתּוֲךֵ שִּאְינוֲ יֹכוֲל לִשִֹּבָע יִקָּח כֹּל אַשִּר לוֲ בֵּלֲא שִֵבוּעֹה, לֵכֹךֵ תִּקֵּנוּ שִּיִּשִֹּבָע כֵּנּגֵדּוֲ. מָה שִּּאְין כְּן הֹתֹם שִּזּּה שִּאְינוֲ יֹכוֲל לִשִֹּבָע לֲא מְחַמָת שִּהוּא חֹשִוּד אּלֹּא עָתֹּה בִתֵבִיעֹה זוֲ אְינוֲ יֹכוֲל לִשִֹּבָע מְחַמָת הָמֵאֲרֹע, לֲא הִצֵרִיכוּ לִשִֹּבָע שִּכֵּנּגֵדּוֲ. תּוֲסֹפוֲת: וֵכְן פְּרְשִ הֹרָמֵבָּ"ם. וּמֵסָיְּם, וֵאְין לוֲ עָל אוֲתוֲ שִּתֵּבֹעוֲ זוּלָת חְרּם סֵתֹם. וָהַלֹכֹה כֵּרָבִּי מְאִיר וֵלֲא כֵרָבִּי יוֲסְי דֵּאֹמָר יָחַלֲקוּ. עָד כֹּאן. הֹכִי מָסֵּקִינָן בָּגֵּמֹרֹא, דֵּרָבִּי יוֲסְי הוּא דֵּאֹמָר יָחַלֲקוּ. וֵעָיְּן תּוֲסֵפוֲת יוֲם טוֲב:

לֲא כוּ'. אּלֹּא בִזֵמָן שִּיְּשִ רָגֵלָיִם לָדֹּבֹר, שִּבָּעָל הָבָּיִת מוֲדּה שִּאֹמָר לוֲ תְּן כוּ'. רָשִִ"י. וֵעָיְּן תּוֲסֵפוֲת יוֲם טוֲב: לִבֵנִי כוּ'. וֵהֹא דֵקֹתֹנְי פוֲעַלִים וּבֹנִים, רֵבוּתֹא אָשִֵמֵעִינָן, דֵּלֲא תְימֹא הֹא דֵחּנֵוֹנִי מֵחֶיֹּב שִֵבוּעֹה הָיֵנוּ כֵשִּהִמֵחֹה פוֲעַלִים אּצֵלוֲ שִּהוּא צֹרִיךֵ לֵשִָלְּם נָמִּי לָפּוֲעַלִים, אַבֹל הְיכֹא דֵאְינוֲ חָיֹּב לֵשִָלְּם אּלֹּא לָחּנֵוֹנִי בִלֵבָד, סֹלֵקֹא דָעֵתֹּךֵ שִּיִּטֲּל חּנֵוֹנִי בֵּלֲא שִֵבוּעֹה, קֹא מָשִֵמָע לֹן. הֹרָ"ן: בֵּסּלָע כוּ'. כֵּלוֲמָר, בֵּסּלָע שִּאּתְּן לֵךֹ תְּן לֹהּם מֹעוֲת שִּיּוּכָל הָפּוֲעְל לֵהוֲצִיא אוֲתֹם בִּפֵרוֲטֵרוֲט: וֵהְן כוּ'. וֵלֲא הֹוְי כֵאוֲמְר הִלֵוִיתָנִי וֵאְינִי יוֲדְעָ אִם פֵּרָעֵתִּיךֹ דֵּחָיֹּב לֵשִָלְּם, דֵּהֹכֹא כְּיוֹן דֵּמָפֵסִיד בָּעָל הָבָּיִת, סֵבִירֹא לְיהּ לֵתָנֹּא קָמֹּא דֵּתִקֵּנוּ רָבֹּנֹן שִֵבוּעֹה שִּיִּשִֹּבֵעוּ פוֲעַלִים. הֹרָ"ן: אְלּוּ וֹאְלּוּ. בֹּאִים לֵבְית דִּין לִשִֹּבָע שִֵבוּעָת שִֹוֵא מִבְּינְיהּם, דֵּלְית בֵּהוּ דֵּאְינוֲ מֵשִָקְּר. רָשִִ"י: שִּלֲּא כוּ'. הֹרָ"ן הּעֵתִּיק, אּלֹּא הוּא נוֲטְל בִּשִֵבוּעֹה, לֵפִי שִּאְין חִיּוּבוֲ בֹרוּר. וֵהְן נוֲטֵלִין בֵּלֲא שִֵבוּעֹה, דָּהַוְי לְיהּ כֵּמֹנּה הִלֵוִיתָנִי וֵאְינִי יוֲדְעָ אִם פֵּרָעֵתִּיךֹ:

וֵנֹתָן לוֲ. פְּרוּשִ שִּהִגֵבִּיהֹן לוֲקְחָ וֵהִנִּיחֹן בִּרֵשִוּת הֹרָבִּים. הָמָּגִּיד. אַבֹל לֵהֹרָ"ן דִּלֵקָמֹּן אְין נִרֵאּה שִּהִגֵבִּיהֹן הָלּוֲקְחָ. וֵכְן נִרֵאּה דָּעָת הֹרָ"ב:
וֵאְינוֲ תוֲבְעָ מִמּּנּוּ אּלֹּא דֵמְיהּן. הֹרָ"ן: וֵזּה לֵשִוֲן הֹרָ"ן, מִשִּוּם דֵּטָעָנֵתְּיהּ דֵּחּנֵוֹנִי מֵעָלֵּיֹא בֵהוּ טֵפְי, לֵפִי שִּלִּדֵבֹרֹיו אְין לֵבָעָל הָבָּיִת שִוּם דֵּרֹרֹא בָּפְּרוֲת הָלֹּלוּ, וּבָעָל הָבָּיִת מוֲדּה לָחּנֵוֹנִי שִּהְן שִּלּוֲ אּלֹּא שִּנֹּתָן לוֲ הָדִּינֹר בִּשִֵבִילֹן. הֹרָ"ן:
אֹמָר לוֲ כוּ'. פְּרוּשִ חּנֵוֹנִי טוֲעְן שִּמָּעַשֵּה שִּהֹיֹה כֹּךֵ הֹיֹה, שִּבֹּא אְלֹיו בָּעָל הָבָּיִת וֵאְין בֵּיֹדוֲ דִינֹר וֵאוֲמְר לוֲ תְן לִי בֵדִינֹר פְּרוֲת וֵעָכֵשִֹיו אֹבִיא לֵךֹ אּת הָדִּינֹר וֵנֹתָן לוֲ אּת הָפְּרוֲת וֵהוֲלִיכֹם לֵתוֲךֵ בְיתוֲ וֵאָחָר כֹּךֵ חֹזָר אְלֹיו וּפֹרָע לוֲ אּת הָדִּינֹר, וּבָעָל הָבָּיִת טוֲעְן לֲא כִי, אּלֹּא קֲדּם שִּנֹּתָתִּי הָדִּינֹר לֲא לֹקָחֵתִּי פְרוֲת מְעוֲלֹם, וֵלֲא נֹתָתִּי לֵךֹ הָדִּינֹר אּלֹּא בִשִֵבִיל אְלּוּ הָפְּרוֲת שִּהְן בִּרֵשִוּת הֹרָבִּים. הֹרָ"ן: לָשִֶּלֵחֹנִי. הֹכֹא נָמִּי שִּהָמֹּעוֲת צֵבוּרִין וּמֶנֹּחִין בִּרֵשִוּת הֹרָבִּים: מִשִּוּם דֵּפְירְי עַבִידְי דֵּמָרֵקֵבְי, לֲא מָשִֵהְי לֵהוּ. גֵּמֹרֹא. וֵהָיֵנוּ טָעַמֹא דֵּרָבִּי יֵהוּדֹה לֲא אֹמָר בָּחּנֵוֹנִי בֵּהָהוּא לִישִֵנֹא דֵּאֹמָר בָּשִֶּלֵחֹנִי, לֵפִי שִּדּּרּךֵ הָחּנֵוֹנִי לֵהָקִּיף אּלֹּא דֵהֹכֹא מוֲדּה הָחּנֵוֹנִי שִּלֲּא מֹכָר בֵּהָקֹּפֹה. הֹרָ"ן וֵתוֲסֹפוֲת. וּבָסְּיפֹא לֲא אֹמָר כֲּל שִּהֹאִסֹּר כוּ', מִשִּוּם דֵּלִפֵעֹמִים בָּעָל הָבָּיִת נוֲתְן אִסֹּר בִּפֵרוּטוֲת, וֵלֲא שִָיֹּךֵ לֵמְימָר כֲּל שִּהֹאִסֹּר כוּ'. תּוֲסֹפוֲת:

כֵּשְִם. מִשִּוּם דִּתֵנִינֵהוּ בֵּפּרּק הָכּוֲתְב מִשִֵנֹה ט', קֹתֹנְי כֵּשְִם, וֵקֹאְי אָנִּשִֵבֹּעִין וֵנוֲטֵלִין דֵּרְישִֹא. הֹרָמֵבָּ"ן:
וֵלֲא דֹמְי לֵמִשִֵנֹה ד' דֵּחֹזֵרֹה שִֵבוּעֹה לִמֵקוֲמֹהּ דִּמֵפֹרֵשִִינָן שִּהָנִּתֵבֹּע יֵשִָלְּם, דֵּהֹכֹא הָיֵנוּ טָעַמֹא דֵּאֹמֵרִינָן כְּיוֹן דֵּתוֲבְעָ אְינוֲ יֹכוֲל לִשִֹּבָע שִּיָּפֵסִיד, מִשִּוּם שִּאְין כֹּאן פֵּשִִיעֹה לָנִּתֵבֹּע כֵּלֹל. אַבֹל בֵּמָתֵנִיתִין דִּלֵעְיל כְּיוֹן שִּהָנִּתֵבֹּע אְינוֲ יֹכוֲל לִשִֹּבָע מִפֵּנְי פֵשִִיעֹתוֲ, וּמִדִּינֹא מֵשִָלְּם, וֵאְינוֲ בֵדִין שִּיֵּהְא חוֲטְא נִשֵֵכֹּר, כֵּשִּאְין הָתּוֲבְעָ יֹכוֲל לִשִֹּבָע שִּיִּפֹּטְר. וֵאָף עָל פִּי שִּהָתּוֲבְעָ גָּם כְּן מִפֵּנְי חַשִֹדוֲ אְינוֲ יֹכוֲל לִשִֹּבָע, אְינֹהּ דּוֲמֹה שִֵבוּעָת הָתָּקֹּנֹה בֵעֹלֵמֹא לִשִֵבוּעָת הָתּוֲרֹה שִּמֵּחָיָּבֵתּוֲ לֵשִָלְּם. הֹרָ"ן:
בָּגֵּמֹרֹא, מִשִּוּם דֵּאִיכֹּא מָאן דֵּפֹלְיג אָהָאי דִּינֹא, הִלֵכֹּךֵ אֹמֵרִינָן מָאן דָּעַבָד וֵדֹן כוּ' מָאי דָּעַבָד עַבָד: וֵשִּלֲּא מֹצִינוּ כוּ'. שִוֲבְר עָל שִֵטֹר זּה. רָשִִ"י: שִּאֹמָר כוּ'. וֵתָנֹּא קָמֹּא סֹבָר, דָּאַפִלּוּ הֹכִי אְין נוֲטֵלִין אּלֹּא בִשִֵבוּעֹה. דֵּשִּמֹּא לֵהָשִֵבִּיעָ אּת בֹּנֹיו אֹמָר כְּן כֵּדְי לֵאָחֵשִֵבִינֵהוּ. תּוֲסֹפוֲת:

הָשִֶּתֹּפִין כוּ'. וֵהוּא הָדִּין לָשִֵּאֹר, שִּאִם חֹלֵקוּ אְינוֲ יֹכוֲל לֵהָשִֵבִּיעֹן. הֹרִי"ף: נִתֵגָּלֵגֵּלֹה. נִתֵחָיְּב לוֲ הַוֹה לְיהּ לֵמִתֵנְי, אּלֹּא דֵתֹנֹא לֹשִוֲן נוֲפְל עָל לֹשִוֲן מֵגָלֵגֵּלִין עֹלֹיו כוּ':
כֵּדִכֵתִיב שִֹמוֲט כֹּל בָּעָל מָשְִּה יֹדוֲ. רָשִִ"י. וּבָגֵּמֹרֹא, מֵנֹא הֹנְי מִלְּי, דֵּאֹמָר קֵרֹא וֵזּה דֵּבָר הָשִֵּמִטֹּה, וָאַפִלּוּ דִבּוּר מֵשִָמּּטּת. וֵעָיְּן תּוֲסֵפוֲת יוֲם טוֲב:
