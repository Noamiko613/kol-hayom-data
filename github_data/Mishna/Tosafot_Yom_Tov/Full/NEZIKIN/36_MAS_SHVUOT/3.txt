פרק ג - משנה א

כְּרַבִּי עֲקִיבָא בְּמִשְׁנָה ה': הֵיכָן כוּ'. וְלֹא, וַהֲרֵי נְמָלָה. בְּרִיָּה שָׁאנֵי. וַהֲרֵי הֶקְדֵּשׁ שֶׁהוּא חַיָּב מְעִילָה בְּשָׁוֶה פְרוּטָה וַאֲפִלּוּ בְּפָחוֹת מִכַּזַּיִת, וּמְשָׁנֵי הָא בָעִינַן שָׁוֶה פְרוּטָה שֶׁזֶּהוּ שִׁעוּרוֹ. וְהָא מְפָרֵשׁ דְּכֻלֵּי עָלְמָא מוֹדוּ דְמִחַיַּב בְּכָל שֶׁהוּא. מְפָרֵשׁ נַמִּי כִּבְרִיָּה דָמֵי. פֵּרֵשׁ רַשִׁ"י כִּבְרִיָּה בִשְׁאָר אִסּוּרִים דָּמֵי. בְּרִיָּה טַעֲמָא מַאי מִשּׁוּם דְּחָשִׁיבָה, מְפָרֵשׁ נַמִּי אִיהוּ אַחַשְּׁבֵיהּ דְּאַסְרֵיהּ עֲלֵיהּ. גְּמָרָא: בִּמְדַבֵּר. כְּלוֹמַר מְדַבֵּר וְאוֹסֵר. דְּהָא אַשְׁכְּחַן שְׁבוּעַת הָעֵדוּת, אֶלָּא דְהָתָם מְדַבֵּר וְחוֹטֵא הוּא. וְכֵן מַשְׁכַּחַת מְגַדֵּף כוּ'. גְּמָרָא. וְתוֹסָפוֹת: מְפָרֵשׁ בַּגְּמָרָא דְּיָלְפִינַן מִקְּרָא כוּ'. וְאִי בָעֵית אֵימָא סְבָרָא, דְּאָמַר לֵיהּ אֱינִישׁ לְחַבְרֵיהּ נִטְעֹם מִידֵי וְעָיְלֵי וְאָכְלוּ וְשָׁתוּ. וּבִלְשׁוֹנָם הָווּ קָרֵי לַאֲכִילָה טְעִימָה. וְאִצְטְרִיךְ סְבָרָא לִקְרָא, דְּמַהוּ דְתֵימָא בִּנְדָרִים הַלֵּךְ אַחַר לְשׁוֹן בְּנֵי אָדָם, קָא מַשְׁמַע לָן. הָרַ"ן בְּשֵׁם בַּעַל הֲלָכוֹת גְּדוֹלוֹת:

פרק ג - משנה ד

רַשִׁ"י. וּכְלוֹמַר וְהוֹאִיל וְאִי לָאו דְּרַחֲמָנָא אַסְרִינְהוּ הָיוּ רְאוּיִין לַאֲכִילָה, הִלְכָּךְ יֵשׁ כָּאן מְקוֹם חִיּוּב, וְכִדְמַסִּיק לְקַמָּן דְּהַיְנוּ בְכוֹלֵל כוּ'. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: קוֹנָם כוּ'. צָרִיךְ לְפָרֵשׁ בְּמִידֵי דְלָא מִשַּׁעְבַּד לָהּ: אָכַל. וְצָרִיךְ שֶׁיֵּדַע כֵּן בִּשְׁעַת הַשְּׁבוּעָה, דְּאִם לֹא כֵן הַוְיָא נִדְרֵי שְׁגָגָה, וְנִדְרֵי שְׁגָגוֹת מֻתָּרִים כְּדִתְנַן בְּפֶרֶק ג' דִּנְדָרִים מִשְׁנָה ב':

פרק ג - משנה ה

שֶׁאֶתֵּן. מַאי אֶתֵּן. אִילֵימָא צְדָקָה לְעָנִי, מֻשְׁבָּע וְעוֹמֵד מֵהַר סִינַי הוּא, שֶׁנֶּאֱמַר נָתוֹן תִּתֵּן, וּתְנַן לְקַמָּן נִשְׁבַּע לְקַיֵּם אֶת הַמִּצְוָה וְלֹא קִיֵּם, פָּטוּר. אֶלָּא מַתָּנָה לְעָשִׁיר. גְּמָרָא: וַאֲפִלּוּ הָכִי רְגִילוּת לְפָרֵשׁ בִּכְלָל וּפְרָט וּכְלָל. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב:

פרק ג - משנה ו

הַמִּצְוָה כוּ'. אָמַר רַחֲמָנָא לְהָרַע אוֹ לְהֵיטִיב, אָמְנָם הוּא מִחַיַּב בִּשְׁבוּעָה עַל דָּבָר שֶׁאִי אֶפְשָׁר בּוֹ שֶׁיֵּיטִיב בִּמְקוֹם לְהָרַע אוֹ יָרַע בִּמְקוֹם לְהֵיטִיב, וְהֵם דִּבְרֵי רְשׁוּת. הָרַמְבַּ"ם. וּכְלוֹמַר דְּאִלּוּ בִדְבַר מִצְוָה כְּשֶׁיִּשָּׁבַע לְהֵיטִיב וְלֶאֱכֹל דָּבָר אָסוּר, הוּא מֵרֵעַ לְנַפְשׁוֹ. וּכְשֶׁיִּשָּׁבַע לְהָרַע שֶׁלֹּא לֶאֱכֹל חָמֵץ בְּפֶסַח, הוּא מֵיטִיב לְנַפְשׁוֹ: שֶׁלֹּא כוּ' שֶׁאִם כוּ'. דְּאִם יִשָּׁבַע לְבַטֵּל מַה שֶּׁנִּשְׁבַּע עָלָיו מֵהַר סִינַי, לֵיכָּא קַל וָחֹמֶר לְחַיְּבֵיהּ כְּשֶׁלֹּא יְבַטֵּל, אֶלָּא יֵעָשֶׂה כְמֻשְׁבָּע כוּ', וּלְפִיכָךְ בָּזֶה אַף רַבִּי יְהוּדָה בֶּן בְּתֵירָא מוֹדֶה:

פרק ג - משנה ז

פֵּרוּשׁ דְּהָכִי מַשְׁמַע, שֶׁלֹּא אֹכַל שִׁעוּר אֲכִילָה מִכִּכָּר זוֹ. וְלֹא נָקַט כִּכָּר זוֹ אֶלָּא לְמִשְׁרֵי נַפְשֵׁיהּ בְּאַחֲרִינָא. כֶּסֶף מִשְׁנֶה בְּשֵׁם הָרִיטְבָ"א: וְכָתְבוּ הַתּוֹסָפוֹת. נִרְאֶה דְיֵשׁ סְפָרִים דְּלֹא גָרְסֵי שֶׁלֹּא אֹכֲלֶנָּה שְׁלִישִׁית, וַאֲפִלּוּ הָכִי שָׁמְעִינַן שֶׁאִם נִשְׁאַל כוּ', מִדְּקָתָנֵי אֵינוֹ חַיָּב אֶלָּא אַחַת וְלֹא קָתָנֵי חַיָּב אַחַת: זוֹ כוּ'. לִמְעוּטֵי אֹכַל וְלֹא אָכַל מִמַּלְקוֹת, שֶׁאֵין בּוֹ מַעֲשֶׂה. גְּמָרָא:

פרק ג - משנה ח

שֶׁל אֶבֶן כוּ'. אֲבָל שֶׁל זָהָב לֹא שְׁכִיחַ. הִלְכָּךְ לֹא אַפֵּיךְ נַמִּי לְמִתְנֵי עַל הָעַמּוּד שֶׁל זָהָב כִּדְמַפִּיךְ עַל הָאִישׁ וְגוֹ': בַּגְּמָרָא. וְכָתְבוּ הַתּוֹסָפוֹת, פֵּרוּשׁ, בִּשְׁבוּעָה. וְכָתַב הָרַ"ן דְּטַעֲמָא דְאֵין שְׁבוּעַת בִּטּוּי וְאֵין הַפֵּרוֹת נֶאֱסָרִין עָלָיו, וַאֲפִלּוּ הָכִי לָקֵי, מִשּׁוּם דְּהָא וַדַּאי לֹא נִתְכַּוֵּן זֶה לֶאֱסֹר הַפֵּרוֹת, שֶׁאִם הָיָה מִתְכַּוֵּן לְכָךְ הֲוָה לֵיהּ לֶאֱסֹר בְּלֹא שׁוּם תְּנַאי, שֶׁהֲרֵי עַכְשָׁיו גַּם כֵּן פֵּרוֹת אֲסוּרִים עָלָיו, וְלָמָּה תָלָה נִדְרוֹ בְּדָבָר שֶׁאֵינוֹ, אֶלָּא וַדַּאי לֹא נִתְכַּוֵּן אֶלָּא לְקַיֵּם שֶׁרָאָה אוֹתוֹ הַדָּבָר, וּמִשּׁוּם הָכִי לוֹקֶה מִשּׁוּם שְׁבוּעַת שָׁוְא. וְלִדְבָרָיו כָּל פֵּרוֹת לָאו דַּוְקָא, אֶלָּא הוּא הַדִּין חַד פֵּירָא, וְכֵן בְּמִשְׁנָה ב' פֶּרֶק ג' דִּנְדָרִים כָּתַב הָרַ"ב כִּכָּר זוֹ. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: בַּגְּמָרָא, בְּטָרוּף. וּפֵרֵשׁ רַשִׁ"י, שֶׁיֵּשׁ בּוֹ חֲרִיצִים כְּקוֹרַת בֵּית הַבַּד. כָּךְ רָאִיתִי נָחָשׁ מְנֻמָּר. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: הַיְנוּ דָּבָר שֶׁאִי אֶפְשָׁר. רַשִׁ"י. וּלְדִבְרֵיהֶם מַתְנִיתִין הָכִי קָאָמַר, שֶׁלֹּא נְעִידְךָ הַיְנוּ נִשְׁבַּע לְבַטֵּל כוּ'. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: זוֹ כוּ'. לִמְעוּטֵי אָכַלְתִּי וְלֹא אָכַלְתִּי דְּחַיָּב קָרְבָּן. וְרַבִּי עֲקִיבָא הִיא דְּרֵישׁ פִּרְקִין. גְּמָרָא:

פרק ג - משנה ט

שָׁוְא. תֵּימַהּ, דִּשְׁתֵּיהֶן שְׁבוּעוֹת בִּטּוּי. דִּשְׁבוּעָה שֶׁאֹכַל הַיְנוּ כַזַּיִת, כְּמוֹ שְׁבוּעָה שֶׁלֹּא אֹכַל. וְשֶׁלֹּא אֹכֲלֶנָּה הַיְנוּ שֶׁלֹּא אֹכֲלֶנָּה כֻלָּהּ. וְשֶׁמָּא יֵשׁ לְחַלֵּק בֵּין אֹכַל לְשֶׁלֹּא אֹכַל. תּוֹסָפוֹת: שֶׁהֲרֵי יָצְאתָה שְׁבוּעָה לַשָּׁוְא עַל דָּבָר שֶׁאִי אֶפְשָׁר לוֹ, וּמֵאוֹתָהּ הַשָּׁעָה הוּא עוֹבֵר, אֲפִלּוּ קִיֵּם שְׁבוּעָתוֹ. גְּמָרָא:

פרק ג - משנה י

מִדִּתְנַן לְקַמָּן אַחַת זוֹ וְאַחַת זוֹ כוּ'. וְיָלְפִינַן לַהּ מִדִּכְתִיב וְאָמְרָה הָאִשָּׁה אָמֵן אָמֵן. גְּמָרָא. וּפֵרֵשׁ הָרַ"ן, דְּאִי לֹא דָמְיָא כְמוֹצִיא שְׁבוּעָה אַמַּאי מִבָּדְקָא, אֲפִלּוּ מֻשְׁבָּע מִפִּי אֲחֵרִים אֵין כָּאן, שֶׁהֲרֵי לֹא אָמְרָה כְלוּם אַחַר שְׁבוּעָה. דְּלֹא דָמֵי לְמַשְׁבִּיעַ עֵדִים שֶׁכּוֹפְרִים אַחַר הַשְּׁבוּעָה וְאוֹמְרִים אֵין אָנוּ יוֹדְעִים כוּ', אֶלָּא וַדַּאי כְּמוֹצִיא שְׁבוּעָה מִפִּיו דָּמֵי:
