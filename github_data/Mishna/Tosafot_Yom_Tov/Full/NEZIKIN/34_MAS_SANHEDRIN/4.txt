פרק ד

הֹרָמֵבָּ"ם בֵּפְרוּשִוֲ. אַבֹל בֵּחִבּוּרוֲ כֹתָב אְין עְדְי מֹמוֲן צֵרִיכִין דֵּרִישִֹה וָחַקִירֹה. וֵכְן הוּא בָגֵּמֹרֹא: מָה בְּין כוּ'. אָף עָל גָּב דֵּשִֹוִין בִּדֵרִישִֹה וָחַקִירֹה מִדִּכֵתִיב מִשִֵפָּט אּחֹד כוּ', לֵעִנֵיָן מִלְּי דֵתֹלֵיֹא בִּזֵכוּת וֵחוֲבֹה אְין לֵהָשִֵווֲתֹם, דֵּלֲא שִָיֹּךֵ בֵּדִינְי מֹמוֲנוֲת. תּוֲסֹפוֲת: לִזֵכוּת. שִּנּּאֱמָר אִם לֲא שִֹכָב אִישִ אֲתֹךֵ כוּ' הִנֹּקִי, דִּתֵחִלָּת הָתֵּנָאי לִזֵכוּת. וֵכְן אוֲמֵרִים לֵכֹל הָנִּדּוֲן, אִם לֲא עֹבָרֵתֹּ אָל תִּירֹא. גֵּמֹרֹא: שִֵנָיִם כוּ'. כִּדֵיֹלֵפִינָן בֵּמָתֵנִיתִין בֵּסוֲף פּּרּק קָמֹּא לֲא תִהֵיּה אָחַרְי רָבִּים לֵרֹעֲת כוּ', וֵלְיכֹּא לֵאוֲקֵמָהּ בֵּדִינְי מֹמוֲנוֲת, כִּדֵאֹמָר לֵקָמֹּן לֲא תָטּּה מִשִֵפָּט אּבֵיֲנֵךֹ, אַבֹל אָתֹּה מָטּּה מִשִֵפָּט שִוֲר הָנִּסֵקֹל, וֵכֹל שִּכְּן שִֵאֹר דִּינְי מֹמוֲנוֲת. רָשִִ"י: לִזֵכוּת כוּ'. בָּגֵּמֹרֹא יֹלֵפִינָן לְיהּ מִקֵּרֹא: הָיּוֲשִֵבִין לִפֵנְי הָדָּיֹּנִין וֵלֲא נִתֵמָנּוּ בְית דִּין עָל כֹּךֵ. רָשִִ"י: אְין יֹכוֲל כוּ'. סֵבֹרֹא הִיא, דֵּבֹעִינָן וֵהִצִּילוּ. וּבָגֵּמֹרֹא מוֲקִים לָהּ בִּשִֵעָת מָשֵֹּא וּמָתֹּן הוּא דֵלֲא יָחַזֲר, דֵּמִתּוֲךֵ שִּאָתֹּה אוֲמְר לוֲ שִּלֲּא לָחַזֲר הוּא מֵחָזְּר אָחַרְי רֵאֹיוֲת לֵהָעַמִיד דֵּבֹרֹיו וֵשִּמֹּא יִמֵצֹא רֵאֹיוֲת נֵכוֲנוֲת. אַבֹל בִּשִֵעָת גֵּמָר דִּין שִּכֵּבֹר חֹזָר עָל כֹּל צֵדֹדִין וֵנִרֵאּה לוֲ שִּאְין דֵּבֹרֹיו נֵכוֲנִים, חוֲזְר וּמֵלָמְּד חוֲבֹה. רָשִִ"י: בָּיּוֲם. וּתֵחִלָּת דִּין בָּלָּיֵלֹה אַפִלּוּ בֵדִיעַבָד אְינוֲ דִין. וֵעָיְּן תּוֲסֵפוֲת יוֲם טוֲב: בָּלָּיֵלֹה. אִם רֹצוּ. רָשִִ"י: פְּרְשִ רָשִִ"י, מִשִֵפָּט נָחַלוֲת, דֵּהָיֵנוּ דִינְי מֹמוֲנוֲת, בָּיּוֲם: דִּכֵתִיב וֵשִֹפֵטוּ, וֵעִקָּר מִשִֵפֹּט הָיֵנוּ גֵּמָר דִּין. רָשִִ"י. וֵעָיְּן תּוֲסֵפוֲת יוֲם טוֲב: הֹכִי קֹאֹמָר, שִּאִם יִרֵאוּהוּ בֵעּרּב שִָבֹּת לֵחוֲבֹה בֹּעוּ לֵמּעֱבָד הַלֹנֹה וֵנִמֵצֹא גֵמָר דִּינוֲ בֵּשִָבֹּת, וֵיִצֵטֹרֵכוּ לֵהָשִֵהוֲתוֲ מִלָּהַמִיתוֲ עָד אָחָר שִָבֹּת, דֵּבוֲ בָיּוֲם אִי אּפֵשִֹר וֵכוּ'. וֵהָשִֵּהִיֹּה מִלָּהַמִיתוֲ אָחָר שִּנִּגֵמָר דִּינוֲ הוּא עִנּוּי הָדִּין. גֵּמֹרֹא. וֵעָיְּן תּוֲסֵפוֲת יוֲם טוֲב: וָדָּאי לֵגוּפְיהּ אִצֵטֵרִיךֵ, אִי לֵחָלְּק אִי לֵלֹאו, אּלֹּא מוֲשִֹבוֲת קֹשִּה לְיהּ, דֵּמִכֵּדְי חוֲבָת הָגּוּף הוּא כוּ'. אּלֹּא מָה לֵּהָלֹּן מָאי מוֲשִֹבוֲת בְּית דִּין, אָף מוֲשִֹבוֲת הֹאֹמוּר כֹּאן בְּית דִּין. גֵּמֹרֹא:

הָטֶּמֵאוֲת כוּ'. אִם בֹּא אֹדֹם לִשִֵאֲל עָל דֵּבָר טֹהֳרוֲת, מָתֵחִילִין מִן הָגֹּדוֲל, שִוֲאַלִין תֵּחִלֹּה לָגֹּדוֲל שִּבָּדָּיֹּנִין מֹה הוּא אוֲמְר בָּדֹּבֹר. רָשִִ"י. וֵיְשִ לִתֵמֲהָּ אָמָּאי לֲא תֹנְי נָמִּי הֹאֹסוּר וֵהָמֶּתֹּר, דֵּמִסֵּתֹמֹא נָמִּי דִינֹא הֹכִי. וֵכְן בָּבֵּרָיֵתֹא בָגֵּמֹרֹא וּבֹרָמֵבָּ"ם לְיתֹא. אָךֵ הֹרָמֵבָּ"ם לֵעִנֵיָן הוֲלֵכִין אָחָר הֹרֲב חֹשְִיב לֵהוּ נָמִּי: מָה שִּּאְין כְּן כֵּשִּאוֲמֵרִים הָקֵּטָנִּים תֵּחִלֹּה, אְינֹם נוֲטִים מִדִּבֵרְי הָמֶּפֵלֹא שִּאוֲמְר לֵאָחַרְיהּם, אּלֹּא שִּעוֲמֵדִים בֵּדִבֵרְיהּם. וֵכֹתֵבוּ הָתּוֲסֹפוֲת דֵּאּפֵשִֹר לֵפֹרְשִ דִּקֵרֹא בֵּדִינְי נֵפֹשִוֲת הוּא דִכֵתִיב. וֵעָיְּן תּוֲסֵפוֲת יוֲם טוֲב: דִּכֵתִיב שֵוֲם תֹּשִֵים עֹלּיךֹ מּלּךֵ מִקּּרּב אָחּיךֹ, כֹּל מֵשִֵימוֲת שִּאָתֹּה מְשִֵים עֹלּיךֹ לֲא יֵהוּ אּלֹּא מִקּּרּב אָחּיךֹ. וֵכְיוֹן דֵּאִמּוֲ מִיִּשֵֵרֹאְל מִקּּרּב אָחּיךֹ קֹרְינֹא בְיהּ. הֹרִי"ף. וֵכֹתֵבוּ הָתּוֲסֹפוֲת, דֵּכֹל שִּכְּן כֵּשִּאֹבִיו מִיִּשֵֵרֹאְל: הָיֵנוּ חוּץ מִירוּשִֹלָיִם. כֵּמוֲ שִּאִיתֹא פּּרּק ד' דֵּקִדּוּשִִין מִשִֵנֹה ה': הָמָּשִֵּיאִין כוּ'. הָמֵיֶחֹסִים וּרֵאוּיִים לֵהָשִֵּיא בֵנוֲתְיהּם לָכֲּהַנִים. רָשִִ"י:

סָנֵהּדֵרִין. אָף שִּל עּשֵֵרִים וּשִֵלשִֹה. רָשִִ"י: לֵפִי שִּאִם הֹיוּ יוֲשִֵבִין בֵּשִוּרֹה אְין פֵּנְי הֹרִאשִוֲנִים רוֲאִים זּה אּת זּה. רָשִִ"י: שִּבְּית הָמִּקֵדֹּשִ בֵּאּמֵצָע שִּל עוֲלֹם. רָשִִ"י. וֵשִֹרֵרְךֵ הוּא הָטָּבּוּר. וּכֵמוֲ שִּהָטָּבּוּר אְינוֲ מָמֹּשִ בֹּאּמֵצָע כְּן גָּם בְּית הָמִּקֵדֹּשִ אְינוֲ מָמֹּשִ בֹּאּמֵצָע, אַבֹל עָל צָד הָקְּרוּב הוּא: הָמֵזָכִּין כוּ'. פֵּלוֲנִי חִיְּב וּמִן הָטָּעָם הָזּּה, וּפֵלוֲנִי זִכֹּה וּמִן הָטָּעָם הָזּּה. רָשִִ"י. וֵטָעַמֹא, כֵּדְי שִּאִם יֲאמֵרוּ שִֵנָיִם טָעָם אּחֹד מִשִֵּנְי מִקֵרֹאוֲת לֲא יִמֵנוּ אּלֹּא אּחֹד (כִּדֵיֹלֵפִינָן לְיהּ מִקֵּרֹא). אִי נָמִּי נֹפֵקֹא מִנָּהּ לָמֵחָיֵּבִין, דֵּאִי לֵמֹחֹר יָחַזֲר מִטָּעֵמוֲ וִיחָיֵּבוֲ מִטָּעָם אָחְר, לִבֹּעְי עוֲד הַלֹנָת דִּין. גֵּמֹרֹא: הֹרָמֵבָּ"ם. וּתֵמְהָנִי דֵּשִָפִּיר קֹאֹמָר רָבִּי יֵהוּדֹה. וֵרָשִִ"י פְרְשִ בִּדֵתָנֹּא קָמֹּא שִּשִֵּנְיהּם כּוֲתֵבִין דִּבֵרְי הָמֵזָכִּין וֵגָם דִּבֵרְי הָמֵחָיֵּבִין. וֵרָבִּי יֵהוּדֹה חֹיְשִ שִּמֹּא מְרֲב הָטֲּרָח יִטֵעוּ:

כֵּלוֲמָר לָעַשֵוֲת שִֵתָּיִם שִּל עּשֵֵרִים וֵאָרֵבֹּעֹה. וּמֵסָיְּם רָשִִ"י וֵלֲא לָעַשֵוֲת שִוּרוֲת קֵטָנּוֲת, וֵלֲא שִֵתָּיִם שִּל עּשֵֵרִים וּשִֵלשִֹה וֵאָחָת שִּל שִֵנְי תָלֵמִידִים, הִלֵכֹּךֵ עַבֵדְי שִֹלשִ שִוּרוֲת: לִפֵנְיהּם. אָף הְן כָּעַגֶלֹּה. אּלֹּא שִּהָגֵּדוֲלִים יוֲשִֵבִין עָל הָסָּפֵסֹלִים וֵהָתָּלֵמִידִים עָל הָקָּרֵקָע. רָשִִ"י: וֵכְן אוֲתוֲ שִּנִּסֵמָךֵ מִן הֹרִאשִוֲנִים לֲא הֹיֹה יוֲשְִב בִּמֵקוֲמוֲ שִּל רִאשִוֲן, אּלֹּא כֶלֹּן בֵּסוֲף הָשִּוּרוֲת, וֵכֹל בֵּנְי הָשִּוּרֹה נִמֵשִֹכִין אִישִ אִישִ לֵמָעֵלֹה מִמֵּקוֲמוֲ וּמֵמָלֵּאִין זּה מֵקוֲמוֲ שִּל זּה. מִפֵּנְי שִּהָקֹּטֹן כוּ'. רָשִִ"י:

וּמֵאָיֵּמִין כוּ'. אָחַרְי שִּאֹמֵרוּ הֹעְדִים רֹאִינוּ זּה אוֲ פֵלוֲנִי שִּעֹבָר עַבְרֹה פֵלוֲנִית, אֹמֵרוּ לֹהּם מָכִּירִים אָתּּם אוֲתוֲ הִתֵרְיתּם בּוֲ, וֵאֹמֵרוּ הֹעְדִים הֹיֵתֹה לוֲ הָתֵרֹאֹה וּמָכִּירִין אֹנוּ אוֲתוֲ. הֹרָמֵבָּ"ם. וֵכְן נָמִּי בֵדִינְי מֹמוֲנוֲת כִּדֵמוּכֹח לֵשִוֲן הָמִּשִֵנֹה ו' פּּרּק ג'. וֵעָיְּן תּוֲסֵפוֲת יוֲם טוֲב: מְאֲמּד. לֹאו דִּבֵדִינְי מֹמוֲנוֲת אֹמֵדִינָן, דֵּהֹא תֹנֹא בֵמִשִֵנֹה ו' פּּרּק ג' אִם אֹמָר הוּא כוּ' לֲא אֹמָר כֵּלוּם. אּלֹּא אָף עָל גָּב דִּפֵסִילְי בֵדִינְי מֹמוֲנוֲת אֹמֵרִינָן לֵהוּ בֵּדִינְי נֵפֹשִוֲת. גֵּמֹרֹא: מִפִּי עְד. כֵּלוֲמָר שִֹמָעֵנוּ עְד מְעִיד בֵּבְית דִּין אָחְר: מִפִּי אֹדֹם נּאֱמֹן שִֹמָעֵנוּ. כֵּלוֲמָר מִפִּי אֹדֹם יֹדוּעָ. מָה שִּּאְין כְּן מִשִֵּמוּעֹה דֵּהָיֵנוּ שִֹמָעֵנוּ אוֲמֵרִים וֵלֲא יֹדָעֵנוּ מִי הוּא אוֲמְר, שִּזּּה אְינוֲ מֵבֲרֹר כֵּלֹל. וּבָגֵּמֹרֹא לֲא גֹרֵסִינָן שִֹמָעֵנוּ. וֵיְשִ לוֲמָר דֵּמִפִּי עְד וּמִפִּי אֹדֹם פְּרוּשִ דֵּמִשִֵּמוּעֹה הוּא: דֹּבֹר אָחְר כוּ'. אְין זּה מִן הֹאִיּוּם, אּלֹּא דֵמָתֵנִיתִין מֵפֹרְשִ לָקֵּרֹא: לֵפִיכֹךֵ כוּ'. כֹּל זּה אוֲמְר לֹהּם. רָשִִ"י: וּמִפֵּנְי כוּ'. הֹרָמֵבָּ"ם לֲא הּעֵתִּיקוֲ: וּלֵהָגִּיד. לָדּוֲרוֲת הָבֹּאִים. רָשִִ"י: בֵּחוֲתֹם. הָיֵנוּ אוֲתוֲ בָרֵזּל שִּהָצּוּרֹה חַקוּקֹה בוֲ. רָשִִ"י: אֹדֹם הֹרִאשִוֲן. חוֲתֹמוֲ שִּל אֹדֹם הֹרִאשִוֲן הוּא צוּרָת מִין הֹאֹדֹם אַשִּר בֹּהּ הֹאֹדֹם אֹדֹם, בּוֲ יִשִֵתָּתֵּפוּ כֹל הֹאֹדֹם. וֵאָף עָל פִּי שִּהָכֲּל מֵקָבֵּלִים אוֲתֹהּ צוּרֹה, יִשִֵתָּנּוּ אִישְִיהּם בִּמֵאוֲרֹעוֲת רָבִּים כֵּמוֲ שִּהוּא מֹצוּי וֵנִרֵאּה. הֹרָמֵבָּ"ם: בִּשִֵבִילִי כוּ'. כֵּלוֲמָר חֹשִוּב אַנִי כֵּעוֲלֹם מֹלְא, לֲא אּטֵרֲד אּת עָצֵמִי מִן הֹעוֲלֹם בָּעַבְרֹה אָחָת, וֵיִמֹּשְִךֵ מִמּּנֹּה. רָשִִ"י: כֵּלוֲמָר אָף עָל פִּי שִּאֹנוּ יוֲדֵעִין הֹאֱמּת, לְב אֹדֹם נוֲקְף עֹלֹיו שִּמֹּא לֲא כִוְּן אּל הֹאֱמּת, וֵיְשִ לוֲ לִדֵאֲג עָל כֹּל הֹאִיּוּמִים הָלֹּלוּ:
