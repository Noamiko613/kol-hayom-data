פרק ט

הָרוֹצֵחַ כוּ'. יָלְפִינַן לְהוּ בַגְּמָרָא דִּבְסַיִף. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: וְכָבַשׁ כוּ' חַיָּב. דְּאָמַר קְרָא אוֹ בְאֵיבָה, לְרַבּוֹת אֶת הַמְצַמְצֵם, הַיְנוּ כוֹבֵשׁ שֶׁצִּמְצְמוֹ שָׁם שֶׁלֹּא יָקוּם. גְּמָרָא. וּמִלַּת כָּבַשׁ עִבְרִית, כְּמוֹ הַגּוֹיִם אֲשֶׁר כִּבֵּשׁ: אַף עַל פִּי שֶׁיּוֹדֵעַ שֶׁסּוֹפוֹ לְהָקִיא, מִיהוּ לָאו מִכֹּחוֹ מָיֵית. רַשִׁ"י: בְּאֶבֶן כוּ'. הָכָא דִפְלִיגֵי בְּאִם יָקוּם וְהִתְהַלֵּךְ כוּ' דִּכְתִיב בְּפָרָשַׁת מִשְׁפָּטִים, וּבְאוֹתָהּ פָּרָשָׁה כְתִיב בְּאֶבֶן אוֹ בְאֶגְרֹף. וְרֵישָׁא נָסֵיב אֶבֶן אוֹ בַרְזֶל, כְּדִכְתִיב בְּפָרָשַׁת רוֹצֵחַ בְּפָרָשַׁת מַסְעֵי:

נִתְכַּוֵּן כוּ'. הָיָה עוֹמֵד אָדָם אֵצֶל בְּהֵמָה וְנִתְכַּוֵּן לַהֲרֹג בְּהֵמָה וְהָרַג אָדָם, פָּטוּר, מִשּׁוּם דְּכִי אַתְרוּ בֵיהּ הַתְרָאַת סָפֵק הוּא, אַף עַל פִּי שֶׁקִּבֵּל עָלָיו הַתְרָאָה שֶׁאָמַר יוֹדֵעַ אֲנִי שֶׁהוּא בֶן בְּרִית וְעַל מְנָת כֵּן אֲנִי עוֹשֶׂה שֶׁאִם אֶהֶרְגֶנּוּ אֲנִי חַיָּב מִיתָה, פָּטוּר, דְּשֶׁמָּא לֹא יַהַרְגֶנּוּ. רַשִׁ"י: לִנְפָלִים. אַנְּפָלִים לֹא מִחַיַּב, דִּכְמַאן דִּקְטִיל דָּמֵי. רַשִׁ"י: דְּאִם לֹא נִתְכַּוֵּן לְמַכַּת מִיתָה, הַתְרָאַת סָפֵק הִיא, דְּהִתְרוּ בוֹ אַל תִּזְרֹק שֶׁמָּא תֵלֵךְ עַל לִבּוֹ. רַשִׁ"י: וְהִתְרוּ בוֹ עָלָיו, דְּהַתְרָאַת וַדַּאי הוּא, וְהָרַג אֶת זֶה, חַיָּב, דְּאִיכָּא הַתְרָאָה עַל בֶּן בְּרִית וְהָוֵי מִתְכַּוֵּן עַל בֶּן בְּרִית. וְרַבִּי שִׁמְעוֹן אוֹמֵר כוּ'. וְהַאי דְּשָׁבֵיק רַבִּי שִׁמְעוֹן לְסֵיפָא בְהֶדְיָא וְהַדַר אַדִּיּוּקָא דְרֵישָׁא, דְּלֹא תֵימָא דְּרַבִּי שִׁמְעוֹן פּוֹטֵר אַכֻּלְּהוּ וַאֲפִלּוּ נִתְכַּוֵּן עַל מָתְנָיו וְהָלְכָה לוֹ עַל לִבּוֹ. רַשִׁ"י:

שֶׁלֹּא לְהַטְרִיחַ בֵּית דִּין. רַשִׁ"י: לְבַת כֹּהֵן. וְרַבִּי שִׁמְעוֹן לְטַעֲמֵיהּ דִּסְבִירָא לֵיהּ דַּאֲרוּסָה גַּם כֵּן בִּשְׂרֵפָה, וּמִדְּאַפְּקַהּ מֵאֲרוּסָה בַת יִשְׂרָאֵל שֶׁבִּסְקִילָה שְׁמַע מִנַּהּ כוּ'. גְּמָרָא: לַמְגַדֵּף כוּ'. שֶׁפּוֹשְׁטִין יָדָן בָּעִקָּר. וּבַת כֹּהֵן לֹא יָצָאת אֶלָּא נְשׂוּאָה, וּמִכְּלַל בַּת יִשְׂרָאֵל נְשׂוּאָה שֶׁהִיא בְחֶנֶק. רַשִׁ"י: רַבִּי שִׁמְעוֹן לְטַעֲמֵיהּ דִּסְבִירָא לֵיהּ מַדִּיחֵי עִיר הַנִּדַּחַת בְּחֶנֶק, דְּדָנִין הַדָּחָה הַדָּחָה מִנָּבִיא, וּמָצִינוּ לְאַנְשֵׁי עִיר הַנִּדַּחַת עַצְמָן שֶׁהֵן בְּסַיִף, אָמַרְתְּ וְכִי אֵיזֶה כֹחַ מְרֻבֶּה, כְּלוֹמַר אֵיזֶה עָוֹן חָמוּר, כֹּחַ הַמַּדִּיחַ אוֹ כֹּחַ הַנִּדָּח, הֲוֵי אוֹמֵר כֹּחַ הַמַּדִּיחַ, שְׁמַע מִנַּהּ דְּחֶנֶק חָמוּר מִסַּיִף. גְּמָרָא: שֶׁכֵּן אַנְשֵׁי עִיר הַנִּדַּחַת שֶׁפּוֹשְׁטִים יָדָם בָּעִקָּר נִדּוֹנִין בְּסַיִף. וּמִמַּדִּיחִים אֵין לְהָשִׁיב, דְּרַבָּנָן סְבִירָא לְהוּ דְּמַדִּיחֵי עִיר הַנִּדַּחַת בִּסְקִילָה כוּ' גְּמָרָא. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב:

בַּגְּמָרָא יָלְפִינַן לַהּ: הָכִי קָאָמַר, שֶׁנִּמְצָא עַכְשָׁיו שֶׁהָיְתָה תְחִלָּה כוּ': וְאַף עַל גַּב דִּבְאִסּוּר מוֹסִיף מוֹדֶה רַבִּי יוֹסֵי כְּדִתְנַן בְּמִשְׁנָה ח' פֶּרֶק ג' דִּכְרֵתוֹת, הָכָא בַחֲמוֹתוֹ וְנַעֲשֵׂית אֵשֶׁת אִישׁ אַף עַל גַּב דְּמוֹסִיף אִסּוּרָא לְעָלְמָא, שֶׁהָיְתָה מֻתֶּרֶת לָהֶם קֹדֶם שֶׁנַּעֲשֵׂית אֵשֶׁת אִישׁ, וּבְמִגּוֹ תֵּאָסֵר נַמִּי עָלָיו בְּאֵשֶׁת אִישׁ, תְּרֵי קְטָלֵי לֹא מָצֵית קָטֵיל לֵיהּ וְנִדּוֹן בַּחֲמוּרָה שֶׁהִיא בַחֲמוֹתוֹ. גְּמָרָא:

דְּגַבְרָא בַר קְטָלָא הוּא בִידֵי שָׁמַיִם, וְקָרוֹבֵי הוּא דְּלֹא מִקָּרַב קְטָלֵיהּ. וְכֵיוָן דְּקָא מְוַתֵּר לֵיהּ לְנַפְשֵׁיהּ לַעֲבֵרוֹת שֶׁל כָּרֵת, מְקָרְבִינַן לֵיהּ לִקְטָלֵיהּ עֲלוֹהִי. וּרְמִיזָא בַקְּרָא, דִּכְתִיב תְּמוֹתֵת רָשָׁע רָעָה. פֵּרֵשׁ רַשִׁ"י מִי שֶׁהֻחְזַק רָשָׁע תְּמִיתָתוֹ רָעָתוֹ, אַלְמָא מֻחְזָק רָשָׁע בַּר מִיתָה הוּא. וּרְמִיזָה הוּא דַהֲוֵי, וְעִקָּרָא דְמִלְּתָא דְּכִפָּה הֲלָכָה לְמֹשֶׁה מִסִּינַי הִיא: גְּמָרָא. אֲבָל שֶׁל שְׁתַּיִם וְשֶׁל שָׁלֹשׁ כְּרֵתוֹת אִסּוּרֵי הוּא דְּקָא טָעֵים. כְּלוֹמַר אֵינוֹ מְוַתֵּר עַצְמוֹ לְכָרֵת אֶלָּא שֶׁרוֹצֶה לִטְעֹם כָּל הָאִסּוּרִים: דְּכֵיוָן דְּשָׁנָה הֻחְזַק דְּשׁוּב לֹא יִתְיַסֵּר, וּכְרַבִּי אָתְיָא דְּקִים לֵיהּ דְּבִתְרֵי זִמְנֵי הָוֵי חֲזָקָה: וְאֵין לוֹ שָׁם מָקוֹם לְהִתְפַּשֵּׁט וְלֹא לִישַׁן. הָרַמְבַּ"ם. וְכִפָּה מִלָּה אֲרַמִּית. וְנָתַתָּה אֹתוֹ אֶל הַמַּהְפֶּכֶת מְתַרְגְּמִינַן כִּפָּתָא: הַהוֹרֵג כוּ'. וְאֵין עוֹשִׂין זֶה לִשְׁאָר מְחֻיָּבֵי מִיתוֹת בֵּית דִּין, אֶלָּא אִם אֵינוֹ חַיָּב מִיתָה פּוֹטְרִין אוֹתוֹ. שֶׁאַף עַל פִּי שֶׁיֵּשׁ עֲוֹנוֹת חֲמוּרִים מִשְּׁפִיכוּת דָּמִים, אֵין בָּהֶן הַשְׁחָתַת יִשּׁוּבוֹ שֶׁל עוֹלָם כִּשְׁפִיכוּת דָּמִים, אֲפִלּוּ עֲבוֹדָה זָרָה וַעֲרָיוֹת וְשַׁבָּת אֵינָן כִּשְׁפִיכוּת דָּמִים, שֶׁאֵלּוּ הֵן בֵּין אָדָם לַמָּקוֹם, אֲבָל שְׁפִיכוּת דָּמִים מֵעֲבֵרוֹת שֶׁבֵּין אָדָם לַחֲבֵרוֹ, וְכָל מִי שֶׁיֵּשׁ בְּיָדוֹ עָוֹן זֶה הֲרֵי הוּא רָשָׁע גָּמוּר, וְאֵין כָּל מִצְוֹת שֶׁעָשָׂה כָּל יָמָיו שְׁקוּלִים כְּנֶגֶד עָוֹן זֶה, וְלֹא יַצִּילוּהוּ מִן הַדִּין כוּ'. הָרַמְבַּ"ם:

כְּלוֹמַר שֶׁהַקַּסְוָה הוּא שֵׁם מֻשְׁאָל: קַנָּאִין כוּ'. גּוֹנֵב קַסְוָה, רָמֵיז לֵיהּ קְרָא, וְלֹא יָבֹאוּ לִרְאוֹת כְּבַלַּע אֶת הַקֹּדֶשׁ וָמֵתוּ . וּפֵרֵשׁ רַשִׁ"י כְּבַלַּע לְשׁוֹן גַּנָּב שֶׁמַּבְלִיעִין וּמַחְבִּיאִין. וְרֶמֶז בְּעָלְמָא הוּא כוּ'. וְהַמְקַלֵּל בַּקּוֹסֵם, דְּמֵחֲזֵי כִמְבָרֵךְ אֶת הַשֵּׁם. גְּמָרָא. וּבוֹעֵל אֲרָמִית, דְּהָא אַשְׁכְּחַן דְּבַר כָּרֵת הוּא יַכְרֵת ה' לָאִישׁ: כְּמוֹ לֵךְ פְּגַע בּוֹ (מְלָכִים א ב). רַשִׁ"י: אִם הוּא כֹהֵן. תּוֹסָפוֹת: לְבֵית דִּין. כְּלוֹמַר לְהַלְקוֹת בְּדִין לָאו שֶׁיֵּשׁ בּוֹ חִיּוּב מִיתָה בִידֵי שָׁמַיִם, אֶלָּא מוֹצִיאִין כוּ', וְאֵין מְמַחִין עֲלֵיהֶם. הָרַמְבַּ"ם: חוּץ כוּ'. דִּבְהַר הַבַּיִת מֻתָּרִים טְמֵאֵי מֵתִים וּמֵת עַצְמוֹ: דִּבְמָאתַיִם וַחֲמִשִּׁים אִישׁ (בַּשְּׂרוּפִים דְּקֹרַח) כְּתִיב. רַשִׁ"י. וְרַבִּי עֲקִיבָא סָבַר אֵין דָּנִין יוּמָת מִיָּמוּת כוּ' גְּמָרָא: וְאִם תֹּאמַר וְתֵיפּוּק לֵיהּ שֶׁנִּכְנַס לַמִּקְדָּשׁ דִּבְכָרֵת. וְיֵשׁ לוֹמַר שֶׁנִּטְמָא בִּפְנִים וְלֹא שָׁהָה כַשִּׁעוּר אֶלָּא בָא בִקְצָרָה וְהָפַךְ בְּצִנּוֹרָא. תּוֹסָפוֹת. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב אַמַּאי לֹא חָשֵׁיב נַמִּי פְּרוּמֵי בְגָדִים:
