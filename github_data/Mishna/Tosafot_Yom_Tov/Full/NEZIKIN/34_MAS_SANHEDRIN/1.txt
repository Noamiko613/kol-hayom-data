פרק א - משנה א

כְּלוֹמַר דֵּאלֹהִים דִּכְתִיב גַּבֵּי שׁוֹמְרִים הָוֵי כְאִלּוּ כְתוּבִים גַּם כֵּן בְּפָרָשַׁת אִם כֶּסֶף תַּלְוֶה, דְּעֵרוּב פָּרָשִׁיּוֹת כָּתוּב כָּאן כִּדְאִיתָא בַגְּמָרָא: שֶׁזֶּה טוֹעֲנוֹ שֶׁשָּׁלַח יָד בְּפִקְדוֹנוֹ, דְּהַיְנוּ גְזֵלוֹת. אֲבָל לָוָה וְלֹא שִׁלֵּם לֹא מִקְּרֵי גַזְלָן, דְּמִלְוָה לְהוֹצָאָה נִתְּנָה. וֵאלֹהִים מַשְׁמַע לְשׁוֹן שְׂרָרָה וְרַבָּנוּת. כְּדִמְתַרְגְּמִינָן רְאֵה נְתַתִּיךָ אֱלֹהִים לְפַרְעֹה (שְׁמוֹת ז) לְשׁוֹן רַבָּנוּת. רַשִׁ"י: וַחֲבָלוֹת. נִרְאֶה לְרַבֵּנוּ תָם דְּכֹפֶר וּשְׁלשִׁים דְּעֶבֶד וְעֶשְׂרִים וְאַרְבָּעָה רָאשֵׁי אֵבָרִים דְּעֶבֶד, בִּכְלַל חֲבָלוֹת. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: דְּשׁוֹר הַמּוּעָד דְּאָמְרָן דְּלָאו דְּשֵׁן וָרֶגֶל אֶלָּא דְּקֶרֶן דְּלֹא מַגְבִּין בְּבָבֶל שֶׁאֵין מֻמְחִין. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב:

פרק א - משנה ב

דְּאֵין צָרִיךְ לְעַבֵּר הַחֹדֶשׁ אֶלָּא כִּי מִקְדַּשׁ יוֹם שְׁלֹשִׁים וְאֶחָד מִמֵּילָא מִעַבַּר הַחֹדֶשׁ שֶׁעָבַר. דְּיוֹם שְׁלשִׁים בָּתְרֵיהּ שָׁדֵית לֵיהּ. רַשִׁ"י. אֲבָל בְּעִבּוּר שָׁנָה לֹא סַגִּי בְלֹא אֲמִירָה, דְּצָרִיךְ שֶׁיֹּאמְרוּ שֶׁתְּהֵא הַשָּׁנָה מְעֻבֶּרֶת. תּוֹסָפוֹת, וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: הַתְּקוּפָה, בֵּין עַל תְּקוּפַת תַּמּוּז שֶׁמִּתְעַכֶּבֶת וְנִמְשֶׁכֶת, שֶׁאֵין תְּקוּפַת תִּשְׁרֵי נוֹפֶלֶת עַד עֲבֹר הֶחָג, מְעַבְּרִין, דְּנָפְקָא לָן מִקְּרָא וְחַג הָאָסִיף תְּקוּפַת הַשָּׁנָה. בֵּין עַל תְּקוּפַת טֵבֵת שֶׁנִּמְשָׁךְ זְמַנָּהּ עַד טז בְּנִיסָן שֶׁהֲבָאַת עֹמֶר דִּילַהּ בְּתוֹךְ תְּקוּפַת טֵבֵת, מְעַבְּרִין. וְעַל הָאָבִיב, אִם לֹא בָשְׁלָה הַתְּבוּאָה מְעַבְּרִין, דִּקְרָא כְתִיב בְּחֹדֶשׁ הָאָבִיב. וְעַל פֵּרוֹת הָאִילָן מִתְאַחֵר בִּשּׁוּלָם יוֹתֵר מִזְּמַן הָעֲצֶרֶת מְעַבְּרִין, שֶׁעֲצֶרֶת זְמַן הַבִּכּוּרִים כוּ'. רַשִׁ"י: וְלֹא אִתְפָּרֵשׁ טַעֲמָא מַאי. וְאֶפְשָׁר לִי לוֹמַר כְּדֵי לְעוֹרֵר הַבְּרָכוֹת לְהַשְׁלִים הַמְכֻוָּן שֶׁמְּעַבְּרִין לִהְיוֹת הָאָבִיב וְהַפֵּרוֹת וְהָאָסִיף בִּזְמַנָּן. וּמָצִינוּ כַיּוֹצֵא בָזֶה שֶׁמְּעוֹרְרִין זְכוּת הָאָבוֹת בְּהַזְכָּרַת שְׁאֵלָה עַד שֶׁבְּחֶבְרוֹן:

פרק א - משנה ג

וְיָהִיב לֵיהּ רְשׁוּתָא . גְּמָרָא: שֶׁאִם לֹא תֹאמַר כֵּן אִי אֶפְשָׁר שֶׁתִּמְצָא בֵית דִּין הַגָּדוֹל לְעוֹלָם, לְפִי שֶׁנִּצְטָרֵךְ שֶׁיִּהְיֶה כָל אֶחָד מֵהֶם סָמוּךְ, וְהַקָּדוֹשׁ בָּרוּךְ הוּא יִעֵד שֶׁיָּשׁוּבוּ, כְּמוֹ שֶׁכָּתוּב וְאָשִׁיבָה שֹׁפְטַיִךְ כְּבָרִאשֹׁנָה וְגוֹ' כוּ'. עַד כָּאן לְשׁוֹן הָרַמְבַּ"ם בְּפֵרוּשׁוֹ. וּבְחִבּוּרוֹ סִיֵּם, וְהַדָּבָר צָרִיךְ הֶכְרֵעַ. וְנִרְאֶה לִי, לְפִי שֶׁאֶפְשָׁר לִדְחוֹת הָרְאָיָה, שֶׁהֲרֵי יִעֵד הַשֵּׁם יִתְבָּרַךְ גַּם כֵּן הִנֵּה אָנֹכִי שֹׁלֵחַ לָכֶם אֵת אֵלִיָּה הַנָּבִיא כוּ', וְהוּא סָמוּךְ בְּלִי סָפֵק וְיָכוֹל לִסְמֹךְ בְּצֵרוּף שְׁנַיִם. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: וַעֲרִיפַת. עֲרִיפָה לָאו דַּוְקָא, דִּמְדִידָה הוּא דְהַוְיָא בִשְׁלֹשָׁה. תּוֹסָפוֹת: תֵּימַהּ, שֶׁזֶּה הַכָּתוּב בִּסְמִיכַת זְקֵנִים לְפַר הֶעְלֵם דָּבָר כָּתוּב, אֲבָל בְּעֶגְלָה נָפְקָא לָן מִדִּכְתִיב זְקֵנֶיךָ. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: וְהַמֵּאוּנִין בִּשְׁלשָׁה. דְּכָל דְּתַקּוּן רַבָּנָן כְּעֵין דְּאוֹרַיְתָא תַקּוּן. וּמֵאוּנִין דְּרַבָּנָן, דְּמִדְּאוֹרַיְתָא אֲפִלּוּ מֵאוּן לֹא בַעְיָא, דְּאֵין קִדּוּשֵׁי קְטַנָּה כְלוּם אֶלָּא אִם כֵּן קִדְּשָׁהּ אָבִיהָ. דִּכְתִיב אֶת בִּתִּי נָתַתִּי לָאִישׁ. רַשִׁ"י: בִּשְׁלשָׁה. דְּמָמוֹנָא הוּא: סְבָרָא הוּא, מַה לִּי עַיּוֹלֵי מַה לִּי אַפּוֹקֵי, אַפּוֹקֵי מַאי טַעֲמָא, דִּלְמָא טָעֵי, עַיּוֹלֵי נַמִּי דִּלְמָא טָעֵי. גְּמָרָא: וְהוּא הַדִּין נַמִּי דְּפָלֵיג אַהֶקְדֵּשׁוֹת, דִּבְהוּ נַמִּי כֹּהֵן כְּתִיב: וּבְפֶרֶק ד' דִּמְגִלָּה מִשְׁנָה ג' פֵּרֵשׁ הָרַ"ב, אִם בָּא לִפְדּוֹתָן. וְכֵן פֵּרֵשׁ רַשִׁ"י. וְהָכִי מִסְתַּבְּרָא:

פרק א - משנה ד

תַּהֲרֹגוּ. בְּנִרְבָּע כְּתִיב. רַשִׁ"י. וּכְלוֹמַר דְּאִתַּקַּשׁ לְאִישׁ: דְּלֵיכָּא לְמֵימַר דִּקְרָא כִפְשָׁטֵיהּ לְחַיֵּב מִיתָה לִבְעָלָיו, דְּהָא אָמַר קְרָא מוֹת יוּמַת הַמַּכֶּה רֹצֵחַ הוּא, עַל רְצִיחָתוֹ אַתָּה הוֹרְגוֹ וְאִי אַתָּה הוֹרְגוֹ עַל רְצִיחַת שׁוֹרוֹ. גְּמָרָא: הַזְּאֵב כוּ'. שֶׁהֵמִיתוּ אָדָם וְנִדּוֹנִין בִּסְקִילָה כְּשׁוֹר. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: זָכָה. לַשָּׁמַיִם. (צְדָקָה עָשָׂה. רַשִׁ"י). דְּלֵיכָּא לְמֵימַר לְעוֹרָן, דְּכֵיוָן שֶׁהֵמִיתוּ אָדָם שַׁוִּינְהוּ רַבָּנָן (דְּלֹא הִצְרִיכוּ לְדוּנָן בְּבֵית דִּין) כְּמַאן דְּנִגְמַר דִּינַיְהוּ, וְאִסּוּרֵי הֲנָאָה נִינְהוּ. גְּמָרָא:

פרק א - משנה ה

פֵּרוּשׁ, עֲלִיָּתוֹ הוּא דַּהֲוֵי בְשִׁבְעִים וְאֶחָד, אֲבָל מִיתָתוֹ בְּעֶשְׂרִים וּשְׁלשָׁה כִּשְׁאָר דִּינֵי נְפָשׁוֹת, כִּדְאִיתָא בַגְּמָרָא. וּפֵרֵשׁ רַשִׁ"י דְּהָא לֹא תָנֵי לֵיהּ בְּמַתְנִיתִין. וּמִשּׁוּם דִּגְמָרוֹ לֹא הָיָה בְשִׁבְעִים וְאֶחָד, לְהָכִי לֹא תָנֵי הַמְרָאָתוֹ וְכֵן מִיתָתוֹ, וְאַף עַל גַּב דְּדַוְקָא בְּשִׁבְעִים וְאֶחָד. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: שֶׁהוּא הָיָה דָן קֹדֶם מִנּוּי הַסַּנְהֶדְרִין כָּל מַה שֶּׁהָיוּ הַסַּנְהֶדְרִין דָּנִין אַחַר כָּךְ, וְלֹא נִתְמַנּוּ הַסַּנְהֶדְרִין אֶלָּא לְהָקֵל מֵעָלָיו. וְרַשִׁ"י פֵרֵשׁ שֶׁהָיָה שׁוֹפֵט וְדָן עַל פִּי הַדִּבּוּר וְשָׁקוּל כוּ': וּבְסַנְהֶדְרִין כָּתַב הָרַ"ב כִּדְאַשְׁכְּחַן בְּמשֶׁה, וְאַף עַל גַּב דְּלֹא כְתִיב וְכֵן תַּעֲשׂוּ, מִשּׁוּם דְּהָכָא מְרַבִּינַן נַמִּי מֶלֶךְ וְנָבִיא כִּדְאִיתָא פֶּרֶק ב' דִּשְׁבוּעוֹת. וְלֹא תָנֵי לְהוּ הָכָא, דְּלֹא חָשֵׁיב אֶלָּא מִלֵּי דְסַנְהֶדְרִין. תּוֹסָפוֹת. וְכֵן צָרִיךְ לוֹמַר בְּמִלְחֶמֶת הָרְשׁוּת דְּלֹא תָנֵי הָכָא מֶלֶךְ. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: סַנְהֶדְרִיּוֹת לַשְּׁבָטִים. שֶׁנֶּאֱמַר שֹׁפְטִים וְשֹׁטְרִים תִּתֶּן לְךָ בְּכָל שְׁעָרֶיךָ לִשְׁבָטֶיךָ: וַאֲפִלּוּ הָכִי אִצְטְרִיךְ לְעֵיל לִמְעוּטֵי שֵׁבֶט. כְּגוֹן אִם יֵשׁ בְּעִיר אַחַת מִשְּׁנֵי שְׁבָטִים. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: הָרַמְבַּ"ם. וְכָתַב הַכֶּסֶף מִשְׁנֶה דְּאַף עַל גַּב דְּלֹא קַיְמָא לָן כְּרַבִּי שִׁמְעוֹן דְּדָרֵישׁ טַעֲמָא דִקְרָא, רָאָה רַבֵּנוּ לְכָתְבוֹ, שֶׁזֶּה טַעַם מִתְקָרֵב אֶל הַשֵּׂכֶל וְאֵין בּוֹ נַפְקוּתָא לְדִינָא: וְלֹא שָׁלֹשׁ. דִּכְתִיב עָרֶיךָ, הֲרֵי שְׁתַּיִם אָמוּר, הָא מָה אֲנִי מְקַיֵּם אַחַת, אַחַת וְלֹא שָׁלשׁ. וְכָתְבוּ הַתּוֹסָפוֹת, אִית לְהַקְשׁוֹת דְּלִשְׁתּוֹק מֵאַחַת וַאֲנָא יָדַעְנָא מֵעָרֶיךָ שְׁתַּיִם וְלֹא שָׁלשׁ, דְּתָפַסְתָּ מֻעָט תָּפַסְתָּ: אַחַת כוּ'. לְדַעַת הָרַ"ב וְהָרַמְבַּ"ם דְּמַחְמִירִין לַעֲשׂוֹת בִּשְׁנַיִם וּשְׁלשָׁה מְקוֹמוֹת וְגַם בִּשְׁנַיִם וּשְׁלשָׁה בָּתֵּי דִינִין, נִרְאֶה לִי דְּתָנָא אַחַת לְאַשְׁמְעִינַן שְׁתַּיִם דּוּמְיָא דְאַחַת, דְּאַף עַל גַּב שֶׁהֵן כְּאַחַת שֶׁמְּקֹרָבִים וְעוֹמְדִים עַל הַסְּפָר שֶׁבֵּין גָּלִיל לִיהוּדָה אַחַת מִזֶּה וְאַחַת מִזֶּה, עוֹשִׂין. וְדַוְקָא שְׁתַּיִם, אֲבָל שָׁלֹשׁ בִּכְהַאי גַּוְנָא לֹא, וְעַיֵּן תּוֹסְפוֹת יוֹם וּכְשֶׁהוּא עִמָּם לֵיכָּא לְמֵימַר שֶׁיְּהֵא שָׁקוּל כְּשִׁבְעִים וְאֶחָד וּלְדוֹרוֹת לִבָּעֵי שִׁבְעִים וְשִׁבְעִים וְאֶחָד, דְּלֵיכָּא לְמֵימַר שָׁקוּל אֶלָּא כְשֶׁהוּא לְבַדּוֹ עוֹמֵד כְּנֶגֶד כֻּלָּם. עַיֵּן תּוֹסְפוֹת יוֹם טוֹב: מְיֻחָסִין וּמְנֻקִּים מִמּוּם. רַשִׁ"י. וְרַבָּנָן, מִוְּהָקֵל מֵעָלֶיךָ וְנָשְׂאוּ אִתְּךָ נָפְקָא לְהוּ. גְּמָרָא. וְאִם תֹּאמַר מִכָּל מָקוֹם תֵּיפּוּק לֵיהּ דְּאֵין בֵּית דִּין שָׁקוּל, וְכֵן נַמִּי לְרַבָּנָן לָמָּה לִּי קְרָא. וּבַגְּמָרָא דַּף ג' עַמּוּד ב', אֵימַר דְּשָׁמַעְתְּ לֵיהּ לְרַבִּי יְהוּדָה (דְּלֹא סְבִירָא לֵיהּ אֵין בֵּית דִּין שָׁקוּל) בְּסַנְהֶדְרִי גְדוֹלָה דִּכְתִיבֵי קְרָאֵי (לְקַמָּן בְּשִׁלְהֵי פִרְקִין. רַשִׁ"י) כוּ'. וְלוּלֵי דְמִסְתְּפִינָא הָיִיתִי אוֹמֵר דְּצָרִיךְ לוֹמַר דִּכְתִיבֵי בַקְּרָא, וְהָכִי פֵרוּשׁוֹ, דִּבְהֶדְיָא כְתִיבֵי בַקְּרָא שִׁבְעִים אִישׁ, וּנְחֵית לְמִנְיָנָא, וְדָרֵישׁ אִתְּךָ בְּדוֹמִים לָךְ, וּלְהָכִי נַמִּי רַבָּנָן בָּעוּ קְרָא, דְּמִנְיָנָא לָאו דַּוְקָא. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: וְאֵין כוּ'. רַבִּי אֱלִיעֶזֶר בְּנוֹ שֶׁל רַבִּי יוֹסֵי הַגְּלִילִי אוֹמֵר, מַה תַּלְמוּד לוֹמַר לִנְטֹת אַחֲרֵי רַבִּים לְהַטֹּת, הַתּוֹרָה אָמְרָה עֲשֵׂה לְךָ בֵית דִּין נוֹטֶה. גְּמָרָא:
