פרק יא

הַמַּכֶּה כוּ'. דִּכְתִיב (שְׁמוֹת כא) וּמַכֵּה אָבִיו וְאִמּוֹ מוֹת יוּמָת. וְכָל מִיתָה הָאֲמוּרָה בַתּוֹרָה סְתָם אֵינָהּ אֶלָּא חֶנֶק. גְּמָרָא. וְכֵן בְּכֻלָּם כְּתִיב מִיתָה סְתָם: מַמְרֵא. כְּמוֹ מַמְרִים הֱיִיתֶם (דְּבָרִים ט). רַשִׁ"י: חַבּוּרָה. וְאֵימָא אַף עַל גַּב דְּלֹא עֲבִיד חַבּוּרָה, כְּתִיב בְּמַכֵּה בְהֵמָה נֶפֶשׁ (דְּמַשְׁמַע דָּם דַּהֲוֵי חַבּוּרָה. רַשִׁ"י), וְאִם אֵינוֹ עִנְיָן לִבְהֵמָה, דְּהָא אִי נַמִּי הִכְחִישָׁהּ בַּאֲבָנִים (דְּלֹא עָשָׂה בָהּ חַבּוּרָה) חַיָּב, תְּנֵהוּ עִנְיָן לְנֶפֶשׁ אָדָם (דְּהַיְנוּ אָבִיו). גְּמָרָא. וּתְמִיהָא לִי, דְּלֹא כְתִיב נֶפֶשׁ בִּבְהֵמָה אֶלָּא בְהוֹרְגָהּ וְלֹא בְמַזִּיקָהּ, כִּדְמוֹכְחֵי קְרָאֵי סוֹף פָּרָשַׁת אֱמֹר. וּבִמְכִילְתָּא דָּרְשֵׁי הָכִי, וּמָה אִם מִדַּת נְזָקִין מְרֻבָּה אֵינוֹ חַיָּב עַד שֶׁיַּעֲשֶׂה בָהּ חַבּוּרָה (דְּאִי לָאו הָכִי אֵין כָּאן נֵזֶק), מִדַּת עֳנָשִׁין מְעוּטָה לֹא יִהְיֶה חַיָּב עַד שֶׁיַּעֲשֶׂה בָהֶן חַבּוּרָה כוּ': אֲבָל אִי לָאו הָכִי הֲוָה אֲמֵינָא נֵילַף מִמְּקַלֵּל בְּהֶקֵּשׁ. עַיֵּן תּוֹסְפוֹת יוֹם טוֹב: לִרְשׁוּתוֹ. וְנִסְתַּפְּקוּ הַתּוֹסָפוֹת אִי הָכָא נַמִּי הִגְבִּיהוֹ כְּמוֹ שֶׁהוֹצִיאוֹ דָמֵי, אוֹ דִלְמָא הָכָא שָׁאנֵי, דְּהָא אַשְׁכְּחַן לְקַמָּן דִּמְכָרוֹ לִקְרוֹבִים פָּטוּר, דְּבָעִינַן שֶׁיּוֹצִיאֶנּוּ מֵרְשׁוּת אֶחָיו, כָּל שֶׁכֵּן דְּבִרְשׁוּת בְּעָלִים לֹא מַהֲנֵי כְלָל: וְהִתְעַמֶּר כוּ'. וְתַנָּא קַמָּא נַמִּי בָעֵי עִמּוּר, כִּדְפֵרֵשׁ הָרַ"ב, וְכָל שֶׁכֵּן מְכִירָה, דְּבָהּ לֵית פְּלֻגְתָּא. וְלֹא הָיָה צָרִיךְ לְהַאֲרִיךְ לְהָבִיא הַכָּתוּב, כָּל שֶׁכֵּן וּמְכָרוֹ: וְאֶלָּא לִמְעוּטֵי מַאי. וְיֵשׁ לוֹמַר דְּכֵיוָן דְּמֵ"ם דְּמִבְּנֵי יִשְׂרָאֵל דָּרְשִׁינַן לֵיהּ, אִם כֵּן לֹא סַגִּי דְלָא לִכְתּוּב מֵ"ם אַחֶרֶת לְגוּפֵיהּ דִּקְרָא דְגוֹנֵב נֶפֶשׁ מִן, וּלְהָכִי כָתַב מֵאֶחָיו, מִשּׁוּם הַמֵּ"ם. וְרַבִּי יְהוּדָה לָאו מֵ"ם דָּרֵישׁ אֶלָּא דְמִמֵּילָא אִמַּעַטוּ עֲבָדִים, דְּאֵינָן בִּכְלָל אֶחָיו. וְעוֹד נִרְאֶה לִי, דְּאִצְטְרִיךְ אֶחָיו, כִּדְתַנְיָא רַבִּי שִׁמְעוֹן אוֹמֵר מֵאֶחָיו, עַד שֶׁיּוֹצִיאֶנּוּ מֵרְשׁוּת אֶחָיו, לַאֲפוֹקֵי מְכָרוֹ לִקְרוֹבִים דְּפָטוּר. וּלְרַבִּי יְהוּדָה תַּרְתֵּי בְמַשְׁמַע מֵאֶחָיו. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב:

רַשִׁ"י. וְנִרְאֶה שֶׁסְּבָרָתָם כְּדֵי לְקָרֵב הַבָּתֵּי דִינִין אֶל הַמָּקוֹם הַיּוֹתֵר מְקֻדָּשׁ שֶׁבְּאֶפְשָׁר. וּלְפִיכָךְ פֵּרֵשׁ פֶּתַח הַר הַבַּיִת הוּא הַפֶּתַח אֶל הָעֵבֶר שֶׁלִּפְנִים שֶׁשָּׁם מַתְחֶלֶת עֶזְרַת נָשִׁים. וְלֹא בַפֶּתַח שֶׁאֶל הָעֵבֶר שֶׁבַּחוּץ: דָּרַשְׁתִּי כוּ'. לְשׁוֹן דְּרָשָׁה, כְּלוֹמַר לָרַבִּים בְּבֵית הַמִּדְרָשׁ. וּלְשׁוֹן לִמּוּד, לְתַלְמִידָיו שֶׁמְּשַׁמְּשִׁין לְפָנָיו בְּבֵיתוֹ. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: אִם שָׁמְעוּ כוּ'. רְצוֹנוֹ לוֹמַר אִם הֵם יוֹדְעִים אוֹתוֹ הַדִּין וְשָׁמְעוּ דִין הַתּוֹרָה בוֹ יַגִּידוּ כוּ'. עַד שֶׁיַּגִּיעַ הַדָּבָר לְבֵית דִּין הַגָּדוֹל. וְאִם בֵּית דִּין הַגָּדוֹל מְקֻבָּלִים בַּדָּבָר הַהוּא יוֹדִיעוּם מַה שֶּׁהֵם יוֹדְעִים מִן הַקַּבָּלָה. וְאִם לֹא שָׁמְעוּ בוֹ, יָדוּנוּ בוֹ כְּפִי הַסְּבָרוֹת וְהֶקֵּשׁוֹת הַתּוֹרִיּוֹת הַיְדוּעוֹת אֶצְלָם עַד שֶׁיַּסְכִּימוּ כֻלָּם אוֹ רֻבָּם שֶׁהַדִּין כָּךְ, וְיֹאמְרוּ לָהֶם שֶׁיִּתְחַיְּבוּ לַעֲשׂוֹת כֵּן. הָרַמְבַּ"ם. וְלוֹמַר שֶׁהַשְּׁנֵי בָתֵּי דִינִין אֵינָן רַשָּׁאִים לְהוֹרוֹת מִלִּבָּם אֶלָּא מַה שֶּׁקִּבְּלוּ וְשָׁמְעוּ יֹאמְרוּ וְלֹא זוּלַת כוּ', וְאִם כֵּן אֵינוֹ נַעֲשֶׂה מַמְרֵא אֲפִלּוּ לֹא יִשְׁמַע לְדִבְרֵיהֶם. וְקַשְׁיָא, אִם כֵּן מָה הוֹעִילוּ בְמַאֲמָרָם שֶׁעַל פִּי קַבָּלָתָם כוּ'. וְנִרְאֶה לִי, שֶׁזֶּה הִכְרִיחוֹ לְהָרַמְבַּ"ם שֶׁבְּחִבּוּרוֹ כָתַב בְּעִנְיָן אַחֵר כוּ'. וְכָתַב הַכֶּסֶף מִשְׁנֶה, נִרְאֶה שֶׁהוּא מְפָרֵשׁ אִם שָׁמְעוּ הַזָּקֵן וְקִבְּלוּ לַעֲשׂוֹת כְּדִבְרֵי אוֹתוֹ בֵית דִּין אוֹמְרִים לָהֶם סְבָרָתָם. עַד כָּאן. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: חַיָּב. וַאֲפִלּוּ הוּא אוֹמֵר גַּם כֵּן מִפִּי הַקַּבָּלָה. וְהַטַּעַם, שֶׁהַתּוֹרָה חָלְקָה לָהֶם כָּבוֹד, כְּמוֹ שֶׁחִיְּבָה הַמְקַלֵּל אָבִיו וְאִמּוֹ מִיתָה לִכְבוֹד אָבִיו וְאִמּוֹ. הָרַמְבַּ"ם: מְסַיֵּם רַשִׁ"י לִמְעוּטֵי תַלְמִיד. וְכָתְבוּ הַתּוֹסָפוֹת, וְאֵין זֶה כִשְׁאָר מֻפְלָא דִבְכָל דּוּכְתֵי. וְהָרַמְבַּ"ם כָּתַב, בַּמֻּפְלָא, מִמִּי שֶׁלֹּא יִפָּלֵא מִמֶּנּוּ אֶלָּא הַדָּבָר הַנִּפְלָא הַקָּשֶׁה:

בְּדִבְרֵי סוֹפְרִים. לֹא שֶׁאַרְבָּעָה בָתִּים הֵם מִדִּבְרֵי סוֹפְרִים וְשֶׁמִּדִּבְרֵי תוֹרָה אֲפִלּוּ חֲמִשָּׁה, דְּאִם כֵּן הֵיאַךְ תְּחַיְּבֵהוּ הַתּוֹרָה מִיתָה מַה שֶּׁהוֹרָאַת הַתּוֹרָה עַצְמָהּ הִיא כָךְ, אֶלָּא כָּל שֶׁאֵינוֹ מְפֹרָשׁ בַּתּוֹרָה אַף עַל פִּי שֶׁהוּא דְאוֹרַיְתָא דִּבְרֵי סוֹפְרִים קָרִינַן לֵיהּ, כְּלוֹמַר שֶׁאִלּוּלֵי קִבְּלוּ סוֹפְרִים פֵּרוּשׁוֹ לֹא הָיִינוּ מְבִינִים אוֹתוֹ כָךְ: פָּטוּר. כְּשֶׁהוֹדָה בְחִיּוּבוֹ בָהֶם, אֲבָל הוּא יְבַטֵּל אוֹתָהּ כְּמוֹ שֶׁאָמַר לַעֲבֹר עַל דִּבְרֵי תוֹרָה עַד שֶׁיּוֹדֶה שֶׁהִיא עֲבֵרָה, וּלְפִיכָךְ אֵינוֹ חַיָּב מִיתַת בֵּית דִּין, לְפִי שֶׁלֹּא יֵהָרֵג מִי שֶׁבִּטֵּל מִצְוַת עֲשֵׂה. אֲבָל אִם אָמַר אֵין תְּפִלִּין עַל דֶּרֶךְ כְּפִירָה יֵהָרֵג מִצַּד שֶׁהוּא מִין. הָרַמְבַּ"ם. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: כֵּן הוּא הַגִּרְסָא בִּזְבָחִים דַּף ל"ג. וְעַיֵּן בְּאוֹר תּוֹרָה וּבְמִנְחַת שַׁי. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב:

אֵין מְמִיתִין כוּ'. אֲבָל דָּנִין אוֹתוֹ בְּבֵית דִּין שֶׁבְּעִירוֹ. גְּמָרָא. וְעַיֵּן פֶּרֶק קַמָּא מִשְׁנָה ה': רַשִׁ"י. וְצָרִיךְ לְפָרֵשׁ אֶלָּא מַעֲלִין כוּ' כְּשֶׁהַבֵּית דִּין חוֹזֵר. וְהַתּוֹסָפוֹת הִקְשׁוּ וְהֶעֱלוּ דְּדַוְקָא כְשֶׁנִּגְמַר דִּינוֹ קֹדֶם שֶׁגָּלְתָה, דִּלְאַחַר שֶׁגָּלְתָה בָּטְלָה דִינֵי נְפָשׁוֹת. וְלִי נִרְאֶה דְּאֶלָּא מַעֲלִין אֵינוֹ חוֹזֵר אֶלָּא עַל בֵּית דִּין שֶׁבְּעִירוֹ: דְּתַנְיָא, אֶל הַמָּקוֹם, מְלַמֵּד שֶׁהַמָּקוֹם גּוֹרֵם: רַבִּי יְהוּדָה אוֹמֵר כוּ'. בַּגְּמָרָא, אָמַר לוֹ רַבִּי יְהוּדָה, וְכִי נֶאֱמַר יִרְאוּ וְיִירָאוּ. וַהֲלֹא לֹא נֶאֱמַר אֶלָּא יִשְׁמְעוּ וְיִירָאוּ, לָמָּה מְעַנִּין וְכוּ':

וּמִיהוּ לְהַתְרָאָתוֹ צָרִיךְ שֶׁיִּהְיוּ נְבִיאִים. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: וְהַמְוַתֵּר כוּ'. הַמַּפְקִירָם, שֶׁלֹּא חָשַׁשׁ לְמַה שֶּׁאָמַר לוֹ הַנָּבִיא. רַשִׁ"י. וּלְשׁוֹן הָרַמְבַּ"ם, שֶׁיָּקֵל וְיַעֲלִים עַיִן מִדִּבְרֵי הַנְּבִיאִים וְלֹא יַעֲשֶׂה מִצְוָתָם עַל הַדִּיּוּק וְהַדִּקְדּוּק: וּבַגְּמָרָא לֹא אָמְרוּ בַלָּשׁוֹן הָרִאשׁוֹן קָרֵי בֵיהּ, אֶלָּא דִכְתִיב וְהָיָה הָאִישׁ אֲשֶׁר לֹא יִשְׁמַע וּקְרִי בֵיהּ לֹא יַשְׁמִיעַ אֶל דְּבָרַי. וְהָכִי פֵרוּשׁוֹ, דִּכְתִיב וְהָאִישׁ אֲשֶׁר לֹא יִשְׁמַע, הַיְנוּ הַמְוַתֵּר, שֶׁהוּא מַשְׁמָעוּת הַכָּתוּב כְּמִי שֶׁאֵינוֹ שׁוֹמֵעַ לְדִבְרֵי ה' שֶׁדִּבֵּר הַנָּבִיא בִּשְׁמוֹ. וּקְרִי בֵיהּ לֹא יַשְׁמִיעַ הַיְנוּ כּוֹבֵשׁ כוּ'. וּקְרִי בֵיהּ לֹא יִשְׁמַע אֶל דְּבָרַי, כְּלוֹמַר פָּסְקִינַן לִקְרָא בְּסַכִּינָא חָרִיפָא וְלֹא מַתְחִיל לִקְרוֹת וְהָיָה הָאִישׁ, אֶלָּא אֲשֶׁר לֹא יִשְׁמַע, וּבַנָּבִיא גוּפֵיהּ קָאָמַר. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב:

בְּשֵׁם עֲבוֹדָה זָרָה. שֶׁנֶּאֱמַר וַאֲשֶׁר יְדַבֵּר בְּשֵׁם אֱלֹהִים אֲחֵרִים. גְּמָרָא: מַקְדִּימִין. כְּלוֹמַר אֵין לָהֶם לְצַפּוֹת מִיתָה אַחֶרֶת, אֶלָּא מַשְׁכִּימִין לְאוֹתָהּ שֶׁנִּגְמַר לָהּ דִּינוֹ שֶׁל נִדּוֹן. רַשִׁ"י. וְהַתּוֹסָפוֹת פֵּרְשׁוּ דִּלְהָכִי קָאָמַר מַקְדִּימִין, כְּלוֹמַר וַדַּאי אַתָּה צָרִיךְ לַהֲמִיתָם בַּמִּיתָה שֶׁרָצוּ לַהֲמִיתוֹ, אֲבָל וַדַּאי אִם אֵינְךָ יָכוֹל, אֲפִלּוּ הָכִי נַמִּי נְמִיתֵם בְּכָל מִיתָה שֶׁנּוּכַל. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: וּבוֹעֲלָהּ. נִרְאֶה דְּהָכִי קָאָמַר, חוּץ מִשֶּׁכְּשֶׁהוּזַמּוּ לְבַת כֹּהֵן עִם בּוֹעֲלָהּ, שֶׁגַּם בּוֹעֲלָהּ נִדּוֹן, דְּמִתְקַיֵּם בְּהוּ כַּאֲשֶׁר זָמַם לַעֲשׂוֹת לְאָחִיו. אֲבָל כְּשֶׁלֹּא זָמְמוּ לְאָחִיו, כְּגוֹן שֶׁאֵין מַכִּירִין אוֹתוֹ אוֹ לֹא הִתְרוּ בוֹ כוּ', דְּהַשְׁתָּא אֵין לָנוּ לְהַקִּישָׁם לוֹ, דְּאִין הָכִי נַמִּי דְּמַקְדִּימִין אוֹתָם לַדִּין שֶׁלָּהּ. וְזֶה נִרְאֶה בִּלְשׁוֹן רַשִׁ"י וְהָרַמְבַּ"ם. אֶלָּא שֶׁהַתּוֹסָפוֹת הוֹכִיחוּ דְּאַף עַל פִּי שֶׁאֵין דִּין בַּבּוֹעֵל, נִדּוֹנִים בְּחֶנֶק. וּלְדִבְרֵיהֶם צָרִיךְ לוֹמַר דְּתַנָּא מִסְּרַךְ סְרִיךְ בְּלִישְׁנָא דְרֵישׁ פִּרְקִין, וְנָקַט וּבוֹעֲלָהּ אַף עַל גַּב דְּלֹא אִצְטְרִיךְ לֵיהּ הָכָא:
