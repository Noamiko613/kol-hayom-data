פרק ד

דֵּכְיוֹן דֵּאְין תָּשִֵמִישִוֲ כֵּשִּל בָּיִת, שִּאְינוֲ עֹשֵוּי אּלֹּא לֵהָצֵנִיעָ שִֹם חַפֹצֹיו לִשִֵמֲר, כֵּמִין תְּבֹה, הִלֵכֹּךֵ אְינוֲ בִּכֵלֹל בָּיִת. רָ"שִ. וּבָגֵּמֹרֹא, הָשִֵתֹּא יֹצִיעָ, חּדּר מִבָּעֵיֹא. לֲא צֵרִיכֹא דֵּאָף עָל גָּב דֵּמֹצָר לְיהּ אָבֵּרֹאִי, שִּכֹּתָב לוֲ הָמְּצָר שִּחוּץ לָחּדּר, אַפִלּוּ הֹכִי לֲא אֹמֵרִינָן שִּקֹּנֹה הָחּדּר כְּיוֹן שִּהוּא בֵּתוֲךֵ הָמְּצָר, אּלֹּא אֹמֵרִינָן מֵצֹרִים הִרֵחִיב, שִּמְּצָר חּדּר אְינוֲ נִכֹּר: הָגָּג. אַבֹל עַלִיֹּה בִּכֵלֹל הָבָּיִת:

אַבֹל הֹרָ"שִ פְּרְשִ בּוֲר, חֹפוּר בֵּקָרֵקָע קֹשִֹה, שִּמָּחֵזִיק מָיִם בֵּלֲא בִּנֵיֹן. דּוּת, חוֲפֵרִין בּוֲר בֵּעֹפֹר תִּחוֲחָ וּבוֲנִין לוֲ כֲּתּל אַבֹנִים וּמִקֵרְי דּוּת. וֵעָיְּן תּוֲסֵפוֲת יוֲם טוֲב: וֵרוּמֹא כוּ'. נִמֵשִָךֵ גָּם עָל וֵלֲא אּת הָגָּג. הָמָּגִּיד. וֵכֹתָב עוֲד וּכֵשִּכֹּתָב לוֲ עֲמּק וֵרוּם אִם בֹּא לֵהָגֵבִּיהָּ וֵלִבֵנוֲת בֹּאַוִיר אוֲ לֵהָעַמִיק וֵלָחֵפֲּר, מָגֵבִּיהָּ וּמָעַמִיק. אַבֹל לֲא קֹנֹה בָּבִּנֵיֹנִים שִּבֹּעַמֹקִים וּבֹאַוִיר. וֵעָיְּן תּוֲסֵפוֲת יוֲם טוֲב: וּלֵרָבֹּנֹן לֲא אֹתְי לִטֵפוּיְי כֵּלֹל דִּבֵלֹאו הֹכִי שִִיְּר לֵעָצֵמוֲ דּּרּךֵ, וּלֵרָוֵחֹא דֵּמִלֵּתֹא אֹמָר כְּן שִּלֲּא יֲאמָר לוֲקְחָ כֹּל הָבָּיִת מֹכָרֵתֹּ לִי וֵלֲא עִכָּבֵתֹּ דּּרּךֵ, שִּאְין כֹּל מוֲכֵרִין בֵּקִיאִין בָּדִּינִין, וֵהִלֵכֹּךֵ פְּרְשִ כֹּל כֹּךֵ. רָ"שִ: גֵּמֹרֹא. וֵאָכֶּלֹּהּ מָתֵנִיתִין קֹאְי:

כֵּלוֲמָר אָף עָל גָּב דִּקֵבִיעֹה, כִּדֵאֹמֵרִינָן בָּגֵּמֹרֹא דֵּקֹתֹנְי מָפֵתְּחָ דּוּמֵיֹא דֵּדּלּת, מִכֹּל מֹקוֲם אְינֹהּ מֵכוּרֹה מִשִּוּם דֵּאְין קֵבִיעוּתֹהּ קֵבִיעוּת גֵּמוּרֹה. וֵעָיְּן תּוֲסֵפוֲת יוֲם טוֲב:

בּוֲרוֲת כוּ'. לֵפִי שִּהּחֹצְר עֹשֵוּי לִכֵבִיסֹה וּלֵהָשִֵקוֲת בֵּהְמוֲת, וֵתָשִֵמִישִ הּחֹצְר וֵהָבּוֲר הוּא תָּשִֵמִישִ אּחֹד, וּלֵפִיכֹךֵ נִמֵכֹּרִין עִמֹּהּ. אַבֹל בָּיִת אְינוֲ עֹשֵוּי לִכֵבִיסֹה כוּ' אּלֹּא לֵדִירֹה וֵאְין תָּשִֵמִישִֹן שִֹוּה. רָשִֵבֹּ"א. וֵעָיְּן תּוֲסֵפוֲת יוֲם טוֲב: הָמִּטָּלֵטֵלִין. תָּשִֵמִישְִי הָבָּיִת שִּאְינֹן קֵבוּעִין כֹּל כֹּךֵ, הֹנֹךֵ דֵּאֹמֵרֹן לֵעְיל דֵּאְינֹן מֵכוּרִין בִּכֵלָל בָּיִת. וֵכֹל שִּכְּן שִֵאֹר מִטָּלֵטֵלִין שִּאְינֹם בִּכֵלֹל תָּשִֵמִישְִי בָּיִת. רָ"שִ: הָמּּרֵחֹץ. דֵּאְין דּּרּךֵ הָחַצְרוֲת שִּיִּהֵיּה לֹהּם כְּן וֵאְינוֲ תָּשִֵמִישִ מֵיֶחֹד לֵחֹצְר אָחָת. דֵּהָרֵבְּה חַצְרוֲת מִשִֵתָּמֵּשִוֲת בֵּמּרֵחֹץ אּחֹד וֵכְן בֵּבְית הָבָּד. נִמּוּקְי יוֲסְף: אַוִירֹהּ. אַבֹל בֹּתִּים וּבוֲרוֲת לֲא. וֵדֹמִים מוֲדִיעִים לְיכֹּא לֵמְימָר, דֵּאְין אוֲנֹאֹה לָקָּרֵקֹעוֲת. רָ"שִ. וֵעָיְּן תּוֲסֵפוֲת יוֲם טוֲב:

בְּית הָבָּד. בָּיִת שִּהָבָּד בּוֲ, וֵאוֲתוֲ בָּיִת לֵתָשִֵמִישִ בָּד לֵבָדּוֲ עֹשֵוּי. רָ"שִ: הָיֹּם. כֹּל הֹנֹךֵ דֵּקֹתֹנְי בֵּהוּ מֹכָר קֵבוּעִין הְן שִֹם. וֵאִידֹךֵ דֵּאְינֹן מֵכוּרִין מִטָּלֵטֵלְי הְן. רָ"שִ: שִּנּוֲעַצִין בֹּאֹרּץ בֵּרֲאשִ הָגָּת מִכֹּאן וּמִכֹּאן. וֵנוֲקֵבִין זּה כֵּנּגּד זּה וֵנוֲתֵנִים בֵּרִיחִים מִזּּה לֹזּה וּמָעַמִידִים הָקּוֲרֹה עָל הָבֵּרִיחָ. הֹרָ"שִ: הָקּוֲרֹה. שִּלֲּא נִקֵרֹא בְּית הָבָּד אּלֹּא עָל שְִם הָקּוֲרֹה. גֵּמֹרֹא:

הָנֵּסֹרִים. מִשִּוּם דֵּמִטָּלֵטֵלְי לֲא מִזֵדָּבֵּנְי. רָשִִ"י: הָמֵּגֶרוֲת כוּ'. דֵּאְין אְלּוּ תָּשִֵמִישִִין הָמֵּיֶחֹדִים לֵמּרֵחֹץ כִּי הֹנֹךֵ דִּלֵעְיל. וֵאְין אְלּוּ מֹאנְי תָּשִֵמִישִֵתְּיהּ אּלֹּא מֹמוֲנוֲ הְם לְהֹנוֲת בֹּהּם אַפִלּוּ בֵּתוֲךֵ בְּיתוֲ, דִּכֵשִּקּוֲנּה אוֲתוֲ קוֲנּה לֵהָסִּיק בֵּבְיתוֲ כוּ' כֵּמוֲ חִטְּי וּשֵֵעֹרְי . מֵגוּרֹה כֵּמוֲ נּהּרֵסוּ מָמֵגוּרוֲת . רָשִִ"י:

שִּנּּאֱמָר שִֵלֹחָיִךֵ פָּרֵדְּס רִמּוֲנִים . וּפְרְשִ רָשִֵבָּ"ם שִֵלֹחָיִךֵ גִּנּוֲת הָמֵּשִָלֵּחִים פְּרוֲת בֵּכֹל שִֹנֹה. וֵכֹתֵבוּ הָתּוֲסֹפוֲת, וֵאִם תֲּאמָר דִּבֵכֶלְּי גֵּמֹרֹא מֵפֲרֹשִִין שִֵלֹחִין בֹּאגְי שֵֹדוֲת שִּמָּשִֵקִין אוֲתֹן כוּ'. וֵיְשִ לוֲמָר דִּפֵשִִיטֹא לָגֵּמֹרֹא דֵּאְין חִלּוּק בְּין בֹּאגְי לֵבָעָל לֵעִנֵיָן מּכּר כוּ', וֵהָשִֵתֹּא דִּמֵפֹרְשִ פָּרֵדְּס, מֵמָעַטְי שֵֵדְה שִֵלֹחִין וּבָעָל. וֵכֹתָב הֹרָ"שִ מִשִּוּם דֵּגָן רֹאוּי לֵטָיְּל וֵשִָיֹּךֵ טֵפְי לֹעִיר מִבֹּאגְי. וֵעָיְּן תּוֲסֵפוֲת יוֲם טוֲב: מֵכוּרִין. דֵּכְיוֹן דֵּיֹצֹא מִכֹּל הֹעִיר לֹלּכּת לֵעִיר אָחּרּת, הוֲצֹאָת הָמִּטָּלֵטֵלִין הַוְי מֵרֶבֹּה מִשִּּבָח. נִמּוּקְי יוֲסְף: לֵשִוֲן נֲטְרֹה אּת הָכֵּרֹמִים . וֵכֹל שִּכְּן שִּחוֲלְק עָל תָּנֹּא קָמֹּא בֵּמָה שִּּאֹמָר דֵּבֹאגְי לֲא. אַבֹל בָּגֵּמֹרֹא מָסֵּקִינָן דִּלֵמָאן דֵּאֹמָר שִֵלֹחִין פָּרֵדְּס, סָנֵטְר בֹּאגְי. וּלֵמָאן דֵּאֹמָר סָנֵטְר שִוֲמְר, שִֵלֹחִין בֹּאגְי. וֵעָיְּן תּוֲסֵפוֲת יוֲם טוֲב:

וֵאָף עָל גָּב דֵּתָשִֵמִישִִין הָמִּטָּלֵטֵלִין הְן, כְּיוֹן דִּקֵבִיעְי לֵהָאי תָּשִֵמִישִ וֵאְין מֵזִיזִין אוֲתֹן מְהָאי שֵֹדּה לֵעוֲלֹם, כִּמֵחֶבֹּר חֹשִִיב לֵהוּ. רָ"שִ: שִּבָּכּּרּם. וּמִמְּילֹא שִֹמֵעִינָן הָמּוֲכְר אּת הָשֵֹּדּה מֹכָר אּת הָכּּרּם. שִּכֹּל אִילֹנוֲת בִּכֵלֹל שֵֹדּה, אּחֹד גֵּפֹנִים וֵאּחֹד כֹּל אִילֹן, חוּץ מְחֹרוּב הָמֶּרֵכֹּב וֵסָדָּן הָשִִּקֵמֹה. רָ"שִ. וֵעָיְּן תּוֲסֵפוֲת יוֲם טוֲב: גֹּרָס כֵּגִרֵסָת הֹרָמֵבָּ"ם. אַבֹל הֹרָמֵבָּ"ם לֲא כֹּתָב הָמֵּחֶבֹּרִים לָקָּרֵקָע. דֵּבָגֵּמֹרֹא אֹמֵרִינָן עַלָהּ אָף עָל גָּב דֵּלֲא קֵבִיעֹא בֵּאָרֵעֹא. וּפְרְשִ הָנִּמּוּקְי יוֲסְף, שִּמֵּכֶסֹּה וֵטוּחֹה מִלֵּמָעֵלֹה, וֵאָף עָל גָּב שִּלֲּא שֵֹם סֵבִיבוֲתּיהֹ הָרֵבְּה כֵּמִנֵהָג הֹעוֲלֹם :

הָתֵּבוּאֹה כוּ'. וֵאָף עָל גָּב דִּצֵרִיכֹא לֵאָרֵעֹא. גֵּמֹרֹא. וּפְרְשִ לֵשִֹטֵחֹהּ בָּשֵֹּדּה לֵאָחָר קֵצִירֹה לֵהִנֹּפְחָ הָתֵּבוּאֹה מְרְיחָ הָקָּרֵקָע וּלֵיָבְּשִ. רָ"שִ: וָהַרְי הוּא כֵּמוֲ שִּמּוֲכְר לָחַבְרוֲ שִֵתְּי שֵֹדוֲת וֵיְשִ לוֲ שֵֹדּה אָחּרּת בֵּתוֲךֵ הָשִֵּתָּיִם. וֵאִם אֹמָר לוֲ שִֵתְּי הָשֵֹּדוֲת וֵכֹל מָה שִּּבֵּתוֲכֹן אַנִי מוֲכְר לֵךֹ, דֵּלֲא מֹכָר הָשֵֹּדּה הָשִֵּלִישִִית שִּבְּינְיהּן, אָף כֹּאן שֵֹדּה אָחָת מֹכָר לוֲ וֵלֲא שִֵתְּי שֵֹדוֲת. וּמֵחִצָּת קֹנִים וֵכוּ' חַשִוּבִין שֵֹדּה בִּפֵנְי עָצֵמֹהּ. רָ"שִ: בֵּטִיט. וֵאָף עָל גָּב דִּמֵחָבֵּרֹא לֵאָרֵעֹא. גֵּמֹרֹא. דֵּכְיוֹן דֵּטוּחֹה בֵּטִיט אְינֹהּ מִטָּלֵטּלּת עוֲד וָחַשִִיבֹה וֵיְשִ לֹהּ שְִם בִּפֵנְי עָצֵמֹהּ, וָהַרְי זּה כֵּמוֲכְר הָשֵֹּדּה שִּלֲּא מֹכָר אּת הָבָּיִת כוּ'. רָ"שִ : וֵזּה לֵשִוֲן הָנִּמּוּקְי יוֲסְף, וֵאָף עָל גָּב דִּמֵחֶבֹּר בֵּטִינֹא, כֵּלוֲמָר הָכֵּלוֲנֵסוֲת, כְּיוֹן שִּאְינֹהּ טוּחֹה וּמֵכֶסֹּה מִלֵּמָעֵלֹה בֵּטִיט: הָבּוֲר. וֵהוּא הָדִּין דּוּת, וֵחֹדֹא מִנָּיֵהוּ נֹקָט וֵעָיְּן תּוֲסֵפוֲת יוֲם טוֲב: הָגָּת. גָּת שִֵכִיחָ בָּשֵֹּדּה, דֵּהָיֵנוּ כּּרּם בִּכֵלֹל כֵּדִלֵעְיל. אַבֹל בְּית הָבָּד שִֵכִיחָ בֹּעִיר. וּלֵפִיכֹךֵ בֵּמוֲכְר הֹעִיר תֵּנָן בְּית הָבָּדִּין וֵלֲא תֵּנָן גָּת, וּבָשֵֹּדּה תֵּנָן גָּת וֵלֲא בְּית הָבָּד: הּחֱזִיק. בֵּגְר דֵּאְין דָּעָת אָחּרּת מָקֵנֹה וֵאְין צָד שִּיִּקֵנּה אּלֹּא בָּחַזֹקֹה, תֹּנְי הּחֵזִיק. אַבֹל בֵּאָחִין דֵּאִיכֹּא נָמִּי קִנֵיֹן אִי נָמִּי חַזֹקֹה גֵּרוּעֹה, תֵּנָן זֹכוּ: מִשִּוּם דִּכֵשִּאְינֹן בִּכֵלָל שֵֹדּה הָוֵיֹן כֵּשֵֹדּה בִּפֵנְי עָצֵמֹהּ וֵדֹמְי כִּשִֵתְּי שֵֹדוֲת, לֵהֹכִי נָקֵטִינֵהוּ לֵאָשִֵמוֲעִינָן דָּהַווּ כֵּאְין מְצָר בְּינְיהּן, וֵאָף עָל גָּב דֵּבוֲר וֵגָת הַווּ רֵבוּתֹא טֹפְי לֵעִנֵיָן זּה כוּ'. וּבֵאָחִין נֹקָט בּוֲר וֵגָת וֵלֲא נֹקָט הֹא דִּשִֵתְּי שֵֹדוֲת, מִשִּוּם דֵּלֲא שִָיֵכֹא אּלֹּא בֵּדִין חַזֹקֹה. אַבֹל בִּשִֵאֹר קִנֵיֹנִין לֲא שִָיֹּךֵ חִלּוּק מְצָר מָפֵסִיק כוּ', דָּהַוֹה כֵּנֹתָן דֵּמְי כֶּלֹּן, דֵּקֹנֹה כֶּלֹּן אַפִלּוּ מְצָר מָפֵסִיק: וֵגִדּוּלְי הּקֵדְּשִ נִינֵהוּ. רָ"שִ. וֵעָיְּן תּוֲסֵפוֲת יוֲם טוֲב:
