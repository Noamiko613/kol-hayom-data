פרק א - משנה א

הַשֻּׁתָּפִין. שֶׁיֵּשׁ לָהֶם חָצֵר בְּשֻׁתָּפוּת וּבָתֵּי שְׁנֵיהֶם פְּתוּחִין לְתוֹכוֹ. וְכָל חֲצֵרוֹת שֶׁשָּׁנוּ חֲכָמִים לִפְנֵי הַבָּתִּים הֵן. וְרֹב תַּשְׁמִישָׁן בֶּחָצֵר. רַשִׁ"י: וּמַאי מְחִצָּה, פְּלֻגְתָּא. דִּכְתִיב וַתְּהִי מֶחֱצַת הָעֵדָה. וְשֶׁרָצוּ לַחְצוֹת מִבַּעְיָא לֵיהּ, אֶלָּא כִּדְאָמְרֵי אֱינָשֵׁי תָּא נַעֲבִיד פְּלֻגְתָּא. גְּמָרָא: דְּאִי לֹא הֶחְזִיק כִּי רָצוּ מַאי הֲוָה, נְהַדְרוּ בְּהוּ. גְּמָרָא: לָאו דַּוְקָא שֶׁהֶחְזִיק כָּל אֶחָד, אֶלָּא כְּשֶׁהֶחְזִיק הָאֶחָד קָנוּ שְׁנֵיהֶם. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: וְלֹא מָצֵי אָמַר לֵיהּ כִּי אִתְרָצִי לָךְ בִּמְחִצָּה דִּנְסָרִים שֶׁהִיא דַקָּה, אֲבָל עָבָה דַּאֲבָנִים שֶׁמְּמַעֵט הַתַּשְׁמִישׁ, לֹא, קָא מַשְׁמַע לָן. גְּמָרָא: בּוֹנִים. וְטַעְמָא דְּבָעִינַן שִׁעוּרִים גְּדוֹלִים כָּל כָּךְ אַף עַל פִּי שֶׁאֵינוֹ אֶלָּא הֶזֵּק רְאִיָּה, מִשּׁוּם דְּחַיְשִׁינַן שֶׁמָּא יִפֹּל וְיִצְטָרֵךְ לְהִתְעַצֵּם עִמּוֹ בְּבֵית דִּין, וּבֵין כָּךְ וּבֵין כָּךְ יְהֵא מַזִּיקוֹ בְּהֶזֵּק רְאִיָּה. נִמּוּקֵי יוֹסֵף: וְדַוְקָא בְּהוּצָא וְדַפְנָא. אֲבָל פָּחוֹת מִכָּאן אֲפִלּוּ נָהֲגוּ מִנְהַג הֶדְיוֹט הוּא וְאֵין הוֹלְכִים אַחֲרָיו. תּוֹסָפוֹת. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: כֹּתֶל קָרֵי לְשֶׁל אֲבָנִים וּלְבֵנִים, וּמְחִצָּה קָרֵי לְשֶׁל לוּלְבֵי גְּפָנִים: שְׁלֹשָׁה טְפָחִים. וְהָא דְּלֹא קָאָמַר שִׁשָּׁה טְפָחִים, מִשּׁוּם דְּלֹא נִטְעֶה לְמֵימַר שִׁשָּׁה לְכָל אֶחָד. וְכָתַב הָרֹא"שׁ דְּהָנָךְ דְּמַתְנִיתִין לֹא תָלֵי בַּמִּנְהָג, דְּקִים לְהוּ לְרַבָּנָן דִּבְצִיר מֵהָכִי לֹא קָאֵי. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: גְּמָרָא. דְּשֻׁתָּפִין לֹא קָפְדֵי אַהֲדָדֵי, וַאֲפִלּוּ נִשְׁתַּהָה בִּרְשׁוּתוֹ. הָרֹא"שׁ. וּמָקוֹם, אַגַּב נַקְטֵיהּ. תּוֹסָפוֹת:

פרק א - משנה ב

גְּמָרָא. דְּאִי לָאו הָכִי קַשְׁיָא, דִּבְגִנָּה מַשְׁמַע דְּאִי סְתָמָא לֹא מְחַיְּבִין, וּבְבִקְעָה מַשְׁמַע דְּאִי סְתָמָא מְחַיְּבִין, וְהָא גִּנָּה חֲמוּרָה מִבִּקְעָה: זֶה לְשׁוֹן הָרַמְבַּ"ם בְּפֵרוּשׁוֹ. מִי שֶׁמָּכַר גִּנָּה סְתָם וְלֹא זָכַר שֶׁהִיא מְעֹרֶבֶת עִם גִּנַּת פְּלוֹנִי, מְחַיְּבִין לַלּוֹקֵחַ לַעֲשׂוֹת מְחִצַּת הַגָּדֵר, וַאֲפִלּוּ בְּמָקוֹם שֶׁנָּהֲגוּ שֶׁלֹּא לִגְדֹּר, אֲבָל בִּקְעָה כוּ'. וְעַיֵּן סוֹף פֶּרֶק ז' מִשְׁנָה ד'. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: לִגְדֹּר. אֲבָל בֶּחָצֵר לֹא אַדְכָּר תַּנָּא מָקוֹם שֶׁנָּהֲגוּ כְּלָל, מִשּׁוּם דְּהֶזֵּק רְאִיָּה דְּחָצֵר שֶׁהוּא מְקוֹם תַּשְׁמִישׁוֹ שֶׁל אָדָם חָמִיר טְפֵי, דְּמַקְפִּידִין בּוֹ יוֹתֵר מִבְּגִנָּה שֶׁהִיא מִשּׁוּם עַיִן הָרַע בִּלְבַד. הָרַשְׁבָּ"א: בְּבִקְעָה. שְׂדֵה לָבָן. רַשִׁ"י. שֶׁהִיא שְׂדֵה תְּבוּאָה שֶׁאֵין הָעַיִן שׁוֹלֵט בָּהּ כָּל כָּךְ כְּמוֹ בְּגִנָּה שֶׁיֵּשׁ בָּהּ יָרָק הַרְבֵּה. נִמּוּקֵי יוֹסֵף: פֵּרוּשׁ חָזִית, מַחֲזֶה, כְּלוֹמַר דָּבָר הַנִּרְאֶה לְהוֹדִיעַ לְמִי הוּא הַכֹּתֶל. הָרַמְבַּ"ם: וְעוֹשִׂין כוּ'. וּבַגְּמָרָא, לֹא יַעֲשֶׂה לֹא זֶה וְלֹא זֶה. וּמְשָׁנֵי כְּגוֹן דְּקָדִים אֶחָד וְנִכְנַס לִרְשׁוּת חֲבֵרוֹ וְעָשָׂה שָׁם חָזִית, אַשְׁמְעִינַן שֶׁיַּעֲשֶׂה גַּם הַשֵּׁנִי. וּפָרֵיךְ וְתַנָּא תַּקַּנְתָּא לָרַמַּאי קָא מַשְׁמַע לָן, בִּתְמִיָּה. בִּשְׁלָמָא רֵישָׁא דְּתָנֵי דִּינֵי דְּכוֹנֵס תָּנָא תַּקַּנְתָּא, אֲבָל הָכָא בִּשְׁנֵיהֶם רוֹצִים לֹא אִצְטְרִיךְ לְדִינָא. אָמַר רָבִינָא, בְּהוּצָא עָסְקִינַן. פֵּרֵשׁ רַשִׁ"י מְחִצַּת נְסָרִים. וְקָא מַשְׁמַע לָן דְּמַהֲנֵי חָזִית, וְלֹא אָמְרִינַן קוּפְלָא דְּטִינָא מֵעִלּוּי הוּצָא לֹא יָדַע. וּלְפִי זֶה כָּתַב הָרֹא"שׁ דְּהוּא הַדִּין דְּחָזִית מַהֲנֵי לְחָצֵר שֶׁאֵין בּוֹ דִּין חֲלֻקָּה. וְלֹא תָּנֵי לֵיהּ מִשּׁוּם דְּלֹא תָּנֵי דִּינָא גַּבֵּיהּ: וּמִכָּאן. לְשׁוֹן הַטּוּר יַעֲשֶׂה כָּל אֶחָד וְאֶחָד סִימָן כָּזֶה לְצַד חֲבֵרוֹ:

פרק א - משנה ג

מִשְּׁלֹשׁ וְכוּ'. שֶׁקָּנָה שָׁלֹשׁ שָׂדוֹת סְבִיב שְׂדֵה חֲבֵרוֹ לְשָׁלֹשׁ מְצָרֶיהָ, וְגָדַר אֶת שְׁלָשְׁתָּן, וְנִמְצָא שָׂדֵהוּ שֶׁל אֶמְצָעִי זֶה מֻקָּף מִשָּׁלֹשׁ רוּחוֹתָיו. רַשִׁ"י: וְאַף עַל גַּב דְּאָמְרִינַן לְעֵיל דִּסְתָם בִּקְעָה מְקוֹם שֶׁנָּהֲגוּ שֶׁלֹּא לִגְדֹּר הוּא, הָנֵי מִילֵי בֵּין בִּקְעָה לְבִקְעָה, וְכֹתֶל גָּבוֹהַּ אַרְבַּע אַמּוֹת לְשָׁם הוּא דְּלֹא בָּעִינַן, מַאי טַעֲמָא דְּלֵיכָּא הֶזֵּק רְאִיָּה, אֲבָל בִּשְׁבִיל הֶזֵּק בְּהֵמוֹת שֶׁלֹּא יִכָּנְסוּ לַשָּׂדֶה צָרִיךְ לִגְדֹּר גָּדֵר שֶׁלּוֹ עֲשָׂרָה טְפָחִים הַמּוֹנֵעַ מִלִּכָּנֵס הַבְּהֵמוֹת. רַשִׁ"י:

פרק א - משנה ד

עַד כוּ'. מְשַׁמֵּשׁ לְמַעְלָה וּלְמַטָּה. כֹּתֶל חָצֵר שֶׁנָּפַל מְחֻיָּב לִבְנוֹתוֹ עַד אַרְבַּע אַמּוֹת וְהָדָר קָתָנֵי עַד אַרְבַּע אַמּוֹת בְּחֶזְקַת כוּ'. בֵּית יוֹסֵף. כְּלוֹמַר וְהָדָר קָתָנֵי עַד אַרְבַּע אַמּוֹת כוּ' אֲפִלּוּ בִּתְחִלַּת בִּנְיַן הַכֹּתֶל וְדוֹ"ק: וְתֵימַהּ הוּא, דְּמִשּׁוּם שֶׁלֹּא נָתַן מִיָּד בְּשָׁעָה שֶׁתּוֹבְעוֹ הֲוָה לֵיהּ בְּחֶזְקַת שֶׁלֹּא נָתַן לְעוֹלָם, וְדִלְמָא אִשְׁתְּמוּטֵי מִשְׁתַּמֵּט כוּ'. אֶלָּא הָכָא כְּגוֹן שֶׁבָּאוּ עֵדִים וְאָמְרוּ עִמָּנוּ הָיָה שִׁמְעוֹן בְּמָקוֹם פְּלוֹנִי מִיּוֹם שֶׁהֵחֵל רְאוּבֵן בִּנְיַן כֹּתֶל זֶה, וְיוֹדְעִין אָנוּ שֶׁלֹּא פְּרָעוֹ. אִי נַמִּי כְּגוֹן שֶׁעָמַד שִׁמְעוֹן בַּדִּין וְחִיְּבָהוּ בֵּית דִּין לִבְנוֹת עַד אַרְבַּע אַמּוֹת, וְסֵרֵב עַל צִוּוּי בֵּית דִּין, הֲרֵי הוּא בְּחֶזְקַת שֶׁלֹּא נָתַן, כְּדִין גַּזְלָן. הָרֹא"שׁ: אֵין כוּ'. וְאַף עַל גַּב דְּמִתְּחִלָּה הִסְכִּים לְכָךְ, הַשְׁתָּא לֹא בָּעֵי. נִמּוּקֵי יוֹסֵף: וּבְכִי הָא לֹא אָמְרִינַן זֶה נֶהֱנֶה וְזֶה לֹא חָסֵר פָּטוּר. תּוֹסָפוֹת. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: שֶׁלֹּא כוּ'. אַיְרֵי כְּגוֹן דְּיַדְעִינַן דְּקָדִים חַד וְאַרְצֵיהּ לְחַבְרֵיהּ וְהָיָה מְסַרְהֵב, כִּדְמַשְׁמַע לִישָׁנָא אֵין מְחַיְּבִין אוֹתוֹ, וּלְכָךְ הֲוֵי בְּחֶזְקַת שֶׁלֹּא נָתַן. וְאֵין צָרִיךְ לְהַעֲמִיד כְּשֶׁעָשָׂה חָזִית. תּוֹסָפוֹת:

פרק א - משנה ה

כְּלוֹמַר לְמַס. דְּאִלּוּ לִצְדָקָה זְמַנִּים אֲחֵרִים יֵשׁ לֵחָשֵׁב כְּאַנְשֵׁי הָעִיר, כִּדְאִיתָא בַּגְּמָרָא:

פרק א - משנה ו

וּבְכָל שֻׁתָּפוּת קָאָמַר, בֵּין שֶׁזָּכוּ מִן הַהֶפְקֵר, בֵּין שֶׁלָּקְחוּ אוֹ יָרְשׁוּ. הָרְמָ"ה: וְלֹא כוּ'. מִשּׁוּם דִּסְבִירָא לֵיהּ דְּבִבְצִיר מֵהָכִי לֹא חָשִׁיב לְמִטְרָח עָלֶיהָ. נִמּוּקֵי יוֹסֵף: הָרַמְבַּ"ם בְּפֵרוּשׁוֹ. אֲבָל בְּחִבּוּרוֹ כָּתַב, דְּהַיְנוּ דַּוְקָא בְּבָבֶל. וְכֵן הוּא בַּגְּמָרָא, דְּתַנָּא דִּידָן בְּאֶרֶץ יִשְׂרָאֵל קָאֵי: כְּגוֹן פַּרְדֵּס אוֹ שָׂדֶה שֶׁמַּשְׁקִין אוֹתוֹ בִּכְלִי, שֶׁיִּהְיֶה כְּדֵי עֲבוֹדַת יוֹם אֶחָד לְכָל אֶחָד. הָרַמְבַּ"ם בְּחִבּוּרוֹ: וְגוֹד, מִלְּשׁוֹן קְצִיצָה. כְּמוֹ קֹץ דָּמִים. וְיִהְיֶה מֵעִנְיַן גֹּדּוּ אִילָנָא. רַשִׁ"י. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: אֲבָל כוּ'. לֹא הָדָר תָּנֵי לֵיהּ אֶלָּא מִשּׁוּם סֵיפָא וּבְכִתְבֵי הַקֹּדֶשׁ כוּ'. גְּמָרָא: וְכִתְבֵי כוּ'. עֶשְׂרִים וְאַרְבָּעָה סְפָרִים. וְהָיוּ רְגִילִין לְכָתְבָן בְּגִלָיוֹן כְּסֵפֶר תּוֹרָה שֶׁלָּנוּ. לְפִיכָךְ גְּנַאי הַדָּבָר לְחָתְכָן. רַשִׁ"י: וּכְשֶׁהֵם שְׁנֵי עִנְיָנִים, דְּכָל חַד וְחַד צָרִיךְ לְהַאי וּלְהַאי. וּלְפִיכָךְ רָצוּ אִין, לֹא רָצוּ לֹא. גְּמָרָא:
