פרק א

גְּמָרָא שְׁבוּעוֹת דַּף ו'. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: בַּהֶרֶת כוּ'. וְאֵין דֶּרֶךְ נֶגַע לָבֹא אֶלָּא בְּאַרְבָּעָה מַרְאוֹת הַלָּלוּ, וּמִיהוּ בְּכָל אֶחָד וְאֶחָד יֵשׁ כַּמָּה מַרְאוֹת שֶׁזֶּה לָבָן מִזֶּה כְּמוֹ שְׁלָגִים וְכֵן כֻּלָּם. הָרַ"שׁ וְתוֹסָפוֹת: וְעַיֵּן בְּהָרַ"ב לְקַמָּן מִשְׁנָה ג':

כְּיַיִן כוּ'. וְהָא דְלֹא שִׁעֲרוּ לְתַרְוַיְהוּ אוֹ בְיַיִן אוֹ בְדָם, וּמַאי טַעְמָא תָּנֵי בְסִיד כְּדָם בְּחָלָב וְלֹא אָמַר כְּיַיִן בְּסִיד. וְעוֹד, שִׁעוּר הַמִּזּוּג מַאי נִיהוּ אוֹ בִּשְׁנֵי חֲלָקִים שֶׁלֶג וְחָלָב אוֹ בְפָחוֹת. וּבָזֶה יֵשׁ לוֹמַר שֶׁהוּא בִכְדֵי שֶׁיֵּרָאֶה זֶה לְתוֹךְ זֶה כִּדְאָמְרִינַן בְּעָלְמָא מַיִם שֶׁדַּם הַצִּפּוֹר נִכָּר בָּהֶם. וְנִרְאֶה דְּרַבִּי יִשְׁמָעֵאל סָבַר בָּעִינַן שֶׁיִּהְיֶה נוֹדָע הַלַּבְנִינוּת וְנִכָּר בָּאַדְמִימוּת, הִלְכָּךְ אִלּוּ אָמַר כְּדָם בְּשֶׁלֶג אוֹ כְּדָם בְּסִיד, מִתּוֹךְ שֶׁשְּׁנֵיהֶם עָבִים אֵין נִמְזָגִים כְּלָל וְאֵין הַמַּרְאוֹת מִתְעָרְבִים. וְאִם אָמַר כְּיַיִן בְּסִיד, אַף עַל פִּי שֶׁהַיַּיִן קָלוּשׁ, הַסִּיד עָב, אֲבָל יַיִן בְּשֶׁלֶג נִמּוֹחַ וְאֵינוֹ נִמּוֹחַ וּשְׁנֵיהֶם נִרְאִים. וְאִם אָמַר יַיִן בְּחָלָב אוֹ בְמַיִם, מִפְּנֵי שֶׁשְּׁנֵיהֶם קְלוּשִׁים הַיַּיִן מַחֲזִירָם לְמַרְאֵהוּ, וְהַלֹּבֶן אֵינוֹ נִכָּר בְּטִבְעוֹ. וְרַבִּי עֲקִיבָא סְבִירָא לֵיהּ אֲדַמְדַּם אָדֹם שֶׁבָּאֲדֻמִּים, הִלְכָּךְ מְשַׁעֵר לָהֶן יַיִן בְּמַיִם כוּ'. הָרַאֲבַ"ד:

וְהַתּוֹעֶלֶת לְחַלֵּק אוֹתָן בְּמִסְפַּר אַרְבָּעָה, וְלֹא תָנֵי סְתָם כָּל מִקְּרוּם כוּ' מִצְטָרְפִין, מְלַמֵּד שֶׁאִם אֵינוֹ בָקִי בָהֶן וּבִשְׁמוֹתֵיהֶן לֹא יִרְאֶה אֶת הַנְּגָעִים. הָרַמְבַּ"ם בְּשֵׁם הַתּוֹסֶפְתָּא: לִפְטֹר כוּ'. נָקַט תְּחִלָּה שְׁנֵי הַקְּצָווֹת, וַהֲדַר מַקְדִּים לְהַסְגִּיר שֶׁהוּא בְשָׁבוּעַ אֶחָד, וְאַחַר כָּךְ לִפְטֹר שֶׁהוּא בְשָׁבוּעַ שֵׁנִי, וַהֲדַר לְהַחְלִיט שֶׁיֵּשׁ בּוֹ כַּמָּה גַוְנֵי: הָעוֹמֵד כוּ'. וְלֹא נָקַט נַמִּי בַּתְּחִלָּה, דְּרֵישָׁא הָכִי פֵרוּשָׁא, אַרְבָּעָה מַרְאוֹת כוּ' מִצְטָרְפִין בַּתְּחִלָּה וְכֵן לִפְטֹר כוּ', וְלֹא הֻצְרַךְ לְפָרֵשׁ צֵרוּף דְּרֵישָׁא דִּפְשִׁיטָא דִּסְתָמָא דְמִלְּתָא אַיְרֵי בְּבָא בַתְּחִלָּה. מַהֲרַ"ם: לִפְטֹר כוּ'. וְאִם תֹּאמַר, מַאי צֵרוּף שַׁיָּךְ כָּאן. נִרְאֶה לִי, כְּגוֹן שֶׁהִסְגִּיר בַּהֶרֶת בִּשְׁנֵי גְרִיסִין וּבְסוֹף שָׁבוּעַ שֵׁנִי יֵשׁ בַּהֶרֶת כַּגְּרִיס וְשְׂאֵת כַּגְּרִיס, וְהַשְׁתָּא אִי הֲווּ שְׁנֵי מִינִים הָוֵי לֵיהּ שְׂאֵת נֶגַע חָדָשׁ וּמַסְגִּירִין עָלָיו, אֲבָל אִם מִצְטָרְפִים דַּהֲוֵי כְנֶגַע אֶחָד הָוֵי לֵיהּ עָמַד בְּעֵינָיו דְּטָהוֹר. מַהֲרַ"ם. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: כָּל דִּבְרֵי הָרַב הֵם עַל פִּי הַתּוֹסֶפְתָּא נִלְמָדִין מִקְּרָאֵי. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב בַּאֲרִיכוּת: שֶׁכָּל כוּ'. לָאו דַּוְקָא, דְּהָא אִיכָּא נְתָקִים. אֶלָּא בְּעוֹר בָּשָׂר קָאָמַר. הָרַ"שׁ:

שִׁשָּׁה עָשָׂר. הָכִי גָּרַס הָרַ"שׁ, וּפֵרֵשׁ דִּסְבִירָא לֵיהּ כְּרַבִּי עֲקִיבָא דְּאָמַר כֻּלְּהוּ לְבַהֲדֵי שְׂאֵת מִצְטָרְפֵי, דִּשְׂאֵת מִצְטָרֶפֶת לְבַהֶרֶת וְסִיד וּקְרוּם, וּקְרוּם וְסִיד גַּם כֵּן מִצְטָרְפִין, הָרֵי אַרְבָּעָה צֵרוּפִים וְאַרְבָּעָה חֲלָקִים, וּפְתוּכִים כְּנֶגְדָּם, הֲרֵי שִׁשָּׁה עָשָׂר. עַד כָּאן. וּמִיהוּ לַמַּסְקָנָה לְרַבִּי עֲקִיבָא כֻּלָּן מִצְטָרְפִין. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: וּבְסֵפֶר קָרְבַּן אַהֲרֹן תֵּרֵץ, דְּהָרַמְבַּ"ם דִּקְדֵּק בִּלְשׁוֹנוֹ וְלֹא כָתַב וּשְׁמֹנָה בִּנְתָקִים, אֶלָּא וְהֵם עַצְמָם בִּנְתָקִים, לוֹמַר שֶׁהֵם הַיּוֹתֵר מְיֻחָדִים בָּהֶם אֲשֶׁר הֵם דּוֹמִים לְמַרְאוֹת הַנְּגָעִים, עִם הֱיוֹת שֶׁמְּטַמְּאִים גַּם בִּשְׁאָר מַרְאוֹת. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: וּבָרַמְבַּ"ם בְּנֻסְחַת אֶרֶץ יִשְׂרָאֵל וְכֵן בְּחִבּוּרוֹ פָּסַק כְּרַבִּי עֲקִיבָא. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב:

דְּאִי קָאֵי בְּסוֹף שָׁבוּעַ שֵׁנִי, פּוֹטְרוֹ. הָרַ"שׁ: נִסְמַךְ. סְבָרָא הוּא בְכָל הָנֵי דִתְנִינַן שֶׁמְּבַטְּלִין סִימָנֵי הַטֻּמְאָה: הַשְּׁחִין. וְקָשֶׁה, תֵּפּוּק לֵיהּ מִשּׁוּם שְׁחִין וּמִכְוָה שֶׁשֵּׂעָר לָבָן גַּם בָּהֶם סִימַן טֻמְאָה. וְיֵשׁ לוֹמַר, כְּגוֹן שֶׁהִיא מַכָּה טְרִיָּה שֶׁאֵינָהּ מְטַמְּאָה. כֶּסֶף מִשְׁנֶה. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: וְנִתְפַּזְּרָה. פֵּרוּשׁ, שֶׁפָּשָׂה הַנֶּגַע לְתוֹכָהּ וְכִסָּה מִקְצָתָהּ, וְאַכַּתִּי נִשְׁאַר שִׁעוּר מִחְיָה דְּהַיְנוּ כָּעֲדָשָׁה בְּהִצְטָרֵף לְכֻלַּהּ מִחְיָה. וּבָא, פֵּרוּשׁ, אוֹ שֶׁבָּא הַשְּׁחִין. מַהֲרַ"ם. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: וְשׁוּב לֹא מְטַמֵּא בְמִחְיָה. מַהֲרַ"ם: לָאו דַּוְקָא, דְּהוּא הַדִּין אַאוֹ נִכְנַס וְכוּ' קָאֵי, אֶלָּא דְּנָקַט בְּמַאי דִּסְלִיק וּסְמִיךְ: פִּסְיוֹן. מִשּׁוּם נִגְעֵי הַגּוּף נַקְטֵיהּ. וּבְנִגְעֵי בָתִּים אֲפִלּוּ עָמַד בְּעֵינָיו טָמֵא: וְאֵין כוּ'. אֲפִלּוּ יֵשׁ בָּהֶם כַּגְּרִיס וְעוֹמֵד בְּשָׁבוּעַ רִאשׁוֹן, נַמִּי הָוֵי לְהָקֵל, דַּהֲוֵי בְהֶסְגֵּר וְלֹא בְהֶחְלֵט, וּכְכָל הָנֵי דְּתָנֵי בְהֶדְיָא. אֶלָּא מִשּׁוּם דְּעַל כָּל פָּנִים אִצְטְרִיךְ לְמִתְנֵי וְנִתְמַעֲטָה, דְּאִם לֹא כֵן קָאֵי בְהֶחְלֵט, נָקַט וְאֵין כוּ' דְּפָטוּר לְגַמְרֵי. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב:

לֹא כוּ'. דְּאַף עַל גַּב דְּאִי נַמִּי פְטָרוֹ בְשַׁבָּת, כְּשֶׁנּוֹלַד לוֹ לְאַחַר הַשַּׁבָּת הָיָה מַחְלִיטוֹ, מִכָּל מָקוֹם הָיָה טָהוֹר אִם טָבַל וְהֶעֱרִיב שִׁמְשׁוֹ, עַד שֶׁיִּוָּלֵד לוֹ וְיַרְאֶה לַכֹּהֵן. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: וּכְלוֹמַר, שֶׁהַנֶּגַע הוּא בְעוֹר הַבָּשָׂר הַזֶּה שֶׁנַּעֲשָׂה צַלֶּקֶת. דְּאִי לָאו הָכִי עֲדַיִן אֵין הַשֵּׂעָר בְּתוֹךְ הַנֶּגַע:
