וְהָרַאֲבַ"ד כָּתַב, שֶׁרַבִּי אֱלִיעֶזֶר מוֹדֶה שֶׁאִם עָלָה עָלֶיהָ זָכָר שֶׁפְּסוּלָה. שֶׁהַמְעֻבֶּרֶת תּוֹבַעַת זָכָר הִיא, וּצְרִיכָה הִיא. אֲבָל עָלָה עָלֶיהָ שֶׁלֹּא בִשְׁעַת תְּבִיעָה, צַעַר הוּא לָהּ וּמַשָּׂא. וַחֲכָמִים פּוֹסְלִין מְעֻבֶּרֶת מִשּׁוּם חֲשָׁשׁ דְּשֶׁלֹּא לִרְצוֹנָהּ: וּפְסוּלָה אֲפִלּוּ לְמַאן דְּאָמַר קָדְשֵׁי בֶדֶק הַבַּיִת הִיא. הָרַ"שׁ:

יָגֹד. כְּשֶׁהַשַּׁחֲרוּת אֵינוֹ מַגִּיעַ לַזַּכְרוּת , דְּאִי מַגִּיעַ יַעֲשֶׂנָּה בַעַל מוּם אִם יַחְתֹּךְ שָׁם, גְּמָרָא: שֶׁמַּאֲמַר הָאֵל, תְּמִימָה, הוֹדִיעָנוּ שֶׁעִנְיָנוֹ שְׁלֵמוּת הָאֹדֶם. הָרַמְבַּ"ם. וְעַיֵּן בְּהָרַ"ב מִשְׁנָה ה':

לַבָּיִת. וְרַבָּנָן, כֵּיוָן דְּקָדְשֵׁי בֶדֶק הַבַּיִת הִיא וּלְקִיחָתָהּ מִתְּרוּמַת הַלִּשְׁכָּה, בָּאָה לַבַּיִת הִיא. כָּךְ נִרְאֶה לִי: אִם כֵּן מַה תַּלְמוּד לוֹמַר עֹל, עֹל פּוֹסֵל בֵּין בִּשְׁעַת עֲבוֹדָה וּבֵין שֶׁלֹּא בִשְׁעַת עֲבוֹדָה, שְׁאָר עֲבוֹדוֹת אֵין פּוֹסְלוֹת אֶלָּא בִשְׁעַת עֲבוֹדָה, כְּלוֹמַר שֶׁעֲבוֹדָה זוֹ לְצֹרֶךְ הִיא וְלֹא בְאַקְרַאי כוּ'. גְּמָרָא:

פְּסוּלָה. דְּנִיחָא לֵיהּ וְהַוְיָא שְׁעַת עֲבוֹדָה. אִי נַמִּי, מַה שֶּׁהַזָּכָר מְשַׁמֵּשׁ בָּהּ חֲשִׁיב עֲבוֹדָה. תּוֹסָפוֹת:

אֲפִלּוּ אַרְבַּע כוּ'. וְלָאו דַּוְקָא. וְאֶלָּא מִיהַת הַרְבֵּה מְאֹד לֹא סְבִירָא לֵיהּ לְרַבִּי עֲקִיבָא. וּפָלֵיג רַבִּי אֱלִיעֶזֶר וְאָמַר אֲפִלּוּ חֲמִשִּׁים: יִתְלֹשׁ. מִפְּנֵי מַרְאִית הָעַיִן. כֶּסֶף מִשְׁנֶה: שְׁתֵּי כוּ'. וְהֵן בְּתוֹךְ גֻּמָּא אֶחָת. הָרַאֲבַ"ד: דְּקָדְשֵׁי בֶדֶק הַבַּיִת לֹא מִתַּסְרֵי בְגִזָּה אֶלָּא מִדְּרַבָּנָן, וּבַפָּרָה לֹא גַזּוּר מִשּׁוּם דְּלֹא שְׁכִיחָא. אִי נַמִּי, דַּהֲוֵי דָבָר שֶׁאֵין מִתְכַּוֵּן. גְּמָרָא:
