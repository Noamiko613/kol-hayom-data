בָּתִּים כוּ' טְהוֹרָה. וַאֲפִלּוּ הֵן טְמֵאִים. הָרַמְבַּ"ם: וְנִרְאֶה לִי דְּמַיְרֵי שֶׁלֹּא בָא בְתוֹכָם לִטְבֹּל. דְּלֹא תִקְשֶׁה מִמַּתְנִיתִין דְּרֵישׁ פֶּרֶק ה':

פרק ב - משנה ב

הָרַ"שׁ. וְקָשֶׁה לִי, דְּאַף עַל גַּב דִּשְׁאוּבִים הֵם, לֹא מִפְּנֵי כֵן הֵן מְטַמְּאִים אַף עַל פִּי שֶׁמַּכְשִׁירִים. וְצָרִיךְ לוֹמַר דִּשְׁאוּבִים וּטְמֵאִים הֵם. וְהָרַמְבַּ"ם מְפָרֵשׁ, דְּמֶרְחָץ שֶׁזֵּעָתוֹ הוּא מִן הַמַּיִם שֶׁבְּתוֹכָהּ וְאִם הָיְתָה הַמֶּרְחָץ טְמֵאָה זֵעָתָהּ טְמֵאָה כוּ':

פרק ב - משנה ג

טָמֵא. שֶׁמַּשְׁקִין טְמֵאִין שֶׁיָּרְדוּ לִטְהוֹרִים, מְטַמְּאִים בְּכָל שֶׁהֵן. הָרַמְבַּ"ם. וְקָשֶׁה, דְּאַדְּרַבָּה הֲוָה לֵיהּ לְמֵימַר קַמָּא קַמָּא בָטֵיל. כוּ'. כֶּסֶף מִשְׁנֶה. וּלְשׁוֹן הָרַאֲבַ"ד, טָמֵא, וְאוּלַי מִפְּנֵי שֶׁמִּשָּׁעָה שֶׁנָּפְלוּ בַתְּחִלָּה, הֻחְזְקוּ גַם הֵם בִּשְׁפִיכוּת שֶׁכְּבָר נִשְׁכְּחָה נְפִילָתָם. וְכָתַב הַכֶּסֶף מִשְׁנֶה, וְאִם תֹּאמַר, לְפִי זֶה אֲפִלּוּ לֹא נָפְלוּ מֵי שְׁפִיכוּת נַמִּי. וְיֵשׁ לוֹמַר, דְּלֹא גַזּוּר אֶלָּא כְשֶׁיֵּשׁ שָׁם מֵי שְׁפִיכוּת:

פרק ב - משנה ו

קָרוֹב. כְּדֵי שֶׁיֵּלֵךְ הַנָּכְרִי וְיִתְלֹשׁ וְיִגְמֹר הַמְּלָאכָה וְיַחֲזֹר לְכָאן. וְשִׁעוּר הַתְּלִישָׁה אֵין צָרִיךְ לְהַזְכִּיר, שֶׁהִיא נַעֲשֵׂית בְּשָׁעָה מֻעֶטֶת. הָרַ"ן: וּבְפֶרֶק כ"ג דְּשַׁבָּת מִשְׁנָה ד', אֶלָּא אִם כֵּן בָּאוּ מִמָּקוֹם קָרוֹב, פֵּרֵשׁ הָרַ"ב אֶלָּא אִם כֵּן נוֹדַע בְּבֵרוּר שֶׁבָּאוּ מִתּוֹךְ הַתְּחוּם, כָּתְבוּ הַתּוֹסָפוֹת, דְּמֵת שָׁאנֵי דְּאָוְשִׁי מִלְּתָא. וּלְמַה שֶּׁכָּתַב הָרַ"ן כוּ', יֵשׁ לוֹמַר דְּמָקוֹם קָרוֹב דְּהָכָא אֵינוֹ כְמָקוֹם קָרוֹב דְּהָתָם, אֶלָּא לַאֲפוֹקֵי מָקוֹם רָחוֹק, וּלְעוֹלָם בְּחוּץ לַתְּחוּם, וְצָרִיךְ לְהַמְתִּין עַד שֶׁיָּבֹאוּ מִמֶּנּוּ, אֲבָל אֵין חוֹשְׁשִׁין לַמָּקוֹם הַיּוֹתֵר רָחוֹק מִזֶּה. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: פֵּרוּשׁ, לֵךְ וּלְקֹט:

פרק ב - משנה ז

וְכָל שֶׁכֵּן לְעִנְיַן אֲבֵדָה וְלִנְזָקִין. רַשִׁ"י. וּבַגְּמָרָא, דִּלְעִנְיַן פִּקּוּחַ נֶפֶשׁ אֵינוֹ כֵן: וְאַף עַל גַּב דְּקַיְמָא לָן אֵין הוֹלְכִין בְּמָמוֹן אַחַר הָרֹב, הָנֵי מִלֵּי כְּשֶׁבָּא לְיָדוֹ הַמָּמוֹן בְּהֶתֵּר וּמִדַּעַת נְתָנָם לוֹ. הָרַ"שׁ וְתוֹסָפוֹת. וּלְיוֹחֲסִין לֹא מָצֵי לְאוֹקְמֵי, דְּבָעִינַן תְּרֵי רֻבָּא: דִּלְהַחֲזִיר אֲבֵדָה אוֹמֵר לוֹ אַיְתֵּי רְאָיָה דְיִשְׂרָאֵל אַתְּ. תּוֹסָפוֹת: תֵּימַהּ, דַּהֲוָה מָצֵי לְמִתְנֵי מֶחֱצָה עַל מֶחֱצָה נָכְרִים, וּמְנַגַּח תּוֹרָא דִידָן לְתוֹרָא דִידֵיהּ. תּוֹסָפוֹת:

פרק ב - משנה ח

עִסָּה. נִרְאֶה לִי, דְּפַת עִסָּה הָיָה מִנְהָגָם דְּבַעֲלֵי בָתִּים אוֹפִין בְּבָתֵּיהֶם יוֹתֵר מִן הַנַּחְתּוֹמִין, וּלְפִיכָךְ אֵין הוֹלְכִין בּוֹ אַחַר רֹב נַחְתּוֹמִין. וְכֵן בְּפַת קִבָּר לְרַבִּי יְהוּדָה. וְנִרְאֶה לִי, דְּתַנָּא קַמָּא וְרַבִּי יְהוּדָה לֹא פְלִיגֵי, דְּמַר כִּי אַתְרֵיהּ כוּ', דִּבְאַתְרֵיהּ דְּתַנָּא קַמָּא הַנַּחְתּוֹמִים אוֹפִים פַּת קִבָּר, וּבְאַתְרֵיהּ דְּרַבִּי יְהוּדָה בַּעֲלֵי בָתִּים אוֹפִין פַּת קִבָּר. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב:

פרק ב - משנה ט

וְזֶה דְּלֹא כִדְמַשְׁמַע מִלְּשׁוֹנוֹ בְּפֶרֶק ז' דִּשְׁקָלִים מִשְׁנָה ג':

פרק ב - משנה י

דַּעֲדַיִן לֹא רָאָה פְנֵי הַבַּיִת וְלֹא נִתְחַיֵּב. הָרַ"שׁ: הָרַמְבַּ"ם. וְצָרִיךְ עִיּוּן, דְּמִשּׁוּם דִּרְגִילִין לְעַשֵּׂר יְהֵא בְחֶזְקַת מְעֻשָּׂר וְיֵצֵא מִכְּלַל דְּמַאי. וְדֹחַק לוֹמַר דְּמַיְרֵי לְעִנְיַן אַכְסַנְיָא וְעָנִי. וּבְחִבּוּרוֹ כָתַב, פָּטוּר מִלְּעַשֵּׂר, שֶׁעֲדַיִן לֹא נִקְבַּע לְמַעֲשֵׂר: זֶהוּ שִׁיטַת הַיְרוּשַׁלְמִי בִּלְבַד. וְהַיְנוּ דְסִיֵּם הָרַ"שׁ, וַאֲפִלּוּ יֵשׁ קִנְיָן מִכָּל מָקוֹם וַדַּאי אֵינוֹ מְעֻשָּׂר הוּא: אֲבָל מַעֲשֵׂר רִאשׁוֹן, הַמּוֹצִיא מֵחֲבֵרוֹ עָלָיו הָרְאָיָה. אֲבָל נִרְאֶה דְּהָכָא אֲנַן אַלִּיבָּא דְרַבִּי מֵאִיר קַיָּמִים דִּסְבִירָא לֵיהּ דְּאַף מַעֲשֵׂר רִאשׁוֹן אָסוּר לְזָרִים, וְצָרִיךְ לְהַפְרִישׁ אַף מַעֲשֵׂר רִאשׁוֹן. אִי נַמִּי אַף לְרַבִּי מֵאִיר לֹא הֶחְמִירוּ בִדְמַאי, מִשּׁוּם דְּרֹב עַמֵּי הָאָרֶץ מְעַשְּׂרִין. וְהַשְׁתָּא נִיחָא דְּלֹא פֵרֵשׁ הָרַ"ב בְּרֵישָׁא מֶחֱצָה לְמֶחֱצָה דְּמַאי שֶׁצָּרִיךְ לְהַפְרִישׁ, אֶלָּא הָכָא אִצְטְרִיךְ לְאַשְׁמְעִינַן דְּאַף לְרַבִּי מֵאִיר אֵין צָרִיךְ לְהַפְרִישׁ מַעֲשֵׂר רִאשׁוֹן. נִרְאֶה לִי: דְּמַאי. וְלֹא יָדַעְנָא טַעֲמָא מַאי, דַּאֲפִלּוּ וַדַּאי תְרוּמָה עוֹלָה בְּמֵאָה וְאֶחָד, וּתְרוּמַת מַעֲשֵׂר אֵינָהּ אֶלָּא אֶחָד מִמֵּאָה, נִמְצָא כְּשֶׁיֵּשׁ עוֹד אֲפִלּוּ מִנָּכְרִי אֶחָד, נִתְבַּטְּלָה. וַאֲפִלּוּ לְהָרִים אֵינוֹ צָרִיךְ. דְּאֵין כָּאן גֶּזֶל הַשֵּׁבֶט, דְּסָפֵק הוּא וְהַמּוֹצִיא מֵחֲבֵרוֹ עָלָיו הָרְאָיָה:

פרק ב - משנה יא

רְבִיעִית עַל שֶׁל חֲמִישִׁית. בִּכְדִי נָקְטוּ, דְּבִשְׁתֵּיהֶן מַעֲשֵׂר שֵׁנִי. וּמַהֲרַ"ם כָּתַב, לְעִנְיַן מְעַשֵּׂר מֵחָדָשׁ עַל הַיָּשָׁן אוֹ לְהֵפֶךְ. וַהֲוָה מָצֵי לְמֵימַר נַמִּי מִן הָרִאשׁוֹנָה עַל הַשְּׁנִיָּה, אֶלָּא דְּהַךְ רְבִיעִית עַל חֲמִישִׁית אַגַּב גְּרָרָא לֹא רָצָה לִדְלֹג:
