פרק ה

פרק ה - משנה א

וְיֵשׁ שֶׁעוֹשִׂין עַל בָּסִיס. וְכִדְפֵרֵשׁ הָרַ"ב בְּרֵישׁ פֶּרֶק ב' דְּבָבָא בַתְרָא. וְעַיֵּן לְקַמָּן רֵישׁ פֶּרֶק ז': וְצָרִיךְ לוֹמַר שֶׁהַקָּטָן מְשֻׁנֶּה בַעֲשִׂיָּתוֹ יוֹתֵר מִן הַגָּדוֹל, אוֹ בְגֹבַהּ אוֹ בְרֹחַב אוֹ בִכְתָלִים דַּקִּים. דְּאִי לֹא תֵימָא הָכִי, גָּדוֹל כְּשֶׁתְּחִלָּתוֹ אַרְבָּעָה כְשֶׁהוּא פָחוֹת מֵאַרְבָּעָה יְהֵא דִין קָטָן עָלָיו, דִּתְחִלָּתוֹ כָל שֶׁהוּא. הָרַ"שׁ: וְצֵרוּף כִּבְשָׁן לֹא בָעֵי. הָרַ"שׁ. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב:

פרק ה - משנה ב

וְהוּא הַדִּין לְעִנְיַן שִׁעוּר גֹּבַהּ, וַחֲדָא מִתַּרְתֵּי נָקַט. וְכֵן הוּא בְהָרַמְבַּ"ם: זֶה לָזֶה. וְאֶבֶן אַחַת מְחַבְּרָן. וְנִרְאֶה לִי דַאֲתָאָן לְתַנָּא קַמָּא. מַהֲרַ"ם. וְהַתּוֹסָפוֹת חֲדָשִׁים כָּתַב דְּאָתֵי כְרַבִּי יְהוּדָה, דְּאִי לְתַנָּא קַמָּא מַאי קָא מַשְׁמַע לָן, דְּהָא אֲפִלּוּ בְתַנּוּר אֶחָד יוֹתֵר מִטֶּפַח טָהוֹר, אִם כֵּן מִכָּל שֶׁכֵּן בֵּין שְׁנֵי תַנּוּרִים, אֶלָּא רַבִּי יְהוּדָה הָא קָא מַשְׁמַע לָן דַּהֲוֵי כְבֵין תַּנּוּר לַכֹּתֶל:

פרק ה - משנה ג

אֲבָל בַּאֲוִיר עַצְמוֹ מִטַּמֵּא הוּא עַצְמוֹ, דִּכְלִי חֶרֶס הוּא. הָרַ"שׁ:

פרק ה - משנה ד

טַעֲמוֹ כִדְכָתַב הָרַ"שׁ, דְּקָשֶׁה דְּאִם כֵּן הֲוֵי מַעֲשֶׂה לִסְתֹּר. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב:

פרק ה - משנה ו

דְּשׁוּלָיו קְרוֹבִין לְשׁוּלֵי הַבּוֹר שֶׁאִם מַסִּיקוֹ כוּ' וְדָפְנוֹתָיו מִלְּמַעְלָה קְרוֹבִים לְדָפְנֵי הַבּוֹר שֶׁלֹּא הֻצְרַךְ לָתֵת בֵּינְתַיִם אֶלָּא אֶבֶן דַּקָּה, וְאֵין אֲוִיר נִכָּר בֵּין דֹּפֶן הַתַּנּוּר לְדֹפֶן הַבּוֹר שֶׁיְּעַכֵּב אֶת הֶסֵּק הַתַּנּוּר שֶׁלֹּא יוֹעִילוּ לוֹ דָפְנֵי הַבּוֹר לְהַחְזִיק אֶת חֻמּוֹ, אוֹ שֶׁהַתַּנּוּר בְּתוֹךְ חֲלַל הַבּוֹר בָּאֲוִיר וְהַהֶסֵּק עוֹלֶה סְבִיב דָּפְנוֹתָיו וְנִסֹּק, כְּגוֹן פִּיו קָצָר מִלְּמַטָּה וְרָחָב מִלְּמַעְלָה, אוֹ הַבּוֹר רָחָב מִלְּמַטָּה וּפִיו קָצָר מִלְּמַעְלָה, טָמֵא. וְאִם לָאו אֶלָּא נִסֹּק כִּשְׁאָר תַּנּוּרִים מִתּוֹכוֹ, טָהוֹר. הָרַ"שׁ. וּמַה שֶּׁכָּתַב דֶּרֶךְ שׁוּלָיו, רוֹצֶה לוֹמַר דֶּרֶךְ מְקוֹם שׁוּלָיו עוֹלֶה לוֹ הָאוּר לְתוֹכוֹ. דְּהָא תַנּוּרִים שֶׁלָּהֶם אֵין לָהֶם שׁוּלַיִם: לֹא שֶׁצָּרִיךְ הֶסֵּק שֵׁנִי, אֶלָּא כְלוֹמַר לְאַחַר הֶסֵּק רִאשׁוֹן:

פרק ה - משנה ז

לִשְׁלֹשָׁה. דְּעַבְדֵיהּ גִּסְטְרָא, שֶׁחִלְּקוֹ לְאָרְכּוֹ בְגָבְהוֹ, וּמִשּׁוּם הָכִי לֹא סַגִּי לְהוּ לְרַבָּנָן כְּשֶׁמְּמַעֲטוֹ בְּפָחוֹת מֵאַרְבָּעָה כְּמוֹ לְרַבִּי מֵאִיר, אֶלָּא בָעִינַן דְּלֹא לֶהֱוֵי רֻבָּא, דְּלֹא דָמֵי לִשְׁיָרָיו אַרְבָּעָה דְרֵישׁ פִּרְקִין, דְּהָתָם דְּצַלְקֵיהּ מַצְלִיק לְרָחְבּוֹ (פֵּרוּשׁ שֶׁצּוֹלְחוֹ וּמְבַקְּעוֹ). גְּמָרָא: דִּבְתַנּוּר שֶׁהֻסַּק קָיְמִינַן, דְּהָא תְנַן שֶׁנִּטְמָא, וְכִי הֻסַּק יֵשׁ בּוֹ גֻמּוֹת וְאִי אֶפְשָׁר לְצַמְצֵם. הָרַ"שׁ: וְגוֹרֵר. נִרְאֶה לִי שֶׁהַגִּרְסָא וְגוֹרֵד בְּדָלֵ"ת. וְכֵן לְקַמָּן לִגְרֹר צָרִיךְ לוֹמַר לִגְרֹד. כְּמוֹ וַיִּקַּח לוֹ חֶרֶשׂ לְהִתְגָּרֵד בּוֹ . וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: וְנִיחָא הַשְׁתָּא הָא דְפָלֵיג רַבִּי מֵאִיר דְּאֵין צָרִיךְ לִגְרֹר כוּ' וְלֹא לַחֲלֹק: אֶלָּא כוּ'. דְּלֹא חָיֵישׁ לְחָלֵים, פֵּרוּשׁ, שֶׁהוּא חָזָק וְעוֹמֵד. גְּמָרָא:

פרק ה - משנה ח

הִרְחִיק כוּ'. פֵּרוּשׁ, זֶה הַתַּנּוּר שֶׁנַּעֲשָׂה מֵחֻלְּיוֹת וּמֵרְחוֹ בְטִיט. וְקָאָמַר הַשְׁתָּא שֶׁאִם עָשָׂה זֶה הַמֵּרוּחַ טְפֵלָה מְרֻחֶקֶת כוּ'. וְכֵן פֵּרֵשׁ הָרַמְבַּ"ם: חֹל כוּ'. קָתָנֵי חֹל דּוּמְיָא דִּצְרוֹר שֶׁהוּא יָבֵשׁ. תּוֹסָפוֹת:

פרק ה - משנה ט

וְלִמּוּדִין, כְּמוֹ עַמּוּדִין, עַל שֵׁם שֶׁהֵם מַעֲמִידִין הַדָּבָר. הָרַאֲבַ"ד:

פרק ה - משנה י

פָּחוֹת מֵאַרְבָּעָה טְפָחִים. הָרַ"שׁ: וְלֹא הִרְחִיק אֶת הַטְּפֵלָה. הָרַ"שׁ: וְהִסְכִּימוּ עַל הַדִּין שֶׁנִּזְכָּר בְּכָאן. לָזֶה יִקָּרֵא זֹאת הַתְּמוּנָה תַנּוּרוֹ שֶׁל בֶּן דִּינַאי, מְיֻחָס אֶל דִּין. הָרַמְבַּ"ם:

פרק ה - משנה יא

כַּאֲשֶׁר נָגְעָה בוֹ כַזַּיִת מִן הַמֵּת, לְפִי מַה שֶּׁהִתְבָּאֵר בַּתּוֹסֶפְתָּא. הָרַמְבַּ"ם: אֲבָל עָשָׂה טְפֵלָה לְשָׁלֵם, טָהוֹר מִדִּין כְּלִי חֶרֶס. הָרַ"שׁ: אִם כֵּן מְשֻׁנֶּה כִירָה מֵהַתַּנּוּר, שֶׁהַתַּנּוּר אֵין לוֹ שׁוּלַיִם כִּדְפֵרֵשׁ הָרַ"ב לְעֵיל מִשְׁנָה ז'. וְהָרַ"שׁ כָּתַב דְּגָרְסִינַן וַעֲשָׂאָהּ בְּטִיט וְעָשָׂה לָהּ פִּטְפּוּטִין, וְהָכִי פֵרוּשׁוֹ, עָשָׂה לָהּ בְּטִיט לְתַקְּנָהּ, וְעָשָׂה לוֹ פִטְפּוּטִין לְאוֹתוֹ הַטִּיט. וּלְשׁוֹן טְפֵלָה שַׁיָּךְ בְּתַנּוּר, וּלְשׁוֹן עֲשָׂאָהּ בְּטִיט שַׁיָּךְ בְּכִירָה. וְתַרְוַיְהוּ לְתִקּוּן הַתַּנּוּר וְהַכִּירָה. וְהַשְׁתָּא כְשֶׁנִּקְּבָה וַעֲשָׂאָהּ בְּטִיט וְעָשָׂה לָהּ פִּטְפּוּטִין, טְמֵאָה נַמִּי מִשּׁוּם כְּלִי חֶרֶס, וְהַפִּטְפּוּטִין מִטַּמְּאִין מֵאֲוִיר. אֲבָל טִיט וּפִטְפּוּטִין בְּלֹא נִקַּב, טְהוֹרָה. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: אוּלָם מֵרוּחַ הַתַּנּוּר שֶׁל מַתָּכוֹת וְשֶׁל אֶבֶן וְהִיא הַטְּפֵלָה, הִנֵּה יֵשׁ תּוֹעֶלֶת לָהּ בּוֹ לְפִי שֶׁהֵן יֹאפוּ מִבִּפְנִים הַלֶּחֶם וְזֶה הַמֵּרוּחַ יַשְׁאִיר חֻמּוֹ וְיִמְנַע מִמֶּנּוּ הָאֲוִיר אֲשֶׁר יְקָרֵר גֶּשֶׁם הַתַּנּוּר. הָרַמְבַּ"ם. וּלְגִרְסַת הָרַ"שׁ דִּבְסָמוּךְ, מוֹעִיל נַמִּי לַכִּירָה. וְכָתַב הָרַ"שׁ, דְּמֵרְחָהּ דְּהָכָא לָאו אַנִּקְּבָה קָאֵי, וּפְלוּגְתָּתָן קָאֵי גַּם אַתַּנּוּר, וְנָטַר לָהּ עַד לְבַסּוֹף, וְשֶׁכֵּן הוּא בַתּוֹסֶפְתָּא:
