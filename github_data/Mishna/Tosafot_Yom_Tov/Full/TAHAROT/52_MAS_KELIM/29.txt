פרק כט

וְכֵן הוּא בָרַ"שׁ. וּלְפִי דְטֻמְאַת שֶׁרֶץ אַף בְּשָׁלֹשׁ עַל שָׁלֹשׁ, לְהָכִי נְקָטוֹ. וְהוּא הַדִּין לְמִדְרָס בִּשְׁלֹשָׁה עַל שְׁלֹשָׁה. וְהָרַמְבַּ"ם סָתַם וּפֵרֵשׁ שֶׁהַבֶּגֶד כַּאֲשֶׁר נִטְמָא. וְגַם הָרַ"ב בַּמִּשְׁנָה דִּלְקַמָּן סָתַם וְכָתַב וְאִם נָגְעָה טֻמְאָה כוּ': וְקָשֶׁה, הָא פִלְיוֹן שֶׁל רֹאשׁ תְּנַן. וְהָרַ"שׁ כָּתַב, פִּלְיוֹן כְּמוֹ אַפִּלְיוֹן, בְּפֶרֶק כָּל כִּתְבֵי תָנֵי לֵיהּ גַּבֵּי שְׁמֹנָה עָשָׂר כֵּלִים. עַד כָּאן. וּמִזֶּה יָצָא לוֹ לְהָרַ"ב לְפִי שֶׁמָּצָא שָׁם שֶׁפֵּרֵשׁ רַשִׁ"י בְּדַף ק"כ פִּלְיוֹן סָדִין לְהִתְעַטֵּף בּוֹ כֻלּוֹ. אַךְ בְּכָאן אִי אֶפְשָׁר לְפָרֵשׁ כֵּן. וְעוֹד דְּהָתָם בְּמִשְׁנָה ד' מָנָה הָרַ"ב שְׁמֹנָה עָשָׂר כֵּלִים, וְתִמְצָא שָׁם שֶׁבִּמְקוֹם פִּלְיוֹן וּמַעֲפֹרֶת שֶׁבַּגְּמָרָא, כָּתַב מִצְנֶפֶת וְטַלִּית קָטָן שֶׁמְּכַסִּין בּוֹ רֹאשׁוֹ. וּבְמִשְׁנָה ב' פֶּרֶק קַמָּא דְשַׁבָּת כָּתַב הָרַ"ב, מַעֲפֹרֶת שֶׁל סַפָּרִים סוּדָר שֶׁנּוֹתֵן הַמִּתְגַּלֵּחַ עַל בִּרְכָּיו כוּ', וְנִשְׁאָר אִם כֵּן מִצְנֶפֶת בִּמְקוֹם פִּלְיוֹן. וְכַיּוֹצֵא בָזֶה פֵּרֵשׁ הָרַמְבַּ"ם כָּאן, שֶׁכָּתַב פִּלְיוֹן שֶׁל רֹאשׁ הוּא הַמִּטְפַּחַת שֶׁיָּשִׂימוּ הַנָּשִׁים עַל רֹאשָׁן. עַד כָּאן. וּבְאוֹתָן שְׁמֹנָה עָשָׂר כֵּלִים חֲשִׁיב מִצְנֶפֶת: כִּפָּה כוּ'. לְשׁוֹן הָרַמְבַּ"ם, כָּךְ שְׁמוֹ כִּפָּה שֶׁל זְקֵנָה, וְלֹא שֶׁיֶּשׁ אַחֵר שֶׁל יַלְדָּה. וּמִשּׁוּם הָכִי כָתַב בְּחִבּוּרוֹ כִּפָּה שֶׁל רֹאשׁ:

תֵּימַהּ לְפָרֵשׁ כֵּן, דְּהָא תְנַן בְּסֵיפָא יָתֵר מִכָּאן חִבּוּר לַטֻּמְאָה וְאֵין חִבּוּר לַהַזָּאָה. וְאֵין לוֹמַר דְּסֵיפָא לֹא קָאֵי אֶלָּא מֵחָלוּק אֶחָד וּלְהַלָּן, דְּאִם כֵּן בְּרֵישָׁא לֹא תָנֵי וְלֹא מִידִי אֶלָּא מוֹנֶה וְהוֹלֵךְ, וְזֶה דָבָר שֶׁאֵין לוֹ שַׁחַר. וְכֵן מְפֹרָשׁ בְּהָרַמְבַּ"ם דְּיָתֵר מִכָּאן דְּסֵיפָא קָאֵי אַכֻּלְּהוּ. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: חָלוּק כוּ'. אֲפִלּוּ הֵן אֲרֻכִּין בְּיוֹתֵר אוֹ רְחָבִין בְּיוֹתֵר כָּל שֶׁהֵן. הָרַמְבַּ"ם:

וְנִרְאֶה לִי דְתַנָּא מִסְּרַךְ סְרִיךְ בְּלִישְׁנָא דְמַתְנִיתִין דְּסוֹף פִּרְקִין:

שֶׁל בַּעֲלֵי בָתִּים. פֵּרוּשׁ, אוֹ שֶׁל בַּעֲלֵי בָתִּים: שְׁיָרֵי יָד. נִרְאֶה לִי שֶׁכֵּן רְגִילִין לִקְרוֹת שְׁיָרֵי יָד וּכְאִינָךְ דְּמִשְׁנָה ז' ח' דְּקָרֵי לְהוּ נַמִּי בִלְשׁוֹן שְׁיָרֵי. וְהָרַמְבַּ"ם כָּתַב, שְׁיָרֵי הַפַּרְגּוֹל. וּמַהֲרַ"ם כָּתַב יַד הַפַּרְגּוֹל שֶׁנִּטַּל וְנִשְׁתַּיֵּר בּוֹ כְשִׁעוּר טֶפַח הָוֵי חִבּוּר:

עַל תְּמוּנַת קוֹפִיץ. הָרַמְבַּ"ם:

וּמְפָרֵשׁ שָׁם (הָרַ"ן) בְּשֵׁם הָרַאֲבַ"ד, כְּלוֹמַר בּוֹרוֹת קְטַנִּים. וְשֶׁצָּרִיךְ לוֹמַר בְּרֵי"שׁ. וְכֵן יַד בָּדִיד דְּהָכָא: וּבְפֵרוּשׁ הָרַמְבַּ"ם בְּנֻסְחַת אֶרֶץ יִשְׂרָאֵל מָצָאתִי, נִכּוּשׁ הוּא אֲשֶׁר מְנַקֶּה עִקָּרֵי הָאִילָנוֹת, וּפַטִּישׁ הוּא עָגֹל אֲשֶׁר יְשַׁבְּרוּ בוֹ אֲבָנִים. וְזֶה נִרְאֶה לְפִי שֶׁנִּמְצָא כַּמָּה פְעָמִים בַּמִּשְׁנָה וּגְמָרָא לְשׁוֹן נִכּוּשׁ וְהוּא עִנְיַן נְקִירָה וַחֲפִירָה. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב:

בֵּית שַׁמַּאי כוּ'. לְעִנְיַן הַזָּיָה אִתְשִׁיל בְּבֵית הַמִּדְרָשׁ, דַּהֲווּ בֵית הִלֵּל לְקֻלָּא, וּמִשּׁוּם הָכִי לֹא תָנֵי לְהוּ בְעֵדֻיּוֹת בַּהֲדֵי קֻלֵּי בֵּית שַׁמַּאי. וְכָהֵנָּה רַבּוֹת: כָּל שֶׁהוּא. אֲפִלּוּ הַבֵּית יָד הוּא בְתַכְלִית הָאֲרִיכוּת, יִטַּמְּאוּ, כִּי הָאָדָם צָרִיךְ לָהֶן לְהִתְרַחֵק בּוֹ מִן הָאֵשׁ כוּ'. הָרַמְבַּ"ם. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב:
