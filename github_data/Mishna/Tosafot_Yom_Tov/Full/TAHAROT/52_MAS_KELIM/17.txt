פרק יז

פרק יז - משנה א

בְּרִמּוֹנִים. בַּעֲלֵי בָתִּים חָסִין עַל כְּלֵיהֶם, הִלְכָּךְ, נִקַּב כְּמוֹצִיא קִטְנִית, מַצְנִיעוֹ לְזֵיתִים. נִקַּב כְּמוֹצִיא זַיִת, מִשְׁתַּמֵּשׁ לֶאֱגוֹזִים. נִקַּב , מִשְׁתַּמֵּשׁ בּוֹ רִמּוֹנִים. אֲבָל שֶׁל אֻמָּן הָעוֹמֵד לִמָּכֵר, לֹא הָוֵי שִׁעוּרֵיהּ בְּהָכִי. רַשִׁ"י: אִם הָיוּ טְמֵאִים נִטְהֲרוּ בִנְקִיבָה זוֹ, וְאִם הֵם טְהוֹרִים אֵינָם מְקַבְּלִים טֻמְאָה. הָרַמְבַּ"ם:

פרק יז - משנה ב

שְׁתִי. יֵשׁ לִתְמֹהַּ לְמַאי נַפְקָא מִנַּהּ, דְּהָא מַסִּיק שֶׁאִם מְקַבֵּל שֶׁל עֵרֶב טְמֵאָה. וּמַהֲרַ"ם מְפָרֵשׁ, דְּכֻלַּהּ הָא בָּבָא דְרַבָּן גַּמְלִיאֵל הִיא, וְהָכִי פֵּרוּשָׁהּ, הַחֵמֶת שִׁעוּרָהּ לִטָּהֵר בִּפְקַעַת שֶׁל שְׁתִי, וְאִם אֵינָהּ מְקַבֶּלֶת שֶׁל שְׁתִי אַף עַל פִּי שֶׁמְּקַבֶּלֶת שֶׁל עֵרֶב טְמֵאָה לְרַבָּנָן. וְכֵן בֵּית קְעָרוֹת כוּ' וּבֵית הָרְאִי כוּ' טָמֵא לְרַבָּנָן. אֲפִלּוּ הָכִי רַבָּן גַּמְלִיאֵל מְטַהֵר בְּכֻלָּן מִפְּנֵי כוּ'. עַיֵּן תּוֹסְפוֹת יוֹם טוֹב: טְמֵאָה. עַד שֶׁתִּנָּקֵב רֻבָּהּ. הָרַמְבַּ"ם:

פרק יז - משנה ג

כְּמִין סְרִיגֵי חַלּוֹנוֹת כוּ'. הָרַמְבַּ"ם: טְמֵאָה. אַף עַל פִּי שֶׁכֻּלָּהּ נְקוּבָה כְמוֹצִיא רִמּוֹן. הָרַמְבַּ"ם:

פרק יז - משנה ד

וּלְהָכִי צָרִיךְ שְׁלֹשָׁה אֲחוּזִים, שֶׁצָּרִיךְ נֶקֶב יוֹתֵר רָחָב מֵאִלּוּ הָיָה הָאֶחָד לְבַד. תּוֹסָפוֹת. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: נִפְרְצוּ. בְּדָפְנוֹתֵיהֶן. עַד הַשְׁתָּא מַיְרֵי שֶׁנִּקְּבוּ בְשׁוּלֵיהֶן. מַהֲרַ"ם:

פרק יז - משנה ה

בֵּינוֹנִי. לְפִי דַּעַת הָרוֹאֶה. הָרַמְבַּ"ם: הֻזְכְּרוּ. בְּבֵית הַמִּדְרָשׁ. מַהֲרַ"ם: כְּדִתְנַן בְּפֶרֶק ג' מֵעָרְלָה מִשְׁנָה ז'. וְאַף עַל גַּב דְּרַבִּי מֵאִיר לֹא מָנֵי לְהוּ אֶלָּא רַבָּנָן, מִכָּל מָקוֹם גַּם רַבִּי מֵאִיר מוֹדֶה דִּשְׁאָר רִמּוֹנִים לֹא מְקַדְּשִׁין דְּאֵין דַּרְכָּן לְהִמָּנוֹת. הָרַ"שׁ:

פרק יז - משנה ו

הָרַ"שׁ. אֲבָל הָרַמְבַּ"ם כָּתַב, יָדוּעַ שֶׁיֶּשׁ מִן הַשִּׁעוּרִים שֶׁל הָעִנְיָנִים הַתּוֹרִיִּים מַה שֶּׁנְּשַׁעֲרֵהוּ בְּכַבֵּיצָה, כְּמוֹ שֶׁאָמְרוּ שֶׁכַּבֵּיצָה מְטַמֵּא טֻמְאַת אֳכָלִים:

פרק יז - משנה ח

מְזֻמָּן לָצֵאת מִמֶּנּוּ, שֶׁאֵינוֹ נִבְלָע בַּפְּרִי כְּמַשְׁקֶה תַּפּוּחִים וְתוּתִים, אֶלָּא אָגוּר כְּמַשְׁקֶה עֲנָבִים. רַשִׁ"י:

פרק יז - משנה ט

שֶׁל מֹשֶׁה. לְפִי מִדַּת אָרְכּוֹ שֶׁל מֹשֶׁה דְהַיְנוּ אַמָּתוֹ. רַשִׁ"י:

פרק יז - משנה י

רַבִּי מֵאִיר כוּ' רַבִּי יְהוּדָה. בִּמְנָחוֹת דַּף צ"ז יָלְפִינַן לְהוּ מִקְּרָא. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: הוּא מִזְבַּח הָעוֹלָה שֶׁל אֲבָנִים שֶׁעָשָׂה שְׁלֹמֹה בִּמְקוֹם מִזְבַּח הַנְּחֹשֶׁת שֶׁל מֹשֶׁה: שֶׁהַמִּזְבֵּחַ מְחֻבָּר לַקַּרְקַע בְּסִיד, כִּדְאִיתָא רֵישׁ פֶּרֶק ג' דְּמִדּוֹת:

פרק יז - משנה יא

וְאֵין לִתְמֹהַּ עַל הַק', כִּדְאַשְׁכְּחַן בְּעֶזְרָא שׁוּשַׁנְכָיֵא, עַל שֵׁם שׁוּשָׁן. תּוֹסָפוֹת:

פרק יז - משנה יב

שֶׁלָּהֶן. לְהָרַ"ב נִרְאֶה לִי דְלָא גָרַס לֵיהּ. וְהָרַמְבַּ"ם גָּרַס לֵיהּ וְזֶה לְשׁוֹנוֹ, כְּמוֹ הַפֶּלֶךְ הַגָּדוֹל אֲשֶׁר יִקְשְׁרוּ בוֹ, לְפִי שֶׁבָּזֶה יִקְשְׁרוּ פֶלֶךְ כוּ': גָּדוֹל. דְּכֵיוָן דְּלֹא נַעֲשָׂה בִידֵי אָדָם, בְּצִיר מֵהָכִי לָא חֲשִׁיב. רַשִׁ"י: יֶשְׁנוֹ. לְאוֹתוֹ אֶגְרוֹף. רַשִׁ"י: לִשְׁכָּה. שֶׁמֻּנָּח בַּלִּשְׁכָּה לִקְדֹּחַ אֶת הַצָּרִיךְ לְבֶדֶק הַבַּיִת. רַשִׁ"י: שֶׁבָּעֹל. שֶׁעוֹשִׂין בּוֹ נֶקֶב וְתוֹקְעִין בּוֹ יָתֵד לִקְשֹׁר רְצוּעוֹת הָעֹל. רַשִׁ"י:

פרק יז - משנה יג

וְחִבֵּר כוּ'. בְּתוֹרַת כֹּהֲנִים יָלֵיף לֵיהּ. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: מְשִׁיחָה. הִיא הַשָּׂפָה אֲשֶׁר יוֹצִיא הַחַיָּט בְּעֵת הִבָּדֵל הַבֶּגֶד. הָרַמְבַּ"ם:

פרק יז - משנה יד

חוּץ. לָאו דַּוְקָא, דְּהָאִיכָּא כֶלֶב הַיָּם: מִכְּנַף כוּ'. נוֹצַת הָעָזְנִיָּה. כְּמוֹ נִמְרְטוּ כְנָפֶיהָ, דְּפֶרֶק אֵלּוּ טְרֵפוֹת, דְּפֵרֵשׁ רַשִׁ"י נוֹצוֹת הַגְּדוֹלוֹת, הָכָא נַמִּי נוֹצוֹת הָעָזְנִיָּה גְדוֹלוֹת מְאֹד וְחוֹתְכִים אוֹתָן וְעוֹשִׂין מֵהֶן כֵּלִים. מַהֲרַ"ם: מַשְׁמַע מִלְּשׁוֹנוֹ דִּמְצֻפֶּה לֹא קָאֵי אַכָּנָף. וּמַהֲרַ"ם כָּתַב, דְּמַשְׁמַע בַּתּוֹסֶפְתָּא דְקָאֵי נַמִּי אַכָּנָף:

פרק יז - משנה טו

וּקְצָת קָשֶׁה, דַּהֲוָה לֵיהּ לְמִתְנֵי כְּלִי קִבּוּל מִכָּל מָקוֹם טָמֵא, מַהוּ הָעוֹשֶׂה דְקָתָנֵי. וְעוֹד קָשֶׁה, אַמַּאי לֹא קָתָנֵי בְהֶדְיָא מַשֶּׁהוּ. וּלְשׁוֹן מַהֲרַ"ם, נִרְאֶה לִי לְפָרֵשׁ, לִקֵּט מִכָּל מָקוֹם מִן הַטְּמֵאִים וּמִן הַטְּהוֹרִים וַעֲשָׂאוֹ כְלִי אֶחָד, טָמֵא. וְהַיְנוּ הָא דִתְנַן לְעֵיל הָעוֹשֶׂה כְלִי מִן הַגָּדֵל בַּיָּם וְחִבֵּר לָהֶן מִן הַגָּדֵל בָּאָרֶץ כוּ', וְתַנְיָא הָכָא מִשּׁוּם מִשְׁכָּב כוּ', וּפֵרוּשׁ, עָשָׂה מִשְׁכָּב וּמוֹשָׁב מֵעוֹר חַיָּה שֶׁבַּיָּם וְשֶׁבַּיַּבָּשָׁה. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: הַמַּצָּה כוּ'. הוֹאִיל וְהוּא דָבָר שֶׁל קְיָמָא שֶׁאֶפְשָׁר שֶׁיַּעֲמֹד. הָרַמְבַּ"ם: שֶׁחֲקָקוּם. נֶקֶב קָטָן בְּצִדֵּיהֶן, וְהוֹצִיאוּ הָאֹכֶל מִתּוֹךְ הַנֶּקֶב. רַשִׁ"י: מַחֲשָׁבָה. שֶׁאִם בֶּן דַּעַת חָשַׁב עֲלֵיהֶן, טְמֵאִים בְּמַחֲשָׁבָה, דִּתְנַן סוֹף פֶּרֶק כ"ה כָּל הַכֵּלִים יוֹרְדִין לִידֵי טֻמְאָתָן בְּמַחֲשָׁבָה. רַשִׁ"י: מַשְׁמַע דְּאִלּוּ מָדְדוּ הֲוָה מַעֲשֶׂה. וְהָא דְסִיֵּם וְלֹא עָשׂוּ בָהֶן מְלָאכָה, הַיְנוּ שֶׁלֹּא מָדְדוּ. אֲבָל רַשִׁ"י כָּתַב, שֶׁאֲפִלּוּ מָדְדוּ בָהּ אֶלָּא שֶׁלֹּא עָשׂוּ בָהּ מַעֲשֶׂה לְשֵׁם כְּלִי:

פרק יז - משנה טז

וּכְלוֹמַר, שֶׁמַּנִּיחַ בּוֹ הַמְּזוּזָה שֶׁאֵין נוֹתְנִין מִמֶּנָּה מֶכֶס, וְתַחְתֶּיהָ טוֹמֵן הַמַּרְגָּלִיּוֹת. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: דִּבְרֵי תוֹרָה. צַדִּיקִים כוּ' יִזָּהֲרוּ בָם שֶׁלֹּא לְרַמּוֹת הַבְּרִיּוֹת. וּפוֹשְׁעִים כוּ' יִלְמְדוּ לְרַמּוֹת. רַשְׁבַּ"ם:

פרק יז - משנה יז

וְאֵינָהּ שֶׁל אֶבֶן, לְפִי שֶׁכְּלֵי אֲבָנִים לֹא יִטַּמְּאוּ כְלָל. וְאֵינָהּ גַּם כֵּן מַשְׁחֶזֶת בַּרְזֶל, לְפִי שֶׁכְּלֵי מַתָּכוֹת פְּשׁוּטֵיהֶן טְמֵאִים. הָרַמְבַּ"ם: הַקַּשׁ. שֶׁכָּל הַגָּדֵל מִן הָאָרֶץ, בִּכְלַל כְּלֵי עֵץ. הָרַמְבַּ"ם: וּלְפִי מַה שֶּׁכָּתַב בְּפֶרֶק כ"ד מִשְׁנָה ד' דְּכָל הַטָּמֵא מִדְרָס טָמֵא אַף בִּשְׁאָר הַטֻּמְאוֹת, צָרִיךְ עִיּוּן. וְהָרַמְבַּ"ם חָזַר בּוֹ בְחִבּוּרוֹ, שֶׁכָּתַב, טְהוֹרָה מִן הַמִּדְרָס. וְכָתַב הַכֶּסֶף מִשְׁנֶה, הַטַּעַם יֵשׁ לוֹמַר שֶׁמִּתּוֹךְ שֶׁהִיא קָשָׁה אֵינָהּ רְאוּיָה לִשְׁכִיבָה:
