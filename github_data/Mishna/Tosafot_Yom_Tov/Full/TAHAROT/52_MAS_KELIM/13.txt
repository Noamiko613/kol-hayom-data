פרק יג

פרק יג - משנה א

הַסַּיִף. תֶּלְיְךָ וְקַשְׁתֶּךָ מְתַרְגְּמִינַן סֵיפָךְ וְקַשְׁתָּךָ. הָרַ"שׁ: וּבָרַ"שׁ אִיתָא מַשְׁחִירִין. וְעַיֵּן שָׁם בַּגְּמָרָא: וְטַעֲמֵיהּ נַמִּי מִשּׁוּם דְּמַשְׁחִיר הַשֵּׂעָר: שֶׁנֶּחֱלְקוּ. כֻּלַּהּ פִּרְקִין מַיְרֵי בַשֶּׁבֶר שֶׁיִּשָּׁבֵר בּוֹ הַכְּלִי וְלֹא יְקַבֵּל טֻמְאָה, אוֹ יִטָּהֵר מִטֻּמְאָתוֹ. כָּךְ הֲבִינוֹתִי מֵהָרַמְבַּ"ם. וְעַיֵּן רֵישׁ פֶּרֶק דִּלְקַמָּן בְּהָרַ"ב: הַתּוֹסָפוֹת הִקְשׁוּ עַל זֶה, דְּבֵית יָד לְמַאי חֲזִי לְעַצְמוֹ. וּמְפָרְשִׁים, דִּשְׁחוֹר הַיְנוּ מִסְפָּרַיִם קְטַנִּים כוּ': טַס שֶׁל בַּרְזֶל אָרֹךְ וְכָפוּף לְאֶמְצָעִיתוֹ, וּבִשְׁנֵי רָאשֶׁיהָ כְּאֹרֶךְ שְׁנֵי סַכִּינִים, וּכְשֶׁנֶּחְלְקָה אַכַּתִּי כוּ'. הָרַ"שׁ:

פרק יג - משנה ב

מַעְדֵּר. מִלְּשׁוֹן וְכֹל הֶהָרִים אֲשֶׁר בַּמַּעְדֵּר יֵעָדֵרוּן . הָרַ"שׁ:

פרק יג - משנה ד

מְלֵאַת פְּגִימוֹת, כְּשֶׁרוֹצִים לַחְתֹּךְ קוֹרָה לִשְׁתַּיִם מוֹלִיכָהּ וּמְבִיאָהּ עַל גַּבָּהּ וְהַפְּגִימוֹת חוֹרְצוֹת אוֹתָהּ עַד סוֹפָהּ. וְנִקְרֵאת מְגֵרָה עַל שֵׁם שֶׁחוֹתֶכֶת דֶּרֶךְ גְּרִירָה. הָרַ"שׁ:

פרק יג - משנה ה

מַפִּילִין רָאשֵׁי פְתִילוֹת כְּשֶׁהֵן שְׂרוּפִין, וּבְלַעַ"ז ממוק"ר, וּמְקַנְּחִין בָּהֶן נֵרוֹת שֶׁל מְנוֹרָה מֵאֶפְרָן שֶׁבְּתוֹכָן. רַשִׁ"י: לְטֻמְאָתָהּ. הַיְשָׁנָה. הָרַמְבַּ"ם:

פרק יג - משנה ו

וְאַף עַל פִּי שֶׁהֵם רְאוּיִין קְצָת קֹדֶם שֶׁנִּקְבְּעוּ, אֲפִלּוּ הָכִי טְהוֹרִין, דְּגָלְמֵי כְלֵי מַתָּכוֹת טְהוֹרִין. קְבָעָן בַּפּוֹתַחַת טְמֵאִין, שֶׁנִּגְמְרָה מְלַאכְתָּן. תּוֹסָפוֹת: דְּשֶׁמָּא יֵשׁ קְבוּעִין וְיֵשׁ שֶׁאֵינָם קְבוּעִין. הָרַ"שׁ:

פרק יג - משנה ז

כְּלוֹמַר צְלוֹחִיּוֹת מְלֵאִים מַיִם:

פרק יג - משנה ח

טָהוֹר. בִּיבָמוֹת דַּף מ"ג פָּרְכִינַן, הָא נִשְׁתַּיְּרוּ בוֹ שְׁתַּיִם בְּמָקוֹם אֶחָד טָמֵא, וַהֲדַר תָּנֵי נִשְׁתַּיֵּר בּוֹ שָׁלֹשׁ טָמֵא, שָׁלֹשׁ אִין, שְׁתַּיִם לֹא. וּמְשָׁנֵי, הָא בְּגַוַּיְתָא הָא בַּבְּרַיְתָא. פֵּרֵשׁ רַשִׁ"י, שְׁתֵּי שׁוּרוֹת שִׁנַּיִם יֵשׁ לְמַסְרֵק שֶׁל צֶמֶר, חִיצוֹנִית וּפְנִימִית, הַחִיצוֹנָה עִקַּר הַמְּלָאכָה, וְהַפְּנִימִית לִקְלֹט הַצֶּמֶר שֶׁלֹּא יִפֹּל, וְהִלְכָּךְ בַּחִיצוֹנִית צָרִיךְ שָׁלֹשׁ בְּמָקוֹם אֶחָד, וְהַפְּנִימִית דַּיָּן בִּשְׁתַּיִם: כְּמוֹ שֶׁשָּׁנִינוּ בְּמַכּוֹת דַּף כ'. רַבִּי אֱלִיעֶזֶר אוֹמֵר אֲפִלּוּ לִקְּטָן בְּמַלְקֵט וּבִרְהִיטְנִי חַיָּב. הָרַ"שׁ: וְהִתְקִינָהּ. הָכָא בַעְיָא תִקּוּן, דְּלֹא חֲזִי לִכְתֹּב, שֶׁאֵלּוּ הַמְּחָטִים קְטַנִּים הֵם. מַה שֶּׁאֵין כֵּן בְּרֵישָׁא שֶׁאוֹתָן הַמְּחָטִים עָבִים וְחַזְיָן לִכְתֹּב וְאֵינָן צְרִיכִין תִּקּוּן. הָרַ"שׁ:
