פרק ז

וְלֹא פוֹסְלִין. פֵּרוּשׁ, וְאֵין צָרִיךְ לוֹמַר שֶׁאֵין פּוֹסְלִין. וְאִם תֹּאמַר, כֵּיוָן דַּחֲשִׁיב לְהוּ מַיָּא דְמַעֲלִין, מַאי טַעֲמָא לֹא פָסְלֵי בִּשְׁאוּבִין. הַיְנוּ טַעֲמָא, דְּמִקְוֵה מַיִם הוּא דְאִתַּקַּשׁ לְמַעְיָן, מַה מַּעְיָן בִּידֵי שָׁמַיִם, אַף מִקְוֶה כוּ', אֲבָל שְׁאָר מַשְׁקִין לֹא אִתַּקּוּשׁ. הָרַאֲבַ"ד: הַנָּרוֹק. אֲפִלּוּ נָרוֹק אֵינוֹ פוֹסֵל. טוּר: וְהֵעִידוּ כוּ'. וְאָמְנָם מַה שֶּׁהָיָה חוֹלֵק עַל רַבִּי עֲקִיבָא הוּא בְּמַשָּׂא וּמַתָּן שֶׁל הַדִּין וְעַל צַד הַוִּכּוּחַ, לֹא שֶׁיִּסְבֹּר כָּךְ. הָרַמְבַּ"ם בְּנֻסְחַת אֶרֶץ יִשְׂרָאֵל:

הָרַמְבַּ"ם. וַעֲדַיִן לֹא פֵרְשׁוּ מַהוּ שְׁלָקוֹת. וְכָתַב הָרַ"שׁ מֵי כוּ' שֶׁנִּכְבְּשׁוּ בָהֶן פֵּרוֹת וִירָקוֹת אוֹ נִשְׁלְקוּ: וְלֹא נֶחְסַר מִמֶּנּוּ כִּי אִם אֶחָד מֵאַרְבָּעִים וְאַחַת לְפִי חֶשְׁבּוֹן. וּבְסוֹף פֶּרֶק הֶעָרֵל אָמְרִינַן דְּכָשֵׁר עַד רֻבּוֹ. אֲבָל בְּיוֹתֵר מֵרֹב, לֹא, כֵּיוָן דְּלֹא חֲזִי לִטְבִילָה כְלָל, דְּמִקְוֵה מַיִם כְּתִיב. אֲבָל מַיִם, אֲפִלּוּ בְיוֹתֵר מֵרֻבּוֹ. הָרַ"שׁ וְהָרֹא"שׁ: וַהֲרֵי הַמַּיִם כְּשֵׁרִים כְּשֶׁהָיוּ. וְהַיְנוּ דְתָנֵי בְרֵישׁ פִּרְקִין יֵשׁ דְּבָרִים דְּלֹא מַעֲלִין וְלֹא פוֹסְלִין. הָרַמְבַּ"ם:

כָּשֵׁר. מִפְּנֵי שֶׁאֵין בָּהֶם מִגּוּף הַדָּבָר הַמְשַׁנֶּה אֶת מַרְאָיו. הָר"א: וּמֹחַל. וַאֲפִלּוּ לְמַאן דְּאָמַר דְּאֵינוֹ מַשְׁקֶה, דְּכָל שֶׁאֵין עוֹשִׂין הֵימֶנּוּ מִקְוֶה פּוֹסֵל בְּשִׁנּוּי מַרְאֶה. גְּמָרָא: פָּסוּל. מִשּׁוּם דְּמִתְחֲזֵי כְמִקְוֶה שֶׁל שְׁאָר מַשְׁקִין, וּמִקְוֵה מַיִם כְּתִיב. הָר"א: וְקָשֶׁה, הָא אֶפְשָׁר בְּהַמְשָׁכָה לְרַבִּי יְהוֹשֻׁעַ בְּפֶרֶק ב' מִשְׁנָה ז'. וְנִרְאֶה לִי, דְּמַתְנִיתִין אַלִּיבָּא דְכֻלֵּי עָלְמָא נַסְבָה. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב:

לֹא פְסָלוּהוּ. וְהוּא שֶׁלֹּא שִׁנּוּ מַרְאֵה הַמִּקְוֶה. הָרֹא"שׁ:

כְּלוֹמַר, וְאֵין הֲלָכָה כְּמוֹתוֹ. וְהָרַמְבַּ"ם פָּסַק לְהַךְ דְּסָגוֹס: שְׁאוּבִין. שֶׁהֲרֵי צָרִיךְ שֶׁיָּבֹאוּ בָהֶם הַמַּיִם: לְפִי שֶׁכְּשֶׁהַכְּלִי בַמִּקְוֶה לֹא נִשְׁאֲבוּ, וּכְשֶׁמַּעֲלֵהוּ וְנִתְהַפֵּךְ דֶּרֶךְ שׁוּלָיו תּוּ לֹא מִקְּרֵי מַיִם שֶׁבְּתוֹכוֹ, שֶׁאֵינָן מִתְקַבְּלִין בְּתוֹכוֹ:

כְּלוֹמַר בַּטִּיט לְבַדּוֹ, אֲבָל כְּשֶׁהַמַּיִם צָפִים עַל גַּבָּיו מַטְבִּילִין בּוֹ: וְקָשֶׁה, דְּמִכָּל מָקוֹם מַאי מַהֲנֵי, הָא אָז לֹא הָיְתָה הַמִּטָּה כֻלָּהּ בַּמַּיִם וַאֲנַן כֻּלָּהּ בְּבַת אַחַת בָּעִינַן. וּלְשׁוֹן הָרֹא"שׁ, קָדְמוּ הַמַּיִם וְנָגְעוּ בְרַגְלֵי כוּ', וְאוֹתָן הַמַּיִם מְחֻבָּרִים לַמִּקְוֶה, הִלְכָּךְ לֹא הָוֵי חֲצִיצָה, כְּדִתְנַן סוֹף פֶּרֶק ח' הָאוֹחֵז בָּאָדָם כוּ' אִם הֵדִיחַ יָדָיו טְהוֹרִים: כִּדְיָלְפִינַן בְּסִפְרָא. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: וְקָשֶׁה, דְּאִם כֵּן מַאי אֲפִלּוּ, הָא דַוְקָא עֵצִים וְקָנִים. אֲבָל לְשׁוֹן הָרַ"שׁ, לֹא מִבַּעְיָא אֲבָנִים אֶלָּא אֲפִלּוּ עֵצִים וְקָנִים שֶׁהַמַּיִם בֵּינֵיהֶם לָא חֲשִׁיב הֶפְסֵק וּמִצְטָרְפִין הַמַּיִם שֶׁבֵּינֵיהֶם לְאַרְבָּעִים סְאָה, וּבִלְבַד שֶׁלֹּא יְחַלְּקוּ כָל הַמִּקְוֶה, שֶׁאַף עַל פִּי שֶׁהַמַּיִם שֶׁבֵּין הַנְּצָרִים מְחַבְּרָן אֵין זֶה חִבּוּר כוּ' . וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: שֶׁיִּתְפְּחוּ. כְּלוֹמַר, שֶׁיַּעֲלוּ כְמִין תַּפּוּחַ וְיִהְיוּ עֲמֻקִּים מִצַּד אֶחָד וְטוֹבֵל בָּהֶם. הָרַ"שׁ: וְלֹא הָוֵי כִמְהַוֶּה מִקְוֶה עַל יְדֵי דָבָר הַמְקַבֵּל טֻמְאָה, מִידִי דַהֲוֵי אַטּוֹבֵל אֶת הַכְּלִי בְיָדָיו בְּתוֹךְ הַמִּקְוֶה, שֶׁאִם הֱדִיחָן, טְהוֹרִים, דְּהָכָא נַמִּי עִקַּר הַמִּקְוֶה בִּמְקוֹמוֹ וְהַגַּל לֹא נִתְלַשׁ לְגַמְרֵי. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב:
