פרק ו - משנה א

וְאִם תֹּאמַר לְרַבִּי מֵאִיר לָמָּה הֻזְכַּר הָעֶלְיוֹן, כֵּיוָן דְּלֹא סָמְכִינַן עֲלֵיהּ. וְיֵשׁ לוֹמַר, דְּאִצְטְרִיךְ לְעִנְיַן עֳנָשִׁין וְקִדּוּשִׁין וּמִלֵּי טוּבָא, דְּרֹב פְּעָמִים תַּחְתּוֹן אָתָא בְּרֵישָׁא כִּדְאָמְרִינַן בַּגְּמָרָא. וְאִם כֵּן אִי חֲזֵינָא עֶלְיוֹן מִסְּתָמָא כְּבָר בָּא תַּחְתּוֹן וּגְדוֹלָה הִיא וּצְרִיכָה גֵּט לְקִדּוּשֶׁיהָ, וְגַם אָסוּר בַּאֲחוֹתָהּ:

פרק ו - משנה ב

וַהֲוָה מָצֵי נַמִּי לְאוֹקְמָא לְעִנְיַן טָהֳרַת טֻמְאָה. דִּכְלִי הֶעָשׂוּי לְמַשְׁקֶה שִׁעוּרוֹ בְּכוֹנֵס מַשְׁקֶה. אֶלָּא דְּנִיחָא לֵיהּ לְאוֹקְמָא בְּעָשׂוּי אַף לָאֳכָלִים, דְּהָא כָּל כְּלִי חֶרֶס תְּנַן:

פרק ו - משנה ג

וְכֵן לְשׁוֹן רַשִׁ"י, שֶׁאֵין לְךָ כְּלִי הָרָאוּי לְמוֹשַׁב הַזָּב שֶׁאֵין שֵׁם כְּלִי עָלָיו וְרָאוּי לְקַבֵּל כָּל טֻמְאוֹת. וְהַאי דְּנָקַט טְמֵא מֵת, מִשּׁוּם דּוּמְיָא דְּמִדְרַס הַזָּב דְּאַב הַטֻּמְאָה הוּא. וְהָכִי קָאָמַר, כָּל הָרָאוּי כוּ'. וְעַיֵּן מַה שֶּׁכָּתַב פֶּרֶק כ"ד דְּכֵלִים מִשְׁנָה ד': מִשּׁוּם דְּבַמֵּת עַצְמוֹ אֵין שַׁיָּךְ לוֹמַר עֲמֹד, נָקַט טְמֵא מֵת. וּטְמֵא מֵת עוֹשֶׂה לִכְלִי אַב הַטֻּמְאָה: דְּכֵיוָן דְּעַל יְדֵי מֵת מִתְטַמֵּא בִּנְגִיעָה לִהְיוֹת אַב הַטֻּמְאָה, לֹא נַפְקָא מִנַּהּ בִּישִׁיבָה:

פרק ו - משנה ד

כֹּל כוּ'. וְאִשָּׁה פְּסוּלָה לְהָעִיד, וְאִם כֵּן פְּסוּלָה לָדוּן. וְקָשֶׁה מִדְּבוֹרָה שֶׁשָּׁפְטָה אֶת יִשְׂרָאֵל. וְיֵשׁ לוֹמַר דְּהָכִי קָאָמַר, כָּל אִישׁ הַכָּשֵׁר כוּ'. אִי נַמִּי, עַל פִּי הַדִּבּוּר הָיְתָה דָּנָה. אוֹ שֶׁהָיְתָה מְלַמֶּדֶת לָהֶם הַדִּינִים. תּוֹסָפוֹת. וְהָרַ"ן תֵּרֵץ, דְּקִבְּלוּהָ שָׁאנֵי: אֲבָל בִּשְׁתֵּי עֵינָיו. אֲפִלּוּ לְעֵדוּת פָּסוּל, דִּכְתִיב אוֹ רָאָה. רַשִׁ"י. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב:

פרק ו - משנה ה

וְאַף דִּבְהֵמָה חַיָּב בְּמַעֲשֵׂר, מִכָּל מָקוֹם מַעַשְׂרוֹ אֵינוֹ טוֹבְלוֹ. וְהָכָא בְּמַעֲשֵׂר דְּטָבֵיל אַיְרֵי. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: הָרַמְבַּ"ם. וְקָשֶׁה, דִּבְרֵישׁ מַעַשְׂרוֹת אַדִּתְנַן וְגִדּוּלָיו מִן הָאָרֶץ, קָאָמַר, יָצְאוּ כְּמֵהִין וּפִטְרִיּוֹת וְכֵן סְטִיס וְקוֹצָה. וְעוֹד קָשֶׁה, דִּבְפֶרֶק קַמָּא דְּמַעֲשֵׂר שֵׁנִי פֵּרֵשׁ דְּבָקָר וְצֹאן בִּכְלַל גִּדּוּלֵי קַרְקַע נִינְהוּ, וְכֵן הוּא בַּגְּמָרָא. וְעוֹד, דְּמָצִינוּ נַמִּי פָּגִים וּבֹסֶר וִכְפָנִיּוֹת דַּהֲווּ אֳכָלִים וּפְטוּרִים מִמַּעֲשֵׂר. לָכֵן נִרְאֶה לִי, דְּהָא דְּנָקְטִינַן בַּגְּמָרָא בָּשָׂר וְדָגִים וּבֵיצִים, לְאַשְׁמְעִינַן דְּאַף עַל פִּי שֶׁהֵם אֳכָלִים חֲשׁוּבִים, פְּטוּרִים מִמַּעֲשֵׂר. וְהַטַּעַם, לְפִי שֶׁאֵינָן דּוֹמִים כְּלָל לְדָגָן תִּירוֹשׁ וְיִצְהָר. וְכֵן רַשִׁ"י לֹא פֵּרֵשׁ שׁוּם טַעַם:

אֲבָל שְׁבִיעִית יֵשׁ לָהֶם, לְעִנְיַן שֶׁחַיָּב לְהַפְקִיר, וְאָסוּר לַעֲשׂוֹתָן סְחוֹרָה וּמְלוּגְמָא וְאַפִּקְטְוִיזִין. רַשִׁ"י:

פרק ו - משנה ט

כֹּל כוּ'. וְהָא דְּכָתַב רַחֲמָנָא סְנַפִּיר, מִשּׁוּם יַגְדִּיל תּוֹרָה וְיַאְדִּיר. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: וְאַף עַל גַּב דְּלֹא נַפְקָא מִנַּהּ, דִּבְלָאו הָכִי טָמֵא לְפִי שֶׁאֵינוֹ מַעֲלֶה גֵּרָה, יֵשׁ לוֹמַר דְּכֻלֵּי גַּוְנֵי נָקַט דְּיֵשׁ לוֹ וְיֵשׁ לוֹ וְיֵשׁ שֶׁיֵּשׁ לוֹ כוּ':

פרק ו - משנה י

וּלְשׁוֹן הָרַמְבַּ"ם, שֶׁהַמִּצְוֹת כֻּלָּן כוּ' כְּמוֹ הַצִּיצִית כוּ'. וּנְטִילַת יָדַיִם, לִפְנֵי הַנִּגּוּב הוּא דִּמְבָרְכִין, וְהָוֵי נַמִּי לְפָנָיו. וּטְבִילָה נַמִּי דִּינָהּ לְפָנֶיהָ, אֶלָּא דְּחָיְשִׁינַן דִּלְמָא מִשּׁוּם בִּיעֲתוּתָא דְמַיָּא מִמְּנַע וְלֹא טָבִיל, וְאַחַר שֶׁהוּא יוֹרֵד, אָז הוּא עָרֹם וְאָסוּר לְבָרֵךְ מִשּׁוּם דְּלִבּוֹ רוֹאֶה אֶת הָעֶרְוָה. תּוֹסָפוֹת. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב:

פרק ו - משנה יא

לֹא דָּק, דְּרָאוּי לְהוֹלִיד הוּא מִשֶּׁיָּבִיא שְׁתֵּי שְׂעָרוֹת. וְגַם מַה שֶּׁתָּלָה דְּרָשָׁה דְּבֵן וְלֹא אָב בְּהַקָּפַת זָקָן, לֹא דָּק, דְּהַךְ דְּרָשָׁה לַאֲתוֹיֵי אַף קֹדֶם שֶׁהִקִּיף מִכִּי מָלוּ שְׁלֹשָׁה חֳדָשִׁים מִשֶּׁהֵבִיא שְׁתֵּי שְׂעָרוֹת, שֶׁאָז נִכָּר הַהֵרָיוֹן וְרָאוּי לְהִקָּרֵא בְּשֵׁם אָב. וְכֵן נַמִּי הִקִּיף אַף עַל גַּב דְּלֹא מָלוּ שְׁלֹשָׁה חֳדָשִׁים. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: רַשִׁ"י. וּבַגְּמָרָא, לֹא שֶׁיִּרְבֶּה מַמָּשׁ, אֶלָּא כְּדֵי שֶׁיִּהְיוּ שְׁתֵּי שְׂעָרוֹת שׁוֹכְבוֹת, וְנִרְאוֹת כְּמוֹ שֶׁיִּרְבֶּה כוּ'. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: וְאִם לֹא נִרְאוּ בָּהּ סִימָנֵי אַיְלוֹנִית, עַד שֶׁתְּהֵא בַּת שְׁלֹשִׁים וְחָמֵשׁ שָׁנִים וְיוֹם אֶחָד. וְכֵן פָּסַק הָרַמְבַּ"ם בְּחִבּוּרוֹ: קֹדֶם שֶׁבָּעַל. דְּוַדַּאי בָּעַל לְשֵׁם קִדּוּשִׁין. וְהָוֵי סְפֵקָא דְּאוֹרַיְתָא. טוּר:
