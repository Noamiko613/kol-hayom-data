נוֹשְׂאִים כוּ'. כֻּלַּהּ מַתְנִיתִין מַיְרֵי בְנוֹשְׂאִים בָּאֲוִיר וְלֹא בְּתוֹךְ הַבַּיִת. הָרַ"שׁ: טְמֵאִים. וַאֲפִלּוּ שֶׁלֹּא כְנֶגֶד הַטֻּמְאָה. הָרַמְבַּ"ם. דְּכֵיוָן שֶׁנַּעֲשׂוּ אֹהֶל הָוֵי כְמַאן דְּמָלֵי טֻמְאָה, שֶׁהֲרֵי מְטַמֵּא לְכָל הַכֵּלִים שֶׁתַּחְתָּיו, וַהֲרֵי כָל מַה שֶּׁעָלָיו הוּא כְנֶגֶד הַטֻּמְאָה: טְמֵאִים. דְּכֵיוָן שֶׁאֵינוֹ חוֹצֵץ וְהַטֻּמְאָה יוֹרֶדֶת שׁוּב נַעֲשֶׂה אֹהֶל עָלֶיהָ לְטַמֵּא כָּל מַה שֶּׁתַּחְתָּיו: מְטַהֵר. הַיְנוּ בְנָתוּן עַל מוֹט שֶׁאֵינוֹ כָעֳבִי הַמַּרְדֵּעַ, שֶׁלֹּא הֻזְקְקוּ הַנּוֹשְׂאִים לַטֻּמְאָה. הָרַ"שׁ. וּמִסְתַּבְּרָא דְפָלֵיג נַמִּי בְנָתוּן עַל אַרְבָּעָה כֵלִים. וּמַהֲרַ"מ כָּתַב דִּבְאַרְבָּעָה כֵלִים אֲפִלּוּ רַבִּי אֱלִיעֶזֶר מוֹדֶה, וּבְאַרְבָּעָה אֲבָנִים רַבָּנָן מוֹדוּ דְחוֹצֵץ, אֶלָּא בָאָדָם פְּלִיגֵי. רַבָּנָן סָבְרֵי כֵּיוָן דִּמְקַבֵּל טֻמְאָה הָוֵי כְּמוֹ כְלִי, וְרַבִּי אֱלִיעֶזֶר מְדַמֶּה לֵיהּ לִבְהֵמָה כֵּיוָן שֶׁיֵּשׁ בּוֹ רוּחַ חַיִּים: כְּלֵי גְלָלִים כוּ'. שֶׁאֵינָם בָּאִים בְּמִדָּה. דְּאִי לָאו הָכִי אֵינָן כְּכֵלִים אֶלָּא כְאֹהָלִים וְחוֹצְצִים. הָרַמְבַּ"ם:

דְּאִי מִנּוֹשְׂאֵי הַמִּטָּה מִכֵּיוָן שֶׁנִּכְנַס הַמֵּת תַּחַת הַקּוֹרָה כְּבָר נִכְנְסָה הַטֻּמְאָה לְפֶתַח הַבַּיִת כוּ'. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: וְצָרִיךְ גַּם כֵּן שֶׁלֹּא יִהְיוּ מֻכְשָׁרִים: וְאִית בֵּיהּ קוֹצִים, דְּלֹא חֲזִי לְטִינָא לִדְרֹס בָּרֶגֶל, וְהוּא לַח, דְּלֹא חֲזִי לְהַסָּקָה. גְּמָרָא: וְאַף בְּלֹא בִטּוּל, כֵּיוָן דְּסוֹתֵם כָּל הַחַלּוֹן. וְדַעַת הָרַמְבַּ"ם נִרְאֶה דַּאֲפִלּוּ הָכִי בָּעֵי בִּטּוּל. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב:

וְהוּא הַדִּין בֵּין בַּיִת לִרְשׁוּת הָרַבִּים אוֹ לְמָבוֹי, אֶלָּא נָקַט כַּאֲשֶׁר הָיוּ בָתֵּיהֶם בִּזְמַן הַמִּשְׁנָה שֶׁהָיוּ חֲצֵרוֹת לִפְנֵיהֶם וְגִנּוֹת לַאֲחוֹרֵיהֶם: בְּתוֹכוֹ. וְהִיא רְצוּצָה: הָרַ"שׁ. וְנִרְאֶה דִרְצוֹנוֹ לוֹמַר שֶׁסָּדוּק וּפָתוּחַ קְצָת לַבַּיִת, וְעַיֵּן רֵישׁ פֶּרֶק ז'. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב:

בַּכֹּתֶל כוּ' טְמֵאִין. דְּאֵין מַצִּיל מֵהַטֻּמְאָה אֶלָּא צָמִיד פָּתִיל אוֹ אֹהֶל אַחֵר, אֲבָל בְּלוּעִים בַּבִּנְיָן לֹא. הָרַמְבַּ"ם: טְמֵאִין. קָשֶׁה, מַאי שְׁנָא מִדִּלְעֵיל הַטֻּמְאָה בַכֹּתֶל מֶחֱצָה עַל מֶחֱצָה דַּחֲכָמִים מְטַהֲרִים הָעוֹמֵד מִלְּמַעְלָן. וְנִרְאֶה לִי דִטְפֵי שַׁיָּךְ לוֹמַר בַּכֹּתֶל שֶׁתִּכָּנֵס טֻמְאָה לַבַּיִת וְלֹא תִּתְפַּשֵּׁט לְמַעְלָה מַה שֶּׁאֵין כֵּן בְּמַעֲזִיבָה שֶׁבְּנוּיָה נַמִּי לַעֲלִיָּה. תּוֹסְפוֹת יוֹם טוֹב:

דְּאִי אֶפְשָׁר לְיַשֵּׁב כְּרַבָּנָן, דְּהַשְׁתָּא בְמַעֲזִיבָה מֵחֶצְיָהּ וּלְמַטָּה הַבַּיִת טָמֵא, כָּל שֶׁכֵּן בֵּין הַקּוֹרוֹת. הָרַ"שׁ. וְעַל כָּל פָּנִים גַּם לְרַבִּי יְהוּדָה צָרִיךְ לְחַלֵּק בֵּין קוֹרוֹת לְמַעֲזִיבָה, דְּלֹא אָמְרִינַן בְּמַעֲזִיבָה רְצוּצָה כֵּיוָן דַּעֲשׂוּיָה לִסָּתֵר. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: דִּיֵּק מִלְּשׁוֹן הַכֹּל, וְלֹא קָתָנֵי בַיִת, מִכְּלָל דְּאַתַּרְוַיְהוּ שָׁדִינַן לַהּ. הָרַ"שׁ: כְּנֶפֶשׁ אֲטוּמָה דְּפֶרֶק ז', דְּמִן הַצְּדָדִים טָהוֹר, אֶלָּא דְבוֹקַעַת וְעוֹלָה בּוֹקַעַת וְיוֹרֶדֶת, וְהָכָא נַמִּי חֲדָא מִתַּרְתֵּי נָקַט. וְאֵין לוֹמַר כְּשֶׁיּוֹרֶדֶת נִמְצָא הַבַּיִת מַאֲהִיל וּמְמַלֵּא כָל הַבַּיִת, דְּהָא הָאֹהֶל חוֹצֵץ לְטַהֵר. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: וְקָשֶׁה, דְּמַתְנִיתִין הֲוָה לַהּ לְפָרֵשׁ כֵּן בְּהֶדְיָא. וְהָרַמְבַּ"ם פֵּרֵשׁ, בִּמְחִצָּה שֶׁל זְכוּכִית. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: טָמֵא. וַעֲלִיָּה טְהוֹרָה. הָרַ"שׁ:

מִשּׁוּם דְּנַעֲשָׂה מֵאֵלָיו. מַהֲרַ"ם. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: אֲבָל אִם יֵשׁ בִּמְקוֹם הַטֻּמְאָה טֶפַח עַל טֶפַח בְּרוּם טֶפַח הָוֵי כְּקֶבֶר סָתוּם וּמְטַמֵּא כָל סְבִיבָיו. תּוֹסֶפְתָּא:

וְקָשֶׁה, דְּהָא לְעֵיל בְּעַמּוּד הָעוֹמֵד בְּתוֹךְ כּוּךְ וּמְעָרָה אָמְרִינַן דְּבוֹקַעַת וְעוֹלָה, וּפְשִׁיטָא דְהֵם מְקֹרִים. וְעוֹד שֶׁהָרַ"ב כָּתַב לְעֵיל דְּהוּא הַדִּין לְעַמּוּד שֶׁבַּבַּיִת: מְטַמֵּא. אַף דִּבְכַמָּה דוּכְתֵּי מַשְׁמַע דְּטֻמְאָה תַחַת הַכְּלִי וּרְצוּצָה אֵינָהּ בּוֹקַעַת אֶלָּא כְנֶגֶד הַטֻּמְאָה, נִרְאֶה לִי דְהָכָא מַיְרֵי שֶׁהַטֻּמְאָה מְמַלְּאָה כָּל תַּחַת הָעַמּוּד, וְהַפֶּרַח בָּטֵל לְגַבֵּי עַמּוּד לְרַבִּי יוֹחָנָן בֶּן נוּרִי. מַהֲרַ"ם: טְהוֹרִין. שֶׁאֵין מְקוֹם הַפֶּרַח שֶׁאֵין שָׁם טֻמְאָה בָּטֵל לִקְצָתוֹ שֶׁיֵּשׁ שָׁם הַטֻּמְאָה כְּמוֹ שֶׁבָּטֵל לְגַבֵּי עַמּוּד. מַהֲרַ"ם: וְלֹא דָמֵי לְדִלְעֵיל מִשְׁנָה ה' טֻמְאָה בֵּין הַקּוֹרוֹת כוּ' כְּאִלּוּ הִיא אֹטֶם וּבוֹקַעַת כוּ' וְלֹא אָמְרִינַן הַכֹּל טָמֵא שֶׁתִּתְפַּשֵּׁט בְּכָל הָעֲלִיָּה אוֹ הַבַּיִת לְאַחַר שֶׁבּוֹקַעַת, דְּשָׁאנֵי הָכָא דְכֹתֶל בַּיִת בָּתַר בַּיִת שָׁדִינַן לֵיהּ וְלָא חֲשִׁיבָה רְצוּצָה לְגַבֵּי בַּיִת, כְּדִלְעֵיל מִשְׁנָה ג', עַיֵּן שָׁם. וְדַוְקָא נִפְתַּח, אֲבָל לֹא נִפְתַּח לֹא שַׁיָּךְ בְּטֻמְאָה רְצוּצָה לוֹמַר סוֹף טֻמְאָה לָצֵאת דֶּרֶךְ הַבַּיִת כוּ'. מַהֲרַ"ם. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: וַחֲבֵרוֹ טָהוֹר. כֵּיוָן דְּבַיִת גּוּפֵיהּ לֹא מִטַּמֵּא מִגּוּפֵיהּ אֶלָּא מִכֹּחַ פַּרְדַּסְקִים שֶׁאֶצְלוֹ, דַּי לָנוּ אִם נְטַמֵּא הַבַּיִת וְלֹא הַכֹּתֶל. מַהֲרַ"ם:
