פרק ה

יֵשׁ גֵּט כוּ'. מִדְּרַבָּנָן. וְטַעְמָא דְּגָזְרוּ מְפָרֵשׁ בַּגְּמָרָא: לֹא קָאֵי אֶלָּא אַגֵּט וּמַאֲמָר, וְלֹא אַבְּעִילָה וַחֲלִיצָה, כְּמוֹ שֶׁאִיתָא בְּמִשְׁנָה ג':

כְּמִצְוָתָהּ. דְּהָכִי תַקִּינוּ רַבָּנָן, שֶׁלֹּא יָבֹא עָלֶיהָ עַד שֶׁיְּקַדְּשֶׁהָ קִדּוּשִׁין גְּמוּרִים, וְאִם בָּעַל וְלֹא קִדֵּשׁ קָנָה, וְלוֹקֶה מַכּוֹת מַרְדּוּת עַל שֶׁנָּהַג קַלּוּת רֹאשׁ בְּעַצְמוֹ, כִּדְאִיתָא בַּגְּמָרָא:

וּמֵהַאי טַעֲמָא אֲפִלּוּ רַבָּן גַּמְלִיאֵל מוֹדֶה דְּיֵשׁ מַאֲמָר אַחַר גֵּט וְיֵשׁ גֵּט אַחַר מַאֲמָר, וְכֵן בִּיאָה יֵשׁ אַחַר מַאֲמָר וְכֵן אַחַר גֵּט כִּדְאִיתָא בַּגְּמָרָא: וּמִכָּל מָקוֹם לֹא הֲוֵי גֵּט כַּחֲלִיצָה לְאַפְקוּעֵיהּ כֻּלָּהּ זִקָּה, הִלְכָּךְ לֹא עָמְדָה עֲלֵיהּ בְּלָאו גָּמוּר, וַאֲפִלּוּ לְרַבִּי עֲקִיבָא דְּמַתְנִיתִין כְּוָתֵיהּ מוֹדֶה דְּקִדּוּשִׁין תַּפְסֵי בָּהּ: חָלַץ כוּ'. אוֹ אוֹ קָתָנֵי, וְעָשָׂה מַאֲמָר אוֹ נָתַן גֵּט אוֹ בָּעַל. וְגֵט דִּלְאַחַר חֲלִיצָה לֹא הֲוָה צָרִיךְ לְמַתְנִיתִין לְאַשְׁמוֹעִינַן דְּלָאו כְּלוּם, דִּלְמַאי אַהֲנֵי, לִתְּסַר בִּקְרוֹבוֹתֶיהָ הָא מִתְּסָר וְקָאֵי מֵחֲמַת חֲלִיצָה. אֶלָּא אַיְּדִי דְּבָעֵי לְמִתְנֵי גֵּט דְּאַחַר בְּעִילָה לְאַשְׁמוֹעִינַן דְּאֵין לְאַחַר בִּיאָה כְּלוּם תָּנָא נַמִּי גֵּט דְּאַחַר חֲלִיצָה, וְהָכִי מְפָרֵשׁ בַּגְּמָרָא. רַשִׁ"י: אוֹ בָעַל כוּ'. (הָכִי נַמִּי אוֹ אוֹ קָתָנֵי) וּמַאֲמָר דְּאַחַר בִּיאָה לֹא הֲוָה צָרִיךְ לְמִתְנֵי, דַּאֲפִלּוּ אַחַר מַאֲמָר בִּיבָמָה אַחַת מַאי מַהֲנֵי, אֶלָּא אַיְּדִי דְּתָנֵי רֵישָׁא חָלַץ וְעָשָׂה מַאֲמָר דְּאִיצְטְרִיךְ לְאַשְׁמְעִינַן כְּרַבִּי עֲקִיבָא דְּאֵין קִדּוּשִׁין תּוֹפְסִין כוּ', וְהָכִי מְפָרֵשׁ בַּגְּמָרָא. רַשִׁ"י: אֵין אַחַר חֲלִיצָה כוּ'. אַרֵישָׁא אַחָלַץ וְעָשָׂה מַאֲמָר כוּ' קָאֵי, וְאַלִּישְׁנָא דְּסֵיפָא אוֹ בָּעַל כוּ' הֲוָה שַׁיָּךְ לְמִתְנֵי אֵין אַחַר בִּיאָה כְּלוּם. וְלִישָׁנָא קְלִילָא הוּא לְמִתְנֵי חֲדָא, וְהַתָּרַת יְבָמָה לַשּׁוּק עֲדִיפָא לֵיהּ לְמִתְנֵי. גְּמָרָא: וְכֵן בְּהָא דְּאֵין אַחַר בִּיאָה כְּלוּם לְרַבָּנָן נַמִּי מַשְׁכַּחַת לָהּ דְּיֵשׁ אַחֲרֶיהָ, בִּשְׁתֵּי יְבָמוֹת וְיָבָם אֶחָד, אִי נַמִּי בִּשְׁנֵי יְבָמִים, שֶׁאִם בָּעַל הָאַחַת וְקִדֵּשׁ הוּא אוֹ אָחִיו הַשְּׁנִיָּה תָּפִיס כִּדְמַסְקֵי הָרִי"ף וְהָרֹא"שׁ דְּלֹא הֲוָה צָרָה אַחַר בְּעִילָה אֶלָּא עֲשֵׂה דְּבַיִת אֶחָד כוּ' וְלָאו הַבָּא מִכְּלַל עֲשֵׂה עֲשֵׂה. וּלְגִרְסַת הַתּוֹסָפוֹת דַּף י"א בְּלָאו. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב:

כֵּיצַד. בִּשְׁתֵּי יְבָמוֹת לְיָבָם אֶחָד. רַשִׁ"י: שְׁנֵי גִטִּין. וְיִבּוּמֵי לֹא. דְּהֵיכִי לִיְעַבַד, אִי לְיִבּוּמֵי לְתַרְוַיְהוּ הֲוָה לֵיהּ שְׁנֵי בָתִּים, וְאִי לַחֲדָא הוֹאִיל וַחֲבֶרְתָּהּ בֵּיהּ אֲגִידָא הֲוָה לֵיהּ בּוֹנֶה שְׁנֵי בָתִּים. רַשִׁ"י. וּמִכֵּיוָן דְּיָהִיב גִּטָּא לַחֲדָא קָם עֲלֵיהּ בְּלֹא יִבְנֶה אַף לַשְּׁנִיָּה: שְׁנֵי גִטִּין וַחֲלִיצָה. דִּבְגֵט לְחוּדָא לֹא נָפְקָא בַּעֲלַת בִּיאָה, דְּבִיאָה פְּסוּלָה הִיא וּכְמוֹ שֶׁכָּתַב הָרַ"ב בְּסָמוּךְ, וּפְסוּלָה לֹא סַגִּי לָהּ בְּגֵט, וְטַעֲמָא מְפֹרָשׁ בַּגְּמָרָא: חֲלִיצָה. הַיְנוּ חֲלִיצָה אַחַת, וּפוֹטֶרֶת צָרָתָהּ. וּמַאי צְרִיכוֹת, צְרִיכוֹת דְּעָלְמָא. גְּמָרָא: דְּלֵיכָּא לְמִגְזָר בַּחֲלִיצָה פְּסוּלָה כְּמוֹ בְּבִיאָה פְּסוּלָה כְּדִמְפָרֵשׁ בַּגְּמָרָא. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב:

שְׁנֵי יְבָמִין כוּ'. וְאֵין צָרִיךְ לוֹמַר בִּשְׁנֵי יְבָמִין לִשְׁתֵּי יְבָמוֹת, דְּאִיכָּא לְמַאן דְּאָמַר בַּגְּמָרָא דְּצָרָה עַל הָאַחִין בְּכָרֵת דְּכִדְקַיְמָא קַיְמָא:

מִשּׁוּם דְּהַאי לִישָׁנָא מִתַּנְיָא בְּמִשְׁנָה ג' בְּיָבָם אֶחָד וִיבָמָה אַחַת, קָא מְפָרֵשׁ הָכִי, אֲבָל לְקוּשְׁטָא דְּמִלְּתָא קָא מְהַדְּרֵי אַכֻּלְּהוּ, דְּכֻלְּהוּ שָׁוִים בֵּין בְּיָבָם אֶחָד וִיבָמָה אַחַת בֵּין שְׁתֵּי יְבָמוֹת. וְהַיְנוּ טַעֲמָא דַּהֲדַר אַסְבְּרָא לְהוּ בִּשְׁתֵּי יְבָמוֹת כְּגוֹן גֵּט לְזוֹ כוּ', וְלֹא בָּא לוֹמַר אֶלָּא דְּאֵין זֶה דִּין חָדָשׁ: וּלְעֵיל בְּמִשְׁנָה ב' בְּעָשָׂה מַאֲמָר וַחֲלִיצָה לֹא תְּנַן אֵין אַחֲרָיו כְּלוּם, כֵּיוָן שֶׁצְּרִיכָה מַעֲשֵׂה דְּגֵט אַחֲרָיו עַל הַמַּאֲמָר שֶׁקְּדָמוֹ. תּוֹסָפוֹת: וַהֲוָה מָצֵי לְמֵימַר וְעָשָׂה מַאֲמָר בְּבַעֲלַת הַגֵּט, דְּהָא תְּנַן לְעֵיל גֵּט וּמַאֲמָר בְּאַחַת שֶׁצְּרִיכָה גֵּט וַחֲלִיצָה, אֶלָּא מִשּׁוּם אִסּוּר קְרוֹבוֹת נָקַט בַּשְּׁלִישִׁית, דְּאִלּוּ בַּעֲלַת גֵּט בְּלָאו הָכִי מִשּׁוּם גִּטָּא נֶאֶסְרוּ הַקְּרוֹבוֹת: אַחַת בְּעִילָה כוּ'. טַעֲמֵיהּ מְפָרֵשׁ בַּגְּמָרָא:
