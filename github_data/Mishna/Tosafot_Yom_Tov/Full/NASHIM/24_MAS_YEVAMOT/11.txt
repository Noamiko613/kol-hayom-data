פרק יא

פרק יא - משנה א

וְהוּא הַדִּין וְכָל שֶׁכֵּן בִּשְׁאָר קְרוֹבוֹת דְּשָׁרֵי: פֵּרֵשׁ רַשִׁ"י שֶׁהָיְתָה הָרִאשׁוֹנָה לָקוּחָתוֹ חַיָּב עַל הַשְּׁנִיָּה כְּשֶׁבָּא עָלֶיהָ, דְּבַשְּׁנִיָּה לֵיכָּא לְמֵימַר לִקּוּחִין דְּהָא לֹא תַּפְסֵי בָּהּ קִדּוּשִׁין. וְכָתַב נִמּוּקֵי יוֹסֵף וּמִיהוּ קָשֶׁה דִּלְמָא כָּל הֵיכָא דְּאִית בָּהּ לִקּוּחִין אָסְרָה תּוֹרָה, וּבֵין בַּשְּׁנִיָּה בֵּין בָּרִאשׁוֹנָה. וְיֵשׁ לוֹמַר דְּמַשְׁמָעוּת קְרָא הָכִי הוּא שֶׁהַקִּיחָה הִיא בָּרִאשׁוֹנָה, דְּהָכִי אָסְרָה תּוֹרָה לִשְׁכַּב בָּאַחַת אַחַר שֶׁיֵּשׁ לוֹ הָאַחֶרֶת. וְאִם הָרִאשׁוֹנָה אֲנוּסָה לְבַד בִּשְׁבִיל זֶה אֵינָהּ שֶׁלּוֹ. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: וּמְנָלָן דְּבַאֲנוּסָה. דִּכְתִיב מֵעִלָּוֵיה דִּקְרָא וְנָתַן הָאִישׁ כוּ' לַאֲבִי הַנַּעֲרָה כוּ'. וְרַבָּנָן הָא לֹא סָמִיךְ לֵיהּ, אֶלָּא כִּי אָתָא קְרָא לְשׁוֹמֶרֶת יָבָם שֶׁל אָבִיו וְכָנָף הָרָאוּי לְאָבִיו הוּא אַף עַל גַּב דַּהֲוָה לֵיהּ דּוֹדָתוֹ וִיבָמָה לַשּׁוּק וְלַעֲבֹר עָלֶיהָ בִּשְׁלֹשָׁה לָאוִין כוּ'. גְּמָרָא: קָאֵי אַרֵישָׁא דְּמַתְנִיתִין נוֹשְׂאִין עַל הָאֲנוּסָה כוּ'. וְהָא דְּאָסְרֵי לְנִטְעָן וְלֹא אָסְרוּ לִבְנֵיהֶן שֶׁל הַמַּתִּירִין דְּסוֹף פֶּרֶק ב', נַשֵׁי לְגַבֵּי נַשֵׁי שְׁכִיחִין, גַּבְרֵי לְגַבֵּי גַּבְרֵי לֹא שְׁכִיחִין. אִי נַמִּי נַשֵׁי דְּלֹא מִתְּסְרָא מִשְּׁכִיבַת הַמְזַנֶּה בִּקְרוֹבוֹתֶיהָ לֹא קַפְּדֵי אַהֲדָדֵי, גַּבְרָא דְּמִתְּסְרָא עָלָיו אִשְׁתּוֹ הַמְזַנָּה קַפְּדִי אַהֲדָדֵי. גְּמָרָא: שֶׁנֶּאֶסְרָה הַבִּיאָה עִמָּהֶם אִם הָיְתָה אִשְׁתּוֹ. הָרַמְבַּ"ם: וַהֲוֵי אוֹנֵס וּמְפַתֶּה עַל הַנְּשׂוּאָה. רַשִׁ"י:

פרק יא - משנה ב

וְהָא דִּתְנַן הַגִּיּוֹרֶת, מִשּׁוּם סֵיפָא דְּלֵדָתוֹ בִּקְדֻשָּׁה. וְהָא דִּתְנַן לֹא חוֹלְצִין וְאִינְהוּ לָאו אַחִין נִינְהוּ כְּלָל, אֶלָּא קָא מַשְׁמַע לָן אֲפִלּוּ מִדְּרַבָּנָן אֵינָן בְּתוֹרַת יִבּוּם וַחֲלִיצָה, וְאִי בָּעֵי לְמִנְסָב נַשֵׁי דַּהֲדָדֵי לְאַחַר מִיתָה אֲפִלּוּ יֵשׁ לָהֶם בָּנִים נַסְבֵי, וּבַגְּמָרָא שָׁקְלֵי וְטָרוּ בָּהּ. נִמּוּקֵי יוֹסֵף. וּמַסְקָנַת הַפּוֹסְקִים דִּכְשֶׁנְּשָׂאוּם לְאַחַר שֶׁנִּתְגַּיְּרוּ, דְּאִסּוּרָא מִדְּרַבָּנָן לְמִנְסָב כוּ': אֲפִלּוּ כוּ'. דְּאַף עַל גַּב דְּהוֹרָתוֹ שֶׁל רִאשׁוֹן שֶׁלֹּא בִּקְדֻשָּׁה וְכִשְׁתֵּי אִמָּהוֹת דַּמְיָא, וְאִיכָּא לְמֵימַר דְּתֶהֱוֵי עֲלֵיהּ כְּנָכְרִית וְאִי בָּעֵי לְיִבּוּמֵי מְיַבֵּם, אֲפִלּוּ הָכִי אֲסִירִי וְקַיְּמִי עֲלֵיהּ בְּאֵשֶׁת אָח. גְּמָרָא. וְאִי גַּם לֵדָתוֹ שֶׁל רִאשׁוֹן שֶׁלֹּא בִּקְדֻשָּׁה, יָכוֹל לִהְיוֹת דְּשָׁרֵי. תּוֹסָפוֹת:

פרק יא - משנה ג

דְּאִי מִשּׁוּם דְּלֹא מִתְרַמְיָא לְכָל חַד עוֹנָה בְּחֹדֶשׁ כְּדִלְעֵיל, הָא לֹא הֲוֵי אֶלָּא עֵצָה טוֹבָה, וְהוּא הַדִּין דִּטְפֵי נַמִּי:

פרק יא - משנה ד

אוֹ מְיַבְּמִין. פֵּרוּשׁ, אַחַר שֶׁבְּנֵי הַכַּלָּה חָלְצוּ תְּחִלָּה. הָרַמְבַּ"ם:

פרק יא - משנה ה

כְּלוֹמַר וְלֹא קַיְמָא לָן כְּוָתֵיהּ, דְּהַאי תַּנָּא רַבִּי יְהוּדָה הוּא וּלְטַעֲמֵיהּ דִּסְבִירָא לֵיהּ מַעֲלִין מִתְּרוּמָה לְיוֹחֲסִין, וְלֹא קַיְמָא לָן הָכִי כְּמוֹ שֶׁתְּנַן מִשְׁנָה ח' פֶּרֶק ב' דִּכְתֻבּוֹת: וְשִׁחְרְרוּ. בַּגְּמָרָא, אֵימָא, כּוֹפִין אוֹתָם וּמְשַׁחְרְרִין זֶה אֶת זֶה, דְּלִשָּׂא שִׁפְחָה אֵינוֹ יָכוֹל בַּת חוֹרִין אֵינוֹ יָכוֹל: אֵינָן מְשַׁלְּמִין כוּ'. אֲבָל צְרִיכִין לְהַפְרִישׁ מִשּׁוּם כַּפָּרָה. תּוֹסָפוֹת: וְאֵינָן חוֹלְקִים כוּ'. שֶׁהַמְשֻׁחְרָר יִשְׂרָאֵל הוּא. רַשִׁ"י. וַאֲפִלּוּ יָבֹאוּ שְׁנֵיהֶם עַל הַגֹּרֶן וְיִהְיוּ מַתְנִין אִם כֹּהֵן אֲנִי תְּנוּ לִי בִּשְׁבִילִי וְאִם חֲבֵרִי כֹּהֵן תְּנוּ לוֹ בִּשְׁבִילוֹ, אֲפִלּוּ הָכִי אֵין חוֹלְקִין כֵּיוָן דְּאֵין אוֹכְלִין, דִּלְמָא אָתֵי לִידֵי תַּקָּלָה. תּוֹסָפוֹת: וּמִלְּתָא דִּפְשִׁיטָא הִיא שֶׁאֲסוּרִים לְהַקְרִיב. אֶלָּא צָרִיךְ לוֹמַר דְּהָכִי קָאָמַר, שֶׁאֵין נוֹתְנִים כוּ' וְשֶׁהֵם יִתְּנוּ אַחַר כָּךְ לְכָל כֹּהֵן שֶׁיִּרְצוּ אֲפִלּוּ שֶׁאֵינוֹ בַּמִּשְׁמָר: כֵּן פֵּרֵשׁ רַשִׁ"י בְּלָשׁוֹן אַחֵר. וּלְאוֹתוֹ פֵּרוּשׁ מְפָרֵשׁ נַמִּי וְאֵין נוֹתְנִים לָהֶם בַּבְּכוֹר. אֲבָל לְפֵרוּשׁ קַמָּא שֶׁפֵּרֵשׁ הָרַ"ב לְהַקְרִיב, מְפָרֵשׁ וְאֵין מוֹצִיאִין כוּ' אִם הָיוּ חַיָּבִים חַטָּאת כוּ' אֵין כּוֹפִין אוֹתָם לְתִתָּם לְאַנְשֵׁי מִשְׁמָר וְלִהְיוֹת עֲבוֹדָתָם וְעוֹרָם לְאַנְשֵׁי מִשְׁמָר, דְּכָל חַד אָמַר אֲנָא כֹּהֵן, וְכֹהֵן מַקְרִיב בְּכָל שָׁעָה שֶׁיִּרְצֶה וַאֲפִלּוּ בְּמִשְׁמָר שֶׁאֵינוֹ שֶׁלּוֹ, אֶלָּא אוֹ יַקְרִיבוּ לָהֶם קָרְבְּנוֹתֵיהֶן וְיַחְזִירוּ לָהֶם הַבָּשָׂר וְהָעוֹר וְיִתְּנוּם לְכָל מִי שֶׁיִּרְצוּ, אוֹ אֵלּוּ יַעֲשׂוּ שָׁלִיחַ לְכָל כֹּהֵן שֶׁיִּרְצוּ לְהַקְרִיב כוּ'. וְהָכִי הֲוָה לֵיהּ לְהָרַ"ב לְפָרֵשׁ (וְעַיֵּן בַּתּוֹסָפוֹת) רֵישָׁא וְסֵיפָא בַּחֲדָא מַחֲתָא: וּבְכוֹרוֹ כוּ'. הַשְׁתָּא אוֹרֵי לָהּ מַאי עָבֵיד בִּבְכוֹר שֶׁלּוֹ. רַשִׁ"י: יְהֵא רוֹעֶה. אֲפִלּוּ בִּזְמַן הַבַּיִת, דְּהָא לֹא מָצֵי לְמִכְפִּינְהוּ לִתֵּן לַכֹּהֲנִים לְאַקְרוּבֵי וְלֶאֱכֹל אֶת בְּשָׂרוֹ, וְאִינְהוּ נַמִּי לֹא מָצֵי לְמִתְבֵיהּ לַכֹּהֵן וּלְהַחְזִיר לָהֶן הַבָּשָׂר, דִּבְכוֹר תַּם אֵינוֹ נֶאֱכָל אֶלָּא לַכֹּהֲנִים, הִלְכָּךְ יִרְעֶה , וּמֻתָּר לְשָׁחֲטוֹ בַּמְּדִינָה וְיֹאכְלוּהוּ הֵם, דִּבְכוֹר בַּעַל מוּם אֵין בּוֹ מִשּׁוּם זָרוּת אֶלָּא מִשּׁוּם גֶּזֶל דְּמַתְּנוֹת כְּהֻנָּה, וְהָכָא גֶּזֶל לֵיכָּא דְּהַמּוֹצִיא מֵחֲבֵרוֹ עָלָיו הָרְאָיָה. רַשִׁ"י: וַאֲפִלּוּ הָכִי צָרִיךְ קְמִיצָה, דְּמִנְחָה שֶׁלֹּא נִקְמְצָה פְּסוּלָה כְּזֶבַח שֶׁלֹּא נִשְׁחַט. רַשִׁ"י:

פרק יא - משנה ז

וְכֵן אָסוּר בַּעֲבוֹדָה, כִּדְאִיתָא רֵישׁ פֶּרֶק ב' דִּזְבָחִים: דְּאִלּוּ חָלָל אֵינוֹ מֻזְהָר עַל הַטֻּמְאָה. וּמִהוּא אוֹנֵן עֲלֵיהֶם לֹא מָצִינַן לְמֵידָק, דְּמָצִינוּ לְמֵימַר בְּמִיתָתוֹ דְּשֵׁנִי וְלִקּוּט עֲצָמוֹת דְּקַמָּא. עַיֵּן סוֹף פֶּרֶק ח' דִּפְסָחִים. וְלֹא מוֹקִים לָהּ שֶׁבָּאוּ עָלֶיהָ שְׁנֵיהֶם בִּזְנוּת דְּלֹא נַעֲשָׂה זַרְעָהּ חָלָל בְּכָךְ, מִשּׁוּם דַּחֲכָמִים גָּזְרוּ שֶׁמְּשַׁתְּקִין אוֹתוֹ מִדִּין כְּהֻנָּה הוֹאִיל וְאֵינוֹ יוֹדֵעַ אָבִיו הַוַּדָּאִי, שֶׁנֶּאֱמַר וְהָיְתָה לוֹ וּלְזַרְעוֹ אַחֲרָיו, עַד שֶׁיִּהְיֶה זַרְעוֹ מְיֻחָס אַחֲרָיו. גְּמָרָא: יוֹרְשִׁין אוֹתוֹ. בְּשֶׁלֹּא הִנִּיחַ זֶרַע. הָרַמְבַּ"ם: וּפָטוּר כוּ'. כְּשֶׁהִכָּה זֶה בִּזְמַן אֶחָד וְזֶה בִּזְמַן אַחֵר, דְּאִיכָּא שְׁתֵּי הַתְרָאוֹת וְכָל חֲדָא הִיא הַתְרָאַת סָפֵק, אֲבָל שְׁנֵיהֶם בְּיַחַד אוֹ שֶׁקִּלֵּל אוֹתָם בְּדִבּוּר אֶחָד, חַיָּב. הָרַמְבַּ"ם: וְתֵימַהּ לָמָּה לֹא יוּכְלוּ לְעַכֵּב וְלִדְחֹתוֹ מִן הָעֲבוֹדָה כְּמוֹ מִן הַחֲלֻקָּה. וּבַגְּמָרָא דַּיְקִינַן עוֹלֶה בְּעַל כָּרְחוֹ וְלֹא בְּעַל כָּרְחָם, דְּאַף עַל גַּב דְּאֵינוֹ חוֹלֵק עִם כָּל אֶחָד וְאֶחָד, שְׁתֵּי הַמִּשְׁפָּחוֹת כָּיְיפִין לֵיהּ לַעֲבֹד מִשּׁוּם פְּגָמָם כוּ': בְּמִשְׁמָר אֶחָד. וּבֵית אָב אֶחָד. גְּמָרָא. וְעַיֵּן מִשְׁנָה ז' פֶּרֶק ב' דְּתַעֲנִית:
