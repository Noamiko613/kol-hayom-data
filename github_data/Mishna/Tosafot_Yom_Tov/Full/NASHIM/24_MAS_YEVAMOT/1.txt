פרק א

חֲמֵשׁ עֶשְׂרֵה כוּ'. מִנְיָנָא דְּהָכָא וְכֵן דְּסֵיפָא הֲרֵי אֵלּוּ כוּ', לְמִעוּטֵי צָרַת אַיְלוֹנִית שֶׁהִכִּיר בָּהּ, אוֹ צָרַת קְטַנָּה מְמָאֶנֶת בְּיָבָם, גְּזֵרָה מִשּׁוּם צָרַת בִּתּוֹ מְמָאֶנֶת אוֹ צָרַת מַחֲזִיר גְּרוּשָׁתוֹ. גְּמָרָא: פּוֹטְרוֹת. לֹא תָּנֵי אוֹסְרוֹת, אַף עַל גַּב דְּוַדַּאי אִסּוּרָא אִיכָּא לְיַבֵּם, וְלַחְלֹץ נַמִּי שַׁיָּךְ לְמֵיסַר, שֶׁאִם אַתָּה אוֹמֵר חוֹלֶצֶת אָתֵי לְמֵימַר דְּתִתְיַבֵּם, אֶלָּא מִשּׁוּם שֶׁכֵּיוָן דְּבִמְקוֹם מִצְוָה הִיא דַּאֲסוּרָה וְשֶׁלֹּא בִּמְקוֹם מִצְוָה כְּגוֹן נָכְרִי שֶׁנַּעֲשָׂה חֲתָנוֹ וְלוֹ אִשָּׁה אַחֶרֶת וּמֵת מֻתֶּרֶת אִשְׁתּוֹ לְחָמִיו כְּדִתְנַן בְּמִשְׁנָה ג', מִשּׁוּם הָכִי תָּנֵי פּוֹטְרוֹת, דְּאִי תָּנֵי אוֹסְרוֹת מַשְׁמַע שֶׁהַבַּת אוֹסַרְתָּהּ עָלָיו לִקַּח, וְהוּא הַדִּין שֶׁלֹּא בִּמְקוֹם מִצְוָה. גְּמָרָא: פּוֹטְרוֹת צָרוֹתֵיהֶן. דְּאִלּוּ הֵן עַצְמָן דִּפְטוּרוֹת מִלְּתָא דִּפְשִׁיטָא הִיא וְלֹא אִיצְטְרִיךְ לְמִתְנֵי, דְּוַדַּאי דְּלֹא אָתֵי עֲשֵׂה דְּיִבּוּם וְדוֹחֶה לֹא תַעֲשֶׂה שֶׁיֵּשׁ בּוֹ כָּרֵת. וְהָא דִּכְתִיב עָלֶיהָ כְּדִמְפָרֵשׁ הָרַ"ב מִשּׁוּם דְּרָשָׁא דְּלִצוֹר דְּאִיצְטְרִיךְ לֵיהּ. גְּמָרָא: דְּאָמַר קְרָא כִּי כָּל אֲשֶׁר יַעֲשֶׂה מִכָּל הַתּוֹעֵבוֹת הָאֵלֶּה וְנִכְרְתָה, הֻקְשׁוּ כָּל הָעֲרָיוֹת כֻּלָּן לַהֲדָדֵי. גְּמָרָא: מִן הַחֲלִיצָה. שֶׁנֶּאֱמַר וְעָלְתָה יְבִמְתּוֹ הַשַּׁעְרָה וְגוֹ' מֵאֵן יַבְּמִי וְגוֹ' וְחָלְצָה נַעֲלוֹ, כָּל הָעוֹלֶה לְיִבּוּם עוֹלֶה לַחֲלִיצָה, וְכָל שֶׁאֵינוֹ עוֹלֶה לְיִבּוּם אֵינוֹ עוֹלֶה לַחֲלִיצָה אֶלָּא תִפָּטֵר: וּמִן הַיִּבּוּם. לְהָכִי לֹא סַגִּי לֵיהּ לְמִתְנֵי מִן הַחֲלִיצָה לְחֻדֵּיהּ, לְאַשְׁמְעִינַן דְּמִצְוַת חֲלִיצָה קוֹדֶמֶת. גְּמָרָא: בְּקַל וָחֹמֶר כוּ'. וְאַף דְּאֵין עוֹנְשִׁין מִן הַדִּין, הָכָא גִּלּוּי מִלְּתָא הִיא. אִי נַמִּי אַתְיָא הֵנָּה הֵנָּה וְאַתְיָא זִמָּה זִמָּה, וְטַעַם הַסֵּדֶר דְּלֹא כַּכָּתוּב עַיֵּן בַּגְּמָרָא: חֲמוֹתוֹ. שֶׁמֵּת בַּעֲלָהּ וּנְשָׂאָהּ אָחִיו וּמֵת בְּלֹא בָּנִים. רַשִׁ"י: דִּכְתִיב הָכָא כִּי יֵּשְׁבוּ אַחִים, וְהָתָם כְּתִיב שְׁנֵים עָשָׂר עֲבָדֶיךָ אַחִים. וְלֹא יָלְפִינַן מִלּוֹט דִּכְתִיב כִּי אֲנָשִׁים אַחִים אֲנַחְנוּ, דִּלְהָכִי כְּתִיב יַחְדָּו (הַמְּיֻחָדִים בְּנַחֲלָה). וְאִי כְּתִיב יַחְדָּו לְחוּד הֲוָה אֲמִינָא עַד דִּמְיַחֲדֵי בְּאַבָּא וּבְאִמָּא. גְּמָרָא: וְלֹא נָקְטָה בְּדִיּוּקָא, אֶלָּא דְּבָעֵי לְפֵרוּשֵׁי לְדִבְרֵי הַכֹּל. מִיהוּ אַלִּיבָּא דְּהִלְכָתָא קַיְמָא לָן כְּרַבָּנָן דְּפֶרֶק ב' מִשְׁנָה ב' דַּאֲסוּרָה עָלָיו בֵּין שֶׁנּוֹלַד וְאַחַר כָּךְ יִבֵּם וּבֵין שֶׁיִּבֵּם וְאַחַר כָּךְ נוֹלַד. נִמּוּקֵי יוֹסֵף: מִיֵּשְׁבוּ יָלְפִינַן, דְּיַחְדָּו הָא אִיצְטְרִיךְ לְעֵיל. וְאֵין לְהַקְשׁוֹת דְּלָמָּה אִיצְטְרִיךְ דְּרָשָׁא דְּיֵשְׁבוּ, הָא גַּבֵּי בְּנֵי יַעֲקֹב לֹא מָצִינוּ אַחִים אֶלָּא כְּשֶׁהָיוּ בָּעוֹלָם כְּאֶחָד, דִּפְשִׁיטָא דַּאֲפִלּוּ הָכִי אַחִין מִקְרוּ. תּוֹסָפוֹת: מִשּׁוּם דְּלִישְׁנָא דְּנִמְצְאוּ הָכִי דַיִּק, וְכָךְ מְפָרֵשׁ רַב אַסִי בַּגְּמָרָא דַּף י"ב. אֲבָל רָבָא פָּסַק הִלְכָתָא אֲפִלּוּ הִכִּיר בָּהּ אָחִיו וְקִיְּמָהּ וְהַוְיָא אִשְׁתּוֹ לֹא מִתְּסְרָא צָרָתָהּ מִשּׁוּם צָרַת הַבַּת. וּפֵרֵשׁ רַשִׁ"י דְּכִי אָסַר רַחֲמָנָא צָרַת עֶרְוָה בִּמְקוֹם מִצְוָה הוּא דַּאֲסָרָהּ, וְהָא כֵּיוָן דְּאַיְלוֹנִית הִיא דִּבְלָאו עֶרְוָה נַמִּי לֹא חַזְיָא לְיִבּוּם כְּדִתְנַן בְּמִשְׁנָה ה' פֶּרֶק ח', הַוְיָא צָרָתָהּ צָרַת עֶרְוָה שֶׁלֹּא בִּמְקוֹם מִצְוָה. וְאֶלָּא הָא דְּקָתָנֵי שֶׁנִּמְצְאוּ, תָּנֵי שֶׁהָיוּ:

מֵתָה כוּ'. הָא דְּלֹא נָקַט מֵאֲנָה אוֹ שֶׁנִּמְצֵאת אַיְלוֹנִית, מִשּׁוּם דְּמַתְנִיתִין אַתְיָא לְאַשְׁמְעִינַן אֲפִלּוּ כָּנַס וּלְבַסּוֹף גֵּרֵשׁ, וּבְמֵאֲנָה וְאַיְלוֹנִית אֵין שׁוּם חִדּוּשׁ, דְּבַטְלֵי לְהוּ קִדּוּשֵׁי קַמָּאִי כוּ'. אִי נַמִּי מִשּׁוּם דְּקָאֵי אַאַחַת מִכָּל הָעֲרָיוֹת הָאֵלּוּ, וּמֵאֲנָה וְאַיְלוֹנִית לֹא שַׁיָּךְ בְּכָל הָעֲרָיוֹת הָאֵלּוּ. תּוֹסָפוֹת: חוֹלֶצֶת כוּ'. וּבְרֵישׁ פֶּרֶק י"ג תְּנַן בֵּית הִלֵּל אוֹמְרִים מְמָאֶנֶת בַּיָּבָם, וְשָׁם מִשְׁנָה ז' אָמַר רַבִּי אֱלִיעֶזֶר מְלַמְּדִין שֶׁתְּמָאֵן, וְכֵן פָּסַק הָרַ"ב שָׁם. וּלְכֻלֵּי עָלְמָא אִם מֵאֲנָה מֵאֲנָה, אֲפִלּוּ הָכִי תְּנַן הָכָא דִּלְעוֹלָם צָרָתָהּ חוֹלֶצֶת וְלֹא מִתְיַבֶּמֶת, וְאֵין לָהּ תַּקָּנָה בְּמֵאוּן כְּדִמְפָרֵשׁ בַּגְּמָרָא, מִשּׁוּם דְּמִשְּׁעַת נְפִילָה נִרְאֵית כְּצָרַת בִּתּוֹ, וְהִלְכָּךְ גָּזְרוּ שֶׁלֹּא תְּמָאֵן לְיִבּוּמֵי צָרָתָהּ, דְּלֹא לְזִלְזוּלֵי בְּצָרַת עֶרְוָה דִּלְמָא אָתֵי לְמִשְׁרֵי:

וְהָנֵי חֲמֵשׁ עֶשְׂרֵה דִּלְעֵיל נַמִּי כְּשֶׁנִּשְּׂאוּ לַאֲחֵרִים צָרוֹתֵיהֶן מֻתָּרוֹת, וְאֵלּוּ חֲמוּרוֹת דַּאֲסוּרוֹת וְצָרוֹתֵיהֶן מֻתָּרוֹת:

וּבֵית הִלֵּל מַכְשִׁירִים. פְּשִׁיטָא. לַאֲפוֹקֵי דְּרַבִּי יוֹחָנָן בֶּן נוּרִי דְּאָמַר בֹּאוּ וּנְתַקֵּן לָהֶם לַצָּרוֹת שֶׁיִּהְיוּ חוֹלְצוֹת וְלֹא מִתְיַבְּמוֹת, קָא מַשְׁמַע לָן דְּבֵית הִלֵּל מַכְשִׁירִין, דַּאֲפִלּוּ תַּקַּנְתָּא דְּרַבָּנָן לֵית לְהוּ בְּצָרוֹת דִּלְחַלְּצוּ. גְּמָרָא: נִתְיַבְּמוּ וְכוּ'. הָא תוּ לָמָּה לִי, אַיְּדִי דְּתָנֵי חָלְצוּ תָּנֵי נַמִּי נִתְיַבְּמוּ. גְּמָרָא: לֹא נִמְנְעוּ בֵּית שַׁמַּאי כוּ'. לֹא נִצְרְכָה אֶלָּא לַצָּרָה עַצְמָהּ, דִּלְדִידְהוּ אֲסוּרָה לַשּׁוּק בְּלֹא תִהְיֶה וְגוֹ', דְּאִלּוּ בְּנֵי הַצָּרוֹת לֹא שַׁיָּךְ לְאִמְנוּעֵי, דִּבְנֵי חַיָּבֵי לָאוִין נִינְהוּ וּכְשֵׁרִים: כָּל כוּ'. מֵהָכָא שָׁמְעִינָא דְּמוֹדִיעֵי לְהוּ, דְּאִי מֵרֵישָׁא אִיכָּא לְמֵימַר דְּלֹא מוֹדִיעֵי, וְהָא דְּלֹא נִמְנְעוּ דְּצָרָה קָלָא אִית לָהּ וַהֲוֵי יָדְעֵי. גְּמָרָא: עוֹשִׂין כוּ'. לְבֵית הִלֵּל נַמִּי דְּאִיצְטְרִיךְ, דְּאַשְׁכְּחָן נַמִּי דְּמַחְמְרֵי בְּטֻמְאָה כְּדִתְנַן בְּמַסֶּכֶת עֵדֻיּוֹת. תּוֹסְפוֹת יוֹם טוֹב: לָאו אַפְּלוּגְתַיְהוּ פָּרְכִינַן, דְּעַד כָּאן לֹא נִשְׁנֵית פְּלֻגְתָּא, וְאַמַּאי לֹא פַּרְכַת עַלָהּ. אֶלָּא כִּי פָּרֵיךְ עַל לֹא נִמְנְעוּ וְכוּ' דְּאֵלּוּ עָשׂוּ כְּדִבְרֵיהֶם. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב:
