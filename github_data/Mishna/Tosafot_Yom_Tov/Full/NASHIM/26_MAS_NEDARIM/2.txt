מֻתָּרִין. הַלָּשׁוֹן אֵינוֹ עוֹלֶה שַׁפִּיר, דְּאַטּוּ הַנְהוּ דְפֶרֶק קַמָּא אֲסוּרִין הֵן שֶׁלֹּא יִדֹּר בָּהֶן. וְאֶפְשָׁר לִי לוֹמַר דְּסָתַם לָן תַּנָּא כְּרַבִּי מֵאִיר דְּדָרֵישׁ טוֹב אֲשֶׁר לֹא תִדֹּר וְגוֹ' (קֹהֶלֶת ה), טוֹב מִזֶּה וּמִזֶּה שֶׁאֵינוֹ נוֹדֵר כָּל עִקָּר, וְאַף עַל גַּב דְּהָתָם לְעִנְיַן נֶדֶר דְּאָמַר עָלַי לְחִיּוּבֵיהּ בְּקָרְבָּן קָאֵי, הָכָא נַמִּי אִיכָּא לְמֵיחַשׁ בְּכֻלְּהוּ נְדָרִים שֶׁמָּא יַעֲבֹר עַל אִסּוּרוֹ שֶׁאָסַר עַל נַפְשׁוֹ: הָרַ"ב לֹא גָּרַס בַּגְּמָרָא וּמִי אִיכָּא לְאַסּוֹקֵי כוּ' אֶלָּא מְחַוַּרְתָּא. וְכֵן הָרַמְבַּ"ם וְהָרֹא"שׁ לֹא גָּרְסֵי לֵיהּ: כִּבְשַׂר חֲזִיר. כָּל מִינֵי אִסּוּרִין קָתָנֵי. אֲסוּרִין בַּאֲכִילָה כִּבְשַׂר חֲזִיר, וַאֲסוּרִים בַּהֲנָאָה כַּעֲבוֹדַת כּוֹכָבִים, וַאֲסוּרִים בַּהֲנָאָה שֶׁאֵין לָהֶם בַּטָּלָה כְּעוֹרוֹת לְבוּבִין שֶׁהֵן תִּקְרֹבוֹת לַעֲבוֹדַת כּוֹכָבִים וְאֵין לָהֶם בִּטּוּל. וְתָנָא נַמִּי שֶׁאֲסוּרִים בַּאֲכִילָה בִּלְבַד וְיֵשׁ בָּהֶן טֻמְאָה כִּנְבֵלוֹת, וְשֶׁטֻּמְאָתָן חֲמוּרָה בְּכַעֲדָשָׁה כִּשְׁקָצִים וּרְמָשִׂים. הָרַ"ן: כִּרְמָשִׂים. כְּדִכְתִיב וְלֹא תְטַמְּאוּ אֶת נַפְשֹׁתֵיכֶם בְּכָל הַשֶּׁרֶץ הָרֹמֵשׂ עַל הָאָרֶץ (וַיִּקְרָא יא): כְּחַלַּת כוּ'. שֶׁאַף עַל פִּי שֶׁחָל אִסּוּר עַל יְדֵי קְרִיאַת שֵׁם, לֹא מִקְּרֵי דָבָר הַנָּדוּר, דְּהָא מֵעִקָּרָא נַמִּי אָסוּר מִשּׁוּם טֶבֶל, וּפִיו לֹא אָסַר אֶלָּא הִתִּיר, שֶׁהוֹצִיא הַשִּׁירַיִם מִידֵי טֶבֶל. הָרֹא"שׁ: מֻתָּר. מִשּׁוּם דְּקָא בָּעֵי לְמִתְנֵי הָאוֹמֵר לְאִשְׁתּוֹ כוּ' דְּצָרִיךְ פֶּתַח, וְאִי לֹא הָדַר וְתָנֵי הֲרֵי זֶה מֻתָּר לֹא יָדַעְנוּ וְאֵלּוּ מֻתָּרִין דְּרֵישָׁא כְּלוֹמַר שֶׁהֵן מֻתָּרִים בְּלֹא פֶתַח, אַהֵי מִנַּיְהוּ קָאֵי, אִי אַכֻּלְּהוּ אִי אַתַּרְתֵּי מִנַּיְהוּ. הָרַ"ן: וּלְכָךְ אֵינוֹ אוֹסֵר עָלָיו הָאֲכִילָה בְּאָמְרוֹ הֲרֵי הוּא כִּבְשַׂר חֲזִיר וְכַעֲבוֹדָה זָרָה, וְאָסַר עָלָיו כְּשֶׁאוֹמֵר כְּקָרְבָּן, לְפִי שֶׁהוּא יָכוֹל לִדֹּר קָרְבָּן וְלוֹמַר עַל מַה שֶּׁיִּרְצֶה הֲרֵי זֶה קָרְבָּן וְיִהְיֶה קָרְבָּן, וְאִי אֶפְשָׁר שֶׁיִּדֹּר וְיֹאמַר עַל שׁוּם דָּבָר הֲרֵי זֶה בְּשַׂר חֲזִיר אוֹ עֲבוֹדָה זָרָה. הָרַמְבַּ"ם: כְּאִמָּא. הוּא הַדִּין בְּמַתְפִּיס בִּשְׁאָר אִסּוּרֵי הֲנָאָה בַּעֲבוֹדָה זָרָה וְעָרְלָה כוּ'. וּבַגְּמָרָא נַמִּי תַנְיָא הָכִי, אֶלָּא הַאי דְּנָקַט כְּאִמָּא, חֲדָא דְּאִית בָּהּ תַּרְתֵּי אַשְׁמוֹעִינַן. חֲדָא, דְּאַף עַל גַּב דְּמַתְפִּיס בְּדָבָר הָאָסוּר, בְּעַם הָאָרֶץ מִדְּרַבָּנָן צָרִיךְ פֶּתַח. וְאַשְׁמוֹעִינַן תּוּ שֶׁצָּרִיךְ שֶׁיְּהֵא מִמָּקוֹם אַחֵר, אֲבָל אֵין פּוֹתְחִין לוֹ בִּכְבוֹד אִמּוֹ, כְּלוֹמַר אִלּוּ הָיָה יוֹדֵעַ שֶׁלֹּא יְהֵא כָבוֹד לְאִמּוֹ בְּכָךְ לֹא הָיָה נוֹדֵר. דְּלֹא תֵימָא נְהִי דְּאֵין פּוֹתְחִין לָאָדָם בִּכְבוֹד אָבִיו וְאִמּוֹ הָנֵי מִלֵּי בִנְדָרִים דְּאוֹרַיְתָא אֲבָל הַאי דְּלֵיתָא אֶלָּא מִדְּרַבָּנָן פּוֹתְחִין, קָא מַשְׁמַע לָן דְּלֹא. וְדַוְקָא בְּמַדִּיר אִשְׁתּוֹ וּמַתְפִּיס בְּדָבָר הָאָסוּר גָּזְרוּ אַטּוּ מַתְפִּיס בְּדָבָר הַנָּדוּר, אֲבָל שְׁאָר נְדָרִים שֶׁאֵינָן מְצוּיִין לֹא גָזְרוּ. הָרַ"ן: כְּלוּם אַתָּה מִתְחָרֵט חֲרָטָה דְּמֵעִקָּרָא, מִלְּשׁוֹן תּוֹהֶה עַל הָרִאשׁוֹנוֹת. הָרַ"ן: כְּלוֹמַר עוֹמֵד אַתָּה עַל דַּעְתְּךָ הָרִאשׁוֹנָה, שֶׁרְצוֹנְךָ עֲדַיִן בְּאוֹתוֹ הַנֶּדֶר: שֶׁאֲנִי. מִשּׁוּם דִּנְדָרִים מַתְפִּיס חֶפְצָא אַנַּפְשֵׁיהּ תָּנֵי בְהוּ בִּלְשׁוֹן שֶׁאֲנִי, כְּלוֹמַר מַה שֶּׁאֲנִי. וּמִשּׁוּם דִּשְׁבוּעוֹת אָסַר נַפְשֵׁיהּ תָּנֵי לְהוּ בִּלְשׁוֹן שֶׁאֵינִי בְּיוּ"ד. הָרַ"ן (וְעַיֵּן בַּתּוֹסְפוֹת יוֹם טוֹב): גְּמָרָא, דְּתַנְיָא דְּבָרִים הַמֻּתָּרִים וַאֲחֵרִים נָהֲגוּ בָהֶם אִסּוּר אִי אַתָּה רַשַּׁאי לְהַתִּירָן בִּפְנֵיהֶם, שֶׁנֶּאֱמַר לֹא יַחֵל דְּבָרוֹ. וְכָתְבוּ הַתּוֹסָפוֹת תֵּימַהּ מָה עִנְיָן זֶה לְנָהֲגוּ בוֹ אִסּוּר, וְיֵשׁ לוֹמַר דְּהַאי נַמִּי שֶׁנָּדַר וְנִתְכַּוֵּן לֶאֱסֹר עַצְמוֹ גַּם בְּדָבָר שֶׁאֵין בּוֹ מַמָּשׁ, אֵין לְךָ נָהֲגוּ גָּדוֹל מִזֶּה: וְאִם תֹּאמַר אַמַּאי מִתְּסַר, וְהָא מִצְוֹת לָאו לֵהָנוֹת נִתְּנוּ. וְיֵשׁ לוֹמַר דְּהַיְנוּ לוֹמַר שֶׁאֵין קִיּוּם הַמִּצְוָה חָשׁוּב הֲנָאָה, אֲבָל מִכָּל מָקוֹם אִי מִתְהַנֵי גוּפֵיהּ בַּהֲדֵי דִּמְקַיֵּם מִצְוָה הֲנָאָה מִקַּרְיָא. וּמִשּׁוּם הָכִי אָמְרִינַן בְּפֶרֶק רָאוּהוּ בֵּית דִּין הַנּוֹדֵר הֲנָאָה מִן הַמַּעְיָן טוֹבֵל בּוֹ בִּימוֹת הַגְּשָׁמִים אֲבָל לֹא בִּימוֹת הַחַמָּה. הָרַ"ן. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: שְׁבוּעָה כוּ'. מַתְמַהּ הַתּוֹסְפוֹת יוֹם טוֹב, אַמַּאי לֹא תָּנֵי שְׁבוּעָה שֶׁאֵינִי מְשַׁמְּשֵׁךְ: מִשּׁוּם שְׁבוּעַת שָׁוְא. הָרַמְבַּ"ם:

וְקָרְבָּן לָאו בַּר אִשְׁתְּבוֹעֵי הוּא, וְאִי אָמְרִינַן דְּקָרְבָּן יְהֵא מַה שֶּׁלֹּא אֹכַל לָךְ נַמִּי שָׁרֵי, דְּלֹא שַׁוְּיֵהּ אִסּוּר קָרְבָּן אֶלָּא מַה שֶּׁלֹּא יֹאכַל. הָרַ"ן. וְאִם תֹּאמַר כֵּיוָן דִּבְאוֹמֵר לֹא חֻלִּין לֹא אֹכַל לָךְ אָסוּר מִשּׁוּם דַּהֲוֵי כְּאוֹמֵר מַה שֶּׁאֹכַל לְךָ לֹא יְהֵא חֻלִּין אֶלָּא קָרְבָּן לְפִיכָךְ לֹא אֹכַל לְךָ, כְּשֶׁאוֹמֵר קָרְבָּן לֹא אֹכַל לָךְ נַמִּי נֵימָא הָכִי וְלִתְּסַר, וְיֵשׁ לוֹמַר דְּהָתָם שָׁאנֵי דְלֵיכָּא לְפָרוֹשֵׁי בֵּיהּ שֶׁנִּשְׁבַּע בְּקָרְבָּן שֶׁלֹּא יֹאכַל לֵיהּ, אֲבָל לִישָׁנָא דְהָכָא כֵּיוָן דְּאִיכָּא לְפַרְשִׁינְהוּ הָכִי אַף עַל גַּב דִּסְתַם נְדָרִים לְהַחְמִיר מֵאַחַר דְּפֵרוּשָׁא דִּלְקֻלָּא מִשְׁתַּמְּעִי טְפֵי בְּלִישָׁנָא הָכִי מְפָרְשִׁינַן לְהוּ וְלֹא לְחֻמְרָא. בֵּית יוֹסֵף: אָסוּר. וְאַף עַל גַּב דְּמַתְנִיתִין רַבִּי מֵאִיר הִיא דְּלֵית לֵיהּ מִכְּלַל לָאו כוּ', אֲפִלּוּ הָכִי בִּשְׁבוּעָה אָסוּר. וְנִרְאֶה דְּהַיְנוּ טַעֲמָא דְּמַסְּקִינַן בְּסוֹף פֶּרֶק שְׁבוּעַת הָעֵדוּת כוּ' דִּבְאִסּוּרָא גְּרֵידָא אִית לֵיהּ כוּ', וּשְׁבוּעוֹת אֵינָן אֶלָּא אִסּוּר גְּרֵידָא שֶׁהוּא חָל בְּגוּפוֹ, אֲבָל נְדָרִים אֵינָן חָלִים עַל דָּבָר שֶׁאֵין בּוֹ מַמָּשׁ. הָרַ"ן: וְהַטַּעַם בָּזֶה, שֶׁהַשְּׁבוּעָה חָלָה עַל גּוּף הָאָדָם, וְהַגּוּף יֵשׁ בּוֹ מַמָּשׁ. טוּר: וְחֹמֶר כוּ'. לֹא דָיְקִינַן מִכְּלָל דִּשְׁבוּעָה מִיהָא אִיתָא, מִשּׁוּם דְּאִין הָכִי נַמִּי דְאִיתָא, וְלָקֵי בָּהּ קָשֶׁה, דְּמַה הֻצְרַךְ לִשְׁנוֹת וּפֵרוּשָׁן לְהָקֵל, וְעוֹד כוּ'. וְיֵשׁ מְפָרְשִׁים אַף עַל פִּי שֶׁסְּתָם נְדָרִים לְהַחְמִיר הַיְנוּ כְּשֶׁלֹּא פֵּרֵשׁ מָה הָיָה בְלִבּוֹ אֶלָּא אָמַר כָּךְ נָדַרְתִּי כְּמוֹ שֶׁפֵּרְשׁוּהָ חֲכָמִים, אָנוּ מְפָרְשִׁים לְחֻמְרָא. אֲבָל אִם אָמַר הֲרֵי עָלַי כְּבָשָׂר מָלִיחַ וְחָזַר וְאָמַר בְּלִבִּי הָיָה כְּשֶׁל עֲבוֹדַת כּוֹכָבִים אָזְלִינַן בָּתַר פֵּרוּשׁוֹ לְהָקֵל. הָרֹא"שׁ. וְעַיֵּן עוֹד: מִדְּלֹא שָׁתַק. הָרֹא"שׁ: עֲבוֹדָה זָרָה. דְּגַם לַעֲבוֹדַת כּוֹכָבִים רְגִילוּת הִיא לִמְלֹחַ, וּלְהָכִי לֹא קָתָנֵי שֶׁל חֻלִּין מִשּׁוּם דְּאֵין רְגִילוּת לִקְרוֹת בָּשָׂר מָלוּחַ אֶלָּא אֵלּוּ דְּקָתָנֵי שֶׁל שָׁמַיִם וְשֶׁל עֲבוֹדַת כּוֹכָבִים וְלֹא שֶׁל חֻלִּין. תּוֹסָפוֹת: וְאִם תֹּאמַר וַהֲלֹא עֲשִׂירִי מֵאֵלָיו הָוֵי קָדוֹשׁ. וְיֵשׁ לוֹמַר דְּבָעִינַן הַעֲבָרַת מִנְיָן, וְאִם הָיוּ עֲשָׂרָה וְנָטַל אֶחָד בְּלֹא מִנְיָן אֵינוֹ קָדוֹשׁ כִּדְאָמַר פֶּרֶק ט' דִּבְכוֹרוֹת. תּוֹסָפוֹת: וְלֹא הָיוּ רְגִילִין בִּתְרוּמַת הַלִּשְׁכָּה, לֹא קָרוּ לַהּ תְּרוּמָה סְתָמָא, אֶלָּא תְרוּמַת הַלִּשְׁכָּה קָרוּ לַהּ, הִלְכָּךְ כִּי נָדְרוּ בִתְרוּמָה סְתָם לֵיכָּא לְסַפּוֹקֵי כְּלָל בִּתְרוּמַת הַלִּשְׁכָּה. הָרַ"ן: פֵּרֵשׁ רַשִׁ"י שֶׁלֹּא הָיוּ כֹּהֲנִים בֵּינֵיהֶם. וּמְסַיֵּם הָרַ"ן וּלְחֶרְמֵי כֹהֲנִים לֹא קָרוּ לְהוּ חֵרֶם סְתָם, הִלְכָּךְ כִּי נָדְרֵי בְחֵרֶם לֵיכָּא לְסַפּוֹקֵי בְמִידֵי, דְּוַדַּאי חֵרֶם גָּבוֹהַּ קָאָמְרֵי. וּבַגְּמָרָא, הָא מַכִּירִין מֻתָּרִין, אָתְיָא כְרַבִּי אֶלְעָזָר בֶּן צָדוֹק דְּפָלֵיג אָרֵישָׁא וּסְבִירָא לֵיהּ סְפֵקָא לְקֻלָּא:
