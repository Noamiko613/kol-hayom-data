פרק ג - משנה א

לְגָרֵשׁ כוּ'. כֻּלְּהוּ יָלֵיף לְהוּ בַּגְּמָרָא מִקְּרָאֵי: אֲפִלּוּ לְמַאן דְּאָמַר בְּעָלְמָא יֵשׁ בְּרֵרָה, הָכָא מוֹדֶה, מִשּׁוּם דִּוְכָתַב לָהּ, לִשְׁמָהּ, מַשְׁמַע שֶׁיְּהֵא מְבֹרָר בִּשְׁעַת כְּתִיבָה. תּוֹסָפוֹת:

פרק ג - משנה ב

מְקוֹם כוּ'. וּמַשְׁמַע שֵׁם עִירוֹ וְשֵׁם עִירָהּ דְּלֹא. הַגָּהוֹת אָשְׁרִי: שֶׁגַּם זֶה עִקָּרוֹ שֶׁל גֵּט. וּמַתְנִיתִין רַבִּי אֶלְעָזָר הִיא דְאָמַר עֵדֵי מְסִירָה כָרְתֵי, הִלְכָּךְ וְכָתַב לָהּ לָאו אַחֲתִימָה קָאֵי, דַּחֲתִימַת הָעֵדִים אֵינָהּ אֶלָּא מִדִּבְרֵי סוֹפְרִים מִפְּנֵי תִקּוּן הָעוֹלָם, וְכִי כְתִיב לָהּ לִשְׁמָהּ אַעִקַּר כְּרִיתוּת קָאֵי, דְּהַיְנוּ כָּל הָנֵי דְּאָמְרָן: חוּץ כוּ'. קָשְׁיָא דְרַבִּי אֶלְעָזָר אַדְּרַבִּי אֶלְעָזָר, דְּרֵישָׁא דְמַתְנִיתִין אוֹקְמִינַן כְּרַבִּי אֶלְעָזָר וְשָׁרֵי לִכְתֹּב לְטָפְסֵי גִּטִּין וּבִלְבַד שֶׁיַּנִּיחַ. תְּרֵי תַנָּאֵי אַלִּיבָּא דְרַבִּי אֶלְעָזָר. גְּמָרָא: וְלֹא קָשֶׁה מִמִּשְׁנָה ה' פֶּרֶק דִּלְעֵיל דְּשָׁרִינַן בְּחֵרֵשׁ שׁוֹטֶה וְקָטָן אֲפִלּוּ לְכַתְּחִלָּה לִכְתֹּב טֹפֶס וְלֹא גָזְרִינַן אַטּוּ תּוֹרֵף, דְּחֵרֵשׁ שׁוֹטֶה וְקָטָן לֹא שְׁכִיחַ כְּלָל לֹא גָזְרוּ. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב שֶׁמֵּבִיא פְּלוּגְתַּת הַפּוֹסְקִים בְּגִטֵּי נָשִׁים אִי פָּסוּל אֲפִלּוּ דִיעֲבַד:

פרק ג - משנה ג

וּבַגְּמָרָא פְלִיגֵי, אִיכָּא מַאן דְּאָמַר כְּדֵי שֶׁיִּהְיֶה אָדָם עוֹמֵד וְרוֹאֶה שֶׁלֹּא עָבַר אָדָם שָׁם, וְאִיכָּא מַאן דְּאָמַר שֶׁלֹּא שָׁהָה אָדָם שָׁם עַד שְׁעַת מְצִיאַת הַגֵּט: לְהָרֹא"שׁ פָּסוּל וַאֲפִלּוּ לֹא הֻחְזְקוּ שְׁנַיִם שֶׁשְּׁמוֹתֵיהֶן שָׁוִים, כָּל זְמַן שֶׁלֹּא רָאָה שֶׁלֹּא עָבַר אָדָם שָׁם. וּלְהָרַמְבַּ"ם אֲפִלּוּ עָבַר, כָּשֵׁר, דְּהִלְכָתָא כְּמַאן דְּאָמַר שֶׁלֹּא שָׁהָה כוּ', אֶלָּא אִם כֵּן הֻחְזְקוּ גַּם כֵּן, דְּאָז בְּעָבַר הֲוָה תַרְתֵּי לְרֵיעוּתָא. וּמוּבָן גַּם כֵּן בְּאֵין הַשַּׁיָּרוֹת מְצוּיוֹת. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: פֵּרְשׁוּ הַתּוֹסָפוֹת דְּהַשָּׁלִיחַ נַמִּי נֶאֱמָן בְּסִימָן מֻבְהָק. אֶלָּא דִבְעֵדִים אַף עַל פִּי שֶׁרָאוּהוּ כְּבָר מְהֵימְנוּ, וְשָׁלִיחַ לֹא מְהֵימָן אֶלָּא שֶׁאָמַר כֵּן קֹדֶם שֶׁרָאָהוּ: מַשְׁמַע שֶׁיּוֹדְעִים שֶׁלֹּא חָתְמוּ אֶלָּא עַל אֶחָד, אֲבָל אֵין מַכִּירִין מִי הוּא אוֹתוֹ יוֹסֵף בֶּן שִׁמְעוֹן, וּכְשֶׁהָעֵדִים אוֹמְרִים כֵּן וְהוּא אוֹמֵר שֶׁעַל שֶׁלּוֹ חָתְמוּ, נֶאֱמָן, דְּלֹא חָשְׁדִינַן לֵיהּ שֶׁיְּשַׁקֵּר בְּמֵזִיד לוֹמַר שֶׁהוּא שֶׁלּוֹ לְקַלְקְלָהּ. אֲבָל כְּשֶׁאֵין הָעֵדִים אוֹמְרִים כְּלוּם חָיְשִׁינַן שֶׁמָּא הוּא סוֹבֵר שֶׁהוּא שֶׁלּוֹ, לְפִי שֶׁאֵינוֹ יוֹדֵעַ שֶׁיֵּשׁ יוֹסֵף בֶּן שִׁמְעוֹן אַחֵר, אוֹ יוֹדֵעַ וְאֵין נִרְאֶה לוֹ לָחוּשׁ, שֶׁגַּם הוּא אִבֵּד גֵּט, וּלְכָךְ אוֹמֵר שֶׁהוּא מַכִּיר אַף עַל פִּי שֶׁאֵינוֹ מַכִּיר. תּוֹסָפוֹת: וְאִם תֹּאמַר וְנֵחוּשׁ לִשְׁאֵלָה, וְיֵשׁ לוֹמַר דְּקִים לֵיהּ בְּנַפְשֵׁיהּ שֶׁלֹּא הִשְׁאִילָם לְשׁוּם אָדָם. אִי נַמִּי בְּיוֹדֵעַ שֶׁבַּחֲפִיסָה אָבְדוּ, וּמִכָּל מָקוֹם צָרִיךְ שֶׁיְּהֵא מַכִּיר הַחֲפִיסָה בִטְבִיעוּת עַיִן: בִּטְבִיעוּת עֵינָא. וּבַחֲפִיסָה וּדְלֻסְקְמָא נַמִּי טְבִיעוּת עַיִן כְּמוֹ סִימָן. אֶלָּא דְאִיכָּא בֵּין סִימָן לִטְבִיעוּת עַיִן, דִּטְבִיעוּת עַיִן אֵין מוֹעִיל לְהַכְשִׁירוֹ אֶלָּא כְשֶׁמְּצָאוֹ אוֹתוֹ בְעַצְמוֹ שֶׁאָבַד לוֹ, וְלֹא כְּשֶׁמְּצָאוּהוּ אֲחֵרִים. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: אוֹ חוֹלֶה. מַה שֶּׁאֵין כֵּן גּוֹסֵס, דְּרֹב גּוֹסְסִים לַמִּיתָה. גְּמָרָא: בְּחֶזְקַת כוּ'. מִדְּאוֹרַיְתָא נָפְקָא לָן, בְּחֻלִּין דַּף י', הַעֲמֵד דָּבָר עַל חֶזְקָתוֹ. רַשִׁ"י: בַּת יִשְׂרָאֵל. צְרִיכָא, דְאִי אַשְׁמְעִינַן גֵּט, מִשּׁוּם דְּלֹא אֶפְשָׁר (לְמֵיחַשׁ לְמִיתָה, דְּאִי חָיְישַׁת אֵין לְךָ שׁוֹלֵחַ גֵּט לְאִשְׁתּוֹ, וְהֵן עֲגוּנוֹת). אֲבָל תְּרוּמָה דְאֶפְשָׁר (לִהְיוֹת נִזּוֹנוֹת בְּחֻלִּין) אֵימָא לֹא, צְרִיכָא. גְּמָרָא: הַשּׁוֹלֵחַ. צְרִיכָא, דְאִי אַשְׁמְעִינַן תְּרוּמָה, זִמְנִין דְּלֹא אֶפְשָׁר (שֶׁהִיא עֲנִיָּה וְתִטֹּל בַּגְּרָנוֹת). אֲבָל חַטָּאת מִסְּפֵקָא לֹא לֵיעוּל חֻלִּין לָעֲזָרָה, צְרִיכָא. גְּמָרָא: מַקְרִיבִין. בַּגְּמָרָא מְפָרֵשׁ דְּדַוְקָא בְּקָרְבַּן נָשִׁים, אִי נַמִּי חַטַּאת הָעוֹף, שֶׁאֵינָם צְרִיכִים סְמִיכָה. דְּאִי לָאו הָכִי לְכַתְּחִלָּה אֵין לַעֲשׂוֹת בְּלֹא סְמִיכָה, דְּהָא מַעֲלִין עָלָיו כְּאִלּוּ לֹא כִּפֵּר:

פרק ג - משנה ד

וְכֵן פֵּרֵשׁ רַשִׁ"י. וּמַשְׁמַע דְּאִלּוּ אָבְדָה דִבְסֵיפָא הַיְנוּ שֶׁטָּבְעָה. וְקָא מַשְׁמַע לָן דְּאַף עַל פִּי כֵן נוֹתְנִין עָלָיו חֻמְרֵי חַיִּים דְּשֶׁמָּא יָצָא, (כִּדְאִיתָא בִּיבָמוֹת פט"ז מ"ד) דִּסְתָם יָם מַיִם שֶׁאֵין לָהֶם סוֹף הוּא. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב בְּשֵׁם הָרַמְבַּ"ם: לֵהָרֵג. בַּגְּמָרָא אִיכָּא תְּרֵי לִישְׁנֵי, לְחַד לִישָׁנָא דַוְקָא בְּבֵית דִּין נָכְרִים, אֲבָל בְּבֵית דִּין יִשְׂרָאֵל אֵין נוֹתְנִין עָלָיו חֻמְרֵי חַיִּים (וְהוּא דַעַת הָרַמְבַּ"ם). וּלְאִידָךְ לִישָׁנָא דַוְקָא בְּבֵית דִּין יִשְׂרָאֵל נוֹתְנִין עָלָיו חֻמְרֵי חַיִּים. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: בַּת יִשְׂרָאֵל כוּ'. וְכָל שֶׁכֵּן לְאִסּוּר אֵשֶׁת אִישׁ. וְאִם הָיָה גֵט אֶחָד מֵהֶם בְּיַד הַשָּׁלִיחַ אֵינוֹ נוֹתְנוֹ לְאִשְׁתּוֹ, וְאִם נוֹתְנוֹ לָהּ הֲרֵי זוֹ סְפֵק מְגֹרֶשֶׁת. וּכְמוֹ שֶׁכָּתַב הָרַמְבַּ"ם:

פרק ג - משנה ה

אַחֵר. דְּשָׁלִיחַ עוֹשֶׂה שָׁלִיחַ כִּדְאִיתָא בְקִדּוּשִׁין דַּף מ"א. וְדַוְקָא חָלָה. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: וּמַשְׁמַע דְּלֹא קָפֵיד אִם יִטֹּל הַחֵפֶץ תְּחִלָּה אִם לֹא, וַהֲרֵי זֶה כְּאוֹמֵר תֵּן לָהּ הַגֵּט וְטֹל הַחֵפֶץ, שֶׁגַּם בָּזֶה לֹא יְשַׁלְּחֶנּוּ בְּיַד אַחֵר כוּ'. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב:

פרק ג - משנה ו

וּבְחָלָה, דְּלֹא עָדִיף מִמִּי שֶׁבָּא מִכֹּחוֹ. פּוֹסְקִים:

פרק ג - משנה ז

כֵּיוָן דְּמִלְּתָא דִפְשִׁיטָא הוּא דִלְדִידְהוּ יָהֵיב לְהוּ אַסָּחֵי לְהוּ שְׁאָר כֹּהֲנִים דַּעְתַּיְהוּ וְהָוֵי כְּמַאן דְּמָטוּ לִידַיְהוּ דְּהָנֵי. רַשִׁ"י וְהָרַ"ן. וּבַהַגָּהוֹת אָשְׁרִי יָהֵיב טַעֲמָא מִשּׁוּם דַּהֲוֵי כְמַתָּנָה מֻעֶטֶת וְאָסוּר לַחֲזֹר בּוֹ: וַאֲפִלּוּ לֹא יָרַשׁ אֶלָּא מְלֹא מַחַט, מִשּׁוּם דְּגוֹבֶה וְחוֹזֵר וְגוֹבֶה. גְּמָרָא. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: בֵּית דִּין. עַל מְנָת שֶׁיַּפְרִישׁ עַל מָעוֹת אֵלּוּ מֵחֶלְקָם. הָרַמְבַּ"ם: אֵינוֹ צָרִיךְ כוּ'. דְּהָיָה כֹּחַ בְּיַד בֵּית דִּין לְהָטִיל חוֹב זֶה עַל הַכְּהֻנָּה וְהַלְוִיָּה, לְפִי שֶׁתַּקָּנָתָם הוּא שֶׁיִּמְצְאוּ מָעוֹת לַהַלְוָאָה. רַשִׁ"י. אֲבָל עַל הֶעָנִי שֶׁהֶעֱשִׁיר לֹא עָבְדוּ תַּקַּנְתָּא, דְּלָא שְׁכִיחַ:

פרק ג - משנה ח

מַפְרִישׁ כוּ'. בְּעַרְבֵי שַׁבָּתוֹת וְיָמִים טוֹבִים אַיְרֵי, דְּשָׁרֵי שֶׁלֹּא מִן הַמֻּקָּף. תּוֹסָפוֹת. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: וּמַשְׁמַע דְּאִלּוּ כְּבָר אֲכָלָן אֵין מְחַיְּבִין אוֹתוֹ לְהַפְרִישׁ, וְתָלִינַן שֶׁעַכְשָׁיו נֶאֶבְדוּ הוֹאִיל וּכְבָר נֶאֶכְלוּ, וְלֵיכָּא אֶלָּא חִיּוּב הַפְרָשַׁת תְּרוּמוֹת וּמַעַשְׂרוֹת. וְהָרַמְבַּ"ם לֹא חִלֵּק. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: בְּקָדִים כוּ'. בַּגְּמָרָא תָנָא בַּתְּקוּפָה, כְּלוֹמַר שֶׁמּוֹצָאֵי הַחַג הִיא בִתְקוּפַת תִּשְׁרֵי, דְּאִלּוּ נִמְשְׁכָה תְקוּפַת תַּמּוּז עַד כָּאן לֹא. רַשִׁ"י. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: סְמָדַר. כְּתֹם פֶּרַח, וַעֲנָבָיו נִרְאִין בָּאֶשְׁכּוֹל כְּסִדְרָן. רַשִׁ"י: וּרְאָיָה לָזֶה מִדִּתְנַן הַבֹּסֶר מִשֶּׁהֵבִיא מַיִם:
