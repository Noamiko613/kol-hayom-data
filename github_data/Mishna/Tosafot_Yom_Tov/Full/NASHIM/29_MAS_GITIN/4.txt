פרק ד

שֶׁשָּׁלַח. מַהוּ דְּתֵימָא לֹא אַלִּימָא שְׁלִיחוּתָא דְבַתְרָא מִשְּׁלִיחוּתָא דְקַמָּא דִלְבַטְּלֵיהּ, קָא מַשְׁמַע לָן. גְּמָרָא: קָדַם כוּ'. מַהוּ דְּתֵימָא כִּי לֹא אָמְרִינַן לִצְעוּרָהּ קָמְכַוֵּן הָנֵי מִלֵּי לַשָּׁלִיחַ, אֲבָל לְדִידָהּ וַדַּאי לִצְעוּרָהּ קָמְכַוֵּן, קָא מַשְׁמַע לָן. גְּמָרָא. וּפֵרֵשׁ הָרַ"ן דְּדוּמְיָא דְּשָׁלִיחַ, דְּהַיְנוּ דְּפָגַע מִמֵּילָא, הָכִי נַמִּי בְאִשְׁתּוֹ דְּפָגַע בָּהּ מִמֵּילָא. וְקָא מַשְׁמַע לָן דְּלֹא אָמְרִינַן לִצְעוּרָהּ כוּ'. וְקָדַם לָאו דַּוְקָא אֶלָּא אוֹרְחָא דְמִלְּתָא נָקַט. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: שָׁלִיחַ. מַהוּ דְּתֵימָא אִיהוּ הוּא דְלָא טָרַח אַדַּעְתָּא לִצְעוּרָהּ, אֲבָל שָׁלִיחַ דְּלֹא אִכְפַּת לֵיהּ כִּי טָרַח וַדַּאי לִצְעוּרָהּ. גְּמָרָא. וְטָרַח דְּנָקַט לָאו טִרְחַת הַדֶּרֶךְ וּרְדִיפָה אַבָּתְרַהּ קָאָמַר, דְּהָכָא מַיְרֵי בְּפָגַע, אֶלָּא טִרְחַת הַדִּבּוּר שֶׁיֵּשׁ טֹרַח וְעָמָל עַל הָאָדָם לְצַעֵר הָאָדָם בְּדִבּוּר עַל יְדֵי עַצְמוֹ, מַה שֶּׁאֵין כֵּן כְּשֶׁיְּדַבֵּר לַשָּׁלִיחַ. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב:

זֶה דַעַת רַב שֵׁשֶׁת וְרֵישׁ לָקִישׁ בַּגְּמָרָא, דְּדָיְקֵי דְבֵית דִּין הַיְנוּ שְׁלֹשָׁה. וְטַעְמָא דְבָעִינַן שְׁלֹשָׁה וַהֲרֵי אֵין זֶה אֶלָּא עֵדוּת בְּעָלְמָא, כָּתַב הָרַ"ן דְּתַקָּנָה קְדוּמָה הָיְתָה כְדֵי שֶׁיְּהֵא לַדָּבָר קוֹל. עַד כָּאן. דְּבֵי תְּרֵי לֹא מַפְקֵי קָלָא, דְּהָכָא בָּעִינַן גִּלּוּי מִלְּתָא טְפֵי לִגְבּוֹת מִמְּשַׁעְבְּדֵי כוּ'. וּלְדִידְהוּ חֲשַׁשׁ מַמְזֵרוּת לֵיכָּא, דְּהָא שְׁלֹשָׁה מַפְקֵי לְקָלָא. אֶלָּא טַעֲמָא מִשּׁוּם תַּקָּנַת עֲגוּנוֹת, דְּכֵיוָן דְּאַטְרְחִינַן עֲלֵיהּ לֵילֵךְ כוּ' לֹא יִטְרַח בִּשְׁבִיל לְעָגְנָהּ. וְרַב נַחְמָן וְרַב יוֹחָנָן סָבְרֵי דְבֵית דִּין הַיְנוּ תְּרֵי, וְקָרִי לְהוּ בֵּית דִּין מִשּׁוּם דְּאֵין דָּבָר שֶׁבָּעֶרְוָה פָחוֹת מִשְּׁנַיִם. וַאֲפִלּוּ הֵיכָא דְלָא חָיְשִׁינַן לְשִׁקְרֵי. וְכֵן פָּסְקוּ כָּל הַפּוֹסְקִים. וּלְדִידְהוּ טַעֲמָא מִשּׁוּם חֲשַׁשׁ מַמְזֵרוּת. וְאִם כֵּן הָרַ"ב מְזַכֵּי שְׁטָרֵי אַבֵּי תְּרֵי וּדְלֹא כְּהִלְכָתָא. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: וְהַיְנוּ מַכַּת מַרְדּוּת. דְּוַדַּאי אֵין כָּאן מַלְקוֹת דְּאוֹרַיְתָא. וְטַעְמָא דְמַלְקִין מִשּׁוּם דְּקָיְמָא לָן דְּתַקַּנְתָּא דְרַבָּן גַּמְלִיאֵל לֹא הָיְתָה אֶלָּא לְכַתְּחִלָּה, אֲבָל בְּדִיעֲבַד אִם בִּטְלוֹ, מִבְטַל. וּמִיהוּ דַּוְקָא לְאַחַר תַּקָּנַת רַבָּן גַּמְלִיאֵל, אֲבָל מַה שֶּׁקֹּדֶם רַבָּן גַּמְלִיאֵל, לְמַר כִּדְאִית לֵיהּ וּלְמַר כִּדְאִית לֵיהּ אֲפִלּוּ בְּדִיעֲבַד הוּא:

אַלְמָנָה כוּ'. וְכָל הַבָּא לִפָּרַע מִנִּכְסֵי יְתוֹמִים. וַהֲוָה מָצֵי לְמִתְנֵי אֵין נִפְרָעִים מִנִּכְסֵי יְתוֹמִים אֶלָּא בִּשְׁבוּעָה, וְנִמְנְעוּ מִלְּהַשְׁבִּיעַ הָאַלְמָנָה, אֶלָּא דְאַלְמָנָה הָוֵי חִדּוּשׁ טְפֵי, סַלְקָא דַּעְתָּךְ אֲמִינָא מִשּׁוּם חִנָּא אֲקִילוּ רַבָּנָן עֲלַהּ. וּלְכָךְ נָקַט רֵישָׁא נַמִּי בְּאַלְמָנָה, כִּדְאִיתָא בַגְּמָרָא וְתוֹסָפוֹת. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: נִמְנְעוּ. בֵּית דִּין. הָרַמְבַּ"ם: וּלְפִיכָךְ בְּאַלְמָנָה הוּא דְנִמְנְעוּ אֲבָל לֹא בִּגְרוּשָׁה. גְּמָרָא: וְזֶה לְשׁוֹן הָרַ"ן, כָּל מַה שֶּׁיִּרְצוּ יִבְחֲרוּ הַיְתוֹמִים דָּבָר קָשֶׁה לְהַדִּירָהּ בּוֹ כְּגוֹן קוֹנָם וְכוּ'. וְלֹא חֲשׁוּדָה לַעֲבֹר עַל נִדְרָהּ, וַאֲפִלּוּ תַּעֲבֹר אֵין עָנְשָׁן שֶׁל נְדָרִים חָמוּר כָּל כָּךְ כְּעָנְשָׁן שֶׁל שְׁבוּעוֹת שֶׁנִּפְרָעִים מִמֶּנּוּ וּמִכָּל הָעוֹלָם. וּבַגְּמָרָא, וְנֵחוּשׁ דִּלְמָא אָזְלָא לְחָכָם וְשָׁרֵי לָהּ. וּמְשָׁנֵי, קָסָבַר צְרִיכָה לְפָרֵט אֶת הַנֶּדֶר הֵיאַךְ נָדְרָה נֶדֶר הַזֶּה וְכִי מְפָרְטָהּ קַמֵּיהּ דְּלִיתוֹמִים נָדְרָה וְכָךְ נָדַרְתִּי אִם נֶהֱנֵיתִי מִכְּתֻבָּתִי כְּלוּם לֹא שָׁרֵי לָהּ: וְאִם תֹּאמַר יַדִּירוּהָ בִדְבָרִים שֶׁאֵין בָּהֶם עִנּוּי נֶפֶשׁ שֶׁאֵין הַבַּעַל יָכוֹל לְהָפֵר. וְיֵשׁ לוֹמַר כֵּיוָן שֶׁאֵין בָּהֶם עִנּוּי נֶפֶשׁ לֹא תָחוּשׁ אִם תְּקַיְּמֵם כוּ'. וְאִם תֹּאמַר יַדִּירוּהָ שֶׁלֹּא תֹאכַל כִּכָּר זֶה כוּ' וְתֹאכְלֶנוּ לְאַלְתַּר בְּפָנֵינוּ, וְיֵשׁ לוֹמַר דְּלֹא תָחוּשׁ לְעֹנֶשׁ זֶה אֶלָּא בְּנֶדֶר, שֶׁעַל יָדוֹ עוֹמֶדֶת בְּאִסּוּר כָּל יָמֶיהָ. תּוֹסָפוֹת וְהָרֹא"שׁ. וּבַגְּמָרָא וְכִי לֹא נִיסֵת נַמִּי לְכִי מִנַסְּבָא מֵפֵר לָהּ בַּעַל. וּמְשָׁנֵי אֵין הַבַּעַל מֵפֵר בַּקּוֹדְמִין. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: פֵּרֵשׁ רַשִׁ"י שֶׁשְּׁבוּעַת הַדַּיָּנִים הַנִּזְכֶּרֶת בַּמִּשְׁנָה, שֶׁל תּוֹרָה הִיא, וְנָקֵיט סֵפֶר תּוֹרָה בְיָדֵיהּ אוֹ תְּפִלִּין, וּמַשְׁבִּיעַ לָהּ בַּשֵּׁם אוֹ בְּכִנּוּי. כִּדְאַשְׁכְּחָן בְּאַבְרָהָם שִׂים נָא יָדְךָ כוּ' וְאַשְׁבִּיעֲךָ . וְעָנְשָׁהּ מְרֻבֶּה. וְחוּץ לְבֵית דִּין, שְׁבוּעָה דְרַבָּנָן בְּקִלְלַת אָרוּר, וְלֹא נָקֵיט מִידֵי. וְאַף עַל פִּי שֶׁשְּׁבוּעַת הַמִּשְׁנָה טַכְסִיסָהּ כְּשֶׁל תּוֹרָה, תִּקְּנוּ כָּאן לְהָקֵל בְּכָךְ כְּדֶרֶךְ שֶׁתִּקְּנוּ שֶׁתְּהֵא גּוֹבָה בְנֵדֶר. הָרַ"ן: תִּקּוּן כוּ'. לָאו לְמֵימְרָא שֶׁהָיוּ צְרִיכִין כָּךְ לְכַתְּחִלָּה, דְּהָא לְעֵיל מַכְשִׁירִין חֲכָמִים דְּהַיְנוּ רַבִּי אֱלִיעֶזֶר אֲפִלּוּ בְּלֹא עֵדֵי חֲתִימָה. אֶלָּא תִּקֵּן וְלִמֵּד לַעֲשׂוֹת כֵּן כְּדֵי שֶׁלֹּא יְהֵא צָרִיךְ לְעֵדֵי מְסִירָה. אֲבָל אִם יִרְצֶה לְכַתְּחִלָּה לִסְמֹךְ עַל עֵדֵי מְסִירָה, יֵשׁ לִסְמֹךְ. הָרֹא"שׁ:

דְּקָנֵי לֵיהּ מִשַּׁבָּאֵי: וְתֵימַהּ דְּתֵיפּוֹק לֵיהּ לְפִי שֶׁנִּתְיָאֵשׁ מִמֶּנּוּ. וְהָכִי אִיתָא בַגְּמָרָא, וְכֵן כָּתַב הָרַמְבַּ"ם. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: יִשְׁתַּעְבֵּד. וְרַבּוֹ נוֹתֵן דָּמָיו. תּוֹסֶפְתָּא. וּפֵרֵשׁ הַכֶּסֶף מִשְׁנֶה דְּהַיְנוּ אִם פְּדָאוֹ לְשֵׁם עֶבֶד. אֲבָל כְּשֶׁפְּדָאוֹ לְשֵׁם בֶּן חוֹרִין, הֲרֵי הִפְקִיר דָּמָיו: דְּאִי יִשְׁתַּעְבֵּד לַשֵּׁנִי אַכַּתִּי יַפִּיל עַצְמוֹ לִגְיָסוֹת כְּדֵי שֶׁיַּרְוִיחַ לְעִנְיַן מְצִיאָה וּמַתָּנָה, דְּלָא קָנֵי לֵיהּ שֵׁנִי אֶלָּא לְמַעֲשֵׂה יָדָיו כְּמוֹ שֶׁקְּנָאוֹ הַשַּׁבָּאִי שֶׁמִּמֶּנּוּ קְנָאוֹ הוּא, וּפְעָמִים נַמִּי שֶׁהוּא שׂוֹנֵא לְרַבּוֹ מִתְּמוֹל שִׁלְשׁוֹם וּמַפְקִיעַ עַצְמוֹ מִמֶּנּוּ. תּוֹסָפוֹת. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: וְהַיְנוּ אַפּוֹתִיקִי מְפֹרָשׁ, לַאֲפוֹקֵי אִם כָּתַב לוֹ גְּבֵה חוֹבְךָ מִזֶּה, דְּהַיְנוּ אַפּוֹתִיקִי סְתָם, לָמָּה לֵיהּ לַמְּשַׁחְרֵר דְּלִכְתֹּב שְׁטָר, הָא כָּתוּב וְקָאֵי, דְהָא מְשַׁעְבְּדֵי לֵיהּ כֻּלְּהוּ נִכְסֵי. הָרַ"ן:

אִי אֶפְשָׁר. מִשּׁוּם לָאו דְּלֹא יִהְיֶה קָדֵשׁ. דִּמְתַרְגְמִינַן לֹא יִסַּב גַּבְרָא מִבְּנֵי יִשְׂרָאֵל אִתְּתָא אָמָה. וְלִשָּׂא חֶצְיָהּ שִׁפְחָה וְחֶצְיָהּ בַּת חוֹרִין אֵינוֹ יָכוֹל כוּ'. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: לֹא תֹהוּ כוּ'. הָא דְלָא נָקַט קְרָא דְּאוֹרַיְתָא דִּפְרוּ וּרְבוּ. מִשּׁוּם דְּאִי הָיָה יָכוֹל לְקַיֵּם לָשֶׁבֶת כָּל דְּהוּ מִשּׁוּם פְּרוּ וּרְבוּ לֹא הֲוָה כָּפִינַן לֵיהּ, דְּלֹא מִחַיַּב בָּהּ מִשּׁוּם דְּאָנוּס הוּא. וְעוֹד יֵשׁ לוֹמַר דְּנָקַט לָשֶׁבֶת מִשּׁוּם דְּשַׁיָּךְ אַף בְּצַד עַבְדוּת, אֲבָל פְּרוּ וּרְבוּ לֹא שַׁיָּךְ אֶלָּא בְצַד חֵרוּת (דְּאִתְּתָא לֹא מִפַּקְדָּה אַפְּרוּ וּרְבוּ, וְכָל מִצְוָה שֶׁאֵין הָאִשָּׁה חַיֶּבֶת בָּהּ אֵין הָעֶבֶד חַיָּב בָּהּ). תּוֹסָפוֹת: בְּרָאָהּ. יוֹצֵר הָאָרֶץ לֹא עַל תֹּהוּ לִהְיוֹת רֵיקָנִית בָּרָא אוֹתָהּ, כִּי אִם לָשֶׁבֶת לִהְיוֹת בָּהּ יִשּׁוּב יְצָרָהּ. רַשִׁ"י:

רְצוֹנוֹ לוֹמַר בַּגְּמָרָא: וְאַלּוֹקֵחַ קָאֵי, שֶׁהוּא הַמּוֹצִיא וְאִסּוּרָא גַּבֵּיהּ אִיתָא, וְאוֹתוֹ קָנְסוּ שֶׁיְּאַבֵּד דָּמָיו. גְּמָרָא: סְפָרִים. שֶׁלֹּא לְהַרְגִּיל אוֹתָן לְגוֹנְבָן וּלְגוֹזְלָן. הָרַמְבַּ"ם:

וְנִרְאֶה לִי שֶׁלֹּא בָא לוֹמַר שֶׁצָּרִיךְ שֶׁיֹּאמַר כֵּן בִּשְׁעַת גֵּרוּשִׁין, דְּאִם כֵּן בְּמוֹצִיא מִשּׁוּם שֵׁם רָע נַמִּי הֲוָה לֵיהּ לְפָרֵשׁ כֵּן. אֶלָּא שֶׁפֵּרוּשׁוֹ מַאי אִכְפַּת לֵיהּ בְּאִם נָדְרָה, וְקָאָמַר שֶׁהוּא אוֹמֵר אִי אֶפְשִׁי כוּ'. תּוֹסְפוֹת יוֹם טוֹב. וְעַיֵּן עוֹד: לָאו דַּוְקָא, דְּהָא לֹא הִתְנָה בְכָךְ, אֶלָּא לַעַז בְּעָלְמָא. וְכֵן כָּתַב הָרִ"י: אֲבָל לְקִלְקוּלָא לֹא חָיֵישׁ הָכָא, דִּבְצָרִיךְ חֲקִירַת חָכָם סְבִירָא לֵיהּ כְּרַבִּי אֱלִיעֶזֶר, וּבְאֵינוֹ צָרִיךְ סְבִירָא לֵיהּ כְּרַבִּי מֵאִיר. גְּמָרָא: נִרְאֶה לִי דְּרַבִּי מֵאִיר לְטַעְמֵיהּ בְּמִשְׁנָה ז' פֶּרֶק י"א דִּנְדָרִים, בְּאוֹמֵר לֹא הָיִיתִי יוֹדֵעַ שֶׁזֶּה נֶדֶר, שֶׁאַף עַל פִּי כֵן הֲוָה לֵיהּ לְהָפֵר, וְרַבִּי אֱלִיעֶזֶר סְבִירָא לֵיהּ כְּרַבָּנָן דְּהָתָם. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: בַּגְּמָרָא, פְּשִׁיטָא מַאי נֵחוּשׁ לָהּ, מַהוּ דְּתֵימָא לִגְזֹר מִשּׁוּם דְּרַבִּי נָתָן, דְּתַנְיָא רַבִּי נָתָן אוֹמֵר הַנּוֹדֵר כְּאִלּוּ בָּנָה בָּמָה וְהַמְקַיְּמוֹ כְּמוֹ שֶׁהִקְרִיב עָלֶיהָ קָרְבָּן, וְהַאי כֵּיוָן דְּקִיֵּם נִדְרוֹ נִקְנְסֵיהּ מִלְּהַחְזִיר, קָא מַשְׁמַע לָן:

אַיְלוֹנִית. פֵּרוּשׁ שֶׁחָשַׁב שֶׁהִיא אַיְלוֹנִית גְּמוּרָה, דְּלֹא יָלְדָה, וְגֵרְשָׁהּ בְּגֵט. וְאַף עַל פִּי שֶׁהָאַיְלוֹנִית אֵינָהּ צְרִיכָה גֵט כְּדִתְנַן רֵישׁ פֶּרֶק קַמָּא דִיבָמוֹת כוּ', הָכָא אַיְרֵי כִדְאָמְרָן דְּלֹא נוֹלְדוּ בָּהּ סִימָנֵי אַיְלוֹנִית גְּמוּרִים אֶלָּא שֶׁהוּא חָשַׁב כֵּן, וְהַיְנוּ דְקָתָנֵי כוּ' וְהָיוּ לָהּ בָּנִים. וְאִם נוֹלְדוּ בָּהּ סִימָנֵי אַיְלוֹנִית אִי אֶפְשָׁר בְּשׁוּם צַד שֶׁיִּהְיוּ לָהּ בָּנִים. הָרַ"ן: לְפִיכָךְ אוֹמְרִים לוֹ קֹדֶם גֵּרוּשִׁין הָוֵי יוֹדֵעַ שֶׁהַמּוֹצִיא אֶת אִשְׁתּוֹ מִשּׁוּם אַיְלוֹנִית לֹא יַחֲזִירֶנָהּ עוֹלָמִית. רַשִׁ"י: כְּלוֹמַר וְהַשְׁתָּא רַבָּנָן נַמִּי חָיְשֵׁי לְקִלְקוּלָא. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: וְקָשְׁיָא, דְּאִם כֵּן הֵטִיל הַתְּנַאי בַּגֵּט עַצְמוֹ וּכְשֶׁלֹּא נִתְקַיֵּם נִמְצָא שֶׁהַגֵּט בָּטֵל. וּבְמַתְנִיתִין דִּלְעֵיל דְּכַפְלֵיהּ כִּי לֹא נִתְקַיֵּם הַתְּנַאי וְהַגֵּט בָּטֵל מַה הוֹעִילוּ חֲכָמִים בְּתַקָּנָתָם דְּלֹא יַחֲזִיר. וְהַתּוֹסָפוֹת וְהָרַמְבַּ"ם כָּתְבוּ בִּדְכַפְלֵיהּ לְמִלְּתֵיהּ וְלֹא כַפְלֵיהּ לִתְנָאֵיהּ. דְּאִי כַפְלֵיהּ לִתְנָאֵיהּ אֵין מוֹעִיל מַה שֶּׁלֹּא יַחֲזִיר, דִּלְעוֹלָם אִיכָּא קִלְקוּלָא. וְכִי לֹא כַּפְלֵיהּ לְמִלְּתֵיהּ אֲפִלּוּ לַעַז לֵיכָּא, דַּחֲשָׁשׁ קִלְקוּל זֶה אֵינוֹ אֶלָּא לַעַז בְּעָלְמָא. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: כִּי הוּא הָיָה סָבוּר שֶׁהָיְתָה אַיְלוֹנִית, אֶלָּא שֶׁאָמַרְנוּ לוֹ שֶׁיְּהֵא גּוֹמֵר וּמְגָרֵשׁ בְּכָל צַד, אֲבָל לֹא גֵּרֵשׁ עַל דַּעַת שֶׁתִּתְבַּע כְּתֻבָּה. הָרַ"ן:

הַלּוֹקֵחַ מֵבִיא. הָכִי גָרְסִינַן בִּסְפָרִים מְדֻיָּקִים, וְזוֹ הִיא גִרְסַת הָרַמְבַּ"ם. וּמְפֹרָשׁ כְּבָר בֵּאַרְנוּ שֶׁהָעִקָּר אֶצְלֵנוּ אֵין קִנְיָן לְנָכְרִי בָּאָרֶץ לְהַפְקִיעַ מִן הַמַּעַשְׂרוֹת, לְפִיכָךְ לָקַח מִן הַנָּכְרִי יָבִיא מִן הַתּוֹרָה. וּכְשֶׁרָאוּ חֲכָמִים שֶׁהָיוּ מִתְיַקְּרִין בִּלְקִיחָתָן מִן הַנָּכְרִי הִתְקִינוּ שֶׁהַלּוֹקֵחַ מִן הַנָּכְרִים לֹא יָבִיא בִכּוּרִים, כְּדֵי שֶׁיִּתְבָּאֵר לוֹ שֶׁיָּרְדָה מִקְּדֻשָּׁתָהּ וְלֹא יִתְאַוּוּ לִקְנוֹת מֵהֶם. וּכְשֶׁרָאוּ גַּם כֵּן שֶׁזֶּה גוֹרֵם שֶׁתִּשָּׁאֵר בְּיַד הַנָּכְרִי וְתִשְׁתַּקֵּעַ בְּיָדָם, הִתְקִינוּ שֶׁהַלּוֹקֵחַ מֵהֶם יָבִיא כְּדֵי שֶׁיֵּדַע שֶׁלֹּא הָיָה קִנְיָן לַנָּכְרִי בְּאֶרֶץ יִשְׂרָאֵל קִנְיָן גָּמוּר, וּלְכָךְ יִשְׁתַּדֵּל בְּפִדְיוֹן זֶה הַקַּרְקַע מִיָּדוֹ, לְפִי שֶׁהִיא עוֹמֶדֶת בִּקְדֻשָּׁתָהּ וּבְמַתָּנוֹת יְדוּעוֹת. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: הָכִי גָרְסִינַן בַּמִּשְׁנָה. וְלֹא גָרְסִינַן וְחָזַר וּלְקָחָהּ כוּ'. וְכֵן הוּא גִרְסַת רַשִׁ"י. וְלֹא מְפָרֵשׁ לְאוֹקִימְתָּא דִּתְרֵי תַקָּנוֹת הָיוּ כַּנִּזְכָּר לְעֵיל, דְּכָל זֶה לְמַאן דְּאָמַר לֹא הָיָה קִנְיָן לְעוֹבֵד כּוֹכָבִים כוּ'. אֲבָל הָרַ"ב פָּסַק כְּמַאן דְּאָמַר דְּיֵשׁ קִנְיָן, וּמְפָרֵשׁ לְמִשְׁנָתֵנוּ אַלִּיבָּא דְהִלְכָתָא וְאֵין צָרִיךְ לָאוֹקִימְתָּא הַנִּזְכֶּרֶת לְעֵיל:
