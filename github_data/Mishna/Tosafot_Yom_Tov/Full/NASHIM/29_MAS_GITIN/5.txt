פרק ה

בֵּעִדִּית. אּת כֹּל נֵאוֲת יָעַקֲב תִּרֵגּוּם יָת כֹּל עִדִּית בְּית יָעַקֲב. נֵאוֲת אֱלֲהִים תָּרֵגּוּם עִדִּית. עֹרוּךֵ: כֵּלוֲמָר דֵּהֹא לֵהֹכִי תֵנָן לֹהּ הֹכֹא בֵגִטִּין וֵאָף עָל פִּי שִּאְין עִנֵיֹנוֲ מְעִנֵיֹן גִּטִּין, אּלֹּא מִשִּוּם דֵּתֹנְי בֵּפּרּק דִּלֵעְיל מִלְּי דֵתִקּוּן הֹעוֲלֹם: שִּנּוֲחָ לוֲ לֵאֹדֹם לִגֵבּוֲת מֶעֹט מִקָּרֵקָע עִדִּית וֵלֲא הָרֵבְּה מִזִּבּוּרִית. רָשִִ"י: גֵּמֹרֹא. וּפְרְשִ הֹרָ"ב אָף עָל פִּי שִּהָזִּבּוּרִית שִֹוֹה כֵנּגּד הָחוֲב לֲא קֹפִיץ עַלְיהּ זֵבִינֹא: בֵּזִבּוּרִית. הָפֹּחוּת וֵהֹרָע. הֹרָמֵבָּ"ם. וֵלֲא נוֲדָע לִי מְאְיזּה לֹשִוֲן הוּא (נִרֵאּה לִי מִלֹּשִוֲן זִבּוּרֹא וֵעָקֵרָבֹּא. בְּית דֹּוִד): וֵקֹשִּה דִלֵמָאן דֵּאֹמָר כֵּתֶבֹּה דֵאוֲרָיֵתֹא דֵּיֹלְיף לֹהּ מִכֵּמֲהָר הָבֵּתוּלוֲת כוּ', מִדֵּאוֲרָיֵתֹא דִינֹהּ בֵּעִדִּית, וֵאָמָּאי הִפֵקִיעוּ חַכֹמִים דִּינֹהּ מְהָאי טָעַמֹא. וֵרָבִּי מְאִיר נָמִּי דֵּאֹמָר בֵּבְינוֲנִית קֹסֹבָר כֵּתֶבֹּה דֵאוֲרָיֵתֹא בֵּרְישִ פּּרּק אָף עָל פִּי. וֵיְשִ לוֲמָר דֵּטָעֵמֹא דֵהִפֵקִיעוּ חַכֹמִים דִּינֹהּ כִּדֵאֹמֵרִינָן בָּגֵּמֹרֹא גָּבְּי בָעָל חוֲב, שִּלֲּא יִרֵאּה אֹדֹם שֵֹדּה נֹאּה וָיֲּאמָר אּקֵפֲּץ וֵאָלֵוּּנּוּ, הֹכִי נָמִּי חֹשִוּ חַכֹמִים בֵּאִשִֹּה שִּמֹּא תָקֵנִיטּנּוּ כֵּדְי שִּיֵּגֹרֵשִּנֹּהּ וֵתִגֵבּּה כֵּתֶבֹּה. וֵטָעֵמֹא דֵיוֲתְר מִשִּּהֹאִישִ כוּ' לֲא הֹוְי אּלֹּא מִשִּוּם דֵּלֲא תְימֹא כֵמוֲ שִּבָּעָל חוֲב בֵּבְינוֲנִית מִשִּוּם נֵעִילָת דּּלּת, אִשִֹּה נָמִּי מִשִּוּם חִנֹּא שִּיֵּהוּ הֹאַנֹשִִים נוֲשֵֵאִים חְן בֵּעְינְי הָנֹּשִִים וֵיִנֹּשֵֵאוּ לֹהּן תֵּהְא בֵבְינוֲנִית. תּוֲסֹפוֲת:

וֵחֹשִִיב תִּקּוּן הֹעוֲלֹם, דֵּאִי לֹאו תִּקּוּן הֹעוֲלֹם הֹיֹה גוֲבּה אַפִלּוּ מִמֵּשִָעֵבֵּדְי בֵּבְינוֲנִית מִשִּוּם נֵעִילָת דּּלּת. עָד כֹּאן. וּמִשִּוּם הֹכִי נֹקָט הֹרָ"ב בָּעָל חוֲב. וֵהוּא הָדִּין לֵנִזֹּקִין: יֵתוֲמִים. מֵפֹרְשִ טָעֵמֹא בָגֵּמֹרֹא מִשִּוּם דֵּלֲא מָסִּיק אָדָּעֵתְּיהּ דֵּמָלֵוּה דֵמִית לֲוּה וֵנֹפֵלְי נִכֵסְי קֹמְי יָתֵמְי דָּהַוֹה לְיהּ נוֲעְל דּּלּת וֵהִלֵכֹּךֵ אַפִלּוּ יֵתוֲמִים גֵּדוֲלִים. וּבֵנִזֹּקִין אָף עָל גָּב דֵּמִדֵּאוֲרָיֵתֹא בֵּעִדִּית, הָיֵנוּ דֵנִזָּק, וֵגָבְּי יָתֵמְי אוֲקֵמוּהֹ אָדֵּאוֲרָיֵתֹא. וֵעָיְּן תּוֲסֹפוֲת:

וּלֵשִוֲן לָאַכִילָת פְּרוֲת לֲא דֹיֵקֹא, דִלֵפְרוֲת בִּלֵחוּד הַוֹה לְיהּ לֵמִתֵנְי. וֵהֹרָמֵבָּ"ם מֵפָרְשִ כֵּשִּגֹּזָל רֵאוּבְן קָרֵקָע מִשִִּמֵעוֲן וֵהֹיֵתֹה מֵלְאֹה פְרוֲת וֵאֹכָל רֵאוּבְן הָפְּרוֲת וֵהִפֵסִיד הָקָּרֵקָע כוּ', כֵּשִּיֹּבֲא שִִמֵעוֲן לִגֵבּוֲת מְרֵאוּבְן הָפְּרוֲת שִּאֹכָל כוּ'. וּשִֵנְי הָפְּרוּשִִים בָּגֵּמֹרֹא בֹּבֹא מֵצִיעֹא דָף י"ד. וּפְרוּשִ הֹרָ"ב נִדֵחֹה: אַבֹל דֵּמְי פְרוֲת הַוֹה כֵמִלֵוּה עָל פּּה. דֵּאָף עָל פִּי שִּכֹּתָב מוֲכְר לָלּוֲקְחָ אַנֹא אְיקוּם וָאָשִָפְּי אִינוּן וֵעָמֵלְיהוֲן וֵשִִבֵחְיהוֲן, אַפִלּוּ הֹכִי אְין קוֲל יוֲצְא אּלֹּא עָל הָמִּקֹּח, לֲא עָל הָשִֵּאֹר, כְּיוֹן שִּעַדָיִן אְינוֲ בֹּעוֲלֹם. רָשִִ"י וֵהֹרָ"ן:

אַבִיהּן. וֵכֹל שִּכְּן הָמֵּמֶנּּה מְהָבְּית דִּין שִּהוּא שִֵלִיחָ בְּית דִּין: לֵעָשְֵּר. קוֲרֵאִין כֹּל הָמָּתֹּנוֲת בֵּשְִם מָעַשְֵר בִּלֵבָד. וּבָגֵּמֹרֹא רֹמִינָן דִּתֵנָן (בֵּרְישִ תֵּרוּמוֲת) וֵהָתּוֲרְם אּת שִּאְינוֲ שִּלּוֲ אְין תֵּרוּמֹתוֲ תֵּרוּמֹה, וּמֵשִֹנִינָן לֲא קֹשִֵיֹא כֹּאן לֵהָאַכִיל כֹּאן לֵהָנִּיחָ. פְּרְשִ רָשִִ"י לֵהָאַכִיל יֵתוֲמִים לֵאָלֵתָּר חָיֹּב לֵעָשְֵּר, לֵהָנִּיחָ הָתֵּבוּאֹה בֹאוֲצֹר עָד שִּיִּגֵדֵּלוּ, לֹמֹּה לִי עִשֵּוּרְי, לִכֵשִּיִּגֵדֵּלוּ יֵעָשֵֵּרוּ הְם: שִּמִּנֹּהוּ. וּמִי שִּסֹּמֵכוּ אּצֵלוֲ שִָבֵקְיהּ מִלֹּדוּן בּוֲ. וֵיְשִ בֹּזּה מָחַלֲקּת הָפּוֲסֵקִים לֵמִי דֹמִי יוֲתְר: וֵאָף עָל גָּב דֵּהָיֵנוּ מֵדָמְּעָ, שִּהוּא נָמִּי מֵעֹרְב, אִצֵטֵרִיךֵ לֵמִתֵנְי בֵּהּדֵיֹא, מִשִּוּם דִּקֵנֹסֹא מִקֵּנֹסֹא לֲא יֹלֵפִינָן, דֵּהֹא לֹאו דִּינֹא הוּא, וֵהְיכֹא דֵקֹנוּס קֹנוּס וֵהְיכֹא דֵלֹא קֹנוּס לֲא קֹנוּס. גֵּמֹרֹא: וֵצֹרִיךֵ לוֲמָר דֵּמָזִּיק שִִעֵבּוּדוֲ שִּל חַבְרוֲ דִתֵנָן בָּפּּרּק דִּלֵעְיל מִשִֵנֹה ד', דִּלֵרָבֹּן שִִמֵעוֲן בּּן גָּמֵלִיאְל דֵּקֹיֵמֹא לֹן כֵּוֹתְיהּ מֵחֹיֵבִינָן לְיהּ, דֵּחֹשִִיב הּזְּק נִכֹּר לֵגָבְּי הֹנֹךֵ דֵּהֹכֹא. תּוֲסֹפוֲת. וֵעָיְּן תּוֲסֵפוֲת יוֲם טוֲב: וֵהוּא הָדִּין לָעַבוֲדָת הוֲלֹכֹה וֵקָבֹּלֹה:

הָחְרּשִּת. מִשִּוּם תָּרֵתְּי בָּתֵרֹיֹתֹא דֵּמִשִּוּם תָּקֹּנֹה נִינֵהוּ, תֹּנֹא לְיהּ הֹכֹא. רָשִִ"י: קֵטָנֹּה כוּ'. אַבֹל חְרּשִּת דֵּאָיֵרְי בָּהּ וֵשִֹבֵקֹה, שִֵמָע מִנָּהּ דֵּלֲא אֹכֵלֹה. וֵטָעֵמֹא בָגֵּמֹרֹא, גֵזְרֹה שִּמֹּא חְרְשִ שִּנֹּשֵֹא פִּקָּחָת יָאַכִילּנֹּהּ, וּגֵזְרֹה מִשִּוּם תֵּרוּמֹה דֵאוֲרָיֵתֹא: וֵאִם תֲּאמָר וֵלוֲקֵמֹא אַפִלּוּ בִּדֵאוֲרָיֵתֹא, דֵקֹטֹן אוֲכְל נֵבְלוֲת הוּא דֵאְין מֵצֶוִּין לֵהָפֵרִישִוֲ. וֵיְשִ לוֲמָר דֵּבָגֵּמֹרֹא מָשִֵמָע דָּאַפִלּוּ מָאַכִילֹהּ בֵּיֹדָיִם. וֵעוֲד מָשִֵמָע דָּאַפִלּוּ לֵכִי גֹּדֵלֹה אוֲכּלּת מִכֲּחָ זּה אָף עָל פִּי שִּעַדָיִן לֲא בֹּעָל. תּוֲסֹפוֲת. וֵעָיְּן תּוֲסֵפוֲת יוֲם טוֲב: יוֲרֵשִֹהּ. דֵּהּפֵקְר בְּית דִּין הּפֵקְר כִּדֵיֹלֵפִינָן לְיהּ בָּגֵּמֹרֹא מִקֵּרֹאִי: שִּלֲּא נוֲדֵעֹה. דֵּאִי נוֲדֵעֹה לֹרָבִּים, מִלֵּתֹא דֵלֹא שִֵכִיחֹא הוּא שִּיֵבִיאּנֹּה הָגָּזֵלֹן, וֵלֲא תִקֵּנוּ בֵּמִלֵּתֹא דֵלֹא שִֵכִיחָ: לֹרָבִּים. אֹמֵרִינָן בָּיֵרוּשִָלֵמִי דֵרָבִּים הָיֵנוּ שִֵלֲשִֹה. וֵעָיְּן תּוֲסֵפוֲת יוֲם טוֲב: מֵכָפּּרּת. וֵאִי קֹשִֵיֹא הֹאְיךֵ הִתֵנוּ בְּית דִּין לָעַקֲר דֹּבֹר מְהָתּוֲרֹה וֵלִפֵטֲר אּת זּה מְחָטֹּאת שִּהוּא חָיֹּב, הֹא אֹמֵרִינָן בִּיבֹמוֲת דָּף צ' שְִב וֵאָל תָּעַשֵּה שִֹאנְי דֵלֹאו מִעָקָּר הוּא. רָשִִ"י. וֵקֹשִּה דָעַדָיִן אְינוֲ מֵיֶשִֹּב הֹאְיךֵ הָכֲּהַנִים אוֲכֵלִין חֶלִּין בֹּעַזֹרֹה. וּלֵהָמָּסֵקֹנֹא מֵתֹרְץ דֵּרָבֹּנֹן אוֲקֵמְיהּ בֵּיָד הָגָּזֵלֹן שִּיֵּהְא שִּלּוֲ וֵשִּיּוּכָל לֵהָקֵדִּישִוֲ. וּפְרְשִ רָשִִ"י דֵהּפֵקְר בְּית דִּין הּפֵקְר, אִם כְּן הֹוְי שִּלּוֲ נָמִּי לֵעִנֵיָן כָּפֹּרֹה, וּמֵיֶשִָּב הָכֲּל. תּוֲסֹפוֲת חַדֹשִִים:

בִּיהוּדֹה. מִלֵחּמּת טִיטוּס שִּהֹיֹה בִּירוּשִֹלָיִם וּבִיהוּדֹה. רָשִִ"י. וּבָיֵרוּשִָלֵמִי, בֹּרִאשִוֲנֹה גֹזֵרוּ גֵּזְרֹה עָל יֵהוּדֹה לֵפִי שִּמָּסֲּרּת בֵּיֹדֹם שִּיֵהוּדֹה הֹרָג עְשֵֹו דִּכֵתִיב יֹדֵךֹ בֵּעֲרּף אוֲיֵבּיךֹ . תּוֲסֹפוֲת: לֹשִוֲן מֵסֲרֹס הוּא, דֵּיִשֵֵרֹאְל אָגָּב אוֲנֵסְיהּ הַוֹה כוּ': אוֲ שִּכֹּתָב בָּכֵּתֶבֹּה אוֲ שִּהִכֵנִיסֹה לוֲ שִוּם בִּכֵתֶבֹּתֹהּ. וֵכֹל שִּכְּן שִֵאֹר נֵכֹסִים. דֵּאִי לֲא עֹבֵדֹה לְיהּ אֹמָר לֹהּ עְינָיִךֵ נֹתָתֵּ בָּגְּרוּשִִין וּבָמִּיתֹה. רָשִִ"י. וֵעָיְּן תּוֲסֵפוֲת יוֲם טוֲב: רֵבִיעָ. כֵּגוֲן שִּלֹּקָח מְהָסִּיקֹרִיקוֲן שֵֹדּה בִשִֵלֲשִִים דִּינֹרִין, הוּא נוֲתְן לֵבָעָל הָבָּיִת עַשֵֹרֹה דִינֹרִין. לֵפִי שִּאֹנוּ אוֲמֵדִין דֵּמְי הָקָּרֵקָע אָרֵבֹּעִים דִּינֹרִין וֵיוֲתְר, וֵהֹרֲב שִּהָסִּיקֹרִיקוֲן עוֲשֵּה מִמּּנּוּ זוֲל אָף עָל פִּי שִּהֹיֵתֹה אּצֵלוֲ בֵלֲא דֹּמִים הוּא זּה הָשִִּעוּר, וֵהוּא עִנֵיֹן מָה שִּּאֹמֵרוּ רֵבִיעָ בָּקָּרֵקָע שִּהִיא שִֵלִישִ בָּמֹּעוֲת. הֹרָמֵבָּ"ם: אְימֹתָי כוּ'. מִלֵּשִוֲן אְימֹתָי מָשִֵמָע דֵּקֹאְי אָדֵּסֹמִיךֵ לְיהּ, נוֲתְן לָבֵּעֹלִים רֵבִיעָ אְימֹתָי כוּ', וּכֵלוֲמָר שִּזּּה שִּהָקָּרֵקָע נִשִֵאָר אְצּל הָלּוֲקְחָ אּלֹּא שִּנּוֲתְן רֵבִיעָ לָבֵּעֹלִים, הוּא בִּזֵמָן שִּאְין בֵּיָד הָבֵּעֹלִים לִקָּח, דֵּאִלּוּ יְשִ בֵּיֹדֹן לִקָּח צֹרִיךֵ שִּיָּחֵזִיר לֹהּם כֹּל הָקָּרֵקָע בֵּדֹמִים שִּנֹּתָן בֹּהּ. וֵמִדֵּלֲא מֵסָיְּם אַבֹל יְשִ בֵּיֹדֹן לִקָּח יָחַזִיר הָקָּרֵקָע לָבֵּעֹלִים, וּמֵסָיְּם הְן קוֲדֵמִין, מָשִֵמָע דֵּמָיֵרְי כֵשִּבֹּאִים לִקָּח הָלּוֲקְחָ וֵהָבָּעָל, שִּהָמִּצֵוֹה עָל הָלּוֲקְחָ שִּלֲּא לִקָּח וֵלֹתְת הֹרֵבִיעָ אּלֹּא יָנִּיחָ לִבֵעֹלֹיו שִּיִּקָּח (אַבֹל אִי קֹדָם הָלּוֲקְחָ וּקֵנֹאֹהּ זֹכֹה). וּמְהֹרָמֵבָּ"ם מָשִֵמָע כִּדֵדֹיֵקִינָן מְהָסְּיפֹא. וֵהָטּוּר כֹּתָב כִּדֵדֹיֵקִינָן מְהֹרְישִֹא כוּ'. וּבֵמְשִֹרִים (נֹתִיב ל"א חְלּק ה') דִּיְּק מְהָיֵרוּשִָלֵמִי, שִּאִם יֹדוֲ מָשֵּּגּת בֵּשִֹעֹה שִּלֵּקֹחֹהּ לוֲקְחָ, בֵּעֹלִים מֵסָלֵּקִין לָלּוֲקְחָ כֹּל זֵמָן שִּיִּרֵצוּ. וֵהָשִֵתֹּא אֹתְי שִָפִּיר לִישִֵנֹא דֵקוֲדֵמִין. דֵּמִלֵּתֹא בֵּטָעַמֹא קֹאֹמָר, שִּהַרְי הֹיוּ קוֲדֵמִים אּלֹּא שִּלֵּאְיזּה סִבֹּה נִמֵנֵעוּ בֵּאוֲתֹהּ שִֹעֹה לִקָּח, לֲא מִפֵּנְי זּה אִבֵּדוּ זֵכוּתֹם. וֵאָף דִּבֵרְי הֹרָמֵבָּ"ם נוּכָל לֵפֹרְשִ כְּן. תּוֲסֵפוֲת יוֲם טוֲב. וֵעָיְּן עוֲד: בִּפֵנְי כוּ'. פְּרוּשִ בִּרֵשִוּתוֲ. הֹרָ"ן:

וֵאָף עָל פִּי שִּזּּה שִּנִּתֵרָצֹּה גָּם כְּן הוּא בִּרֵמִיזֹה וֵאִם כְּן הָיֵנוּ רוֲמְז. לֲא קֹשִֵיֹא דָּהַוֹה אַמִינֹא כִּי נִרֵמָז מְאַחְרִים מָה שִּּנִּתֵרָצֹּה בִּרֵמִיזֹה לֲא לִסֵגְי דֵּדִלֵמֹא טֹעֹה בָּמּּה שִּנִּרֵמָז לוֲ קֹא מָשִֵמָע לֹן. וֵהָאי נִתֵרָצֹּה לֲא קֹאְי אּלֹּא אָנִּרֵמָז דֵּאִלּוּ בֹּרוֲמְז מָאי בֹעְי טֵפְי וּמֹה רִצּוּי שִָיֹּךֵ: בֵּמִטָּלֵטֵלִין. אַבֹל בֵּגִטִּין דִּבֵרְי הָכֲּל בִּרֵמִיזֹה. כֵּדִתֵנָן בֵּרְישִ פּּרּק י"ד דִּיבֹמוֲת וֵאִיכֹּא דֵּאֹמֵרְי כֵמָחַלֲקּת בָּמִּטָּלֵטֵלִין כֹּךֵ מָחַלֲקּת בֵּגִטִּין וֵאְימֹא אָף בָּמִּטָּלֵטֵלִין. גֵּמֹרֹא. וּמִיהוּ הֹא דֵגִטִּין כֵּשִּנֹּשֵֹא כֵּשִּהוּא חְרְשִ, כֵּדִתֵנָן הֹתֹם כֵּשְִם שִּכֹּנָס בִּרֵמִיזֹה כוּ': הֹכִי גֹּרָס הֹרִי"ף בִּדֵרָב כֹּהַנֹא: וֵזּה הֹעִיּוּן תֹּלוּי בָּדָּיֹּן. הֹרָמֵבָּ"ם: מִקֹּחוֲ מִקֹּח. וֵטָעֵמֹא מִשִּוּם כֵּדְי חָיֹּיו. גֵּמֹרֹא. דֵּאִי לֹאו זֵבִינְיהּ זֵבִינֹא לֲא מִזָּבֵּנְי לְיהּ מֵזוֲנְי וֵלֲא מֵזָבֵנְי מִנְּיהּ. רָשִִ"י: לֵפִי שִּדֵּבָר תּוֲרֹה הוּא מָה שִּּכֲּהְן קוֲרְא רִאשִוֲן וּמִקֵּרֹאִי יֹלְיף לֵהוּ וֵאִיכֹּא דֵמְילָף מִוֵקִדָּשִֵתּוֲ:

דִּשִֵכִיחְי נָמִּי רָבִּים. תּוֲסֹפוֲת: וּפְרוּשִ נִתֵפֹּרֵדֹה הָחַבִילֹה, נִפֵסָק הָקּּשִּר, אִבְּד הָלְּוִי אּת כֵּבוֲדוֲ בִּשִֵבִיל חַבִילֹתֹם הָנִּפֵרּדּת וֵאְינוֲ קוֲרְא כֵלֹל. רָשִִ"י: וֵצֹרִיךֵ לוֲמָר שִּאִם יָחֵשִֵדוּ יֹבֲאוּ עָל יֵדְי כֹּךֵ לִידְי מָחַלֲקּת. רָשִִ"י. וֵרָבְּנוּ תָם מֵפֹרְשִ חַשִֹדֹא שִּלֲּא יֲאמֵרוּ מִפֵּנְי הָחַשִֹד שִּחוֲשִֵדִים אוֲתֹם לִגֵנֲב פָּת שִּל עְרוּב אְין מָנִּיחִין אוֲתוֲ שִֹם: וּכֵשִּהוּא מֵמָלֵּאוֲ סוֲכְר אּת הֹאָמֹּה עָד שִּיִּתֵמָלֵּאוּ. רָשִִ"י: וֵטָעֵמֹא דֵרָבִּי יוֲסְי בֵּכוּלֵהוּ, דֵּקֹסֹבָר עֹשֵוּ מִפֵּנְי דָּרֵכְי שִֹלוֲם אּת שִּאְינוֲ זוֲכּה לִקֵנוֲת קִנֵיֹן גֹּמוּר כֵּזוֲכּה. רָשִִ"י: חְרְשִ שִוֲטּה וֵקֹטֹן. דֵּמִדֵּאוֲרָיֵתֹא לֲא זֹכֹה, לֵפִי שִּאְין לוֲ יֹד לִזֵכּוֲת בֵּמָה שִּּאְינוֲ שִּלּוֲ וֵאְין דָּעָת אָחּרּת מָקֵנֹה לוֲ. בְּית יוֲסְף: וּמָיֵרְי שִּכֵּשִּחוֲתְךֵ הָפֵּרִי אְינוֲ אוֲחְז לָפֵּרִי בֵּיֹדוֲ. וֵהָיֵנוּ דֵּתֹנֹא בָגֵּמֹרֹא אִם לִקְּט וֵנֹתָן בָּיֹּד הַרְי זּה גֹּזְל גֹּמוּר. וֵזּה לֵשִוֲן הָתּוֲסֹפוֲת, נִרֵאּה דֵאִצֵטֵרִיךֵ לֵאָשִֵמֵעִינָן בֵּתוֲלֵשִֹן בֵּיֹדוֲ וּמָפִּילֹן אָף עָל פִּי שִּלֲּא נִתֵעָכֵּבוּ בֵּיֹדוֲ וֵלֲא אֹחָז בֹּהּן אּלֹּא לֵהָפִּילֹן:
