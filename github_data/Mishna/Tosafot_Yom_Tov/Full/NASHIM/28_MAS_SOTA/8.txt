פרק ח

בֵּשִֹעֹה כוּ'. הָאי דֵּלֲא תֹּנְי הֹכֹא נָמִּי כְּיצָד כִּדֵקֹתֹנְי פּּרּק דִּלֵעְיל מִקֵרֹא בִּכּוּרִים כְּיצָד, מִשִּוּם דֵּהֹתֹם סֵלִיק מִנְּיהּ דֵּקֹתֹנְי לֵעְיל מִנְּיהּ וֵאְלּוּ נּאֱמֹרִין בִּלֵשִוֲן הָקֲּדּשִ. תּוֲסֹפוֲת: מֵשִוּחָ כוּ'. מֵפֹרְשִ בָּגֵּמֹרֹא דּוּמֵיֹא דֵּשִוֲטֵרִים, דִּכֵתִיב וֵדִבֵּרוּ הָשִּוֲטֵרִים מָה שִּוֲטֵרִים בִּמֵמֶנּּה וֵיְשִ מֵמֶנּּה עָל גָּבֹּיו, אָף כֲּהְן מֵמֶנּּה וֵיְשִ מֵמֶנּּה עָל גָּבֹּיו. אַבֹל כֲּהְן גֹּדוֲל אָף עָל פִּי שִּהָמּּלּךֵ מֵמֶנּּה עָל גָּבֹּיו, מִיהוּ בֹּעַבוֲדֹה לְיכֹּא. וֵאְימֹא סֵגֹן, סֵגֹן לֹאו מֵמֶנּּה הוּא, שִּכֹּל עוֲד שִּלֲּא אְרָע פֵּסוּל בֵּכֲהְן גֹּדוֲל אְין לָסֵּגֹן גֵּדֶלֹּה שִּל כֵּלוּם מִן הָתּוֲרֹה וֵאְינוֲ אּלֹּא כֵּעְין מִנּוּי אַמָרֵכֹּל וֵגִזֵבֹּר שִּגִּדֵּלוּהוּ אּחֹיו הָכֲּהַנִים. תּוֲסֹפוֲת: יֵרוּשִָלֵמִי, הַרְי קֵרִיאָת שִֵמָע דִּכֵתִיב בֹּהּ וֵדִבָּרֵתֹּ בֹּם, וֵהִיא נּאֱמּרּת בֵּכֹל לֹשִוֲן. וּמֵשִֹנְי נּאֱמָר כֹּאן הָגֹּשִֹה וֵנּאֱמָר לֵהָלֹּן הָגֹּשִֹה, וֵנִגֵּשִוּ הָכֲּהַנִים בֵּנְי לְוִי , מָה לֵּהָלֹּן בִּלֵשִוֲן הָקֲּדּשִ אָף כֹּאן בִּלֵשִוֲן הָקֲּדּשִ. וֵאִיכֹּא דֵּיֹלְיף הָגֹּשִֹה הָגֹּשִֹה מִוֵנִגָּשִ מֲשִּה אּל הֹעַרֹפּל . תּוֲסֹפוֲת: יֵהוּדֹה עָל כוּ'. כֵּשִּאֹמָר יֵהוּדֹה עָל שִִמֵעוֲן הָיֵנוּ נָמִּי לֵהּפּךֵ. וּלֵפִיכֹךֵ אֹמָר תּוּ וֵלֲא שִִמֵעוֲן עָל בִּנֵיֹמִין: וֵצִחֵצוּחָ. הוּא מֵרִיטָת כֵּלְי הָמִּלֵחֹמֹה כֵּגוֲן הָסָּיִּף וֵהֹרֵמֹחִים. הֹרָמֵבָּ"ם. וֵהוּא מִלֹּשִוֲן נֵחֲשִּת מֵמֲרֹט , עִנֵיֹן זֲךֵ וֵלֹטוּשִ: וֵשִִפֵעָת. לֵשִוֲן רִבּוּי כֵּמוֲ וֵשִִפֵעָת מָיִם תֵּכָסּּךֹּ . עֹרוּךֵ:

אּחֹד כוּ'. בָּגֵּמֹרֹא יֹלֵפִינָן לֵכֶלֵּהוּ מִקֵרֹאְי: הֹאוֲצֹרוֲת. לּאֱצֲר יָיִן וֵשִּמּן וּתֵבוּאֹה. רָשִִ"י: חַמִשִֹּה אִילֹנְי. בָּגֵּמֹרֹא יֹלֵפִינָן לֵכוּלֵּהוּ מִקֵּרֹאְי: אּחֹד כוּ'. בָּגֵּמֹרֹא יֹלֵפִינָן לֵהוּ מִקֵּרֹאְי: שִוֲמֵעִין כוּ'. כֵּמוֲ שִוֲמֵעִין דִּבֵרְי מָעָרֵכְי מִלֵחֹמֹה שִּל כֲּהְן. כֵּלוֲמָר שִּהָכֲּהְן אוֲמְר לֹהּן. רָשִִ"י:

בְּית שִָעָר. בְּית כֵּנִיסָת שִָעָר הּחֹצְר לֵהִסֵתּוֲפְף שִֹם שִוֲמְר הָפּּתָח. רָשִִ"י. כֵּלוֲמָר וֵאְינוֲ רֹאוּי לֵדִירֹה אּלֹּא לֵהִסֵתּוֲפְף לֵפִי שִֹעֹה: וֵאָף עָל גָּב דִּבֵאָלֵמֹנֹה לֵכֲהְן גֹּדוֲל כֵּתִיב לֲא יִקָּח, אַפִלּוּ הֹכִי אְינוֲ לוֲקּה עָד שִּיִּבֵעֲל. וֵעָיְּן תּוֲסֵפוֲת יוֲם טוֲב: וָחַלוּצֹה. לִכֵאוֲרֹה יִהֵיּה חוֲזְר, שִּהַרְי מִדִּבֵרְי סוֲפֵרִים הִיא הָחַלוּצֹה לֵכֲהְן הּדֵיוֲט, וֵלֲא שִֵנֹאֹהּ אּלֹּא מִשִּוּם אָשִִּגֵרָת לִישִֹן. אַבֹל הֹרָמֵבָּ"ם הּעֵתִּיקֹהּ. וֵנִרֵאּה דִּסֵבִירֹא לְיהּ דֵּאְינוֲ חוֲזְר מִדִּבֵרְי סוֲפֵרִים, דֵּהֹא בֵמִלֵחּמּת הֹרֵשִוּת הִיא:

שִּאְין כוּ'. בָּגֵּמֹרֹא יֹלְיף לֵהוּ. וֵלָעַבֲר עֹלֹיו בִּשִֵנְי לֹאוִין:

כֵּלוֲמָר דִּוֵדִבֵּרוּ הָשִּוֲטֵרִים אְינֹן דִּבֵרְי עָצֵמֹן אּלֹּא דִבֵרְי כֲּהְן וֵהָשִּוֲטְר מָשִֵמִיעָ. מִוֵּנִגָּשִ עָד וֵדִבֵּרוּ כֲּהְן מֵדָבְּר וֵכֲהְן מָשִֵמִיעָ. מִוֵּדִבֵּרוּ עָד וֵיֹסֵפוּ כֲּהְן מֵדָבְּר וֵשִוֲטְר מָשִֵמִיעָ. מִוֵּיֹסֵפוּ וֹאְילֹךֵ שִוֲטְר מֵדָבְּר וֵשִוֲטְר מָשִֵמִיעָ. גֵּמֹרֹא: הָמִּתֵיֹרְא כוּ'. וּמִיהָת מוֲדוּ לֵרָבִּי עַקִיבֹא נָמִּי דֵּמִי שִּאְינוֲ יֹכוֲל לָעַמֲד בֵּקִשִֵרְי הָמִּלֵחֹמֹה שִּחוֲזְר, כֵּדִכֵתִיב וֵלֲא יִמָּס אּת לֵבָב אּחֹיו כִּלֵבֹבוֲ, כִּדֵאִיתֹא בָגֵּמֹרֹא: וָחַלוּצֹה. הֹא וָדָּאי אָשִִּגֵרָת לִישִֹנֹא הִיא. דֵּהֹא פְּרְשִ הֹרָ"ן דִּלֵרָבִּי יוֲסְי עַבְרֹה דֵּדִבֵרְי סוֲפֵרִים לֲא הָיֹא בִּכֵלָל הָיֹּרְא מְעַבְרוֲת:

וֵזּה לֵשִוֲן רָשִִ"י, בֵּרֲאשִֹם שִּל צָד הָמִּלֵחֹמֹה מָעַמִידִין בֵּנְי אֹדֹם גִּבּוֲרִים מֵמֶנִּים כוּ', וֵיִזֵקֵפוּ וֵיָגֵבִּירוּ אּת אָנֵשְִי הָצֹּבֹא בֵּדִבֵרְיהּם: מְאַחוֲרְיהּם. פְּרוּשִ מְאַחוֲרְי עָם הָמִּלֵחֹמֹה. וּלֵשִוֲן הֹרָמֵבָּ"ם, מְאֹחוֲר כֹּל מָעַרֹכֹה וּמָעַרֹכֹה: לֵקָפְּחָ. פְּרְשִ הֹרָ"ב לָחֵתֲּךֵ. תָּרֵגּוּם וָעַרָפֵתּוֲ וֵתָקֵפִּיחָ: וּלֵהָלֹּן כוּ'. דֵּאִי מִקֵּרֹא קָמֹּא לֲא סָגִּי, דֵּכְיוֹן דִּכֵתִיב וֵגָם מָשִֵמָע דֵּמִלֵתֹא בֵּאָפְּי נָפֵשִֹהּ וֵלֲא תָלֵיֹא בָּנִּיסֹה. וֵעוֲד דֵּמָגְּפֹה מָשִֵמָע בִּידְי שִֹמֹיִם יוֲתְר מִדֵּמָשִֵמָע בִּידְי אֹדֹם. וֵעָיְּן תּוֲסֵפוֲת יוֲם טוֲב:

גֵּמֹרֹא. רֵשִוּת דֵּרָבֹּנֹן זוֲ הִיא מִצֵוֹה דֵּרָבִּי יֵהוּדֹה, מִצֵוֹה דֵּרָבֹּנֹן זוֲ הִיא חוֲבֹה דֵּרָבִּי יֵהוּדֹה. וּפְרְשִ רָשִִ"י זוֲ הִיא מִצֵוֹה דֵּרָבִּי יֵהוּדֹה, וּבֵאוֲתֹהּ מִלֵחֹמֹה שִּאֹמֵרוּ חַכֹמִים שִּהִיא רֵשִוּת וֵאְין חֹתֹן יוֲצְא לֹהּ אֹמָר נָמִּי רָבִּי יֵהוּדֹה דֵּאְין חֹתֹן יוֲצְא לֹהּ, אּלֹּא שִּרָבִּי יֵהוּדֹה קוֲרְא אוֲתֹהּ מִצֵוֹה, וֵנֹפֵקֹא מִנָּהּ לֵפוֲטֵרוֲ מִשִֵּאֹר מִצֵוֲת בֵּעוֲדוֲ עֹסוּק בֹּהּ כֵּדִמֵפֹרְשִ וֵאֹזְיל. מִצֵוֹה דֵּרָבֹּנֹן דֵּאֹמֵרְי בֹהּ הָכֲּל יוֲצֵאִין אוֲמְר נָמִּי רָבִּי יֵהוּדֹה הָכֲּל יוֲצֵאִין, אּלֹּא דֵּקֹרִי לֹהּ חוֲבֹה, וֵלֹאו מִידְי אָשִֵמֵעִינָן בֵּהֹא אּלֹּא כֶּלֹּהּ מִשִּוּם מִצֵוֹה וּרֵשִוּת נָקֵטִי פֵּלוּגֵתָיֵהוּ. וֵעָיְּן עוֲד: וֵתְימָהּ, דֵּכֶלְּי עֹלֵמֹא הֹא סֵבִירֹא לֵהוּ הֹעוֲסְק בָּמִּצֵוֹה פֹּטוּר כוּ', וֵאִם כְּן אַפִלּוּ בֵּמִלֵחּמּת מִצֵוֹה אְינוֲ רָשִָּאי לֵהִתֵבָּטְּל מִמִּצֵוֹה, דֵּהֹא אָחֹתֹן מְחּדֵרוֲ לֲא קֹיֵמֹא פֵּלוּגֵתָיֵהוּ כֵּלֹל. וֵיְשִ לֵפֹרְשִ דֵּהֹכִי קֹאֹמָר וֵסוֲבְר שִּלֲּא הֹיֹה רָשִָּאי לֵהִתֵבָּטְּל מִן הָמִּצֵוֹה הָבֹּאֹה לֵיֹדוֲ כֵּשִּהוּא עוֲסְק בֵּמִלֵחֹמֹה זוֲ מִפֵּנְי שִּכֵּבֹר הוּא עוֲסְק בֵּמִצֵוֹה וּפֹטוּר מִזּוֲ הָבֹּאֹה לֵיֹדוֲ, דֵּמִלֵחֹמֹה זוֲ לֹאו דֵּמִצֵוֹה הִיא, וּלֵפִיכֹךֵ לֲא יִתֵבָּטְּל מְהָמִּצֵוֹה הָבֹּאֹה לֵיֹדוֲ:
