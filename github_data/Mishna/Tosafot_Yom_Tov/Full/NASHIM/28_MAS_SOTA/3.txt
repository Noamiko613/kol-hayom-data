פרק ג

יְרוּשַׁלְמִי, וְאֵין הַדָּבָר כִּעוּר, וּמֵנִיחַ מַפָּה וְאֵינוֹ חוֹצֵץ, וּמֵבִיא כֹּהֵן זָקֵן. וַאֲפִלּוּ תֵּימָא כֹּהֵן יֶלֶד, שֶׁאֵין יֵצֶר הָרַע מָצוּי לְשָׁעָה. תּוֹסָפוֹת:

מַקְרִיב. גְּמָרָא, הָא אַקְרְבָה, הָכִי קָאָמַר כוּ', וּבַהַשְׁקָאָה גּוּפָהּ פְּלִיגֵי רַבִּי שִׁמְעוֹן וְרַבָּנָן: לָאו דַּוְקָא נָקַט, דִּקְרָא אֲתָא לְרִשּׁוּמָן נִכָּר, דְּכֵיוָן דִּכְתִיב וּמָחָה מִמֵּילָא שָׁמְעִינַן אַרִשּׁוּמוֹ נִכָּר דְּלָאו מִחוּי הוּא כוּ', אֶלָּא עִקַּר קְרָא אִם הִקְרִיב מִנְחָתָהּ וְאַחַר כָּךְ הִשְׁקָהּ כְּשֵׁרָה, כִּדְדָרִישׁ רַבִּי שִׁמְעוֹן וְהִשְׁקָהּ. תּוֹסָפוֹת:

אֵינִי כוּ'. וְאַף עַל פִּי שֶׁאֵינָהּ אוֹמֶרֶת טְמֵאָה אֲנִי שֶׁאֵין כּוֹפִין אוֹתָהּ לִשְׁתּוֹת. הָרַמְבַּ"ם: נִגְנֶזֶת. דְּכָל כִּתְבֵי הַקֹּדֶשׁ שֶׁאֵינָן לִקְרֹא בָּהֶן נִגְנָזִין, שֶׁלֹּא יִתְבַּזּוּ. רַשִׁ"י: וְהַאי מִתְפַּזֶּרֶת הַיְנוּ נִשְׂרֶפֶת כִּדְאָמַר לְקַמָּן וְאֵלּוּ שֶׁמִּנְחוֹתֵיהֶן נִשְׂרָפוֹת כוּ': מְעַרְעֲרִים. לְשׁוֹן עַרְעֵר תִּתְעַרְעַר . שֶׁאִם נִמְחֲקָה מְגִלָּה וְאָמְרָה אֵינִי שׁוֹתָה מְעַרְעֲרִין אוֹתָהּ וּמַשְׁקִין אוֹתָהּ בְּעַל כָּרְחָהּ: רַשִׁ"י. וְנִרְאֶה שֶׁנִּתְכַּוְּנוּ לְהַעֲמִיד הַמִּשְׁנָה כְּרַבִּי עֲקִיבָא דְּסָבַר דְּאוֹמֶרֶת אֵינִי שׁוֹתָה כִּי אוֹמֶרֶת מֵחֲמַת בִּעֲתוּתָא הוּא דִמְעַרְעֲרִין, וְהָכָא מֵחֲמַת בִּעֲתוּתָא קָיְמִינַן. וְאִי נַמִּי לְרַבִּי יְהוּדָה דְּסָבַר דִּלְעוֹלָם מְעַרְעֲרִין, וְטַעְמָא מְפָרְשִׁים אַלִיבֵּיהּ דִּלְעוֹלָם אַף עַל פִּי שֶׁאֵין אָנוּ רוֹאִין בִּעֲתוּתָא יֵשׁ לָנוּ לָחוּשׁ דִּלְמָא מֵחֲמַת בִּעֲתוּתָא:

אֵימָא סֵיפָא אִם יֵשׁ לָהּ זְכוּת תּוֹלָה, וְהָא רַבִּי שִׁמְעוֹן סָבַר דְּאֵין זְכוּת תּוֹלָה. וּמְשָׁנֵי בִּזְכוּת כְּרַבָּנָן סְבִירָא לֵיהּ: דְמֵחֲמַת בְּעִיתוּתָא פִּתְאוֹמִית רְגִילָה לִפְרֹס, שֶׁמַּרְפִּים הַדָּמִים וְנִפְתַּח הַמָּקוֹר. וְלֹא דָּמֵי לַחֲרָדָה מְסַלֶּקֶת הַדָּמִים דִּתְנַן סוֹף פֶּרֶק ד' דְנִדָּה, שֶׁהִיא דְּאָגָה אֵינָהּ בָּאָה פִתְאוֹם. גְּמָרָא: וְטַעְמָא שֶׁהוּא נִקְרָא מַחֲנֵה לְוִיָּה. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: לְפִי שֶׁעַד שַׁעַר נִקָּנוֹר הוּא מַחֲנֵה לְוִיָּה, וְקוֹרְאֵהוּ בִּכְלַל הַר הַבַּיִת, וְשַׁעַר נִקָּנוֹר עַצְמוֹ אַף עַל פִּי שֶׁהוּא בְּעֶזְרַת יִשְׂרָאֵל לֹא נִתְקַדֵּשׁ כוּ', וּקְדֻשַּׁת עֲזָרָה רְצוֹנוֹ לוֹמַר עֶזְרַת יִשְׂרָאֵל וּמִשָּׁם וָאֵילַךְ שֶׁהוּא מַחֲנֶה שְׁכִינָה: אֶלָּא מִיהַת הַכֹּל לְפִי זְכוּתָהּ, שֶׁאֵינָהּ דּוֹמָה עוֹשָׂה מְרֻבֶּה לְעוֹשָׂה מֻעָט: שָׁנָה. וְאַף עַל פִּי שֶׁאֵין רְאָיָה לַדָּבָר זֵכֶר לַדָּבָר, דִּכְתִיב וַחֲטָיָךְ בְּצִדְקָה פְּרֻק וְגוֹ' הֵן תֶּהֱוֵה אַרְכָה לִשְׁלֵוְתָךְ. וּכְתִיב כֹּלָא מְּטָה עַל נְבוּכַדְנֶצַּר מַלְכָּא וְגוֹ' לִקְצָת יַרְחִין תְּרֵי עֲשָׂר. גְּמָרָא: שְׁתֵּי שָׁנִים כוּ'. יְרוּשַׁלְמִי. שָׁנָה אַחַת מִנְבוּכַדְנֶצַּר. שְׁתֵּי שָׁנִים מֵאַמְנוֹן, וַיְהִי לִשְׁנָתַיִם יָמִים . שָׁלֹשׁ שָׁנִים מֵאַחְאָב, וַיֵּשְׁבוּ שָׁלֹשׁ שָׁנִים אֵין מִלְחָמָה : תֵּדַע כוּ'. תִּרְאֶה וְתִלְמֹד בְּפָרָשַׁת סוֹטָה, דְּיָלְפִינַן וְנִקְּתָה וְנִזְרְעָה זֶרַע. וְנִקְּתָה עַל יְדֵי זְכֻיּוֹת כְּדֵי זְרִיעַת זֶרַע תִּשְׁעָה חֳדָשִׁים, וְיֵשׁ אוֹמְרִים יוֹתֵר כְּדִמְפָרֵשׁ בִּגְמָרָא: כְּאִלּוּ כְּתִיב בְּטֵי"ת, מֵעִנְיַן חָבִית שֶׁנִּתְרוֹעֲעָה וּטְפָלָה בִּגְלָלִים, וּטְפֵלָה שֶׁל תַּנּוּר. כִּי הַתָּי"ו וְהַטֵּי"ת מִתְחַלְּפִין בְּמוֹצָא דטלנ"ת. וְהָרַמְבַּ"ם פֵּרֵשׁ תִּפְלוּת דִּבְרֵי הֲבַאי וְדִבְרֵי מְשָׁלִים, מִלְּשׁוֹן הֲיֵאָכֵל תָּפֵל בְּלִי מֶלַח , שֶׁעִנְיָנוֹ דָּבָר חָסֵר וְגָרוּעַ: וְרַבָּנָן כְּלוֹמַר בֶּן עַזַּאי, מִבַּעְיָא לֵיהּ לְכִדְרַבִּי יוֹסֵי בַּר חֲנִינָא דְּאָמַר אֵין דִּבְרֵי תּוֹרָה מִתְקַיְּמִין אֶלָּא בִּידֵי מִי שֶׁמַּעֲמִיד עַצְמוֹ עָרוּם עָלֶיהָ. פֵּרֵשׁ רַשִׁ"י שֶׁפָּרַשׁ מִכָּל עֲסָקָיו וְנַעֲשָׂה עָלֶיהָ עָנִי וַחֲסַר כֹּל. לִישָׁנָא אַחֲרִינָא שֶׁמַּעֲרִים שֶׁיִּתְקַיֵּם תּוֹרָתוֹ לִקְבֹּץ עַל יָד לִלְמֹד מִכָּל אָדָם: לְשׁוֹן הָרַמְבַּ"ם, הַחֲכָמִים קוֹרִין עַצְמָם פְּרוּשִׁים לֶהֱיוֹתָם מֻפְרָשִׁים מִבְּנֵי אָדָם שֶׁיֵּשׁ לָהֶם חֶסְרוֹנוֹת וְהַמִּדּוֹת הַפְּחָתִיּוֹת וְהָרְדִיפָה אַחַר תַּאֲוַת הָעוֹלָם. הֵם מְיַחֲלִים לִשְׂכַר הָעוֹלָם הַבָּא וְלַמִּדּוֹת הַמְּעֻלּוֹת: מְבַלֵּי כוּ'. אוֹ מְכַלֵּי עוֹלָם. שֶׁאֵלּוּ הַדְּבָרִים הַמַּפְסִידִים מְצִיאוּת הָאָדָם. הָרַמְבַּ"ם. וְרָצָה מְצִיאוּת הָאָדָם לְפִי שֶׁהוּא עִקַּר הַבְּרִיאָה בָּעוֹלָם הַשָּׁפֵל. וְכֵן אָמְרוּ עַל שְׁלֹשָׁה דְּבָרִים הָעוֹלָם עוֹמֵד:

מִתְנַוְּנָה. פְלִיגֵי בָּהּ, רַב יוֹסֵף סְבִירָא לֵיהּ דִּלְתַנָּא קַמָּא מִתְנַוְּנָה נַמִּי לֹא, וְרַב שֵׁשֶׁת סוֹבֵר דְּתַנָּא קַמָּא נַמִּי סְבִירָא לֵיהּ דְּתוֹלָה אֲבָל מִתְנַוְּנָה וְכוּ'. וּפֵרֵשׁ רַשִׁ"י וְלֹא פְלִיגֵי עֲלֵיהּ דְּרַבִּי אֶלָּא בְּסוֹף שֶׁהִיא מֵתָה בְּאוֹתָהּ מִיתָה לְחוּד. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב:

עֵדִים כוּ' בָּא עָלֶיהָ. וְאַף עַל גַּב דְּבְהָנֵי אִגַּלֵּי מִלְּתָא לְמַפְרֵעַ דְּכִי קַדּוּשׁ בְּטָעוּת קַדּוּשׁ, גְּזֵרָה שֶׁמָּא יֹאמְרוּ מוֹצִיאִים מִכְּלֵי שָׁרֵת לְחֻלִּין. גְּמָרָא. תּוֹסָפוֹת חֲדָשִׁים: שֶׁלֹּא בְדִקְדּוּק כָּתַב כֵּן, דִּקְרֵיבָה הַיְנוּ עַל הַמִּזְבֵּחַ, וְאִלּוּ בֵּית הַדֶּשֶׁן הַיְנוּ מָקוֹם בָּעֲזָרָה לִפְסוּלֵי מֻקְדָּשִׁים הַנִּשְׂרָפִים, כְּמוֹ שֶׁכָּתַב הָרַ"ב לְעֵיל מִשְׁנָה ג'. אֶלָּא כְּתַנָּאֵי. וְכֻלֵּי עָלְמָא כָּל שֶׁמִּמֶּנּוּ לָאִשִּׁים הֲרֵי זֶה בְּבַל תַּקְטִירוּ מִדִּכְתִיב לֹא תַקְטִירוּ מִמֶּנּוּ כוּ'. כָּל שֶׁמִּמֶּנּוּ כוּ'. דְּרַבִּי אֱלִיעֶזֶר סָבַר מַקְרִיב הַשְּׁיָרִים עַל הַמִּזְבֵּחַ וּלְשֵׁם עֵצִים. וּלְרַבָּנָן לֵית לְהוּ הַאי סְבָרָא אֶלָּא נִשְׂרָפִים שְׁיָרֶיהָ עַל בֵּית הַדֶּשֶׁן. כִּדְאִיתָא בִּגְמָרָא:

לָאו דַּוְקָא, דְּהוּא הַדִּין חוֹבָה. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: כִּדְדָרְשִׁינַן בְּקִדּוּשִׁין דַּף ע"ז: וּלְשׁוֹן סִפְרָא, לְהֵחַלּוֹ, בִּזְמַן שֶׁהוּא נוֹהֵג מִנְהָג זֶה הֲרֵי הוּא חֻלִּין: וְאַף עַל גַּב דְּמִבְּנֵי אַהֲרֹן מְמַעֲטִינַן וְלֹא בְנוֹת אַהֲרֹן, טָרַח קְרָא וְכָתַב זָכָר לְרַבּוֹת בַּעֲלֵי מוּמִין לְמַחֲלוֹקֶת, כְּמוֹ שֶׁכָּתַב הָרַ"ב בְּרֵישׁ פֶּרֶק י"ב דִּזְבָחִים:

מַה בֵּין כוּ'. שִׁיֵּר טוּבָא, הָנָךְ דְּסוֹף פֶּרֶק קַמָּא דְּקִדּוּשִׁין, וְכֵן הָא דְּאֵין אִשָּׁה מֵפֵרָה נִדְרֵי בִתָּהּ: עַיֵּן פֵּרוּשׁוֹ פֶּרֶק ד' דְּנָזִיר מִשְׁנָה ו' כוּ'. וְנִרְאֶה לִי דְּפֵרוּשׁ לִכְשֶׁיִּגְדַּל כְּלוֹמַר לְאַחַר שֶׁנַּעֲשָׂה גָּדוֹל לְעִנְיָן נְדָרִים, וּמִיהוּ עַד שֶׁיָּבִיא שְׁתֵּי שְׂעָרוֹת אַחַר שְׁלֹשׁ עֶשְׂרֵה שָׁנִים וְתוּ לֹא, דִפְשִׁיטָא דְּמִכֵּיוָן שֶׁיָּצָא מֵרְשׁוּתוֹ לְגַמְרֵי פָּקְעָה לָהּ הַדָּרַת אָבִיו וּמִמֵּילָא אֵין צָרִיךְ לְהַשְׁלִים עוֹד נְזִירוּתוֹ. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: בִּתּוֹ. לְאָמָה כְּשֶׁהִיא קְטַנָּה. רַשִׁ"י: בִּתָּהּ. דִּכְתִיב כִּי יִמְכֹּר אִישׁ אֶת בִּתּוֹ . גְּמָרָא: לָאו לַאֲפוֹקֵי נַעֲרָה, דִּבְהֶדְיָא תְּנַן הָאִישׁ מְקַדֵּשׁ אֶת בִּתּוֹ כְּשֶׁהִיא נַעֲרָה. אֶלָּא לַאֲפוֹקֵי בּוֹגֶרֶת: מְקַדֶּשֶׁת. מֵרֵישָׁא דִּקְרָא דְּאֶת בִּתִּי נָתַתִּי לְאִישׁ יָלֵיף לֵיהּ, וְאָמַר אֲבִי הַנַּעֲרָה אֶל הַזְּקֵנִים, מַשְׁמַע לְמַעֵט אִמָּהּ דְּאֵינָהּ זַכָּאָה בְּבִתָּהּ לָתֵת. תּוֹסָפוֹת: אוֹתוֹ לְגוּפֵיהּ אִצְטְרִיךְ, דִּבְלֹא כְּסוּתוֹ, וְעִקַּר טַעְמָא מִשּׁוּם בִּזְיוֹנָא. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב:
