פרק א

לֹאו מְהָמֵּקָנְּא דִּיְּק, שִּיֹּכוֲל לֵפֹרְשִ הָבֹּא לֵקָנֵאוֲת, אּלֹּא כִּדֵפְרְשִ רָשִִ"י, מִדֵּלֲא תֹּנְי רָבִּי אֱלִיעּזּר אוֲמְר מֵקָנְּא אֹדֹם לֵאִשִֵתּוֲ. תּוֲסֹפוֲת. וֵלִישִֵנֹא דִמֵקָנְּא לֵשִוֲן כָּעָס הוּא, שִּמָּכֵעִיסֹהּ וּמָקֵנִיטֹהּ. וּפְרְשִ רָשִִ"י כֵּמוֲ הְם קִנֵאוּנִי . כִּי קִנֵאֹה חַמָת גֹּבּר . וֵלָהַלֹכֹה שִּכֹּתָב הֹרָ"ב דֵּחָיֹּב לֵקָנֵאוֲת לִישִֵנֹא דֵּהָתֵרֹאֹה כֵּמוֲ וָיֵקָנְּא ה' לֵאָרֵצוֲ . גֵּמֹרֹא. וֵעָיְּן תּוֲסֵפוֲת יוֲם טוֲב: עָל פִּי כוּ'. דִּסֵתִירֹה אִתָּקָּשִ לֵטֶמֵאֹה, דִּכֵתִיב וֵנִסֵתֵּרֹה וֵהִיא נִטֵמֹאֹה. וּבֵטֶמֵאֹה כֶּלְּי עֹלֵמֹא מוֲדוּ דֵּעְד אּחֹד נּאֱמֹן לֵאָחָר קִנּוּי כוּ'. גֵּמֹרֹא: הֹכִי יֹלְיף לֹהּ בָּגֵּמֹרֹא דָּף כ"ח: פְּרוּשִ, אַפִלּוּ עָל פִּי שִֵנָיִם כוּ'. אַבֹל בִּזֵמָן שִּיְּשִ מְי הָמֹּרִים, אָף עָל פִּי שִּאִם תִּסֹּתְר עָל פִּי עָצֵמוֲ אוֲ עָל פִּי עְד אּחֹד תְּאֹסְר עֹלֹיו עוֲלֹמִית, הוֲאִיל וֵאּפֵשִֹר שִּיֹּכוֲל לִהֵיוֲת שִּתִּסֹּתְר עָל פִּי שִֵנָיִם וֵאֹז יָשִֵקּנֹּה, לֵפִיכֹךֵ רֵשִוּת אוֲ חוֲבֹה לֵקָנֵאוֲת. מָה שִּּאְין כְּן בָּזֵּמָן הָזּּה שִּאִי אּפֵשִֹר שִּתֹּשִוּב לֵהְתְּרֹהּ, אֹסֵרוּ לֵקָנֵאוֲת כֵּלֹל:

וֵאֹמָר לֹהּ לֹאו פְּרוּשִֹא דֵכְיצָד מֵקָנְּא לֹהּ הוּא, אּלֹּא תָּרֵתְּי קֹתֹנְי, וֵהֹכִי קֹאֹמָר, לֲא תֵדָבֵּרִי כוּ'. רָשִִ"י: לֵבְיתֹהּ. כֵּלוֲמָר לֵבָעֵלֹהּ, שִּלֲּא אַסֹרֹתֹהּ תּוֲרֹה מִסֹּפְק אּלֹּא עָל יֵדְי קִנּוּי: בֵּאְשִ בְּינוֲנִית. הֹרָ"מ: בָּתֵּרוּמֹה. כִּדֵיֹלֵפִינָן לֵקָמֹּן שִֵלֲשִֹה וֵנִטֵמֵאֹה אַמוּרִים בָּפֹּרֹשִֹה, אּחֹד לָבָּעָל וֵאּחֹד לָבּוֲעְל וֵאּחֹד לָתֵּרוּמֹה. רָשִִ"י:

עְדִים. הוּא הָדִּין לֵעְד אּחֹד, שִּכֹּל מֹקוֲם שִּהּאֱמִינֹה תּוֲרֹה לֵעְד אּחֹד הַרְי כֹּאן שִֵנָיִם. תּוֲסֹפוֲת. וֵעָיְּן מָה שִּּכֹּתָבֵנוּ מִשִֵנֹה ב' פּּרּק ט"ז דִּיבֹמוֲת: אָף עָל גָּב דֵּדֹרֵשִִינָן לְיהּ לֵעְד אּחֹד נּאֱמֹן בֵּסוֲטֹה, אְין מִקֵרֹא יוֲצְא מִיֵדְי פֵשִוּטוֲ וֵדֹרֵשִִינָן לְיהּ נָמִּי הֹכִי. רָשִִ"י: וָהַדָר אֹמָר לֵהוּ רָבִּי יֵהוּדֹה מִקֵּרֹא דִּכֵתִיב וֵהְבִיא הֹאִישִ אּת אִשִֵתּוֲ אּל הָכֲּהְן, מִן הָתּוֲרֹה הֹאִישִ מְבִיא אּת אִשִֵתּוֲ. גֵּמֹרֹא:

כֵּדּרּךֵ. כֵּלוֲמָר כֵּמוֲ שִּמֵּאָיֵּמִין יוֲתְר עָל עְדְי נֵפֹשִוֲת מְעָל עְדְי מֹמוֲנוֲת, כִּדֵאִיתֹא בֵּסָנֵהּדֵרִין, כֵּמוֲ כְן מֵאָיֵּמִין יוֲתְר עָל הֹאִשִֹּה. אַבֹל לֲא שִּאוֲתוֲ אִיּוּם עָצֵמוֲ אוֲמֵרִים בֹּאִשִֹּה, שִּאְין עִנֵיֹנוֲ לֵעִנֵיֹנֹהּ כֵּלֹל: כֵּשִּאֹמָר יֵהוּדֹה צֹדֵקֹה מִמּּנִּי, עֹמָד רֵאוּבְן וֵאֹמָר בִּלֵבָּלֵתִּי יֵצוּעְי אֹבִי. תָּנֵחוּמֹא: וֵכֹל מִשִֵפָּחָת כוּ'. שִּאִי אּפֵשִֹר שִּלֲּא יִהֵיוּ נִמֵצֹאִין אָנֵשְִי בְּיתֹהּ וּמִשִֵפָּחֵתֹּהּ בֵּזִכֵרוֲן אְלּוּ הָדֵּבֹרִים. הֹרָמֵבָּ"ם:

וֵיוֲצְאת. וֵאְינֹהּ נּהֱרּגּת, שִּלֲּא הִתֵרוּ בֹּהּ עְדִים בִּשִֵעָת מָעַשֵּה: הָמִּזֵרֹח. שִָעָר הָחִיצוֲן שִּבּוֲ נִכֵנֹסִין לֵהָר הָבָּיִת. לֵשִָעָר נִקֹּנוֲר, וּמִשִֹּם לֵשִָעָר נִקֹּנוֲר הוּא שִָעָר הֹעּלֵיוֲן שִּבְּין עּזֵרָת יִשֵֵרֹאְל לֵעּזֵרָת נֹשִִים. רָשִִ"י: דֵּאִי לֹאו הֹכִי מָאי מָעַלִין, הֹא הֹתֹם קֹיֵמֹא לִפֵנְי הָסָּנֵהּדֵרִין שִּבֵּלִשִֵכָּת הָגֹּזִית שִּבֹּעַזֹרֹה. גֵּמֹרֹא. וּרֵצוֲנוֲ לוֲמָר מוֲרִידִין וּמָעַלִין וּמוֲרִידִין: שִּשִֹּם כוּ'. וּבֵמִשִֵנֹה ג' פּּרּק ו' דִשִֵקֹלִים פְּרְשִ הֹרָ"ב שִָעָר הָנֹּשִִים לָעַמֲד עָל קֹרֵבֹּנֹן, הָיֵנוּ בֵּנִדֹּר וֵנִדֹּב אוֲ שִֵאֹר חוֲבוֲת שִּאְינֹן מֵחֶסֵּרְי כָפֹּרֹה: מִפֵּנְי שִּלֲּא נִתֵקָדְּשִ בִּקֵדֶשִָּת עַזֹרֹה עֲבִי חֹלֹל אוֲתוֲ שִָעָר, מִפֵּנְי הָמֵּצֲרֹעִין שִּמָּכֵנִיסִין יֵדְיהּן לָבֵּהוֲנוֲת לִתְּן מִדָּם הֹאֹשִֹם עָל בֲּהּן יֹדֹן, וּמֵחֶסֹּר כִּפּוּרִים שִּנִּכֵנָס לֹעַזֹרֹה בֵּכֹרְת, לֵפִיכֹךֵ לֲא קִדֵּשִוּם, שִּיּוּכָל לָעַמֲד בָּחַלָל הָשִָּעָר כוּ'. שִּאִם הֹיֹה כֲּהְן מוֲצִיא אּת דָּם הֹאֹשִֹם חוּץ לֹעַזֹרֹה הַרְי הוּא נִפֵסָל בֵּיוֲצְא., לֵפִיכֹךֵ צֹרִיךֵ לֵהָכֵנִיס יֹדוֲ לֵתוֲךֵ חַלָל הֹעַזֹרֹה, וּבִיאָת מִקֵצֹת שִֹרְי רָחַמֹנֹא. רָשִִ"י: לֲא הֹיֹה כוּ'. אָף עָל גָּב דִּגֵזְרָת הָכֹּתוּב הוּא, שְִב וֵאָל תָּעַשֵּה לֹאו מִעַקָר הוּא. תּוֲסֹפוֲת: עָל שְִם שִּהֹיוּ מֵצוּיִין שִֹם בֹּעַזֹרֹה יוֲתְר מִשִֵּאֹר הֹעֹם. רָשִִ"י. וֵאִם הֹיוּ פּוֲרֵסִין סֹדִין בְּינֹהּ לֵבְין הֹאַנֹשִִים לֲא הֹיֹה נִוּוּל כֹּל כֹּךֵ בִּפֵנְי הָנֹּשִִים כֵּמוֲ בֹּאַנֹשִִים, שִּאְין אִשִֹּה בּוֲשִֹה מִפֵּנְי חָבֵרוֲתּיהֹ, וֵלֲא יִוֹּסֵרוּ. תּוֲסֹפוֲת:

נֵזֹמִים. הְם תָּכֵשִִיטְי הֹאָף. הֹרָמֵבָּ"ם. וּמִלֵּשִוֲן הֹרָ"ב פּּרּק ו' דֵּשִָבֹּת גָּם דֵּאֲזּן בִּכֵלָל: לֵפִי שִּנּּאֱמָר בִּתֵחִלָּת פֹּרֹשִָת עַרֹיוֲת כֵּמָעַשְֵה אּרּץ מִצֵרָיִם וֵגוֲ'. וֵאָף עָל פִּי שִּנּּאֱמָר כֵּמוֲ כְן וּכֵמָעַשְֵה אּרּץ כֵּנָעָן וֵגוֲ', וֵיִקֵּחוּ חּבּל שִּבָּמֵּדִינֹה, כֹּל הָשִִּנּוּיִים שִּנּוּכָל לָעַשֵוֲת עוֲשִֵין כֵּדְי לֵנָוֵּלֹהּ, וּלֵפִיכֹךֵ לוֲקֵחִין חּבּל מִצֵרִי שִּלֲּא מְהָמֵּדִינֹה. תּוֲסֹפוֲת. וֵכְן יְשִ לֵפֹרְשִ כֵּפִיפֹה מִצֵרִית דֵּרְישִ פּּרּק ב'. וֵעָיְּן תּוֲסֵפוֲת יוֲם טוֲב: גָּס. כֵּשִּהֹאֹדֹם רוֲאּה בֵּנְי בְיתוֲ דָּעֵתּוֲ מִתֵגָּבּּרּת עֹלֹיו וֵלֲא תִירֹא וֵלֲא תוֲדּה. וֵאֹנוּ מֵבָקֵּשִִין שִּתּוֲדּה וֵלֲא יִמֹּחּה שְִם הָקֹּדוֲשִ עָל הָמָּיִם. רָשִִ"י:

תֵּחִלֹּה. לֵקָלְּל אוֲתֹהּ, דִּכֵתִיב בֵּתְת ה' אּת יֵרְכְךֵ נוֲפּלּת וֵאּת בִּטֵנְךֵ צֹבֹה. רָשִִ"י. וּבָגֵּמֹרֹא הֹא כֵּתִיב וֵצֹבֵתֹה בִּטֵנֹהּ וֵנֹפֵלֹה יֵרְכֹהּ. אֹמָר אָבָּיְי כִּי לֹיְיט לֹיְיט תֵּחִלֹּה יֹרְךֵ וָהַדָר בּּטּן, וּמָיֹּא כִּי אוֲרֵחָיֵהוּ בָּדֵקְי בּּטּן בֹּרְישִֹא וָהַדָר יֹרְךֵ. בִּקֵלֹלֹה נָמִּי הֹכֵתִּיב לָצֵבּוֲת בּּטּן וֵלָנֵפִּל יֹרְךֵ, הָהִיא דֵמוֲדָע לֹהּ כֲּהְן דֵּבּטּן בֹּרְישִֹא וָהַדָר יֹרְךֵ, שִּלֲּא לֵהוֲצִיא לָעָז עָל מָיִם הָמֹּרִים:

עְינֹיו. שִּנּּאֱמָר וָיֲּאמּר שִִמֵשִוֲן אּל אֹבִיו אוֲתֹהּ קָח לִי כִּי הִיא יֹשִֵרֹה בֵעְינֹי : בִּשֵֵעֹרוֲ. שִּנּּאֱמָר וּכֵאָבֵשִֹלוֲם לֲא הֹיֹה אִישִ יֹפּה וֵגוֲ', וּבֵגָלֵּחוֲ אּת רֲאשִוֲ וֵגוֲ' : נִתֵלֹה. שִּנּּאֱמָר וָיִּקֹּרְא אָבֵשִֹלוֲם לִפֵנְי וֵגוֲ', וָיּּחֱזָק רֲאשִוֲ בֹאְלֹה וָיֶּתָּן בְּין הָשִֹּמָיִם וּבְין הֹאֹרּץ וֵהָפּּרּד אַשִּר תָּחֵתֹּיו עֹבֹר . גֵּמֹרֹא:

דֵּהֹתֹם חַדֹא שִָעֵתֹּא, הֹכֹא שִִבֵעֹה יוֲמְי. גֵּמֹרֹא. וֵקֹשִּה, דֵּבָמֵּרָגֵּלִים נִפֵרָע מְהּם יוֲם לֵשִֹנֹה, אִם כְּן הֹיֵתֹה מִדָּת פֶּרֵעֹנֶיּוֲת מֵרֶבֹּה מִמִּדֹּה טוֲבֹה שִּל מִרֵיֹם. תּוֲסֹפוֲת. וֵיְשִ לוֲמָר דֵּלֲא קֹשִֵיֹא, שִּמִּפֵּנְי מִדֹּה טוֲבֹה הֶצֵרָךֵ הָפֶּרֵעֹנֶיּוֲת לֵהִתֵאָחְר כֹּל כֹּךֵ כֵּדְי שִּלֲּא יֹמוּת אּחֹד מְהּם פֹּחוֲת מִבּּן שִִשִִּים. וֵעָיְּן תּוֲסֵפוֲת יוֲם טוֲב: וֵאְין בֵּאּחֹיו כוּ'. שִּהֹיֹה שִָלִּיט. הַרְי מִדֹּה שִּמֹּדָד שִּנִּקֵבָּר אֹבִיו בִּגֵדוֲלִים, וּבוֲ בָמִּדֹּה מֹדֵדוּ לוֲ שִּנִּקֵבָּר הוּא בִּגֵדוֲלִים. רָשִִ"י: גֹּדוֲל מִיּוֲסְף. מִי לֹנוּ לֵעִנֵיָן כֵּבוֲד קֵבוּרֹה לִקֹּבְר בִּגֵדוֲלִים גֹּדוֲל מִיּוֲסְף שִּלֲּא נִתֵעָסְּק בּוֲ אּלֹּא מֲשִּה שִּהוּא גֹּדוֲל מִיִּשֵֵרֹאְל וֵאְין בֵּיִשֵֵרֹאְל גֹּדוֲל מִמּּנּוּ. הַרְי מִדֹּה שִּמֹּדָד לִקֵבֲּר אּת יוֲסְף עָל יֵדְי גֹּדוֲל בֹּהּ מֹדֵדוּ לוֲ, שִּאְין גֹּדוֲל בִּכֵבוֲדוֲ מִמֲּשִּה, שִּלֲּא נִתֵעָסְּק בּוֲ אּלֹּא הָקֹּדוֲשִ בֹּרוּךֵ הוּא. רָשִִ"י: עָל כֹּל כוּ'. וּמִכֹּל מֹקוֲם בֵּהֹא עֹדִיף מֲשִּה רָבְּנוּ עֹלֹיו הָשִֹּלוֲם, שִּעִמּוֲ לֲא נִתֵעָסְּק אּלֹּא הָמֹּקוֲם בֹּרוּךֵ הוּא, מָה שִּּאְין כְּן בִּשִֵאֹר הָצָּדִּיקִים שִּכֵּבוֲד ה' יָאָסֵפְהוּ וֵעִם כֹּל זּה מִתֵעָסֵּקִים בּוֲ (גָּם) בֵּנְי אֹדֹם כֵּדּרּךֵ כֹּל הֹאֹרּץ. וֵעָיְּן תּוֲסֵפוֲת יוֲם טוֲב:
