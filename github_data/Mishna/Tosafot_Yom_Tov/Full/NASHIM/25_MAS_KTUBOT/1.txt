פרק א - משנה א

שֶׁפַּעֲמַיִם כוּ'. אַבְּתוּלָה קָאֵי. רַשִׁ"י. וְכָתַב הָרַ"ן דְּמַקְשׁוּ הָכָא, דְּלִתְנֵי נַמִּי טַעֲמָא דְּשָׁקְדוּ לְמִעוּטֵי יוֹם רִאשׁוֹן בְּשַׁבָּת, וְטַעֲמָא דְּאַלְמָנָה בְּיוֹם חֲמִישִׁי כוּ'. עוֹד יֵשׁ לוֹמַר דְּמִשּׁוּם שָׁקְדוּ לֹא הֲווֹ מְתַקְּנֵי רַבָּנָן לְעִכּוּבֵי מִצְוָה דְנִשּׂוּאִין, אֶלָּא מִשּׁוּם טַעֲמָא דִבְתוּלִים דְּמִלְּתָא דְאִסּוּרָא הִיא הֻצְרְכוּ לְיַחֵד לַבְּתוּלָה יָמִים, וְכֵיוָן שֶׁהֻצְרְכוּ לְיַחֵד יִחֲדוּ לַבְּתוּלָה רְבִיעִי דְּמַעֲלֵי טְפֵי מֵאֶחָד בְּשַׁבָּת מִשּׁוּם שָׁקְדוּ, וְאַגַּב, יִחֲדוּ לָאַלְמָנָה גַּם כֵּן, דְּאִי לֹא, אָתוּ לְמֵימַר דְּנִשּׂוּאֵי אַלְמָנָה זִילוּ לְרַבָּנָן וְלֹא מְתַקְּנֵי בְהוּ מִלְּתָא, וּמִשּׁוּם הָכִי תָּנֵי טַעֲמָא דִבְתוּלִים, לוֹמַר דְּעִקַּר הַתַּקָּנָה הָיְתָה בִּשְׁבִילָהּ. וְעַיֵּן עוֹד: הַשֵּׁנִי כוּ'. דִּשְׁכִיחֵי, דְּאָתוּ לְמִקְרֵי בְּסִפְרָא. גְּמָרָא: פֵּרוּשׁ, לְאַחַר שֶׁנִּתְאָרְסָה, וַהֲרֵי הִיא אֲסוּרָה עָלָיו, כְּגוֹן אֵשֶׁת כֹּהֵן, אוֹ שֶׁקִּבֵּל בָּהּ אָבִיהָ קִדּוּשֶׁיהָ פְּחוּתָה מִבַּת שָׁלֹשׁ, דְּלֵיכָּא אֶלָּא חַד סְפֵקָא, בְּכֹהֵן, שֶׁמָּא קֹדֶם שֶׁנִּתְאָרְסָה, וּבִפְחוּתָה מִבַּת שָׁלֹשׁ, שֶׁמָּא בְּאֹנֶס. וּמִשּׁוּם הָכִי אַף בִּשְׁאָר נָשִׁים דְּאִיכָּא סְפֵק סְפֵקָא וְאֵינָהּ אֲסוּרָה עָלָיו לֹא פְּלוּג רַבָּנָן בְּתַקַּנְתָּא. תּוֹסָפוֹת: גְּמָרָא. וְהָכִי קָאָמַר, מִצַּד תַּקָּנָה זוֹ דְמִשְׁנָתֵנוּ נִשֵּׂאת בְּכָל יוֹם. אֲבָל וַדַּאי יֵשׁ יָמִים דְּאִסּוּר אַחֵר חָל עֲלַיְהוּ, כְּגוֹן שַׁבָּת וְיוֹם טוֹב וְחֹל הַמּוֹעֵד: וְאִי נִשֵּׂאת בְּיוֹם אַחֵר לֹא מִקַּיְמָא תַּקַּנְתָּא אֶלָּא מַשְׁכִּים לְמָחָר לִמְלַאכְתּוֹ וְהוֹלֵךְ לוֹ:

פרק א - משנה ב

מָאתַיִם. בְּסוֹף פֶּרֶק י"ג פָּסַק הָרַ"ב דִּכְתֻבַּת אִשָּׁה דְּרַבָּנָן, וְהַשְׁתָּא הָנֵי מָאתַיִם בְּכֶסֶף מְדִינָה הֵם, שֶׁכֵּן כָּל שֶׁל דִּבְרֵיהֶם, וְיֵשׁ בְּכָל דִּינָר שִׁבְעָה חֲלָקִים נְחֹשֶׁת וּשְׁמִינִית כֶּסֶף, נִמְצָא מָאתַיִם, עֶשְׂרִים וַחֲמִשָּׁה שֶׁל כֶּסֶף טָהוֹר, וְזוֹ דַעַת הָרַמְבַּ"ם. וְהָרֹא"שׁ סוֹבֵר דְּבִכְתֻבָּה תִּקְּנוּ בְּסֶלַע שֶׁל תּוֹרָה שֶׁכֻּלּוֹ כֶּסֶף טָהוֹר, דְּזֶה הַכְּלָל לֹא נֶאֱמַר אֶלָּא בְּסֶלַע, וְלֹא בְזוּזֵי וּמָנֶה: וַחֲלוּצָה. לִרְבוּתָא נָקַט, דְּסָלְקָא דַעְתָּךְ אֲמִינָא כֵּיוָן דְּשׁוֹמֶרֶת יָבָם לֹא מִחַסְּרָה כְּנִיסָה לַחֻפָּה שֶׁהֲרֵי יְבָמָהּ יָבֹא עָלֶיהָ בְּעַל כָּרְחָהּ וַהֲוָה אֲמִינָא דְּכִנְשׂוּאָה דַּמְיָא, קָמַשְׁמַע לָן. תּוֹסָפוֹת: שֶׁנִּפְדּוּ כוּ'. בְּשֵׁם רַבֵּנוּ מוהר"ר לִיוָאֵי ז"ל שָׁמַעְתִּי טַעַם שֶׁשּׁוֹנֶה שֶׁנִּפְדּוּ , שֶׁכֵּן מִצְוַת פִּדְיוֹן שְׁבוּיִם קוֹדֶמֶת. וּלְהִתְגַּיֵּר אַף עַל פִּי שֶׁדּוֹחִין אוֹתוֹ בִּתְחִלָּה, אֵין מַרְבִּין עָלָיו וְאֵין מְדַקְדְּקִין עָלָיו, אֲבָל לְשַׁחְרֵר עוֹבֵר בַּעֲשֵׂה דִּוְהִתְנַחַלְתֶּם אוֹתָם: פְּחוּתוֹת כוּ'. שֶׁאֲפִלּוּ נִבְעֲלוּ, בְּתוּלֵיהֶן חוֹזְרִין. רַ"ן:

פרק א - משנה ג

וּמִכָּל מָקוֹם לֹא דָמִי לִקְטַנָּה הַנִּבְעֶלֶת, דְּהָתָם בְּתוּלֶיהָ חוֹזְרִין, אֲבָל קָטָן וְהִשִּׁיר הַבְּתוּלִין לֹא עֲדִיפָא מִמֻּכַּת עֵץ, אֶלָּא הָכָא בְּלֹא הִשִּׁיר בְּתוּלִים אַיְרֵי. תּוֹסָפוֹת וְהָרֹא"שׁ:

פרק א - משנה ד

הַנִּשּׂוּאִין. שֶׁנִּכְנְסָה לַחֻפָּה וְיֵשׁ לָהּ עֵדִים שֶׁלֹּא נִסְתְּרָה. אִי נַמִּי נִסְתְּרָה וְלֹא שָׁהֲתָה כְּדֵי בִיאָה. גְּמָרָא: וְאֵין כוּ'. אֵין יָכוֹל לְהַפְסִידָהּ אוֹתָהּ מָנֶה וְלוֹמַר בְּחֶזְקַת בְּתוּלָה נְשָׂאתִיךְ וַהֲרֵי מִקְחִי מֶקַּח טָעוּת. רַשִׁ"י:

פרק א - משנה ה

הָאוֹכֵל. מִדְּקָתָנֵי הָאוֹכֵל, מִכְּלַל דְּאִיכָּא דוּכְתָּא בִּיהוּדָה (עַצְמוֹ) דְּלֹא אָכֵיל, וְנַפְקָא מִנָּהּ דְּהֵיכָא דְּלֹא נְהִיגֵי לְיַחֵד יָכוֹל לִטְעֹן. גְּמָרָא וְרַשִׁ"י: אַלְמְנַת כֹּהֵן. אַלְמָנָה בַת כֹּהֵן. רַשִׁ"י: לַבְּתוּלָה. בַּת כֹּהֵן כְּשֶׁנִּשֵּׂאת לְכֹהֵן. וְכָל שֶׁכֵּן כְּשֶׁנִּשֵּׂאת לְיִשְׂרָאֵל: וְלֹא מִחוּ כוּ'. אַף עַל פִּי שֶׁלֹּא כָתַב בִּלְשׁוֹן תּוֹסֶפֶת, אֶלָּא אַרְבַּע מֵאוֹת דְּחָזוּ לִיכִי, וְקָא סָלְקָא דַעְתָּךְ דְּלֹא תִגְבֶּה דְּלֹא חָזוּ , וּבִלְשׁוֹן תּוֹסֶפֶת לֹא כָתַב, קָמַשְׁמַע לָן דְּמִנְהָג טוֹב וְרָאוּי לִהְיוֹת בַּכְּהֻנָּה, וְשַׁיָּךְ בְּהוּ דַּחֲזוּ לִיכִי. תּוֹסָפוֹת:

פרק א - משנה ו

לֹא כִי. אִי אֶפְשָׁר לְפָרֵשׁ שֶׁמַּכְחִישָׁהּ בְּבָרִי, דְּמֵהֵיכָא יָדַע, וְעוֹד, לָמָּה כְּנָסָהּ וּבְעָלָהּ. אֶלָּא הָכִי קָאָמַר, שֶׁמָּא לֹא כִי: אֲפִלּוּ מָנֶה. דְּלֹא דָמִי לְמֻכַּת עֵץ דִּלְעֵיל. גְּמָרָא: נֶאֱמֶנֶת. דְּמוֹקְמִינַן אִתְּתָא אַחֲזָקָה. דִּמְסַפְּקִינַן אַשְּׁעַת אֵרוּסִין אִי בְּתוּלָה הֲוַאי, וְהַעֲמֵד הָאִשָּׁה אַחֶזְקָתָהּ וּבְתוּלָה נוֹלְדָה, וְהַךְ שָׁעָה הוּא דְּאִיתְנִיסָא לְאַחַר אֵרוּסִין. גְּמָרָא: הָרַמְבַּ"ם. אֲבָל בְּחִבּוּרוֹ כָּתַב, וְיֵשׁ לוֹ לְהַחֲרִים סְתָם עַל מִי שֶׁטּוֹעֵן שֶׁקֶר כוּ'. וְזֶה עִקָּר, דְּכֵיוָן שֶׁאֵינוֹ טוֹעֵן עָלֶיהָ בְּבָרִי, אֵין כָּאן מָקוֹם לִשְׁבוּעָה אֲפִלּוּ הֶסֵּת, שֶׁלֹּא מְצָאנוּהָ אֶלָּא בְּבָרִי: גְּמָרָא. וְכָתַב הָרֹא"שׁ וְאֵין לְהַאֲמִינָהּ בְּמִגּוֹ דְנֶאֱנַסְתִּי, דְּלֹא בַעְיָא לְמִפְסָל נַפְשָׁהּ מִכְּהֻנָּה. וְאֵין לְהַאֲמִין בְּמִגּוֹ דְמֻכַּת עֵץ, דְּלֹא שְׁכִיחַ. אִי נַמִּי, לֹא אָמְרִינַן מִגּוֹ לְאַפּוּקֵי מָמוֹנָא: פֵּרֵשׁ רַשִׁ"י, וְאֵין לָחוּשׁ שֶׁמָּא יִטְעֹן אָדָם כָּךְ אֶלָּא בֶּאֱמֶת, שֶׁאִם הָיָה שׂוֹנְאָהּ מִתְּחִלָּה לֹא הָיָה טוֹרֵחַ בַּסְּעֻדָּה בְּחִנָּם, אֶלָּא הָיָה מְגָרְשָׁהּ תְּחִלָּה. וְכָתַב הָרַ"ן דְּאֵין לְהַקְשׁוֹת דְּתֵיפּוֹק לֵיהּ דְּהוֹאִיל וְאִיהוּ טוֹעֵן נַמִּי בָּרִי, מֵהֵיכָא תֵּיתֵי שֶׁתְּהֵא הִיא נֶאֱמֶנֶת לְאַפּוּקֵי מָמוֹנָא. דְּלָאו קוּשְׁיָא הִיא כְּלָל, דְּכֵיוָן דְּטַעֲמָא דִכְתֻבָּה כְּדֵי שֶׁלֹּא תְהֵא קַלָּה בְּעֵינָיו לְהוֹצִיאָהּ, אִם אַתָּה אוֹמֵר נֶאֱמָן וַעֲלָהּ דִּידָהּ רָמֵי לְבֵרוּרֵי וְאִי אֶפְשָׁר לָהּ לְהָבִיא רְאָיָה, הֲרֵי הִיא קַלָּה בְּעֵינָיו לְהוֹצִיאָהּ, הִלְכָּךְ צָרִיךְ לְטַעֲמָא דַּחֲזָקָה וְכוּ': לֹא כוּ'. דְּהָכָא נָמִי אוֹקֵי מָמוֹנָא בְּחֶזְקַת מָרֵיהּ:

פרק א - משנה ז

כִּדְפֵרֵשׁ בְּמִשְׁנָה ג' דְּאִית לָהּ מָנֶה אֲפִלּוּ לֹא הִכִּיר בָּהּ. וּמִשּׁוּם הָכִי אִצְטָרִיךְ לְמִתְנֵי פְּלוּגְתַּיְהוּ בְּמֻכַּת עֵץ אַף עַל גַּב דְּשָׁמְעִינַן בְּנֶאֱנַסְתִּי, לְאַשְׁמְעִינַן הָא דְּמָנֶה יֵשׁ לָהּ. וְנֶאֱנַסְתִּי אִצְטָרִיךְ לְאַשְׁמְעִינַן, דְּאִם הוֹדֵית מִקְחָהּ מֶקְּח טָעוּת וְאֵין לָהּ כְּלוּם. גְּמָרָא: וְהָכִי נַמִּי בְּאֵשֶׁת יִשְׂרָאֵל וְקִבֵּל בָּהּ אָבִיהָ קִדּוּשִׁין בְּפָחוֹת מִבַּת שָׁלֹשׁ, וְטַעֲמָא, דְּלֵיכָּא אֶלָּא חַד סְפֵקָא. וְכָתַב הַמַּגִּיד מִשְׁנֶה, וְסָפֵק דְּשֶׁמָּא מֻכַּת עֵץ הִיא אֵינוֹ מָצוּי וְאֵין חוֹשְׁשִׁין לוֹ. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב:

פרק א - משנה ח

וְתָנֵי לָהּ לְהוֹדִיעֲךָ כֹּחוֹ דְרַבִּי יְהוֹשֻׁעַ דְּאַף עַל גַּב דְּלֹא רָאוּהָ שֶׁנִּבְעֲלָה, פְּסוּלָה. גְּמָרָא: טִיבוֹ. טִיב הִיא סִבָּה כְּמוֹ טִיב גִּטִּין וְקִדּוּשִׁין. הָרַמְבַּ"ם: נֶאֱמֶנֶת. לְפִי שֶׁהִיא בְּחֶזְקַת כַּשְׁרוּת:

פרק א - משנה ט

מְעֻבֶּרֶת. לְהוֹדִיעֲךָ כֹּחוֹ דְרַבָּן גַּמְלִיאֵל. וְעוֹד, דַּאֲפִלּוּ בִּתָּהּ מַכְשְׁרִינַן: רוֹצֶה לוֹמַר הִיא, אֲבָל בִּתָּהּ הָא פָּסְקִינַן הֲלָכָה כְּאַבָּא שָׁאוּל בְּפֶרֶק ד' דְּקִדּוּשִׁין, דִּלְכַתְּחִלָּה הֵקֵלּוּ. וְטַעֲמָא, לְפִי שֶׁאֶפְשָׁר דַּאֲפִלּוּ לַקָּהָל פְּסוּלָה, וְאִם הָיוּ מַחְמִירִין בָּהּ שֶׁלֹּא לָבֹא בַקָּהָל לֹא הָיָה לָהּ תַּקָּנָה, הִלְכָּךְ לֹא הֶחְמִירוּ שֶׁלֹּא תִנָּשֵׂא לְכַתְּחִלָּה. הָרַ"ן וְהַמַּגִּיד. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: אֲבָל בָּעִיר, אֲפִלּוּ פֵּרַשׁ הַבּוֹעֵל וְאָזֵיל לְגַבָּהּ, לֹא אָמְרִינַן כָּל דְּפָרִישׁ כוּ', אֶלָּא חָשְׁבִינַן לֵיהּ כְּקָבוּעַ דִּכְמֶחֱצָה עַל מֶחֱצָה דָּמֵי, גְּזֵרָה אַטּוּ הֵיכָא דְאַזְלָא אִיהִי לְגַבַּיְהוּ. גְּמָרָא:

פרק א - משנה י

הָעַיִן. כְּדִכְתִיב עַל הָעַיִן בְּדֶרֶךְ שׁוּר: מַשִּׂיאִין. בְּנוֹתֵיהֶן וְאַלְמְנוֹתֵיהֶן. שֶׁאֵין רֹב בְּנֵי הָעִיר פּוֹסְלִין אִשָּׁה לַכְּהֻנָּה בְּבִיאָתָן. רַשִׁ"י: וּבְמַתְנִיתִין דִּלְעֵיל הֲלָכָה כְּרַבָּן גַּמְלִיאֵל וְלֹא פֵרֵשׁ דְּבָעִינַן תְּרֵי רֻבֵּי. מְחַלְּקִים הָרֹא"שׁ וְהָרַ"ן, דִּלְעֵיל בְּטוֹעֶנֶת וְהָכָא בְּאֵינָהּ טוֹעֶנֶת. וּסְתָמָא דְמִלְּתָא הָכִי אִיתָא, דְּתִינוֹקֶת הֲוָה וְלֹא יָדְעָה לְמִטְעָן, וְכֵיוָן דַּהֲוֵי לֵיהּ שֶׁמָּא, אֲפִלּוּ לְרַבָּן גַּמְלִיאֵל בָּעֵי תְּרֵי רֻבֵּי. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב פֵּרוּשׁ אַחֵר: רַשִׁ"י פֵּרֵשׁ, רֹב סִיעָה שֶׁל כְּשֵׁרִים הַבָּאִים מִמָּקוֹם אַחֵר עוֹבְרִים שָׁם. וְהָרַמְבַּ"ם מְפָרֵשׁ דְּרֹב סִיעָה הֵם שֶׁפָּרְשׁוּ מִן הַמָּקוֹם הַזֶּה וְעוֹבְרִים שָׁם בְּפָרָשַׁת דְּרָכִים אוֹ בִקְרָנוֹת שֶׁבַּשָּׂדוֹת. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב. פֵּרוּשׁ סִיעָה, שַׁיָּרָא. מִלְּשׁוֹן וַאֲחוּזַת מֵרֵעֵהוּ תִּרְגֵּם אוּנְקְלוּס וְסִיעַת מְרַחֲמוֹהִי:
