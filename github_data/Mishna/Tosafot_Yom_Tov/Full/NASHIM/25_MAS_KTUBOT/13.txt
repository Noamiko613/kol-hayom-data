פרק יג

פרק יג - משנה א

חִלּוּפֵי גִרְסָאוֹת יֵשׁ, אִי תָנֵי גְּזֵרוֹת אוֹ גְּזֵלוֹת: וְצָרִיךְ עִיּוּן, דְּהָכָא אָמְרִינַן דִּמְזוֹנוֹת גַּבְיָא בְּלֹא שְׁבוּעָה, מַה שֶּׁאֵין כֵּן כְּתֻבָּה, וּלְעִנְיַן מְשַׁעְבְּדֵי חָיְשִׁינַן בִּמְזוֹנוֹת טְפֵי לִצְרָרֵי מִבִּכְתֻבָּה, דִּכְתֻבָּה נִפְרַעַת בִּשְׁבוּעָה מִנְּכָסִים מְשֻׁעְבָּדִין וּמְזוֹנֵי לֹא גַבְיָא כְּלָל אֲפִלּוּ בִּשְׁבוּעָה. תּוֹסָפוֹת גִּטִּין דַּף נ"א:

פרק יג - משנה ב

אִבֵּד כוּ'. כְּמוֹ שֶׁכָּתַב הָרַ"ב בְּמִשְׁנָה ב' פֶּרֶק ד' דִּנְדָרִים, שֶׁאֵין זֶה אֶלָּא מְנִיעַת תְּבִיעָה, וְכֵיוָן שֶׁאֵין מֵבִיא לוֹ תּוֹעֶלֶת אֶלָּא מְסַלֵּק מִמֶּנּוּ נֶזֶק חִיּוּבוֹ, אֵין זֶה חַיָּב לוֹ כְּלוּם. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: קֶרֶן הַצְּבִי. שֶׁהַצְּבִי קַל לָרוּץ כְּדִתְנַן פֶּרֶק ה' דְּאָבוֹת, וְנִמְצָא שֶׁזֶּה שֶׁמַּנִּיחַ מְעוֹתָיו עַל קֶרֶן הַצְּבִי הוּא מְזֻמָּן לְהֶפְסֵד:

פרק יג - משנה ג

יוֹרְשִׁים כוּ'. פֵּרוּשׁ, הַבָּנוֹת נִזּוֹנוֹת תַּחַת יַד אַחֵיהֶן וְכָל הַנְּכָסִים עוֹמְדִים בִּרְשׁוּתָם. הַמַּגִּיד: רוֹצֶה לְרַבּוֹת גַּם כְּסוּת וּמַלְבּוּשׁ. נִמּוּקֵי יוֹסֵף: וְהִקְשׁוּ בַגְּמָרָא רֵישׁ פֶּרֶק ט' דְּבָבָא בַתְרָא, וְאִי לֵיכָּא לְכֻלְּהוּ יִקְחוּ הַבָּנוֹת אֶת הַכֹּל, דְּוַדַּאי מַה שֶּׁהוּא עוֹדֵף עַל מְזוֹן הַבָּנוֹת לֹא יִטְּלוּהוּ הַבָּנוֹת. וּמְתָרֵץ, מוֹצִיאִין מֵהֶם מְזוֹנוֹת לַבָּנוֹת עַד שֶׁיִּבָּגְרוּ, וְהַשְּׁאָר לַבָּנִים, וְאִי לֵיכָּא שְׁאָר, יִשְׁאֲלוּ עַל הַפְּתָחִים: הָכִי סָתִים לָהּ בְּרֵישׁ פֶּרֶק ט' דְּבָבָא בַתְרָא. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב:

פרק יג - משנה ד

בַּקַּנְקַנִּים. נִרְאֶה לִי לְפָרֵשׁ, דְּכַד וְקַנְקַן שְׁנֵי מִינִים הֵם, וְלֹא שֶׁטַּעֲנוֹת שְׁנֵיהֶם הָיָה זֶה טוֹעֵן כַּד וְזֶה מֵשִׁיב קַנְקַן, דְּאִם כֵּן פְּשִׁיטָא דְּלֹא הֲוֵי מִמִּין הַטַּעֲנָה, , תּוֹסְפוֹת יוֹם טוֹב. וְעַיֵּן עוֹד: כְּלוֹמַר, לַאֲפוֹקֵי אִם אָמַר עֲשָׂרָה כַדִּים מְלֵאִים, דִּלְכֻלֵּי עַלְמָא טוֹעֲנוֹ שֶׁמֶן וְכַדִּים. בַּגְּמָרָא:

פרק יג - משנה ה

לְכָל הַפֵּרוּשִׁים מַיְרֵי שֶׁלֹּא הִנִּיחַ כָּאן נְכָסִים, לְפִי דְמַיְרֵי שֶׁהַפְּסִיקָה הָיְתָה בְּעִנְיַן שֶׁקָּנָה וּמְחֻיָּב לְשַׁלֵּם, וְאִלּוּ הָיָה לוֹ נְכָסִים הָיָה הֶחָתָן מוֹצִיא מִמֶּנּוּ בְּדִין. תּוֹסְפוֹת יוֹם טוֹב:

פרק יג - משנה ו

זְכוּתוֹ. פֵּרוּשׁ, אֲפִלּוּ יֵשׁ לוֹ עֵדֵי זְכוּת אוֹ עֵדֵי גְזֵלָה וְאֵין לַחֲבֵרוֹ עֵדֵי מֶקָּח. בֵּית יוֹסֵף: עֲשָׂאָהּ כוּ'. פֵּרֵשׁ רַשִׁ"י, עֲשָׂאָהּ הַמֻּחְזָק בָּהּ סִימָן לְאַחֵר, לֹא לִרְאוּבֵן זֶה הָעוֹרֵר עָלָיו, אֶלָּא לְאִישׁ אַחֵר הָיָה מוֹכֵר שָׂדֶה בְּצַד זֶה, וּמָסַר לוֹ אַרְבָּעָה מְצָרֵי הַשָּׂדֶה וְכָתַב לוֹ מְצָרִים, מֵצַר פְּלוֹנִי שָׂדֶה שֶׁלִּי, וְזֶה חָתַם עַל הַשְּׁטָר וְלֹא עִרְעֵר, אִבֵּד אֶת זְכוּתוֹ אֲפִלּוּ לְאַדְמוֹן, דְּאֵין לוֹמַר כָּאן הַשֵּׁנִי נֹחַ לִי. וְהַתּוֹסָפוֹת מְפָרְשִׁים, עֲשָׂאָהּ הַמְעַרְעֵר סִימָן לְאַחֵר, כְּגוֹן שֶׁכָּתַב בַּמְּצָרִים מֵצַר פְּלוֹנִי שָׂדֶה שֶׁל שִׁמְעוֹן:

פרק יג - משנה ז

פֵּרֵשׁ רַשִׁ"י, וְאוֹזִיל גַּבָּךְ כְּשֶׁתִּקְנֶה הַדֶּרֶךְ מִמֶּנִּי. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב:

פרק יג - משנה ח

וּכְשֶׁהִגִּיעַ זְמַן פִּרְעוֹן הַחוֹב קֹדֶם. הָרַ"ן: דְּאַף שֶׁצָּרִיךְ זֶה לִמְכֹּר, הָיָה מוֹצֵא הַרְבֵּה, דְּקוֹנִים מְצוּיִים הֵם, דְּאָמְרֵי אֱנָשֵׁי, זָבֵין קָנִית זָבֵינְתְּ אַפְסַדְתְּ:

פרק יג - משנה ט

זֶה עַל זֶה. רְאוּבֵן עַל שִׁמְעוֹן וְשִׁמְעוֹן עַל רְאוּבֵן. וּשְׁטָר שֶׁהוֹצִיא רְאוּבֵן מֻקְדָּם וְשֶׁל שִׁמְעוֹן מְאֻחָר. אַדְמוֹן אוֹמֵר יָכוֹל לוֹמַר אִלּוּ כוּ' כֵּיצַד כוּ' הָיָה לְךָ לִתְבֹּעַ חוֹבְךָ מִמֶּנִּי. וַחֲכָמִים אוֹמְרִים זֶה גּוֹבֶה כוּ', שֶׁאַף הָרִאשׁוֹן גּוֹבֶה. וּפָרְכִינַן בַּגְּמָרָא, אִי דְמָטֵי זִמְנֵיהּ שֶׁל חוֹב הָרִאשׁוֹן מַאי טַעֲמָא דְרַבָּנָן שֶׁהָרִאשׁוֹן גּוֹבֶה חוֹבוֹ. וְאִי לֹא מָטֵי זִמְנָא מַאי טַעֲמָא דְאַדְמוֹן דְּהָא לֹא מָצֵי אָמַר לֵיהּ אִלּוּ כוּ', דְּמִשּׁוּם דְּלֹא הִגִּיעַ זְמַנּוֹ הֻצְרַךְ לִלְוֹת מִמֶּנּוּ. וּמוֹקְמִינַן לָהּ בַּגְּמָרָא כְּגוֹן שֶׁלָּוָה מִמֶּנּוּ בַּיּוֹם דְּמִשְׁלִים זִמְנֵיהּ. וּפְלִיגֵי אִי עֲבֵיד אֱינַשׁ דְּיַזִּיף לְיוֹמָא אוֹ לֹא. הָרַ"ן:

פרק יג - משנה י

שָׁלֹשׁ אֲרָצוֹת כוּ'. הֵן בְּאֶרֶץ יִשְׂרָאֵל חֲלוּקוֹת לְעִנְיַן נִשּׂוּאֵי אִשָּׁה. רַשִׁ"י: וְהוּא מֵאוֹתָהּ הָאָרֶץ שֶׁנְּשָׂאָהּ:

פרק יג - משנה יא

וְאֵין הַכֹּל כוּ'. לַאֲתוֹיֵי עֶבֶד שֶׁבָּרַח מֵחוּצָה לָאָרֶץ, דְּאָמְרִינַן לֵיהּ זַבְּנֵיהּ הָכָא וְזִיל. גְּמָרָא: אֶרֶץ יִשְׂרָאֵל. וּבִלְבָד שֶׁלֹּא יְהֵא פָּחוֹת מִתַּקָּנַת חֲכָמִים, לִבְתוּלָה מָאתַיִם וּלְאַלְמָנָה מָנֶה, שֶׁכָּל הַפּוֹחֵת כוּ' : בְּקַפּוֹטְקִיָּא. וְיִחֵד אוֹתָהּ עַל דֶּרֶךְ מָשָׁל, מִפְּנֵי שֶׁהִיא קְרוֹבָה לְאֶרֶץ יִשְׂרָאֵל. הָרַמְבַּ"ם: הִלְכָּךְ בָּתַר שִׁעְבּוּדָא אָזְלִינַן כְּדִין שִׁטְרֵי חוֹב. רַשִׁ"י. וְכָתַב עַל זֶה הָרַ"ן, וְאִי תֵימָא אִם כֵּן הֵיכָא דְנָשָׂא בְּאֶרֶץ יִשְׂרָאֵל דְּנוֹתֵן לָהּ מִמְּעוֹת אֶרֶץ יִשְׂרָאֵל, וְאִם כֵּן לְעוֹלָם יְהֵא לָהּ מָאתַיִם דְּצוֹרִי, כְּמוֹ בְאוֹנֵס וּמְפַתֶּה דְּהֵן שֶׁל צוֹרִי כוּ', אֶלָּא דְּרַבָּן שִׁמְעוֹן בֶּן גַּמְלִיאֵל סָבַר דְּאַף דִּכְתֻבָּה לְאִשָּׁה מִן הַתּוֹרָה, אֲבָל אֵיזוֹ כְתֻבָּה שֶׁיַּסְכִּימוּ בָהּ שְׁנֵיהֶן מִשְׁתַּעְבֵּד לָהּ, וּלְפִיכָךְ חָל שִׁעְבּוּדוֹ בִּמְקוֹם הַנִּשּׂוּאִין. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: נָשָׂא כוּ'. אֲתָאן לְכֻלֵּי עַלְמָא:
