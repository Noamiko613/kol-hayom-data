פרק יג

פָּטוּר. רְצוֹנוֹ לוֹמַר עַל הָעֲלִיָּה כוּ': כְּדִכְתִיב בְּהַעֲלָאָה וְאֶל פֶּתַח אֹהֶל מוֹעֵד לֹא יְבִיאֶנּוּ, בִּרְאוּיִין לַהֲבִיאָן לִפְנִים וּלְהַעֲלוֹתָם הַכָּתוּב מְדַבֵּר. רַשִׁ"י. וּבַגְּמָרָא יָלֵיף מִקְּרָא גַּם לִשְׁחוּטֵי חוּץ שֶׁהֶעֱלָן בַּחוּץ:

אָמְרוּ לוֹ כוּ'. דְּרַבָּנָן הָיוּ סְבוּרִין דְּטַעְמֵיהּ דְּרַבִּי יוֹסֵי מִשּׁוּם דִּמְחֻלָּל וְעוֹמֵד הוּא. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: מִי שֶׁטֻּמְאָה פּוֹרַחַת מִמֶּנּוּ. רַשִׁ"י:

וּבַגְּמָרָא, דְּבִשְׁחִיטָה נַמִּי כְתִיב לְהַקְרִיב קָרְבָּן לַה'. וּמִבָּעֵי לֵיהּ לְהוֹצִיא שָׂעִיר הַמִּשְׁתַּלֵּחַ: וְאִישׁ אִישׁ דִּכְתִיב גַּבֵּי שְׁחִיטָה, אָמְרִינַן בַּגְּמָרָא לְחַיְּבוֹ כְּשֶׁשָּׁחַט לְהֶדְיוֹט, אֲבָל לְרַבּוֹת לִשְׁנַיִם שֶׁאָחֲזוּ לֹא, דְּהָא כְתִיב מִעוּטָא הָאִישׁ הַהוּא. וְתֵימַהּ עַל הָרַ"ב לְעֵיל שֶׁפֵּרֵשׁ מִדָּם יֵחָשֵׁב וְגוֹ'. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: עַל כָּל עֲלִיָּה כוּ'. נֶאֱמַר בְּמַעֲלֶה חוּץ, לַעֲשׂוֹת אוֹתוֹ לַה', עַל כֵּן אָמַר רַבִּי שִׁמְעוֹן אוֹתוֹ נוֹפֵל עַל כָּל אֵבָר וְאֵבָר, וּלְפִיכָךְ הֶעֱלָה בַחוּץ חַיָּב עַל כָּל אֵבָר וְאֵבָר וַאֲפִלּוּ הָיָה חָסֵר. וְרַבִּי יוֹסֵי אוֹמֵר שֶׁהוּא נוֹפֵל עַל הָאֵבָר כֻּלּוֹ, וְעַל כֵּן אוֹמֵר עַל הַשָּׁלֵם חַיָּב וְלֹא עַל הֶחָסֵר. הָרַמְבַּ"ם: הַמִּזְבֵּחַ. שֶׁיִּבְנֶה מִזְבֵּחַ לְשֵׁם כָּךְ וְיַעֲלֶה. רַשִׁ"י: פֵּרֵשׁ רַשִׁ"י בְּהֶתֵּר הַבָּמוֹת. וּמְסַיֵּם בַּגְּמָרָא וּלְפִיכָךְ בִּשְׁעַת אִסּוּר הַבָּמוֹת הֶעֱלָה עַל הַסֶּלַע אוֹ עַל אֶבֶן חַיָּב:

קָדָשִׁים כוּ'. יָלְפִינַן לְכָל קָדָשִׁים מִדִּכְתִיב זֶבַח. וְיָלְפִינַן לְהַקֹּמֶץ וְכֻלְּהוּ, מִדִּכְתִיב וְאֶל פֶּתַח אֹהֶל מוֹעֵד לֹא יְבִיאֶנּוּ, כָּל הַבָּא לְאֹהֶל מוֹעֵד חַיָּבִים עָלָיו בַּחוּץ: כַּזַּיִת כוּ'. כֵּיוָן דִּלְרַבִּי יוֹסֵי לְעֵיל הַמַּעֲלֶה אֵבָר חָסֵר פָּטוּר, הָכָא בְּשֶׁאוֹתוֹ כַּזַּיִת הוּא אֵבָר. אִי נַמִּי, כְּשֶׁהוּא מִמָּקוֹם שֶׁאֵינוֹ אֵבָר כְּגוֹן שֶׁהוּא בָשָׂר אוֹ חֵלֶב. כֶּסֶף מִשְׁנֶה: חַיָּב. קָסָבַר כִּי כְתִיב אוֹתוֹ עַל הַשָּׁלֵם הוּא חַיָּב וְאֵינוֹ חַיָּב עַל הֶחָסֵר, אַשִּׁעוּר הַקְטָרָה כְתִיב וְהַיְנוּ כַּזַּיִת. רַשִׁ"י: אֶת כֻּלּוֹ. מִדְּלֹא פָלֵיג אַרֵישָׁא גַּבֵּי עוֹלָה, אוֹתוֹ הָכִי נַמִּי דָּרֵישׁ לֵיהּ אַשִּׁעוּר הַקְטָרָה כְרַבָּנָן. וּמִיהוּ עוֹלָה דְּלֹא מִפַּסְּלָה בְּחִסָּרוֹן, דְּכָל עַצְמָהּ בַּדָּם תְּלוּיָה כַפָּרָתָהּ, חַיָּב בְּכַזַּיִת. אֲבָל בְּהָנֵי פָלֵיג דְּקָא סָבַר קֹמֶץ כֻּלֵּיהּ הַיְנוּ שִׁעוּר הַקְטָרָתוֹ אַף בִּפְנִים דְּהָא מִפְסַל בְּחִסָּרוֹן. וְכֵן כֻּלָּם, שֶׁכֻּלָּם מַתִּירִין כוּ'. רַשִׁ"י: כָּל שֶׁהֵן. מִשִּׁעוּר הַכָּשֵׁר. דְּבִלְבוֹנָה שֶׁחָסְרָה פְּלִיגֵי בִּמְנָחוֹת פֶּרֶק קַמָּא, דִּלְרַבִּי שִׁמְעוֹן כְּשֵׁרָה אֲפִלּוּ עָמְדָה עַל קֹרֶט אֶחָד, וּלְרַבִּי יוֹסֵי בִּשְׁנֵי קְרָטִין. רַשִׁ"י:

כְּרַבִּי יוֹסֵי דְּאָמַר אֵינוֹ חַיָּב עַד שֶׁיַּעֲלֶה לְרֹאשׁ הַמִּזְבֵּחַ כְּהַקְטָרַת פְּנִים. רַשִׁ"י: וּכְלוֹמַר, דְּאַף בִּפְנִים לֹא אָמְרִינַן בִּכְהַאי גַּוְנָא לְפִי שֶׁמִּין בְּמִינוֹ כוּ'. כָּךְ נִרְאֶה לִי. וְכֵן מוּכָח מִסֵּיפָא. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: פָּטוּר. לְפִי שֶׁהַקֹּמֶץ אֵינוֹ יָדוּעַ הוּא כְּמוֹ הָאֵמוּרִים. הֲלֹא תִרְאֶה כִּי כְּשֶׁנִּקְמָץ וְחָזַר לְתוֹכָהּ הֲרֵי דוֹמֶה לְאֵמוּרִים וְחַיָּב. הָרַמְבַּ"ם:

הַזּוֹרֵק כוּ'. נַפְקָא לֵיהּ בַּגְּמָרָא מִקְּרָא: חַיָּב. וְלֹא פָלֵיג רַבִּי אֱלִיעֶזֶר, דַּאֲפִלּוּ בְּחַטָּאוֹת הַפְּנִימִיּוֹת אִם נִשְׁפַּךְ הַדָּם אַחַר שֶׁנָּתַן מַתָּנָה אַחַת, מְהַנְּיָא הַךְ דְּיָהֲבָא, וּמֵבִיא פַּר אַחֵר וּמַתְחִיל. רַשִׁ"י: וְכֵן פָּסַק הָרַמְבַּ"ם. וּתְמֵהַנִי, דְּמַה לִּי מְפֹרָשׁ מִדְּאוֹרַיְתָא מַה לִּי הֲלָכָה לְמֹשֶׁה מִסִּינַי, שֶׁהֲרֵי שְׁתַּיִם זוּ שָׁמַעְנוּ מִפִּי הַגְּבוּרָה וְלָמָּה לֹא יִהְיֶה חַיָּב עַל מַה שֶּׁהוּא הֲלָכָה לְמֹשֶׁה מִסִּינַי כְּמוֹ עַל מַה שֶּׁהוּא מִדְּאוֹרַיְתָא. וְתוּ, דְּבַגְּמָרָא מַסִּיק דְּטַעְמֵיהּ דְּרַבִּי אֶלְעָזָר דַּהֲלָכָה לְמֹשֶׁה מִסִּינַי הִיא. וְהַשְׁתָּא קָשֶׁה לִי דִּבְמִשְׁנָה ט' פֶּרֶק ד' דְּסֻכָּה כָּתַבְתִּי דְּמַסְקָנָא דִגְמָרָא דַּהֲלָכָה לְמֹשֶׁה מִסִּינַי הִיא, וְאִם כֵּן אַמַּאי אֵין הֲלָכָה כְּרַבִּי אֶלְעָזָר כוּ'. וּלְפִי זֶה יִרְאֶה לִי דִּבְשִׁעוּרָא פָלֵיג רַבִּי אֶלְעָזָר דִּסְבִירָא לֵיהּ דְּאֵין צָרִיךְ שִׁעוּר כְּלָל, וְכִדְאִיתָא בַגְּמָרָא, דְּמֵי הֶחָג מַשְׁמַע אֲפִלּוּ פָּחוֹת מִשְּׁלֹשָׁה לֻגִּין. וּלְפִי דְקַיְמָא לָן כְּמַאן דְּאָמַר בְּסֻכָּה דְּבָעִינַן שְׁלֹשָׁה לֻגִּין, לְכָךְ פָּסַק הָרַמְבַּ"ם דְּלֹא כְרַבִּי אֶלְעָזָר דְּהָכָא. וְכֵן פָּסַק בְּחִבּוּרוֹ, הַמְנַסֵּךְ שְׁלֹשָׁה לֻגִּין חַיָּב, פָּחוֹת מִכֵּן פָּטוּר. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: כְּלוֹמַר דְּלֹא כִסְתָמָא דְמַתְנִיתִין רֵישׁ פֶּרֶק ה':

כְּמוֹ שֶׁכָּתַבְתִּי בְּרֵישׁ פִּרְקִין. וּשְׁחִיטָה בַחוּץ בְּעוֹף פֵּרֵשׁ רַשִׁ"י דְרַחֲמָנָא רַבְּיֵהּ מֵאוֹ אֲשֶׁר יִשָּׁחֵט. וְאַהַעֲלָאָתוֹ נַמִּי אָמְרִינַן וַאֲלֵיהֶם תֹּאמַר, לְעָרֵב פָּרָשִׁיּוֹת, דְּכָל שֶׁמִּתְחַיֵּב עַל שְׁחִיטָתוֹ בַחוּץ אִם חָזַר כוּ': וַאֲפִלּוּ עָלְתָה תֵּרֵד. וְרַבִּי יְהוּדָה הוּא דְאָמַר הָכִי בְּפֶרֶק ט' מִשְׁנָה ב'. רַשִׁ"י: דְּבִשְׁחִיטַת חוּץ לֹא שְׁנָא לַיְלָה מִיּוֹם, דְּהָא רָאוּי הוּא לָבֹא אֶל פֶּתַח אֹהֶל מוֹעֵד לְמָחָר. וּבְהַעֲלָאָתוֹ נַמִּי חַיָּב דְּהָא כְתִיב וַאֲלֵיהֶם תֹּאמַר. רַשִׁ"י: דְּהָא בִפְנִים אִם עָלְתָה לֹא תֵרֵד אִית לֵיהּ בְּפֶרֶק ט' שָׁם. רַשִׁ"י: וְאִם תֹּאמַר מַאי שְׁנָא מִשּׁוֹחֵט בְּהֵמָה בַלַּיְלָה בִּפְנִים. וְיֵשׁ לוֹמַר דְּשׁוֹחֵט עוֹף בִּפְנִים מִקְטַל קַטְלֵיהּ, וְאִם עָלָה יֵרֵד אֲפִלּוּ לְרַבִּי שִׁמְעוֹן, אַף עַל גַּב דְּלֹא מְטַמֵּא בְגָדִים אַבֵּית אַבְּלִיעָה כְּדִלְעֵיל סוֹף פֶּרֶק ז'. תּוֹסָפוֹת:

וְאִם כֵּן מַתְנִיתִין דַּוְקָא בְדָמִים הַפְּנִימִים, כִּדְפֵרֵשׁ הָרַ"ב בְּמִשְׁנָה ו': לְמַה. בַּגְּמָרָא מְפָרֵשׁ דְּאַגָּמְרִינַן בְּמַאי דִמְדַמִּינְהוּ, דְּמַתְנִיתִין רַבִּי הִיא דְאָמַר בְּסוֹף פֶּרֶק ד' דִּתְמוּרָה, אֲבוּדָה בִּשְׁעַת הַפְרָשָׁה מֵתָה. וְהָכִי קָאָמַר, טַעֲמָא דְאָבְדָה דְּדָמְיָא לְכוֹס דָּחוּי, הָא הִפְרִישׁ שְׁתֵּי חַטָּאוֹת לְאַחְרָיוּת, כִּי מִתְכַּפֵּר בְּאַחַת הַשְּׁנִיָּה תִרְעֶה, וְאִם שְׁחָטָהּ סְתָם כָּשֵׁר לְעוֹלָה. וּמַשְׁכַּחַת לָהּ בְּחַטָּאת שְׂעִיר נָשִׂיא שֶׁהוּא זָכָר עוֹלָה: וְהָכִי נַמִּי כְּשֶׁיֵּשׁ יְדִיעָה בֵּינְתַיִם: כְּשֵׁם כוּ'. וְאַשָּׁחַט שְׁנֵיהֶם בִּפְנִים קָאֵי כוּ'. אֲבָל מִשּׁוּם שְׁחוּטֵי חוּץ לֹא אִצְטְרִיכָא לֵיהּ, דִּכְמַאן דְּקַטְלִינְהוּ דָמֵי וְלֵית בְּהוּ מְעִילָה, כִּדְאָמְרִינַן קָדָשִׁים שֶׁמֵּתוּ יָצְאוּ מִידֵי מְעִילָה:
