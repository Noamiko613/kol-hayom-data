פרק ו - משנה א

שֶׁשְּׁחָטָן. מִדְּאוֹרַיְתָא שׁוֹחֵט לְכַתְּחִלָּה. כִּדְדָרְשִׁינַן מִוְּזָבַחְתָּ עָלָיו. וְשֶׁמָּא הָא דְנָקַט דִּיעֲבַד, מִדְּרַבָּנָן, שֶׁלֹּא תַרְבִּיץ גְּלָלִים. תּוֹסָפוֹת: וְהָא דְּדָרְשִׁינַן עָלָיו שֶׁיְּהֵא שָׁלֵם, הָכָא דָיֵק מִשּׁוּם דִּוְזָבַחְתָּ עָלָיו מַשְׁמַע שֶׁמִּצְוָה לִזְבֹּחַ עָלָיו. תּוֹסָפוֹת:

פרק ו - משנה ב

כְּלוֹמַר דְּדַלֵּי דַּלּוּת. וְהַיְנוּ דְכָתַב וּמַשְׁמַע נַמִּי וְכוּ', דְּכֻלְּהוּ חֲדָא קְרָא וַחֲדָא דְרָשָׁא: תֵּימַהּ דְּלֹא יָלְפִינַן נַמִּי לִמְלִיקָה. וְשֶׁמָּא לֹא יָלְפִינַן אֶלָּא מִידִי דְּאַשְׁכְּחַן דְּבָעֵי מִזְבֵּחַ כְּמִנְחָה. תּוֹסָפוֹת: וְהָרַמְבַּ"ם כָּתַב וּשְׁיָרֵי הַדָּם שֶׁל עוֹלוֹת וְחַטָּאוֹת הַנֶּאֱכָלוֹת וַאֲשָׁמוֹת וּשְׁלָמִים שֶׁשּׁוֹפְכִים עָלֶיהָ:

פרק ו - משנה ג

וְלִשְׁלֹשָׁה דְבָרִים הַנַּעֲשִׂים בָּהּ לְמַטָּה אֵינוֹ עוֹלֶה וְאֵינוֹ מַקִּיף, אֶלָּא עוֹמֵד עַל הָרִצְפָּה בַּקַּרְקַע. רַשִׁ"י: וּלְפִי זֶה כֶּבֶשׁ קָטָן הָיָה גַם כֵּן לִשְׂמֹאל. רַשִׁ"י. וְכֵן כָּתְבוּ הַתּוֹסָפוֹת. וּכְמוֹ שֶׁהָיָה כֶּבֶשׁ קָטָן שֶׁפָּנָה אֶל הַסּוֹבֵב לְמִזְרַח הַכֶּבֶשׁ. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: וְחוֹזְרִים כוּ'. בַּדֶּרֶךְ שֶׁעָלוּ בוֹ. וְאֵין מַקִּיפִין, דְּכָל כַּמָּה דְּמָצִינַן לְמַעוֹטֵי בַּהִלּוּךְ שֶׁל שְׂמֹאל מְמַעֲטִין. רַשִׁ"י. וְתוֹסָפוֹת כָּתְבוּ דְּטַעֲמָא דְהָכָא חֲזָרָתוֹ עַל הֶעָקֵב הָוֵי דֶרֶךְ יָמִין:

פרק ו - משנה ד

חַטַּאת כוּ'. מִשּׁוּם דְּדַם חַטָּאת קוֹדֵם לְעוֹלָה, מְפָרֵשׁ חַטָּאת תְּחִלָּה. תּוֹסָפוֹת: וְאֵין כֵן לְשׁוֹן הַבְּרַיְתָא בְּסוֹף פִּרְקִין. אֶלָּא חוֹתֵךְ שִׁדְרָה וּמִפְרֶקֶת בְּלֹא רֹב בָּשָׂר עַד שֶׁיַּגִּיעַ לַוֶּשֶׁט אוֹ לַקָּנֶה חוֹתֵךְ סִימָן אֶחָד אוֹ רֻבּוֹ וְרֹב בָּשָׂר עִמּוֹ. וּפֵרֵשׁ רַשִׁ"י שִׁדְרָה, חוּט הַשִּׁדְרָה. מִפְרֶקֶת, עֶצֶם הַצַּוָּאר. בְּלֹא רֹב בָּשָׂר הַמַּקֶּפֶת, דְּאִי לָאו הָכִי הַוְיָא נְבֵלָה. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: יָלְפִינַן לַהּ בַּגְּמָרָא מִקְּרָא: לְשׁוֹן רַשִׁ"י, הַזָּאָה אוֹחֵז בְּגוּף הָעוֹף וּמַעֲלֶה וּמוֹרִיד בְּכֹחַ וְהַדָּם נִתָּז מֵעַצְמוֹ: וּבִמְנָחוֹת פֵּרֵשׁ רַשִׁ"י, מִצּוּי, סוֹחֵט בֵּין אֶצְבְּעוֹתָיו וְנִצּוֹק הַדָּם הוֹלֵךְ לְמֵרָחוֹק: לֹא יָדַעְנָא לְמַאי אִצְטְרִיךְ לְהַאי, דִּקְרָא גוּפֵיהּ הָא קָאָמַר יִמָּצֶה אֶל יְסוֹד הַמִּזְבֵּחַ. אֲבָל בַּגְּמָרָא מַיְתֵי הַבְּרַיְתָא עַל הַהַזָּאָה שֶׁתְּהֵא לְמַטָּה וְלֹא לְמַעְלָה. וְכֵן כָּתַב הָרַ"ב בְּעַצְמוֹ רֵישׁ קִנִּים. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב:

פרק ו - משנה ה

וְלֹא סַגִּי לָן בְּהָכִי, דְּהָא גְזֵרָה שָׁוָה אִית לָן לְמוּל עֹרֶף, דְּנֵילַף נַמִּי לַיְסוֹד. אֶלָּא כִּדְאִיתָא בַּבְּרַיְתָא, דִּכְתִיב וּמָלַק וְהִקְטִיר, מָה הַקְטָרָה בְּרֹאשׁ הַמִּזְבֵּחַ, אַף מְלִיקָה בְּרֹאשׁ הַמִּזְבֵּחַ. וְעוֹד נֶאֱמַר וְהִקְטִיר וְנִמְצָה דָמוֹ, וְכִי תַעֲלֶה עַל דַּעְתְּךָ לְאַחַר שֶׁהִקְטִיר חוֹזֵר וּמְמַצֶּה, אֶלָּא לוֹמַר לָךְ מַה הַקְטָרָה בְּרֹאשׁ הַמִּזְבֵּחַ אַף מִצּוּי בְּרֹאשׁ הַמִּזְבֵּחַ: וְזֶה לְשׁוֹן רַשִׁ"י, עָרְפָּהּ, גֹּבַהּ רֹאשָׁהּ. מוּל עָרְפָּהּ, לְמַטָּה לְצַד הָאֲגַפִּים מוּל הָרוֹאֶה אֶת הָעֹרֶף. עַד כָּאן. וְצָרִיךְ עִיּוּן, דַּהֲוָה לֵיהּ לְפָרֵשׁ בַּמִּשְׁנָה הַקּוֹדֶמֶת: בִּגְזֵרָה שָׁוָה וּמָלַק וּמָלַק. וְכָתְבוּ הַתּוֹסָפוֹת, דְּלֵיכָּא לְמֵילַף מְלִיקָה דְחַטָּאת דִּתְהֵא גַם כֵּן בְּרֹאשׁ הַמִּזְבֵּחַ, דְּכֵיוָן דִּמְלִיקָה דְעוֹלָה לֹא יָלְפִינַן אֶלָּא בְהֶקֵּשׁ אֵינוֹ חוֹזֵר וּמְלַמֵּד בִּגְזֵרָה שָׁוָה: הָרַמְבַּ"ם. וְלֹא סַגִּי לָן כוּ'. וְנֵילַף נַמִּי בִּגְזֵרָה שָׁוָה לְלֹא יַבְדִּיל. אֶלָּא כִדְאִיתָא בַּבְּרַיְתָא דְּתַלְמוּד לוֹמַר וּמָלַק וְהִקְטִיר, מַה הַקְטָרָה הָרֹאשׁ לְעַצְמוֹ וְהַגּוּף לְעַצְמוֹ, אַף מְלִיקָה כוּ'. וּמִנַּיִן שֶׁהַקְטָרַת הָרֹאשׁ לְעַצְמוֹ וְהַגּוּף כוּ', שֶׁנֶּאֱמַר וְהִקְטִיר אוֹתוֹ, הֲרֵי הַקְטָרַת הַגּוּף אֲמוּרָה, הָא מָה אֲנִי מְקַיֵּם וְהִקְטִיר הַמִּזְבֵּחַ כוּ', בְּהַקְטָרַת הָרֹאשׁ הַכָּתוּב מְדַבֵּר: וּמְמַצֶּה. דְּלֹא כְתִיב אֶלָּא מִצּוּי בִּלְבַד: מִלְּשׁוֹן אֵין מַקִּיפִים שְׁתֵּי חָבִיוֹת: בְּמֶלַח. כְּדִכְתִיב וְכָל קָרְבַּן מִנְחָתְךָ בַּמֶּלַח תִּמְלָח. רַשִׁ"י:

פרק ו - משנה ו

שֶׁעִקַּר הַדָּם הַכָּבֵד וְהַלֵּב וְהַגִּידִים הַיּוֹצְאִים מִשְּׁנֵיהֶם. הָרַמְבַּ"ם. וְתָנוּ רַבָּנָן עוֹלָה אַף עַל פִּי שֶׁמִּצָּה דַם הַגּוּף וְלֹא מִצָּה דַם הָרֹאשׁ. יָכוֹל מִצָּה דַם הָרֹאשׁ וְלֹא דַם הַגּוּף, תַּלְמוּד לוֹמַר הִיא:

פרק ו - משנה ז

כֵּיצַד. כָּל הָנָךְ מִשְׁנָיוֹת דְּהָכָא וּדְפֶרֶק ב' דִּמְכִילְתִין וּדְפֶרֶק קַמָּא דִמְנָחוֹת, צְרִיכֵי לְדַקְדֵּק אַמַּאי צְרִיכֵי לְכֻלְּהוּ. תּוֹסָפוֹת:
