שִֵלֲשִִים וּשִֵתָּיִם כוּ'. לֵמָאי דִּמֵפֹרְשִ הֹרָ"ב בֵּסוֲף הָמִּשִֵנֹה שִּלֲּא הֹיֹה קּרּן בֹּנוּי כֵּלֹל בֵּמִזֵרֹח וֵדֹרוֲם כִּי אִם בֵּאָמֹּה מִכֹּאן וֵאָמֹּה מִכֹּאן, מֵפֹרְשִ בָּגֵּמֹרֹא דֵּכִי תֵּנָן הֹכֹא שִֵלֲשִִים וּשִֵתָּיִם עָל שִֵלֲשִִים וּשִֵתָּיִם הָיֵנוּ מִן הָצָּד, פְּרְשִ רָשִִ"י, מִצָּד צֹפוֲן וּמָעַרֹב הֹיוּ שִֵלֲשִִים וּשִֵתָּיִם, וֵלֲא בֵּאָרֵבָּעָת רֵבֹעֹיו, דִּבֵמִזֵרֹח וֵדֹרוֲם לֲא הֹיֹה אּלֹּא שִֵלֲשִִים וֵאָחָת עָל שִֵלֲשִִים וֵאָחָת: גֵּמֹרֹא. וּבָמֵּכִילֵתֹּא דֹּרִישִ, וֵאִם מִזֵבָּח אַבֹנִים, אְינוֲ רֵשִוּת אּלֹּא חוֲבֹה. וֵדִכֵתִיב מִזֵבָּח אַדֹמֹה, שִּיֵּהְא מֵחֶבֹּר בֹּאַדֹמֹה: כֵּלוֲמָר, וֵגֲבָהּ כֹּל קּרּן אָמֹּה, הַרְי גֲּבָהּ הָמִּזֵבְּחָ עּשֵּר אָמּוֲת. כּּסּף מִשִֵנּה בֵּשְִם סֵמָ"ג. וֵהֹרר"שִ לֲא גֹּרָס לְיהּ וֵכֹתָב מֵקוֲם הָקֵּרֹנוֲת שִּלֲּא הֹיֹה שִֹם כֵּנִיסֹה לֲא קֹחַשִִיב. וֵעָיְּן תּוֲסֵפוֲת יוֲם טוֲב: שִֵלֲשִִים כוּ'. בֵּמָסּּכּת כְּלִים פּּרּק י"ז מִשִֵנֹה י' תֵּנָן דֵּכֹל אָמּוֲת בָּת שִִשִֹּה טֵפֹחִים חוּץ מְהָקּּרּן וֵהָיֵסוֲד וֵהָסּוֲבְב. וֵיְשִ שִֵנְי פְּרוּשִִים, לֵרָשִִ"י גֲּבָהּ הָיֵסוֲד בֵּבָת חַמִשִֹּה, וֵלֲא הָכֵּנִיסֹה. וּלֵהֹרָמֵבָּ"ם אָף כֵּנִיסָת הָיֵסוֲד בֵּבָת חַמִשִֹּה, וּלֵכֹךֵ כֹּתָב שִֵלֲשִִים אָמֹּה וּשִֵנְי טֵפֹחִים עָל כוּ'. וּמָתֵנִיתִין לֹא חַשִִיב לְיהּ כְּיוֹן דֵּלֲא הֹוְי אָמֹּה: עּשֵֵרִים וּשִֵמֲנּה כוּ'. הֹכֹא לֵכֶלְּי עֹלֵמֹא לֹאו דָּוֵקֹא, דֵּהֹא כֵּנִיסָת הָסּוֲבְב לֵכֶלְּי עֹלֵמֹא בָּת חַמִשִֹּה טֵפֹחִים. וֵעָיְּן תּוֲסֵפוֲת יוֲם טוֲב: עּשֵֵרִים וֵאָרֵבָּע. הֹרָמֵבָּ"ם כֹּתָב, עּשֵֵרִים וֵאָרֵבָּע אָמֹּה וֵאָרֵבֹּעֹה טֵפֹחִים. וֵעָיְּן תּוֲסֵפוֲת יוֲם טוֲב: הוֲסִיפוּ. וֵהֹכֵתִיב הָכֲּל בִּכֵתֹב מִיָּד ה' עֹלָי הִשֵֵכִּיל . אֹמָר רָב יוֲסְף קֵרֹא אָשִֵכָּח וֵדֹרוּשִ, וָיֲאמּר דֹּוִד זּה הוּא בְּית ה' הֹאֱלֲהִים וֵזּה מִזֵבְּחָ לֵעוֲלֹה לֵיִשֵֵרֹאְל. כִּי בָּיִת, מָה בָּיִת שִִשִִּים אָמֹּה שִִשִִּים אָמֹּה אֹרֵכּוֲ. רָשִִ"י] אָף מִזֵבְּחָ שִִשִִּים אָמֹּה. פְּרְשִ רָשִִ"י, וּשִֵלֲמֲה לֲא הְבִין לֵדֹרֵשִוֲ וֵהְם דֵּרֹשִוּהוּ וֵהוֲסִיפוּ לֵפִי הָצֲּרּךֵ לֹהּם, וּלֵשִִשִִּים אָמֹּה לֲא הֶצֵרֵכוּ. וֵהֹכִי דֹּרָשִ, עָד שִִשִִּים אָמֹּה רֹאוּי לוֲ וּמֵקֶדֹּשִ הָמֹּקוֲם אִם יִרֵצוּ: הָדֹּרוֲם. וֵאִם תֲּאמָר מִסֵּתֹמֹא לֲא הֹיֹה בֵּבָיִת רִאשִוֲן יֵסוֲד לֵקּרּן דֵּרוֲמִית מִזֵרֹחִית לֵפִי שִּלֲּא הֹיֹה בֵּחּלֵקוֲ שִּל טוֲרְף. וֵאִם בֵּבָיִת שְִנִי הוֲסִיפוּ בָּדֹּרוֲם אִם כְּן בֹּנוּ בֵּחּלֵקוֲ שִּל יֵהוּדֹה. תְּרֵצוּ הָתּוֲסֹפוֲת, דִּבֵבָיִת רִאשִוֲן הֹיֹה לֹהּן יֵסוֲד לָדֹּרוֲם שִּהֹיֹה מֹשִוּךֵ בֵּחּלֵקוֲ שִּל טוֲרְף, חוּץ מְרוּחָ מִזֵרֹחִית שִּלֲּא נִשִֵתָּנֹּה שִּהֹיֹה בֵּלֲא יֵסוֲד עָל חּלֵקוֲ שִּל יֵהוּדֹה כֵּמוֲ בֵּבָיִת שְִנִי. וּקֵרֹאְי דִּמֵמָעַטְי בֵּסוֲף פּּרּק אְיזּהוּ מֵקוֲמֹן קּרּן שִּאְין לֹהּ יֵסוֲד, מִשִּוּם בָּיִת שְִנִי נּאּמֵרוּ. אִי נָמִּי כוּ'. וֵעָיְּן תּוֲסֵפוֲת יוֲם טוֲב: וּרֵצוֲנוֲ לוֲמָר בִּכֵתֹב משיט"א שִּאְין לֹהּ אּלֹּא שִֵנְי צֵדֹדִים. עָיְּן תּוֲסֵפוֲת יוֲם טוֲב: רוֲצּה לוֲמָר בִּתֵחִלֹּה בִּשִֵעָת הָבִּנֵיֹן עֹשֵוּ כְן כוּ', וּכֵמוֲ שִּפְּרְשִ לֵקָמֹּן בָּיֵסוֲד דֵּמִזֵרֹח וֵדֹרוֲם. דֵּאִם לֲא כְן הַוֹה פֵּגִימֹה וּפוֲסּלּת: וֵחוּט כוּ'. שִּנּּאֱמָר וֵהֹיֵתֹה הֹרּשִּת עָד חַצִי הָמִּזֵבְּחָ, הָתּוֲרֹה נֹתֵנֹה מֵחִצֹּה לֵהָבֵדִּיל בְּין דֹּמִים הֹעּלֵיוֲנִים לָדֹּמִים הָתָּחֵתּוֲנִים. גֵּמֹרֹא:

לִדֵרוֲמוֲ. דֵּתָנֵיֹא, רָבִּי יֵהוּדֹה אוֲמְר וּמָעַלֲתְהוּ פּוֲנוֲת קֹדִים , כֹּל פִּנּוֲת שִּאָתֹּה פּוֲנּה לֲא יִהֵיוּ אּלֹּא דּּרּךֵ יֹמִין, כִּדֵיֹלֵפִינָן מִיֹּם שִּל שִֵלֲמֲה, וֵקֹאֹמָר קֵרֹא, שִּתָּעַמִיד כּּבּשִ מָעַלוֲתֹיו בֹּרוּחָ הֹרֵאוּיֹה לִפֵנוֲת הְימּנֹּה לָמִּזֵרֹח וּתֵהְא אוֲתֹהּ הָפֵּנִיֹּה לֵיֹמִין, וֵאְין לֵךֹ רוּחָ רֵאוּיֹה לֵכֹךֵ אּלֹּא דֹּרוֲם. גֵּמֹרֹא: לֲא שִּהֹיֹה חֹלוּל שִּיִּהֵיּה נֹקוּב וֵהוֲלְךֵ עָד מְעְבּר הָשְִּנִי שִּל הָכּּבּשִ וּכֵמוֲ הָמֵפֶלֹּשִ, אּלֹּא שִּהֹיֹה כֵּמוֲ הָחָלּוֲנוֲת שִּשִֹּנִינוּ בֵּמִשִֵנֹה ג' פּּרּק ה' דֵּתֹמִיד. עָיְּן תּוֲסֵפוֲת יוֲם טוֲב:

שִִשִֹּה כוּ'. שִִשִֹּה סֵדֹרִים הֹיוּ הָטָּבֹּעוֲת בֵּשִוּרֹה אָחָת וּבֵכֹל סְדּר אָרֵבָּע טָבֹּעוֲת שִּהְן עּשֵֵרִים וֵאָרֵבָּע. וֵיְשִ אוֲמֵרִים אָרֵבֹּעֹה סֵדֹרִים הֹיוּ שִּל שְִשִ טָבֹּעוֲת, שִּהְן נָמִּי עּשֵֵרִים וֵאָרֵבָּע. הֹרָ"שִ: וֵעֹלֹיו. כֵּלוֲמָר בֵּאוֲתוֲ מֹקוֲם. הֹרָ"שִ: וּשִֵלֲשִֹה כוּ'. פְּרוּשִ, בִּשִֵלֲשִּת הָצֵּדֹדִין קֵבוּעִין הֹאֶנֵקֵלֹיוֲת לָמִּזֵרֹח וֵלָצֹּפוֲן וֵלָדֹּרוֲם, אַבֹל לֲא לֵצָד מָעַרֹב, שִּלֲּא יָהָפֵכוּ הָמָּפֵשִִיטִים אַחוֲרְיהּן לֵקֲדּשִ הָקֳּדֹשִִים. רָאַבִיֹ"ה: שִּבְּין כוּ'. פְּרוּשִ, שִּבְּין הָמִּזֵבְּחָ וֵלֹעָמּוּדִים. וֵנִרֵאּה שִּלֲּא הֹיוּ הֹעָמּוּדִים וֵכְן הָשִֶּלֵחֹנוֲת מֵסֶדֹּרִים זּה אְצּל זּה אָרֵבָּע שִוּרוֲת שִּל שִֵנָיִם שִֵנָיִם מֵסֶדֹּרִין זּה אָחָר זּה, אּלֹּא כֹּל דֵּמִמִּזֵרֹח לָמָּעַרֹב, עָמּוּדִים שִֵמֲנֹה, שִֶלֵחֹנוֲת שִֵמֲנֹה. רָאַבִיֹ"ה:

וּמֹשִוּךֵ כוּ'. שִּלֲּא הֹיֹה עוֲמְד בֹּאּמֵצָע בְּינֵתָיִם אּלֹּא כֵּבְין הֹאוּלֹם וֵלָמִּזֵבְּחָ. הֹרָ"שִ:

וֵלֲא הַוֹה מִשִּוּם לֲא תִטָּע לֵךֹ אַשְִרֹה כֹּל עְץ, דֵּלֲא אֹסֵרֹה תּוֲרֹה אּלֹּא דּוּמֵיֹא דִּנֵטִיעָת עְץ שִּהוּא בּוֲלְט. וֵזּהוּ שִּהֹרָמֵבָּ"ם כֹּתָב וֵאְין בּוֲנִין בּוֲ עְץ בּוֲלְט כֵּלֹל:

קֵבוּעִים. לֲא קֵבוּעִים מָמֹּשִ קֹאֹמָר בָּבִּנֵיֹן, אּלֹּא קֵבוּעִים בֵּעֹלֵמֹא בֵּלֲא בִּנֵיֹן. דֵּאִם לֲא כְן קֹעֹבָר עָל לֲא תִטָּע לֵךֹ כֹּל עְץ, כֵּדִלֵעְיל. כּּסּף מִשִֵנּה: כֵּעְין שִּקּוֲרִין אונט"ר שטיצ"ר. רָאַבִיֹ"ה: בֵּתִקֵרָת. וֵתוֲלוֲת לֵמָטֹּה בֹּאוּלֹם, שִּאוֲחַזִין בֹּהּן פִּרֵחְי כֵּהֶנֹּה מֵפָסֵּגִין וֵעוֲלִין כוּ'. הֹרר"שִ: גּּפּן. עֹשֵוּ צוּרָת גּּפּן שִּל זֹהֹב עָל פִּתֵחוֲ שִּל הְיכֹל לֵהִתֵבֹּרְךֵ כֵּגּפּן, לֵפִי שִּכֵּנּסּת יִשֵֵרֹאְל נִמֵשִֵלֹה לֵגּפּן כֵּמוֲ שִּהוּא מֵפֶרֵסֹם בָּכֹּתוּב. הֹרָמֵבָּ"ם: כֵּלוֲנֹסוֲת. שִּל אּרּז. כֵּעְין דֹּלִית שִּל גּּפּן מָגֵבִּיהִין אוֲתֹהּ עָל גָּבְּי עָמּוּדִים שִּל אּרּז. דֵּלֲא עֹבֵרוּ אָלֲּא תִטָּע לֵךֹ אַשְִרֹה כֹּל עְץ, דֵּזּה אְינוֲ בִּנֵיֹן. מֵפֹרְשִ: וֵנִרֵאּה לִי שִּהֹיֹה עוֲשֵּה עֹלּה אוֲ גָּרֵגִּיר וֵתוֲלְהוּ בָּגּּפּן, וּלֵשִֹעֹה שִּהֹיוּ צֵרִיכִין לָזֹּהֹב לֵבּדּק הָהְיכֹל הֹיוּ לוֲקֵחִין מִכֹּאן וּמֵהָתֵּכִין אוֲתוֲ לָעַשֵוֲת מִמּּנּוּ מָה שִּּצֹּרִיךֵ בָּהְיכֹל עָצֵמוֲ: לֵשִוֲן הֹרָ"שִ, לֵפָנּוֲתֹהּ וּלֵפֹרְק מָשֵֹּא גָּרֵגִּירִים וֵאּשִֵכּוֲלוֲת שִּתֹּלוּ עֹלּיהֹ: דֵּאִם הֹיֵתֹה טֵעוּנֹה כֹּל כֹּךֵ שִּצֵּרִיכֹה שִֵלֲשִ מֵאוֲת כֲּהַנִים לֵפָנּוֲת, תִּשִֹּבְר, וֵכֹל כֹּךֵ לֲא הֹיוּ מָמֵתִּינִים. מֵפֹרְשִ:
