פרק א - משנה א

הָרַמְבַּ"ם. וְאֵין נִרְאִין דִּבְרֵי הַמְפָרֵשׁ דְּמַסֶּכֶת תָּמִיד שֶׁכָּתַב שֶׁיִּהְיוּ שׁוֹמְרִים יוֹם וְלַיְלָה: אַבְטִינָס. עַיֵּן מַה שֶּׁכָּתַבְתִּי בְּסוֹף מְכִילְתִּין עַל לִשְׁכַּת הָעֵץ, וְתִמְצָא מְקוֹם לִשְׁכָּה זוֹ: פֶּרֶק א' פָּסוּק כ"ו. וּבְסוֹף פִּרְקִין כָּתַב הָרַ"ב עוֹד, שֶׁגַּם בַּמְּקוֹמוֹת שֶׁשָּׁמְרוּ הַכֹּהֲנִים הָיוּ הַלְוִיִּם גַּם כֵּן מְשַׁמְּרִים: פִּנּוֹתָיו. אנגל"ש. לַמָּקוֹם שֶׁכָּלָה הַחוֹמָה וּמַתְחֶלֶת חוֹמָה. מְפָרֵשׁ: דִּתְרֵי מִנַּיְהוּ לֹא צְרִיכֵי שִׁמּוּר. גְּמָרָא. לְפִי שֶׁהָיוּ תְּכוּפִין לִשְׁאָר הַשְּׁעָרִים. הָרַ"ר שְׁמַעְיָה: וּמִבַּחוּץ דְּקָתָנֵי כְּלוֹמַר וְאֵלּוּ וָאֵלּוּ מִבַּחוּץ. הָרַ"שׁ: הַכַּפֹּרֶת. בֵּית קָדְשֵׁי קָדָשִׁים נִקְרָא בֵּית הַכַּפֹּרֶת. עָרוּךְ. וְכֵן בַּכָּתוּב . וְעַל שֵׁם שֶׁשָּׁם הַכַּפֹּרֶת בַּכְּרוּבִים לְתַבְנִית הַמֶּרְכָּבָה:

פרק א - משנה ב

מְחַזֵּר. בְּכָל לַיְלָה וְלַיְלָה עַל כָּל מִשְׁמָר וּמִשְׁמָר. וְהָיָה לוֹ מַפְתְּחוֹת הַר הַבַּיִת. מְפָרֵשׁ. וּרְצוֹנוֹ לוֹמַר לַאֲפוֹקֵי מִשְׁמְרוֹת הַכֹּהֲנִים, וּשְׁמוֹ יוֹכִיחַ. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: וְהֵם כוּ'. בִּירוּשָׁלַיִם. הָרַמְבַּ"ם:

פרק א - משנה ג

הַדָּרוֹם. יִתָּכֵן לוֹמַר שֶׁעַל שֵׁם חֻלְדָּה הַנְּבִיאָה נִקְרָאִים כֵּן. וְהָיוּ כָּאן שְׁנַיִם לְפִי שֶׁרֹב תַּשְׁמִישׁ הַר הַבַּיִת בַּדָּרוֹם כִּבְרֵישׁ פֶּרֶק דִּלְקַמָּן. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: קִיפוֹנוֹס. לֹא יָדַעְתִּי לוֹ טַעַם. וּבְשִׁלְטֵי גִּבּוֹרִים חָשַׁב שֶׁהוּא לְשׁוֹן יָוָן עִנְיָנוֹ עֲבוֹדַת הַגַּן, שֶׁאֶפְשָׁר שֶׁהָיָה שָׁם גַּן שֶׁל וְרָדִים מִחוּץ לְהַר הַבַּיִת . וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: טָדִי. בְּדל"ת. פֵּרוּשׁוֹ הַצְנַע, כְּמוֹ בִּטְעָמִים יְטִידוּךָ שֶׁיְּדַבְּרוּ בִּצְנִיעוּתֶיךָ. אִי נַמִּי, לְשׁוֹן פִּיּוּט, כְּמוֹ הָנֵי טַיְדֵּי וּפַיְטֵי רְגִילִים בְּהוּ קָמֵי רֵישׁ גָּלוּתָא. פִּסְקֵי תּוֹסָפוֹת. וְאוּלַי שֶׁהַפַּיְטָנִים הָיָה לָהֶם שָׁם לִשְׁכָּה שֶׁבָּהּ מְלַמְּדִים לַלְוִיִּם נִגּוּנֵי הַמִּזְמוֹרִים, כִּי יֵשׁ לְכָל מִזְמוֹר נִגּוּן מְיֻחָד. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: כָּךְ כָּתַב הָרַמְבַּ"ם. וְעַיֵּן פֶּרֶק ג' דְּפָרָה מִשְׁנָה ח' בְּדִבְרֵי הָרַ"ב:

פרק א - משנה ד

כְּלוֹמַר אַף בַּדָּרוֹם, בְּכָל מָקוֹם בַּעֲזָרָה, לַאֲפוֹקֵי קָדְשֵׁי קָדָשִׁים שֶׁשְּׁחִיטָתָן בַּצָּפוֹן. וּלְפִי שֶׁהֵם מַתָּנָה לַכֹּהֲנִים, נִתְיַחֵד לָהֶם שַׁעַר, לַאֲפוֹקֵי שְׁאָר קָדָשִׁים קַלִּים דְּשֶׁל בְּעָלִים הֵם:

פרק א - משנה ו

כִּדְמוּת בָּתֵּי הַשֵּׁנָה שֶׁיֵּשׁ בְּטִירוֹת הַמְּלָכִים שֶׁהוּא בַּיִת קָטָן בְּתוֹךְ בַּיִת גָּדוֹל כוּ'. וּטְרַקְלִין הוּא בֵּית הַמַּלְכוּת, וְקִיטוֹן מְקוֹם בֵּית הַשֵּׁנָה שֶׁיְּשֵׁנִים בּוֹ בִּזְמַן הַקַּיִץ. תַּרְגּוּם קַיִץ, קַיְטָא. הָרַמְבַּ"ם: מִמָּקוֹם שֶׁהוּא חֹל עַד הַמָּקוֹם הַקֹּדֶשׁ, וְלֹא מִמְּקוֹם הַקֹּדֶשׁ, מִשּׁוּם אִסּוּר דְּלֹא תִטַּע וְגוֹ' כָּל עֵץ אֵצֶל מִזְבַּח: לֶחֶם הַפָּנִים. וּסְמוּכִים לָעֲזָרָה, וְנִמְצָא שֶׁהָיְתָה מְקֻדֶּשֶׁת. אַתְיָא כְּמַאן דְּאָמַר אֲפִיָּתָן בִּפְנִים, בְּפֶרֶק י"א דִּמְנָחוֹת:

פרק א - משנה ז

שְׁנַיִם. וְלֹא קָא חֲשִׁיב רֵישׁ פִּרְקִין דְּהָכָא וּדְתָמִיד אֶלָּא חַד שׁוֹמֵר, כֵּיוָן דְּגַבֵּי אַהֲדָדֵי הֲווּ קַיְמֵי , סַגִּי לְהוּ בְּחַד שׁוֹמֵר, דָּוֵי לְהָכָא וְדָוֵי לְהָכָא. גְּמָרָא. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב:

פרק א - משנה ט

אַמָּה עַל אַמָּה. בָּאָרֶץ, וְטַבְלָא עָלָיו. הָרַ"שׁ: הִגְבִּיהַּ. הַכֹּהֵן הַשּׁוֹעֵר. הָרַ"שׁ. וְכָתַב הָרַמְבַּ"ם, לְפִי הַנִּרְאֶה מִמִּשְׁנָה זוֹ שֶׁהָיוּ צְרִיכִים מַפְתְּחוֹת לִסְגֹּר הַדְּלָתוֹת, וְהָיוּ מַנְעוּלֵי דַּלְתוֹתֵיהֶן בְּלִי סָפֵק דּוֹמִים לַמַּנְעוּלִים כוּ' שֶׁעוֹשִׂין אוֹתָם אֶצְלֵנוּ, וְאֵינָם נִסְגָּרִים אֶלָּא בְּמַפְתְּחוֹתֵיהֶם שֶׁבָּהֶם נִפְתָּחִים:
