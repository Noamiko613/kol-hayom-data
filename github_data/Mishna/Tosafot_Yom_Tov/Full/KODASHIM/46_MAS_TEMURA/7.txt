פרק ז

דְּכִי קָתָנֵי אַעִקַּר זִבְחָא. וּמִשּׁוּם הָכִי לֹא תָּנֵי וָלָד וּתְמוּרָה כִּדְאִיתָא בַּגְּמָרָא: פֵּרוּשׁ דְּטָמֵא כְּתִיב גַּם כֵּן בְּכָל הַקֳּדָשִׁים. שֶׁנֶּאֱמַר כָּל אִישׁ אֲשֶׁר יִקְרַב מִכָּל זַרְעֲכֶם אֶל הַקֳּדָשִׁים אֲשֶׁר יַקְדִּישׁוּ וְגוֹ' . וַהֲדַר כְּתִיב בִּשְׁלָמִים וְהַנֶּפֶשׁ אֲשֶׁר תֹּאכַל בָּשָׂר מִזֶּבַח הַשְּׁלָמִים וְטֻמְאָתוֹ עָלָיו . וּפִגּוּל וְנוֹתָר צָרִיךְ לוֹמַר דְּיָלֵיף לְהוּ בִּגְזֵרָה שָׁוָה כְּמוֹ שֶׁכָּתַב הָרַ"ב סוֹף פֶּרֶק ד' דִּזְבָחִים: דְּכֵיוָן דְּאִיעַבַּר לִפְנֵי פִּדְיוֹנָן קַדְּשֵׁי מִיָּד לְמַאן דְּאָמַר רֵישׁ פֶּרֶק ה' בִּמְעֵי אִמָּן הֵם קְדוֹשִׁים. וַאֲפִלּוּ לְמַאן דְּאָמַר בַּהֲוָיָתָן וּבִשְׁעַת הֲוָיָתָן אִמָּן חֻלִּין, מִכָּל מָקוֹם תַּפְסִינְהוּ קְדֻשָּׁה בִּשְׁעַת הֲוָיָתָן. תּוֹסָפוֹת: כְּלוֹמַר שֶׁאֵין בָּהֶן לַכֹּהֵן כְּלוּם כְּדִלְקַמָּן:

וּלְשׁוֹן הָרַמְבַּ"ם, עַל בַּעֲלֵי מוּמִין וּבְהֵמָה טְמֵאָה, וְרַשִׁ"י כָּתַב, עַל הָאֲבָנִים וְעַל הָעֵצִים. עַיֵּן תּוֹסְפוֹת יוֹם טוֹב:

אֵין כוּ'. שֶׁנֶּאֱמַר בִּבְכוֹר לֹא יַקְדִּישׁ אִישׁ אוֹתוֹ, שֶׁלֹּא יַעֲשֶׂנּוּ עוֹלָה אוֹ שְׁלָמִים, וְהוּא הַדִּין לִשְׁאָר קָדָשִׁים שֶׁאֵין מְשַׁנִּין אוֹתָן מִקְּדֻשָּׁה לִקְדֻשָּׁה, אֶחָד קָדְשֵׁי מִזְבֵּחַ וְאֶחָד קָדְשֵׁי בֶּדֶק הַבַּיִת. הָרַמְבַּ"ם. וְאַף עַל פִּי שֶׁיֵּשׁ לוֹמַר שֶׁלֹּא לָדוּן קָדְשֵׁי בֶּדֶק הַבַּיִת מִנֵּיהּ, יֵשׁ לוֹמַר דְּזוֹ אֵינָהּ אֶלָּא אַסְמַכְתָּא. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: הָרַמְבַּ"ם. וּמַשְׁמַע דְּמִקַּל לְחָמוּר מְשַׁנִּין. וְהָרַאֲבַ"ד הִשִּׂיג דִּבְבֶדֶק הַבַּיִת מְשַׁנִּין מִזֶּה לָזֶה אֲפִלּוּ מֵחָמוּר לְקַל, וְכֵן פֵּרֵשׁ רַשִׁ"י, קָדְשֵׁי בֶּדֶק הַבַּיִת שֶׁהִקְדִּישׁ לַמִּזְבֵּחַ לֹא עָשָׂה כְּלוּם כוּ'. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: טַעֲמוֹ כִּדְמַסִּיק בְּדִבּוּר וּמַחֲרִימִין. בְּשֵׁם הָרַמְבַּ"ם. וְזֶה לְשׁוֹן הַתּוֹסָפוֹת, מִשּׁוּם דְּקָדְשֵׁי מִזְבֵּחַ יֵשׁ עֲלֵיהֶן שֵׁם הַבְּעָלִים לִפְדּוֹתָן כְּשֶׁהוּמְמוּ, אֲבָל בְּקָדְשֵׁי בֶּדֶק הַבַּיִת שֶׁהִקְדִּישָׁן אֵין לַבְּעָלִים עֲלֵיהֶן יוֹתֵר מִשְּׁאָר אָדָם: כְּדֵי שֶׁיִּטֹּל אֶת הָעוֹר. רַשִׁ"י. וְהַתּוֹסָפוֹת כָּתְבוּ, כַּמָּה אָדָם רוֹצֶה לִתֵּן בְּשׁוֹר זֶה לְעוֹלָה וְאַף עַל פִּי שֶׁאֵינוֹ חַיָּב: וּכְפֵרוּשׁ רַשִׁ"י בַּמִּשְׁנָה, וְלָאו לְמֵידַק הָא קָדְשֵׁי בֶּדֶק הַבַּיִת לֹא, דְּאִם כֵּן בְּמַאי פְּלִיגֵי כוּ', אֶלָּא דְּהוּא הַדִּין קָדְשֵׁי בֶּדֶק הַבַּיִת. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב:

יִקָּבְרוּ. הַנְּפָלִים. רַשִׁ"י: וּפֶטֶר חֲמוֹר. וּרְמִינְהוּ הָאוֹרֵג מְלֹא הַסִּיט מִשְּׂעַר נָזִיר בְּשַׂק יִדְלַק הַשַּׂק. וּמְשָׁנֵי כָּאן בְּשַׂק כָּאן בְּשֵׂעָר. פֵּרֵשׁ רַשִׁ"י בְּשַׂק אִי אָמַרְתְּ יִקָּבֵר אָתֵי אֱינָשׁ וּמִתְהַנֵּי מִנֵּיהּ הוֹאִיל וְאֵינוֹ כָּלֶה עַד לְאַחַר זְמַן. וּמַתְנִיתִין בַּשֵּׂעָר עַצְמוֹ שֶׁלֹּא נֶאֱרַג, וְאִידִי וְאִידִי בְּטָהוֹר אוֹ טָמֵא:

שֶׁחֲכָמִים חוֹלְקִים עָלָיו. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: דְּיֵשׁ אֵם לַמִּקְרָא. וּלְמַאן דְּאָמַר יֵשׁ אֵם לַמָּסֹרֶת יֵשׁ דְּרָשׁוֹת אֲחֵרוֹת בְּמַסֶּכֶת שַׁבָּת דַּף כ"ה:

וּמֵהַאי טַעֲמָא נַמִּי בְּשַׂר קֹדֶשׁ וְכֵן הַמִּנְחָה שֶׁנִּטְמְאוּ אוֹ נִפְסְלוּ אוֹ נוֹתְרוּ כוּ'. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: רַשִׁ"י. וּלְפוּם רִיהֲטָא כָּתַב כֵּן. דְּהָא רַבָּנָן סְבִירָא לְהוּ בְּמִשְׁנָה ד', דְּחֻלִּין שֶׁנִּשְׁחֲטוּ בָּעֲזָרָה בִּקְבוּרָה. אֲבָל בְּרֵישׁ פֶּרֶק ו' דִּכְרֵתוֹת כָּתַב דְּאַף עַל גַּב דְּבִקְבוּרָה הַאי הָוֵי כְּזֶבַח פָּסוּל. וְעַיֵּן שָׁם: יִקָּבֵר. וְהָכִי נַמִּי פָּלֵיג בְּחַטַּאת הָעוֹף. וּבְתוֹרַת כֹּהֲנִים דָּרֵישׁ מִכִּי קֹדֶשׁ הֵם, הֵם בִּשְׂרֵפָה וְלֹא חַטַּאת הָעוֹף וַאֲשַׁם תָּלוּי בִּשְׂרֵפָה. תּוֹסָפוֹת: אֲבָל בְּנִקְבָּרִין אַף עַל גַּב דְּאִיכָּא נַמִּי לְמֵיחַשׁ לְהָכִי, מִכָּל מָקוֹם אִי אֶפְשָׁר לְשָׂרְפָן, דַּאֲפָרָן אָסוּר בַּהֲנָאָה. וְהוֹאִיל וְהַנִּשְׂרָפִין אֲפָרָן מֻתָּר אָתֵי לְמִתְהַנֵּי מִנֵּיהּ, וְאִי אֶפְשָׁר לוֹמַר שֶׁאֵפֶר אֵלּוּ יִקָּבֵר, לְפִי שֶׁאֵין בְּיַד כָּל אָדָם לְחַלֵּק בֵּין אֵפֶר לְאֵפֶר אֵיזֶהוּ בִּקְבוּרָה וְאֵיזֶהוּ בִּשְׂרֵפָה, וְאָתֵי לְמִטְעֵי. הִלְכָּךְ אָמְרוּ אוֹתָן שֶׁאֲפָרָן אָסוּר יִקְבְּרֶנּוּ בְּעַצְמָן. נִרְאֶה לִי: צָרִיךְ עִיּוּן טַעֲמָא מַאי. וְאוֹמֵר מוֹרִי הָרמ"ר דְּנִשְׂרָפִין כֵּיוָן שֶׁצִּוָּה הַכָּתוּב לְשָׂרְפָן אַחַר שֶׁעָשָׂה כְּאִלּוּ נַעֲשֵׂית מִצְוָתָן, וְאֵין לְךָ דָּבָר שֶׁנַּעֲשֵׂית מִצְוָתוֹ וּמוֹעֲלִין בּוֹ, הָכִי נַמִּי כֵּיוָן שֶׁנַּעֲשֵׂית מִצְוָתוֹ הָלַךְ אִסּוּרֵיהּ. אֲבָל הַנִּקְבָּרִים דְּלֹא הִטְעִין הַכָּתוּב לְשָׂרְפָן מָשׁוּךְ אִסּוּרַיְהוּ לְעוֹלָם. תּוֹסָפוֹת: וְכֵיוָן דְּטָעוּן גְּנִיזָה אָסוּר בַּהֲנָאָה. רַשִׁ"י: אֵינוֹ כוּ'. שֶׁאַף עַל פִּי שֶׁהוּא מַחֲמִיר בִּשְׂרֵפָתָהּ. הֲרֵי הֵקֵל בַּאֲפָרָן. שֶׁאֵפֶר הַנִּקְבָּרִים אָסוּר. הָרַמְבַּ"ם: תֵּימַהּ, דִּלְמַאי נַפְקָא מִנַּהּ מִתְחַשְּׁבִין בְּאַפֵּי נַפְשַׁיְהוּ כוּ'. אֲבָל הָרַמְבַּ"ם כָּתַב וּמְחַשְּׁבִין לָהּ מַה שֶּׁרָאוּי לָהֶם כוּ'. וּלְפִיכָךְ נִרְאֶה לִי לְהַגִּיהַּ בְּדִבְרֵי הָרַ"ב וְנִפְדִּין בְּשָׁוְיָן וְהַשְּׁאָר לְפִי חֶשְׁבּוֹן. וְעוֹד יֵשׁ לוֹמַר, דְּדַעְתּוֹ כְדַעַת הַתּוֹסָפוֹת דְּהַנָּפְקָא מִנַּהּ שֶׁלֹּא חָשְׁבִינַן לְהוּ הַמִּדְרוֹן. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב:
