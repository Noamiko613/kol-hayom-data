פרק י

תַּלְמוּד לוֹמַר כוּ'. וְאִי לָאו קַל וָחֹמֶר לֹא הֲוָה צָרִיךְ קְרָא, אַף עַל גַּב דִּסְתָמָא כְּתִיב דְּמַשְׁמַע בֵּין חֻלִּין בֵּין מֻקְדָּשִׁין דַּהֲוָה אֲמֵינָא דָּבָר הַלָּמֵד מֵעִנְיָנוֹ, בַּמֶּה הַכָּתוּב מְדַבֵּר, בְּחֻלִּין, דִּכְתִיב לְעֵיל מֵהַאי קְרָא גֵּז צֹאנְךָ, וְרֵאשִׁית הַגֵּז אֵינוֹ נוֹהֵג בְּמֻקְדָּשִׁין. תּוֹסָפוֹת:

כָּל כוּ'. לָאו בְּקָדְשֵׁי בֶדֶק הַבַּיִת אַיְרֵי, אֶלָּא בְּהִקְדִּישָׁן לַמִּזְבֵּחַ לִדְמֵיהֶן כוּ'. וְאַף עַל פִּי שֶׁעוֹבֵר כְּשֶׁמַּקְדִּישׁ בַּעַל מוּם לַמִּזְבֵּחַ, מִכָּל מָקוֹם אַשְׁמְעִינַן דִּינוֹ. תּוֹסָפוֹת: רַשִׁ"י. וּבַגְּמָרָא דִבְכוֹרוֹת גָּרְסִינַן נַמִּי וּמִן הַמַּתָּנוֹת. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: וְיוֹצְאִין כוּ'. לְאַחַר פִּדְיוֹנָן. כֵּיוָן דִּקְדֻשַּׁת דָּמִים לַמִּזְבֵּחַ נִינְהוּ דְּמִחַלְּפָא בִּקְדֻשַּׁת הַגּוּף לַמִּזְבֵּחַ, גָּזְרוּ בְּהוּ רַבָּנָן. אִי נַמִּי דְּקָדְשֵׁי בֶדֶק הַבַּיִת נַמִּי אֲסִירֵי בְּגִיזָה וַעֲבוֹדָה. גְּמָרָא. וְעוֹד קָאָמַר הָתָם דְּמִדְּרַבָּנָן הוּא בְּקָדְשֵׁי בֶדֶק הַבַּיִת: לְהִגָּזֵז. דְּאִלּוּ תְּמִימִים וְנִפְדּוּ, כְּתִיב תִּזְבַּח וְלֹא גִיזָה. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: פָּטוּר. אֲפִלּוּ קֹדֶם פִּדְיוֹן. וַאֲפִלּוּ הֲוֵי מוּמָן קַל כְּגוֹן דֻּקִּין שֶׁבָּעַיִן דִּלְגַבֵּי קָדָשִׁים מִחַיַּב הַשּׁוֹחֲטָן בַּחוּץ כְּדִלְקַמָּן. הָכָא פָּטוּר מִשּׁוּם דְּלֹא חָלָה עֲלֵיהֶן מִתְּחִלָּה קְדֻשָּׁה. רַשִׁ"י: וְאַשְׁמְעִינַן נַמִּי דְּלֹא בָעֵי הַעֲמָדָה וְהַעֲרָכָה. רַשִׁ"י: הַמַּתָּנוֹת. בִּגְמָרָא דִבְכוֹרוֹת יָלֵיף לְהוּ: פֵּרֵשׁ רַשִׁ"י, אָסוּר בַּהֲנָאָה בְּלֹא פְדִיָּה, וּמִקְרָב לֹא קָרְבֵי, דְּמִכֹּחַ קְדֻשָּׁה דְחוּיָה אָתֵי, דְּאִמָּם לֹא הֻקְדְּשָׁה לַמִּזְבֵּחַ. אַךְ תְּמֵהַנִי דְּמִנֵּיהּ וּבֵיהּ הֲוָה לֵיהּ לְפָרֵשׁ דְּאִי אִתְיַלַּד לִפְנֵי פִדְיוֹנוֹ מִקְדַּשׁ נַמִּי קַדִּישׁ, דְּהָכִי אִיתָא בַגְּמָרָא שָׁם. וּמַפִּיק לֵיהּ מִקְּרָא: עַיֵּן מַה שֶּׁכָּתַבְתִּי רֵישׁ פֶּרֶק ה':

פּוֹטְרִים. עַל כָּרְחֲךָ כְּשֶׁכֻּלָּן בַּעֲלֵי מוּמִין אַיְרֵי. דְּאֵין שׁוֹחֲטִין בְּכוֹר אֶלָּא אִם כֵּן נָפַל בּוֹ מוּם, וּמִדְּנִתְעָרֵב וְאֵין מַכִּירִין בּוֹ אִם כֵּן גַּם הָאֲחֵרִים בַּעֲלֵי מוּמִין הֵן. תּוֹסָפוֹת: הַשּׁוֹחֵט כוּ'. וְלִתְנֵי הַכֹּהֵן וְהַנָּכְרִי פְּטוּרִין. אָמַר רָבָא, זֹאת אוֹמֶרֶת הַדִּין עִם הַטַּבָּח. כְּלוֹמַר דְּיָכוֹל הַכֹּהֵן לִתְבֹּעַ מַתְּנוֹתָיו מֵהַטַּבָּח. וְיָלְפִינַן לֵיהּ בַּגְּמָרָא מִקְּרָא. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: פָּטוּר כוּ'. דִּכְתִיב מֵאֵת הָעָם, וְלֹא מֵאֵת הַכֹּהֲנִים. גְּמָרָא. וְכָל שֶׁכֵּן דְּאֵין נָכְרִי בִּכְלַל הָעָם זוֹבְחֵי זֶבַח: הַלּוֹקֵחַ. שֶׁהֲרֵי שֶׁל כֹּהֵן הֵן. וְדִכְוָתָהּ בְּיִשְׂרָאֵל הֲוָה מִיחַיַּב הַלּוֹקֵחַ, דְּכֵיוָן שֶׁהוּא שְׁחָטָהּ עֲלֵיהּ רַמְיָא לְמִתְבִינְהוּ. רַשִׁ"י: דְּמִדְּרַבָּנָן גָּזְרוּ כְּשֶׁהוּא טַבָּח לִתֵּן מַתְּנוֹת כְּהֻנָּה אֲפִלּוּ הַבְּהֵמָה שֶׁלּוֹ, שֶׁלֹּא יַרְגִּילוּ טַבָּחֵי יִשְׂרָאֵל לְשַׁתֵּף כֹּהֲנִים אֲחֵרִים עִמָּהֶם לִפָּטֵר מִן הַמַּתָּנוֹת. וְעַד שָׁלֹשׁ שַׁבָּתוֹת דְּאִיכָּא לְמֵימַר לְדִידֵיהּ שָׁחַט לֹא גָזְרוּ רַבָּנָן וְאוֹקְמוּהוּ עַל דְּבַר תּוֹרָה, וּמִכָּאן וָאֵילָךְ גָּזְרוּ. וְכִי קָבַע אַמְּסַחְתָּא מִיָּד מוּכָח דְּלָאו דִּידֵיהּ שָׁחֵיט וּלְכָךְ גָּזְרוּ לְאַלְתָּר. תּוֹסָפוֹת: יֵרָאֶה דְרָצָה לוֹמַר הַכֹּל לְפִי תֹאַר הָעִנְיָן, לִפְעָמִים שְׁתַּיִם וְלִפְעָמִים שָׁלֹשׁ, לְפִי רְאוֹת הַדַּיָּן דְּאִיכָּא פִּרְסוּמֵי דְּלָאו לְדִידֵיהּ שָׁחֵיט. וְהַתּוֹסָפוֹת כָּתְבוּ בְּשֵׁם רִ"י, דְּנָקַט שְׁתַּיִם לְרַבִּי דְּאִית לֵיהּ בִּתְרֵי זִמְנֵי הֲוֵי חֲזָקָה, וְשָׁלֹשׁ לְרַבָּן שִׁמְעוֹן בֶּן גַּמְלִיאֵל:

פָּטוּר. וּרְמִינְהוּ דִתְנַן סוֹף פֶּרֶק ד' דְּפֵאָה שֶׁסְּפֵק לֶקֶט לֶקֶט, הָכָא פָּרָה בְּחֶזְקַת פְּטוּרָה קַיְמָא, קָמָה בְּחֶזְקַת חִיּוּבָא קַיְמָא. וַהֲרֵי עִסָּה נַעֲשֵׂית עַד שֶׁלֹּא נִתְגַּיֵּר כוּ' סָפֵק חַיָּב, סְפֵק אִסּוּרָא לְחֻמְרָא וּסְפֵק מָמוֹנָא לְקֻלָּא. פֵּרֵשׁ רַשִׁ"י, חַלָּה סְפֵק אִסּוּרָא שֶׁיֵּשׁ בָּהּ עֲוֹן מִיתָה, הִלְכָּךְ לֹא סָמְכִינַן אַחֲזָקָה, אֲבָל מַתָּנוֹת אֵין בָּהֶן קְדֻשָּׁה אֶלָּא דִּין מָמוֹן, וְהַמּוֹצִיא מֵחֲבֵרוֹ עָלָיו הָרְאָיָה: כַּוָּנָתוֹ לוֹמַר, שֶׁהָעֶצֶם הָעֶלְיוֹן הַמְחֻבָּר לַגּוּף הוּא בִכְלַל הַזְּרוֹעַ, וְהוּא אֶחָד מִשְּׁנֵי עֲצָמוֹת, וְעַד דִּתְנַן הוּא עַד בִּכְלָל. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: וְאִם כֵּן לְרַבִּי יְהוּדָה אֵין בְּמַתְּנַת הַזְּרוֹעַ רַק עֶצֶם הָאֶמְצָעִי לְבַדּוֹ, וּבְהָא פָּלֵיג אַתַּנָּא קַמָּא. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: אֵיזֶהוּ כוּ'. וּבַגְּמָרָא. וְהַלְּחָיַיִם לְמַאי אֲתָא, לְהָבִיא צֶמֶר שֶׁבְּרֹאשׁ כְּבָשִׂים וְשֵׂעָר שֶׁבִּזְקַן תְּיָשִׁים. וְהַקֵּבָה לְמַאי אֲתָא, לְהָבִיא חֵלֶב שֶׁעַל גַּבֵּי הַקֵּבָה וְשֶׁבְּתוֹכָהּ:
