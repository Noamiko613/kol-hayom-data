פרק ב - משנה א

דִּלְכַתְּחִלָּה הִצְרִיכוּהוּ רַבָּנָן תַּרְוַיְהוּ מִשּׁוּם הַרְחָקָה לַעֲבֵרָה דִּלְמָא לֹא אָתֵי לְמֶעְבַּד רֻבָּא דְחַד. רַשִׁ"י: דְּלָאו בְּנֵי שְׁחִיטָה כְּלָל נִינְהוּ, מִשּׁוּם דִּכְתִיב אֲסֵפָה דִּידְהוּ בִּמְקוֹם שְׁחִיטָה דְּאַחֲרִינֵי, כְּדִכְתִיב הֲצֹאן וּבָקָר יִשָּׁחֵט לָהֶם וּמָצָא לָהֶם אִם אֶת כָּל דְּגֵי הַיָּם יֵאָסֵף לָהֶם, וּמִדְּשָׁנֵי קְרָא בְּדִבּוּרֵיהּ וְכָתַב בְּהוּ אֲסֵפָה, שְׁמַע מִנַּהּ דַּוְקָא הוּא. גְּמָרָא: לָאו מִקְּרָא יָלֵיף. אֶלָּא כְּלוֹמַר עַל הִלְכוֹת שְׁחִיטָה נִצְטַוָּה. רַשִׁ"י. וְהַתּוֹסָפוֹת כָּתְבוּ דְּיֵשׁ מְפָרְשִׁים בְּגִימַטְרִיָּא דְּכַאֲשֶׁר, א' אֶחָד בָּעוֹף, ש' שְׁנַיִם בַּבְּהֵמָה, ר' רֻבּוֹ שֶׁל אֶחָד כָּמוֹהוּ, רָאשֵׁי תֵבוֹת דְּכַאֲשֶׁר לְמַפְרֵעַ. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: שֶׁל אֶחָד. רֻבּוֹ שֶׁל כָּל אֶחָד וְאֶחָד. רַשִׁ"י: חֲצִי אֶחָד כוּ'. הוּא הַדִּין חֲצִי כָל אֶחָד וְאֶחָד. וּמֵרֵישָׁא נַמִּי שְׁמִיעַ לֵיהּ דְּדַוְקָא רֹב בָּעִינַן. וְלֹא נָקְטֵיהּ אֶלָּא אַגַּב דְּבָעֵי לְמִתְנֵי אֶחָד וָחֵצִי בַּבְּהֵמָה, אַף עַל גַּב דַּהֲווּ טְפֵי מֵרֹב הַשְּׁנַיִם בֵּין הַכֹּל, שְׁחִיטָה פְסוּלָה. תּוֹסָפוֹת:

פרק ב - משנה ב

הַשּׁוֹחֵט כוּ'. דִּיעֲבַד אִין לְכַתְּחִלָּה לֹא. בְּקָדָשִׁים מַיְרֵי. דְּתָנֵי רַב יוֹסֵף, תִּזְבַּח, שֶׁלֹּא יִהְיוּ שְׁנַיִם שׁוֹחֲטִין זֶבַח אֶחָד. תִּזְבָּחוּהוּ, שֶׁלֹּא יְהֵא אֶחָד שׁוֹחֵט שְׁנֵי זְבָחִים. גְּמָרָא. וְהַשְׁתָּא סֵיפָא שְׁנַיִם אוֹחֲזִין דִּלְכַתְּחִלָּה, דַּוְקָא בְּחֻלִּין:

פרק ב - משנה ג

חֲבֵרוֹ כוּ'. הוּא הַדִּין כְּשֶׁחָזַר הוּא בְעַצְמוֹ וְשָׁחַט. אֶלָּא אוֹרְחָא דְמִלְּתָא נָקַט דִּכְשֶׁהוּא מִתְיַגֵּעַ בָּא חֲבֵרוֹ וְשׁוֹחֵט. כֶּסֶף מִשְׁנֶה: כְּלוֹמַר בְּגַסָּה לְגַסָּה, וּבְדַקָּה לְדַקָּה. וְכֵן לְעוֹף:

פרק ב - משנה ד

פֵּרְשׁוּ הַתּוֹסָפוֹת, דְּצָרִיךְ לוֹמַר דְּמַיְרֵי שֶׁשָּׁחַט בְּסַכִּין פְּגוּמָה אוֹ בְמַגָּל, וּפְסוּלָהּ הוּא מִשּׁוּם דְּאֵין שׁוֹחֲטִין אֶלָּא חוֹנְקִין. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: אַחַד מֵהֶן. לְרַבִּי עֲקִיבָא אִצְטְרִיךְ: בִּשְׁחִיטָתָהּ. כְּלוֹמַר בַּמָּקוֹם הָרָאוּי לִשְׁחִיטָה. גְּמָרָא:

פרק ב - משנה ה

וְלֹא יָצָא כוּ'. וְאֵין אוֹמְרִים שֶׁמָּא מֵתִים הָיוּ. הָרַמְבַּ"ם: דִּבְקָדָשִׁים לֹא, דְּהָא תְּנַן חַיָּה. גְּמָרָא. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: הַיְנוּ בְּדָבָר שֶׁאֵינוֹ מִין לֶחֶם. כְּמוֹ שֶׁכָּתַבְתִּי בְּפֶרֶק ב' דַּחֲגִיגָה מִשְׁנָה ה': שֶׁלֹּא כוּ'. וּלְצָלִי אַיְרֵי, דְּלֹא בָעֵי הֲדָחָה. אִי נַמִּי לִקְדֵרָה וְהוּדְחוּ בְּמֵי פֵרוֹת. תּוֹסָפוֹת:

פרק ב - משנה ו

הַמְסֻכֶּנֶת. שֶׁלֹּא אָסְרָה הַתּוֹרָה אֶלָּא כְּעֵין טְרֵפַת חַיַּת הַיַּעַר שֶׁהֲרֵי עָשָׂה בָהּ מַכָּה הַמְּמִיתָה אוֹתָהּ. הָרַמְבַּ"ם. וַאֲפִלּוּ לְכַתְּחִלָּה שָׁרֵי. וְנָקַט לְשׁוֹן דִּיעֲבַד מִשּׁוּם דְּבָעֵי לְמִתְנֵי עַד שֶׁתְּפַרְכֵּס: זִנְּקָה. זִנּוּק, קִלּוּחַ. רַשִׁ"י. וְנִגְזַר מִן יְזַנֵּק מִן הַבָּשָׁן . הָרַמְבַּ"ם: וְנָקַט לִישְׁנָא דְתַנָּא קַמָּא אַף עַל גַּב דִּכְמִדַּת רַבִּי אֱלִיעֶזֶר קַיְמִינַן דְּדַיָּהּ אִם זִנְּקָה. וּבַגְּמָרָא כָּתַב רַשִׁ"י, וּכְמִדַּת רַבִּי אֱלִיעֶזֶר דְּאָמַר זִנּוּק הֲוֵי פִרְכּוּס:

פרק ב - משנה ז

שְׁחִיטָתוֹ כְשֵׁרָה. מְפָרֵשׁ בַּגְּמָרָא מִשּׁוּם דִּסְבִירָא לֵיהּ דְּאֵין סְתָם מַחֲשֶׁבֶת נָכְרִי לַעֲבוֹדָה זָרָה, וּבְהָא הוּא דְפָלֵיג אַדְּרַבִּי אֱלִיעֶזֶר. אֲבָל תַּרְוַיְהוּ סְבִירָא לְהוּ דִּבְעָלִים מְפַגְּלִים. וְכָתַב הָרַ"ן דְּהָא דְקָתָנֵי הַשּׁוֹחֵט, הוּא הַדִּין דַּאֲפִלּוּ לְכַתְּחִלָּה, אֶלָּא דְּלֹא מָצֵי לְמִתְנֵי שׁוֹחֲטִין לַנָּכְרִי, דְּאָז הֲוֵי מַשְׁמַע דִּלְמִשְׁרֵי לְמִשְׁחָט הוּא דַאֲתָא דְּלֵית בֵּיהּ מִשּׁוּם לִפְנֵי עִוֵּר לֹא תִתֵּן מִכְשׁוֹל אֲבָל לֹא שֶׁתְּהֵא שְׁחִיטָתוֹ כְשֵׁרָה, הִלְכָּךְ הֻצְרַךְ לִשְׁנוֹת שְׁחִיטָתוֹ כְשֵׁרָה. אֲבָל אִין הָכִי נַמִּי דַּאֲפִלּוּ לְכַתְּחִלָּה: וַאֲסוּרָה בַהֲנָאָה כְּתִקְרֹבֶת עֲבוֹדָה זָרָה. וְאַף עַל גַּב דִּבְמַתְנִיתִין דִּלְקַמָּן דִּתְנַן נַמִּי פְּסוּלָה וּמְפָרְשִׁינַן דְּאֵינָהּ אֲסוּרָה בַהֲנָאָה, יֵשׁ לוֹמַר דִּלְקַמָּן הֲוָה לֵיהּ לְמִתְנֵי הֲרֵי אֵלּוּ זִבְחֵי מֵתִים, אֲבָל הָכָא אַיְדֵי דְאָמְרֵי רַבָּנָן כְּשֵׁרָה, אָמַר רַבִּי אֱלִיעֶזֶר פְּסוּלָה. תּוֹסָפוֹת: מֵחֲצַר כָּבֵד. לָאו דַּוְקָא נָקַט יוֹתֶרֶת הַכָּבֵד מִשּׁוּם דְּעוֹלָה לְגָבוֹהַּ וְהַוְיָא כְּחֵלֶב וָדָם, אֶלָּא אֲפִלּוּ כַּזַּיִת שֶׁל בָּשָׂר נַמִּי, וְאוֹרְחָא דְמִלְּתָא הוּא דְנָקַט. תּוֹסָפוֹת:

פרק ב - משנה ח

פְּסוּלָה. וְהָא דְמַסִּיק הָרַ"ב בְּמִשְׁנָה ו' פֶּרֶק ז' דְּכִלְאַיִם דְּאֵין אָדָם אוֹסֵר דָּבָר שֶׁאֵינוֹ שֶׁלּוֹ, אִיכָּא הָכָא בַּגְּמָרָא תְּרֵי אוֹקִימְתּוֹת. חֲדָא, דְּהָכָא בְּיֵשׁ לוֹ בָהּ שֻׁתָּפוּת. וְכֵן פָּסַק הָרַמְבַּ"ם. וַחֲדָא בְּיִשְׂרָאֵל . אִי נַמִּי שֶׁהִתְרוּ בוֹ וְקִבֵּל הַתְרָאָה. וְכֵן פֵּרֵשׁ הָרֹא"שׁ: וּדְרָשָׁא פְּשׁוּטָה נָקַט, אַף עַל גַּב דְּרַבִּי יוֹסֵי אֲמָרָהּ. וְתַנָּא קַמָּא נָסֵיב דְּרָשָׁא אַחֲרִיתִי בְּמִשְׁנָה ה' פֶּרֶק ג' דַּעֲבוֹדָה זָרָה:

פרק ב - משנה ט

מִשּׁוּם הָכִי לֹא שְׁנָא צְלוּלִין לֹא שְׁנָא עֲכוּרִין. כֶּסֶף מִשְׁנֶה: וְאַף עַל גַּב דְּפָסַק הָרַ"ב בְּמִשְׁנָה ד' פֶּרֶק כ"ב דְּשַׁבָּת דְּכָל מָקוֹם שֶׁאָסְרוּ חֲכָמִים מִפְּנֵי מַרְאִית הָעַיִן אֲפִלּוּ בְּחַדְרֵי חֲדָרִים אָסוּר, הַיְנוּ כְּמוֹ שׁוֹטְחוֹ בַחַמָּה אֲבָל לֹא כְנֶגֶד הָעָם, שֶׁאִם הָיוּ רוֹאִין אוֹתוֹ עוֹשֶׂה כֵּן בְּצִנְעָה הָיָה שָׁם חֲשָׁד כְּמוֹ בַשּׁוּק. אֲבָל כָּאן הָרוֹאֶה שֶׁעוֹשֶׂה בְּבֵיתוֹ אוֹמֵר לְנַקֵּר חֲצֵרוֹ הוּא עוֹשֶׂה. תּוֹסָפוֹת:

דְּקָא שָׁחֵיט פֶּסַח בִּשְׁאָר יְמוֹת הַשָּׁנָה, וְקַיְמָא לָן דִּשְׁלָמִים הוּא. רַשִׁ"י: תּוֹדָה. דְּבָא בְּנֶדֶר וּנְדָבָה: פְּסוּלָה. הִקְשָׁה בֵּית יוֹסֵף לְהָרֹא"שׁ דַּאֲפִלּוּ יֵשׁ לוֹ בָהּ שֻׁתָּפוּת אֵינוֹ אוֹסֵר, וְאַמַּאי פָּסֵיל לָהּ הָכָא. וְתֵרֵץ דִּסְבִירָא לֵיהּ דְּלֹא אָמְרוּ אֵין אָדָם כוּ', אֶלָּא בְּדָבָר שֶׁאִסּוּרוֹ מִן הַדִּין, מִשּׁוּם דְּמִסְּתָמָא לְצִעוּרֵי קָא מְכַוֵּן, אֲבָל הָכָא שֶׁאֵין אִסּוּרוֹ מִן הַדִּין אֶלָּא מִפְּנֵי מַרְאִית עַיִן, כִּי לֹא הֲוֵי דִּידֵיהּ מַאי הֲוֵי, הָא מַאן דְּחָזֵי סָבַר דִּידֵיהּ הוּא וַאֲתָא לְמִשְׁרֵי קָדָשִׁים בַּחוּץ, הִלְכָּךְ אֲפִלּוּ אֵין לוֹ בָהּ שֻׁתָּפוּת אֲסוּרָה. וְהָרַמְבַּ"ם מַצְרִיךְ גַּם כָּאן שֻׁתָּפוּת. דְּסָבַר כָּל הַשּׁוֹחֵט שֶׁאֵינוֹ שֶׁלּוֹ לְשֵׁם דָּבָר פָּסוּל קָלָא אִית לֵיהּ. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: חַטָּאת. כְּשֶׁלֹּא חָטָא. אֲבָל כְּשֶׁחָטָא וְחַיָּב חַטָּאת כִּי אָמַר זוֹ לְחַטָּאת, פְּסוּלָה. גְּמָרָא: וְכֵן הוּא לְשׁוֹן רַשִׁ"י. וּתְמִיהָה לִי שֶׁהִשְׁמִיטוּ לַאֲשַׁם מְצֹרָע. אֲבָל אֲשַׁם נָזִיר אָתִי בְּזֶה הַכְּלָל כְּדִלְקַמָּן: תְּמוּרָה. כְּשֶׁאֵין לוֹ זֶבַח בְּבֵיתוֹ. אֲבָל יֵשׁ לוֹ וְאָמַר לְשֵׁם תְּמוּרַת זִבְחִי, פְּסוּלָה. גְּמָרָא: אָסוּר. וּכְפִי שִׁיטַת לִישְׁנָא דְּמִשְׁנָתֵנוּ הֲוָה לֵיהּ לְמֵימַר פָּסוּל, אֶלָּא דַחֲדָא מִלְּתָא הוּא פּוֹסְלִין וּמַכְשִׁירִין וְאוֹסְרִין וּמַתִּירִין, כִּדְאִיתָא בַגְּמָרָא: כֵּן פֵּרֵשׁ הָרַמְבַּ"ם. וְתָמוּהַּ הוּא דְּהָא אָמְרִינַן הֵיכָא דְשַׁיָּךְ חוֹבָה אִין הָכִי נַמִּי דְּמִפְּסַל, וְאִם כֵּן הָכָא מִמָּה נַפְשָׁךְ מִפְּסַל. וּבְחִבּוּרוֹ כָּתַב שֶׁאֵין חוֹשְׁשִׁין לָהּ שֶׁמָּא הִפִּילָה, שֶׁכָּל הַמַּפֶּלֶת יֵשׁ לָהּ קוֹל. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב:
