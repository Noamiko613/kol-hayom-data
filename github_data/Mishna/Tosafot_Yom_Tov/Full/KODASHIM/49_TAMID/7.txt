דְּהָא דִכְתִיב וְאַל יָבֹא בְּכָל עֵת אֶל הַקֹּדֶשׁ , דָּרְשִׁינַן בִּמְנָחוֹת דְּקֹדֶשׁ הוּא הַהֵיכָל בְּאַל יָבֹא, וְאֶל מִבֵּית לַפָּרֹכֶת כוּ' בְּמִיתָה. וּמִלְּשׁוֹן הָרַ"ב מַשְׁמַע דְּכָל הִשְׁתַּחֲוָיָה שָׁרְיָא וְלֹא הַוְיָא שֶׁלֹּא לְצֹרֶךְ. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: שֶׁשָּׁמַע כוּ'. עַל יְדֵי פַעֲמוֹנֵי זָהָב שֶׁבְּשׁוּלֵי הַמְּעִיל: יוֹצֵא. פֵּרוּשׁ, יוֹצֵא מִמְּקוֹמוֹ בַּאֲשֶׁר הוּא שָׁם וְרוֹצֶה לִכָּנֵס לַהֵיכָל, קָדַם הוּא וְהִגְבִּיהַּ כוּ'. אֲבָל הָרַמְבַּ"ם הֶעְתִּיק כוּ' וְיִכָּנֵס לַהֵיכָל וְיִשְׁתַּחֲוֶה וְכֵיוָן שֶׁיִּשְׁמַע הַסְּגָן קוֹל רַגְלָיו שֶׁל כֹּהֵן גָּדוֹל שֶׁהוּא יוֹצֵא מַגְבִּיהַּ לוֹ אֶת הַפָּרֹכֶת. נִרְאֶה דְּשֶׁהוּא יוֹצֵא הַיְנוּ מֵהַהֵיכָל כוּ'. וְקָשֶׁה, דְּלָמָּה לֹא מַגְבִּיהַּ נַמִּי בִּכְנִיסָתוֹ לְשָׁם. וְנִרְאֶה לִי שֶׁהָיָה הַפָּרֹכֶת כָּל הַיּוֹם מֻגְבָּהּ וְעוֹמֵד עַל יְדֵי קְשָׁרִים וִיתֵרוֹת לִכָּנֵס וְלָצֵאת הַכֹּהֲנִים הַצְּרִיכִים לַעֲבוֹדַת מְנוֹרָה וּמִזְבֵּחַ פְּנִימִי, אֶלָּא שֶׁכְּשֶׁנִּכְנָס הַכֹּהֵן גָּדוֹל הָיוּ מְסִירִים הַפָּרֹכֶת מֵהַקְּשָׁרִים וִיתֵרוֹת כְּדֵי שֶׁיְּהֵא נִתְלֶה עַל עָמְדוֹ, וְכָבוֹד גָּדוֹל נַעֲשָׂה לוֹ לַכֹּהֵן גָּדוֹל בָּזֶה לְהַצְנִיעַ אַחֲרָיו הַקֹּדֶשׁ כְּאִלּוּ מִתְיַחֵד שָׁם עִם הַשּׁוֹכֵן בַּבַּיִת בָּרוּךְ הוּא, וּלְפִיכָךְ אַחַר שֶׁחָזַר וְרָצָה לָצֵאת הֻצְרָךְ לְהַגְבִּיהַּ, וְלֹא כֵן בִּשְׁעַת כְּנִיסָתוֹ. כֵּן נִרְאֶה לִי:

לְשׁוֹן הָרַמְבַּ"ם, רוֹמֵז אֶל כָּל הַכֹּהֲנִים שֶׁקָּרְאוּ בְּלִשְׁכַּת הַגָּזִית וְאַחַר כָּךְ עָשׂוּ מִן הָעֲבוֹדוֹת מַה שֶּׁעָשׂוּ: שֶׁזָּכַר אוֹתָם וְאָמַר שֶׁגָּמְרוּ עֲבוֹדָתָן תְּחִלָּה וְעָמְדוּ עַל מַעֲלוֹת הָאוּלָם קֹדֶם שֶׁיַּגִּיעוּ שְׁאָר אֲחֵיהֶם הָעוֹסְקִים בְּהַעֲלָאַת אֵבְרֵי הַתָּמִיד. הָרַמְבַּ"ם. וְכֵן כָּתַב בְּחִבּוּרוֹ שֶׁאַחַר שֶׁמַּעֲלִין הָאֵבָרִין לַמִּזְבֵּחַ מַתְחִילִין אֵלּוּ שֶׁעַל מַעֲלוֹת הָאוּלָם וּמְבָרְכִין בִּרְכַּת כֹּהֲנִים וְכוּ':

לְהַקְטִיר. עַל שֵׁם שֶׁעוֹלֶה קִטּוּר וְתִמְרוֹת עָשָׁן מִן הַחֲלָבִים וְהָאֵבָרִים קָרֵי לְהוּ בִּלְשׁוֹן הַקְטָרָה. עַיֵּן תּוֹסְפוֹת יוֹם טוֹב: דְּאִלּוּ סְמִיכָה דִקְרָא לֹא אַשְׁכְּחַן אֶלָּא בְּחַיִּים וְלֹא בִשְׁחוּטִין: הַשֵּׁנִי כוּ'. הַאי נַמִּי מִשּׁוּם יְקָרָא דְכֹהֵן גָּדוֹל, וּכְעֵין קְרִיאַת כֹּהֵן גָּדוֹל בְּיוֹם הַכִּפּוּרִים: לְנַסֵּךְ. הוּא הַדִּין לַהַעֲלָאַת סֹלֶת וְהַקְטָרַת הַחֲבִתִּין, אֶלָּא דְּלֹא תָנֵי אֶלָּא אֵלּוּ שֶׁנִּתְחַדֵּשׁ בָּהֶם דָּבָר בְּכֹהֵן גָּדוֹל: שֶׁאֵין אוֹמְרִים שִׁירָה אֶלָּא עַל הַיַּיִן, כִּדְיָלֵיף בַּגְּמָרָא מִקְּרָא. וּפֵרְשׁוּ הַתּוֹסָפוֹת, אֵין אוֹמְרִים שִׁירָה עַל שׁוּם אֲכִילַת מִזְבֵּחַ, כְּגוֹן זְרִיקַת דָּמִים וְנִסּוּךְ הַמַּיִם, כִּי אִם עַל הַיַּיִן. אֲבָל וַדַּאי מָצִינוּ שִׁירָה בְּלֹא יַיִן, כְּגוֹן הַלֵּל בִּשְׁחִיטַת פְּסָחִים: תָּקְעוּ. לְפִי שֶׁאֶפְשָׁר לָאָדָם שֶׁלֹּא יִרְאֶה הֲנָפַת הַסּוּדָרִים כְּשֶׁיִּפְנֶה לִבּוֹ לְדָבָר אַחֵר, וְאִי נַמִּי כְּשֶׁהָיוּ בְתוֹךְ הַלְּשָׁכוֹת אוֹ בְהַר הַבַּיִת לֹא בָעֲזָרָה, אֲבָל קוֹל הַתְּקִיעוֹת אִי אֶפְשָׁר שֶׁלֹּא יִשְׁמָעֵם, וּלְפִיכָךְ תָּקְעוּ כְּדֵי שֶׁיִּפְנוּ לִרְאוֹת בַּהֲנָפַת הַסּוּדָרִים. נִרְאֶה לִי: בַּצֶּלְצָל. כְּלֵי זֶמֶר אֲחֵרִים הָיוּ גַם כֵּן שָׁם, אֶלָּא שֶׁזֶּה הָיָה כְּלִי זֶמֶר מְיֻחָד שֶׁאֵין שָׁם אֶלָּא הוּא, וְקוֹל גָּדוֹל הָיָה לוֹ וְהוּא הַמַּתְחִיל, וּמְמֻנֶּה מְיֻחָד הָיָה לוֹ: וּבְמִשְׁנָה ה' פֶּרֶק ה' דְּסֻכָּה כָּתַב הָרַ"ב, שֶׁבַּשִּׁיר הָיוּ שְׁלֹשָׁה פְּרָקִים שֶׁמַּפְסִיקִים בּוֹ, וּמַשְׁמַע דְּלֹא קָחֲשִׁיב לְהָנָךְ דְּהָכָא כוּ', יֵשׁ לוֹמַר מִשּׁוּם דְּאֶפְשָׁר בְּלֹא הֵם עַל יְדֵי הַסּוּדָרִים כוּ'. אִי נַמִּי, תַּנָּאֵי נִינְהוּ. עַיֵּן תּוֹסְפוֹת יוֹם טוֹב: הָעָם. כָּל הָעָם שֶׁבָּעֲזָרָה:

לַה' כוּ'. כָּל הַמִּזְמוֹר. רַשִׁ"י: וְנִתְעַלָּה וְיָשַׁב בַּמָּרוֹם דֻּגְמַת שָׁכְנוֹ בְּעִירוֹ וְהַר קָדְשׁוֹ. רַשִׁ"י: וּלְשׁוֹן הָרַמְבַּ"ם, שֶׁעָלֶיהָ יִהְיֶה הַדִּין וְהַדַּיָּנִין. עַד כָּאן. וְאַסֵּיפָא דִקְרָא קָיְמֵי, בְּקֶרֶב אֱלֹהִים יִשְׁפֹּט. וּבַגְּמָרָא דְרֹאשׁ הַשָּׁנָה שָׁם, עַל שׁוּם שֶׁגִּלָּה אֶרֶץ בְּחָכְמָתוֹ וְהֵכִין תֵּבֵל לַעֲדָתוֹ. פֵּרֵשׁ רַשִׁ"י שֶׁגִּלָּה אֶרֶץ מְקוֹם מַצַּב עֲדָתוֹ: לְחַיֵּי. וְהָרַ"ב הֶעְתִּיק לְחֵי הָעוֹלָמִים. וּשְׁנֵיהֶם צָדְקוּ יַחְדָּיו. שֶׁאִם אַתָּה אוֹמֵר לְחַיֵּי הָעוֹלָמִים, הַכַּוָּנָה לַהַשֵּׁם יִתְבָּרַךְ שֶׁהוּא חַיֵּי הָעוֹלָמִים כֻּלָּם, כָּעִנְיָן שֶׁנֶּאֱמַר וְאַתָּה מְחַיֶּה אֶת כֻּלָּם. וַאֲפִלּוּ אִם תֹּאמַר חַיֵּי הָעוֹלָם הוּא נָכוֹן, כִּי הוּא סָמוּךְ מִשֵּׁם חַיִּים שֶׁלֹּא נִמְצָא נִפְרָד בִּלְשׁוֹן הַקֹּדֶשׁ. וּכְשֶׁאַתָּה אוֹמֵר חֵי הָעוֹלָמִים גַּם כֵּן נָכוֹן, וְתִהְיֶה הַקְּרִיאָה חֵי בְּצֵירֵ"י, כָּעִנְיָן שֶׁנֶּאֱמַר וַיִּשָּׁבַע בְּחֵי הָעוֹלָם , שֶׁהוּא סָמוּךְ, וְיִהְיֶה עִנְיָנוֹ חַיִּים שֶׁל הָעוֹלָמִים וְהוּא הַשֵּׁם יִתְבָּרַךְ שֶׁהוּא חִיּוּתוֹ שֶׁל עוֹלָם כָּאָמוּר. מַה שֶּׁאֵין כֵּן חַי בְפַתָּ"ח, שֶׁאֵינוֹ סָמוּךְ, יִהְיֶה עִנְיָנוֹ הָעוֹלָם שֶׁהוּא חַי, וְאֵין הָעוֹלָם עַצְמוֹ חַי עַל צַד הָאֱמֶת. כֵּן כָּתַב הָרַמְבַּ"ם. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: לַאֲפוֹקֵי מִמַּאן דְּאָמַר תְּרֵי חֲרוּב, שֶׁנֶּאֱמַר יְחַיֵּינוּ מִיֹּמַיִם. גְּמָרָא:
