פרק ב - משנה א

חַטַּאת כוּ'. עַד הַשְׁתָּא אַיְרֵי בְדִין פְּסוּלֵי קָדָשִׁים מָתַי יִמְעֲלוּ בָהֶם. וּמִכָּאן וָאֵילָךְ אַיְרֵי בִּכְשֵׁרִים. תּוֹסָפוֹת. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: מִשֶּׁהֻקְדְּשָׁה. בַּפֶּה, אֲפִלּוּ אִם נֶהֱנָה מִמֶּנָּה מֵחַיִּים, כְּגוֹן שֶׁמְּכָרָהּ דְּהוֹצִיאָהּ לְחֻלִּין. רַשִׁ"י: הֻכְשְׁרָה. כָּל מַה שֶּׁאָמַר בְּזֶה הַפֶּרֶק הֶכְשֵׁר, רָצָה לוֹמַר הֲכָנָה לַדְּבָרִים שֶׁעֲתִידִים לִזְכֹּר אוֹתָם, לֹא עִנְיַן הֶכְשֵׁר טֻמְאָה הַנִּזְכָּר בְּכָל מָקוֹם. וְדַע זֶה, וְלֹא יַטְעֲךָ שִׁתּוּף הָעִנְיָן. הָרַמְבַּ"ם: וּבְלִינָה. בְּלִינַת דָּם בִּשְׁקִיעַת הַחַמָּה, אִי נַמִּי בְלִינַת בָּשָׂר לְיוֹם וָלַיְלָה. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב. וְלֹא קָתָנֵי וּבַיּוֹצֵא, דְּמַתְנִיתִין רַבִּי עֲקִיבָא הִיא וּלְדִידֵיהּ הַזָּיָה מַהֲנֵי לֵיהּ לַיּוֹצֵא, וּמִדְּלֹא פָסֵיק לֵיהּ לֹא קָתָנֵי: וְאֵין כוּ'. לְפִי שֶׁאֵין בְּחַטַּאת הָעוֹף אֵמוּרִים רַק כֻּלָּהּ לַכֹּהֲנִים כִּבְשַׂר חַטָּאת. הָרַמְבַּ"ם. וְעַיֵּן סוֹף פִּרְקִין דִּלְעֵיל:

פרק ב - משנה ב

וְשׁוּב כֵּיוָן שֶׁנַּעֲשֵׂית מִצְוָתוֹ אֵין מוֹעֲלִין בּוֹ. רַשִׁ"י. כְּלוֹמַר דְּלֹא אִקְרוּ עוֹד קָדְשֵׁי ה'. וְעַד שֶׁתֵּצֵא, עַד שֶׁתֵּרָאֶה. גְּמָרָא. פֵּרֵשׁ רַשִׁ"י אַחַר שֶׁהֵרִים כוּ' אוֹתוֹ הַנִּשְׁאָר רָאוּי וְעוֹמֵד לְהוֹצָאַת הַדֶּשֶׁן:

פרק ב - משנה ג

וְהוּא הַדִּין בִּשְׁאָר הָעֲבוֹדוֹת: שֶׁיֻּתַּךְ. גָּזוּר מִן תּוֹךְ. רְצוֹנוֹ לוֹמַר עַד שֶׁיֵּעָשֶׂה הַבָּשָׂר חָלוּל וְיִתְהַפְּכוּ חֲלָקָיו בָּאֵשׁ עַד שֶׁיִּדְמֶה לִסְפוֹג הַיָּם. וְעִנְיָן זֶה יִהְיֶה בַבָּשָׂר אַחַר שֶׁיִּשָּׂרֵף לְגַמְרֵי. הָרַמְבַּ"ם. וְעַיֵּן סוֹף פֶּרֶק י"ב דִּזְבָחִים. וְצָרִיךְ לוֹמַר דְּכָאן גָּרַס שֶׁיִּתֹּךְ. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: וַהֲרֵי תְּרוּמַת הַדֶּשֶׁן שֶׁנַּעֲשֵׂית מִצְוָתוֹ וְיֶשׁ בּוֹ מְעִילָה כוּ', מִשּׁוּם דַּהֲווּ תְּרוּמַת הַדֶּשֶׁן וְעֶגְלָה עֲרוּפָה שְׁנֵי כְתוּבִין הַבָּאִין כְּאֶחָד וְאֵין מְלַמְּדִין. וּלְמַאן דְּאָמַר מְלַמְּדִים, תְּרֵי מִעוּטֵי הָעֲרוּפָה (הֵ"א קַמַּיְתָא) וְשָׂמוֹ (וָי"ו בַּתְרָאֵי), הָנֵי אִין מִידִי אַחֲרִינֵי לֹא. גְּמָרָא:

פרק ב - משנה ד

שֶׁיֵּצֵא. מוּסָב עַל הַבָּשָׂר שֶׁהוּא לְשׁוֹן זָכָר. וְהָרַ"ב הֶעְתִּיק עַד שֶׁתֵּצֵא, וּמוּסָב עַל הָעוֹלָה שֶׁהִיא לְשׁוֹן נְקֵבָה, כְּדִכְתִיב וְנִתַּח אוֹתָהּ לִנְתָחֶיהָ : הָרַמְבַּ"ם. וּלְעֵיל מִשְׁנָה ב' כָּתַב עַד שֶׁתִּנָּטֵל מֵאֵפֶר. וְכֵן פֵּרֵשׁ רַשִׁ"י כוּ'. וּפְלֻגְתָּא דְרַב וְרַב יוֹחָנָן הִיא. וְהָרַמְבַּ"ם מְפָרֵשׁ כְּרַב יוֹחָנָן, וְרַשִׁ"י מְפָרֵשׁ כְּרַב, וְהָרַ"ב תָּפַס שְׁנֵיהֶן וְלֹא דָק:

פרק ב - משנה ו

רַשִׁ"י. אֲבָל הַתּוֹסָפוֹת כָּתְבוּ דְשַׁפִּיר גָּרְסִינַן דְּמִדְּקָתָנֵי קָרְמוּ בַתַּנּוּר הֻכְשְׁרוּ לִפָּסֵל כוּ', אַלְּמָא דְאִית לֵיהּ דְּתַנּוּר מְקַדֵּשׁ, אִם כֵּן עַל כָּרְחֲךָ לִינָה פָסְלָה וְנֶאֱפוֹת בְּשַׁבָּת. וְהָא דִתְנַן בְּסוֹף פֶּרֶק י"א דִּמְנָחוֹת דְּנֶאֱפוֹת מֵעֶרֶב יוֹם טוֹב, אִית לֵיהּ דְּתַנּוּר אֵינוֹ מְקַדֵּשׁ. עַיֵּן תּוֹסְפוֹת יוֹם טוֹב: הַזָּבַח. דִּשְׁחִיטַת הַכְּבָשִׂים מְקַדַּשְׁתָּן וּמְפַגַּלְתָּן, כִּדְאִיתָא בְּמִשְׁנָה ב' פֶּרֶק ב' דִּמְנָחוֹת: וְאֵין כוּ'. פֵּרְשׁוּ הַתּוֹסָפוֹת דְּכֵיוָן שֶׁאֵינָן גּוּף הַזֶּבַח לֵיכָּא לְפָרוֹשֵׁי כְּמוֹ בְאִינָךְ דְּאֵין מוֹעֲלִין בָּהֶן, הָא אִסּוּרָא אִיכָּא, וּבַיּוֹצֵא וְאַלִּיבָּא דְּרַבִּי עֲקִיבָא. וְכֵן צָרִיךְ לוֹמַר בְּלֶחֶם הַפָּנִים:

פרק ב - משנה ז

פִּגּוּל. שֶׁאִם חָשַׁב בִּשְׁעַת הַקְטָרַת הַבְּזִיכִין הֲרֵינִי מַקְטִיר עַל מְנָת לֶאֱכֹל חוּץ לִזְמַנּוֹ, הֻקְבַּע הַלֶּחֶם בְּפִגּוּל, דְּכִי הֵיכִי דִזְרִיקַת דָּם בִּבְהֵמָה כוּ':

פרק ב - משנה ט

בַּכְּלִי. לֹא נָקַט בַּכְּלִי מִשּׁוּם קֹמֶץ, דְּמֵעִקָּרָא נַמִּי קֹדֶם שֶׁנָּתַן הַקֹּמֶץ בַּכְּלִי לְקַדְּשׁוֹ, הֻכְשַׁר לִפָּסֵל בִּטְבוּל יוֹם מִשְּׁקָּדְשׁוּ בִּכְלִי רִאשׁוֹן עִם כָּל הַמִּנְחָה כֻלָּהּ קֹדֶם שֶׁקָּמַץ. אֶלָּא אַגַּב הָנָךְ דִּלְעֵיל נָקַט לֵיהּ. תּוֹסָפוֹת: לֹא לַמִּזְבֵּחַ בִּלְבַד, דְּפָרִים הַנִּשְׂרָפִים עַד שֶׁיִּתֹּךְ הַבָּשָׂר בְּבֵית הַדֶּשֶׁן מוֹעֲלִין בָּהֶן. וְאִם כֵּן נִתָּר בְּשָׂרָן וְעוֹרָן לְבֵית הַדֶּשֶׁן נַמִּי. וּלְשׁוֹן רַשִׁ"י עוֹלַת הָעוֹף וְעוֹלַת הַבְּהֵמָה שֶׁדָּמָן כוּ': וְכִי יֵשׁ נוֹגֵעַ שֶׁהוּא חַיָּב. אֶלָּא בְכָל הַקֳּדָשִׁים הָרְאוּיִם לִקָּרֵב. וְהוּא מִשֶּׁקָּדַשׁ בִּכְלִי, שֶׁאָכַל בְּטֻמְאָה הוּא מְדַבֵּר. גְּמָרָא:
