פרק ב - משנה א

אַרְבָּעָה. מִנְיָנָא לַאֲפוֹקֵי מִדְּרַבִּי אֱלִיעֶזֶר בֶּן יַעֲקֹב: וְאַרְבָּעָה. מִנְיָנָא לַאֲפוֹקֵי מִדְּרַבִּי שִׁמְעוֹן, דְּתָנֵי רַבִּי שִׁמְעוֹן אוֹמֵר שְׁבוּעַת הַפִּקָּדוֹן לֹא נִתָּן זְדוֹנָהּ לְכַפָּרָה. גְּמָרָא: וְאִלּוּ בִּשְׁאָר דְּבָרִים חֲלוּקִים הֵם כְּדִתְנַן סוֹף פֶּרֶק ג' דְּסוֹטָה. גְּמָרָא: דְּגֵר צָרִיךְ מִילָה וּטְבִילָה וְקָרְבָּן, וְכֻלְּהוּ יָלְפִינַן לְהוּ מִקְּרָאֵי בַּגְּמָרָא: וּבַגְּמָרָא יָלֵיף לֵיהּ דְּגַם בְּעוֹף סַגִּי לֵיהּ, וְהַיְנוּ שְׁתֵּי פְּרִידוֹת וּשְׁתֵּיהֶן עוֹלוֹת:

פרק ב - משנה ב

גְּמָרָא. וְהָא קְרָא כִּי כְּתִיב בְּמֵזִיד כְּתִיב , אֶלָּא אֵימָא לַעֲשׂוֹת שׁוֹגֵג כַּמֵּזִיד: כְּלוֹמַר כִּדְאָמְרִינַן בְּעָלְמָא אֵין עוֹנְשִׁין אֶלָּא אִם כֵּן מַזְהִירִין. דְּהַיְנוּ עֹנֶשׁ בֵּית דִּין שֶׁל מַטָּה. דְּוַדַּאי דְּאַף עַל הַשּׁוֹגֵג מִתְיַסֵּר בְּיִסּוּרִין מִבֵּית דִּין שֶׁל מַעְלָה, דְּהָא צָרִיךְ חַטָּאת לְכַפָּרָה, וְעַל לֹא הוֹדַע אָשָׁם תָּלוּי שֶׁתּוֹלֶה וּמֵגֵן מִן הַיִּסּוּרִים. וּבַגְּמָרָא מֵבִיא קְרָאֵי דְּפִתְאֹם מַשְׁמַע נַמִּי שׁוֹגֵג וְאֹנֶס, וּפָרֵיךְ לִכְתֹּב פִּתְאֹם דְּמַשְׁמַע כֻּלְּהוּ וְלֹא לִכְתֹּב פֶּתַע. וּמְשָׁנֵי דַּהֲוָה אֲמֵינָא דְּהָכָא מַשְׁמָעוּתוֹ דַּוְקָא שׁוֹגֵג וְלֹא מֵזִיד מִידִי דַּהֲוָה אַכָּל הַתּוֹרָה דְּאֵין קָרְבָּן אַמֵּזִיד, לְהָכִי כָּתַב פֶּתַע דְּשׁוֹגֵג הוּא לְגַלּוֹיֵי דְּפִתְאֹם אֹנֶס וּמֵזִיד הוּא: וְתֵימַהּ דִּבְרֵישׁ פֶּרֶק שְׁבוּעַת הַפִּקָּדוֹן פֵּרֵשׁ הָרַ"ב מִדְּלֹא כְּתִיב בָּהּ וְנֶעְלָם, וְכֵן הוּא בַּגְּמָרָא דִּשְׁבוּעוֹת לְהֶדְיָא. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב שֶׁהֶאֱרִיךְ:

פרק ב - משנה ג

חֲמִשָּׁה. מִנְיָנָא לַאֲפוֹקֵי רַבִּי דְּסָבַר וְקִדַּשׁ אֶת רֹאשׁוֹ בַּיּוֹם הַהוּא, בַּיּוֹם הַשְּׁמִינִי, וְנָזִיר שֶׁנִּטְמָא בִּשְׁבִיעִי טֻמְאָה אֲרִיכְתָּא הוּא כְּאִלּוּ נִטְמָא בְּתוֹךְ הַשִּׁבְעָה: וַחֲמִשָּׁה. מִנְיָנָא לַאֲפוֹקֵי מִדְּרַבִּי אֱלִיעֶזֶר דְּאָמַר נָשִׂיא מֵבִיא שָׂעִיר בְּטֻמְאַת מִקְדָּשׁ וְקָדָשָׁיו. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: בְּזָדוֹן אוֹ בְּשׁוֹגֵג וּבִידִיעָה בֵּינְתַיִם, דְּאִי לָאו הָכִי אֲפִלּוּ בִּשְׁאָר אִסּוּרִים אֵינוֹ חַיָּב אֶלָּא אַחַת כְּדִלְקַמָּן פֶּרֶק ג' מִשְׁנָה ב': פֵּרוּשׁ לְאַחַר שֶׁגִּלַּח וְטָבַל לִחַיַּב שְׁתֵּי קָרְבָּנוֹת, שֶׁכֵּיוָן שֶׁלֹּא הֵבִיא קָרְבְּנוֹתָיו וְהוּא כְּבָר נִטְהַר רָאוּי לִכְשֶׁיַּחֲזֹר וְיִתְטַמֵּא שֶׁיָּבִיא קָרְבָּן עַל כָּל טֻמְאָה וְטֻמְאָה, דְּטֻמְאָה אַחֲרִיתִי הִיא הוֹאִיל וְטָבַל בּוֹ בַּיּוֹם: וְחַד מִיהָא חַיָּב מִשּׁוּם טֻמְאָה זוֹ. רַשִׁ"י בְּנָזִיר. וְאֵין נִרְאֶה כֵּן מִדְּבָרָיו בִּמְכִילְתִין רֵישׁ פֶּרֶק קַמָּא. אֲבָל כְּשֶׁנִּטְמָא טֻמְאוֹת הַרְבֵּה בַּיּוֹם הַשְּׁמִינִי, שֶׁהוּא רָאוּי לְהָבִיא קָרְבְּנוֹתָיו, חַיָּב קָרְבָּן עַל כָּל טֻמְאָה וְטֻמְאָה כָּל זְמַן שֶׁלֹּא הֵבִיא קָרְבְּנוֹתָיו: הַמְקַנֵּא כוּ'. מֵבִיא מִנְחַת שְׂעֹרִים אַחַת לְכֻלָּן כְּשֶׁהוּא מַשְׁקָהּ. רַשִׁ"י: וְאִיכָּא לְמֵידַק אִי הָכִי נַמִּי כְּשֶׁהָיָה רָאוּי לְהָבִיא קָרְבְּנוֹתָיו אִי מִחַיַּב קָרְבָּן עַל כָּל נֶגַע וְנֶגַע כְּמוֹ בַּנָּזִיר. וְנִרְאֶה דְּשָׁוִין הֵן. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: צִפֳּרָיו. אוֹתָן שְׁתֵּי צִפֳּרִים חַיּוֹת אַחַת לַשְּׁחִיטָה וְאַחַת לַשִּׁלּוּחַ. וְלָאו קָרְבָּן נִינְהוּ. רַשִׁ"י: וְכֵן כָּתַב הָרַמְבַּ"ם. וְתֵימַהּ דִּבְפֶרֶק י"ד דִּנְגָעִים מִשְׁנָה י"א פָּסְקוּ כְּרַבִּי יְהוּדָה כוּ', וּבְנֻסַּח אֶרֶץ יִשְׂרָאֵל הֻגָּה בְּכָאן, וְהַנִּרְאֶה לִי שֶׁהֲלָכָה כְּרַבִּי יְהוּדָה. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב:

פרק ב - משנה ד

וְכֵן כָּתַב הָרַמְבַּ"ם, דְּאַטּוּ בְּרַשִּׁיעֵי עָסְקִינַן: רְצוֹנוֹ לוֹמַר מַה שֶּׁאִמּוֹ טְמֵאָה לַמִּקְדָּשׁ וְקָדָשָׁיו. הָרַמְבַּ"ם: בִּנְקֵבָה. כָּל חַטַּאת יָחִיד נְקֵבָה הִיא דְּהַיְנוּ כִּשְׂבָּה אוֹ שְׂעִירָה. רַשִׁ"י: וְלֹא אֵת הָאִשָּׁה לָאִישׁ. אֵימָא הוּא מַעֲטֵיהּ קְרָא אֲבָל הִיא תִּלְקֵי וְתֵיְתֵי קָרְבָּן. וְהֵבִיא אֶת אֲשָׁמוֹ לַה' כְּתִיב, גְּמָרָא: וְהָא דִּכְתִיב לְזָרַע בְּאֵשֶׁת אִישׁ וְכֵן בְּסוֹטָה, מְתָרֵץ בַּגְּמָרָא דְּאֵשֶׁת אִישׁ פְּרָט לִמְשַׁמֵּשׁ בְּאֵבָר מֵת, אִי נַמִּי פְּרָט לִמְשַׁמֵּשׁ מֵתָה, סָלְקָא דַעְתָּךְ אֲמֵינָא לְאַחַר מִיתָה נַמִּי אִקְרֵי שְׁאֵרוֹ וְלִחַיַּב, קָא מַשְׁמַע לָן. דְּסוֹטָה פְּרָט לְשֶׁקִּנֵּא לַהּ בִּנְשִׁיקָה אִי נַמִּי דֶּרֶךְ אֵבָרִים, סָלְקָא דַעְתָּךְ אֲמֵינָא בִּקְפִידָא דְּבַעַל תָּלָה רַחֲמָנָא וְהָא קָא קָפֵיד, קָא מַשְׁמַע לָן:

פרק ב - משנה ה

וְאִי אֶפְשָׁר לוֹמַר שֶׁבּוֹעֵל שִׁפְחָה מְשֻׁחְרֶרֶת פְּנוּיָה שֶׁיּוּמַת , אֶלָּא לְפִי שֶׁהִיא מְקֻדֶּשֶׁת וְיֵשׁ בָּהּ צַד עַבְדוּת. שֶׁאִלּוּ נִגְמַר פִּדְיוֹנָהּ הָיְתָה אֵשֶׁת אִישׁ וְנֶהֱרָגִים שְׁנֵיהֶם. הָרַמְבַּ"ם: וַדָּאִית. וּמְאֹרֶסֶת לְעֶבֶד עִבְרִי. דִּכְתִיב כִּי לֹא חֻפָּשָׁה, מִכְּלָל דְּהוּא חֻפַּשׁ. גְּמָרָא:

פרק ב - משנה ו

וְכָל שֶׁכֵּן אִם הוּא מֵזִיד, דְּהָא קְרָא כִּי כְּתִיב בַּמֵּזִיד כְּתִיב כְּמוֹ שֶׁכָּתַבְתִּי בְמִשְׁנָה ב':
