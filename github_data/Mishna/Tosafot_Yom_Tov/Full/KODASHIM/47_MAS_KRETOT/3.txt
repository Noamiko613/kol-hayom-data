פרק ג - משנה א

כְּדִתְנַן רֵישׁ פֶּרֶק י' דִּיבָמוֹת. הָאִשָּׁה כוּ' וּבָאוּ וְאָמְרוּ כוּ' דְּקַיְמָא לָן אֲפִלּוּ בְּחַד. גְּמָרָא: דְּתָנוּ רַבָּנָן אוֹ הוֹדַע אֵלָיו חַטָּאתוֹ, וְלֹא שֶׁיּוֹדִיעוּהוּ אֲחֵרִים. יָכוֹל אֲפִלּוּ אֵינוֹ מַכְחִישָׁן, תַּלְמוּד לוֹמַר אוֹ הוֹדַע אֵלָיו, מִכָּל מָקוֹם. בְּמַאי עָסְקִינַן, אִילֵימָא בִּתְרֵין, תְּרֵין וְאֵין מַכְחִישָׁן קְרָא בָּעֵי, אֶלָּא לָאו בְּחַד, וְקָתָנֵי כִּי אֵין מַכְחִישָׁן מְהֵימָן, שְׁמַע מִנַּהּ. גְּמָרָא. וְכָתְבוּ הַתּוֹסָפוֹת דְּהָא דְּאִי מַכְחִישׁוֹ לֹא מְהֵימָן הָעֵד עַל כָּרְחָךָ רְצוֹנוֹ לוֹמַר דְּאוֹמֵר אֵינִי יוֹדֵעַ, דְּאִי אוֹמֵר בְּהֶדְיָא לֹא אָכַלְתִּי פְּשִׁיטָא דְּפָטוּר, דְּעֵד אֶחָד אֵינוֹ קָם אֶלָּא לִשְׁבוּעָה: לֹא כוּ'. צָרִיךְ לוֹמַר דְּמַיְרֵי שֶׁהֵעִידוּ שְׁנֵיהֶם בְּבַת אַחַת, דְּאִי בָּזֶה אַחַר זֶה הָא כָּל מָקוֹם שֶׁהֶאֱמִינָה תּוֹרָה עֵד אֶחָד הֲרֵי כָּאן שְׁנַיִם כוּ': פָּטוּר. וְהוּא הַדִּין אִם אָמַר אֵינִי יוֹדֵעַ כְּמוֹ שֶׁפֵּרַשְׁנוּ. תּוֹסָפוֹת: מְחַיֵּב. וְאִם תֹּאמַר הַאֵיךְ מְחַיֵּב, וְהָאָמַר לֹא אָכַלְתִּי, וּבְעַל כָּרְחֵיהּ לֹא יָבִיא חַטָּאת, כִּדְאָמְרִינַן חַיָּבֵי חַטָּאת אֵין מְמַשְׁכְּנִין אוֹתָן. וְיֵשׁ לוֹמַר דְּהָתָם מַיְרֵי כְּשֶׁאָמַר דְּלֹא נִיחָא לֵיהּ בְּכַפָּרָה. אֲבָל הַאי נִיחָא לֵיהּ בְּכַפָּרָה אֶלָּא שֶׁהוּא סָבוּר לִהְיוֹת פָּטוּר, מְמַשְׁכְּנִין אוֹתוֹ. תּוֹסָפוֹת: מָה כוּ'. הַשְׁתָּא נַמִּי מַה שֶּׁאָמַר לֹא אָכַלְתִּי, כָּךְ רְצוֹנוֹ לוֹמַר, לֹא אָכַלְתִּי שׁוֹגֵג אֶלָּא מֵזִיד. וְהָכִי מְתָרְצִינַן מִלְּתֵיהּ דִּבְהָכִי הֲוָה דַּעְתֵּיהּ, וְאַף עַל גַּב דְּלֹא מְשַׁוֵּי אֱינָשׁ נַפְשֵׁיהּ רְשִׁיעָא כוּ'. אֲבָל אִי אָמַר מֵזִיד אָכַלְתִּי חֵלֶב מְהֵימָן, דְּאוֹתוֹ מִתְכַּוֵּן לְטוֹב, שֶׁאֵינוֹ רוֹצֶה לְהָבִיא חֻלִּין לָעֲזָרָה. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב:

פרק ג - משנה ב

וְחָזַר כוּ'. וְלֹא הָיְתָה לוֹ יְדִיעָה בֵּינְתַיִם, דְּיֵשׁ יְדִיעָה לַחֲצִי שִׁעוּר כַּחֲכָמִים סוֹף פֶּרֶק י"ב דְּשַׁבָּת:

פרק ג - משנה ג

וּפְלֻגְתָּא דְּתַנָּאֵי הוּא בְּמִשְׁנָה ב' פֶּרֶק ח' דְּעֵרוּבִין, וְשָׁם פָּסַק הָרַ"ב כְּהָרַמְבַּ"ם: לֹא דָּק, דְּאֵין בּוֹ אֶלָּא חִיּוּב מִיתָה, וּכְמוֹ שֶׁכָּתַב בְּעַצְמוֹ סוֹף פֶּרֶק ט' דְּסַנְהֶדְרִין: וְנִכְנַס כוּ'. פֵּרוּשׁ וְעָבַד עֲבוֹדָה. אֲבָל בִּכְנִיסָה בְּלֹא עֲבוֹדָה לֹא מִחַיַּב. וְהָכִי פֵּרְשׁוּ בְּתוֹרַת כֹּהֲנִים. תּוֹסָפוֹת כוּ'. וּמִכָּל מָקוֹם לְהָרַמְבַּ"ם יֵשׁ אִסּוּר אַף בְּלֹא עֲבוֹדָה, וְאַף מִן הַמִּזְבֵּחַ וְלִפְנִים יֵשׁ אִסּוּר:

פרק ג - משנה ד

כְּגוֹן שֶׁאָכַל כּוּלְיָא בְּחֶלְבָּהּ. אִי נַמִּי דְּמַלְּיֵהּ בְּתַמְרֵי, דְּלֹא מִחַיַּב בְּיוֹם הַכִּפּוּרִים אֶלָּא עַל כַּכּוֹתֶבֶת. גְּמָרָא. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: פֵּרוּשׁ בְּחֵלֶב זֶה דְּמֻקְדָּשִׁים: וְאִי נַמִּי מֻקְדָּשִׁים דְּיֵשׁ שֶׁהֵן לַכֹּהֲנִים אִינְהוּ נַמִּי כִּי קָא זָכוּ מִשֻּׁלְחָן גָּבוֹהַּ קָא זָכוּ. וּמִיהוּ הָכָא בְּחֵלֶב דְּאֵינוֹ אֶלָּא לְגָבוֹהַּ: דְּאִי לָאו הָכִי אֵין דֶּרֶךְ הוֹצָאָה בְּכָךְ, כְּדִתְנַן בְּמִשְׁנָה ג' פֶּרֶק י' דְּשַׁבָּת. אֲבָל כְּשֶׁאוֹכְלוֹ כָּךְ כֵּיוָן דְּקָא מְכַוֵּן , מַחֲשַׁבְתּוֹ מְשַׁוְּיָא לֵיהּ מָקוֹם. גְּמָרָא. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: וְכָתַב רַשִׁ"י עֵרוּב כְּדִי נַסְבַהּ. אֶלָּא דְּהָא בְּהָא תַּלְיָא, דְּלֹא תִּקְּנוּ חֲכָמִים עֵרוּבֵי חֲצֵרוֹת אֶלָּא בִּשְׁבִיל אִסּוּר הוֹצָאָה. שֶׁגָּזְרוּ מֵרְשׁוּת לִרְשׁוּת, גְּזֵרָה מִשּׁוּם הוֹצָאָה מֵרְשׁוּת הַיָּחִיד לִרְשׁוּת הָרַבִּים. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב:

פרק ג - משנה ה

וְהוּא הַדִּין אִם גֵּרְשָׁהּ כְּדִלְקַמָּן, וַחֲדָא מִתַּרְתֵּי נָסֵיב: וְהַתּוֹסָפוֹת כָּתְבוּ וְאִם תֹּאמַר וְהָא מַמְזֶרֶת הִיא וְאִם כֵּן הֵיאַךְ אֲסוּרָה לְאָחִיו מִשּׁוּם אֵשֶׁת אָח, אֵיךְ חָל אִסּוּר אֵשֶׁת אָח עַל אִסּוּר מַמְזֵרוּת. וְיֵשׁ לוֹמַר דְּמַיְרֵי דְּאָחִיו נַמִּי מַמְזֵר כוּ', וְעוֹד יֵשׁ לוֹמַר אִסּוּר חָמוּר עַל אִסּוּר קַל חָל. וְתֵרוּץ בַּתְרָא אֵין נִרְאֶה מִדִּבְרֵי הָרַ"ב, וְאִם כֵּן לְפֵרוּשׁ קַמָּא קָשָׁה קֻשְׁיַת הַתּוֹסָפוֹת. וְיֵשׁ לוֹמַר דְּמַיְרֵי שֶׁיֵּשׁ כָּאן אָח אֶחָד שֶׁהוּא מַמְזֵר שֶׁהָיְתָה מֻתֶּרֶת לוֹ וְעַכְשָׁיו כְּשֶׁנְּשָׂאָהּ זֶה נֶאֶסְרָה עַל אוֹתוֹ הָאָח מִשּׁוּם אֵשֶׁת אָח, וּבְמִגּוֹ נֶאֱסַר גַּם כֵּן עַל זֶה. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: כְּלוֹמַר שֶׁמֵּת זֶה וְנִשֵּׂאת לְאַחֵר. וְכֵן כָּתַב הָרַמְבַּ"ם בְּהֶדְיָא, וְכֵן כָּתַב הָרַ"ב בְּהֶדְיָא בְּמַתְנִיתִין דִּלְקַמָּן. וּבְחִנָּם נִדְחֲקוּ לְפָרֵשׁ כֵּן, דְּהָא אִסּוּר אֵשֶׁת אִישׁ חָיֵיל בְּבַת אַחַת עִם אִסּוּר אֵשֶׁת אֲחִי אָבִיו כְּמוֹ בִּתּוֹ וַאֲחוֹתוֹ. וְכֵן לְקַמָּן בְּאִם עָבַר הַזָּקֵן פֵּרֵשׁ הָרַמְבַּ"ם דְּאִסּוּר אֵשֶׁת אָב וְאִסּוּר אֵשֶׁת אִישׁ הֵם שְׁנֵי אִסּוּרִים הַבָּאִים כְּאֶחָד: כְּגוֹן שֶׁהָיְתָה לוֹ בַּת מֵאִשָּׁה אַחֶרֶת וּנְשָׂאָהּ זֶה הָאָב חָמִיו שֶׁל בַּעַל בִּתּוֹ, וּכְשֶׁבָּא זֶה עַל אֲחוֹת בִּתּוֹ שֶׁל חֹתְנוֹ שֶׁהִיא לוֹ בַּת בִּתּוֹ, הָיְתָה אֲחוֹת אִשְׁתּוֹ. רַשִׁ"י. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב כַּמָּה מְצִיאוּת שֶׁיְּהֵא חַיָּב יוֹתֵר, וְהַתַּנָּא לֹא חַשׁ לְמִתְנִינְהוּ דְּמִמֵּילָא מַשְׁמַע כְּשֶׁתַּעֲמֹד לְמִנְיָן: עַיֵּן תּוֹסְפוֹת יוֹם טוֹב עוֹד מְצִיאוּת אַחֵר עַל זֶה, וּמַסִּיק בְּשֵׁם הָרַמְבַּ"ם דְּכָל אֵלּוּ שֶׁאִסּוּרָן מִשּׁוּם אִסּוּר מוֹסִיף צָרִיךְ שֶׁיִּהְיוּ הָאֲנָשִׁים הַנֶּאֱסָרִים בָּעוֹלָם, אֲבָל אִם אֵינָם בָּעוֹלָם אֵין אוֹמְרִים הוֹאִיל אִם הָיוּ כוּ':

פרק ג - משנה ז

לְשׁוֹן הָרַמְבַּ"ם וּבָא אוֹתוֹ הַבֵּן עַל הַבַּת הָאַחֶרֶת, הִיא אֲחוֹתוֹ מֵאָבִיו, וַאֲחוֹת אִמּוֹ מִן הָאָב וּמִן הָאֵם וַאֲחוֹת אָבִיו מֵאִמּוֹ:

פרק ג - משנה ח

וְהוּא לָשׁוֹן עִבְרִי, כְּקוֹל הַסִּירִים , וְכֵן מִבּוֹר הַסִּירָה כְּמִדְרַשׁ חֲכָמֵינוּ זַ"ל : קַל וָחֹמֶר. וּבְפֶרֶק ט' דְּחֻלִּין מִשְׁנָה ז' יָלְפִינַן לַהּ מִקְּרָא. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב:

פרק ג - משנה ט

וּבְמִשְׁנָה ב' דְּפֵרֵשׁ הָרַ"ב דְּאֵין תַּמְחוּיִין מְחַלְּקִין, אָתְיָא נַמִּי כְּרַבִּי יְהוֹשֻׁעַ, דְּהָכָא דְּאָמַר מְחַלְּקִין לְחֻמְרָא הוּא דְּאָמַר, אֲבָל הָתָם לְקֻלָּא לֹא אָמַר. גְּמָרָא. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: כַּנֶּהֱנֶה. וְהוּא הַדִּין דְּמָצֵי לְמֵימַר הַנֶּהֱנֶה כָּאוֹכֵל, מַה שֶּׁאֵין כֵּן בְּנוֹתָר. אֶלָּא אַגַּב דְּנָקֵיט מַאֲכִיל כָּאוֹכֵל קָאָמַר מְהַנֶּה כַּנֶּהֱנֶה. תּוֹסָפוֹת: בְּהֶעְלֵם אֶחָד. עַיֵּן לְקַמָּן בְּסוֹף פֶּרֶק ה' דִּמְעִילָה וּבְמַה שֶּׁכָּתַב הָרַ"ב שָׁם. תּוֹסָפוֹת חֲדָשִׁים:

פרק ג - משנה י

וְאַף עַל גַּב דְּמִנִּדָּה לֹא מָצֵי לְמִפְשַׁט לֵיהּ אֶלָּא חֲדָא. דְּהָא בְּנִדָּה לֵיכָּא תּוֹלָדוֹת. רַשִׁ"י: וְקַשְׁיָא לִי דְּפָסוּק זֶה עַל הָעֹנֶשׁ נֶאֱמַר וַעֲדַיִן אַזְהָרָה לֹא שָׁמַעְנוּ, וּכְדִלְקַמָּן בִּבְהֵמָה, וְרַשִׁ"י בַּחֻמָּשׁ פֵּרֵשׁ בַּפָּסוּק לֹא תִקְרְבוּ לְגַלּוֹת עֶרְוָה , לְהַזְהִיר הַנְּקֵבָה כַּזָּכָר, לְכָךְ נֶאֱמַר לְשׁוֹן רַבִּים: הַבְּהֵמָה כוּ'. שֶׁאַף עַל פִּי שֶׁהִיא נֶעֱנֶשֶׁת, כְּדִכְתִיב וְאֶת הַבְּהֵמָה תַּהֲרֹגוּ, אֵינָהּ מֻזְהֶרֶת:
