פרק ו - משנה א

שִׁשִּׁים יוֹם. רְצוּפִים בַּבֹּקֶר וּבָעֶרֶב. הָרַמְבַּ"ם.: וּמַשְׁכַּחַת לַהּ נַמִּי בְּשׁוֹר הַמּוּעָד כְּדִמְפֹרָשׁ בְּמִשְׁנָה ד' פֶּרֶק קַמָּא דְּבָבָא קַמָּא. וְעַיֵּן שָׁם: וְאִם תֹּאמַר כֵּיוָן שֶׁרִבִּית אוֹכֶלֶת בָּהֶן לֹא יַכְרִיזוּ, דְּשֶׁמָּא יַפְסִידוּ בְהַאֲרָכַת הַזְּמַן יוֹתֵר מִמַּה שֶּׁיַּרְוִיחוּ. וְכִי תֵימָא כְגוֹן שֶׁקִּבֵּל עָלָיו לָדוּן בְּדִין יִשְׂרָאֵל, אִי הָכִי רִבִּית נַמִּי לֹא לִשְׁקוֹל. תֵּרְצוּ בַּגְּמָרָא שֶׁקִּבֵּל לְזוֹ וְלֹא קִבֵּל לְזוֹ. וּפֵרֵשׁ הָרַ"ן שֶׁקִּבֵּל עָלָיו לְזוֹ לְהַמְתִּין יְמֵי הַהַכְרָזָה וּבְלֹא רִבִּית, וְלֹא קִבֵּל עָלָיו שֶׁאִם לֹא יִרְצוּ לְהַגְבּוֹתוֹ שֶׁלֹּא יַעֲלֶה לוֹ חוֹבוֹ רִבִּית: וְאִם תֹּאמַר בְּסוֹף פֶּרֶק ד' דִּכְתֻבּוֹת פָּסַק דְּלֹא הִפְסִידָה אֶלָּא אִם כֵּן שֶׁתָּבְעָה, וּמִכֵּיוָן שֶׁתָּבְעָה הֲרֵי הִפְסִידָה וְתוּ לֹא נִזְקַק לָהּ. תֵּרְצוּ בַגְּמָרָא, כֵּיוָן דְּאִיזְקַקְנוּ לָהּ מֵעִקָּרָא בִּתְחִלַּת תְּבִיעַת כְּתֻבָּתָהּ לְהַפְסִידָהּ מְזוֹנוֹתֶיהָ, מִזְדַּקְּקִינַן לַהּ לְבַסּוֹף לְהַגְבּוֹתָהּ כְּתֻבָּתָהּ. וְכָתַב הָרַ"ן, כְּלוֹמַר שֶׁאִם אֵין אָנוּ נִזְקָקִים הֲרֵי הִיא נִמְנַעַת מִתְּחִלָּה לָבֹא, וְנִמְצֵאת מַפְסִדְתָּם מְזוֹנוֹת: וְכֵן כָּתַב הָרַמְבַּ"ם. וּתְמִיהָה טוּבָא, דְּמַה מָּקוֹם לִשְׁבוּעָה זוֹ, כֵּיוָן שֶׁעַל יְדֵי הוֹדָאַת הָאָב אוֹ מֵחֲמַת חֲזָקָה מַגְבִּין, מַה צָּרִיךְ לִשְׁבוּעָה. וְהָרַמְבַּ"ם בְּחִבּוּרוֹ כָתַב, שֶׁכָּל אֵלּוּ גוֹבִין בְּלֹא שְׁבוּעָה, וְלֹא חִלֵּק בֵּין מְקַרְקְעֵי לְמִטַּלְטְלִין. וְכֵן פָּסַק הַטּוּר: עֲבָדִים שֶׁמָּא יִשְׁמְעוּ וְיִבְרְחוּ. מִטַּלְטְלִין וּשְׁטָרוֹת, שֶׁמָּא יִגָּנְבוּ, פֵּרֵשׁ רַשִׁ"י, כְּשֶׁנֶּאֱסָפִין לִרְאוֹתָם כְּדֵי לְלָקְחָם: יַדִּיר כוּ'. תֵּימַהּ, וְכִי מַה הוּא עוֹשֶׂה וַהֲלֹא אֵינוֹ רוֹצֶה שֶׁתִּגְבֶּה מִן הַהֶקְדֵּשׁ, וְאַדְּרַבָּה לָאִשָּׁה הֲוָה לָן לְקוֹנְסָהּ שֶׁגּוֹבָה מִן הַהֶקְדֵּשׁ. וְיֵשׁ לוֹמַר כֵּיוָן דְּחָיְשִׁינַן שֶׁמָּא עָשָׂה קְנוּנְיָא עַל הַהֶקְדֵּשׁ אוֹ עַל הֶעָרֵב שֶׁהִכְנִיס בַּעֲרֵבוּת כוּ', לְדִידָהּ לֹא קָנְסוּהָ רַבָּנָן, וְלוֹ רָאוּי לִדּוֹר כְּדֵי לֵידַע אִם כַּוָּנָתוֹ לְהַעֲרִים וְלֹא תוֹעִיל לוֹ כַוָּנָתוֹ שֶׁהִדִּירוּהוּ. תּוֹסָפוֹת: וְהֵן הֵן דִּבְרֵי בֵית שַׁמַּאי וּבֵית הִלֵּל דְּרֵישׁ פֶּרֶק ה' דְּנָזִיר. גְּמָרָא: אַף. אֲבָל הָרַ"ב לֹא גָּרַס לֵיהּ, וּלְכָךְ פָּסַק כְּרַבִּי יְהוֹשֻׁעַ דַּאֲתָא כְבֵית הִלֵּל. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב שֶׁהֶאֱרִיךְ:

פרק ו - משנה ב

וְטַעֲמָא דִבְדָבָר מֻעָט סַגִּי, מִשּׁוּם דְּאֵין אוֹנָאָה לְהֶקְדֵּשׁוֹת. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: כֵּיוָן דְּאֵינָן קְדֻשַּׁת הַגּוּף אֶלָּא קְדֻשַּׁת דָּמִים, אֵינוֹ מַפְקִיעַ מִידֵי שִׁעְבּוּד: לִתֵּן כוּ'. אַחַר שֶׁיִּשָּׁבְעוּ, כְּדִין הַבָּא לִפָּרַע מִמְּשֻׁעְבָּדִים. הָרַמְבַּ"ם. וְהוּא גְמָרָא בִשְׁבוּעוֹת דַּף מ"ב, דְּחָיְשִׁינַן לִקְנוּנְיָא אַף בְּהֶקְדֵּשׁ. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: אֲבָל פִּי שְׁנַיִם קוֹנֶה. וּמִכָּאן יֵשׁ לְהִסְתַּיֵּעַ לַסְּבָרָא שֶׁכָּתַבְתִּי בְּמִשְׁנָה ה' פֶּרֶק י"א דִּכְתֻבּוֹת, דְּדַוְקָא בְיָתֵר מִפַּלְגָּא יֵשׁ אוֹנָאָה: מוֹסִיף. פֵּרֵשׁ רַשִׁ"י, בַּעַל חוֹב יוֹסִיף לְהַלְווֹתוֹ דִינָר וּפוֹדֶה הַנְּכָסִים הָאֵלּוּ. וְכָתְבוּ הַתּוֹסָפוֹת, מַשְׁמַע לְפֵרוּשׁוֹ שֶׁהַבַּעַל חוֹב עַצְמוֹ פּוֹדֶה. וְאֵין לְשׁוֹן הַמִּשְׁנָה מַשְׁמַע כֵּן, דְּקָתָנֵי עַל מְנָת לִתֵּן כוּ'. לָכֵן נִרְאֶה לְפָרֵשׁ, שֶׁהַלֹּוֶה יוֹסִיף לִלְווֹת עוֹד דִּינָר וּפוֹדֶה הַנְּכָסִים הָאֵלּוּ כְּדֵי לִתֵּן כוּ'. וְלִי נִרְאֶה דְגַם רַשִׁ"י הָכִי קָאָמַר, הַלֹּוֶה פּוֹדֶה כוּ', וּכְמוֹ שֶׁאִיתָא בָרֵישָׁא הַבַּעַל פּוֹדֶה, וְהָא דְכָתַב בַּעַל חוֹב, מִשּׁוּם דְּלֹא שְׁכִיחַ שֶׁיַּלְוֵהוּ אַחֵר שֶׁהֲרֵי כָּל נְכָסָיו הִקְדִּישׁ וְאֵין לוֹ עוֹד, וּלְכָךְ מְפָרֵשׁ דְּמוֹסִיף בְּבַעַל חוֹב אַף עַל גַּב דְּפוֹדֶה הַלֹּוֶה. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב:

לֹא כוּ'. אַף עַל פִּי שֶׁהוּא חַיָּב בִּמְזוֹנוֹתָן וּבִכְסוּתָן. הָרַמְבַּ"ם: חָרָשׁ כוּ'. נִרְאֶה לִי, דְּהָא דִתְנַן שְׁנֵי מַעֲצָדִין כוּ', לָאו דַּוְקָא אֵלּוּ בִלְחוֹד, אֶלָּא לְסִימָנָא בְעָלְמָא, וְכֵן לְכָל הַכֵּלִים שֶׁצָּרִיךְ לְאֻמָּנוּתוֹ זֹאת. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב:

פרק ו - משנה ד

וְלֹא דָמִי לְמָזוֹן וּכְסוּת דִּמְשַׁיְּרִינַן לוֹ מָעוֹת לִקְנוֹתָן, מִשּׁוּם דְּלֹא סַגִּי לֵיהּ בְּלָאו הָכִי: נְכָסָיו. תְּפִלִּין בִּכְלַל נְכָסָיו. וּמַעֲלִין לוֹ אֶת תְּפִלָּיו וּפוֹדֶה אוֹתָן. רַשִׁ"י. וּבְפֶרֶק ט' דְּבָבָא קַמָּא כָּתַב מַעֲלִין, שָׁמִין וְלֹוֶה הַמָּעוֹת וְיִתֵּן בִּשְׁבִילָם לַהֶקְדֵּשׁ. וְהָרַמְבַּ"ם בְּנֻסַּח אֶרֶץ יִשְׂרָאֵל כָּתַב, מַה שֶּׁאָנוּ מַעֲלִין לוֹ תְפִלָּיו רְצוֹנוֹ לוֹמַר שֶׁאֲפִלּוּ תְפִלָּיו אֵין מַנִּיחִין לוֹ אֶלָּא הַכֹּל הֶקְדֵּשׁ. מַה שֶּׁאֵין כֵּן בָּעֲרָכִין שֶׁנּוֹתְנִין לוֹ תְפִלָּיו. עַד כָּאן. וְנִרְאֶה לִי דִלְשׁוֹן מַעֲלִין לִדְבָרָיו לְשׁוֹן סִלּוּק. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב:

פרק ו - משנה ה

בַּצֶּבַע. לְשׁוֹן הָרַמְבַּ"ם, וְלֹא בְגָדִים שֶׁצְּבָעָן. וְיֵרָאֶה דִבְגָדִים צְבוּעִים קְרוּיִים צֶבַע בִּלְשׁוֹן הַמִּשְׁנָה. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: וְאַף עַל פִּי שֶׁלֹּא לְבָשׁוּם עֲדַיִן. רַשִׁ"י: מְקוֹמוֹ כוּ'. שֶׁהַמַּעֲרִיךְ עַצְמוֹ וּמָךְ הוּא נִדּוֹן בְּהֶשֵּׂג יָד וּלְפִי מַה שֶּׁיֵּשׁ לוֹ נוֹתֵן לַהֶקְדֵּשׁ, וְהָיָה לוֹ עֶבֶד כוּ', אֵין אוֹמְרִים רוֹאִין כְּאִלּוּ לוֹקְחִין לוֹ כְּסוּת כוּ' וְכָךְ יִתֵּן זֶה לַהֶקְדֵּשׁ, אֶלָּא לְפִי שְׁעָתוֹ שָׁמִין אוֹתוֹ. וְכֵן פָּרָה וְכֵן מַרְגָּלִית בַּכְּפָר, אֵין שָׁמִין אוֹתָן אֶלָּא לְפִי מְקוֹמָן כוּ' לְפִי מַה שֶּׁשָּׁוְיָן כָּאן שָׁמִין. וְזֶה מַשְׁהֶא אוֹתָהּ וְיִמְכְּרֶנָּה כַּמָּה שֶׁיִּרְצֶה. וּלְפִי דָמִים שֶׁשָּׁמְאוּהָ יִתֵּן, שֶׁהֲרֵי בִזְמַן שֶׁהֶעֱרִיכוֹ הַכֹּהֵן לֹא הֲוָה לֵיהּ יוֹתֵר. רַשִׁ"י. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: וּשְׁעָתוֹ. הָכָא מַיְרֵי בְמִטַּלְטְלִין דְּאֵין מַכְרִיזִין עָלָיהֶן שֶׁמָּא יִגָּנְבוּ. אֲבָל בְּרֵישׁ פִּרְקִין דְּמַצְרִיךְ הַכְרָזָה שִׁשִּׁים יוֹם, מַיְרֵי בְקַרְקָעוֹת. תּוֹסָפוֹת: אֶלָּא אִם כֵּן רָצָה שֶׁיְּהֵא הֶפְסֵד שֶׁלּוֹ וְשָׂכָר לַהֶקְדֵּשׁ, מֻתָּר:
