פרק ח

וֵאָף עָל פִּי דֵּלֲא פֹּסְיק לֵפֹרוֲשְִי אּלֹּא אָמִּן הּחֹדֹשִ כוּ', קֹאְי נָמִּי אָמִּן הֹאֹרּץ כוּ'. דֵּאִי מִן הֹאֹרּץ אָף בִּזֵבֹחִים, הַוֹה לְיהּ לֵמִתֵנְי חוּץ מִבֵּכוֲר וּמָעַשְֵר דֵּאְין בֹּאִין מִחוּץ לֹאֹרּץ כֵּדִתֵנָן בֵּסוֲף פּּרּק ג' דִּתֵמוּרֹה: וֵלֲא דֹק, דֵּבִשִֵתְּי הָלּּחּם הוּא דִּכֵתִיב כוּ'. וֵהֹיֹה נִרֵאּה דֵּעֲמּר יֹלֵפִינָן מִדִּקֹרְי לְיהּ קֵרֹא מִנֵחָת בִּכּוּרִים. אַבֹל בָּגֵּמֹרֹא מוּכֹח דֵּיֹלְיף לָהּ מִדִּכֵתִיב רְאשִִית קֵצִירֵכּם: הָמֶּבֵחֹר. שִּנּּאֱמָר וֵכֹל מִבֵחָר נִדֵרְיכּם. תּוֲסּפֵתֹּא: חַפֹרָיִם. סֵפֹרִים אַחְרִים עַפֹרָיִם. וֵכְן גִּרֵסָת הָתּוֲסֹפוֲת. וֵכֹתֵבוּ עִיר הִיא:

דֵּהֹא זוֲרֵעֹהּ קֲדּם לֵפּסָח שִִבֵעִים יוֲם, הָיֵנוּ בֹּעֲמּר. וּמִיהוּ הָתּוֲסֹפוֲת כֹּתֵבוּ דִּבֵכֹל הָמֵּנֹחוֲת קֹאֹמֵרִינָן. עָיְּן תּוֲסֵפוֲת יוֲם טוֲב: שִִבֵעִים יוֲם. שִּכֵּבֹר יְשִ כֲּחָ בָּחָמֹּה שִּזּוֲרָחָת עֹלּיהֹ. רָשִִ"י: וּבֹעְי בָּגֵּמֹרֹא, בֵּרֲב חִטֹּה אוֲ בֵּרֲב סֵאֹה, תְּיקוּ. וּפְרְשִ הֹרָמֵבָּ"ם, בֵּהִתֵלִיעֹה רֲב חִטֹּה אָחָת אָף עָל פִּי שִּהִיא מֵעֲרּבּת בִּסֵאֹה חִטִּים שִּלֲּא הִתֵלִיעוּ:

מִמָּה כוּ'. מִן הָזְּיתִים שִּנִּזֵרֵעוּ זֵרֹעִים בְּינְיהּם. לֵפִי שִּהָזֵּרֹעִים מָכֵחִישִִים לֹאִילֹנוֲת. רָשִִ"י: הָגָּרֵגֵּרִים. שִּל זְיתִים שִּנֹּפֵלוּ. רָשִִ"י: שִּנִּשִֵרוּ כוּ'. נִרֵאּה דֵּמִלֵּתֹא בֵּאָפְּי נָפֵשִָהּ הִיא, וֵלֲא קֹאְי אָגָּרֵגֵּרִים. וֵעָיְּן תּוֲסֵפוֲת יוֲם טוֲב: הָכֵּבוּשִִים כוּ'. וֵלֲא תֵּנָן מֵבֶשִֹּל. אַבֹל בֵּפְרוּשִ רָשִִ"י, אוֲ שִּנִּתֵבָּשִֵּלוּ אוֲ שִּנִּשִֵלֵקוּ:

וֵכֹתֵבוּ הָתּוֲסֹפוֲת, וֵקֹשִּה, דִּבֵמָסּּכּת פְּאֹה מָשִֵמָע דֵּחָיֹּבִין בֵּפְאֹה, וֵכֹל דֹּבֹר שִּאְין לֵקִיטֹתוֲ כֵּאּחֹד לֲא מִחָיָּב בֵּפְאֹה כִּדֵאִיתֹא שִֹם. וֵזּה לֵשִוֲן הֹרָמֵבָּ"ם, זָיִת שִּגִּרֵגֵּרוֲ בֵּרֲאשִ הָזָּיִת וּבְרֵרוּ אָחָת אָחָת וֵכֹתֵשִוּ כוּ'. זְיתִים שִּמֵּסֹקֹן כֶּלֹּן בֵּעִרֵבּוּבֵיֹא וֵהּעֱלֹן לָגָּג וֵחֹזָר וּבְרְר גָּרֵגְּר גָּרֵגְּר וֵכוֲתֵשִֹן כוּ'. זְיתִים שִּמֵּסֹקֹן וֵטוֲעַנֹן בֵּתוֲךֵ הָבָּיִת עָד שִּיִּלֵקוּ וֵהּעֱלֹן וֵנִגֵּבֹן וֵאָחָר כֹּךֵ כֹּתָשִ כוּ': כֵּלוֲמָר מֵלָקְּט זְיתִים שִּבֵּאּמֵצָע הֹאִילֹן שִּאְינֹן בֵּרֲאשִ הָזָּיִת, כֵּלוֲמָר אוֲתֹן שִּיֹּכוֲל לָעַמֲד בֵּרֲאשִ גָּגּוֲ וֵנוֲטֵלֹן, שִּאוֲתֹן אְין מִתֵבָּשִֵּלִין בִּמֵהְרֹה כֵּמוֲ אוֲתֹן שִּבֵּרֲאשִ הָזָּיִת. וֵהָאי דֵּאֹמָר בֵּרֲאשִ הָגָּג, לֵסִימֹנֹא קֹאֹמָר. רָשִִ"י: וֵהְם עַצֹמוֲת שִּיְּשִ בֹּהּם מֲחָ, דִּמֹּה אוֲתֹן לֵמָעַטֹן, לֵפִי שִּהְם רֵחֹבִים וּגֵדוֲלִים בָּחַלֹלֹם. וּמְהָשְִּם הָזּּה נִגֵזָר הָפֲּעָל וֵאוֲמְר עוֲטֵנוֲ. הֹרָמֵבָּ"ם:

מִדֵּתֹנְי רָבִּי חִיֹּא, יָיִן מִגִּתּוֲ לֲא יֹבִיא, וֵאִם הְבִיא כֹּשְִר. תּוֲסֹפוֲת: מֵעֶשִֹּן. כֵּשִּהֹיוּ הֹעַנֹבִים מֹרִים הֹיוּ מֵעָשִֵּנִין אוֲתֹן כֵּדְי לֵמָתֵּקֹן. אוֲ מֵבָשִֵּלִין אוֲתֹן. רָשִִ"י. וֵעָיְּן תּוֲסֵפוֲת יוֲם טוֲב:

נוֲדְף. יָעַמֲד רְיחוֲ, מִפֵּנְי שִּיָּעַלּה הּבֵלוֲ בָּמֹּקוֲם שִּהִנִּיחָ בֵּרֲאשִ הָכֵּלִי וֵיָעַמֲד רְיחוֲ וֵלֲא יִתֵעֹרְב בּוֲ שִוּם דֹּבֹר. הֹרָמֵבָּ"ם. וֵנוֲדְף לֹשִוֲן אַרֹמִית, וֵרְיחָ שִֵמֹנָיִךֵ מִכֹּל בֵּשֵֹמִים מֵתֶרֵגֹּם וֵשִוּם טָב דֵּצָדִּיקָיִךֵ נֹדְיף מִכֹּל בֶּשֵֵמִין: וֵנִרֵאּה לִי דֵּמְאּמֵצֹעֹהּ קֹאְי אָמִּשִֵּלִישִֹהּ, כֵּלוֲמָר מְאּמֵצָע הָשִֵּלִישִ הֹאּמֵצֹעִי כוּ'. וֵהֹכִי פְּרוּשִוֲ, מְאּמֵצֹעֹהּ שִּל שִֵלִישִֹהּ, וֵהָשִֵּלִישִֹהּ הִיא הֹאּמֵצֹעִית: תֵּמִימִים. וֵאִם הֹיוּ בּוֲ קֵמֹחִין הֹוְי כֵּבָעָל מוּם. רָשִִ"י:
