פרק י

הֹעֲמּר. הִיא שִּל שֵֵעֲרִין. וֵיְשִ לוֲ רֵמֹזִים בָּתּוֲרֹה, וּמִכֵּלֹלֹם שִּקֹּרֹא אוֲתֹהּ מִנֵחָת בִּכּוּרִים שִּנּּאֱמָר וֵאִם תָּקֵרִיב מִנֵחָת בִּכּוּרִים, וֵהָפְּרוּשִ הָמֵקֶבֹּל שִּיִּרֵמֲז לֵמִנֵחָת הֹעֲמּר. וּמִלָּת אִם אְינֹהּ רֵשִוּת אּלֹּא רֵמִיזָת הָזֵּמָן. וֵנִקֵרֹא שִֵתְּי הָלּּחּם בִּכּוּרִים, וּבְאְר זּה וֵאֹמָר בִּכּוּרְי קֵצִיר חִטִּים. וֵזּהוּ רֵאֹיֹה עָל שִּהָבִּכּוּרִים שִּקֹּדֵמוּ לֹהּן בֵּנִיסֹן אְינֹן אּלֹּא בִּכּוּרְי שֵֵעֲרִים. וֵהֹעִקֹּר שִּעֹלֹיו אֹנוּ סוֲמֵכִים הוּא קָבֹּלֹה. הֹרָמֵבָּ"ם:

גֵּמֹרֹא. וֵאִי בֹּעְית אְימֹא מִשִּוּם כָּרֵמּל תָּקֵרִיב. פְּרְשִ רָשִִ"י. רָךֵ וּמֹלְא. שִּתֵּהְא הָתֵּבוּאֹה רָכֹּה וֵנִמֵלּלּת בָּיֹּד, וֵאִי מָיֵתְי מְרֹחוֲק נֹשְִיב בְּיהּ זִיקֹא וּמִתֵקָשִֹּה בָּדּּרּךֵ:

אוֲמְר כוּ'. כֹּל אּחֹד מְהָשִֵּלֲשִֹה קוֲצֵרִים. אַבֹל הָמֵּשִִיבִים הְם רָבִּים מִכֹּל עַיֹרוֲת, הִלֵכֹּךֵ כָּלֵלִינֵהוּ וֵתֹנְי אוֲמֵרִים: קֵצֲר. בָּקֵּצִירֹה עָצֵמֹהּ הְשִִיבוּ וֵאֹמֵרוּ קֵצֲר, וֵלֲא הֹיֹה דָּי לֹהּם לוֲמָר הְין: וֵזּה לֵשִוֲן הֹרָמֵבָּ"ם, וֵאִם תֲּאמָר שִּאוֲתוֲ פּּסָח בֵּשִָבֹּת אְרָע כֵּמוֲ שִּדִּמּוּ הָטִּפֵּשִִים, הָאְיךֵ תֹּלֹה הָכֹּתוּב הּתְּר אַכִילֹתֹם לּחֹדֹשִ, בֵּדֹבֹר שִּאְינוֲ הֹעִקֹּר וֵלֲא הָסִּבֹּה, אּלֹּא נִקֵרֹה נִקֵרֹה. אּלֹּא מְאָחָר שִּתֹּלֹה הָדֹּבֹר בֵּמֹחֳרָת הָפּּסָח הָדֹּבֹר בֹּרוּר שִּמֹּחֳרָת הָפּּסָח הוּא הֹעִלֹּה הָמָּתּּרּת אּת הּחֹדֹשִ, וֵאְין מָשִֵגִּיחִין עָל אְיזּה יוֲם הוּא מִימְי הָשִֹּבוּעָ: שִּכְּן הוּא בֵּכֹל מֹקוֲם חוּץ מִבָּקֹּרֵבֹּנוֲת: וֵהֹא דֵּקֹתֹנְי וֵהָשִֵּאֹר נִפֵדּּה, לֵרָבִּי עַקִיבֹא לְית לְיהּ, דִּלֵדִידְיהּ לֲא בֹּעְי פֵּדִיֹּה. אִי נָמִּי שִּלֲּא יֲאמֵרוּ הּקֵדְּשִ יוֲצְא בֵּלֲא פִּדֵיוֲן. תּוֲסֹפוֲת:

שִָמֵנוֲ. וֵזּה לֲא יֵהְא אּלֹּא בָּיּוֲם, רֵצוֲנוֲ לוֲמָר יוֲם שִִשִֹּה עֹשֵֹר בֵּנִיסֹן. הֹרָמֵבָּ"ם: יֹצָק כוּ'. בֵּמִשִֵנֹה ג' פּּרּק ו' תֵּנָן דֵּכֹל הָמֵּנֹחוֲת טֵעוּנוֲת שִֹלֲשִ מָתֵּנוֲת שִּמּן. וֵנִרֵאּה דֵּכֶלֵּהוּ תֹּנְי לֵהוּ הֹכֹא. דֵּנֹתָן שִָמֵנוֲ הוּא נֵתִינָת שִּמּן תֵּחִלֹּה בָּכֵּלִי, וֵיֹצָק וּבֹלָל הוּא יֵצִיקֹה וּבֵלִילֹה. וּבְין הֹכֹא וּבְין הֹתֹם הַוֹה לְיהּ לֵמִתֵנְי בֵּלִילֹה בֵּרְישִֹא, דֵּיֹצָק הוּא יֵצִיקֹה אָחַרוֲנֹה. תּוֲסֹפוֲת: וֵקֹמָץ. לֵאָחָר שִּבְּרְר כֹּל הָלֵּבוֲנֹה לֵצָד אּחֹד:

דֵּאִלּוּ קֵצִירֹה עָצֵמֹהּ אּפֵשִֹר בֵּבְית הָשִֵּלֹחִין, אִי נָמִּי עָל יֵדְי קִטּוּף: מֶתֹּרִים. לֵכֶלְּי עֹלֵמֹא. וָאַפִלּוּ רָבִּי יֵהוּדֹה מוֲדּה בִּזֵמָן שִּיְּשִ עֲמּר עֲמּר מָתִּיר, דִּכֵתִיב עָד הַבִיאַכּם. רָשִִ"י: מִפֵּנְי כוּ'. וֵלֹכְן צֵרִיכִין שִּיִּשִֵמֵרוּם חַצוֲת יוֲם יֵרוּשִֹלָיִם. לֲא חַצוֲת אוֲתֹן הָמֵּקוֲמוֲת. לֵפִי שִּאּפֵשִֹר שִּיִּהֵיּה מֶקֵדֹּם אוֲ מֵאֶחֹר כוּ'. הֹרָמֵבָּ"ם. וֵעָיְּן תּוֲסֵפוֲת יוֲם טוֲב:

אְין לִי אּלֹּא חַדֹשִֹה שִּל חִטִּים, חַדֹשִֹה שִּל שֵֵעֲרִים מִנָּיִן, תָּלֵמוּד לוֲמָר חַדֹשִֹה חַדֹשִֹה , אִם אְינוֲ עִנֵיֹן לָחַדֹשִֹה שִּל חִטִּים, תֵּנְהוּ עִנֵיֹן לָחַדֹשִֹה שִּל שֵֵעֲרִים. וּמִנָּיִן שִּתֵּהְא קוֲדּמּת לָבִּכּוּרִים, תָּלֵמוּד לוֲמָר כוּ', וּמִנָּיִן שִּתֵּהְא קוֲדּמּת לִנֵסֹכִים וּפְרוֲת הֹאִילֹן, נּאֱמָר כוּ': פֹּסוּל. וֵאִם הְבִיא לֲא נִתֵקָדֵּשִוּ לִהֵיוֲת רֵאוּיִין לֵקֹרֵבֹּן. אַבֹל נִתֵקָדֵּשִוּ לֵהִפֹּסְל וֵיִהֵיוּ כֵּקֹדֹשִִים שִּנִּפֵסֵלוּ. הֹרָמֵבָּ"ם: וֵקָשִֵיֹא, תִּנָּח חֹדֹשִ שִּנּּאֱסָר וָעַדָיִן לֲא הֶתָּר, אַבֹל בִּכּוּרְי פְּרוֲת וֵנִסֵכְּי יָיִן דֵּמְעוֲלֹם לֲא נּאּסֵרוּ אָמָּאי אִם הְבִיא פֹּסוּל. וֵכְן הוּא בָּגֵּמֹרֹא, דִּלֵהָאי טָעַמֹא דֵּלֲא הֶתָּר מִכֵּלֹלוֲ כוּ', נֵסֹכִים וּבִכּוּרִים שִּהִקֵרִיבֹם קֲדּם לֹעֲמּר כֵּשְִרִים. וֵכֹתָב רָשִִ"י, וּדֵקֹתֹנְי וֵאִם הְבִיא פֹּסוּל, לֹאו אָבִּכּוּרְי פְּרוֲת קֹאְי כוּ'. וּלֵדִינֹא מָשִֵמָע בָּגֵּמֹרֹא דֵּלְיתֹא לֵהָאי טָעַמֹא, וֵהָאי וֵאִם הְבִיא פֹּסוּל קֹאְי גָּם אָבִּכּוּרְי פְּרוֲת. וֵגָם הֹרָמֵבָּ"ם בֵּחִבּוּרוֲ פֹּסָק כְּן. וֵעָיְּן תּוֲסֵפוֲת יוֲם טוֲב:

וֵאִם תֲּאמָר וֵהֹנֹךֵ מִינִין שִּהְן כִּלֵאָיִם זּה בֹּזּה כֵּדִתֵנָן בֵּרְישִ כִּלֵאָיִם הְיאָךֵ הְן מִצֵטֹרֵפִין, הֹתֵנָן בֵּסוֲף פּּרּק ב' דִּתֵרוּמוֲת כֹּל שִּהוּא כִּלֵאָיִם בָּחַבְרוֲ לֲא יִתֵרֲם מִזּּה עָל זּה. וֵיְשִ לוֲמָר דֵּגָבְּי חָלֹּה תֹּלוּי הָטָּעָם בֹּעִסֹּה לֵפִי שִּעִסּוֲתְיהּן דּוֲמוֲת זוֲ לֹזוֲ. וֵעוֲד יְשִ לוֲמָר דִּכֵשִּיְּשִ בֵּכֹל אּחֹד בִּפֵנְי עָצֵמוֲ כֵּדְי חִיּוּב חָלֹּה אְין תּוֲרֵמִין מִזּּה עָל זּה מִמִּין עָל שִּאְינוֲ מִינוֲ. תּוֲסֹפוֲת: מִדִּכֵתִיב בָּשֵֹּדּה, דֹּיֵקִינָן לָהּ בָּגֵּמֹרֹא דֵּצֹרִיךֵ הָשִֵרֹשִֹה:

כְּן הוּא גִּרֵסָת הָתּוֲסֹפוֲת בֵּמָתֵנִיתִין. וֵכֹתֵבוּ, דֵּצֹרִיךֵ לוֲמָר כְּיוֹן דִּלֵכָתֵּחִלֹּה לֲא יֹבִיא, אָף עָל גָּב דֵּאִם הְבִיא כֹּשְִר, מִמֹּקוֲם שִּאִי אָתֹּה מְבִיא קֹרְינֹא בְּיהּ כוּ'. וֵהֹא דֵּלֲא חַשִִיב הֹכֹא בְּית הָזֵּבֹלִים כוּ', יְשִ לוֲמָר דֵּלֲא פֵסִידְי וֵאָחֵמוּר רָבֹּנֹן כֵּמוֲ דֵּאָחֵמוּר בִּגֵדִישִֹה. וּלֵנֶסֵחֹא אָחַרִינֹא דֵּגֹרָס בְּית הָשִֵּלֹחִין שִּבֹּעַמֹקִים, יְשִ לוֲמָר דֵּהָהוּא אִם הְבִיא פֹּסוּל דֵּאִיכֹּא תָּרֵתְּי לֲא קֹרִינָן בְּיהּ שִּאָתֹּה מְבִיא, אַבֹל הְיכֹא דֵּאִם הְבִיא כֹּשְִר קֹרִינָן בְּיהּ כוּ'. וֵעָיְּן תּוֲסֵפוֲת יוֲם טוֲב: מִשִּוּם דֵּבִגֵדִישִֹה לְיכֹּא פֵּסִידֹא אִי שִֹבֵקִינָן עָד אָחָר עֲמּר. וֵהֹא דֵּשִֹרְי רָבִּי יֵהוּדֹה בֵּמִשִֵנֹה ה' קּמָח וֵקֹלִי, מִשִּוּם עוֲלְי רֵגֹלִים הִתִּירוּ. תּוֲסֹפוֲת. וֵעָיְּן תּוֲסֵפוֲת יוֲם טוֲב:

וּלֵהֹכִי נֹקָט נֵטִיעוֲת, מִשִּוּם אוֲרֵחֹא דֵּמִלֵּתֹא שִּאֹדֹם זוֲרְעָ כוּ'. רָשִִ"י: וֵיְרֹאּה לֵפֹרְשִ בְּית כֵּמוֲ שִּפְּרְשִ רָשִִ"י וָיִּשִֵמָע בְּית פָּרֵעֲה . כֵּמוֲ בְּית יִשֵֵרֹאְל, בְּית יֵהוּדֹה, עַבֹדֹיו וֵאָנֵשְִי בְּיתוֲ: צֵבֹתִים. מָאי טָעַמֹא דֵּכָמֹּה דֵּאּפֵשִֹר לֲא טֹרֵחִינָן. גֵּמֹרֹא: לֲא כוּ'. כֶּלֵּהוּ יֹבִיא דִתֵנִינָן נָפֵקֹא לֹן מִוֵּתָקֵרִיב יֵתְרֹה כוּ'. בָּגֵּמֹרֹא: כֵּדִכֵתִיב מְהֹחְל חּרֵמְשִ בָּקֹּמֹה. רָשִִ"י: כְּן פְּרְשִ רָשִִ"י בֵּמִשִֵנֹה ב', רָךֵ וּמֹלְא. וֵכֹתֵבוּ הָתּוֲסֹפוֲת, דָּהַוְי כֵּמוֲ שִֵמֵלֹה שֵָלֵמֹה. אַבֹל לֵשִוֲן רָשִִ"י בֵּמָתֵנִיתִין, כָּר וּמֹל. וֵזּה לֵשִוֲנוֲ בֵּפְרוּשִ הָחֶמֹּשִ, בֵּעוֲד הָכָּר מֹלְא שִּהָתֵּבוּאֹה לָחֹה וּמֵלְאֹה בָּקָּשִִּין שִּלֹּהּ, וֵעָל כְּן נִקֵרֵאוּ הָמֵּלִילוֲת כָּרֵמּל, וֵכְן כָּרֵמּל בֵּצִקֵלוֲנוֲ : בָּיּוֲם כוּ'. וּמָתֵנִיתִין דֵּסוֲף פּּרּק ב' דִּמֵגִלֹּה דֵּדָוֵקֹא בָּלָּיֵלֹה וֵלֲא בָּיּוֲם, וֵכְן הֹא דֵּרְישִ פִּרֵקִין דֵּנִקֵצָר בֵּשִָבֹּת, אָתֵיֹא כֵּרָבִּי אּלֵעֹזֹר בֵּרָבִּי שִִמֵעוֲן, וּדֵהֹכֹא כֵּרָבִּי. גֵּמֹרֹא: הָשִָּבֹּת. לֵהָקֵרֹבֹה, אַבֹל לִקֵצִירֹה לֲא. דֵּהֹא נִקֵצָר בָּיּוֲם כֹּשְִר, וֵיוּכָל לֵקֹצֵרֹהּ מְעּרּב שִָבֹּת, וֵאָמָּאי תִּדֵחּה שִָבֹּת. גֵּמֹרֹא:
