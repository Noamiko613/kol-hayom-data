פרק א

שִּנִּקֵמֵצוּ. עַבוֲדֹה קָמָּיֵתֹא נֹקָט. וֵהוּא הָדִּין נֵתִינֹה בִּכֵלִי וֵהוֲלֹכֹה וֵהָקֵטֹרֹה, כִּדֵּקֹתֹנְי סְיפֹא. תּוֲסֹפוֲת: דֵּהָיֵנוּ שִִנּוּי קֲדּשִ כֵּדִמֵפֹרְשִ בֵּסְיפֹא לֵשְִם מִנֵחָת חוֲטְא כוּ'. וֵהוּא הָדִּין שִִנּוּי בֵּעֹלִים. תּוֲסֹפוֲת: גֵּמֹרֹא. אִי בֹּעְית אְימֹא סֵבֹרֹא, מִשִּוּם דִּמֵשִָנְּי בֹּהּ כֹּל הֹנְי לְשִָנְּי וֵלְיזִיל. אִי בֹּעְית אְימֹא קֵרֹא כוּ'. וֵעָיְּן פְּרוּשִ הֹרָ"ב בֵּרְישִ זֵבֹחִים: וּלֵפִי זּה הוּא הָדִּין אַשִָם נֹזִיר וּמֵצֲרֹע דֵּבֹאוּ לֵהָכֵשִִיר וֵלֲא הִכֵשִִירוּ. אַבֹל כֹּל זּה אִיתּוֲתָב נָמִּי בָּגֵּמֹרֹא, וּלֵהֹכִי נָמִּי לֲא כֵּתֹבֹם הֹרָ"ב בֵּרְישִ זֵבֹחִים. וֵכְן הֹרָמֵבָּ"ם בֵּחִבּוּרוֲ חֹזָר בּוֲ וֵלֲא פֹּסָל לֹעֲמּר שִּלֲּא לִשִֵמוֲ: וֵהִקֵטִיר. בִּמֵקוֲם זֵרִיקֹה:

אָחָת כוּ'. וּמִשִּוּם דֵּרָבִּי שִִמֵעוֲן אֹמָר טָעַמֹא דֵּלֲא בֹּעְי מִנֵחָת חוֲטְא לֵבוֲנֹה וֵשִּמּן שִּלֲּא יֵהְא קֹרֵבֹּנוֲ מֵהֶדֹּר, סֹלֵקֹא דָעֵתֹּךֵ אַמְינֹא כִּי קֹמֵצְי לָהּ פֵּסוּלִין נָמִּי תִּתֵכָּשִָּר, קֹא מָשִֵמָע לֹן. גֵּמֹרֹא: קֹמָץ כוּ'. בָּגֵּמֹרֹא יֹלֵפִינָן לָהּ מִקֵּרֹא: וֵעֹלֹה. וֵאִם תֲּאמָר מָאי אִירֵיֹא עֹלֹה אַפִלּוּ נִשִֵאָר בֵּשִִירָיִם נָמִּי, דֵּהָוֵיֹא מִנֵחֹה חַסְרֹה. וֵיְשִ לוֲמָר שִּאְין רֵגִילִים לֵצָמֵצְם הֹעִשֵֹּרוֲן שִּלֲּא יִהֵיּה בּוֲ מֵעָט יוֲתְר. תּוֲסֹפוֲת: צֵרוֲר כוּ'. כֹּל הֹנְי לֹמֹּה לִי, צֵרִיכֹא, דֵּאִי תֹּנֹא צֵרוֲר מִשִּוּם דֵּלֹאו בָּת הָקֵרֹבֹה הִיא, אַבֹל מּלָח דֵּבָת הָקֵרֹבֹה הִיא אְימֹא תִּתֵכָּשִָּר. וֵאִי תֹּנֹא מּלָח דֵּלֲא אִקֵּבָע בָּהַדְי מִנֵחֹה מְעִקֹּרֹא, שִּאְינוֲ מוֲלְחָ אּלֹּא הָקֲּמּץ בִּלֵבָד, אַבֹל לֵבוֲנֹה דֵּאִקֵּבָע בָּהַדְי מִנֵחֹה מְעִקֹּרֹא אְימֹא תִּתֵכָּשִָּר, קֹא מָשִֵמָע לֹן. גֵּמֹרֹא: גָּרֵגִּיר כוּ'. אְין לֵהֹבִיא רֵאֹיֹה מִזּּה שִּמָּשִֵלִיךֵ הָמּּלָח עָל כֹּל הָמִּנֵחֹה, שִּזּּה אְינוֲ כוּ'. אּלֹּא דּוּמֵיֹא דִּצֵרוֲר, רֵצוֲנוֲ לוֲמָר שִּאִם נִזֵדָּמְּן וֵנִתֵעֹרְב עִם הָסֲּלּת שִוּם דֹּבֹר מִזּּה וֵיָעַלּה בֵּיֹדוֲ. הֹרָמֵבָּ"ם: וֵשְִם אוֲתוֲ הָתּוֲסּפּת שִּמּוֲסִיפִין בָּגֵּדִישִֹה בְּרוּצִין. הֹרָמֵבָּ"ם. וּבּעֹרוּךֵ פְּרְשִ מִלֵּשִוֲן רִיצֹה, שִּהוּא יוֲצְא וֵרֹץ לָחוּץ: וֵחֹסְר כוּ'. אָגָּב דִּמֵפֹרְשִ יֹתְר, מֵפֹרְשִ חֹסְר כָּיּוֲצְא בּוֲ, אָף עָל גָּב דִּצֵרוֲר כוּ' מִשִּוּם חֹסְר הוּא. תּוֲסֹפוֲת: גֵּמֹרֹא. וֵהוּא הָדִּין מָאַפְה תָּנּוּר. וֵהֹא דֵּנֹקָט קּמָח, מִשִּוּם דֵּפוֲתֵתֹן, שִָיֹּךֵ לֵקֹרֵאֹן קּמָח. וֵעָיְּן תּוֲסֵפוֲת יוֲם טוֲב:

דֵּמָאי אִירֵיֹא דֵּתֹנְי רִבֹּה שִָמֵנֹהּ, לִתֵנְי רִבֹּה לֹהּ שִּמּן, אּלֹּא הֹא קֹא מָשִֵמָע לֹן דֵּאָף עָל גָּב דֵּהִפֵרִישִ לֹהּ שִֵנְי לֶגִּין וֵלֲא אֹמֵרִינָן הוֲאִיל וֵהָאי חַזִי לְיהּ לֲא לִפֹּסְל. גֵּמֹרֹא. וֵרִבֹּה פֹּחוֲת מִשִֵּנְי לֶגִּין, יְשִ פֵּלֶגֵתֹּא בְּין רָשִִ"י וֵהֹרָמֵבָּ"ם. וֵדָעָת הֹרָ"ב אְין הּכֵרְעָ. וֵעָיְּן תּוֲסֵפוֲת יוֲם טוֲב: דִּשִֵנְי קֵמֹצִין לֵגָבְּי שִֵנְי קֳרֹטִין הַוֹה יוֲתְר מִדָּאי הָרֵבְּה. מָה שִּּאְין כְּן שִֵנְי לֶגִּין לֵגָבְּי לֲג אּחֹד: הָקּוֲמְץ כוּ'. וֵנוֲתְן בִּכֵלִי. בֵּאָרֵבָּע עַבוֲדוֲת הָלֹּלוּ מֵפָגֵּלִים בָּמִּנֵחֹה. דֵּגֹמֵרִינָן מִזֵּבֹחִים בֵּהּקְּשִ. כִּדֵאֹמָר בֵּרְישִ פִּרֵקִין דֵּאִתָּקָּשִ מִנֵחֹה לֵחָטֹּאת וֵאֹשִֹם. וּמְהָהִיא הּקְּשִֹא גֹּמֵרִינָן מָחַשִֹבֹה דֵּפוֲסּלּת בָּמִּנֵחֹה. וּמִנָּהּ, מָה זּּבָח נִפֵגֹּל בֵּאָרֵבָּע עַבוֲדוֲת כוּ' אָף מִנֵחֹה כוּ'. רָשִִ"י: לֵמֹחֹר. דֵּהָיֵנוּ חוּץ לִזֵמָנּוֲ, דֵּמִנֵחֹה אְינֹהּ נּאֱכּלּת אּלֹּא לֵיוֲם וֵלָיֵלֹה. דֵּאִתָּקָּשִ לֵחָטֹּאת וֵאֹשִֹם. רָשִִ"י: לִרֵבוּתֹא כֹּתָב כְּן, דָּאַפִלּוּ פֵּסוּל לְיכֹּא:
