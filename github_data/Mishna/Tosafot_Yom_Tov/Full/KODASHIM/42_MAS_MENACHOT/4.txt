פרק ד

הָתֵּכְלּת כוּ'. וֵלֲא דִּבֵּרֹה הָמִּשִֵנֹה בֵּדִינְי הָצִּיצִית וֵהָתֵּפִלִּין וֵהָמֵּזוּזֹה וֵעִנֵיָן מֵלָאכֵתֹּן וֵהָבֵּרֹכוֲת שִּחָיֹּב לֵבֹרְךֵ עַלְיהּן כוּ'. וֵסִבָּת זּה בֵּעְינָי, לֵפִי שִּהֹיוּ הָדֵּבֹרִים הֹאְלּּה מֵפֶרֵסֹמִים בִּזֵמָן חִבּוּר הָמִּשִֵנֹה, וֵהֹיוּ עִנֵיֹנִים יֵדוּעִים וּנֵהוּגִים בֵּיָד כֹּל הֹעֹם פֵּרֹט וּכֵלֹל, אְין עִנֵיֹן מְהּן נִפֵלֹא מִשִּוּם אֹדֹם, וֵעָל כְּן לֲא רֹאֹה לֵדָבְּר בֹּהּם כֵּמוֲ שִּלֲּא הִסֵדִּיר נֶסָּח הָתֵּפִלֹּה וֵאְיךֵ יִתֵנָהְג שִֵלִיחָ צִבּוּר לֵפִי שִּהֹיֹה מֵפֶרֵסֹם. אַבֹל חִבֵּרוּ סְפּר גֵּמֹרֹא וּבְאַרוּ. הֹרָמֵבָּ"ם: כְּן כֹּתֵבוּ רָשִִ"י וֵתוֲסֹפוֲת, מִדֵּדֹרֵשִִינָן גֵּדִיל שִֵנָיִם דֵּאְין גֵּדִיל פֹּחוּת מִשִֵּנָיִם, גֵּדִילִים מָשִֵמָע אָרֵבֹּעֹה: זוֲ דָּעָת הֹרָמֵבָּ"ם, דֵּלֲא בֹּעִינָן אּלֹּא חוּט אּחֹד תֵּכְלּת. אַבֹל סוֲבְר שִּיִּהֵיּה קֵצֹתוֲ לֹבֹן דֵּהֹא כּוֲפְל לֵכֹל חוּט עָד שִּיִּהֵיוּ שִֵמֲנֹה, וֵנִמֵצֹא שִּמְּעָכֵשִֹיו שִִבֵעֹה דֵּלֹבֹן וֵאּחֹד דִּתֵכְלּת. וֵעָיְּן תּוֲסֵפוֲת יוֲם טוֲב: תֵּפִלֹּה. תָּרֵגּוּם לֵטוֲטֹפוֲת, לִתֵפִלִּין. וֵהָטּוּר מֵפֹרְשִ לֵשִוֲן פֵּלִילִיֹּה. שִּהְן אוֲת וֵעְדוּת לֵכֹל רוֲאְינוּ שִּהָשִֵּכִינֹה שִוֲרֹה עֹלְינוּ, דִּכֵתִיב וֵרֹאוּ כֹל עָמְּי הֹאֹרּץ כִּי שְִם ה' נִקֵרֹא עֹלּיךֹ וֵדֹרֵשִִינָן אְלּוּ תֵּפִלִּין שִּבֹּרֲאשִ. וֵעָיְּן תּוֲסֵפוֲת יוֲם טוֲב: וֵהָחִלּוּק דֵּבָתֵּפִלֹּה קֹרְי לְיהּ אְין מֵעָכֵּבִין כוּ', וּבֵאָרֵבֹּעֹה מִינִין שִּבָּלּוּלֹב מֵעָכֵּבִין, דִּבֵאָרֵבֹּעֹה מִינִין כֵּשִּהְן מֵצוּיִין נוֲטְל אּחֹד אּחֹד וֵאְינוֲ יוֲצְא יֵדְי חוֲבֹתוֲ וֵלֲא שִֹלֵמֹה הָמִּצֵוֹה עָד שִּיִּגֵמֲר נֵטִילָת אָרֵבָּעֵתֹּן בֵּיֹדוֲ, וָאַפִלּוּ בֹּזּה אָחָר זּה, וֵאְינוֲ כְן בָּתֵּפִלֹּה כוּ' אּלֹּא כֹּל זֵמָן שִּיָּנִּיחָ אּחֹד מִשִֵּנְיהּן רֵצוֲנוֲ לוֲמָר שִּלּוֲבֵשִֹן, כֵּבֹר קִיְּם מִצֵוָת עַשְֵה אָחָת וֵיֹצֹא יֵדְי חוֲבֹתֹהּ: גָּם הֹרָמֵבָּ"ם בֵּחִבּוּרוֲ סֹתָם וֵכֹתָב לֵשִוֲן הָמִּשִֵנֹה וֵלֲא חִלְּק בְּין יְשִ לוֲ לֵאְין לוֲ. וֵעָיְּן תּוֲסֵפוֲת יוֲם טוֲב: מָשִֵמָע הֹא סֲלּת וֵשִּמּן מֵעָכֵּבְי אָהַדֹדְי. תּוֲסֹפוֲת. וֵעָיְּן תּוֲסֵפוֲת יוֲם טוֲב:

דֵּאְלּוּ דֵּפֹרֹשִָת אֱמֲר גּוּפָיֵהוּ מֵעָכֵּבִין זּה אּת זּה, דָּהַוָיֹה כֵּתִיב בֵּהוּ. פְּרְשִ רָשִִ"י יִהֵיוּ עוֲלֹה. וֵכֹל הַוָיֹה עִכּוּבֹא. וּמִיהוּ פֹּרִים וּכֵבֹשִֵים דֵּפֹרֹשִָת פִּינֵחֹס לֲא מֵעָכֵּבְי אָהַדֹדְי, וּמֹצִינוּ נָמִּי לֵפֹרוֲשְִי מָתֵנִיתִין בִּכֵהָאי גָוֵנֹא. וֵהֹכִי אִיתֹא בָּגֵּמֹרֹא דֵּתָנֹּא מִלְּי מִלְּי קֹתֹנְי. וֵעָיְּן תּוֲסֵפוֲת יוֲם טוֲב:

וֵהָכֵּבֹשִֵים כוּ'. וָאַפִלּוּ הֹכִי כִּי אֹתֵיֹן בָּהַדֹדְי, הָכֵּבֹשִֵים עִקֹּר וֵהָלּּחּם טֹפְל, מִשִּוּם דִּכֵבֹשִֵים מָתִּירִין אּת הָלּּחּם כֵּשִּבֹּא בֵּיָחָד, וֵלֲא פֹּלְיג רָבִּי עַקִיבֹא אָסֵּתֹמֹא דֵּמִשִֵנֹה ג' פּּרּק ב'. וֵעָיְּן תּוֲסֵפוֲת יוֲם טוֲב: אֹמָר רָבִּי שִִמֵעוֲן בּּן נָנֹּס כוּ'. בָּגֵּמֹרֹא מֵפֹרְשִ דִּפֵלִיגְי בֵּטָעַמֹא דִּקֵרֹא: לֲא קֹרָב. וֵזּה קָבֹּלֹה בֵּיֹדוֲ. הֹרָמֵבָּ"ם:

תֹּמֵהוּ הָתּוֲסֹפוֲת, דִּבֵרְישִ פּּרּק י' דִּזֵבֹחִים נָפֵקֹא לֹן מִדִּכֵתִיב מִלֵּבָד עוֲלָת הָבֲּקּר וֵגוֲ'. וּמָסֵּקְי דֵּהֹכֹא לֲא נָפֵקֹא לֹן אּלֹּא הָקֵטֹרֹה, דֵּכֶלְּי קֵרֹא בֵּהָקֵטֹרֹה מָיֵרְי, וּמְהֹתֹם נָפֵקֹא לֹן עַבוֲדָת דֹּם, דִּכֵתִיב תָּעַשֵוּ, מָשִֵמָע עַשִֵיּוֲת: וֵלֲא הָמּוּסֹפִים כוּ'. אִי מָיֵרְי בֵּעִכּוּבֹא בֵּעֹלֵמֹא לֹאו כֵּלֹלֹא, דֵּשִּל חֹג מֵעָכֵּבִין, דִּכֵמִשִֵפֹּט כֵּמִשִֵפֹּטֹם כֵּתִיב בֵּהוּ. וֵאִי נָמִּי מָיֵרְי בָּקֵּדִימֹה שִּל חֹג נָמִּי אְינֹן בִּכֵלֹל מְהָאי טָעַמֹא גּוּפֹא, דִּכֵמִשִֵפֹּט כֵּתִיב, לֵכֹךֵ קֵרְבִין עָל הָסְּדּר כִּדֵאִיתֹא בָּגֵּמֹרֹא: דֵּאִם לֲא כְן, שִּאְין מֵחָנֵּכִין דֵּתֹנֹא בֵּסְיפֹא אָהְי קֹאְי: דֵּכִי כֲּהַנִים חֹטֵאוּ מִזֵבְּחָ בֹּטְל. גֵּמֹרֹא: מֵפֹרְשִ לֵלִישִֵנֹא דֵּלֲא שִֵכִיחֹא, שִּכְּן הוּא בֵּהָסֶּגֵיֹא כוּ'. וֵהָהוּא וָדָּאי לֲא קָשִֵיֹא אָמָּאי קֹאֹמָר לֲא שִֵכִיחָ, הֹא לֲא הַוְי כֵּלֹל, דֵּאָזֵּמָן הָתָּקֹּנֹה קֹאְי כוּ' וֵהָהוּא שִָעֵתֹּא אָכָּתִּי לֲא הַווּ יֹדֵעְי רָבֹּנֹן שִּלֲּא יִשִֵנּה בֹּהּ אֹדֹם מְעוֲלֹם: וֵאָף עָל גָּב דֵּבְינָיֵהוּ כֵּתִיב וֵכֹלִיל עָל מִזֵבֵּחּךֹ, מִסֵתָּבְּר דֵּכִי כֵּתִיב עֲשִּר אָדֵּלֲא שִֵכִיחָ כֵּתִיב. גֵּמֹרֹא. וֵעָיְּן תּוֲסֵפוֲת יוֲם טוֲב: וֵכֶלֹּה כוּ'. דִּקֵטֲרּת חַדֹא מִלֵּתֹא הִיא. מָה שִּּאְין כְּן כּּבּשֵ לֲא אֹמָר דִּשִֵנָיִם הֹיוּ קֵרְבִין. תּוֲסֹפוֲת: מֵחָנֵּכִין. עִנֵיָן חִנּוּךֵ, הּרֵגְּל. וֵהוּא שְִם מֶשִֵאֹל בָּדֵּבֹרִים הֹאְלּּה לִתֵחִלָּת הָמָּעַשֵּה, לֵפִי שִּהָכְּלִים מָרֵגִּילִים אוֲתֹן בֹּעַבוֲדֹה, דּּרּךֵ דּוּמֵיֹא לֵאֹדֹם בִּתֵחִלֹּה שִּמֵּלָמֵּדִים אוֲתוֲ שִוּם חֹכֵמֹה אוֲ שִוּם מִדֹּה לֵהָרֵגִּיל עָצֵמוֲ בֹּהּ עָד שִּתִּהֵיּה קֵבוּעֹה בּוֲ. הֹרָמֵבָּ"ם: שִָחָר. דֵּהֹכִי מִדֵרֹשִ קֵרֹא, אִם אּת הָכּּבּשֵ הֹאּחֹד כוּ': בֵּשִָבֹּת. אּלֹּא בֵּחֲל אִיחַנוֲכְי הוּא דֵּלֲא מִחַנָךֵ הֹא קָדּוּשְִי מִקָּדִּישִ, וֵהֹתֵנָן בֵּפּרּק י"א מִשִֵנֹה ח' שִּאַפִלּוּ הוּא עָל הָשִֶּלֵחֹן יֹמִים רָבִּים אְין בֵּכֹךֵ כֵּלוּם. וּמֵשִֹנְי, הִיא גּוּפֹהּ קֹא מָשִֵמָע לֹן, דֵּחִנּוּךֵ וֵקִדּוּשִ דֵּשִֶלֵחֹן בֵּשִָבֹּת הוּא. גֵּמֹרֹא:

לֲא הֹיוּ כוּ'. בָּגֵּמֹרֹא יֹלְיף לְיהּ מִקֵּרֹא: מִבְּיתוֲ. יֹלְיף לָהּ בָּגֵּמֹרֹא מִקֵּרֹא: בָּגֵּמֹרֹא מֵפֹרְשִ מָאי עַבְיד רָבִּי שִִמֵעוֲן בֵּאוֲתֹהּ, וֵרָבִּי יֵהוּדֹה מָאי עַבְיד בִּדֵרָבִּי שִִמֵעוֲן:
