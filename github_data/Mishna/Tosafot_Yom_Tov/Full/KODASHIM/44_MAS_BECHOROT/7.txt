פרק ז - משנה א

כָּל זֶה יָלֵיף לְהוּ בְתוֹרַת כֹּהֲנִים: יוֹתֵר כוּ'. גְּמָרָא, דְּאָמַר קְרָא כָּל אִישׁ אֲשֶׁר בּוֹ מוּם מִזֶּרַע אַהֲרֹן, אִישׁ שֶׁשָּׁוֶה בְזַרְעוֹ שֶׁל אַהֲרֹן. פֵּרֵשׁ רַשִׁ"י, אִישׁ הַשָּׁוֶה בְזַרְעוֹ שֶׁל אַהֲרֹן הוּא דְבָעֵי מוּמָא, הָא אֵין שָׁוֶה בְזַרְעוֹ שֶׁל אַהֲרֹן בְּלֹא מוּמָא נַמִּי מִפְסַל. וּבְפֶרֶק קַמָּא כָּתַב, שָׁוֶה בְזַרְעוֹ שֶׁדּוֹמֶה לִבְנֵי אָדָם: שָׁקוּעַ. וְכֵן גָּרַס הָרַמְבַּ"ם. וּמְפָרֵשׁ, שֶׁיִּהְיֶה דְמוּת רֹאשׁוֹ כְּמוֹ שֶׁנָּתַן יָדוֹ עַל גֹּבַהּ רֹאשׁוֹ וְנִתְפַּשֵּׁט עַד שֶׁבָּלְטוּ צְדָדָיו. וּמִי שֶׁרֹאשׁוֹ כִדְמוּת זֶה קוֹרִין אוֹתוֹ בַּעֲרָבִי קוב"י. וּבְחִבּוּרוֹ כָּתַב, מִי שֶׁאֶמְצַע קָדְקֳדוֹ שׁוֹקֵעַ לְמַטָּה כְּמִי שֶׁדְּחָקוֹ בְיָדוֹ: לֹא דְרַבִּי יְהוּדָה לֹא סְבִירָא לֵיהּ הַךְ דְּרָשָׁא, דְּאִם כֵּן לִפְלוֹג בְּכֻלְּהוּ. אֶלָּא דִסְבִירָא לֵיהּ דַּחֲתִיכַת בָּשָׂר בְּעָלְמָא הוּא, וְאֵינוֹ בִּכְלַל שֶׁאֵינוֹ שָׁוֶה. וְכֵן הוּא בַּגְּמָרָא:

פרק ז - משנה ב

שֶׁהוּא פָּסוּל. כִּדְאִיתָא בַגְּמָרָא: מִלְּשׁוֹן גַּב הַמִּזְבֵּחַ. עַל גַּבִּי חָרְשׁוּ חוֹרְשִׁים. רַמְבַּ"ן: וּלְשׁוֹן הָרַמְבַּ"ם עַל עַפְעַפֵּי עֵינָיו:

פרק ז - משנה ג

אֵיזֶהוּ כוּ'. וּלְשׁוֹן חָרוּם מִלְּשׁוֹן כָּל חֵרֶם אֲשֶׁר יָחֳרַם, וְהַחֲרַמְתִּי עָרֵיהֶם, עִנְיַן חֻרְבָּן. וְיִקָּרֵא חָרוּם כִּי הַחֹטֶם הַדְרַת פָּנִים, כְּמוֹ שֶׁאָמְרוּ אֵין מְעִידִין אֶלָּא עַל פַּרְצוּף פָּנִים עִם הַחֹטֶם. וַאֲשֶׁר אֵין חָטְמוֹ כְתֹאַר בְּנֵי אָדָם צוּרָתוֹ חֲרֵבָה. הָרַמְבַּ"ם: לְמַעְלָה כוּ'. דְּתָנוּ רַבָּנָן, בְּעֵינוֹ, כָּל מַה שֶּׁבְּעֵינוֹ, מִכָּאן אָמְרוּ שְׁתֵּי עֵינָיו כוּ': מִפְּנֵי כוּ'. אַנָּשְׁרוּ רִיסֵי עֵינָיו קָאֵי. רַשִׁ"י: וְדוֹחֵק זֶה לָזֶה. הָרַמְבַּ"ם: כָּךְ כָּתַב הָרַמְבַּ"ם בְּהֶדְיָא דְּלוֹקֶה. אֲבָל רַשִׁ"י כָּתַב עוֹבֵר בַּעֲשֵׂה, דְּהָכִי מַשְׁמַע, הַשָּׁוֶה בְזֶרַע אַהֲרֹן וְאֵין בּוֹ מוּם הָוֵי כֹהֵן וְיַעֲבֹד, שֶׁאֵינוֹ שָׁוֶה לֹא יַעֲבֹד. וְלָאו הַבָּא מִכְּלָל עֲשֵׂה עֲשֵׂה. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב:

פרק ז - משנה ד

עֵינָיו כוּ'. אַף עַל גַּב דְּבִבְהֵמָה לֹא הֲוָה מוּמָא בִּשְׁתֵּיהֶן שָׁווֹת, כְּמוֹ שֶׁכָּתוּב לְעֵיל פֶּרֶק ו' מִשְׁנָה י', הָכָא מִשּׁוּם שֶׁאֵינוֹ שָׁוֶה בְזַרְעוֹ שֶׁל אַהֲרֹן מִפְּסִיל אַף בִּשְׁתֵּיהֶן שָׁוִין. גְּמָרָא: קְטַנּוֹת. אַף עַל פִּי שֶׁשְּׁתֵּיהֶן קְטַנּוֹת. דּוּמְיָא דְעֵינָיו. וְזֶה לְשׁוֹן הָרַמְבַּ"ם, מִי שֶׁשְּׁתֵּי אָזְנָיו קְטַנּוֹת הַרְבֵּה וּמִי שֶׁשְּׁתֵּי אָזְנָיו נְפוּחוֹת דּוֹמוֹת לִסְפוֹג. הָרַמְבַּ"ם: רַשִׁ"י. שֶׁהַסְּפוֹג צֶמֶר הַיָּם כוּ' כְּשֶׁשּׁוֹאֶבֶת מַיִם הִיא מִתְנַפַּחַת וּנְקָבֶיהָ מִתְרַחֲבִים, וּכְשֶׁכּוֹבְשִׁין אוֹתָהּ וּמוֹצִיאִין מִמֶּנָּה מַה שֶּׁשָּׁאֲבָה הִיא מִתְיַבֶּשֶׁת וְנִכְוֶצֶת וְכָל נְקָבֶיהָ נַעֲשִׂים צָרִים. וְאִם כֵּן דּוֹמֶה לִכְשֶׁהִיא רֵיקָנִית. וּלְהָרַמְבַּ"ם שֶׁכָּתַבְתִּי בְּסָמוּךְ, דּוֹמֶה לִכְשֶׁהִיא מְלֵאָה, דְּהַשְׁתָּא הָוֵי כְּמוֹ הֵפֶךְ מֵאָזְנָיו קְטַנּוֹת:

פרק ז - משנה ה

שְׂפָתוֹ כוּ'. וַאֲפִלּוּ אֵין בָּהּ עֶצֶם. מַה שֶּׁאֵין כֵּן בַּבְּהֵמָה. מִשּׁוּם דַּהֲוֵי שֶׁאֵינוֹ שָׁוֶה כוּ': יוֹצֵא. רֹב בְּנֵי אָדָם מְקוֹם טַבּוּרָם דּוֹמֶה לְגֻמָּא בְּשֶׁטַח הַגּוּף. וְיֵשׁ מֵהֶן שֶׁטַּבּוּרָן בּוֹלֵט עַל שֶׁטַח הַגּוּף דּוֹמֶה לְיַבֶּלֶת: אֵין כוּ'. לֹא זְכָרָם הֵנָּה מִפְּנֵי שֶׁהֵם מְיֻחָדִים בָּאָדָם, לְפִי שֶׁכְּבָר זָכַר שֶׁהֵם גַּם כֵּן מוּמֵי בְהֵמָה. אָמְנָם זְכָרָם לְהוֹדִיעֲךָ שֶׁאֵלּוּ הַמּוּמִין הֵם הַנִּקְרָאִים בַּתּוֹרָה מְרוֹחַ אָשֶׁךְ. הָרַמְבַּ"ם: רַבִּי יִשְׁמָעֵאל אוֹמֵר כוּ'. דְּאִי כְתַנָּא קַמָּא הֲוָה לֵיהּ לִכְתֹּב חֲסַר אָשֶׁךְ. גְּמָרָא: מַשְׁמַע שְׁתֵּי בֵיצָיו. וְכִדְאָמַר אֲשָׁכָיו לְשׁוֹן רַבִּים. וְהָא דְכָתַב קְרָא מְרוֹחַ אָשֶׁךְ לְשׁוֹן יָחִיד, פֵּרוּשׁ שֶׁאֵין לוֹ שׁוּם אֶשֶׁךְ. וְלֹא דָמֵי לְאֵין לוֹ אֶלָּא בֵּיצָה אַחַת כוּ', דְּשָׁאנֵי הָתָם שֶׁנִּבְרָא מִתְּחִלַּת בְּרִיָּתוֹ חֲסַר בֵּיצָה אַחַת. כֶּסֶף מִשְׁנֶה: רַבִּי עֲקִיבָא אוֹמֵר כוּ'. גְּמָרָא, דְּאִי כְרַבִּי יִשְׁמָעֵאל מִמְּרוֹחַ אָשֶׁךְ מִבָּעֵי לֵיהּ. פֵּרֵשׁ רַשִׁ"י, דְּהָא מֵ"ם דִּמְרוֹחַ מְשַׁמֶּשֶׁת הִיא: רַבִּי חֲנִינָא כוּ'. גְּמָרָא, דְּאִי כְרַבִּי עֲקִיבָא רוּחַ אָשֶׁךְ מִבָּעֵי לֵיהּ. אֶלָּא שֶׁמַּרְאָיו חֲשׁוּכִין, וְקָסָבַר גּוֹרְעִין וּמוֹסִיפִין וְדוֹרְשִׁין. פֵּרֵשׁ רַשִׁ"י, גּוֹרְעִין חֵי"ת מִמְּרוֹחַ, וְאָלֶ"ף מֵאָשֶׁךְ, וּמוֹסִיף זֶה בִּמְקוֹם זֶה וְהָוֵי מַרְאוֹ חשֶׁךְ:

פרק ז - משנה ו

אֲבָל לְהָרַמְבַּ"ם לֹא פֵרֵשׁ בְּמַתְנִיתִין אֶלָּא עִקֵּל בִּלְבַד. דְּפִיקָה יוֹצְאָה מִגּוּדָלוֹ לֹא הָוֵי פֵּרוּשׁ דְּבַעַל פִּיקָה. וּמְפָרֵשׁ בַּעַל פִּיקָה הוּא שֶׁתְּהֵא עֲקֵבוֹ מֵאַחַר רַגְלָיו בּוֹלֶטֶת וְיוֹצֵאת יוֹתֵר מִן הָרָאוּי: בַּגְּמָרָא, לֹא תֵימָא דְטַרְפָא וְלֹא סְדִיקָה, אֶלָּא כֵּיוָן דְּטַרְפָא אַף עַל גַּב דִּסְדִיקָה. פֵּרֵשׁ רַשִׁ"י טַרְפָא קְלוּשָׁה. וְהֶעָרוּךְ פֵּרֵשׁ רְחָבָה וּמְשֻׁוָּה: אֶצְבְּעוֹתָיו. בְּיָדָיו וּבְרַגְלָיו. הָרַמְבַּ"ם: הַפֶּרֶק. פֶּרֶק רִאשׁוֹן הַסָּמוּךְ לְכַף הַיָּד. הָרַמְבַּ"ם: וְתֵימַהּ, דִּלְקַמָּן מְפָרֵשׁ הָרַ"ב כָּל שֶׁכֵּן אִם הָיָה בְיָדוֹ אַחַת חָמֵשׁ וּבְיָדוֹ אַחַת שֵׁשׁ דַּהֲוֵי מוּם כוּ', וְהָאֵיךְ תָּלֵי טַעֲמָא הָכָא בְהוֹאִיל וַחֲתָכָהּ כוּ'. וְהָרַמְבַּ"ם כָּתַב, אִם חָתַךְ אֶת הַיְתֵרָה כָּשֵׁר, וְאִם הָיָה בָהּ עֶצֶם אֲפִלּוּ חֲתָכָהּ פָּסוּל. וְלִדְבָרָיו פֵּרוּשׁ וַחֲתָכָהּ כְּלוֹמַר אֲפִלּוּ חֲתָכָהּ: עֶשְׂרִים וְאַרְבַּע. לִישְׁנָא דִקְרָא נָקַט. רַשִׁ"י. וּבַגְּמָרָא עֲבִיד צְרִיכוּתָא אַקְּרָא: הַכּוּשִׁי. לְפֵרוּשׁ הָרַ"ב לְעֵיל שֶׁמַּרְאָיו חֲשׁוּכִים שָׁחֹר וְאֵינוֹ כְּכוּשִׁי, וְכֵן נִרְאֶה שֶׁפֵּרֵשׁ הָרַמְבַּ"ם, קָשֶׁה כּוּשִׁי לָמָּה לִּי, דְּכָל שֶׁכֵּן הוּא. וְרַשִׁ"י פֵּרֵשׁ הָתָם שָׁחֹר כְּכוּשִׁי. וּבַגְּמָרָא רַבִּי חֲנִינָא בֶּן אַנְטִיגְנוֹס לֹא תָנֵי כּוּשִׁי: וְהִשָּׁמֵר שֶׁמָּא תַעֲלֶה עַל דַּעְתְּךָ שֶׁאֵלּוּ הַשֵּׁמוֹת נֶאֶמְרוּ עַל הַגְּוָנִים, לְפִי שֶׁאֵינָן אֶלָּא שֵׁמוֹת לִבְנֵי אָדָם שֶׁיֵּשׁ בָּהֶן אֵלּוּ הַגְּוָנִים. הָרַמְבַּ"ם: פְּסוּלִין כוּ'. אַף עַל גַּב דִּכְבָר תָּנֵי יָתֵר עֲלֵיהֶן בָּאָדָם כוּ', מִשּׁוּם דִּבְתוֹרַת כֹּהֲנִים רָצוּ לְמֵילַף בְּקַל וָחֹמֶר אָדָם מִבְּהֵמָה עַד דִּמְרַבֵּי לֵיהּ מִקְּרָא, לְהָכִי קָאָמַר אַף עַל גַּב דִּכְשֵׁרִין בַּבְּהֵמָה פְּסוּלִין בָּאָדָם:

פרק ז - משנה ז

אוֹתוֹ כוּ'. כְּגוֹן אַהֲרֹן וּבָנָיו. וְדִכְוָתֵיהּ תַּיִשׁ וּבְנוֹ אָסוּר. וּמַתְנִיתִין חֲנַנְיָא הִיא דְאָמַר נוֹהֵג אַף בִּזְכָרִים. גְּמָרָא: וְשֶׁנֶּעֶבְדָה כוּ'. לֹא גָרְסִינַן. וְאֶפְשָׁר מִשּׁוּם הָא דִבְרָכוֹת , אָמַר רַבִּי יוֹחָנָן כָּל כֹּהֵן שֶׁהָרַג אֶת הַנֶּפֶשׁ לֹא יִשָּׂא אֶת כַּפָּיו, שֶׁנֶּאֱמַר יְדֵיכֶם דָּמִים מָלֵאוּ. וְכָל שֶׁכֵּן לָעֲבוֹדָה, וְהוּא הַדִּין לִמְגַלֶּה עֲרָיוֹת. אֶלָּא דְקָשֶׁה לִי דְּאִם כֵּן לִתְנֵי דִּפְסוּלִים בָּאָדָם. וְגַם לֹא רָאִיתִי לְהָרַמְבַּ"ם שֶׁפְּסָלָן בָּאָדָם: מִשּׁוּם דְּמִסְּתָמָא נִיחָא לְהוּ לְהָרַבִּים. תּוֹסָפוֹת:
