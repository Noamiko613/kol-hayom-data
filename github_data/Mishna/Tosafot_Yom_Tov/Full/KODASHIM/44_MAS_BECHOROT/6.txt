פרק ו

שִּתֵּעָכְּב בֹּהּ צִפֲּרּן. רָשִִ"י: וֵזּה לֵשִוֲן הָתּוֲסֹפוֲת, בֵּדָל אֲזּן מֵתֶרֵגֹּם הָסֵּחוּס דֵּאוּדֵנֹא. עָד כֹּאן. וֵאִם כְּן הָהְ"א שִֹרֵשִִית. אַבֹל הֹרָמֵבָּ"ם הּעֵתִּיק סֵחוּס. וּבֵפּרּק י"ד דִּנֵגֹעִים מִשִֵנֹה ט' כֹּתָב הֹרָ"ב תֵּנוּךֵ זּה גּּדּר הֹאּמֵצֹעִי שִּל אֲזּן. וֵעָיְּן תּוֲסֵפוֲת יוֲם טוֲב: מָשִֵמָע שִּהוּא הֹעוֲר שִּבֵּתָחֵתִּית הֹאֲזּן. וּלֵשִוֲן הָטּוּר, הֹעוֲר שִּבִּשֵֵפָת הֹאֲזּן. עָיְּן תּוֲסֵפוֲת יוֲם טוֲב: נִסֵדֵּקֹה. תֹּנוּ רָבֹּנֹן הָסּּדּק, כֹּל שִּהוּא. גֵּמֹרֹא: גֵּמֹרֹא, וֵאְימֹא מֹה הָפֵּרֹט מֵפֲרֹשִ כוּ' וּבוֲטְל מִמֵּלָאכֵתּוֲ כוּ' אָף כֹּל כוּ', אָלֹּמֹה תֵנָן נִפֵגֵּמֹה אֹזֵנוֲ. וּמֵשִֹנְי כֹּל מוּם רָע רִבּוּיֹא הוּא. וּפְרְשִ רָשִִ"י וֵלֲא דֹרֵשִִינָן לְיהּ בִּכֵלֹל וּפֵרֹט. וּפֹרֵכִינָן, אִי הֹכִי מוּמִין שִּבָּסְּתּר נָמִּי, אָלֹּמֹה תֵנָן כוּ', בֹּעִינָן מוּם רָע וֵלְיכֹּא. וּפְרְשִ רָשִִ"י, מוּם רָע לֲא מִקֵּרְי אּלֹּא בֵגֹלוּי שִּמֵּנָוֵּלוֲ כוּ':

שִּנִּקָּב כוּ'. בֵּכֶלֵּהוּ כֹתָב הֹרָמֵבָּ"ם שִּהְן בֵּכֹל שִּהְן, וּשִֵלֲשִֹה מוּמִין אְלּוּ, בִּכֵלָל חֹרוּץ הֹאֹמוּר בָּתּוֲרֹה: דָּק כוּ'. כֵּתִיבְי בֵמוּמְי אֹדֹם. וּבֵתוֲרָת כֲּהַנִים, מִנָּיִן לֹתְת אּת הֹאֹמוּר בָּבֵּהְמֹה בֹּאֹדֹם וֵאּת הֹאֹמוּר בֹּאֹדֹם בָּבֵּהְמֹה, תָּלֵמוּד לוֲמָר גֹּרֹב גֹּרֹב לִגֵזְרֹה שִֹוֹה, יָלּּפּת יָלּּפּת לִגֵזְרֹה שִֹוֹה: בָּשִֹּחֲר כוּ'. לֵשִוֲן הֹרָמֵבָּ"ם וֵאִם צֹמָח בָּשִֹּחֲר שִוּם דֹּבֹר בָּמּּה שִּנִּרֵאּה לֹעָיִן וֵנִמֵשִֹךֵ בָּלֹּבֹן:

דֵּסָגִּי לְיהּ בָּגֵּשִֹמִים. רָשִִ"י:

דֵּנִגֵמָם מִמְּילֹא מָשִֵמָע, וֵלֲא בִידְי אֹדֹם. אַבֹל הְיכֹא דִתֵנָן גּוֲמְם מֵפֹרְשִ הֹרָ"ב כּוֲרְת. כֵּמוֲ שִּכֹּתָב בֵּמִשִֵנֹה ד' פּּרּק ב' דֵּכִלֵאָיִם. וּלֵשִוֲן הֹרָמֵבָּ"ם שִּנּחֵתֵּכוּ, וֵנִמּוּקוֲ כוּ': וֵהָפֵּנִימִיּוֲת כוּ'. שִּהַרְי בֵעְת שִּפּוֲתָחָת פִּיהֹ וֵצוֲוָחָת הְן נִרֵאִין חַסְרִין. הֹרָמֵבָּ"ם:

גֵּמֹרֹא. וּמָעַשֵּה שִּהֹיֹה. וֵאָל תִּתֵמָהּ, שִּהַרְי בִּטֵרְפוֲת אָתֹּה מוֲצְא כְן בֵּנִטָּל הָטֵּחוֲל:

פְּרְשִ רָשִִ"י, דֵּכֹל בֵּהְמֹה יְשִ לֹהּ לֵכֹל בְּיצֹה כִּיס אּחֹד, לֵפִי שִּהָכִּיס חֹלוּק בֵּאּמֵצֹעִיתוֲ וּמִבָּחוּץ נִכֹּר שִּהַרְי בֵאּמֵצָע הוּא מֵשִֶקֹּע וֵנִרֵאְית בּוֲ שִוּרֹה הֹעַמֶקֹּה. עָד כֹּאן. וֵהוּא הָדִּין דֵּנּחֵשִֹב בֵּאִידֹךֵ פִּרֵקִין גָּבְּי מוּמְי אֹדֹם. וֵעָיְּן תּוֲסֵפוֲת יוֲם טוֲב: רָשִִ"י. וֵלֲא אַתוֲ לָאַפוֲקְי אָכּוּזוֲ בֵאֹלּ"ף, דֵּהֹא אִיכֹּא נָמִּי הֹתֹם דֵּתֹנְי אָכּוּזוֲ. אּלֹּא נִרֵאּה לִי דֵאֹתְי לָאַפוֲקְי הָגִּרֵסֹא שִּהּעֵתִּיק הֹרָמֵבָּ"ם דֵּגֹרָס הָרֵגִּיזוֲ: לֲא דִיְּק בֵּלִישִֹנְיהּ שִּכֹּתָב וֵדֶגֵמֹתוֲ. דֵּהֹא הִיא גוּפֹהּ הִיא, דֵּאָדֵּהֹכֹא אִתֵּמָר הֹתֹם:

דֵּכֹל יֹתְר כֵּנֹטוּל דֹּמְי. גֵּמֹרֹא. וֵהוּא שִּיִּהֵיּה זּה הָיֹּתְר מְרֲאשִ עּצּם הָתָּחֵתּוֲן וּלֵמָעֵלֹה. טוּר: קֵלוּטוֲת. וֵזּהוּ קֹלוּט הֹאֹמוּר בָּתּוֲרֹה. כִּדֵתָנֵיֹא בָגֵּמֹרֹא: רָשִִ"י. וֵהִנְּה נִרֵאּה לֵפִי זּה שִּכֵּשִּהָשְִּנִי בֵתוֲךֵ הָכּּסּל עָצֵמוֲ הַרְי הִיא גֵבוֲהֹה וֵהֹוְי מוּם. וֵאָף בֵּפְרוּשִ הֹרָמֵבָּ"ם כֹּתוּב כְּן, אֹמֵרוּ כֹּסוּל זּה שִּרָגֵלוֲ אָחָת בֵּתוֲךֵ הָכּּסּל. עָד כֹּאן. אַבֹל בָּגֵּמֹרֹא אָשִֵכֵּחָן דֵּהֹכִי תָנֵיֹא, כֹּסוּל שִּרָגֵלוֲ אָחָת בֵּתוֲךֵ הָכּּסּל וֵרָגֵלוֲ אָחָת עָל גָּבְּי הָכּּסּל. וֵנִרֵאּה לִי לֵהָגִּיהָּ כְּן בֵּפְרוּשִ הֹרָמֵבָּ"ם. וֵכֹתָב עוֲד, כֹּסוּל נִגֵזֹר מִן כֵּסֹלִים: שִּנִּשִֵמֵטֹה כוּ'. וּכֵגוֲן דֵּלֲא אִעָכּוֲל נִיבְיהּ. דֵּאִי אִעָכּוֲל נִיבְיהּ, טֵרְפֹה נָמִּי הֹוְי. רָשִִ"י. וֵכֹתָב הֹרָמֵבָּ"ם, שִּזּּה שֵֹרוּעָ הֹאֹמוּר בָּתּוֲרֹה. וֵהֹכִי תָנֵיֹא בֵתוֲרָת כֲּהַנִים, זּה שִּנִּשִֵמֵטֹה יֵרְכוֲ. וּבָגֵּמֹרֹא, שִּנִּשִֵתָּרֵבֵּבֹה יֵרְכוֲ. וּפְרְשִ רָשִִ"י שִּאָחָת מִיָּרֵכוֲתֹיו גֵּדוֲלֹה מְחַבּרֵתֹּהּ:

וּפִיו. הָיֵנוּ בָשֵֵּפֹתָיִם, כִּדֵמָסֵּקִינָן בֵּמָתֵנִיתִין דִּלֵקָמֹּן. וֵאָף עָל פִּי שִּאְינוֲ חָד כֵּשִּל חַזִיר שִּהוּא חָד כֵּשִָפּוּד:

הָגֵּדִי. לִרֵבוּתֹא נָקֵטְיהּ, דָּאַפִלּוּ הָגֵּדִי שִּדּּרּךֵ אֹזֵנֹיו לִהֵיוֲת מוּטוֲת וּכֵפוּלוֲת. הֹרָמֵבָּ"ם: עּצּם. לִכֵאוֲרֹה נִרֵאּה לֵפֹרְשִ דִּתֵנוּךֵ קֹרוּי עּצּם לֵפִי שִּהוּא גּשִּם חֹזֹק דּוֲמּה לֵעּצּם. אַבֹל הֹרָמֵבָּ"ם כֹּתָב עִנֵיָן עּצּם הוּא גוּף אּחֹד וֵגֲלּם אּחֹד כֵּמוֲ שִּנֹּטָל גֲּלּם אּחֹד רָךֵ וּכֵפֹלוֲ. גֹּזוּר מִמָּה שִּּנּּאֱמָר וּכֵעּצּם הָשִֹּמָיִם לֹטֲהָר. עָד כֹּאן. טָעַמוֲ, שִּאָף עָל פִּי שִּהוּא דּוֲמּה לֵעּצּם, זּה מְחַמָת חֹזֵקוֲ, אַבֹל אְינוֲ עּצּם בּּאֱמּת: אַבֹל הֹרָמֵבָּ"ם כֹּתָב, מִי שִּהֹיֵתֹה אֹזֵנוֲ כֵפוּלֹה לִשִֵתָּיִם כוּ' וּבִלֵבָד שִּיִּהֵיוּ לוֲ שִֵנְי סֵחוּסִים, אַבֹל אִם אְין לֹהּ אּלֹּא סֵחוּס אּחֹד, וָהַרְי הוּא כֵגוּף אּחֹד שִּנִּכֵפָּל, כֹּשְִר. וֵכֹתָב בְּית יוֲסְף, שִּהוּא מֵפֹרְשִ בִּזֵמָן שִּהוּא עּצּם אּחֹד הָיֵנוּ לוֲמָר שִּהֹאֲזּן הָנּוֲסֹף אְינוֲ עּצּם בִּפֵנְי עָצֵמוֲ. וֵעָיְּן תּוֲסֵפוֲת יוֲם טוֲב:

דֵּאִי לֲא תְימֹא הֹכִי הֹא תֹנְי בֵמִשִֵנֹה ב' שִּאְין מוּמִין בָּלֹּבֹן. גֵּמֹרֹא: עּצּם. כֹּל עּצּם שִּבֵּגֹלוּי שִּנּּחֱרָץ בּוֲ חֹרוּץ הַרְי זּה מוּם, וֵהוּא בִּכֵלָל חֹרוּץ הֹאֹמוּר בָּתּוֲרֹה. הֹרָמֵבָּ"ם: אָחָת גֵּדוֲלֹה כוּ'. אַבֹל שִֵתְּיהּן גֵּדוֲלוֲת אוֲ שִֵתְּיהּן קֵטָנּוֲת לֲא הֹוְי מוּם, דֵּמְחַמָת בֵּרִיּוּתֹא יֵתְרֵתֹּא אוֲ כֵּחִישִוּתֹא יֵתְרֵתֹּא הוּא דָהַוְי. אַבֹל אָחָת גֵּדוֲלֹה כוּ', אִי מְחַמָת בֵּרִיאוּת הוּא, תָּרֵוָיֵהוּ בֹעְי מִיבֵרֹא כוּ' אוֲ תָרֵוָיֵהוּ בֹעְי מִיכֵחָשִ. גֵּמֹרֹא. וֵעָיְּן תּוֲסֵפוֲת יוֲם טוֲב: וֵלֲא כוּ'. וֵעָד כָּמֹּה תֵהְא , תָּנֵיֹא כוּ' אַפִלּוּ אְינֹהּ בָּשִֵּנִיֹּה אּלֹּא כֵּפוֲל, כֵּשְִרֹה. וֵעָיְּן תּוֲסֵפוֲת יוֲם טוֲב:

כֹּךֵ הִגָּהֵתִּי מִפְּרוּשִ רָשִִ"י. שִּאְין לֵפֹרְשִ שִּדָּעֵתּוֲ כֵהֹרָמֵבָּ"ם שִּכֹּתָב כֹּל מָרֵבִּית עַגֹלִים כְּן וֵאְינוֲ מוּם לֵפִי שִּגֵּדְלִים לֵאָחָר מִכֹּאן, דֵּאִם כְּן הַוֹה לְיהּ לֵהֹרָ"ב לִכֵתֲּב נִמֵתֹּחוֲת עָד לֵמָטֹּה דֵּהָיֵנוּ לֵמָטֹּה מִן הֹעָרֵקוֲב: וֵאִי מִשִּוּם סְיפֹא פֵּסוּלְי הָמֶּקֵדֹּשִִין נִפֵדִּין עַלְיהּן, הֹא נָמִּי פֵּשִִיטֹא, מִשִֵחֹט שִֹחֵטִינָן, מִיפֵרֹק פֹּרֵקִינָן מִבָּעֵיֹא. גֵּמֹרֹא: הֹרָמֵבָּ"ם. וֵצֹרִיךֵ לוֲמָר דֵּלְית לֹן לֵמְימָר דֵּכֹלִיל טֵפְי מָהַנְי, מִשִּוּם דִּבֵהֹנְי דֵבְית דִּין שִּל אָחַרְיהּן אֹמֵרוּ שִּהְן מוּמִין. אַבֹל גִּרֵסֹתְנוּ בָּגֵּמֹרֹא, אָיֵדְי דִתֵנָן שִֵלֲשִֹה הוֲסִיף וֵאֹמֵרוּ לוֲ לֲא שִֹמָעֵנוּ אּת אְלּוּ, וֵקֹתֹנְי וֵאֹתְי כִיחִידֹאְי קֹא סֹתְים בֵּכֶלֵּהוּ עָל אְלּוּ מוּמִין שִוֲחַטִים כוּ':

בְּין בִּפֵנִים בְּין מִבָּחוּץ. גֵּמֹרֹא: וֵכְיוֹן דֵּיֹבְשִ מִבִּפֵנִים לֲא . רָשִִ"י. וֵנִמּוּקוֲ עִמּוֲ, שִּהַרְי יֹבְשִ בְּין מִבִּפֵנִים בְּין מִבָּחוּץ נָמִּי מוּם קֹבוּעָ הוּא כֵדִלֵעְיל, שִֵמָע מִנָּהּ דִּיבְשִוּת נָמִּי לֲא יוּכָל לֵהְרֹפְא. וּמִכֹּל מֹקוֲם גִּרֵסֹתְנוּ בָּגֵּמֹרֹא דֵמִצֵרִית לָח מִבִּפֵנִים וֵיֹבְשִ מִבָּחוּץ. וֵכְן כֹּתָב רָשִִ"י עָצֵמוֲ בֵּפְרוּשִ הָחֶמֹּשִ פֹּרֹשִָת תֹבֲא. תּוֲסֵפוֲת יוֲם טוֲב. וּמוֲהַרָ"מ שִֹם הּעֵתִּיק לֵשִוֲן רָשִִ"י בֵּהְפּךֵ, לָח מִבָּחוּץ. תּוֲסֹפוֲת חַדֹשִִים: וֵזֹקְן כוּ'. בָּגֵּמֹרֹא יֹלְיף לֵהוּ מִקֵּרֹא: וֵשִּנּּעּבֵדֹה כוּ'. דֵּתֹנוּ רָבֹּנֹן מִן הָבֵּהְמֹה, לֵהוֲצִיא רוֲבְעָ וֵנִרֵבֹּע. וּמִן הָצֲּאן, לֵהוֲצִיא אּת הָנּוֲגְחָ. וֵהֹרָ"ב לֲא דִּקֵדְּק בִּלֵשִוֲנוֲ, שִּפֹּתָח בֵּרוֲבְעָ וֵסִיְּם בֵּנִרֵבֹּע: וֵכְן רָבֹּנֹן נָמִּי לֲא קָיֵמוּ אּלֹּא אָאָנֵדֵּרוֲגִינוֲס, אַבֹל טֶמֵטוּם , מְטִיל מָיִם בִּמֵקוֲם זָכֵרוּת, זֹכֹר, וּבֹעְי לֵמְתֵבְיהּ לָכֲּהְן . הֹרֲא"שִ:
