וּבַמִּשְׁנָה דִלְעֵיל כָּתַבְתִּי דְּאִיכָּא דְסָבַר דְּמִנַּיְהוּ גוּפַיְהוּ יָכוֹל לְהַתְנוֹת. וְיֵשׁ לְתָרֵץ, דִּטְפֵי יֵשׁ לָחוּשׁ בִּשְׁנֵי סְלָעִים מִבְּהַרְבֵּה מָעוֹת. וְעוֹד יֵשׁ לוֹמַר דְּבִשְׁנֵי סְלָעִים מִסְּתָמָא אֵין אֶחָד חָרִיף מֵהַשֵּׁנִי, מַה שֶּׁאֵין כֵּן בְּהַרְבֵּה מָעוֹת דְּמַשְׁכַּחַת בְּהוּ חֲרִיפִים וְשֶׁאֵינָם חֲרִיפִים כְּמוֹ שֶׁכָּתַבְנוּ לְעֵיל, וְדִבְּרוּ חֲכָמִים בַּהוֶֹה:

לְשׁוֹן הַגְּמָרָא בָּבָא מְצִיעָא דַּף מ"ה: דְּזִמְנִין דְּלֹא מָלוּ זוּזֵי בְּדִינְרָא, , (וְעַיֵּן שָׁם בְּהָרֹא"שׁ דִּלְמַסְקָנָא גַם בֵּית הִלֵּל אוֹסְרִין, וְאוּלַי מַיְרֵי כָּאן בִּדְמַאי, וְעַיֵּן דְּמַאי פֶּרֶק א' מִשְׁנָה ב', מִשְׁנָה ח'):

פרק ב - משנה ח

כֵּן פֵּרֵשׁ רַשִׁ"י בְּבָבָא מְצִיעָא דַּף מ"ה, וְהִקְשׁוּ עָלָיו הַתּוֹסָפוֹת דַּהֲרֵי בְּמַסֶּכֶת עֲדֻיּוֹת פֶּרֶק קַמָּא תָּנֵי לַהּ בַּהֲדֵי דְבָרִים דְּבֵית שַׁמַּאי לְחֻמְרָא. לְכָךְ נִרְאֶה דְּבֵית שַׁמַּאי דַּוְקָא בְּכָל הַסֶּלַע קָאָמְרֵי, וְלֹא יַחֲלִיף שְׁתַּיִם וְשָׁלֹשׁ פְּעָמִים, שֶׁבְּכָל פַּעַם מִשְׂתַּכֵּר הַחֶנְוָנִי וְנִמְצָא מַפְסִיד מַעֲשֵׂר שֵׁנִי. וְעוֹד יֵשׁ לוֹמַר, דְּיִתֵּן מָעוֹת דַּוְקָא בְּכָל הַסֶּלַע, שֶׁלֹּא יִתְעַפְּשׁוּ בַּדֶּרֶךְ קֹדֶם שֶׁיַּגִּיעוּ שָׁם: מַשְׁמַע דְּטַעֲמָא דִשְׁרִיּוּתָא הוּא מִשּׁוּם שֶׁאֵין לוֹ בַּדִּינָר כֶּסֶף, וְאִלּוּ בַּפֵרוֹת לֹא אִכְפַּת לָן. וּבַיְרוּשַׁלְמִי אִיפְּכָא, דְּאָמַר רֵישׁ לָקִישׁ, מַה פְּלִיגִין רַבִּי מֵאִיר וְרַבָּנָן, בְּפֵרוֹת שֶׁאֵין בָּהֶם כְּדֵי כֶסֶף, אֲבָל בְּפֵרוֹת שֶׁיֵּשׁ בָּהֶן כְּדֵי כֶסֶף אַף רַבָּנָן מוֹדוּ. וּמְפָרֵשׁ הָרַ"מ הַטַּעַם, דִּכְשֶׁהַפֵּרוֹת מֻעָטִים מִדִּינָר אֵינָם רְאוּיִם לִפָּדוֹת בִּפְנֵי עַצְמָם, וּלְפִיכָךְ הֵם נִפְדִּים עִם הַכֶּסֶף. וְהָרַאֲבַ"ד מְפָרֵשׁ, דְּהוֹאִיל וְהַפֵּרוֹת מֻעָטִים וְאֵין מַתְפִּיסִים סֶלַע, מְצָרְפִים אוֹתָם עִם הַמָּעוֹת וּמַתְפִּיס סֶלַע עֲלֵיהֶם, אֲבָל אִם יֵשׁ בַּפֵּרוֹת דָּבָר הָרָאוּי אֵין זוֹקְקִים כֶּסֶף עִמָּהֶם לְחַלְּלָן עַל כֶּסֶף אַחֵר, שֶׁאֵין מְחַלְּלִין כֶּסֶף עַל כֶּסֶף:

לְפֵרוּשׁוֹ הֲוֵי מַתְנִיתִין בֵּית שַׁמַּאי לְקֻלָּא וּבֵית הִלֵּל לְחֻמְרָא. וְקָשֶׁה דִבְּעֵדֻיּוֹת תָּנֵי לַהּ בַּהֲדֵי הַנֵּי דְבֵית שַׁמַּאי לְחֻמְרָא (וּבְסֵפֶר בֵּית דָּוִד תֵּרֵץ דְּלֹא חָשִׁיב שָׁם מַה דִּפְלִיגֵי בֵּיהּ נַמִּי תַּנָּאֵי אַחֲרִינֵי). וְהָרַ"שׁ מְפָרֵשׁ, דְּהָכָא נַמִּי בֵּית שַׁמַּאי לְחֻמְרָא וְסָבְרֵי דְדַוְקָא בְּכָל הַסֶּלַע מָעוֹת, שֶׁלֹּא יִצְטָרְכוּ לִקְנוֹת (כְּלוֹמַר לְהַחֲלִיף) שְׁתֵּי פְעָמִים וְלִתֵּן שְׁתֵּי קַלְבּוֹנוֹת (דְּמֵי הַחְלָפָה לַשֻּׁלְחָנִי): וּבְסוֹף פֶּרֶק קַמָּא דְסַנְהֶדְרִין חָשִׁיב נַמִּי חֲנַנְיָא בֶן חֲכִינַאי, וְתוּ מוֹסִיף שִׁמְעוֹן בֶּן נַנָּס, וּפֵרֵשׁ רַשִׁ"י שֶׁכָּל אֵלּוּ לֹא בָאוּ לְזִקְנָה, וְיוֹשְׁבִין בַּקַּרְקַע וּמְשִׁיבִין תְּשׁוּבוֹת קֻשְׁיוֹת לַזְּקֵנִים, וְהַיְנוּ דָּנִין בַּהֲלָכָה:

דְּהַשְׁתָּא חָל הַחִלּוּל קֹדֶם שֶׁיַּתְחִילוּ לִשְׁתּוֹת, וְלֹא מֻפְלָג אֶלָּא סָמוּךְ בְּיוֹתֵר, בְּשָׁעָה שֶׁפֵּרֵשׁ הַיַּיִן מֵהַכַּד לְפִיהֶם. וְאֵינוֹ עִנְיָן לְדִין בְּרֵרָה. כֶּסֶף מִשְׁנֶה: מִכַּד אֶחָד. וְהוּא שֶׁיִּהְיוּ טְמֵאִים בִּשְׁאָר טֻמְאוֹת מִלְּבַד זִיבוּת, וְשֶׁיִּהְיֶה הַכְּלִי מֵחֶרֶס, וּבִתְנַאי שֶׁלֹּא יִגְּעוּ בַיַּיִן. וּכְלִי חֶרֶס אֵינוֹ מִטַּמֵּא מִגַּבּוֹ, אֲבָל אִם הָיוּ טְמֵאִים בְּזִיבוּת מִטַּמֵּא בְּהֶסֵּט. הָרַ"מ. וְדִתְנַן שׁוֹתִין, רוֹצֶה לוֹמַר זֶה אַחַר זֶה, בַּתְּחִלָּה הַטְּהוֹרִים וְאַחַר כָּךְ הַטְּמֵאִים, אֲבָל הַטְּמֵאִים רַשָּׁאִים לְהַשְׁקוֹת לַטְּהוֹרִים עַל פִּי אֵלּוּ הַתְּנָאִים. תּוֹסְפוֹת יוֹם טוֹב:
