פרק ד - משנה א

דַּהֲוָה מָצֵי לְמֵימַר יִהְיֶה: יְרוּשַׁלְמִי. וְהָכִי דָרִישׁ דְּאוֹתָם מִעוּטָא הוּא וּמִסְתַּבֵּר לְהוֹצִיא אֶת אֵלּוּ שֶׁיֵּשׁ בָּהֶם סַכָּנָה: וּמְחַלֵּק. שֶׁלֹּא יִרְאֶה לֶעָנִי קְרוֹבוֹ וְיַשְׁלִיךְ לְפָנָיו. יְרוּשַׁלְמִי: וְהָרַ"מ פֵּרֵשׁ אֱגוֹזֵי פֶרֶךְ וְהֵם מִין אֱגוֹזִים וְהוּא מְאֹד גָּבֹהַּ: תִּשְׁעִים וְתִשְׁעָה. פֵּרוּשׁ עֲנִיִּים: לָזֶה שׁוֹמְעִין. וַאֲפִלּוּ הוּא בָּרִיא וְחָזָק יוֹתֵר מֵחֲבֵרָיו אוֹ אוֹכֵל יוֹתֵר מֵהֶם:

פרק ד - משנה ב

שֶׁאָמַר כַּהֲלָכָה. וַאֲפִלּוּ הוּא זָקֵן וְתַשׁ כֹּחַ אוֹ חוֹלֶה, שֶׁהֲרֵי מַה שֶּׁאוֹמֵר לְחַלֵּק מִפְּנֵי חֲלִישׁוּתוֹ הוּא, אֲפִלּוּ הָכִי שׁוֹמְעִין לוֹ. שָׁם:

פרק ד - משנה ג

רַשִׁ"י. וְלֹא נִתְכַּוְּנוּ לוֹמַר דִּבְהָכִי הֲוֵי קִנְיָן, אֶלָּא רְצוֹנוֹ לוֹמַר לְפִי כַּוָּנַת הֶעָנִי, אֲבָל לְפִי הָאֱמֶת אֵין זֶה קִנְיָן כְּלָל: שֶׁאַחֲרֵי שֶׁכִּסָּה הַפֵּאָה שֶׁלֹּא יוּכְלוּ הָעֲנִיִּים לִרְאוֹתָהּ לְקַחְתָּהּ, וְהוּא לֹא קְנָאָהּ בָּזֶה שֶׁאֵין זֶה קִנְיָן כְּלָל, לְכָךְ קוֹנְסִין אוֹתוֹ אַף מַה שֶּׁלָּקַח וְזָכָה בָהּ. הָרַ"מ. וּלְשׁוֹן בַּעַל כַּף נַחַת, כְּדֵי שֶׁלֹּא יַעֲשֶׂה כֵּן פַּעַם אַחֶרֶת: וּפֵרֵשׂ. דַּעַת הָרַ"מ דִּבְנָפַל אוֹ בְּפֵרֵשׂ גַּם כֵּן קוֹנְסִים אוֹתוֹ, וְאֵין נִרְאֶה כֵּן מִן הַמִּשְׁנָה אֶלָּא דַּוְקָא כְּשֶׁנָּטַל וּזְרָקוֹ כוּ' שֶׁהֲרֵי חוֹלְקָם לִשְׁתֵּי בָּבוֹת. תּוֹסְפוֹת יוֹם טוֹב: בְּעֹמֶר הַשִּׁכְחָה. הוּא הַדִּין בְּשִׁכְחַת קָמָה, אֶלָּא דְשִׁכְחַת עֹמֶר בְּהֶדְיָא כְּתִיב וְשָׁכַחְתָּ עֹמֶר, וְשִׁכְחַת קָמָה מֵרִבּוּי קָאָתֵי:

פרק ד - משנה ד

בְּמַגָּלוֹת. הֵם חֶרְמֵשִׁים, וְקַרְדֻּמּוֹת הֵם מִמִּינֵי הַפַּטִּישִׁים. הָרַ"מ: שֶׁלֹּא יַכּוּ אִישׁ אֶת רֵעֵהוּ. מֵרֹב עַם הַבָּאִים לִלְקֹט וְדוֹחֲקִים זֶה אֶת זֶה אוֹ שֶׁמָּא יִתְקוֹטְטוּ:

פרק ד - משנה ה

שֶׁבְּשַׁחֲרִית וּמִנְחָה יְשֵׁנִים. הָרַ"שׁ. וְהִנֵּה הִשְׁאִילוּ לַפֵּאָה שֵׁם לֶקֶט. וְעַיֵּן רֵישׁ פֶּרֶק ח'. הָרַ"שׁ (עַל שֵׁם שֶׁהָעֲנִיִּים מְלַקְּטִין אוֹתָהּ. תּוֹסְפוֹת יוֹם טוֹב): הָרַ"שׁ. וּפֵרֵשׁ רַבֵּנוּ זַ"ל בְּגוּר אַרְיֵה (פָּרָשַׁת דְּבָרִים) לְפִי שֶׁהַשּׁוּרָה מְשׁוּכָה בְּשָׁוֶה וְזֶה אֻמָּנוּת נִקְרֵאת אֻמָּן וּבַפֵּרוּשׁ הַנִּזְכָּר לְעֵיל כָּתַב פְּאַת רָאשֵׁיכֶם תִּרְגֵּם יוֹנָתָן אֻמָּנָה רֵישְׁכוֹן: יְרוּשַׁלְמִי. וְטַעֲמָא שֶׁיּוּכְלוּ הָעֲנִיִּים לָקַחַת מִכָּל אֻמָּן וְאֻמָּן וְלֹא יִצְטָרְכוּ לְהַמְתִּין וּלְצַפּוֹת אֵימָתַי יַנִּיחוּ הַפֵּאָה. תּוֹסְפוֹת יוֹם טוֹב:

פרק ד - משנה ו

מִן הַלֶּקֶט. תַּנָּא נָקְטִינְהוּ בְּהוֶֹה וְרָגִיל לִהְיוֹת, שֶׁכֵּן בִּתְחִלַּת הַקְּצִירָה אֶפְשָׁר וּשְׁכִיחַ לִהְיוֹת נוֹשֵׁר, אֲבָל שִׁכְחַת עֹמֶר וְקָמָה אֵינוֹ רָגִיל שֶׁיִּשְׁכַּח בִּתְחִלַּת הַקְּצִירָה, וְהָדָר תָּנִי פֵּאָה שֶׁמְּקוֹמָהּ בְּסוֹף הַשָּׂדֶה. אֲבָל בַּקְּרָאֵי כְּתִיבֵי בְּהֵפֶךְ כְּסֵדֶר אַזְהָרָתָן, כְּשֶׁקּוֹצֵר מַזְהִירוֹ שֶׁלֹּא יִקְצֹר כָּל שָׂדֵהוּ, וּכְשֶׁקּוֹצֵר דֶּרֶךְ לְלַקֵּט הַנּוֹשֵׁר, וְאִם שָׁכַח עֹמֶר וְקָמָה דְּלֹא שְׁכִיחַ כוּ'. תּוֹסְפוֹת יוֹם טוֹב:

פרק ד - משנה ז

חַיָּב. בְּשִׁכְחָה. הָרַ"מ:

פרק ד - משנה ח

וְכֵן כָּתַב הָרַ"שׁ, וְנָקְטֵי חִיּוּב דִּתְבוּאָה דְּבָהּ אַיְרִינַן בְּפִרְקִין. אֲבָל לְמַאי דְתָנִינַן פֵּירוֹתָיו מְדֻיָּק פֵּרוּשׁ הָרַ"מ עוֹנַת וְכוּ' וּזְמַנָּם מִשְׁתַּנֶּה כְּפִי שִׁנּוּי הַזְּרָעִים, וְכֵן כָּתַב הָרַ"ב (וְהָרַ"שׁ) בְּפֵרוּשׁ מִשְׁנָה זוֹ שֶׁשְּׁנוּיָה בְּפֶרֶק ג' מִשְׁנָה ד' דְּחַלָּה:

פרק ד - משנה ט

שֶׁנִּמְצָא רִאשׁוֹן. כְּלוֹמַר שֶׁלֹּא יְעַכְּבֵם עַד שֶׁיָּבֹא אוֹתוֹ עָנִי שֶׁרָצָה לְזַכּוֹתוֹ, אֶלָּא כָּל עָנִי שֶׁיִּמְצָא רִאשׁוֹן יִתְּנֶנָּהּ לוֹ וְאַל יְאַחֲרָהּ בְּיָדוֹ, וְאִם אֵרַע שֶׁיָּבֹא אוֹתוֹ עָנִי רִאשׁוֹן אֵין הָכִי נַמִּי דְּרַשַּׁאי לְתִתּוֹ לוֹ, שֶׁאֵין כָּאן קְנָס כְּדִלְעֵיל מִשְׁנָה ג', דְּאוֹתוֹ עָנִי לֹא עֲבֵיד וְלֹא מִידֵי:

פרק ד - משנה י

רְצוֹנוֹ לוֹמַר תּוֹךְ הַיָּד קָאֵי אַנָּשַׁר, וּכְאִלּוּ תָּנֵי תּוֹךְ הַיָּד וְנָשַׁר, וְכֵן נַמִּי תּוֹךְ הַמַּגָּל כְּאִלּוּ תָנֵי תּוֹךְ הַמַּגָּל וְנָשַׁר: רְצוֹנוֹ לוֹמַר שֶׁמַּה שֶּׁיִּפּוֹל מֵחֲמַת נִדְנוּד הַיָּד וְיִפֹּל לַאֲחוֹרֵי הַיָּד הֲרֵי הוּא אֹנֶס, וְכֵן מֵחֲמַת נִדְנוּד הַמַּגָּל וְיִפֹּל אֲחוֹרֵי הַמַּגָּל גַּם הוּא אֹנֶס: כְּלוֹמַר בְּמָקוֹם שֶׁרָאשֵׁי אֶצְבְּעוֹתָיו מֻנָּחִים עַל הַפַּס יָד. וְאֵין זֶה תּוֹךְ הַיָּד, גַּם אֵין זֶה אַחַר הַיָּד:

פרק ד - משנה יא

דְּתַנָּא קַמָּא נַמִּי סְבִירָא לֵיהּ סְפֵק לֶקֶט לֶקֶט, וְלֹא פָּלֵיג אַדְּרַבִּי מֵאִיר אֶלָּא בַּתַּחְתּוֹנִים, מִשּׁוּם דִּסְבִירָא לֵיהּ דְּוַדַּאי מִשֶּׁל קָמָה כְּנָסוּם: קְצַת קָשֶׁה מַה יָּשָׁן שַׁיָּךְ בֵּין קָמָה לַקָּצִיר, וְנִרְאֶה לִי שֶׁדְּבָרָיו מֻרְכָּבִים, שֶׁמַּה שֶּׁכָּתַב דְּמִן הַקָּמָה הָיָה הוּא מְהָרַ"שׁ שֶׁכָּתַב הַיְרוּקִים כְּכַרְתִי כוּ' שֶׁעֲדַיִן לֹא נִגְמְרוּ כוּ' וְלֹא נִתְבַּשְּׁלוּ כְּדֵי לִקְצֹר וּכְשֶׁהַתְּבוּאָה יְבֵשָׁה וּמִתְלַבֶּנֶת קוֹצְרִים אוֹתָהּ. וּמַה שֶּׁכָּתַב נוֹטִים לְהַשְׁחִיר הוּא מֵהָרַ"מ וְהוּא כָּתַב וּמִזֶּרַע הַשָּׁנָה שֶׁעָבְרָה הוּא. תּוֹסְפוֹת יוֹם טוֹב: מִכָּל מָקוֹם הוּצְרַךְ לְפָרֵשׁ שֶׁאִי אֶפְשָׁר לַגֹּרֶן כוּ' דְּאִם לֹא כֵן מַה סְפֵקָא אִיכָּא, כְּמוֹ לְתַנָּא קַמָּא אוֹת כ"ד: וּבְרֵישׁ פִּרְקִין דָּרְשִׁינַן הַנַּח לִפְנֵיהֶם וְהֵם יָבוֹזוּ. וְנִרְאֶה לִי דְּאִי לְהַךְ לְחוּדָא אָתָא הֲוָה לֵיהּ לִכְתֹּב עֲזֹב, וּכְתִיב תַּעֲזֹב, מַשְׁמַע אַתָּה מִשֶּׁלְּךָ עֲזֹב, וְאִי לְהַךְ לְחוּדָא לִכְתֹּב תִּתֵּן, אֶלָּא יָבוֹזוּ כְּדִלְעֵיל נַמִּי. תּוֹסְפוֹת יוֹם טוֹב:
