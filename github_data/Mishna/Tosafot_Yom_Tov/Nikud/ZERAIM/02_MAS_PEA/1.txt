פרק א

אְלּוּ דֵבֹרִים. הֹא דֵלֲא תֵנָן תֵּרוּמֹה עִמֹּהּן כְּיוֹן דֵּאִיכֹּא תֵּרוּמָת מָעַשְֵר שִּיְּשִ לֹהּ שִִעוּר לֲא נֹקָט אָף סֵתֹם תֵּרוּמֹה. יֵרוּשִָלֵמִי. וֵעוֲד יְשִ לוֲמָר שִּהִיא חַלוּקֹה מִן הָשִֵּאֹר כֵּדִמֵפֹרְשִ אָחָר כֹּךֵ בֵּפְאֹה בִּקְּשִ לָעַשֵוֲת כֹּל שֵֹדְהוּ פְּאֹה עוֲשֵּה, בִּקְּשִ לָעַשֵוֲת כֹּל כִּרֵיוֲ תֵּרוּמֹה אְינוֲ עוֲשֵּה. (וֵעוֲד יְשִ לוֲמָר דֵּלֲא תֹנְי רָק מָה שִּּיְּשִ בֵּהוֲסֹפֹתֹן מִצֵוֹה כֵּדִמֵשִֹנְי בָּיֵרוּשִָלֵמִי אָאְינֹךֵ שִּלֲּא נִשִֵנוּ עָיְּן שִֹם, וּתֵרוּמֹה גֵדוֲלֹה אָדֵּרָבֹּא מָאַכִיל טּבּל לָכֲּהְן שִּמּוֲסִיף אָחָר שִּנֹּתָן כָּשִִּעוּר. הֹרָ"שִ. וֵצֹרִיךֵ עִיּוּן הֹא פְּאֹה נָמִּי דִּכֵוֹתֹהּ, עָיְּן לֵהָלֹּן מִשִֵנֹה ג' וֵצֹרִיךֵ עִיּוּן). תּוֲסֵפוֲת יוֲם טוֲב: הָפְּאֹה. וֵעוֲשֵּה מִצֵוֹה בֵּהוֲסֹפֹתוֲ. מִיהוּ הְיכֹא דֵנֹתָן פְּאֹה כֵּדִינוֲ בֵּסוֲף שֵֹדְהוּ חָיֹּב בֵּמָעָשֵֵרוֲת מָה שִּּהוֲסִיף: וֵאָף מִדֵּרָבֹּנֹן אְין לֹהּן שִִעוּר: תָּקֹּנָת אוּשִֹא הוּא, בֵּמָסּּכּת כֵּתֶבּוֲת דָּף נ'. וֵהָשִֵתֹּא מִדֵּאוֲרָיֵתֹא אְין שִִעוּר לִגֵמִילוּת חַסֹדִים בֵּמֹמוֲנוֲ: שִּאֹדֹם אוֲכְל. כֹּתָב הֹרָ"מ כִּי הָמִּצֵוֲת כֶּלֹּם נּחֱלֹקִים בִּתֵחִלֹּה עָל שִֵנְי חַלֹקִים. הֹאּחֹד, בֵּמָה שִּּיְּשִ בְּינוֲ וּבְין הָקֹּדוֲשִ בֹּרוּךֵ הוּא כֵּגוֲן צִיצִית וּתֵפִילִין כוּ'. וֵהָשְִּנִי, מִצֵוֲת הָתֵּלוּיוֲת בֵּתוֲעּלּת בֵּנְי אֹדֹם קֵצֹתֹם עִם קֵצֹתֹם כֵּגוֲן הֹאָזֵהֹרֹה עָל הָגֵּנְבֹה וּכֵגוֲן הָצִּוּוּי בֵּאָהַבָת אִישִ אּת רְעְהוּ. וּכֵשִּיָּעַשֵּה הָמִּצֵוֲת שִּבָּחְלּק הֹרִאשִוֲן יִגֵמֵלְהוּ הָקֹּדוֲשִ בֹּרוּךֵ הוּא עַלְיהּם לֹעוֲלֹם הָבֹּא. וּכֵשִּיָּעַשֵּה הָמִּצֵוֲת שִּבָּחְלּק הָשְִּנִי תְּחֹשְִב לוֲ לִצֵדֹקֹה לֹעוֲלֹם הָבֹּא וֵיִמֵצֹא טוֲבֹה בֹּעוֲלֹם הָזּּה בָּעַבוּר שִּנֹּהָג מִנֵהָג הָטּוֲב בְּין בֵּנְי אֹדֹם כוּ', וֵכֹל הָמִּצֵוֲת שִּבְּין אֹדֹם לָחַבְרוֲ נִכֵנֹסוֲת בִּגֵמִילוּת חַסֹדִים: וֵהָקּּרּן. וֵהְשִִיב אּת אַשִֹמוֲ בֵרֲאשִוֲ תָּרֵגּוּם יֵרוּשִָלֵמִי אֹשִֹם קּרּן. עֹרוּךֵ. וֵהֹרָמֵבָּ"ן כֹּתָב שִּבָּעַבוּר שִּהָכֲּחָ בָּקָּרֵנָיִם יִקֹּרְא גוּף הָדֹּבֹר קָרֵנוֲ: כִּבּוּד אֹב וֹאְם. כֵּתִיב לֵמָעָן יָאַרִיכוּן יֹמּיךֹ וּלֵמָעָן יִיטָב לֹךֵ, חָד בֹּעוֲלֹם הָזּּה וֵחָד בֹּעוֲלֹם הָבֹּא. בִּגֵמִילוּת חַסֹדִים כֵּתִיב רוֲדְף צֵדֹקֹה וֹחּסּד יִמֵצֹא חָיִּים צֵדֹקֹה וֵכֹבוֲד. בָּהַבֹאָת שִֹלוֲם כֵּתִיב בָּקְּשִ שִֹלוֲם וֵרֹדֵפְהוּ כוּ' אָתֵיֹא רֵדִיפֹה מִגֵּמִילוּת חַסֹדִים. בֵּתָלֵמוּד תּוֲרֹה כֵּתִיב כִּי הוּא חָיּּיךֹ וֵאֲרּךֵ יֹמּיךֹ: וֵתָלֵמוּד תּוֲרֹה כוּ'. כִּי בֵּתָלֵמוּד תּוֲרֹה יִזֵכּּה הֹאֹדֹם בֵּכֹל זּה, שִּהָתָּלֵמוּד מְבִיא לִידְי מָעַשֵּה. הֹרָ"מ:

וּתֵמְהָנִי לֹמֹּה יוֲסִיף שִּהַרְי הָפְּאֹה נָמִּי גֵּדוֲלֹה כֵּשִּהִיא שִִשִִּים מְהָשֵֹּדּה הָגֵּדוֲלֹה וָהַרְי הֹעַנִיִּים מֶעֹטִים. וֵהֹרָ"שִ לֲא כֹּתָב דֵּמוֲסִיף אּלֹּא שִּלֲּא יִפֵחוֲת מִשִִּשִִּים אָף שִּהֹעַנִיִּים מֶעֹטִים:

נוֲתֵנִין כוּ'. וֵלֲא אֹמֵרִינָן דִּפֵאָת שֵֹדֵךֹ מָשִֵמָע דָּוֵקֹא בָּסּוֲף כִּדֵמָשִֵמָע לֵשִוֲן פְּאֹה, וֵאִם נֹתָן מִתֵּחִלֹּה וּמְאּמֵצָע לֲא תִקֵדָּשִ בֵּדִין פְּאֹה וּלֵחָיְּב בֵּמָעַשְֵר. הֹרָ"שִ: וֵהָתּוֲסֵפוֲת יוֲם טוֲב הִקֵשִֹה עָל זּה, וּמֵחֶוֹּר פְּרוּשִ הֹרָ"שִ אִם שִִיְּר קּלָח אּחֹד כוּ', כֵּשִּנֹּתָן שִִעוּר פְּאֹה וּמֵבָקְּשִ לֵהוֲסִיף אָיֵרְי דֵּיֹכוֲל לֵהוֲסִיף עָל הָמֵחֶבֹּר וֵיְשִ לוֲ תּוֲרָת פְּאֹה לִפֵטֲר מִן הָמָּעַשְֵר, וֵאִם לֹאו, כֵּלוֲמָר שִּכֵּבֹר תֵּלֹשִוֲ, אְינוֲ נוֲתְן אּלֹּא מִשִּוּם הּפֵקְר, דֵּכֹל מָה שִּּמּוֲסִיף מִן הָתֹּלוּשִ אְין לוֲ תּוֲרָת פְּאֹה וֵאְינוֲ נִפֵטֹר מִן הָמָּעַשְֵר עָד שִּיָּפֵקִיר. וֵעָיְּן עוֲד: וֵנָפֵקֹא מִנָּהּ דֵהּפֵקְר בְּין לָעַנִיִּים בְּין לָעַשִִירִים, מָה שִּּאְין כְּן פְּאֹה דֵּלָעַנִיִּים כוּ'. אַבֹל לֵעִנֵיָן מָעַשְֵר פְּאֹה נָמִּי פֹּטוּר כֵּדִתֵנָן בֵּסוֲף פִּרֵקִין. תּוֲסֵפוֲת יוֲם טוֲב:

וֵנֹקָט סֵפִיחְי וֵלֲא סִטִּיס עָצֵמוֲ דִּפֵשִִיטֹא דֵּלֲא חֹזְי כֵּלֹל. הֹרָ"שִ. וֵעָיְּן תּוֲסֵפוֲת יוֲם טוֲב: פְּרוּשִ בֵּאִילֹן אּחֹד בֵּעָצֵמוֲ, וֵזּה לֵשִוֲן הֹרָ"מ, יְשִ בֵּאִילֹן זּה מָה שִּּנִּגֵמָר בּוֲ הָיּוֲם וֵיְשִ בּוֲ מָה שִּּנִּגֵמָר לֵאָחָר כָּמֹּה יֹמִים: לֵשִוֲן הֹרָ"שִ, דִּסֵתֹם קֹצִיר מִתֵבָּשְִּל בֵּיָחָד: רֲב יֹרֹק קֹאֹמָר, דֵּיְשִ יֹרֹק חָיֹּב בֵּפְאֹה כֵּגוֲן שִוּמִים וּבֵצֹלִים כֵּדִלֵקָמֹּן פּּרּק ג' מִשִֵנֹה ג' הָנֵהוּ מָכֵנִיסֹן לֵקִיּוּם. הֹרָ"שִ: וּפְרוּשִוֲ דֹּחוּק. וֵהֹרָ"שִ כֹּתָב עָל פִּי תּוֲרָת כֲּהַנִים, וּסֵתֹם קֹצִיר מָכֵנִיסֹן לֵקִיּוּם, יֹצֵאוּ יֵרֹקוֲת שִּאָף עָל פִּי שִּלֵּקִיטֹתוֲ כֵּאּחֹד אְינוֲ מָכֵנִיסוֲ לֵקִיּוּם: וֵהָתֵּבוּאֹה. בָּיֵרוּשִָלֵמִי דֹּרֵשִִינָן וּבֵקֶצֵרֵכּם אְין לִי אּלֹּא תֵּבוּאֹה קִטֵנִיּוֲת מִנָּיִן תָּלֵמוּד לוֲמָר אָרֵצֵכּם: וֵדִבֵרְי תְימָהּ הְם דָּהַרְי וָדָּאי שִּנִּשִֵלֵמוּ בֹהּם כוּ'. וּלֵשִוֲן הֹרָ"מ בֵּחִבּוּרוֲ כֹּל הָדּוֲמּה לֵקֹצִיר בֵּחֹמְשִ דֵּרֹכִים אְלּוּ הוּא שִּחָיֹּב בָּפְּאֹה כֵּגוֲן הָתֵּבוּאֹה כוּ', וֵאוּלָי מִשִּוּם נִשִֵמֹר כֹּתָב כְּן. תּוֲסֵפוֲת יוֲם טוֲב:

אַבֹל בֵּרְישִ דֵּמָאי כֹּתָב בִּשִֵמוֲ שִּהְם אִילֹנוֲת, וּלֵכֹךֵ יְשִ לוֲמָר דֵּהֹכֹא גָּם כְּן אִילֹן, וּבָיֵרוּשִָלֵמִי אִילֹנוֲת מִנָּיִן תָּלֵמוּד לוֲמָר שֵֹדֵךֹ. וֵאִיכֹּא דֵיֹלִיף מִכּּרּם וֵזָיִת:

וּבֵפּרּק ד' מִשִֵנֹה ח' פְּרְשִ כֵּשִּהוּא מָשִֵוּה פֵּנְי הָכֵּרִי בֵרָחָת. פְּרוּשִ מֵרִיחֹה מֵרִיקֹה, וֵהוּשִֵאַלֹה זֲאת הָמִּלֹּה בֵּעִנֵיָן הָשִֵלֹמָת צֹרֵכְי הָזּּרָע עָד שִּלֲּא יִשִֹּאְר בּוֲ אּלֹּא שִּיּוֲלִיכוֲ אּל בְּית הֹאוֲצֹר. הֹרָ"מ. וֵהוּא דּוֲמּה כֵּמוֲ הָמֵמֹרְק לֵכְלִים רְישִ פּּרּק י' דֵּמִקֵוֹאוֲת (וֵעָיְּן שִֹם בֹּרָ"מ) (וֵעָיְּן יוֲמֹא פּּרּק ג' מִשִֵנֹה ד'), שִּעוֲשִֵין כֵּשִּנִּשִֵלָם הָכֵּלִי: נִרֵאּה לִי דֵסֹמָךֵ אָמֹּה שִּכֹּתוּב גָּבְּי פְּאֹה שִּצֹּרִיךֵ שִּיָּפֵרִישִ תֵּרוּמֹה וּמָעַשְֵר תֵּחִלֹּה וֵאָחָר כֹּךֵ יִטֲּל הָפְּאֹה, הוֲרֹה לֹנוּ שִּהָמִּצֵוֹה הִיא שִּיָּפֵרִישִ כוּ', וֵכִדֵדֹרֵשִִינָן מִשִֵנֹה י"א פּּרּק ה' דֵּמָעַשְֵר שְִנִי הֹא אִם הִקֵדִּים כוּ'. אִי נָמִּי פּּן לֲא יָפֵרִישִוּ הֹעַנִיִּים וֵהָמָּכֵשְִלֹה הָזֲּאת תָּחָת יֹדוֲ יִהֵיּה. וֵהֹכִי נָמִּי בֵּהּפֵקְר דִּכֵוֹתֹהּ בֵּפְאֹה וֵעָיְּן מָה שִּכֹּתוּב מִשִֵנֹה ב' פּּרּק ג' דִּדֵמָאי. תּוֲסֵפוֲת יוֲם טוֲב: וֵיִקֵנוּ טֵבֹלִים כֵּדְי לֵהָפֵקִיעָ מָתֵּנוֲת אַחְיהּם. הֹרָ"מ: וֵאָשִֵמֵעִינָן הָמָּקֵדִּישִ דֵּאָף עָל גָּב דֵּמִתֵחִלֹּה הֹיֵתֹה גָם כְּן הָתֵּבוּאֹה שִּלּוֲ, הַוֹה אַמְינֹא דֵכִי הֹדֹר וּפֵדֹאֹהּ אָף עָל גָּב דִּלֵאָחָר מְרוּחָ הוּא שִָפִּיר קֹרִינָן בְּיהּ תֵּבוּאָת זָרֵעַךֹ, קֹא מָשִֵמָע לֹן דֵּלֲא. תּוֲסֵפוֲת יוֲם טוֲב:
