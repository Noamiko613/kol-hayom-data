פרק ט

יֵרוּשִָלֵמִי כֵּלוֲמָר אְילִים וֵכֹרִים, וֵצּמּר מוּסֹב עָל שִֵנְיהּם. וּשִֵנְיהּם מִין אּחֹד, אּלֹּא שִּאְילִים זֵכֹרִים וֵכֹרִים נֵקְבוֲת: צּמּר וּפִשִֵתִּים. בֵּמִשִֵנֹה ג' פּּרּק י"א דִּנֵגֹעִים תֵּנָן דִּבֵגֹדִים צֵבוּעִים אְינֹן מֵטָמֵּאִין בִּנֵגֹעִים, וּלֵעִנֵיָן כִּלֵאָיִם בְּין לֵבֹנִים בְּין צֵבוּעִים, וֵכְן בִּגֵדְי כֲּהַנִים לֵבֹנִים וּצֵבוּעִים. וּבֵהֹא לֲא דָמֵיֹן הֹנְי כֵלֹלְי לָהַדֹדְי: רֵצוֲנוֲ לוֲמָר כֲּהַנִים הּדֵיוֲטִים, וּלֵהֹכִי נֹקָט בֹּתָר הֹכִי אָבֵנְט, דֵּאִלּוּ בֵּבִגֵדְי כֲהְן גֹּדוֲל אִיכֹּא נָמִּי מֵעִיל אְפוֲד וֵחֲשִּן: וֵיֹלֵפִינָן לְיהּ בֵּפּרּק קָמֹּא דֵחֶלִּין דָּף י"א מְאָחַרְי רָבִּים לֵהָטּוֲת דֵּכֹל הֹאִסּוּרִים בֵּטְלִים בֵּרֶבֹּא דֵּאִיתֹא קָמֹּן, וּבֵמִידְי דֵיֹבְשִ בֵּיֹבְשִ מִין בֵּמִינוֲ אַפִלּוּ מִדֵּרָבֹּנֹן אְין צֹרִיךֵ יוֲתְר. וּבֹתָר שִֵמֹא אֹזֵלִינָן בֵּמִין בֵּמִינוֲ, וָאַפִלּוּ אִית לֵחָד שְִם לֵוָי, כִּדֵאִיתֹא בֵּפּרּק בָּתֵרֹא דָעַבוֲדֹה זֹרֹה:

וֵיִדֵמוּ כֵּאִלּוּ הְם צּמּר וּפִשִֵתִּים מִפֵּנְי שִּהֹאּחֹד מְהָמִּינִים הֹהְם חֹלֹק כֵּמוֲ פִּשִֵתֹּן וֵהָשְִּנִי יְשִ בּוֲ יֵבְשִוּת וֵהוּא שֵֹעִיר כֵּמוֲ הָצּּמּר: הוּא לֵשִוֲן הֹרָמֵבָּ"ם. וּבֵפּרּק בָּמּּה מָדֵלִיקִין כֹּתָב פְּרוּשִ אָחְר: אְינוֲ מוּסֹב עָל וֵהֹאִידֵנֹא אּלֹּא בֹּא לוֲמָר דֵּהָקָּנֵבּוֲס שִֹרְי לֵעוֲלֹם שִּהָכֲּל מָכִּירִין בּוֲ: יֵרוּשִָלֵמִי. וּפְרְשִ הֹרָ"שִ מִטֹּה שִּל חַבֹלִים שִּנִּשִֵקָע תָּחֵתֹּיו וֵצִדְּי הָכָּר עוֲלִים עֹלֹיו, אַבֹל נֵסֹרִים שִּל עְץ קֹשִּה וֵאְינוֲ נִשִֵקֹע: נוֲגְעָ בֹּהּן. דֵּאֹז חָיֵשִִינָן שִּמֹּא תִכֹּרְךֵ כוּ' כֵּמוֲ שִּכֹּתָב הֹרָ"ב מִשִֵנֹה ד': וּבֵרְישִ הָתֵּכְלּת מָיֵתְי לָהּ, וּמָשִֵמָע דֵּאָיֵרְי בֵּטָלִּית קֵטָנֹּה אָף עָל פִּי שִּאְין הָגֹּדוֲל יוֲצְא בֹהּ עַרָאי. הֹרָ"שִ: וֵהֹרָמֵבָּ"ם לֲא פְּרְשִ דִּפֵלִיגְי, וֵכְן בֵּחִבּוּרוֲ פֹּסָק לֵתָרֵוָיֵהוּ. וֵכֹתָב בָּכּּסּף מִשִֵנּה שִּטָּעַמוֲ דֵּהֹכֹא שִֹאנְי דֵּכְיוֹן שִּהוּא לוֲבְשִ מָמֹּשִ הַרְי עֹבָר עָל לֲא תִלֵבָּשִ שִָעָטֵנְז, אַבֹל מוֲכֵרְי כֵסוּת שִּאְינֹם לוֲבֵשִִים מָמֹּשִ אּלֹּא מָעַלִים עַלְיהּם מֶתֹּר אִם אְינוֲ מִתֵכָּוְּן:

וֵאְינֹם חָיֹּבִים הָחָיִּים לִנֵהֲג בֹּהּם אִסּוּר וֵלֲא שִוּם מִצֵוֹה מִמִּצֵוֲת הָכֹּתוּב כֵּגוֲן צִיצִית וּתֵפִלִּין וּמֵזוּזוֲת וֵדוֲמּה לֹאְלּוּ. הֹרָמֵבָּ"ם: עָל כֵּתְפוֲ. וֵטָעַמֹא מִשִּוּם דֵּלֲא שִֹרִינָן מִפֵּנְי שִּהוּא קֹשִּה אּלֹּא בֵּהָצֹּעֹה שִּהוּא מִדֵּרָבֹּנֹן, אַבֹל לֲא בֵהָעַלֹאֹה שִּהוּא מִדֵּאוֲרָיֵתֹא. הֹרָ"מ:

מֵסָיְּם הֹרָ"שִ וּפֵלִיגְי אָסֵּתֹמֹא דִלֵעְיל מִשִֵנֹה ב':

וֵלֲא יֹדָעֵנוּ טָעַמֹא מָאי דֵאְין חוֲשִֵשִִין שִּיִּהֵיוּ עוֲשְֵי הָמִּנֵעֹל הָהוּא נִזֵהֹרִים לִמֵנֲעָ אַפִלּוּ חוּט אּחֹד שִּל פִּשִֵתֹּן. וֵהֹרָמֵבָּ"ם בֵּחִבּוּרוֲ כֹּתָב מִנֵעֹל שִּהוּא כִּלֵאָיִם וֵאְין לוֲ עֹקְב וּמֶתֹּר לֵפִי שִּעוֲר הֹרּגּל קֹשִּה הוּא וֵאְינוֲ נּהֱנּה כִּשִֵאֹר עוֲר הָגּוּף, וּמֵפֹרְשִ הָכּּסּף מִשִֵנּה דֵּפְרוּשִ זּרּב הָיֵנוּ שִּבִּימוֲת הָחָמֹּה לוֲבֵשִִים אוֲתוֲ וֵיְשִ בּוֲ פִּשִֵתֹּן וֵלִפֵעֹמִים מֵשִֵימִים בּוֲ גָּם כְּן צּמּר כֵּדְי לֵלֹבֵשִוֲ בִּימוֲת הָקֲּר, וּמָה שִּּכֹּתָב אְין לוֲ עֹקְב הוּא דֵּצוּרֹתוֲ כֹּךֵ הוּא:

שִוּעָ טֹווּי וֹנוּז. תְּימָהּ דֵּפֹתָח בֵּטֹווּי וֵאֹרוּג וֵסִיְּם בֵּשִוּעָ טֹווּי וֹנוּז. וּבֵתוֲסֹפוֲת הִקֵשִֹה גָם כְּן לָפְּרוּשִ הֹרִאשִוֲן שִּבֵּדִבֵרְי הֹרָ"ב דֵּמָתֵנִיתִין מָשִֵמָע דֵּאֹרוּג הָיֵנוּ נוּז כֵּדִתֵנָן שִּנּּאֱמָר וֵכוּ'. וֵאוֲמְר רָבְּנוּ תָּם דֵּעָל כֹּרֵחַךֹ לֹאו לֵמִידֵרֹשִ כֹּל שִָעָטֵנְז קֹאֹתְי וֵכוּ'. וֵאִם תֲּאמָר אָמָּאי תֹּנְי אֹרוּג, יְשִ לוֲמָר דִּכֵמוֲ דֵתֹנְי טֹווּי דֵּמִנּוֲטֵרִיקוֲן נָפֵקֹא, הֹכִי נָמִּי תֹּנְי אֹרוּג דֵּמִדֵּרֹשִֹא דֵיָחֵדֹּו נָפֵקֹא. וֵשִוּעָ וֹנוּז דִּכֵתִיבְי בֵּהּדֵיֹא לֲא הֶצֵרָךֵ לֵהָתֵנוֲת בֹּהּם בֵּרְישִֹא. וֵעָיְּן בָּאַרִיכוּת בֵּתוֲסֵפוֲת יוֲם טוֲב:

וֵהֹרֲא"שִ בֵּמָסּּכּת נִדֹּה כֹּתָב דֵּהָךֵ שִֹרְי טֵפְי:

וֵהָקֶּפֹּה. שִּהוּא גָּם כְּן דּּרּךֵ לֵלֹבֵשִוֲ:
