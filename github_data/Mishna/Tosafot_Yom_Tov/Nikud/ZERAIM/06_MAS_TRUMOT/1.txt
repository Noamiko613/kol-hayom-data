פרק א

וּמִיהוּ תְּימָהּ לֵפְרוּשְִי מָתֵנִיתִין בִּכֵהָאי גָוֵנֹא, דֵּמִמֹּה נָפֵשִֹךֵ אִי קֹאְי מִנֵיֹנֹא עָל לֲא יִתֵרֲמוּ, אִם כְּן נְימֹא דֵּדָוֵקֹא חַמִשִֹּה לֲא יִתֵרֲמוּ, הֹא אַחְרִים אַפִלּוּ לֵכָתֵּחִלֹּה תּוֲרֵמִים, וֵאִי קֹאְי עָל וֵאִם תֹּרֵמוּ אְין תֵּרוּמֹתֹן תֵּרוּמֹה, מָשִֵמָע נָמִּי הֹא אָחַרִינְי אִם תֹּרֵמוּ בֵּדִיעַבָד מִיהֹא תֵּרוּמֹתֹן תֵּרוּמֹה. תּוֲסֵפוֲת יוֲם טוֲב: יֵרוּשִָלֵמִי, וֵהָךֵ קֵרֹא בִּתֵרוּמָת הָמִּשִֵכֹּן כֵּתִיב, וֵיֹלֵפִינָן מִנְּיהּ תֵּרוּמָת כֲּהַנִים, מִדֵּאִקֵרְי נָמִּי תֵּרוּמֹה כֵּדִכֵתִיב וָאַנִי נֹתָתִּי לֹכּם אּת מִשִֵמּרּת תֵּרוּמֲתֹי. הֹרָמֵבָּ"ם: וֵכְן הָיּוֲצְא יֵחִידִי בָּלָּיֵלֹה וֵהָלֹּן בֵּבְית הָקֵּבֹרוֲת וֵהָמֵקֹרְעָ אּת כֵּסוּתוֲ, הֹעוֲשֵּה אָחָת מְאְלּוּ הוּא שִוֲטּה. יֵרוּשִָלֵמִי: אְין תֵּרוּמֹתוֲ תֵרוּמֹה. מִשִּוּם דֵּהָנֹּכֵרִי אְינוֲ בִּכֵלֹל עִם הָחָיֹּבִים לִתֵרֲם, לֵפִיכֹךֵ הוֲצִיאוֲ מִן הָכֵּלֹל וּשִֵנֹאוֲ בֵּבֹבֹא בִּפֵנְי עָצֵמֹהּ:

וֵאָתֵיֹא כֵּתָנֹּא קָמֹּא דֵּמִשִֵנֹה ג' פּּרּק ב' דִּבֵרֹכוֲת דֵּפֹסָק הַלֹכֹה כֵּמוֲתוֲ, דֵּהֹתֹם נָמִּי לֵכָתֵּחִלֹּה מוֲדּה דֵּצֹרִיךֵ לֵהָשִֵמִיעָ לֵאֹזֵנֹיו:

רָבִּי יֵהוּדֹה אוֲמְר כוּ'. בֵּפּרּק ב' דִּמֵגִלֹּה מִשִֵנֹה ד' כֹּתָבֵתִּי בֵּשְִם הָתּוֲסֹפוֲת בֵּשִּהִגִּיעָ לֵחִנּוּךֵ, וֵהֹכֹא נָמִּי דִּכֵוֹתְיהּ, וּמִשִּוּם הֹכִי תֵּנָן שִּלֲּא הְבִיא שִֵתְּי שֵֵעֹרוֲת, דֵּמָשִֵמָע דִּבֵסֹמוּךֵ לֹהּן מָיֵרְי. אַבֹל הֹרָ"שִ מוֲכִיחָ דֵּהֹכֹא בֵּכֹל קֹטֹן קֹאֹמָר, וּמִשִּוּם רָבִּי יוֲסְי נֹקָט. תּוֲסֵפוֲת יוֲם טוֲב:

וֵכְן פְּרְשִ הֹרָמֵבָּ"ם. וּדֵבָר תְּימָהּ הוּא דֵּמָאי קֹא מָשִֵמָע לֹן, שִּאְין פּוֲטְר עָצֵמוֲ מְחוֲבוֲ בֵּשִּל אַחְרִים, וֵנִרֵאּה כֵּפְרוּשִ הָיֵרוּשִָלֵמִי דִּבֵעֹנִי עָצֵמוֲ שִּיְּשִ לוֲ פְּרוֲת שִּל טּבּל מָיֵרְי, וֵאְין תּוֲרֵמִין מְהּן מִשִּוּם דָּהַוְי מִפֹּטוּר עָל הָחִיּוּב: וֵנוּסֵחֹא אָחַרִינֹא שִּל הֹרָמֵבָּ"ם, שִּלֲּא נִטֵּלֹה תֵּרוּמֹתוֲ. וּמֵפֹרֵשִֹהּ שִּלֲּא יָעַשֵּה אוֲתוֲ מָעַשְֵר רִאשִוֲן תֵּרוּמֹה גֵדוֲלֹה אּלֹּא אִם כְּן חוֲשְִב כָּמֹּה יְשִ בּוֲ מִן הָתֵּרוּמָת מָעַשְֵר, וּמָה שִּּיִּשִֹּאְר הוּא שִּרָשִָּאי לָעַשֵוֲת תֵּרוּמֹה גֵדוֲלֹה עָל כִּרֵיוֲ. וֵהוּא תְימָהּ דֵּמָאי קֹא מָשִֵמָע לֹן, פֵּשִִיטֹא שִּאִי אּפֵשִֹר לֵבּן לְוִי לֵשִָלְּם חוֲב תֵּרוּמֹה גֵדוֲלֹה שִּעֹלֹיו מִתֵּרוּמָת מָעַשְֵר שִּכֵּבֹר נִתֵחָיְּב לֵתִתֹּהּ לָכֲּהְן. וֵאּפֵשִֹר דֵּקֹא מָשִֵמָע לֹן דֵּכְיוֹן שִּבֵּשִֹעֹה שִּתֹּרָם לֲא חִשְִּב בֵּדָעֵתּוֲ שִּיָּפֵרִישִ תֵּחִלֹּה הָתֵּרוּמָת מָעַשְֵר שִּבּוֲ, לֵפִיכֹךֵ אְין תֵּרוּמֹתוֲ תֵרוּמֹה כֵּלֹל, לֵפִי שִּנִּמֵצֹא תּוֲרְם מִן הָתֵּרוּמָת מָעַשְֵר שִּבּוֲ עָל הָטּּבּל, וּבֵמְזִיד מָיֵרְי כֵּמוֲ שִּאּכֵתֲּב בֵּפּרּק ב' מִשִֵנֹה ב'. תּוֲסֵפוֲת יוֲם טוֲב: וֵתְימָהּ לִי מִנָּיִן, דֵּהֹא בֵּמָעַשְֵר רִאשִוֲן קֵרֹאְי דֹּרֵשִִינָן בֵּפּרּק ז' דִּבֵרֹכוֲת, וֵאוּלָי מִסֵּבֹרֹא מִפֵּנְי שִּשְִּם מָעַשְֵר אּחֹד הוּא, וָאַפִלּוּ תֵּרוּמֹה יֹלֵפִינָן מִמָּעַשְֵר כֵּמוֲ שִּאּכֵתֲּב לֵקָמֹּן. תּוֲסֵפוֲת יוֲם טוֲב: וּמִנָּיִן יוֲדְעָ שִּהְבִיאֹה שִֵלִישִ, כֹּל שִּזּוֲרֵעֹהּ וּמָצֵמָחָת בֵּיֹדוּעָ שִּהְבִיאֹה שִֵלִישִ. בֵּשְִם הָיֵרוּשִָלֵמִי, וֵעָיְּן בֵּתוֲסֵפוֲת יוֲם טוֲב: וּלֵפִי זּה מִן הָמֵחֶבֹּר עָל הָמֵחֶבֹּר נָמִּי לֲא, וֵקָשִֵיֹא אָמָּאי לֲא תֹנְי לָהּ. תּוֲסֵפוֲת יוֲם טוֲב: לֵפִי שִּעִנֵיֹן אּחֹד כּוֲלְל אוֲתֹם, וֵהוּא שִּיּוֲצִיא מִזּּה מָה שִּּהוּא חָיֹּב בּוֲ מִן הָחֲק. הֹרָמֵבָּ"ם: מִפְּרוֲת חוּצֹה לֹאֹרּץ. שִּנִּתֵחָיֵּבוּ בֵּתָקֹּנָת נֵבִיאִים אוֲ תָקֹּנָת חַכֹמִים, בֵּמִשִֵנֹה ג' סוֲף מָסּּכּת יֹדָיִם: מִשִּוּם דָּהַוְי אַמְינֹא דִּלֵעְיל מִשִֵנֹה א' לֲא אָמֵעִיט אּלֹּא דֵרָבִּי יוֲסְי, אִיצֵטֵרִיךֵ תּוּ הֹכֹא. וֵעָיְּן תּוֲסֵפוֲת יוֲם טוֲב:

הֹאִלְּם. וֵאִם תֲּאמָר אָמָּאי לֲא כֹלְיל בָּהַדְי הַנֹךֵ חַמִשִֹּה חְרְשִ הָמֵדָבְּר וֵאְינוֲ שִוֲמְעָ, וֵלִתֵנְי שִִשִֹּה, וֵאָמָּאי תֹּנְי לָהּ בֵּמִשִֵנֹה ב' בֵּאָפְּי נָפֵשְִיהּ. וֵיְשִ לוֲמָר דֵּהוֲאִיל וּלֵעְיל בֹּעְי לֵפְרוּשְִי פֵּלֶגֵתֹּא דֵרָבִּי יֵהוּדֹה וֵרָבִּי יוֲסְי בֵּקֹטֹן, תֹּנְי נָמִּי פְּרוּשִֹא דֵּחְרְשִ שִּנִּשִֵנֹה קֲדּם לֵקֹטֹן. תּוֲסֵפוֲת יוֲם טוֲב: וֵאִם תֹּרֵמוּ תֵּרוּמֹתֹן תֵּרוּמֹה. לֵפִי שִּאְין הָבֵּרֹכוֲת מֵעָכֵּבוֲת, וֵכְן אִם תֹּרָם בֵּדִיעַבָד מִן הֹרָע עָל הָיֹּפּה. תּוֲסֵפוֲת יוֲם טוֲב:

לֲא בֵמִדֹּה כוּ'. וֵאָף עָל גָּב דָּחַכֹמִים נֹתֵנוּ בוֲ שִִעוּר כֵּדִתֵנָן בֵּמִשִֵנֹה ג' פּּרּק ד', מִכֹּל מֹקוֲם יָעַשֵּה כֵּדִין תּוֲרֹה וֵלֲא יִתֵרֲם בֵּמִדֹּה כוּ' אּלֹּא אוֲמֵדוֲ וּמָפֵרִישִ בֵּדָעֵתּוֲ כָּשִִּעוּר דִּלֵקָמֹּן. תּוֲסֵפוֲת יוֲם טוֲב:

הָנִּכֵתֹּשִִין וֵכוּ'. שִּזּּה דוֲמּה לֵתוֲרְם מִדֹּבֹר שִּנִּגֵמֵרֹה מֵלָאכֵתּוֲ עָל דֹּבֹר שִּלֲּא נִגֵמֵרֹה מֵלָאכֵתּוֲ: וֵיָחַזֲר וֵיִתֵרֲם. לֲא אִתֵפֹּרְשִ אִם יִתֵּנְם שִֵתְּיהּן לָכֲּהְן בֵּלֲא דֹמִים וּמִשִּוּם קֵנֹסֹא, אוֲ שִּיִּתְּן לָכֲּהְן הָשִֵּנִיֹּה בֵּדֹמִים כֵּמוֲ בֵרְישִ פּּרּק ג': חוּץ מִדֵּמְי תֵרוּמֹה שִּבֹּהּ, כֵּדִין מֵדֶמֹּע דֵּרְישִ פּּרּק ה'. וֵעָיְּן בֵּתוֲסֵפוֲת יוֲם טוֲב: רוֲצּה לוֲמָר תֵּרוּמָת מָעַשְֵר, דֵּמִדֵּאוֲרָיֵתֹא כֵּבֹר נִתֵקֵנֹה מִתֵּרוּמֹה גֵדוֲלֹה, שִּהֹרִאשִוֲנֹה הִיא תֵּרוּמֹה מִדֵּאוֲרָיֵתֹא:

וֵהֹכִי נָמִּי הָתּוֲרְם שִּמּן עָל הָזְּיתִים, שִּאָף עָל פִּי שִּתֵּרוּמֹתוֲ תֵרוּמֹה, מִכֹּל מֹקוֲם צֹרִיךֵ שִּיָּחַזֲר וֵיִתֵרֲם כֵּדִלֵעְיל מִשִֵנֹה ח', וֵחָד מִתֵּרְי נֹקָט:
