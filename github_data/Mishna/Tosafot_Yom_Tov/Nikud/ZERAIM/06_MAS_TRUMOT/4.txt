פרק ד

פרק ד - משנה א

וְנֻסְחָא אַחֶרֶת מִמָּקוֹם אַחֵר. וְזֶה גִירְסַת הָרַמְבַּ"ם, וּפֵרוּשׁוֹ שֶׁצָּרִיךְ שֶׁיּוֹצִיא הַתְּרוּמָה עַל זֶה שֶׁהִפְרִישׁ מִנֵּיהּ וּבֵיהּ וְלֹא יַפְרִישׁ עָלָיו מִמָּקוֹם אַחֵר:

פרק ד - משנה ב

וְזֶה לְשׁוֹן הָרַמְבַּ"ם, דִּבְדִיּוּק הַחֶשְׁבּוֹן אֵינוֹ טָעוּן לָתֵת לֶעָנִי אֶלָּא תֵּשַׁע עֲשִׂירִיּוֹת מֵהַסְּאָה, וְיַפְרִישׁ גַּם כֵּן אוֹתָן שְׁמֹנֶה סְאִין וַעֲשִׂירִית סְאָה: כְּלוֹמַר וְאֵינוֹ יָכוֹל לִסְמֹךְ שֶׁכֻּלָּם בְּחֶזְקַת קַיָּמִים, אֶלָּא צָרִיךְ שֶׁיִּתְבָּרֵר לוֹ כַּמָּה קַיָּמִים, וּלְפִי מַה שֶּׁנִּתְבָּרֵר לוֹ שֶׁקַיָּמִים כָּךְ יַפְרִישׁ עֲלֵיהֶם. וְלֹא דָמוּ לְדִתְנַן בְּסוֹף פֶּרֶק ג' דְּגִטִּין הַמַּנִּיחַ פֵּרוֹת כוּ' מַפְרִישׁ עֲלֵיהֶן בְּחֶזְקַת שֶׁהֵן קַיָּמִים, דְּהָתָם הַפֵּרוֹת אֶצְלוֹ נִשְׁאָרִים וְאֵין כָּאן חֲשָׁשׁ שֶׁנֶּאֶכְלוּ אֶלָּא שֶׁמָּא נֶאֶבְדוּ, וּלְהָא לֹא חַיְשִׁינַן דְלֹא שְׁכִיחַ, אֲבָל הָכָא שֶׁנּוֹתְנָם לַאֲחֵרִים, לַלֵּוִי וְלֶעָנִי, יֵשׁ לָחוּשׁ שֶׁמָּא נֶאֶכְלוּ:

פרק ד - משנה ג

מֵאַרְבָּעִים. דִּכְתִיב (יְחֶזְקֵאל מה) זֹאת הַתְּרוּמָה אֲשֶׁר תָּרִימוּ שִׁשִּׁית הָאֵפָה מֵחֹמֶר הַחִטִּים וְשִׁשִּׁיתֶם הָאֵיפָה מֵחֹמֶר הַשְּׂעוֹרִים. וְחֹמֶר, שְׁלֹשִׁים סְאִין, וְאֵיפָה, שְׁלֹשָׁה סְאִין, נִמְצָא שִׁשִּׁית הָאֵיפָה חֲצִי סְאָה, וְשִׁשִּׁיתֶם הַיְנוּ תְּרֵי שִׁשִּׁית עַל חַד שִׁשִּׁית, הֲוֵי סְאָה וּמֶחֱצָה לִשְׁנֵי חֳמָרִים, לְחֹמֶר חִטִּים וּלְחֹמֶר שְׂעוֹרִים, הֲוֵי אֶחָד מֵאַרְבָּעִים. יְרוּשַׁלְמִי: מִשְּׁלֹשִׁים. וְדָרְשׁוּ שְׁתֵּי שִׁשִּׁית מֵחֹמֶר דְּהַיְנוּ סְאָה מִשְּׁלֹשִׁים סְאִין: מֵחֲמִשִּׁים. דִּכְתִיב (בְּמִדְבָּר לא) וּמִמַּחֲצִית בְּנֵי יִשְׂרָאֵל תִּקַּח אֶחָד אָחֻז מִן הַחֲמִשִּׁים, כָּל מַה שֶּׁאַתָּה אוֹחֵז הֲרֵי הוּא כָּזֶה, מַה זֶּה מֵחֲמִשִּׁים אַף מַה שֶּׁתֹּאחַז בְּמָקוֹם אַחֵר מֵחֲמִשִּׁים, וְהַיְנוּ מִדָּה בֵּינוֹנִית שֶׁהִיא בְּרֹב בְּנֵי אָדָם: מִשִּׁשִּׁים. דְּדָרְשִׁינַן נַמִּי שִׁשִּׁית הָאֵיפָה מֵחֹמֶר חִטִּים לְחוּדֵיהּ, וְשִׁשִּׁית הָאֵיפָה חֲצִי סְאָה וְהוּא חֵלֶק שִׁשִּׁים מֵחֹמֶר שֶׁהוּא שְׁלֹשִׁים סְאִין:

פרק ד - משנה ד

אֵין תְּרוּמָתוֹ תְרוּמָה. דְּאִיכָּא פְּסֵידָא לְבַעַל הַבַּיִת, דְּאִי תֵימָא דִּכְשִׁעוּר הֲוֵי תְּרוּמָה, וְאִם כֵּן נִמְצְאוּ חֻלִּין שֶׁהוֹסִיף הֵם מְדֻמָּעִים עִם הַתְּרוּמָה. וְכֵן אִם פִּחֵת נַמִּי אֵין תְּרוּמָתוֹ תְּרוּמָה, לְפִי שֶׁאִם נֹאמַר דִּכְשִׁעוּר הֲוֵי תְרוּמָה, נִמְצָא שֶׁמִּקְצָת מְתֻקָּן וּמִקְצָת טֶבֶל מְדֻמָּעִין זֶה בָזֶה. הָרַ"שׁ:

פרק ד - משנה ה

עוֹד יֵשׁ לוֹמַר כִּדְפֵרֵשׁ הָרַ"ב בְּפֶרֶק ה' דִּדְמַאי דְמַתְנִיתִין כְּאַבָּא אֶלְעָזָר בֶּן גּוֹמֵל דְּסָבַר דְּיִשְׂרָאֵל רַשַּׁאי לִתְרֹם תְּרוּמַת מַעֲשֵׂר (וּפֵרוּשׁ זֶה יִתָּכֵן גַּם לְנֹסַח אַחֵר, וְהוּא גִרְסַת הָרַ"שׁ, וְעַיֵּן תּוֹסָפוֹת חֲדָשִׁים). וּפֵרוּשׁ יַעֲשֶׂנָּה, לֹא כֻלָּהּ אֶלָּא אוֹתוֹ הַמּוֹתָר. תּוֹסְפוֹת יוֹם טוֹב: יְרוּשַׁלְמִי דַּיּוֹ כְּלַפֵּי רַבִּי טַרְפוֹן וְרַבִּי עֲקִיבָא אֲמָרוֹ. תּוֹסְפוֹת יוֹם טוֹב:

פרק ד - משנה ו

מְשַׁעֲרִים. לְאַשְׁמְעִינַן דְּלֹא יַפְרִישׁ בְּמִדָּה בִּזְמַן אֶחָד עַל זְמַן אַחֵר הַיָּפֶה מִמֶּנּוּ, מִשּׁוּם דִּמְעַשֵּׂר מִן הָרַע עַל הַיָּפֶה (סִיּוּם דְּבָרָיו הֵם אֶחָד לְפֵרוּשׁ הָרַ"שׁ וְלֹא לְפֵרוּשׁ הָרַ"ב). תּוֹסָפוֹת חֲדָשִׁים: מִשְּׁלָשְׁתָּן. פֵּרוּשׁ מִן הַמּוֹדֵד וְהַמּוֹנֶה וְהָאוֹמֵד, וְהָרַמְבַּ"ם כָּתַב דְּאֵין מַפְרִישִׁין מֵאֹמֶד לִתְרוּמַת מַעֲשֵׂר וּלְמַעֲשֵׂר, לְכָךְ כָּתַב שָׁם מְשֻׁבָּח מִשְּׁנֵיהֶם, וְסוֹבֵר כְּשִׁיטַת הַיְרוּשַׁלְמִי (וּפֵרוּשׁ מִשְּׁלָשְׁתָּן כְּמוֹ בִשְׁלָשְׁתָּן):

פרק ד - משנה ז

עוֹלָה כוּ'. אֲפִלּוּ כְּשֶׁעֲדַיִן בְּיַד יִשְׂרָאֵל כוּ' וְאִי בָּעֵי מָצֵי מִתְּשִׁיל עֲלַהּ כוּ' לֶהֱוֵי כְּדָבָר שֶׁיֵּשׁ לוֹ מַתִּירִין כוּ', וּמְתָרֵץ סוֹף פֶּרֶק ז' דִּנְדָרִים דְּכֵיוָן דְּלֵיכָּא מִצְוָה שֶׁיִּשְׁאַל עַל הַפְרָשָׁה לֹא אָתֵי לְאִתְּשׁוֹלֵי עֲלַהּ: שְׁתוּת. וִיבָאֵר לְךָ שֶׁהַקַּב הַנּוֹסָף צָרִיךְ שֶׁיִּהְיֶה כֻּלּוֹ חֻלִּין כְּדֵי שֶׁיִּהְיֶה שְׁתוּת סְאָה שֶׁל תְּרוּמָה. הָרַמְבַּ"ם:

פרק ד - משנה ח

וְאַף שֶׁאֵין מְבַטְּלִין אִסּוּר לְכַתְּחִלָּה, כַּמְּפֹרָשׁ בְּפֶרֶק ה' מִשְׁנָה ט', הַנֵּי מִילֵי לְהַרְבּוֹת בְּהֶתֵּר, אֲבָל בִּכְהַאי גַוְנָא שֶׁאֵינוֹ מַרְבֶּה בְּהֶתֵּר אֶלָּא מְשַׁנֶּה הָאִסּוּר מִכְּמוֹת שֶׁהָיָה וּמַפְרִידָן כְּדֵי לְעָרְבָם יַחַד, בִּכְהַאי גַוְנָא נִרְאָה לִי דְּשָׁרֵי לְבַטֵּל לְכַתְּחִלָּה, וְרַבִּי אֱלִיעֶזֶר אוֹסֵר. תּוֹסְפוֹת יוֹם טוֹב:

פרק ד - משנה י

וּבְזוֹ כוּ'. וְצָרִיךְ טַעַם בֵּין לְרַבִּי אֱלִיעֶזֶר וּבֵין לְרַבִּי יְהוֹשֻׁעַ מַאי שְׁנָא לְבָנוֹת וּשְׁחוֹרוֹת מִתַּחְתּוֹנוֹת לְעֶלְיוֹנוֹת. וְהַטַּעַם כָּתְבוּ הַתּוֹסָפוֹת זְבָחִים דַּף ע"ג דְּלֹא דָמִי, דְּהַשְּׁחוֹרוֹת אֵינָן רְאוּיִין לִהְיוֹת בִּסְפֵק דִּמּוּעַ, שֶׁלְּעוֹלָם הַשְּׁחוֹרוֹת נִכָּרוֹת מִתּוֹךְ הַלְּבָנוֹת. (וְטַעֲמָא דְרַבִּי יְהוֹשֻׁעַ דְּכָאן לֹא שַׁיָּךְ לוֹמַר אִם הָיָה רוֹצֶה הָיָה דוֹרְסָן וּמְעָרְבָן, כֵּיוָן דִּדְרוּסוֹת וְעוֹמְדוֹת הֵן. בֵּית דָּוִד): מַעֲלוֹת כוּ'. אַף עַל פִּי שֶׁאֵינָן בִּסְפֵק דִּמּוּעַ, הֵקֵלּוּ בִּתְרוּמַת פֵּרוֹת דְּרַבָּנָן. תּוֹסָפוֹת שָׁם:

פרק ד - משנה יא

וּפֵרוּשׁ הַפָּסוּק שֶׁשָּׁט עַל פְּנֵי הַמַּיִם וְלֹא הֵסִיר. וַחֲכָמִים הִשְׁאִילוּ מִלָּה אֲרָמִית זֹאת לְהָסִיר מַה שֶּׁלְּמַעְלָה: בְּגִרְסַת הָרַב לֹא גָרְסִינַן מֵאָה סְאָה. וּמִיהוּ גַּם אוֹתוֹ גִרְסָא יֵשׁ לְתַקֵּן עַל פִּי פֵּרוּשׁ הָרַ"ב, וְהָכִי קָאָמַר אִם יֵשׁ בַּקִּפּוּי וְגַם מַה שֶּׁתַּחְתָּיו מֵאָה סְאָה. תּוֹסְפוֹת יוֹם טוֹב:

פרק ד - משנה יב

עֲיָרוֹת. בַּעַל כַּף נַחַת פֵּרֵשׁ, קֻפּוֹת גְּדוֹלוֹת הַרְבֵּה עֲשׂוּיוֹת לְמִדָּה, וְהִיא מִלָּה בִלְשׁוֹן עֲרָב. אֲבָל לְפֵרוּשׁ הָרַ"ב נִרְאָה כִּפְשׁוּטוֹ לְשׁוֹן עִיר, דִּלְפִי שֶׁהֵן רְחוֹקוֹת אֵין מִתְעָרְבִין. תּוֹסְפוֹת יוֹם טוֹב:

פרק ד - משנה יג

מֵאָה וּשְׁנֵי כוּ'. לֹא שֶׁצָּרִיךְ מֵאָה וּשְׁתַּיִם, שֶׁהֲרֵי בְּאֶחָד וּמֵאָה עוֹלָה לְהַמַּחְמִיר שֶׁבְּמִשְׁנָה ז', אֶלָּא שֶׁכָּךְ הָיָה הַמַּעֲשֶׂה. תּוֹסְפוֹת יוֹם טוֹב:
