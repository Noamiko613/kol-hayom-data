פרק ט

וְלִכְאוֹרָה מַשְׁמַע דְּרַבִּי עֲקִיבָא דַּעַת שְׁלִישִׁית דְּאַף טְמֵאִים אֲסוּרִים. אֲבָל מִשִּׁיטַת הָרַמְבַּ"ם מַשְׁמַע דְּרַבִּי עֲקִיבָא הַיְנוּ תַּנָּא קַמָּא:

כְּלוֹמַר תְּרוּמָה גְדוֹלָה וּתְרוּמַת מַעֲשֵׂר. וְלֹא חָשׁ לְהַזְכִּיר אֶלָּא דְבָרִים הַנִּתָּנִים אַף לְזָרִים, כְּדֵי לְמָכְרָהּ לַכֹּהֵן בִּדְמֵי תְרוּמָה. הָרַ"שׁ: וַעֲנִיֵּי יִשְׂרָאֵל מוֹכְרִין. וְכָךְ הַלְוִיִּם מוֹכְרִים לַכֹּהֲנִים מַעֲשֵׂר שֶׁלָּהֶן. הָרַמְבַּ"ם:

וְהַבִּכּוּרִים כוּ'. עַיֵּן בְּמִשְׁנָה ב' פֶּרֶק ב' דְּבִכּוּרִים, וּמַתְנִיתִין דְּהָכָא בְּזוֹרֵעַ חוּץ לִירוּשָׁלַיִם, וְכֵן לְעִנְיַן מַעֲשֵׂר שֵׁנִי, וְהָכִי מוֹקִים לַהּ בַּיְרוּשַׁלְמִי:

וְהוּא מִלְּשׁוֹן כִּסְלֵי לִגְיָא בִּבְרָכוֹת פֶּרֶק קַמָּא דַּף ו' וְהַגִּרְסָא שָׁם לְאוֹגְיָא: תּוֹסֶפְתָּא. וְטַעֲמָא לְפִי שֶׁהַתְּרוּמָה מֻפְרֶשֶׁת הִיא וְלֹא נִתְעָרְבָה בַּחֻלִּין. הָרַמְבַּ"ם. וְלֹא דָמִי לְמִשְׁנָה י' פֶּרֶק ד', דְּהָתָם בְּתָלוּשׁ רְאוּיִן הֵן לְהִתְעָרֵב. תּוֹסְפוֹת יוֹם טוֹב:

וּרְצוֹנוֹ לוֹמַר דַּוְקָא כְּשֶׁאָכַל אוֹתָן גִּדּוּלִין קֹדֶם שֶׁנִּגְמְרָה מְלַאכְתָּן. וּפֵרוּשׁ נִגְמְרָה מְלַאכְתָּן הַיְנוּ לְמַעַשְׂרוֹת, וְהִתְבָּאֵר זְמַן שֶׁל כָּל פְּרִי וּפְרִי פֶּרֶק קַמָּא דְמַעַשְׂרוֹת: גִּדּוּלֵי גִדּוּלִין אֲסוּרִין. דְּטֶבֶל דָּבָר שֶׁיֵּשׁ לוֹ מַתִּירִין, וַאֲפִלּוּ בְּאֶלֶף לֹא בָטִיל: לַפֵּרוּשׁ הָרִאשׁוֹן הַגִּרְסָא כַּשְּׂעוֹרִים, וְלַפֵּרוּשׁ הַשֵּׁנִי כַּשְּׂעוֹרָה:

שֶׁפֵּרוֹתָיו טֶבֶל. וּכְמַאן דְּאָמַר בִּמְנָחוֹת (דַּף סז) מֵרוּחַ הַנָּכְרִי פּוֹטֵר (דִּלְמַאן דְּאָמַר אֵינוֹ פוֹטֵר מִדְּאוֹרַיְתָא אֲפִלּוּ עֲרַאי אָסוּר), דְּלֹא מְחַיֵּב אֶלָּא מִשּׁוּם בַּעֲלֵי כִיסִין, וּבַאֲכִילַת עֲרַאי לֹא גָזוּר, וַאֲפִלּוּ גָזוּר כֵּיוָן דִּשְׁתָלָהּ פָּקַע לַהּ טִבְלָהּ מִנָּהּ כוּ'. וַאֲפִלּוּ לְמַאן דְּאָמַר (שָׁם דַּף ע) כֵּיוָן שֶׁמֵּרְחָהּ טָבְלָה לַהּ וְכִי קָרָא עָלֶיהָ שֵׁם קָדְשָׁה לַהּ, הַנֵּי מִילֵי בְּמֵרוּחַ יִשְׂרָאֵל דְּחַיָּב מִדְּאוֹרַיְתָא. הָרַ"שׁ: וְכֵן כָּתְבוּ הָרַ"שׁ וְהַתּוֹסָפוֹת. וּמִיהוּ אִיכָּא לְמֵימַר שֶׁלֹּא הֻצְרְכוּ לְתָרֵץ כָּךְ אֶלָּא לְרַבָּה דִּמְפָרֵשׁ וַאֲסוּרִים לֶאֱכֹל לְזָרִים. אֲבָל לַמַּסְקָנָא דְּמַעֲלָה בְּעַלְמָא הוּא וְאַף לַכֹּהֲנִים אֲסוּרִים לֶאֱכֹל, אֵין כָּאן קֻשְׁיָא כְּלָל, דְּכֵיוָן שֶׁאָסוּר לֶאֱכֹל עַד שֶׁיָּגוֹם, וַדַּאי דְּלֹא מַשְׁהֵי לַהּ. תּוֹסְפוֹת יוֹם טוֹב:
