פרק ד

שְׁבוּעַת כוּ'. שְׁבוּעָה לֹא כְתִיב בְּהֶדְיָא בַקְּרָא, אֶלָּא יָלְפִינַן מִקּוֹל, כְּמוֹ שֶׁכָּתוּב בְּרֵישׁ מַתְנִיתִין דְּסוֹף פִּרְקִין: כְּלוֹמַר, אַף עַל גַּב דְּאֵינוֹ בַר מַלְקוֹת, שֶׁנִּתָּק לַעֲשֵׂה דְהִשָּׁבוֹן: מִכְּלָל דְּמֶלֶךְ מִדְּאוֹרַיְתָא אֵינוֹ כָשֵׁר לְהָעִיד, וְהָכִי אִיתָא בְהֶדְיָא בַגְּמָרָא דִּמְמַעֲטִינַן מְשַׂחֵק בְּקֻבְיָא וְכָל שֶׁכֵּן מֶלֶךְ. וּכְלוֹמַר שֶׁהוּא מִדְּאוֹרַיְתָא. וּפֵרֵשׁ רַשִׁ"י, דִּכְתִיב (דְּבָרִים יז) שׂוֹם תָּשִׂים עָלֶיךָ מֶלֶךְ, שֶׁתְּהֵא אֵימָתוֹ עָלֶיךָ, וְהַבָּא לְהָעִיד צָרִיךְ לַעֲמֹד לִפְנֵי הַדַּיָּן וְדֶרֶךְ בִּזָּיוֹן הוּא לוֹ. וּלְפִי זֶה אַף בְּמַלְכֵי בֵית דָּוִד אָיְרִינַן, דְּאֵינָן רְאוּיִין לְהָעִיד. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: מִפִּי עַצְמוֹ. דְּיָלֵיף לַהּ מִגְּזֵרָה שָׁוָה דְתֶחֱטָא תֶּחֱטָא מִפִּקָּדוֹן. וְסָבַר דּוּן מִנַּהּ. וּמִנַּהּ כְּלוֹמַר, דְּדָיְנִינַן מִנַּהּ לְחַיְּבוֹ אַף מִפִּי עַצְמוֹ, וּמִנַּהּ כִּי הָתָם דְּמִחַיְּבֵי בֵּין בְּבֵית דִּין בֵּין שֶׁלֹּא בְבֵית דִּין. דְּנִשְׁבַּע עַל שָׁקֶר כְּתִיב, מַשְׁמַע בֵּין בְּבֵית דִּין בֵּין שֶׁלֹּא בְבֵית דִּין. רַשִׁ"י: וְכֵן אִם עָנָה אָמֵן אַחַר הַשְּׁבוּעָה. רַשִׁ"י: אַף עַל גַּב דְּלֹא עָנָה אָמֵן. שֶׁכֵּן הוּא אוֹמֵר וְשָׁמְעָה קוֹל אָלָה אִם לוֹא יַגִּיד וְנָשָׂא עֲוֹנוֹ: שֶׁנֶּאֱמַר וְשָׁמְעָה קוֹל אָלָה: בֵּין כוּ'. דִּסְבִירָא לְהוּ דּוּן מִנַּהּ וְאוֹקֵי בְאַתְרַהּ. כְּלוֹמַר, לְאַחַר שֶׁהֵבֵאתָ מִפִּי עַצְמוֹ לִשְׁבוּעַת הָעֵדוּת מִפִּקָּדוֹן, אוֹקֵי בְאַתְרַהּ בְּדִין הַכָּתוּב בִּשְׁבוּעַת הָעֵדוּת, דְּמִפִּי אֲחֵרִים שֶׁמִּשְׁפָּטוֹ דַוְקָא בְבֵית דִּין, הָכִי נַמִּי מִפִּי עַצְמוֹ שֶׁהֵבֵאתָ אֵינוֹ אֶלָּא בְבֵית דִּין:

וּמִמֵּילָא שֶׁיּוֹדֵעַ לוֹ עֵדוּת. דְּאִי לָאו הָכִי אֵינָהּ אֲסוּרָה. וְהַיְנוּ זְדוֹן הָעֵדוּת. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: רַשִׁ"י. וְתָמְהוּ הַתּוֹסָפוֹת, דְּתֵיפּוּק לֵיהּ דִּבְקוּשְׁטָא קָא מִשְׁתַּבְעֵי. וְתֵרֵץ הָרַ"ן, דְּרַשִׁ"י הָכִי קָאָמַר, שֶׁבְּאוֹתָהּ שָׁעָה שֶׁנִּשְׁבְּעוּ אֵין אָנוּ יוֹדְעִין, יוֹדְעִין הָיוּ אֶלָּא שֶׁלֹּא הָיוּ מִתְכַּוְּנִים לָשׂוּם לֵב, אֲבָל לֹא הָיוּ צְרִיכִין יְדִיעָה אַחֶרֶת, וְנִמְצָא מַה שֶּׁנִּשְׁבְּעוּ אֵין אָנוּ יוֹדְעִים לֹא נִשְׁבְּעוּ עַל הָאֱמֶת, שֶׁאִלּוּ הָיוּ שָׂמִים לֵב בְּאוֹתָהּ שָׁעָה יוֹדְעִים הָיוּ, שֶׁהֲרֵי מִיָּד כְּשֶׁנִּתְיַשְּׁבוּ נִזְכְּרוּ בְּלֹא שׁוּם חִדּוּשׁ יְדִיעָה, וַאֲפִלּוּ הָכִי כֵּיוָן שֶׁהָיוּ סְבוּרִים לִשָּׁבַע עַל הָאֱמֶת, פְּטוּרִים, מִשּׁוּם דַּאֲנוּסִים הָיוּ כוּ'. אֲבָל וַדַּאי דְּהוּא הַדִּין וְכָל שֶׁכֵּן כְּשֶׁסְּבוּרִים שֶׁמֻּתָּרִים לְגַמְרֵי. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב:

שֶׁאָמְרוּ. בְּלֹא שְׁבוּעָה, אֵין אָנוּ כוּ', וְאָמַר לָהֶם מַשְׁבִּיעַ אֲנִי עֲלֵיכֶם שֶׁיְּהֵא אֱמֶת מַה שֶּׁאֲמַרְתֶּם, וְאָמְרוּ אָמֵן, חַיָּבִים. אֲבָל אִם שָׁתְקוּ אַחַר הַשְּׁבוּעָה, אֵין כָּאן קַבָּלַת שְׁבוּעָה. רַשִׁ"י. וְנִרְאֶה מִלְּשׁוֹנוֹ דְּדַוְקָא שָׁתְקוּ, אֲבָל כָּפְרוּ וְאָמְרוּ אֵין אָנוּ יוֹדְעִים כוּ' חַיָּבִים. וּמַיְרֵי בְבֵית דִּין. וּמַתְנִיתִין אֲפִלּוּ כְרַבָּנָן. וְאָמֵן לָאו דַּוְקָא, וְלֹא נָקְטוּ אֶלָּא כְדֵי לְקַצֵּר. וְכֵן כָּתְבוּ הַתּוֹסָפוֹת, וְעַיֵּן בְּסָמוּךְ (אוֹת יג): חוּץ לְבֵית דִּין. אָתְיָא נַמִּי כְרַבָּנָן, דִּלְרַבָּנָן נַמִּי לֹא בָעִינַן שְׁבוּעָה בְבֵית דִּין, אֶלָּא הַכְּפִירָה. דְּאַהַגָּדָה דַוְקָא קָפֵיד קְרָא דְּלֶהֱוֵי בְמָקוֹם הָרָאוּי לְהַגָּדָה. תּוֹסָפוֹת:
וְרֵישָׁא דְתָנֵי הֲרֵי אֵלּוּ חַיָּבִין, וְלֹא אָמְרִינַן דַּהֲוֵי כְפִירַת דְּבָרִים בְּעָלְמָא כֵּיוָן דִּכְבָר אָמְרוּ אֵין אָנוּ יוֹדְעִים כוּ' וְשׁוּב אֵין יְכוֹלִין לַחֲזֹר וּלְהוֹדוֹת, נִרְאֶה לִי מִשּׁוּם דְּלֹא קַפְּדָה הַתּוֹרָה אַחוֹזֵר וּמַגִּיד אֶלָּא בָתַר שְׁבוּעָה. דְּאִם לֹא יַגִּיד, בָּתַר וְשָׁמְעָה קוֹל אָלָה כְּתִיב. וְהַטַּעַם מִפְּנֵי שֶׁדֶּרֶךְ הָעֵדִים לְהִשָּׁמֵט שֶׁלֹּא לְהָעִיד, וּכְשֶׁאוֹמְרִים בָּתַר שְׁבוּעָה יוֹדְעִים אָנוּ, אֵין זֶה חוֹזֵר כוּ'. עוֹד נִרְאֶה לִי, דְּאֵין אָנוּ יוֹדְעִים לֹא מִקְּרֵי מַגִּיד כְּלָל, דְּהָא לֹא הִגִּידוּ כְלוּם, וְהָכָא לָאו מִשּׁוּם חוֹזֵר וּמַגִּיד אַיְתִּינַן עֲלַהּ, אֶלָּא מִשּׁוּם דְּכָפְרוּ בָתַר שְׁבוּעָה שׁוּב לֹא מָצוּ מְשַׁוּוּ נַפְשַׁיְהוּ רְשִׁיעֵי לוֹמַר יוֹדְעִים אָנוּ. וּבְהָכִי מִתָּרְצָא אַמַּאי נָקַט רֵישָׁא בְּמֻשְׁבָּע מִפִּי עַצְמוֹ, דִּלְאַשְׁמוֹעִינַן הַךְ רְבוּתָא נָקַט בְּמִפִּי עַצְמוֹ. הָרַ"ן. כְּלוֹמַר דְּאִלּוּ הֲוָה נָקַט בְּמֻשְׁבָּע מִפִּי אֲחֵרִים, לֹא הָיָה יָכוֹל לְהַתְחִיל אָמְרוּ אֵין אָנוּ יוֹדְעִים כוּ', לְפִי שֶׁשּׁוּב אַחַר כָּךְ כְּשֶׁמַּשְׁבִּיעָן אֵין רְגִילוּת לְהָשִׁיב אֶלָּא אָמֵן, וְאִם כֵּן לֹא הֲוָה יָדְעִינַן רְבוּתָא זוֹ:

גְּמָרָא. דְּאִי אֶפְשָׁר לְצַמְצֵם. וּפֵרוּשׁ תּוֹךְ כְּדֵי דִבּוּר, עַיֵּן בְּרֵישׁ פֶּרֶק ד' דְּנָזִיר. וְלֹא שֶׁיִּהְיֶה כָל הַכְּפִירָה תּוֹךְ כְּדֵי דִבּוּר אֶלָּא שֶׁתַּתְחִיל תּוֹךְ כְּדֵי דִּבּוּר. גְּמָרָא. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: דְּאַף עַל גַּב דְּקָם לִשְׁבוּעָה (וְרֹב בְּנֵי אָדָם אֵין נִשְׁבָּעִין לַשֶּׁקֶר וְהָיָה מְשַׁלֵּם), דָּבָר הַגּוֹרֵם לְמָמוֹן לָאו כְּמָמוֹן דָּמֵי. גְּמָרָא: הַכּוֹפֵר כוּ'. גְּמָרָא הַשְׁתָּא בָּזֶה אַחַר זֶה דְתַרְוַיְהוּ קָא כָפְרֵי אָמַרְתְּ הָרִאשׁוֹן חַיָּב וְשֵׁנִי פָטוּר, כָּפַר אֶחָד וְהוֹדָה אֶחָד מִבַּעְיָא. לֹא צְרִיכָא כְּגוֹן שֶׁכָּפְרוּ שְׁנֵיהֶם וְחָזַר אֶחָד מֵהֶן וְהוֹדָה בְתוֹךְ כְּדֵי דִבּוּר, וְהָא קָמַשְׁמַע לָן דְּמַהוּ דְתֵימָא דְהָא דְרֵישָׁא תּוֹךְ כְּדֵי דִבּוּר מַהֲנֵי, הָנֵי מִלֵּי כְפִירָה וּכְפִירָה, אֲבָל כְּפִירָה וְהוֹדָאָה אֵימָא לֹא, קָא מַשְׁמַע לָן. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: שֶׁהָיוּ יְכוֹלִים לוֹמַר יָדַעְנוּ כִי הָיָה שָׁם כַּת אַחֶרֶת בִּשְׁעַת תְּבִיעָתָן. הָרַמְבַּ"ם: נָשְׂאוּ שְׁתֵּי אֲחָיוֹת, וְאֵינָן כְּשֵׁרִים בְּעֵדוּת אַחַת. וְקָשֶׁה גּוֹסְסוֹת דְּקָאָמַר דְּמַשְׁמַע דְּתַרְוַיְהוּ גוֹסְסוֹת לְמַאי, דִּבְאַחַת סַגִּי. וְכָתְבוּ הַתּוֹסָפוֹת דְּמַיְרֵי כְּגוֹן שֶׁבִּשְׁעַת רְאִיַּת הָעֵדוּת הָיוּ רְחוֹקִים. דְּבָעִינַן תְּחִלָּתָן וְסוֹפָן בְּכַשְׁרוּת: וְלִמֶּדְךָ שֶׁהַגּוֹסֵס הָוֵי כְחַי לְכָל דְּבָרָיו וְאַף עַל פִּי שֶׁרֻבָּן לְמִיתָה. רַשִׁ"י וְהָרַ"ן. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: בִּשְׁתֵּיהֶן. שֶׁאַחַר שֶׁמֵּתוּ הַנָּשִׁים אָז כָּפְרָה הַשְּׁנִיָּה הָעֵדוּת גַּם כֵּן. הָרַמְבַּ"ם:

שְׁבוּעָה כוּ'. שְׁבוּעָה זוֹ אֵינִי יוֹדֵעַ לָמָּה, דְּכֵיוָן שֶׁאָמַר הַלָּה מַשְׁבִּיעַ אֲנִי עֲלֵיכֶם מֻשְׁבָּעִין הֵם כוּ'. וְיֵשׁ לוֹמַר לִרְבוּתָא נָקַט דְּאַף עַל גַּב דְּהוֹצִיאוּ הֵם עַצְמָם שְׁבוּעָה מִפִּיהֶם אֲפִלּוּ הָכִי אֵינָן חַיָּבִין אֶלָּא אַחַת. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב:

גְּמָרָא, וְהָא קָא מַשְׁמַע לָן. וּפֵרֵשׁ רַשִׁ"י וְאַיְדֵי דְתַנְיָא הָא, תָּנָא בַהֲדַיְהוּ תַּשְׁלוּמֵי אַרְבָּעָה וַחֲמִשָּׁה וְאוֹנֵס וּמְפַתֶּה, דְּמָמוֹן שֶׁאֵין מִשְׁתַּלֵּם בָּרֹאשׁ הוּא, דּוּמְיָא דַחֲצִי נֶזֶק שֶׁאֵין מִשְׁתַּלֵּם בָּרֹאשׁ. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: דְּקַיְמָא לָן מוֹדֶה בִקְנָס פָּטוּר, הִלְכָּךְ לֹא הָוֵי אֶלָּא גוֹרֵם לְמָמוֹן, דְּאִלּוּ הוֹדָה הָיָה פָטוּר, וְלָאו מָמוֹן הוּא. גְּמָרָא: דְּמִסְּתָמָא מָמוֹנָא קָא תָבַע וְלֹא קְנָס, שֶׁאִם הוֹדָה לֹא יְשַׁלֵּם. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב:

שֶׁאֵינִי כוּ'. נָקַט לָהּ בִּלְשׁוֹן חִיּוּב וּבִלְשׁוֹן שְׁלִילָה: פְּלוֹנִי כוּ'. מִשְׁנָה יְתֵרָה הִיא, דְּהָא אֲפִלּוּ מָנֶה לִפְלוֹנִי כוּ' תְּנַן בְּמִשְׁנָה י"ב דְּפָטוּר. אֶלָּא הָא קָא מַשְׁמַע לָן דְּדַוְקָא שֶׁפְּלוֹנִי כֹהֵן פָּטוּר בְּכָל עִנְיָן, דְּלֹא שַׁיָּךְ בָּהּ הַרְשָׁאָה. אֲבָל מָנֶה כוּ' זִמְנִין דְּחַיָּב כְּגוֹן בְּהַרְשָׁאָה. גְּמָרָא. וּמִנַּהּ דְּהוּא הַדִּין סֵיפָא שֶׁאָנַס כוּ', דַּהֲוֵי נַמִּי מִשְׁנָה יְתֵרָה, בְּבָא בְהַרְשָׁאָה אַיְרֵי. וְזֶהוּ שֶׁכָּתַב הָרַ"ב (וְכֵן כָּתַב רַשִׁ"י) וּבַגְּמָרָא מוֹקֵי לַהּ בְּבָא בְהַרְשָׁאָה. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב:

וַאֲפִלּוּ הוּא עָנִי אֵינוֹ חַיָּב לוֹ אֶלָּא מִטַּעַם נֶדֶר, וְנִמְצָא שֶׁאֵין עָלָיו חִיּוּב מָמוֹן בְּשׁוּם צַד. שֶׁאֵין בֵּית דִּין יוֹרְדִין לִנְכָסָיו אִם אֵינוֹ רוֹצֶה לְקַיֵּם נִדְרוֹ, אֶלָּא שֶׁמְּנַדִּין אוֹתוֹ אוֹ מַכִּין אוֹתוֹ עַד שֶׁיְּקַיֵּם מוֹצָא שְׂפָתָיו. נִרְאֶה לִי:

הִקְשָׁה הָרַמְבַּ"ן, לָמָּה לִּי לְמִתְנֵי הָכִי, פְּשִׁיטָא, דְּאִי מִשּׁוּם קְרוֹבִים כוּ' הָא תְנַן בְּרֵישׁ פִּרְקִין, וְעֵד מִפִּי עֵד תְּנַן בְּפֶרֶק ד' דְּסַנְהֶדְרִין מִשְׁנָה ו' כוּ'. וְתֵרֵץ דְּאַשְׁמְעִינַן אִם הֶאֱמִינָן הַנִּתְבָּע עַל עַצְמוֹ שֶׁאִם יָדְעוּ יָבֹאוּ וְיָעִידוּ, אֲפִלּוּ הָכִי פְּטוּרִין, כֵּיוָן שֶׁמִּן הַדִּין אֵינָם נֶאֱמָנִים. וְנִסְתַּיֵּעַ מֵהַיְרוּשַׁלְמִי. וְכֵן מַשְׁמַע בַּגְּמָרָא דִידָן. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב:

פְּטוּרִים. כְּשֶׁלֹּא בָא בְהַרְשָׁאָה. דְּהָא בְמִשְׁנָה ז' דָּיְקִינַן בַּגְּמָרָא דְּבָא בְהַרְשָׁאָה יָכוֹל לְהַשְׁבִּיעָם וּמִתְחַיְּבִים בְּהַשְׁבָּעָתוֹ. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב:

מַשְׁבִּיעַ. דִּקְרָא וְשָׁמְעָה קוֹל אָלָה כְּתִיב, וּבַגְּמָרָא יָלְפִינַן דְּאָלָה הִיא שְׁבוּעָה. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: גְּמָרָא, דְּאִי לָאו הָכִי מַאי קָאָמַר. וּפֵרֵשׁ רַשִׁ"י, בִּשְׁבוּעָה מַזְכִּיר אֶת הַשֵּׁם בְּאֶחָד מִכָּל הַלְּשׁוֹנוֹת הַלָּלוּ. וּכְלוֹמַר שֵׁם אוֹ כִנּוּי. כִּדְאָמְרִינַן בַּגְּמָרָא. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: בְּחַנּוּן כוּ'. דַּהֲוֵי כְמִי שֶׁנִּשְׁבָּע בּוֹ יִתְבָּרַךְ, אַף עַל פִּי שֶׁאֵין אֵלּוּ שְׁמוֹתָיו אֶלָּא כִנּוּיִין בְּעָלְמָא, הַיְנוּ טַעֲמָא דִּבְמִי שֶׁהוּא חַנּוּן בְּמִי שֶׁהוּא רַחוּם קָאָמַר, וְלֹא דָמֵי לְנִשְׁבָּע אֶת הַכִּנּוּיִין, תַּלְמוּד לוֹמַר אִישׁ כִּי יְקַלֵּל אֱלֹהָיו כוּ', מִכָּל מָקוֹם, דִּבְרֵי רַבִּי מֵאִיר. וּמֵהָכָא שָׁמְעִינַן דִּפְלוּגְתַּיְהוּ בְּשֵׁם הַמִּתְבָּרֵךְ. וְכֵן נִרְאֶה מִלְּשׁוֹן הָרַ"ב בְּסָמוּךְ. וּלְפִיכָךְ לְשׁוֹן בְּכֻלָּן דִּתְנַן, כְּמוֹ לְכֻלָּן. וְכֵן מַה שֶּׁכָּתַב הָרַ"ב וְכֵן כָּתַב הָרַמְבַּ"ם, בְּאֶחָד מֵהֶם כְּמוֹ לְאֶחָד: יַכְּכָה אֱלֹהִים כוּ'. לְשׁוֹן רַשִׁ"י יַכְּכָה ה' אֱלֹהִים. וְכֵן יַכְּכָה אֱלֹהִים גָּרְסִינַן. וּלְדַעְתִּי לֹא גָרְסִינַן בָּרֵישָׁא אֱלֹהִים כְּלָל. אֶלָּא יַכְּכָה ה'. וּבָרַשַׁ"ל כָּתַב וְזֶה לְשׁוֹנוֹ, הָרִי"ף גָּרַס בְּתַרְוַיְהוּ שֵׁם יוּ"ד הֵ"י, וְלֹא אֱלֹהִים. אָכֵן גַּם גִּרְסָא זוֹ יְשָׁרָה. וְקָא מַשְׁמַע לָן אַף שֶׁזֶּה קוֹרֵא בַתּוֹרָה שֵׁם יוּ"ד הֵ"י וְהוּא אוֹמֵר וְכֵן יַכְּכָה אֱלֹהִים אֲפִלּוּ הָכִי הָוֵי אָלָה: הַכְּתוּבָה. דְּהָא כְתִיב בָּהּ כְּכֹל אָלוֹת הַבְּרִית (דְּבָרִים כט). רַשִׁ"י: פּוֹטְרִין. פֵּרֵשׁ הָרַמְבַּ"ן זַ"ל, דְּאַף עַל גַּב דִּבְעָלְמָא סְבִירָא לְהוּ דְּמִכְּלַל לָאו אַתָּה שׁוֹמֵעַ הֵן, הָכָא מִשּׁוּם הָכִי פָטְרִינַן, דְּבָעִינַן קְרָא כְדִכְתִיב וְשָׁמְעָה קוֹל אָלָה, אֲבָל לֹא שֶׁשָּׁמְעָה בְרָכָה וְנִשְׁמְעָה מִתּוֹכָהּ קְלָלָה. הָרַמְבַּ"ן וְהָרַ"ן:
