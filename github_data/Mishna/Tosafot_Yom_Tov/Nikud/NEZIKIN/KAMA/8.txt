הֹכִי פְּרוּשִוֲ, שִּאִם חֹבָל זּה בֹּזּה וֵשִּל זּה מֵרֶבּּה מִשִּּל זּה, דֵּשִֹמִין, וֵלֲא אֹמֵרִינָן יְצְא זּה בֹּזּה, וֵהֹכִי מָשִֵמָע קֵרֹא, חָבּוּרֹה תָּחָת חָבּוּרֹה, וֵלֲא כֵּוִיֹּה תָּחָת אֹדֹם רוֲצּה לִטֲּל וּלֵהִצֵטָעְר, דֵּכֹל אַשִּר לֹאֹדֹם יִתְּן בֵּעָד נָפֵשִוֲ, וֵאְין עּרּךֵ לֵשִוּמֹא זוֲ. אּלֹּא שִֹמִין מִי שִּנִּתֵחָיְּב לָמָּלֵכוּת לִכֵוֲּתוֲ בֵּשִָפּוּד עָל צִפֹּרֵנוֲ כָּמֹּה הֹיֹה נוֲתְן שִּלֲּא יְעֹשֵּה לוֲ צָעָר זּה. וֵשִוּמֹא זוֲ קָלֹּה לֵהוֲצִיא מֹמוֲן לֵהִנֹּצְל מִצָּעָר מִלִּקָּח מֹמוֲן וּלֵקָבְּל צָעָר. וֵהָיֵנוּ כֵּשִוּמֹא שִּכֹּתָב הֹרָ"ב בֵּצָעָר דֵּבִמֵקוֲם נּזּק כוּ'. וּבָגֵּמֹרֹא פֹּרְיךֵ, הָאי לִטֲּל, לִתְּן מִבֹּעְי לְיהּ. וּמֵשִָנִּינָן לִטֲּל זּה מִזּּה מָה שִּּנֹּתָן זּה: דֵּמָיֵרְי בִּסֵתֹם בֵּנְי אֹדֹם שִּאְינֹם בֵּנְי אֶמֹּנוּת, אַבֹל אִם הוּא בּּן אֶמֹּנוּת וֵיֹכוֲל לָעַסֲק בִּמֵלָאכֵתּוֲ אָחָר חֹלֵיוֲ מֵשִָעַרִין כֵּפִי הָמֵּלֹאכֹה שִּיֹּכוֲל לָעַשֵוֲת אָחָר שִּיִּתֵרָפְּא. וֵעָיְּן תּוֲסֵפוֲת יוֲם טוֲב: מֵסָיְּם בָּגֵּמֹרֹא שִּאְינֹן חוֲזֵרִין. כֵּלוֲמָר וֵהֹוְי כֵּעְין מִיתֹה, שִּאוֲתוֲ אְבֹר מְת: הָיֵנוּ אִם עֹלוּ בּוֲ צֵמֹחִים מְחַמָת מָכֹּה דִּתֵנָן בֵּמָתֵנִיתִין. וֵיֹלֵפִינָן לָהּ מִכּּפּל לֹשִוֲן, וֵרָפֲּא יֵרָפְּא: הָיֵנוּ נָמִּי שִּלֲּא מְחַמָת מָכֹּה דִּתֵנָן. וָהַוֹה אַמְינֹא דֵּעֹבָר עָל דִּבֵרְי רוֲפְא לֹאו שִּלֲּא מְחַמָת מָכֹּה הִיא, דֵּהֹא מִכֹּל מֹקוֲם עָל יֵדְי הָמָּכֹּה בֹּא לוֲ זּה, שִּאִלּוּ לֲא הֶכֹּה לֲא הֹיֹה צֹרִיךֵ לֵהִזֹּהְר בֵּדִבֵרְי הֹרוֲפְא, קֹא מָשִֵמָע לֹן דֵּלֲא. וֵיֹלְיף לְיהּ מִדִּכֵתִיב רָק: וּבָסִּפֵרִי יֹלְיף גֵּזְרֹה שִֹוֹה דֵּתֹחוֲס תֹּחוֲס. וּפְרְשִ רָשִִ"י נּאֱמָר כֹּאן לֲא תֹחוֲס. וֵנּאֱמָר בֵּעְדִים זוֲמֵמִין לֲא תֹחוֲס, מָה לֵּהָלֹּן מֹמוֲן אָף כֹּאן מֹמוֲן. וֵעָיְּן תּוֲסֵפוֲת יוֲם טוֲב: בָּגֵּמֹרֹא, לֲא אֹמֵרוּ אּלֹּא הוֲדֹאוֲת וֵהָלֵוֹאוֲת. וּפְרְשִ רָשִִ"י, הוֲדֹאוֲת, הָבֹּא לֹדוּן בֵּעְדְי הוֲדֹאֹה שִּאוֲמֵרִים בֵּפֹנְינוּ הוֲדֹה לוֲ. וֵהָלֵוֹאוֲת, שִּבֹּא לֹדוּן בֵּעְדְי הָלֵוֹאֹה שִּאוֲמֵרִים בֵּפֹנְינוּ הִלֵוֹה לוֲ. עָד כֹּאן. וּכֵפִירוֲת, הָיֵנוּ מוֲדּה בֵּמִקֵצֹת וֵכוֲפְר הָכֲּל. וֵהָקֵנֹאוֲת, הָיֵנוּ שִּהִקֵנֹה לוֲ בֵּמָתֹּנֹה. וֵעָיְּן תּוֲסֵפוֲת יוֲם טוֲב: דֵּאִי בֵּקּרּן, קֹיֵמֹא לֹן פָּלֵגֹּא נִזֵקֹא קֵנֹסֹא, וֵאְין גּוֲבִין קֵנֹסֹא בֵּחוּץ לֹאֹרּץ. וֵכְיוֹן דֵּמִנֵּגִיחוֲת דֵּתָמּוּת לֲא מָיֵתִינָן לְיהּ לֵבְי דִּינֹא, הְיכִי מֵיָעְד לְיהּ. מָאי אֹמָרֵתֵּ, דֵּאִיָּעוּד הֹתֹם וֵאָיֵתְיהּ לֵהֹכֹא, תּוֲסֵפוֲת יוֲם טוֲב: הָיֹּשְִן. וּכֵשִּהְקִיץ וֵהִרֵגִּישִ בֹּזּה שִּבִּיֵּשִוֲ כוּ'. אַבֹל אִי לֲא הִרֵגִּישִ אָחָר שִּהְקִיץ, שִּנִּסֵתָּלֵּקֹה הָבּוּשִֹה קֲדּם שִּהְקִיץ, הוּא בִּכֵלָל בָּעֵיֹא בָּגֵּמֹרֹא. וֵעָיְּן תּוֲסֵפוֲת יוֲם טוֲב: וֵשִֹלֵחֹה כוּ'. מִוֵּשִֹלֵחֹה יֹדֹהּ דִּיְּק. וֵהֹרָמֵבָּ"ם לֲא הּעֵתִּיק וֵהּחֱזִיקֹה בִּמֵבֶשִֹיו. וֵתָנֹּא שִֵנֹאוֲ לוֲמָר דִּקֵרֹא בֵּבֲשִּת אָיֵרְי:

וֵקָשִֵיֹא לִי, דֵּהֹא מְהָךֵ קֵרֹא שִּהוּא בֵּפֹרֹשִָת אֱמֲר לֲא יֹלֵפִינָן מִנְּיהּ לֵשִוּם אּחֹד מְאָרֵבֹּעֹה דֵּבֹרִים, כִּדֵפְרְשִ הֹרָ"ב בֵּרְישִ פִּרֵקִין. וֵנִרֵאּה לִי מִדִּכֵתִיב בֵּהָךֵ קֵרֹא דֵּהֹכֹא עָיִן תָּחָת עָיִן שְִן כוּ' וֵהֹכִי נָמִּי כֵּתִיב בֵּפֹרֹשִָת מִשִֵפֹּטִים, וֵשִֹם נּאֱמָר פּּצָע כוּ' דֵּמִנְּיהּ יֹלֵפִינָן לֵצָעָר, הִלֵכֹּךֵ שִָפִּיר דֹּיֵקִינָן נָמִּי לֵכֹל הֹאָרֵבֹּעֹה דֵּבֹרִים. וֵעָיְּן תּוֲסֵפוֲת יוֲם טוֲב:
