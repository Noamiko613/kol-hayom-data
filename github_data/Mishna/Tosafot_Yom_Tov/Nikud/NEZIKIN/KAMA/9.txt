פרק ט

עֵצִים כוּ'. מַסְּקִינַן בַּגְּמָרָא דְּדַוְקָא בְּשִׁנּוּי דְּלֹא הֲדַר לִבְרִיָּתוֹ כְּבַתְּחִלָּה. וּכְגוֹן שֶׁגָּזַל עֵצִים שֶׁאֵינָם מְשֻׁפִּין וַעֲשָׂאָן כֵּלִים בּוּכָאנֵי, דְּהַיְנוּ שִׁפָּן, דְּלֹא הֲדַר לִבְרִיָּתוֹ, וְנִשְׁתַּנָּה שְׁמוֹ שֶׁנִּקְרָא בּוּכָאנֵי. וְהַיְנוּ דִּתְנַן וַעֲשָׂאָן כֵּלִים: צֶמֶר כוּ'. הָכִי נַמִּי שֶׁגָּזַל צֶמֶר שֶׁאֵינוֹ טָווּי וְעָשָׂה מֵהֶן לְבָדִים, שֶׁאֵינוֹ חוֹזֵר לִבְרִיָּתוֹ לִהְיוֹת כָּל נִימָא וְנִימָא בִּפְנֵי עַצְמָהּ, וְנִשְׁתַּנָּה שְׁמוֹ שֶׁנִּקְרָא לֶבֶד. וְהַיְנוּ דִּתְנַן וַעֲשָׂאָן בְּגָדִים: כִּדְיָלֵיף לֵיהּ בַּגְּמָרָא דַּף ס"ז מִקְּרָא . רַשִׁ"י: הָעוֹמֶדֶת כוּ'. כְּלוֹמַר לַאֲפוֹקֵי שֶׁיָּשׁוּמוּ כְּאִלּוּ יָלְדָה, דְּהָא עַל יְדֵי הַלֵּידָה קְנָאָהּ. וְזֶה לְשׁוֹן הַתּוֹסָפוֹת, וְכָל שֶׁבַח שֶׁמִּשְּׁעַת גְּזֵלָה עַד שֶׁתֵּלֵד, לַגַּזְלָן. עַד כָּאן. וְטַעֲמָא, דִּלְעוֹלָם מְשַׁלֵּם כִּשְׁעַת הַגְּזֵלָה, כִּדְכָיֵל בַּסֵּיפָא, וְקָאֵי אַכּוּלְּהוּ בָּבֵי דְּמַתְנִיתִין, וְלֹא אָזְלִינַן בָּתַר שְׁעַת קְנִיָּה, אֶלָּא מִשֶּׁקְּנָאָהּ חָזְרָה הַשּׁוּמָא לִשְׁעַת הַגְּזֵלָה. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: רְצוֹנוֹ לוֹמַר דְּדַוְקָא הַוָּלָד וְהַגִּזָּה קוֹנֶה בְּשִׁנּוּי וּמְשַׁלֵּם מַה שֶּׁהָיוּ דְּמֵיהֶם, וְהָעֹדֶף שֶׁשָּׁוֶה עַכְשָׁיו יוֹתֵר הוּא שֶׁלּוֹ. אֲבָל פָּרָה עַצְמָהּ חוֹזֶרֶת בְּעֵינָהּ. דְּלֵידָה וְגִזָּה לָאו שִׁנּוּי הוּא בְּגוּף הַפָּרָה כוּ'. נִמּוּקֵי יוֹסֵף. אֲבָל לְהָרֹא"שׁ, דְּמֵי פָּרָה מַמָּשׁ, דַּהֲוֵי שִׁנּוּי אַף לַפָּרָה עַצְמָהּ וּקְנָאָהּ בּוֹ: וְיָלְדָה. הוּא הַדִּין אֲפִלּוּ לֹא יָלְדָה, וְאַיְדֵי דְּנָסֵיב רֵישָׁא וְיָלְדָה, נָסֵיב נַמִּי סֵיפָא וְיָלְדָה. גְּמָרָא. וְאַף עַל גַּב דִּבְרֵישָׁא נַמִּי מְשַׁלֵּם כִּשְׁעַת הַגְּזֵלָה, סֵיפָא הָא קָא מַשְׁמַע לָן דְּעִבּוּר וּטְעִינָה הָוֵי שִׁנּוּי כְּמוֹ לֵידָה. וְנִרְאֶה לִי דְּהָכִי נַמִּי דִּינָא דְּמַחֲזִיר הַפָּרָה לְפֵרוּשׁ הַנִּמּוּקֵי יוֹסֵף וּכְדִלְעֵיל. דְּמַאי שְׁנָא שִׁנּוּי הַמִּלּוּי אוֹ שִׁנּוּי הָרֵקוּת. וְקַשְׁיָא לִי, דְּלִתְנֵי נַמִּי בָּרֵישָׁא מְשַׁלֵּם כִּשְׁעַת הַגְּזֵלָה. וְצָרִיךְ עִיּוּן. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: וְלֹא אָמְרִינַן דְּעַל כָּל פָּנִים יְשַׁלֵּם אַרְבָּעָה וַחֲמִשָּׁה כִּשְׁעַת הַגְּזֵלָה, לְפִי דְּשֶׁלּוֹ הוּא טוֹבֵחַ כוּ'. וּמִמֵּילָא דְּמִכָּל שֶׁכֵּן שֶׁאֵינוֹ מְשַׁלֵּם הַכֶּפֶל אֶלָּא כִּשְׁעַת הַגְּזֵלָה:

מִשּׁוּם דְּבַגְּמָרָא מְהַפְּכִינַן לַהּ כוּ' וְתָנֵי וַחֲכָמִים אוֹמְרִים בַּעֲבָדִים כוּ': כֵּיוָן דְּאוֹרְחַיְהוּ בְּהָכִי, כְּהֶזֵּק שֶׁאֵינוֹ נִכָּר חָשְׁבִינַן לֵיהּ, וְאוֹמֵר לוֹ הֲרֵי כוּ', וְהָוֵי כְּחָמֵץ בְּפֶסַח. אֲבָל מַתְנִיתִין בְּשֶׁהִרְקִיבוּ כֻּלָּן שֶׁהוּא הֶזֵּק נִכָּר כְּיַיִן וְהֶחְמִיץ מַיְרֵי. נִמּוּקֵי יוֹסֵף: שֶׁיָּכוֹל לוֹמַר לוֹ לֵךְ וְהוֹצִיאוֹ שָׁם. רַשִׁ"י: תְּרוּמָה כוּ'. הַטַּעַם דְּאֵלּוּ אַף שֶׁנִּפְסְלוּ אֵצֶל הַגַּזְלָן לְגַמְרֵי, מִכָּל מָקוֹם תְּבוּאַת תְּרוּמָה וְלֶחֶם חָמֵץ לֹא נִשְׁתַּנָּה מִשְּׁאָר תְּבוּאוֹת וּלְחָמִין, וַהֲרֵי הֵן כְּמוֹ שֶׁהָיוּ בִּשְׁעַת גְּזֵלָה וּכְמוֹ שֶׁהֵן עַכְשָׁיו שְׁאָר תְּבוּאוֹת וּלְחָמִין, מִשּׁוּם הָכִי מָצֵי אָמַר לֵיהּ הֲרֵי כוּ'. מַה שֶּׁאֵין כֵּן בִּפְסַלְתּוֹ מַלְכוּת, כֵּיוָן דְּצוּרַת מַטְבֵּעַ זוֹ אֵינָהּ יוֹצְאָה כְּלָל בְּכָל מְדִינוֹת הַמֶּלֶךְ וְנִטְבְּעָה צוּרָה אַחֶרֶת בִּמְקוֹמָהּ, הֲרֵי נִכָּר הֶפְסֵדָהּ. סְמַ"ע: וְנֶעֶבְדָה כוּ'. בְּעֵד אֶחָד, שֶׁלֹּא נִסְקְלָה. אֲבָל פְּסוּלָה לְקָרְבָּן. וְטַעֲמָא דְּאָמַר לֵיהּ הֲרֵי כוּ', דְּכָל בְּהֵמוֹת לָאו לְמִזְבֵּחַ קַיְּמֵי. רַשִׁ"י: לִסָּקֵל. סְבִירָא לֵיהּ לְהַאי תָּנָא דְּגוֹמְרִין דִּינוֹ שֶׁל שׁוֹר שֶׁלֹּא בְּפָנָיו וְאֵין צָרִיךְ לְהָבִיאוֹ לְבֵית דִּין, דְּאָמַר לֵיהּ מַאי עָבְדֵי לֵיהּ, גְּמָרָא:

דְּאֵין אֻמָּן קוֹנֶה בְּשֶׁבַח כְּלִי. גְּמָרָא. וְלֹא דָּמֵי לְגַזְלָן שֶׁמְּשַׁלֵּם כִּשְׁעַת הַגְּזֵלָה, דְּהָכָא לֹא כִּוֵּן מֵעוֹלָם לִגְזֹל וְלֹא זָכָה בָּהּ. וְעַיֵּן מַה שֶּׁכָּתַבְנוּ לְקַמָּן בְּדִבְרֵי רַבִּי מֵאִיר וְרַבִּי יְהוּדָה: חַיָּב. שֶׁזֶּה כְּזוֹרֵק חֵץ וְהִזִּיק בּוֹ הוּא. הָרַמְבַּ"ם:

דְּמֵי כוּ'. וּבְהִקְדִּיחָה בִּשְׁעַת נְפִילָה, שֶׁלֹּא הָיָה בּוֹ שֶׁבַח מֵעוֹלָם. אֲבָל אִם הִקְדִּיחָה לְאַחַר שֶׁקָּלַט וְנַעֲשָׂה שֶׁבַח בַּצֶּמֶר, חַיָּב לִתֵּן לוֹ גַּם דְּמֵי שִׁבְחוֹ, כְּמוֹ נָתַן לָאֻמָּן לְתַקֵּן בְּמַתְנִיתִין דִּלְעֵיל. גְּמָרָא. וְלֹא יָדַעְתִּי לָמָּה לֹא הֶעְתִּיקוּם הַפּוֹסְקִים: הַיְצִיאָה. פֵּרֵשׁ הָרַמְבַּ"ם מַה שֶּׁהוֹצִיא מִדָּמָיו. וְרַשִׁ"י הוֹסִיף שְׂכַר טָרְחוֹ כִּשְׁאָר שְׂכִיר יוֹם: רַשִׁ"י. וּלְקַמָּן בְּדִבְרֵי יְהוּדָה בְּהוֹצָאָה יְתֵרָה עַל שֶׁבַח, פֵּרְשׁוּ שֶׁאִם הַשָּׂכָר פָּחוֹת מִן הַשֶּׁבַח שֶׁנּוֹתֵן לוֹ אֶת הַשָּׂכָר, וּמִסְתַּבֵּר וַדַּאי דְּהָכִי נַמִּי אִם הַשָּׂכָר פָּחוֹת מִן הַיְצִיאוֹת דְּנוֹתֵן לוֹ: וּמַשְׁמַע דְּכִי יָהֵיב לֵיהּ אַגְרָא מִחַיַּב לְמֵיתַב הַצֶּמֶר וְלֹא אָמְרִינַן דְּקָנֵי לֵיהּ לְגַמְרֵי, שֶׁהֲרֵי לֹא נִתְכַּוֵּן הַצּוֹבֵעַ לִזְכּוֹת בּוֹ כְּלָל. מַה שֶּׁאֵין כֵּן בְּגַזְלָן: רַשִׁ"י. וּמַשְׁמַע דַּוְקָא לִדְשִׁנָּה קָנְסִינַן. וְהִלְכָּךְ אִם בַּעַל הַצֶּמֶר אֵינוֹ חָפֵץ בַּצֶּמֶר אֶלָּא שֶׁיִּתֵּן לוֹ דְּמֵי צַמְרוֹ, הָרְשׁוּת בְּיָדוֹ. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: אֶת הַיְצִיאָה. לְפִי שֶׁאֵין מִתְכַּוֵּן לִקְנוֹתוֹ. הִלְכָּךְ לֹא דָּמֵי לְגַזְלָן דְּכֻלָּא שְׁבָחָא דִּידֵיהּ הִיא כְּדִתְנַן רֵישׁ פִּרְקִין. תּוֹסָפוֹת:

שָׁוֶה פְרוּטָה. לְאַשְׁמָה בָהּ, פְּרָט לְפָחוֹת מִשָּׁוֶה פְרוּטָה. סִפְרָא: וְנִשְׁבַּע לוֹ. לְדִבְרֵי הָרַ"ב בְּפֵרוּשׁ מִשְׁנָה ז' פֶּרֶק ד' דְּבָבָא מְצִיעָא דְּעַל כְּפִירַת פְּרוּטָה נִשְׁבָּעִין, אָתֵי שַׁפִּיר. אֲבָל לְפִי דְּבָרָיו בְּרֵישׁ פֶּרֶק ו' דִּשְׁבוּעוֹת אֵין נִשְׁבָּעִין עַל כְּפִירָה פְּחוּתָה מִשְּׁתֵּי כֶּסֶף. פֵּרְשׁוּ הַתּוֹסָפוֹת דְּעַל יְדֵי בֵּית דִּין אֵין נִשְׁבָּעִין, וְהָכָא בְּנִשְׁבָּע עַל יְדֵי עַצְמוֹ: וּמְסַיֵּם נִמּוּקֵי יוֹסֵף, דִּכְתִיב בְּיוֹם אַשְׁמָתוֹ, דְּמַשְׁמַע שֶׁנִּשְׁבַּע לַשֶּׁקֶר וּמֵבִיא אֲשָׁמוֹ. וְהָרַמְבַּ"ם נָתַן טַעַם לַדָּבָר מִפְּנֵי שֶׁכְּבָר נִתְיָאֲשׁוּ מֵאַחַר שֶׁנִּשְׁבַּע וְאֵין בָּאִין עוֹד לְתָבְעוֹ. וְלִדְבָרָיו בְּאֵינוֹ נִשְׁבַּע אֲפִלּוּ לְהוֹדִיעוֹ אֵינוֹ צָרִיךְ. אֲבָל הַטּוּר שֶׁכָּתַב שֶׁצָּרִיךְ לְהוֹדִיעוֹ לוֹמַר לוֹ כָּךְ וְכָךְ גְּזַלְתִּיךָ בֹּא וְקַח מִמֶּנִּי: לְהוֹלִיכוֹ שֶׁאִם יַאַנְסוּהוּ חַיָּב לְהַחְזִיר וְלֹא הָוֵי הֲשָׁבָה כוּ'. רַשִׁ"י: לִשְׁלוּחוֹ. עַיֵּן פֶּרֶק ח' דְּבָבָא מְצִיעָא מִשְׁנָה ג' בְּדִבְרֵי הָרַ"ב, דִּפְלֻגְתָּא דְּשָׁלִיחַ בְּעֵדִים שָׁיְכָא נַמִּי הָכָא. וְכֵן הוּא בְּהֶדְיָא בַּגְּמָרָא: רַשִׁ"י. וְכָתְבוּ הַתּוֹסָפוֹת, מִתּוֹךְ פֵּרוּשׁוֹ מַשְׁמַע דְּנוֹתְנוֹ לִשְׁלִיחַ בֵּית דִּין וְהוּא שׁוֹמְרוֹ בְּיָדוֹ עַד שֶׁיָּבֹא הַנִּגְזָל. וְאֵין נִרְאֶה כוּ'. וּמַסְּקֵי דְּנִרְאֶה דְּמַתְנִיתִין אַיְרֵי לְעִנְיַן אַחֲרָיוּת הַדֶּרֶךְ, דְּמִפְטַר כְּשֶׁיִּתֵּן לִשְׁלִיחַ בֵּית דִּין וְיֵשׁ לוֹ כַּפָּרָה מִיָּד וְאֵין צָרִיךְ לוֹ לְהַמְתִּין עַד שֶׁיַּגִּיעַ לְיַד הַנִּגְזָל, כֵּיוָן שֶׁקִּבְּלוֹ שְׁלִיחַ בֵּית דִּין. וְכֵן פֵּרֵשׁ הָרַמְבַּ"ם: לְיוֹרְשָׁיו. נִרְאֶה דְּאָתֵי לְאַשְׁמוֹעִינַן דַּאֲפִלּוּ לַיּוֹרְשִׁים יוֹלִיךְ לְמָדַי. אִי נַמִּי לְעִנְיַן קֶרֶן וְחֹמֶשׁ וְאָשָׁם. וְלֹא דָּמִי לְיוֹרְשֵׁי גַּזְלָן דְּאֵין מְשַׁלְּמִין חֹמֶשׁ. תּוֹסָפוֹת:

לֵילֵךְ כוּ'. אֲבָל לִכְשֶׁיָּבֹא מְחֻיָּב לָתֵת לוֹ מִדִּבְרֵי תּוֹרָה:

נָתַן כוּ'. הוּא הַדִּין אִם לֹא נָתַן לוֹ אֶת הַקֶּרֶן וְנִשְׁבַּע אַתַּרְוַיְהוּ אַקֶּרֶן וְאַחֹמֶשׁ הָיָה מוֹסִיף חֹמֶשׁ, אֶלָּא אוֹרְחָא דְּמִלְּתָא נָקַט, דְּאַקֶּרֶן אֵין נִשְׁבָּעִין שְׁתֵּי פְּעָמִים. וְהוּא הַדִּין אִם נִשְׁבַּע עַל חֹמֶשׁ דְּיוֹלִיכֶנּוּ אַחֲרָיו לְמָדַי, כֵּיוָן שֶׁנִּשְׁבַּע עָלָיו נַעֲשָׂה כְּגַזְלָן. תּוֹסָפוֹת: הֲרֵי זֶה כוּ'. וְכֵן לְעִנְיַן דִּינֵי נִשְׁבַּע וְהוֹדָה. וְתַנָּא דִּידָן נָקַט בְּגָזֵל. מִשּׁוּם דְּפִרְקִין בְּדִינֵי גָּזֵל אַיְרֵי:

שֶׁגְּנָבוֹ. לְעֵיל נָקַט שֶׁאֲכָלוֹ, לִרְבוּתָא אַף עַל פִּי שֶׁאִבְּדוֹ מִן הָעוֹלָם פָּטוּר מִן הַכֶּפֶל בְּטַעֲנַת אָבַד. וְהָכָא נָקַט שֶׁגְּנָבוֹ לִרְבוּתָא דְּאַף עַל פִּי שֶׁהוּא בְּעֵין חַיָּב כֶּפֶל כֵּיוָן שֶׁטּוֹעֵן טַעֲנַת גַּנָּב. תּוֹסָפוֹת: כֶּפֶל. אֲבָל תַּשְׁלוּמֵי אַרְבָּעָה וַחֲמִשָּׁה לֹא קָתָנֵי דְּלִתְנֵי טָבַח וּמָכַר כוּ', מִשּׁוּם דְּבִקְרָא לֹא כְּתִיב אַרְבָּעָה וַחֲמִשָּׁה אֶלָּא בַּגַּנָּב עַצְמוֹ. תּוֹסָפוֹת:

וְנִשְׁבַּע לוֹ. מַמָּשׁ, דְּאִי לֹא נִשְׁבַּע לוֹ לֹא הָיָה צָרִיךְ לְהוֹצִיא חֵלֶק הַנּוֹגֵעַ לִירֻשָּׁתוֹ מֵהַגְּזֵלָה. וְאֶפְשָׁר דְּהוּא הַדִּין לֹא נִשְׁבַּע, וְנִשְׁבַּע דְּקָתָנֵי מִשּׁוּם סֵיפָא דְּלֹוֶה וּבַעַל חוֹב בָּאִין וְנִפְרָעִין, לְאַשְׁמְעִינַן דְּאַף עַל גַּב דְּנִשְׁבַּע שָׁרֵי בְּהָכִי. בֵּית יוֹסֵף. וְהַסְּמַ"ע כָּתַב דְּהַדַּרְכֵי מֹשֶׁה הִכְרִיעַ דְּנִשְׁבַּע דַּוְקָא: לְבָנָיו כוּ'. פֵּרְשׁוּ הַתּוֹסָפוֹת, דִּלְבָנָיו שֶׁל גַּזְלָן אוֹ לְאֶחָיו שֶׁל גַּזְלָן, כְּלוֹמַר בַּתְּחִלָּה לְבָנָיו אִם יֵשׁ לוֹ, דְּהֵם קוֹדְמִים לְכָל אָדָם בַּחֵלֶק הַמַּגִּיעוֹ, וְנַחְשֹׁב אוֹתוֹ כְּאִלּוּ מֵת הוּא וַהֲרֵי בָּנָיו יוֹרְשִׁים חֶלְקוֹ עִם אֶחָיו, וְהֵם אֵין בָּאִין מִכֹּחוֹ אֶלָּא מִכֹּחַ אֲבוּהַּ דְּאַבָּא כִּדְאָמַר בְּבָבָא בַּתְרָא דַּף קנ"ט תַּחַת אֲבוֹתֶיךָ יִהְיוּ בָנֶיךָ. הִלְכָּךְ אַף עַל פִּי שֶׁאֵין לַאֲבִיהֶן חֵלֶק בָּהֶם כְּלוּם שֶׁהֲרֵי יֵשׁ לוֹ לְהַחֲזִיר הַגְּזֵלָה, אֲפִלּוּ הָכִי הֵם קוֹדְמִין בְּחֶלְקוֹ מֵהַהוּא טַעֲמָא דְּבָבָא בַּתְרָא. אוֹ לְאֶחָיו כְּשֶׁאֵין לוֹ בָּנִים. וְכֵן פָּסַק הָרַמְבַּ"ם, אֶלָּא דִּסְבִירָא לֵיהּ דְּמַתְנִיתִין לָאו דַּוְקָא נָקַט בָּנָיו בָּרֵישָׁא, דְּוַדַּאי כָּל יְמֵי חַיָּיו אֵין חֵלֶק לְבָנָיו בִּירֻשַּׁת אָבִיו. וְרַשִׁ"י מְפָרֵשׁ לְבָנָיו שֶׁל אָבִיו אוֹ לְאֶחָיו שֶׁל אָבִיו. וּלְשׁוֹן הָרַ"ב אִי אֶפְשָׁר לְפָרֵשׁ כְּשׁוּם אֶחָד מֵהַפֵּרוּשִׁים. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: הָרַמְבַּ"ם. וְטַעֲמָא דְּאִי קְנָאָהּ בְּשִׁנּוּי אֵין עָלָיו אֶלָּא לְשַׁלֵּם דְּמֵי הַגְּזֵלָה וְלֹא הָוֵי בִּכְלַל וְהֵשִׁיב אֶת הַגְּזֵלָה, דְּהָא קְרָא קָאָמַר אֲשֶׁר גָּזַל, אִם כְּעֵין שֶׁגָּזַל, יַחֲזִיר, כְּמוֹ שֶׁכָּתַבְתִּי בְּרֵישׁ פִּרְקִין. שְׁמַע מִנַּהּ דִּקְרָא לֹא אַיְרֵי אֶלָּא אִם הוּא כְּעֵין שֶׁגָּזַל שֶׁלֹּא נִשְׁתַּנָּה. נִרְאֶה לִי: עֵרוּב פֵּרוּשִׁים אֲנִי רוֹאֶה בְּדִבְרֵי הָרַ"ב. דְּזֶה יָצָא לוֹ מֵהָרַמְבַּ"ם, וְהוּא מְפָרֵשׁ שֶׁאֵין לוֹ אַחִים. וְהָרַ"ב כְּבָר פֵּרֵשׁ אֵין לוֹ נְכָסִים, כְּשִׁיטַת שְׁאָר הַמְפָרְשִׁים, וּלְדִידְהוּ יֵשׁ לוֹמַר דְּנָקַט בָּנָיו כוּ' דִּשְׁכִיחֵי, וְהוּא הַדִּין אִם אֵין לוֹ דְּהַיְרֻשָּׁה מְמַשְׁמֶשֶׁת וְעוֹלָה עַד יַעֲקֹב אָבִינוּ, וַאֲלֵיהֶם הוּא נוֹתֵן. וְזֶה לְשׁוֹן הָרַמְבַּ"ם, וְאִם לֹא הָיָה אֶצְלוֹ אָח מֵאָבִיו וְלֹא בָּנִים (וְהוּא פֵּרוּשׁ דִּוְאִם אֵין לוֹ) אוֹ לֹא הֶחֱזִיר הוּא אֶת הַגְּזֵלָה לְבָנָיו (וְהוּא פֵּרוּשׁ דִּוְאִם אֵינוֹ רוֹצֶה) יִתֵּן אוֹתוֹ בְּחוֹבוֹ וְכוּ'. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: וְנִפְרָעִים. דְּמַה שֶּׁמַּחֲזִיר לָהֶם לֹא חָשֵׁיב כְּאִלּוּ נָתַן לָהֶם בְּמַתָּנָה וְשֶׁלֹּא יוּכְלוּ בַּעֲלֵי חוֹב לִטְרֹף, דְּאֵין אַחֲרָיוּת עַל הַמִּטַּלְטְלִים. דְּאֵינוֹ נוֹתֵן לָהֶם לְשֵׁם מַתָּנָה אֶלָּא בִּסְתָמָא בִּשְׁתִיקָה מַחֲזִיר לָהֶם וְאֵין דַּעְתּוֹ אֶלָּא לְקַיֵּם מִצְוַת הֲשָׁבָה בְּעָלְמָא, הִלְכָּךְ לֹא הַוְיָא מַתָּנָה, וְיָכוֹל בַּעַל חוֹב לִטְרֹף מֵהֶם. וְהוּא וַדַּאי אֵינוֹ יָכוֹל לַחֲזֹר וְלִקַּח מֵהֶן, דְּאָז לֹא הָיָה מִתְקַיֵּם הֲשָׁבָה כְּלָל אִם הָיָה יָכוֹל לִקַּח בְּעַצְמוֹ מַה שֶּׁנָּתַן. אֲבָל מַה שֶּׁבַּעַל חוֹב נוֹטֵל מִתְקַיֵּם שַׁפִּיר מִצְוַת הֲשָׁבָה, כֵּיוָן שֶׁבְּעַצְמוֹ אֵינוֹ נוֹטֵל. תּוֹסָפוֹת:

קוֹנָם. מְבֹאָר בְּרֵישׁ נְדָרִים שֶׁהָאוֹמֵר קוֹנָם תּוֹפֵס הַחֵפֶץ בְּקָרְבָּן, שֶׁקּוֹנָם כִּנּוּי לְקָרְבָּן הוּא, וְיִהְיֶה פֵּרוּשׁ הַלָּשׁוֹן, קוֹנָם יִהְיֶה אִם אַתָּה נֶהֱנֶה כוּ'. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: יִירָשֶׁנּוּ. שֶׁזֶּה כְּאוֹמֵר נְכָסַי עָלֶיךָ אֲסוּרִים (נְדָרִים פ"ה מ"ג). הָרַמְבַּ"ם: לֹא כוּ'. שֶׁזֶּה כְּמוֹ שֶּׁאָמַר לֵיהּ נְכָסִים אֵלּוּ אֲסוּרִים עָלֶיךָ. שָׁם. וְכָתַב הָרַ"ן לָאו לְמֵימְרָא דְּכֵיוָן דִּנְכָסִים אֲסוּרִים לוֹ בַּהֲנָאָה לֹא יִזְכֶּה בָּהֶן, דְּהָא קָתָנֵי סֵיפָא וְיִתֵּן לְבָנָיו וּלְאֶחָיו וְאִם אֵין לוֹ לֹוֶה וּבַעֲלֵי חוֹב כוּ', וְאִם לֹא זָכָה בְּגוּף הַנְּכָסִים הֵיאַךְ נוֹתְנָם לְבָנָיו אוֹ לְאֶחָיו, וְהֵיאַךְ בַּעֲלֵי חוֹב בָּאִים וְנִפְרָעִים, אֶלָּא וַדַּאי נְכָסִים דִּידֵיהּ נִינְהוּ אֶלָּא שֶׁאֵינוֹ רַשַּׁאי לֵהָנוֹת מֵהֶם. עַד כָּאן. וּמִשּׁוּם הָכִי נִיחָא לְשׁוֹן וְיַחֲזִיר. אֲבָל הָרַ"ן הֶעְתִּיק וְיִתֵּן. וְכֵן הוּא בַּיְרוּשַׁלְמִי: וְיַחֲזִיר. לָאו לְמֵימְרָא שֶׁיִּתְּנוּ לָהֶם מַמָּשׁ, אֶלָּא שֶׁמַּרְאֶה לָהֶם מָקוֹם וְאוֹמֵר לָהֶם נְכָסַי אֵלּוּ אָסַר עָלַי אַבָּא וְאֵינִי יוֹדֵעַ מַה אֶעֱשֶׂה בָּהֶם, טְלוּ לְעַצְמְכֶם וַעֲשׂוּ מֵהֶם מַה שֶּׁתִּרְצוּ. וּכְהַאי גַּוְנָא לָאו מַתָּנָה הוּא אֶלָּא גַּוְנָא דְּהֶפְקֵר. אֲבָל הַבֵּית יוֹסֵף כָּתַב שֶׁאֲפִלּוּ לִתְּנָם מַמָּשׁ מֻתָּר, כֵּיוָן שֶׁמּוֹדִיעָם שֶׁמִּפְּנֵי שֶׁאֲסוּרִים עָלָיו נוֹתְנָם לָהֶם אֵין כָּאן טוֹבַת הֲנָאָה. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: לָאו דַּוְקָא. וְעַיֵּן פֶּרֶק ד' דִּנְדָרִים מִשְׁנָה ז'. וְלֹא תָּנֵי הָכָא נַמִּי אֵינוֹ רוֹצֶה, מִשּׁוּם דְּוַדַּאי אֵינוֹ רוֹצֶה. תּוֹסָפוֹת. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: וְנִפְרָעִים. אַחַר מִיתָה. וַאֲפִלּוּ בְּחַיָּיו הֲוָה שָׁרֵי בִּכְהַאי גַּוְנָא, כְּדִתְנַן (שָׁם) הַמֻּדָּר הֲנָאָה מֵחֲבֵרוֹ וְאֵין לוֹ מַה יֹּאכַל יֵלֵךְ אֵצֶל חֶנְוָנִי כוּ'. אֶלָּא לְפִי שֶׁזֶּה אֵין רוֹצֶה שֶׁיֵּהָנֶה מִמֶּנּוּ כְּלָל בְּחַיָּיו וְאֵינוֹ יָכוֹל לַעֲשׂוֹת תַּקָּנָה אֶלָּא מִדַּעְתּוֹ. תּוֹסָפוֹת:

לַכֹּהֵן. תָּנוּ רַבָּנָן אָשָׁם זֶה קֶרֶן. הַמּוּשָׁב זֶה חֹמֶשׁ. כְּשֶׁהוּא אוֹמֵר מִלְּבַד אֵיל הַכִּפּוּרִים, הֱוֵי אוֹמֵר אָשָׁם זֶה קֶרֶן. וְלָמָּה קוֹרֵהוֹ אָשָׁם, שֶׁאִם הֶחֱזִירוֹ בַּלַּיְלָה דְּלֹא יָצָא, הֶחֱזִירוֹ לַחֲצָאִין לֹא יָצָא, וְאֵין חוֹלְקִים גֶּזֶל הַגֵּר כְּנֶגֶד גֶּזֶל הַגֵּר, כְּדִין אָשָׁם. גְּמָרָא:

מַשְׁמַע דִּרְצוֹנוֹ לוֹמַר טַעֲמָא שֶׁזָּכוּ בַּמֶּה שֶׁבְּיָדָם כְּדִין נִכְסֵי הַגֵּר כוּ' שֶׁכָּל הַקּוֹדֵם בָּהֶן זָכָה. וּתְמֵהַנִי, דְּאִי הָכִי קְרָא לְמַאי אִצְטְרִיךְ כוּ'. וּבַגְּמָרָא אָמַר אַבַּיֵי שְׁמַע מִנַּהּ כֶּסֶף מְכַפֵּר מֶחֱצָה, דְּאִי לֹא מְכַפֵּר הֲוָה אֲמֵינָא מְהַדַּר לַיּוֹרְשִׁין, מַאי טַעֲמָא, אַדַּעְתָּא דְּהָכִי (דְּלֹא תַּהֲוֵי לֵיהּ כַּפָּרָה) לֹא יָהֵיב לֵיהּ כוּ'. הָא קַמָּן דְּהַטַּעַם מִשּׁוּם דִּמְכַפֵּר מֶחֱצָה. וּלְהַךְ סְבָרָא בָּעִינַן קְרָא דְּלֹא הֲוָה יָדְעִינַן דְּכִפֵּר מֶחֱצָה. אֶלָּא הֲוָה אֲמֵינָא דְּאֵין כַּפָּרָה עַד שֶׁיָּבִיא קָרְבָּן: יָצָא. יָלְפִינַן לֵיהּ בַּגְּמָרָא מִקְּרָא: אֵין כוּ'. בַּגְּמָרָא יָלֵיף לֵיהּ: אֵינוֹ מְדֻקְדָּק, דְּמַשְׁמַע דְּאִי לֹא נָתַן לְבַסּוֹף עִכֵּב, וּכְבָר פֵּרֵשׁ דְּהָעַכָּבָה הַיְנוּ עַל הַהַקְרָבָה, וְכֵיוָן שֶׁכְּבָר קָרְבָה אֵין עוֹד עַכָּבָה כוּ'. וְאוּלַי רְצוֹנוֹ לוֹמַר דְּלָא תֵימָא דִּבְדִיעֲבַד סַגִּי לֵיהּ וְיִפָּטֵר מִן הַחֹמֶשׁ, לְהָכִי קָאָמַר וּלְבַסּוֹף נָתַן, כְּלוֹמַר שֶׁלְּבַסּוֹף יִתֵּן וְלֹא יִפָּטֵר בְּכָךְ. וְקָרֵי בֵּיהּ נוֹתֵן. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב:
