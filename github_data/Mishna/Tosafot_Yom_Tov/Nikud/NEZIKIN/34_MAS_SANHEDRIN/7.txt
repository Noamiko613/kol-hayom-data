פרק ז

מִשְׁנָה ד' פֶּרֶק ט'. וְהָא דְּלֹא נָקַט הַמִּשְׁנָה דִלְעֵיל מֵהַהִיא דִשְׁנַיִם מֵחַיָּבֵי מִיתוֹת בֵּית דִּין שֶׁנִּתְעָרְבוּ נִדּוֹנִין בַּקַּלָּה. וְיֵשׁ לוֹמַר דִּרְבוּתָא דַּחֲמִירָא עֲדִיפָא. וְעוֹד יֵשׁ לוֹמַר דְּנִיחָא לְמִנְקַט נַפְקוּתָא בְּאָדָם אֶחָד: חֶנֶק. אַף עַל גַּב דְּלֹא כְתִיב בְּהֶדְיָא, אַרְבַּע מִיתוֹת גְּמָרָא גְמִירֵי לַהּ. וְכָל מִיתָה הָאֲמוּרָה בַתּוֹרָה בִסְתָם אִי אַתָּה רַשַּׁאי לְהַחֲמִיר עָלֶיהָ, אֶלָּא לְהָקֵל עָלֶיהָ. וּלְרַבִּי שִׁמְעוֹן כוּ', מַה מִּיתָה בִידֵי שָׁמַיִם אֵין בָּהּ רֹשֶׁם אַף כוּ'. וְאֵימָא שְׂרֵפָה, מִדְּבַת כֹּהֵן בִּשְׂרֵפָה מִכְּלָל כוּ'. גְּמָרָא:

אַרְכֻּבּוֹתָיו. לְשׁוֹן וְאַרְכֻבָּתֵהּ דָּא לְדָא נָקְשָׁן , וְהוּא הַפֶּרֶק הָאֶמְצָעִי שֶׁבֵּין הָרֶגֶל וְהַיָּרֵךְ: לְשׁוֹן רַשִׁ"י. מִפְּנֵי שֶׁהָרַכָּה אֵינָהּ חוֹנֶקֶת לִפְתֹּחַ פִּיו וְהַקָּשָׁה מְחַבֶּלֶת אֶת גְּרוֹנוֹ וּמְנַוַּלְתּוֹ, וּלְפִיכָךְ קָשָׁה מִבִּפְנִים. וְהָרַמְבַּ"ם כָּתַב כְּדֵי שֶׁלֹּא יִצְטַעֵר בְּשָׂרוֹ בַּלֶּבֶד (מִלְּשׁוֹן לְבָדִין) הַקָּשֶׁה בִּשְׁעַת חֶנְקוֹ: זֶה כוּ'. הַיְנוּ שְׁנֵי עֵדָיו. הָרַמְבַּ"ם: וְאַף. לִישָׁנָא בְעָלְמָא הוּא, כְּלוֹמַר אִם יַחַנְקוּהוּ אַף הוּא יָמוּת בְּיָדָם וְאֵין זוֹ שְׂרֵפָה. רַשִׁ"י: בִּצְבָת. תַּרְגּוּם וּמַלְקָחֶיהָ, וּצְבָתָהָא:

הָכִי תַנְיָא, אָמַר לָהֶן רַבִּי יְהוּדָה לְרַבָּנָן אַף אֲנִי יוֹדֵעַ שֶׁמִּיתָה מְנֻוֶּלֶת הִיא, אֲבָל מָה אֶעֱשֶׂה שֶׁהֲרֵי אָמְרָה תוֹרָה כוּ'. וְנִמְצֵינוּ לְמֵדִין דְּרַבִּי יְהוּדָה לֹא יִקְרָא נִוּוּל לַמִּיתָה שֶׁאָמְרוּ רַבָּנָן כוּ'. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: נְקִימָה זוֹ אֵינִי יוֹדֵעַ מָה הִיא, כְּשֶׁהוּא אוֹמֵר וְהֵבֵאתִי עֲלֵיכֶם חֶרֶב נֹקֶמֶת נְקַם בְּרִית הֱוֵי אוֹמֵר נְקִימָה זוֹ סַיִף:

וְנִשְׁאֲרוּ מִקְצָת שֶׁלֹּא נִכְתַּב בָּהֶן דָּמָיו כוּ', וְהֵם הַבָּא עַל הַבְּהֵמָה וְהָאִשָּׁה הַמְּבִיאָה כוּ', וְיָלְפִינַן לְהוּ בַגְּמָרָא. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: הַזְּכוּר. נֶקֶב הַמִּשְׁכָּב מִקְּרֵי זְכוּר. רַשִׁ"י. וְכָתְבוּ הַתּוֹסָפוֹת דְּהָא דְלָא קָתָנֵי וְהֵבִיא זָכָר עָלָיו, אַף עַל גַּב דִּכְתִיב בְּהֶדְיָא מוֹת יוּמְתוּ שְׁנֵיהֶם, אִיכָּא לְמֵימַר דְּלִישְׁנָא דִקְרָא נָקַט דִּכְתִיב וְאֶת זָכָר לֹא תִשְׁכַּב: וְעַל הַבְּהֵמָה. וְהַמֵּבִיא בְהֵמָה עָלָיו נַמִּי חַיָּב. וְכָתְבוּ הַתּוֹסָפוֹת דְּמִשּׁוּם דְּלֹא כְתִיב בְּהֶדְיָא אֶלָּא מִדְּרָשָׁא, לֹא קָתָנֵי לַהּ: הַבָּא כוּ'. בְּשׁוֹגֵג. רַשִׁ"י: וּמִשּׁוּם כוּ'. הָא דְלָא קָתָנֵי אֵשֶׁת אִישׁ, מִשּׁוּם דְּהָכָא לְאַחַר מִיתָה אַיְרֵי. תּוֹסָפוֹת: אֶלָּא כוּ'. בַּגְּמָרָא פְלִיגֵי בִּדְרָשָׁא דִּקְרָאֵי. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: וּמִשּׁוּם כוּ'. וְרַבִּי יְהוּדָה פָּלֵיג נַמִּי בְהָא, כִּדְאִיתָא בַגְּמָרָא. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: לְאַחַר מִיתַת כוּ'. אַחִיּוּב דְּאֵשֶׁת הָאָב קָאֵי, דִּמְרַבֵּי לֵיהּ קְרָא. אֲבָל שֵׁם אֵשֶׁת אִישׁ לְאַחַר מִיתָה לֹא קָאָמַר. רַשִׁ"י: כְּלוֹמַר דְּקִיחָה זוֹ לָאו קִיחַת חֻפָּה הִיא, אֶלָּא קִיחַת קִדּוּשֵׁי כֶסֶף, דְּגָמַר מִשְּׂדֵה עֶפְרוֹן: הַזְּכוּר. תֵּימַהּ לְמַאי נָפְקָא מִנַּהּ נָקַט לֵיהּ הָכָא: אִם אָדָם כוּ'. אַף עַל פִּי שֶׁלֹּא נִשְׁנָה מְפֹרָשׁ בְּמִשְׁנָתֵנוּ שֶׁהַבְּהֵמָה נֶהֱרֶגֶת, מִכָּל מָקוֹם בְּכֹחַ מַה שֶּׁנִּשְׁנָה שֶׁהַשּׁוֹכֵב עִמָּהּ בִּסְקִילָה שֶׁלֹּא לְמַדְנוּהוּ אֶלָּא מִמַּה שֶׁהַבְּהֵמָה נִסְקֶלֶת (כִּדְאִיתָא בַגְּמָרָא), שַׁיָּךְ שַׁפִּיר לְמִתְנֵי הָא דְּאִם אָדָם חָטָא כוּ': תַּקָּלָה. מִכְשׁוֹל עָוֹן. רַשִׁ"י:

שֶׁיְּפָרֵשׁ. שֶׁיַּזְכִּיר אֶת הַשֵּׁם. אֲבָל אִם לֹא הוֹצִיא שֵׁם מִפִּיו אֶלָּא שָׁמַע שֵׁם יוֹצֵא מִפִּי אַחֵר וּבֵרְכוֹ, פָּטוּר. רַשִׁ"י: וּבַגְּמָרָא יָלְפִינַן דְּהַאי נוֹקֵב לִישְׁנָא דְבָרוֹכֵי הוּא. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: כָּל זֶה כָתוּב בְּפֵרוּשׁ רַשִׁ"י. וּלְשׁוֹן הָרַמְבַּ"ם, שֵׁם בֶּן אַרְבַּע אוֹתִיּוֹת, אָלֶף דָּלֶ"ת נוּ"ן יוּ"ד, וְיֵשׁ מִי שֶׁמְּפָרֵשׁ שֶׁאֵינוֹ חַיָּב אֶלָּא עַל שֵׁם הֲוָיָ"ה, וַאֲנִי אוֹמֵר שֶׁעַל שְׁנֵיהֶם הוּא נִסְקָל. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: אֱמֹר כוּ'. בַּיְרוּשַׁלְמִי פָרֵיךְ, וְאָמְרִינַן לֵיהּ גְּדֹף. וּמְשָׁנֵי אֶלָּא אוֹתוֹ הַשֵּׁם שֶׁאָמַרְתִּי לִפְנֵיכֶם קִלֵּל, וּבוֹ קִלֵּל, כְּלוֹמַר מַזְכִּיר שֵׁם בֶּן אַרְבַּע אוֹתִיּוֹת, וְאוֹמֵר הַשֵּׁם הַזֶּה הִזְכִּיר הַמְגַדֵּף וְקִלֵּל בּוֹ, כְּלוֹמַר יַכֶּה יוֹסֵי אֶת יוֹסֵי. וְלֹא יָדַעְתִּי לָמָּה לֹא כְתָבוֹ הָרַמְבַּ"ם. כֶּסֶף מִשְׁנֶה. וְלִי נִרְאֶה דְּהָרַמְבַּ"ם סְבִירָא לֵיהּ דִּגְמָרָא דִידָן פָּלֵיג. וְקֻשְׁיַת הַיְרוּשַׁלְמִי לֹא קַשְׁיָא, דְּמִידִי דַהֲוָה אַמְּחִיקַת שֵׁם בְּסוֹטָה, דְּכָךְ צִוְּתָה הַתּוֹרָה הֵיכָא דְאִי אֶפְשָׁר בְּלָאו הָכִי. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: עוֹמְדִין כוּ' וְקוֹרְעִין וְלֹא מְאַחִין. בַּגְּמָרָא יָלְפִינַן כָּל זֶה מִקְּרָאֵי (שׁוֹפְטִים ג וּמְלָכִים ב יח). וְהָא דְלָא תְנַן שֶׁהָעֵדִים קוֹרְעִים, לְפִי שֶׁכְּבָר קָרְעוּ בְּשָׁעָה שֶׁשָּׁמְעוּ, וְשׁוּב אֵין חַיָּבִים כִּדְיָלְפִינַן מִקְּרָא. גְּמָרָא: כָּמוֹהוּ. שְׁמַע מִנַּהּ דְּכָשֵׁר בְּדִינֵי מָמוֹנוֹת וּבְדִינֵי נְפָשׁוֹת, וּמַעֲלָה הוּא דְּעָבוּד רַבָּנָן. וְהָכָא כֵּיוָן דְּלֹא אֶפְשָׁר אוֹקְמוּהוּ רַבָּנָן אַדְּאוֹרַיְתָא. דְּאִי סָלְקָא דַעְתָּךְ פָּסוּל, הָכָא מִשּׁוּם דְּאִי אֶפְשָׁר קָטְלִינַן לְגַבְרָא, בִּתְמִיהָה. גְּמָרָא:

הַמְנַסֵּךְ. וּבִכְלָלוֹ נַמִּי זוֹרֵק. כְּדִכְתִיב בַּל אַסִּיךְ נִסְכֵּיהֶם מִדָּם. גְּמָרָא. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: תַּרְגּוּם וַיְחַבֶּק לוֹ, וְגַפֵּף לֵיהּ. הָרַמְבַּ"ם: אִם אֶרְחַץ. רַשִׁ"י: נִשְׁבָּע מְתַרְגֵּם מְקַיֵּם. רַשִׁ"י: וְכֵן פָּסַק הָרַמְבַּ"ם. דְּאִלּוּ בְמֵזִיד וּלְחִיּוּב מִיתָה אִי אֶפְשָׁר, דְּהָא מֻטְעֶה הוּא. כֶּסֶף מִשְׁנֶה. וְדַעַת רַשִׁ"י דְּחַיָּב מִיתָה בְמֵזִיד כֵּיוָן דְּהִתְרוּ בוֹ. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: לְמַרְקוּלִיס. מְצַדְּדִין שָׁלֹשׁ אֲבָנִים אַחַת מִכָּאן וְאַחַת מִכָּאן וְאַחַת מִלְמַעְלָה עַל גַּבֵּיהֶן, וְקוֹרִין אוֹתוֹ מַרְקוּלִיס. רַשִׁ"י:

הַעֲבָרָה זוֹ אֵינָהּ שְׂרֵפָה כְּמוֹ שֶׁחָשַׁב הָמוֹן רַב לִמְאֹד. אֲבָל עִקָּרוֹ הוּא כְּשֶׁהָיוּ מַבְעִירִין אֵשׁ וְהָיוּ מְלַהֲטִין אוֹתוֹ לְשֵׁם אוֹתוֹ הַפֶּסֶל הַיָּדוּעַ שֶׁהָיוּ עוֹבְדִין אוֹתוֹ בְכָךְ וְהוּא הַנִּקְרָא מֹלֶךְ, וְהָיָה אוֹחֵז אֶחָד מִבָּנָיו וְנָתַן אוֹתוֹ בְּיַד הַמְשַׁמֵּשׁ בְּעֵסֶק אוֹתוֹ הַנֶּעֱבָד וְהָיָה מַעֲבִירוֹ עַל הָאֵשׁ מִצַּד זֶה לְצַד זֶה. הָרַמְבַּ"ם. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: מַשְׁמַע לֵיהּ לְתַנָּא נְתִינָה הַבָּאָה לִכְלַל הַעֲבָרָה. וְלֹא מַשְׁמַע לֵיהּ נְתִינָה עַל מְנָת לְהַעֲבִיר לְחַיּוֹבֵיהּ עַל נְתִינָה לְחוּדָהּ. רַשִׁ"י: פִּתּוֹם. שֵׁם הַמְכַשֵּׁף. רַשִׁ"י: וְאֵין זֶה דּוֹרֵשׁ אֶל הַמֵּתִים, לְפִי שֶׁדּוֹרֵשׁ אֵינוֹ עוֹשֶׂה מַעֲשֶׂה אֶלָּא כִדְתַנְיָא וְדוֹרֵשׁ כוּ' זֶה הַמַּרְעִיב עַצְמוֹ וְהוֹלֵךְ וְלָן בְּבֵית הַקְּבָרוֹת כְּדֵי שֶׁתִּשְׁרֶה עָלָיו רוּחַ טֻמְאָה. פֵּרֵשׁ רַשִׁ"י שֵׁד שֶׁל בֵּית הַקְּבָרוֹת יְהֵא אוֹהֲבוֹ וּמְסַיְּעוֹ בִכְשָׁפָיו. וְהָרַמְבַּ"ם כָּתַב כְּדֵי שֶׁיּוֹדִיעֵהוּ הַמֵּת בַּחֲלוֹם מַה שֶּׁשָּׁאַל עָלָיו: בְּוָי"ו. וְכֵן מַשְׁמַע בַּגְּמָרָא דְּתַרְתֵּי נִינְהוּ: לְשׁוֹן רַשִׁ"י, מַעֲלֶה הַמֵּת מִן הָאָרֶץ וּמוֹשִׁיב לוֹ בַשֶּׁחִי תַּחַת זְרוֹעוֹתָיו וּמְדַבֵּר מִשֶּׁחְיוֹ: עַיֵּן פֶּרֶק ח' דְּכִלְאַיִם מִשְׁנָה ה'. וּבַגְּמָרָא, יִדְּעוֹנִי זֶה הַמַּנִּיחַ עֶצֶם יַדּוּעַ בְּפִיו וְהוּא מְדַבֵּר מֵאֵלָיו. (עַל יְדֵי כְשָׁפִים. רַשִׁ"י): וְאַזְהָרָה דִמְכַשְּׁפִים גּוּפַיְהוּ מִלֹּא יִמָּצֵא בְךָ. רַשִׁ"י. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב:

בְּדָבָר כוּ'. מִכְּלָל דְּאִיכָּא מִידֵי דְּחִלּוּל שַׁבָּת הָוֵי וְאֵין חַיָּבִין לֹא חַטָּאת וְלֹא כָרֵת, מַאי נִיהוּ, תְּחוּמִין וְאַלִּיבָּא דְרַבִּי עֲקִיבָא, הַבְעָרָה וְאַלִּיבָּא דְרַבִּי יוֹסֵי דְּאָמַר לְלָאו יָצְאָה: פּוֹטְרִין. בַּגְּמָרָא יָלֵיף לֵיהּ מִקְּרָא. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב:

זוֹ דִּבְרֵי רַבִּי מֵאִיר דִּקְטַנָּה אֵין לָהּ קְנָס, דְּנַעֲרָה דַוְקָא וְלֹא קְטַנָּה. אֲבָל חֲכָמִים שֶׁנֶּחְלְקוּ עָלָיו חֲלוּקִין עָלָיו אַף כָּאן. דְּלֹא מִעֵט הַכָּתוּב אֶלָּא בוֹגֶרֶת. גְּמָרָא: בְּרֵישׁ פֶּרֶק ו' דִּיבָמוֹת אִתְרַבֵּי שֶׁלֹּא כְדַרְכָּהּ לְכָל עֲרָיוֹת. וְהָא דְכֻלָּם בִּסְקִילָה לָאו כֻּלֵּי עָלְמָא הִיא, כְּמוֹ שֶׁכָּתַבְנוּ בָּאוֹת שֶׁאַחֲרֵי זֶה: הָרִאשׁוֹן כוּ'. כְּשֶׁבָּא עָלֶיהָ כְּדַרְכָּהּ. הָרַמְבַּ"ם. וְקָשֶׁה לִי מַאי קָא מַשְׁמַע לָן, פְּשִׁיטָא בְּתוּלָה תְנַן. וְרַשִׁ"י פֵּרֵשׁ כְּגוֹן שֶׁבָּאוּ עָלֶיהָ שֶׁלֹּא כְדַרְכָּהּ, הָרִאשׁוֹן בִּסְקִילָה דְּמִשְׁכְּבֵי אִשָּׁה כְתִיב, וְהַשֵּׁנִי בְחֶנֶק דִּבְעוּלָה הִיא. וְהָא סְתָמָא רַבִּי הִיא דְּאָמַר שֶׁלֹּא כְדַרְכָּהּ נָפְקָא לֵיהּ מִכְּלַל בְּתוּלָה כוּ':

דְּמִיתָה כְּתִיבָא בֵיהּ, וּסְתָם מִיתָה אֵינָהּ אֶלָּא חֶנֶק. גְּמָרָא: דְּגָמְרִינַן הַדָּחָה הַדָּחָה מִנָּבִיא, וְלֹא גָמְרִינַן הַדָּחָה הַדָּחָה מִמֵּסִית, דְּדָנִין מֵסִית רַבִּים מִמֵּסִית רַבִּים, וְאֵין דָּנִין מֵסִית רַבִּים מִמֵּסִית יָחִיד. וְכִי תֵימָא דָּנִין הֶדְיוֹט מֵהֶדְיוֹט וְאֵין דָּנִין הֶדְיוֹט מִנָּבִיא, כֵּיוָן שֶׁהִדִּיחַ אֵין לְךָ הֶדְיוֹט גָּדוֹל מִזֶּה. גְּמָרָא: כָּל כוּ'. כְּלוֹמַר שֶׁכָּל חַיָּבֵי מִיתוֹת צְרִיכִין הַתְרָאָה. רַשִׁ"י. וְחוּץ מִזּוֹ, כְּלוֹמַר חוּץ מִמִּיתָה זוֹ. וְהָרַמְבַּ"ם הֶעְתִּיק חוּץ מִזֶּה: דִּמְתַרְגְּמִינַן וְאָרַב לוֹ, וּכְמַן לֵיהּ. רַשִׁ"י: לִשְׁנַיִם. יַחַד, הֵן עַצְמָן נַעֲשׂוּ עֵדִים, וְאֵין צָרִיךְ לְהַתְרוֹת בּוֹ כִּשְׁאָר חַיָּבֵי מִיתוֹת, אֶלָּא מְבִיאִים אוֹתוֹ לְבֵית דִּין, כְּדִכְתִיב וְלֹא תַחְמֹל וְלֹא תְכַסֶּה עָלָיו. רַשִׁ"י: הוּא אוֹמֵר. הַנִּסָּת צָרִיךְ לַהֲשִׁיבוֹ יֵשׁ לִי חֲבֵרִים כוּ'. רַשִׁ"י: עֵדִים. וְלֹא סַגִּי בְּחַד וְאוֹתוֹ שֶׁאוֹמֵר לוֹ, כֵּיוָן שֶׁמְּקִלִּין בְּהַכְמָנָה, מַה שֶּׁאֵין כֵּן בִּשְׁאָר מִיתוֹת בֵּית דִּין. וּמִשּׁוּם הָכִי נַמִּי בְּיָחִיד צָרִיךְ לוֹמַר לוֹ הֵיאַךְ נַנִּיחַ כוּ': דְּהַיְנוּ שֶׁאוֹמֵר אֶעֱבֹד אֲנִי בִתְחִלָּה וְאַתָּה אַחֲרַי. כֶּסֶף מִשְׁנֶה. וּתְמֵהַנִי, דְּבַגְּמָרָא מוּכָח דִּבְנִסָּת הָאוֹמֵר מַיְרֵי, וּכְלוֹמַר שֶׁנִּתְרַצָּה לוֹ, חַיָּב אַף הַנִּסָּת וַאֲפִלּוּ לֹא עָבַד, וְלֹא דָמֵי לְנִסָּת מֵעַצְמוֹ דְּצָרִיךְ שֶׁיַּעֲבֹד. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: אֵלֵךְ כוּ'. שֶׁמְּחֻסַּר הֲלִיכָה. אַף עַל גַּב דְּאִיכָּא לְמֵימַר אַדְּהָכִי הֲדַר בֵּיהּ. רַשִׁ"י: נֵלֵךְ. אַף עַל פִּי שֶׁכָּלַל עַצְמוֹ עִם אֲחֵרִים, דְּאִיכָּא לְמֵימַר הָנֵי אֲחֵרִים לֹא אָזְלֵי וְאִיהוּ נַמִּי לֹא לֵיזוּל. רַשִׁ"י: וְטַעֲמָא דְבִסְקִילָה, דְּגָמְרִינַן הַדָּחָה הַדָּחָה מִמֵּסִית. וְאַל תַּקְשֶׁה דְּלִגְמַר מִנָּבִיא, דְּהַךְ סֵיפָא רַבָּנָן דִּסְבִירָא לְהוּ דְּנָבִיא נַמִּי בִסְקִילָה, וְרֵישָׁא (דְּמַדִּיחַ עִיר וְנָבִיא) בְחֶנֶק, רַבִּי שִׁמְעוֹן הִיא, כִּדְאִיתָא בַגְּמָרָא:

לוֹמַר שֶׁאוֹתוֹ מַעֲשֶׂה שֶׁנִּרְאֶה לִבְנֵי אָדָם שֶׁעוֹשֶׂה, לֹא עֲשָׂאוֹ בֶאֱמֶת אֶלָּא שֶׁנִּדְמֶה לָהֶם כָּךְ, אֲבָל וַדַּאי שֶׁעוֹשֶׂה מַעֲשֶׂה כְּדֵי לֶאֱחֹז הָעֵינַיִם. כֶּסֶף מִשְׁנֶה: רַבִּי עֲקִיבָא אוֹמֵר. לִכְאוֹרָה נִרְאֶה דְּרַבִּי עֲקִיבָא לַאֲפְלוֹגֵי אֲתָא, וְאִיכָּא לְמֵימַר דִּלְתַנָּא קַמָּא אוֹחֵז מְמַעֵט מִסְּקִילָה וְלֹא מִמַּלְקוֹת, דְּחַיָּב מִשּׁוּם (וַיִּקְרָא יט) לֹא תְעוֹנֵנוּ, וּפָלֵיג רַבִּי עֲקִיבָא וְאָמַר פָּטוּר אַף מִמַּלְקוֹת. אֲבָל בַּגְּמָרָא מוּכָח דְּלָאו לַאֲפְלוֹגֵי אֲתָא, וְגַם לְתַנָּא קַמָּא פָּטוּר אֲבָל אָסוּר. וְלֹא קַשְׁיָא לִישְׁנָא דְרַבִּי עֲקִיבָא אוֹמֵר, דְּטוּבָא אִיכָּא כְוָתֵיהּ. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב:
