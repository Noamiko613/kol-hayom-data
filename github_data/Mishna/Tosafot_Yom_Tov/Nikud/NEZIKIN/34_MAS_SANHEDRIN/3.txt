פרק ג

זּה בּוֲרְר לוֲ אּחֹד. וֵהֹא דֵכֹתָב הֹרָ"ב בֵּרְישִ פּּרּק קָמֹּא, דִּשִֵלֲשִֹה הּדֵיוֲטוֲת כּוֲפִין לֹדוּן בִּפֵנְיהּם, מָיֵרְי בִמֵסֹרְב וֵאְינוֲ רוֲצּה לֹדוּן כֵּלֹל. וֵעָיְּן תּוֲסֵפוֲת יוֲם טוֲב: רָשִִ"י. וּפְרְשִ הֹרֲא"שִ דִּרֵצוֲנוֲ לוֲמָר שִּסּוֲבְר כֹּךֵ שִּיָּהַפֲךֵ בִּזֵכוּתוֲ יוֲתְר מִבִּזֵכוּת הֹאָחְר, וּמִתּוֲךֵ כֹּךֵ צֹיֵתְי לֵדִינָיֵהוּ. אַבֹל הָדָּיֹּן עָצֵמוֲ חֹלִילֹה לִמֵצֲא סֵבֹרֹא לֵזָכּוֲתוֲ אִם לֲא שִּיְּרֹאּה לוֲ דִּין גֹּמוּר. אַבֹל אִם הֹיֹה יֹכוֲל לֵהָטֵעוֲת אּת חַבְרוֲ שִּיֵּקָבְּל אּת סֵבֹרֹתוֲ אָף עָל פִּי שִּהוּא מֵסֶפֹּק בֹּהּ, הוּא בִכֵלָל מָטּּה מִשִֵפֹּט. אַבֹל דָּעָת הֹרָ"ב מִתּוֲךֵ שִּזּּה בוֲרֵרוֲ מְבִין דֵּבֹרֹיו לָאַשִוּרוֲ כוּ' נִמֵצֹא שִּלֲּא יִשִֹּאְר שִוּם צָד זֵכוּת נּעֵלֹם מְעְינְי הָדָּיֹּנִים שִֵנְיהּם, וֵהָשִֵּלִישִִי שִוֲמְעָ מָשֵֹּא וּמָתֹּן שִּל שִֵנְיהּם וּמָכֵרִיעָ בְּינְיהּם וֵיוֲצְא הָדִּין לָאַמִתּוֲ. עָד כֹּאן: וּבָגֵּמֹרֹא מָסֵּקִינָן דֵּכֶלְּי עֹלֵמֹא דָּעָת הָדָּיֹּנִים בֹּעִינָן שִּיֵּבֹרֵרוּ מִי שִּיְּשְִב עִמֹּהּן אֹדֹם הֹגוּן, וֵכִי פֵלִיגְי בֵּדָעָת בָּעַלְי דִינִין, רָבִּי מְאִיר סֹבָר דָּעָת בָּעַלְי דִינִין נָמִּי בֹעִינָן כוּ': פּוֲסְל דָּיֹּנוֲ. בָּגֵּמֹרֹא, כֲּל כֵּמִנְּיהּ דֵּפֹסְיל דָּיֹּנְי. אֹמָר רָבִּי יוֲחֹנֹן בֵּעָרֵכֹּאוֲת שִּבֵּסוּרֵיֹא שִֹנוּ, שִּהְם הּדֵיוֲטוֲת שִּאְין יוֲדֵעִין דִּבֵרְי תוֲרֹה אּלֹּא שִּהֶרֵגֵּלוּ לֹדוּן. וֵרָבִּי מְאִיר יֹהְיב לֵהוּ דִין יוֲשִֵבְי קֵרֹנוֲת, וֵיֹכוֲל לֵפֹסֵלֹן. וֵרָבֹּנֹן סֹבֵרְי כְּיוֹן שִּהִמֵחוּם רָבִּים עַלְיהּם לֲא כֲל הְימּנּוּ לֵפֹסֵלֹן, וָעַדִיפְי מִיּוֲשִֵבְי קֵרֹנוֲת. אָחַרוֲנִים: זּה הוּא לֵשִוֲן רָשִִ"י בֵּאוֲקִימֵתֹא קָמָּיֵתֹא. אַבֹל בָּמָּסֵקֹנֹא כֹתָב לֲא אֹדוּן לֵפֹנֹיו: וָאַפִלּוּ נֹשֵֹא וֵנֹתָן בָּיֹּד. וֵעָיְּן תּוֲסֵפוֲת יוֲם טוֲב: וֵקָיֵמֹא לֹן כֵּמָאן דֵּדֹאְין דִּינֹא דֵגָרֵמְי: וֵלֲא יֹדָע שִּכֵּבֹר פֹּשִָט הָמָּעַשֵּה בֵּכֹל הֹעוֲלֹם כֵּדִבֵרְי הֹאּחֹד. וֵעָיְּן תּוֲסֵפוֲת יוֲם טוֲב: וֵחוֲזְר וֵנוֲטְל כוּ' בֵּמִכֹּל שִּכְּן מְהֹא דִּלֵקָמֹּן: נֹפֵקֹא מִנְּיהּ לֵעִנֵיָן זֵמֹן. אִי נָמִּי זִמֵנִין דֵּלְיכֹּא לֵבֹרוֲרְי טֹעוּתֹא לֲא מְהָמִּשִֵנֹה וֵלֲא מְהָסּוּגֵיֹא דֵעֹלֵמֹא, וָאַפִלּוּ הֹכִי אּפֵשִֹר שִּאְינוֲ דִין צּדּק וִיכוֲלִים הָדָּיֹּנִים לִפֵסֲק דֵּלֲא כֵמוֲ שִּפֹּסָק זּה הָדָּיֹּן, דֵּאָטּוּ כֶלֵּהוּ דִינְי נִשִֵנוּ בָמִּשִֵנֹה וֵסוּגֵיֹא דֵעֹלֵמֹא: שִּזּּה מִתֵכָּוְּן לֵהָזִּיק הוּא כְּיוֹן דֵּלֹאו דָּיֹּנֹא הוּא: דֵּהֹכֹא בֵּבֹבּל שְִבּט, לֵשִוֲן שֵֵרֹרֹה, וֵיְשִ לֹהּם רֵשִוּת . וֵהֹתֹם בֵּאּרּץ יִשֵֵרֹאְל מֵחוֲקְק, שֵֵרֹרֹה מֶעּטּת. כִּדֵאִיתֹא בָגֵּמֹרֹא. וֵעָיְּן תּוֲסֵפוֲת יוֲם טוֲב: כֵּלוֲמָר וֵדִינֹיו דִּין. אַבֹל אִם טֹעֹה, אְינוֲ פֹטוּר, הוֲאִיל וֵלֲא נֹקָט רֵשִוּתֹא: רָשִִ"י. וּקֵצֹת קֹשִּה לוֲמָר כְּן דֵּמִפֵּנְי שִּסֹּבוּר הֹיֹה שִּיְּשִ לוֲ שִֵתְּי כִתְּי עְדִים וֵאֹמָר כְּן וֵאָחָר כֹּךֵ בִּקְּשִ וֵלֲא מֹצֹא יָפֵסִיד. וֵהֹרָמֵבָּ"ם מֵפֹרְשִ דִּפֵלוּגֵתָיֵהוּ כֵּשִּהְבִיא שִֵתְּי כִתְּי עְדִים שִּאָחָר שִּהְעִידוּ כָת אָחָת כִּדֵבֹרֹיו הְבִיא עוֲד אָחּרּת וֵהְעִידוּ גָּם כְּן כֵּמוֲ הֹרִאשִוֲנִים, וֵטֹעָן הָלֹּה שִּאָרֵבֹּעֹה הֹעְדִים פֵּסוּלִים הְם, רָבִּי מְאִיר אוֲמְר שִוֲמֵעִים לוֲ, וֵנֲאמָר לֵזּה שִּהֱבִיאֹם הֹבְא רֵאֹיֹה שִּהְם כֵּשְִרִים, לֵפִי שִּאָתֹּה לֲא הֹיִיתֹ מְבִיא עְדִים אָחָר עְדִים אּלֹּא שִּאָתֹּה יוֲדְעָ שִּיְּשִ בֹּהּן מִי שִּאְינוֲ כֹשְִר, וָאַנָחֵנוּ לֲא נְדָע אְיזּהוּ הֹרִאשִוֲנִים אוֲ הֹאָחַרוֲנִים. וָחַכֹמִים אוֲמֵרִים אְין חוֲשִֵשִִים לֹזּה, לֵפִי שִּלֲּא עֹשֵֹה זּה אּלֹּא לֵחָזְּק טָעַנוֲתֹיו וּלֵהוֲדִיעָ כִּי זּה יֹדוּעָ לֹרָבִּים:

וֵכֹל שִּכְּן לֵהֹעִיד. עָיְּן פּּרּק ו' דֵּנִדֹּה מִשִֵנֹה ד': רוֲעְי בֹקֹר. הוּא הָדִּין כֹּל בְּי תֵלֹתֹא דֵּלֲא גֵמִירְי דִינֹא פֵּסוּלִים. וֵלִרֵבוּתֹא נֹקָט רוֲעְי בֹקֹר, דֵּלֲא שִֵכִיחְי בֵיִשִּוּב וֵלֲא רֹאוּ וֵלֲא שִֹמֵעוּ עִסֵקְי דִין בְּין אֹדֹם לָחַבְרוֲ, וָאַפִלּוּ לֵעְדוּת נָמִּי פָסֵלִינֵהוּ רָבֹּנֹן, דֵּקֹא חֹזוּ רָבֹּנֹן דֵּקֹא מֵכָוֵּנְי וֵשִֹרְי לֵהוּ לִרֵעוֲת בִּשֵֵדוֲת אַחְרִים. גֵּמֹרֹא: וֵכְן דָּעָת נִמּוּקְי יוֲסְף דִּלֵגָבְּי עְדוּת נָמִּי הֹוְי גֵמָר דִּין כֵּשִּפֹּסֵקוּ בְית דִּין הָדֹּבֹר עָל פִּיהּן. אַבֹל הֹרִיבֹ"שִ וּגֵאוֲנִים סֵבִירֹא לֵהוּ דִּלֵגָבְּי עְדוּת הֹוְי גֵמָר דִּין כֹּל שִּנִּתֵקָבְּל עְדוּתֹן בֵּבְית דִּין. וֵעָיְּן תּוֲסֵפוֲת יוֲם טוֲב: וּבֵרְישִֹא נָמִּי בֵּאּתְּן לֵךֹ מָחַלֲקּת, אַבֹל בֵּמֹחוּל לֵךֹ אַפִלּוּ רָבִּי מְאִיר מוֲדּה. עָיְּן תּוֲסֵפוֲת יוֲם טוֲב: לֹאו דָּוֵקֹא. דֵּאִם כְּן פֵּשִִיטֹא, דֵּהֹא בִלֵאָחָר גֵּמָר דִּין מָיֵרְי וּמֵחִילֹה גֵמוּרֹה הִיא. אּלֹּא מָיֵרְי כֵּגוֲן דֵּאֹמָר לְיהּ תּוֲבְעָ לָנִּתֵבֹּע הַרְי עֹלָי לָעַשֵוֲת מִתֵּבִיעֹה זוֲ כֵּכֹל מָה שִּּיֲּאמֵרוּ אָבֹּא אוֲ אֹבִיךֵ הְן לִפֵטוֲר הְן לֵחִיּוּב. שִּאְין בָּלֹּשִוֲן הָזּּה לֵשִוֲן מֵחִילֹה. נִמּוּקְי יוֲסְף:

הָפֵּסוּלִין. פֵּסוּלְי דֵרָבֹּנֹן קֹא חֹשְִיב. וּמִשִּוּם הֹכִי לֲא גֹרֵסִינָן וָעַבֹדִים, דֵּאִינֵהוּ פֵסוּלִים מִדֵּאוֲרָיֵתֹא מִקָּל וֹחֲמּר. תּוֲסֹפוֲת: וֵהֹא דִתֵנָן מָלֵוּה בֵרִבִּית. מִלֵוֹה הָבֹּאֹה בֵרִבִּית. גֵּמֹרֹא: וֵאָף עָל גָּב דֵּעֹרְב וֵעְדִים עוֲבֵרִים נָמִּי בֵלֲא תֵשִֵימוּן, לֲא תֵשִֵימוּן לּאֱנֹשְִי, לֵמָלֵוּה וֵלֲוּה מָשִֵמָע לֵהוּ. הֹרֲא"שִ. וֵעָיְּן תּוֲסֵפוֲת יוֲם טוֲב: וֵאָף עָל גָּב דֵּהָיֵנוּ מֵשֵָחְק בֵּקֶבֵיֹא, תֹּנֹא תוֲלּה בֵדָעָת יוֲנוֲ וֵתֹנֹא תוֲלּה בֵדָעָת עָצֵמוֲ. וּצֵרִיכְי כוּ'. בָּגֵּמֹרֹא: דֵּלֲא זֹכֹה בָעָל הָשִּוֲבֹךֵ, דֵּמִמְּילֹא קֹאֹתוּ וֵרֹבוּ לֵהֹתֹם. רָשִִ"י: וֵכֹל שִּכְּן לֹדוּן: בֵּרִבִּית קֵצוּצֹה. וּמָתֵנִיתִין בָּאַבָק רִבִּית. וֵעָיְּן תּוֲסֵפוֲת יוֲם טוֲב: דֵּלֲא תָחֵמֲד לּאֱנֹשְִי בֵּלֲא דֵמְי מָשִֵמָע לֵהוּ: וּמָיֵרְי שִּהּחֱזִירוּ, דֵּאִם לֲא כְן פֵּסוּלִים מִדֵּאוֲרָיֵתֹא, אּלֹּא מִשִּוּם דֵּלֲא יֹדֵעְי לֵמָאן נֵהָדֵרְי, אִי נָמִּי שִּמֹּא יָחֵזֵרוּ לֵקִלֵקוּלֹם. תּוֲסֹפוֲת: וֵאּפֵשִֹר לְיהּ לֵאִתֵזוֲנְי בֵצִנֵעֹא וּמֵבָזְּי נָפֵשְִהּ בֵּפָרֵהּסֵיֹא. דֵּאִי לֲא אּפֵשִֹר לְיהּ, חָיּוּתְיהּ הוּא. גֵּמֹרֹא: אֹמָר רָבִּי יֵהוּדֹה. אָמֵּשֵָחְק בֵּקֶבֵיֹא קֹאְי. אִי נָמִּי אָמָּפֵרִיחְי יוֲנִים לֵפְרוּשִ הֹרִאשִוֲן: כֵּשְִרִין. וֵדָוֵקֹא כֵשִּהָמֹּעוֲת עָל הָדָּף. אַבֹל הָמֵשֵָחַקִים בָּאַמֹנֹה, אַפִלּוּ הִקֵנוּ, אָסֵמָכֵתֹּא גֵמוּרֹה הִיא וֵאְין קִנֵיֹן מוֲעִיל בֹּהּ. תּוֲסֹפוֲת: שִִבֵרְי עְצִים. . וֵנִמּוּקְי יוֲסְף כֹּתָב לֵפִי שִּהָקֶּבֵיֹא מֵרֶבָּעָת כֵּאּבּן פָּסֵפֹּס קֹרוּי כְּן:

הָקֵּרוֲבִים. בָּגֵּמֹרֹא יֹלְיף לֵהוּ מִקֵּרֹא דִּפֵסוּלִין בְּין לִזֵכוּת בְּין לֵחוֲבֹה, בְּין בֵּדִינְי נֵפֹשִוֲת בְּין בֵּדִינְי מֹמוֲנוֲת: אֹבִיו. סֵפֹרִים אַחְרִים לֲא גֹרֵסִינָן. וֵהָטָּעָם שִּלֲּא מֹנֹה אוֲתוֲ בָמִּשִֵנֹה, לֵפִי שִּהוּא מֵבֲאֹר בָּקֵּרֹא, וֵלֲא לֹמָדֵנוּ אִסּוּר עְדוּת הָקֵּרוֲבִים אּלֹּא מִמָּה שִּּהִזֵהִיר הָשְִּם יִתֵבֹּרָךֵ מִלֵּהֹעִיד הֹאֹב עָל הָבְּן אוֲ הָבְּן עָל הֹאֹב, כֵּמוֲ שִּנּּאֱמָר לֲא יוּמֵתוּ כוּ'. הֹרָמֵבָּ"ם: בָּגֵּמֹרֹא יֹלְיף לְיהּ מִקֵּרֹא: וֵהוּא הָדִּין לֵכֶלֵּהוּ בָעָל כֵּאִשִֵתּוֲ כוּ'. אּלֹּא נֹקָט גִּיסוֲ דִּסֵמִיךֵ: דּוֲדוֲ וֵכוּ'. לָאַפוֲקְי אַחְי הֹאְם וּקֵרוֲבְי הֹאִישִוּת. עָיְּן בֵּנִמּוּקְי יוֲסְף: הֹרָמֵבָּ"ם. וֵכֹתָב רָשִָ"ל שִּכְּן נִרֵאּה לוֲ דֵּלֲא כֵפְרוּשִ רָשִִ"י כוּ'. וּפֹשִוּט הוּא, דֵּהֹא בִכֵלָל דּוֲדוֲ וּבּן דּוֲדוֲ אָף מִן הֹאְם וֵאְינֹם רֵאוּיִים לֵיֹרֵשִוֲ. וֵכִי תְימֹא דֵאָדֵּרָבֹּא וֵכֹל הֹרֹאוּי לֵיֹרֵשִוֲ פְּרוּשִ דּוֲדוֲ וּבּן דּוֲדוֲ דֵּדָוֵקֹא מִן הֹאֹב, אִם כְּן לֲא לִתֵנְי דוֲדוֲ וּבּן דּוֲדוֲ כֵּלֹל. וּמִיהוּ בָּעָל אַחוֲתוֲ רֹאוּי לִירָשִ הֹאֹחוֲת וֵהִיא רֵאוּיֹה לֵיֹרֵשִוֲ וּבָעָל כֵּאִשִֵתּוֲ. וֵכְן בָּעָל אַחוֲת אֹבִיו, וֵגִיסוֲ, נָמִּי הֹאַחֹיוֲת רֵאוּיוֲת לִירָשִ זוֲ אּת זוֲ, וּבָעָל כֵּאִשִֵתּוֲ, רִאשִוֲן בֵּרִאשִוֲן אַפִלּוּ בִתֵרְי בָעָל לֵכֶלְּי עֹלֵמֹא: כֵּדִתֵנָן בֵּרְישִ פּּרּק ח' דֵּבֹבֹא בֹתֵרֹא, אַחְי הֹאְם מָנֵחִילִין וֵלֲא נוֲחַלִין:

הֹאוֲהְב. סִיּוּמֹא דֵדִבֵרְי רָבִּי יֵהוּדֹה הוּא כוּ'. וֵעָיְּן תּוֲסֵפוֲת יוֲם טוֲב: וּבָגֵּמֹרֹא יֹלֵפִינָן נָמִּי שִֵנְי תָלֵמִידְי חַכֹמִים שִּשֵּוֲנֵאִין זּה אּת זּה, שִּאְין יוֲשִֵבִין בֵּדִין אּחֹד. וֵעָיְּן תּוֲסֵפוֲת יוֲם טוֲב:

וּמֵאָיֵּמִין כוּ'. בִּפֵנְי הָכֲּל. וֵעָיְּן מָה שִּּכֹּתָבֵתִּי פּּרּק ד' מִשִֵנֹה ה': וֵכְן לֵשִוֲן רָשִִ"י וֵנִמּוּקְי יוֲסְף. וֵאָגָּב רִהַטֹא כֹּתֵבוּ כְן. שִּהַוֹה לֵהוּ לִכֵתֲּב אִיזּבּל שִּיֹּעַצֹה: כֹּל כוּ'. כֵּדְי לָחֵקֲר הֹעְדִים לֵבָדֹּן, שִּחַקִירֹה כֹזוֲ עַדִיפֹא. לֵבוּשִ. וּמִסֵתָּבֵּרֹא דֵּהָיֵנוּ חוּץ מִבָּעַלְי הָדִּינִין, דֵּאָדֵּרָבֹּה עַדִיפֹא שִּיִּהֵיוּ שִֹם בִּשִֵעָת הָחַקִירֹה. וּלֵנֶסָּח אָחְר , סֵבֹרֹא, דָּחַקִירֹה נָמִּי עַדִיפֹא כֵּשִּהִיא בִּפֵנְי רָבִּים, אּלֹּא שִּהֹעְדִים לֵבָד מוֲצִיאִין וֵאְין מֵשִָיֵּרִין אּלֹּא אּחֹד, מִשִּוּם דִּמֵרָמֵזְי רָמּוֲזְי: אִישִ. לֵשִוֲן חַשִִיבוּת, כִּדֵאָשִֵכֵּחָן בֵּאֹבוֲת, אִישִ סוֲכוֲ כוּ'. נִרֵאּה לִי: מֹאתָיִם. נֹקָט שִִעוּר הָמָּסֵפִּיק לֹאֹדֹם. שִּכְּן שִֹנִינוּ בֵּפְאֹה מִי שִּיְּשִ לוֲ מֹאתָיִם זוּז לֲא יִטֲּל כוּ': יוֲסִיפוּ כוּ'. שִֵנָיִם. הֹרָמֵבָּ"ם פּּרּק ח' מְהִלֵכוֲת סָנֵהּדֵרִין. רֵצוֲנוֲ לוֲמָר דֵּהֹכִי פְרוּשִֹא דֵמָתֵנִיתִין, הָדָּיֹּנִים יוֲסִיפוּ, וֵכָמֹּה, שִֵנָיִם. וּמֵסָיְּם הֹרָמֵבָּ"ם, וֵאִם אְרָע עוֲד כוּ' יוֲסִיפוּ כוּ' עָד שִִבֵעִים וֵאּחֹד. וֵעָיְּן תּוֲסֵפוֲת יוֲם טוֲב:

אָחָר קָבֹּלָת עְדוּת. דֵּבִשִֵעָת קָבֹּלָת עְדוּת וָדָּאי בָּעַלְי דִין הֹתֹם קָיֵמְי. נִמּוּקְי יוֲסְף. וֵהֹרָ"ב נֹקָט מִלֵּתֹא דִפֵסִיקֹא, דֵּלִפֵעֹמִים אְין כֹּאן עְדִים כֵּלֹל: כֵּדְי שִּיִּהֵיוּ הָדָּיֹּנִים אַהוּבִים אְצּל בֵּנְי אֹדֹם. הֹרָמֵבָּ"ם: הָגֹּדוֲל כוּ'. לָחַלֲק לוֲ כֹבוֲד. שִּכְּן מִדָּת חַכֹמִים שִּאְינֹם מֵדָבֵּרִים בִּפֵנְי מִי שִּהוּא גֹדוֲל מִמּּנּוּ, כִּדֵאִיתֹא בֵאֹבוֲת . וּמָה שִּּכֹּתָב הָסֵּמָ"ע סִימֹן י"ט, לֲא נִרֵאּה לִי: וֵאוֲמְר כוּ'. יְשִ לוֲמָר דֵּאִי מִלֲּא תְלְךֵ רֹכִיל הַוֹה אַמְינֹא דֵּדָוֵקֹא כֵשִּהוֲלְךֵ לֵרָגְּל וּלֵגֹרוֲת מֹדוֲן הוּא דֵאֹסוּר, אַבֹל לֵזָכּוֲת אּת עָצֵמוֲ שִּלֲּא יִשֵֵנֹאְהוּ הָלֹּה לֲא אֹסָר קֵרֹא, תֹּא שִֵמָע הוֲלְךֵ כוּ' שִֵמָע מִנָּהּ הָמֵגָלּּה סוֲד מִקֵּרְי הוֲלְךֵ רֹכִיל אָף עָל פִּי שִּאְינוֲ עוֲשֵּה לֵגֹרוֲת מֹדוֲן. וֵיְשִ סֵפֹרִים דֵּלֲא גֹרֵסְי אּלֹּא הוֲלְךֵ רֹכִיל. וֵכְן מָשִֵמָע בָּגֵּמֹרֹא. וֵעָיְּן תּוֲסֵפוֲת יוֲם טוֲב:
