פרק ה - משנה א

הָעֲבָדִים. שֶׁיֵּשׁ לוֹ לְבַעַל הַסְּפִינָה לְהַנְהִיגָהּ וְלִשְׁמֹר פְּרַקְמַטְיָא שֶׁלּוֹ בְּתוֹכוֹ. רַ"שׁ. וּפְשִׁיטָא הִיא. וְכֵן הָאַנְתִּיקִי כָּל שֶׁכֵּן הִיא מִן הַמַּרְצוּפִין, אֶלָּא מִשּׁוּם סֵיפָא בִּזְמַן שֶׁאָמַר לוֹ כוּ' נַקְטִינְהוּ. תּוֹסָפוֹת. וְנִמּוּקֵי יוֹסֵף: מִלָּשׁוֹן זֶה יֵרָאֶה דַּאֲפִלּוּ לְעֵיל נַמִּי לֹא הֲווּ בִּכְלַל מַתָּנָה וְהֶקְדֵּשׁ אֶלָּא הֶחָשׁוּב כְּקַרְקַע, לַאֲפוֹקֵי פֵּרוֹת תְּלוּשִׁים וְכָל שֶׁכֵּן מִטַּלְטְלִין דְּלֹא, אַף עַל פִּי שֶׁכְּשֶׁאָמַר הִיא וְכָל מַה שֶּׁבְּתוֹכָהּ הַכֹּל מָכוּר. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: מָכַר אֶת הַקָּרוֹן לֹא כוּ'. וַאֲפִלּוּ אִיכָּא דָּמִים יְתֵרִים מוֹדֶה רַבִּי יְהוּדָה, מִשּׁוּם דִּלְעוֹלָם אֵין קָרוֹן וּפְרָדוֹת נִקְרָאִים בְּשֵׁם אֶחָד. נִמּוּקֵי יוֹסֵף: אֲבָל מָכַר הַפְּרָדוֹת אֲפִלּוּ אֲדוּקִין לֹא מָכַר הַקָּרוֹן. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: הַצֶּמֶד. מְפָרֵשׁ בַּגְּמָרָא דְּאַיְרֵי בְּאַתְרָא דְּקָרֵי לְצַמְדָא צַמְדָא וּלְבָקָר בָּקָר, וְאִיכָּא נַמִּי דְּקָרֵי לַבָּקָר צַמְדָא. דְּאִי בְּאַתְרָא דְּלֵיכָּא דְּקָרֵי לַבָּקָר צַמְדָא, פְּשִׁיטָא צַמְדָא זַבֵּין לֵיהּ בָּקָר לֹא זַבֵּין לֵיהּ. וְאִי דְּקָרֵי כֻּלְּהוּ נַמִּי לַבָּקָר צַמְדָא, כֻּלְּהוּ זַבֵּין לֵיהּ. וּפֵרֵשׁ הָרַ"שׁ דְּבִכְהַאי גַּוְנָא וַדַּאי הַדָּמִים מוֹדִיעִים כוּ'. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: לֹא כוּ'. וַאֲפִלּוּ כְּשֶׁאֲדוּקִים בּוֹ נַמִּי לֹא קָנָה. רַ"שׁ. וְהָרִ"י הַלֵּוִי פֵּרֵשׁ דְּהָכָא נַמִּי דַּוְקָא כְּשֶׁאֵין אֲדוּקִים, אֲבָל בַּאֲדוּקִים, מְכוּרִים. וְהָכִי מִסְתַּבְּרָא. טוּר: הַצֶּמֶד. דִּבְרֵי הַכֹּל הוּא. אֶלָּא דִּפְלִיגֵי רַבִּי יְהוּדָה וְרַבָּנָן בְּפֵרוּשָׁא דְּהַךְ מִלְּתָא. דִּלְרַבִּי יְהוּדָה זִמְנִין מָכַר זִמְנִין לֹא מָכַר, וּלְרַבָּנָן לְעוֹלָם לֹא מָכַר. רַ"שׁ. וּכְלוֹמַר דְּלֹא תַקְשֶׁה חֲכָמִים הַיְנוּ תַּנָּא קַמָּא. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: שֶׁהַדָּבָר יָדוּעַ שֶׁאֵינוֹ נִמְכַּר בְּמָאתַיִם זוּז. מַה שֶּׁאֵין כֵּן סֵפֶר תּוֹרָה בְּהֵמָה וּמַרְגָּלִיּוֹת דְּסָבְרֵי חֲכָמִים בְּמִשְׁנָה ט' פֶּרֶק ד' דְּבָבָא מְצִיעָא דְּיֵשׁ לָהֶן אוֹנָאָה, וּמִמֵּילָא דְּהוּא הַדִּין דַּהֲוֵי נַמִּי בִּטּוּל מִקָּח, מִשּׁוּם דְּהַוְיָן דְּבָרִים שֶׁהַדַּעַת טוֹעָה בָּהֶן. נִמּוּקֵי יוֹסֵף: דְּתַכְשִׁיטֵי חֲמוֹר הֲווּ וְקָנִי לְהוּ הֵיכָא דְּאִיתְנְהוּ. תּוֹסָפוֹת:

דְּקָסָבַר חֲמוֹר לִרְכִיבָה קָאֵי וְלֹא לְמַשְּׂאוֹי, וְאִבָּעֵי לְהוּ אִי בְּעוֹדָן עָלָיו מַחֲלֹקֶת כוּ'. וְלֹא אִפְשִׁיטָא. הָרִי"ף וְהָרֹא"שׁ. וְהַיְנוּ שֶׁהָרַ"ב לֹא פֵּרֵשׁ בַּתְּחִלָּה הַאֵיךְ מַיְרֵי, וְלַסּוֹף מַסִּיק וַאֲפִלּוּ הָיוּ עָלָיו. אֲבָל הוּא מִטַּעַם הַמּוֹצִיא מֵחֲבֵרוֹ עָלָיו הָרְאָיָה. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: נַחוּם כוּ'. סָבַר סְתָם חֲמוֹר לְמַשְּׂאוֹי קָאֵי. פֵּרֵשׁ הָרַ"שׁ, וְקָנָה כֻּלָּן גַּם הָאֻכָּף וְהַמַּרְדַּעַת, דְּלִטְעֹן מַשְּׂאוֹי עַל גַּבָּהּ עֲשׂוּיִין אֻכָּף וּמַרְדַּעַת בֵּין לִרְכִיבָה בֵּין לְמַשְּׂאוֹי: לְשׁוֹן הָרַ"שׁ, דִּסְבִירָא לֵיהּ לְרַבִּי יְהוּדָה דְּהַמּוֹכֵר חֲמוֹר סְתָם כוּ':

וְאַף עַל גַּב דְּבִנְקֵבָה מַיְרֵי, נָקַט בְּנוֹ, עַל שֵׁם שֶׁהַחֲמוֹר בִּלְשׁוֹן זָכָר הוּא. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: בּוֹר. שֶׁל מַיִם הַמְּכֻנָּסִים, רַ"שׁ: וַאֲפִלּוּ לְמַאן דְּאָמַר אֵין אָדָם מַקְנֶה דָּבָר שֶׁלֹּא בָּא לָעוֹלָם, מִכָּל מָקוֹם הֵיכָא שֶׁאֵינָן חוֹזְרִין וְרוֹצִין לְקַיֵּם הַמִּקָּח, צָרִיךְ הַתַּנָּא לְאַשְׁמְעִינַן הַדִּין. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: וְזֶה לְשׁוֹן הָרַ"שׁ, דֶּרֶךְ הַיּוֹנִים לְהוֹלִיד שְׁנֵי וְלָדוֹת בְּכָל חֹדֶשׁ זָכָר וּנְקֵבָה, חוּץ מֵחֹדֶשׁ אֲדָר שֶׁאֵין יוֹלְדוֹת. וְהַנְּהוּ וְלָדוֹת גַּם הֵם לְסוֹף שְׁנֵי חֳדָשִׁים עוֹשִׂין וְלָדוֹת בְּכָל חֹדֶשׁ. וְכָל שְׁנֵי כוּ' שֶׁלֹּא יִבְרְחוּ הָאִמָּהוֹת וְהוּא שׁוֹבָכוֹ לֹא מָכַר, הִלְכָּךְ מִסְּתָמָא גַּם בְּרֵכָה רִאשׁוֹנָה שֶׁל זוֹ עִכֵּב עִם הַשּׁוֹבָךְ, דְּזוֹהִי קִיּוּם הַשּׁוֹבָךְ: מִפְּנֵי שֶׁמִּשְּׁלֹשָׁה הָרִאשׁוֹנִים וָאֵילַךְ הֵם גְּרוּעִים. וּמִשּׁוּם הָכִי יִקָּח מֵהֶם הַמּוֹכֵר בְּסֵרוּס לְצֹרֶךְ כַּוַּרְתּוֹ, כְּדֵי שֶׁלֹּא יִשָּׁאֵר רֵיקָן וְיַפְסִיד, וְלָכֵן יִשָּׁאֲרוּ לוֹ גַּם מִן הַטּוֹבִים קְצָת כוּ'. נִמּוּקֵי יוֹסֵף: לְפִי שֶׁהוּא לֹא מָכַר לוֹ כַּוֶּרֶת כְּדִלְקַמָּן: אִם הַמְּכִירָה הִיא לְעוֹלָם. אִי נַמִּי לְעוֹלָמוֹ שֶׁל מְכִירָה קָאָמַר, דְּהַיְנוּ שָׁנָה: שְׁלֹשָׁה נְחִילִין. כִּי מִתִּשְׁעָה יָמִים לְתִשְׁעָה יָמִים יוֹצֵא מִן כַּוֶּרֶת נְחִיל שֶׁל דְּבוֹרִים וְיוֹשְׁבִין עַל עֲנַף אִילָן, וּמֵבִיא כַּוֶּרֶת חֲדָשָׁה וּמַכְנִיסָן לְתוֹכָהּ. הָרַ"שׁ: חַלּוֹת כוּ'. אִם לָקַח חַלּוֹת דְּבַשׁ. שֶׁהַלּוֹקֵחַ פֵּרוֹת כַּוֶּרֶת אֵין הַחַלּוֹת בִּכְלָל אֶלָּא הַדְּבוֹרִים הָעוֹשִׂים פֵּרוֹת. וְהַלּוֹקֵחַ הַחַלּוֹת מַנִּיחַ שְׁתֵּי חַלּוֹת: כְּדֵי שֶׁלֹּא תֶּחֱרַב כַּוַּרְתּוֹ. שֶׁהוּא לֹא מָכַר לוֹ הַכַּוֶּרֶת. רַ"שׁ: רַ"שׁ. וְיֵשׁ מְפָרְשִׁים שְׁתֵּי אֶגְרוֹפִין. וְהוּא יוֹתֵר נָכוֹן. נִמּוּקֵי יוֹסֵף. וְכֵן הוּא בְּפֵרוּשׁ הָרַמְבַּ"ם:

בְּפֶרֶק קַמָּא דְּבִכּוּרִים מִשְׁנָה ו' כָּתַב הָרַ"ב דִּמְסָפְקָא לֵיהּ אִי קָנָה קַרְקַע. אֶלָּא דְּהָכָא לֹא נָפְקָא מִנַּהּ בָּזֶה, לְהָכִי סָתַם כָּאן: וּמִלָּה זוֹ דּוֹמָה לַאֲרַמִּי. תַּרְגּוּם וָאֶכֹּת אוֹתוֹ טָחוֹן , וְשָׁפִית יָתֵיהּ: וּכְגוֹן שֶׁיֵּשׁ בֵּין הַגֶּזַע לָאִילָן כְּשִׁעוּר הַמְּפֹרָשׁ לְפָנֵינוּ. הָרַ"שׁ. וְכָתַב הַנִּמּוּקֵי יוֹסֵף, וְאַף עַל פִּי שֶׁעַל הַלּוֹקֵחַ לְהָבִיא רְאָיָה כִּי הוּא הַמּוֹצִיא, מִיהַת מַיְתֵי לֵיהּ לִידֵי דִּינָא וְדַיְנָא: פֵּרוּשׁ אוֹרֶה, מְלַקֵּט. כְּמוֹ וְאָרוּהָ כָּל עוֹבְרֵי דֶּרֶךְ: שֶׁתְּהֵא רְאוּיָה הַמַּחְרֵשָׁה לַחְרֹשׁ בֵּינְתַיִם. הָרַ"שׁ: דַּחֲשִׁיבֵי מְפֻזָּרִים כִּדְאַשְׁכְּחָן בְּכִלְאַיִם סוֹף פֶּרֶק ד' כוּ'. וְאַף עַל גַּב דְּהָתָם פָּסְקוּ הָרַמְבַּ"ם וְהָרַ"ב כְּרַבִּי מֵאִיר וְרַבִּי שִׁמְעוֹן דְּסַגִּי בִּשְׁמֹנֶה עַל שְׁמֹנֶה, כָּתַב הַנִּמּוּקֵי יוֹסֵף דְּצָרִיךְ לוֹמַר דְשִׁעוּרָא דְּכִלְאַיִם וְדִקְנִיָּה אֵינָם שָׁוִים. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: יְשַׁפֶּה. דְּכֵיוָן דְּאֵין נְטוּעִין בְּשֶׁל מוֹכֵר כִּי אִם בְּקַרְקַע שֶׁל לוֹקֵחַ, וְהַמּוֹכֵר לֹא שִׁעְבֵּד לוֹ קַרְקַע שֶׁלּוֹ, אֵינוֹ מַנִּיחָן לִכָּנֵס בְּשֶׁלּוֹ, מֵאַחַר שֶׁמַּזִּיקִין לוֹ. רַ"שׁ:

גַּסָּה. בְּגַסָּה אֵין דֶּרֶךְ לִמְכֹּר אֶלָּא כָּל אֶחָד מֵאֵלּוּ בִּפְנֵי עַצְמוֹ. אֲבָל בְּדַקָּה דֶּרֶךְ לִמְכֹּר הָרַגְלַיִם עִם הָרֹאשׁ, דְּלֹא חֲשִׁיבֵי כֻּלֵּי הַאי. וְכֵן הַכָּבֵד עִם הָרֵאָה, אֲבָל אִפְּכָא לֹא, שֶׁהֶחָשׁוּב אֵינוֹ דִּין שֶׁיְּהֵא טָפֵל לְשֶׁאֵינוֹ חָשׁוּב. נִמּוּקֵי יוֹסֵף: הוּא הַדִּין בְּכָל הַשְּׁנוּיִים בְּפִרְקִין הַלָּלוּ, שֶׁאִם יֵשׁ מִנְהָג יָדוּעַ יָדוּעַ שֶׁהוֹלְכִין אַחֲרָיו. הָרַמְבַּ"ם. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: אֲבָל הַתּוֹסָפוֹת כָּתְבוּ, קָנֶה הִיא רֵאָה עִם הַלֵּב:

פרק ה - משנה ו

אַרְבַּע. מִנְיָנָא לֹא לְמִעוּטֵי מִידֵי אָתָא. אֶלָּא לְהוֹדִיעֲךָ שֶׁצָּרִיךְ לָתֵת טַעַם לְכָל הָאַרְבַּע לָמָּה הֵם חֲלוּקִים זֶה מִזֶּה. רַ"שׁ: כְּלוֹמַר אַף עַל גַּב דְּלֹא נִתְאַנָּה בְּדָמִים כְּלָל, אֲפִלּוּ הָכִי הֲוֵי כְּאוֹנָאָה, לְפִי שֶׁיֹּאמַר דְּלֹא נִיחָא לֵיהּ בְּרָעוֹת אֲפִלּוּ בְּזוֹל. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: יָפוֹת. שֶׁהִטְעָהוּ הַלּוֹקֵחַ וְאָמַר לוֹ חִטִּין הַלָּלוּ נֶחְשָׁבִים רָעוֹת בָּעִיר הַזֹּאת וּבַעֲבוּר זֶה מְכָרָן, וְאַחַר כָּךְ מָצָא שֶׁהָיוּ מְחַשְּׁבִין אוֹתָן לְיָפוֹת. נִמּוּקֵי יוֹסֵף: שְׁחַמְתִּית. עוֹשָׂה קֶמַח מְרֻבֶּה. וּלְבָנָה עוֹשָׂה הַפַּת יוֹתֵר נָאָה וְיָפָה. נִמּוּקֵי יוֹסֵף: לְפִי שֶׁהוּא צָרִיךְ עַכְשָׁיו לְחֹמֶץ:

פרק ה - משנה ז

כֵּן הִיא בָּרַ"שׁ, בְּבֵי"ת. אֲבָל לֹא נָקַט מֵרְשׁוּת הָרַבִּים. וּבַרַמְבַּ"ם כָּתַב מֵרְשׁוּת הָרַבִּים לְסִמְטָא, בְּלָמֶ"ד כוּ': לְפִי שֶׁאֵין לוֹ רְשׁוּת לְהַנִּיחָן שָׁם: וּמִלְּתָא בְּאַפֵּי נַפְשָׁהּ הִיא. הָרַ"שׁ. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: וְאַף עַל גַּב דְּמַתְנִיתִין בְּמַשָּׂאוֹת גְּדוֹלוֹת, וּבְפִשְׁתָּן נַמִּי מִי לֹא עֲבִידֵי שְׁלִיפֵי רַבְרְבֵי, שָׁאנֵי פִּשְׁתָּן דְּמִשְׁתַּמֵּט, כְּלוֹמַר שֶׁאֶפְשָׁר לְשׁוֹמְטוֹ מְעַט מְעַט וּלְהַגְבִּיהוֹ, וְאֵין בּוֹ טֹרַח. וְלֹא אָמְרוּ מַשָּׂאוֹת גְּדוֹלוֹת שֶׁאֵין צְרִיכִים הַגְבָּהָה אֶלָּא כְּגוֹן טָעוּן שֶׁל אֱגוֹזִים אוֹ פִּלְפְּלִים אוֹ שְׁקֵדִים, וְהָיָה גָּדוֹל שֶׁאֵין אֶחָד יָכוֹל לְהַגְבִּיהַּ, וְאִם יַתִּירוֹ יִתְפָּרֵד וְיִהְיֶה לוֹ בּוֹ טֹרַח גָּדוֹל. הָרַמְבַּ"ם. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: לְשׁוֹן הָרַ"שׁ, דְּאֵין דֶּרֶךְ לְהַגְבִּיהַּ מִקְּחוֹ אֶלָּא כְּדֵי לִטְּלוֹ וְלֵילֵךְ: הָרַמְבַּ"ם. וּמְפָרֵשׁ לֵיהּ בְּפִשְׁתָּן שֶׁנִּתְיַבֵּשׁ וְעוֹמֵד לִתָּלֵשׁ, דִּכְתָלוּשׁ דָּמֵי, אֲפִלּוּ הָכִי בִּתְלִישַׁת וְהַגְבָּהַת הַמִּקְצָת קָנָה הַשְּׁאָר עַל יְדֵי קִנְיַן הַשְּׂכִירוּת. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב:

פרק ה - משנה ח

יַיִן וָשֶׁמֶן. וְהוּא הַדִּין לְפֵרוֹת. אֶלָּא אַגַּב דְּבָעֵי לְאוֹרוֹיֵי בְּסֵיפָא חַיָּב לְהַטִּיף שְׁלֹשָׁה טִפִּין נָקַט יַיִן וָשֶׁמֶן. רַ"שׁ: וְהוּקָרוּ כוּ'. לָאו דַּוְקָא. אֶלָּא שֶׁאֵין דֶּרֶךְ לַחֲזֹר בְּחִנָּם. תּוֹסָפוֹת: דְּאִי שֶׁל לוֹקֵחַ, רִאשׁוֹן רִאשׁוֹן קָנָה כְּשֶׁיֵּשׁ שְׁנָתוֹת בַּמִּדָּה. וּסְתָמָא תְּנַן אֲפִלּוּ בְּמִדָּה שֶׁיֵּשׁ לָהּ שְׁנָתוֹת. וְאִי דְּמוֹכֵר, אֲפִלּוּ מִשֶּׁנִּתְמַלֵּאת לֹא קָנָה וַאֲפִלּוּ בִּרְשׁוּת לוֹקֵחַ, דִּכְשֵׁם שֶׁכֵּלָיו שֶׁל לוֹקֵחַ אֵין קוֹנוֹת בִּרְשׁוּת מוֹכֵר, כָּךְ כֵּלָיו שֶׁל מוֹכֵר אֵין קוֹנוֹת בִּרְשׁוּת לוֹקֵחַ: הָרַמְבַּ"ם. וּלְאַשְׁמְעִינַן דַּאֲפִלּוּ הָכִי לֹא קָנָה אֶלָּא אִם כֵּן הַמִּדָּה אֵינָהּ שֶׁל מוֹכֵר. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: הָרַ"שׁ. וְתֵימַהּ, דְּאִם כֵּן אַמַּאי שָׁבִיק יֻקְרָא וְזוֹלָא דְּאַיְרֵי בָּהּ. וּפֵרֵשׁ רִיבַ"ם, שֶׁדֶּרֶךְ הַסַּרְסוּר לִמְדֹּד בְּשָׂכָר, וְנִשְׁבְּרָה עַל יְדֵי הַמְּדִידָה וּבְלֹא אֹנֶס. תּוֹסָפוֹת: שֶׁל מוֹכֵר. לְפִי שֶׁנִּתְיָאֲשׁוּ הַבְּעָלִים. הָרַמְבַּ"ם. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב:

פרק ה - משנה ט

אֲבָל בְּחִיּוּב הַפֻּנְדְּיוֹן, לֹא קַשְׁיָא, דְּכֵיוָן שֶׁעִכְּבוֹ הַחֶנְוָנִי בְּיָדוֹ נִתְחַיֵּב בַּחֲלִיפָיו וּבְתַשְׁלוּמִין גְּמוּרִים כוּ'. דְּאִם הָיָה מַחֲזִיר אוֹתוֹ הַפֻּנְדְּיוֹן עַצְמוֹ הָכִי נַמִּי דְּפָטוּר. תּוֹסָפוֹת. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: שֶׁעַל מְנָת כוּ'. וַאֲפִלּוּ צְלוֹחִית שֶׁלָּקַח לָמֹד בָּהּ לַאֲחֵרִים לֹא מְחַיֵּב בֵּיהּ רַבִּי יְהוּדָה, דִּסְבִירָא לֵיהּ דְּלֹא הֲוֵי אֶלָּא שׁוֹאֵל, וְכִי מַחֲזִירוֹ לְמָקוֹם שֶׁשְּׁאָלוֹ מִשָּׁם דַּי. גְּמָרָא: וּמוֹדִים כוּ'. הֵיכָא דְּלֹא נְטָלָהּ לָמֹד בָּהּ לַאֲחֵרִים אֲפִלּוּ נְטָלָהּ לָמֹד לַתִּינוֹק כוּ', וְאִם כֵּן לֹא אַפְקָהּ מֵרְשׁוּת בְּעָלִים, וּמִשּׁוּם הָכִי פָּטוּר מִן הַצְּלוֹחִית. רַ"שׁ:

פרק ה - משנה י

חִלּוּף כוּ'. סִיטוֹן פַּעַם אַחַת בְּשָׁנָה. דְּכֵיוָן שֶׁמּוֹכֵר תָּדִיר אֵין הַמַּשְׁקֶה נִקְרָשׁ בְּתוֹכוֹ. אֲבָל בַּעַל הַבַּיִת כוּ'. רַ"שׁ. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: חֶנְוָנִי כוּ' בַּשַּׁבָּת. וַאֲפִלּוּ רַבָּן שִׁמְעוֹן בֶּן גַּמְלִיאֵל מוֹדֶה. רַ"שׁ וְנִמּוּקֵי יוֹסֵף: כְּמוֹ וּמָחָה ה' דִּמְעָה מֵעַל כָּל פָּנִים . רַ"שׁ: כְּבָשָׂר וְשֶׁמֶן וּדְבַשׁ וְדָג מָלִיחַ, מִפְּנֵי שֶׁנִּדְבַּק בּוֹ מֵהַלִּכְלוּךְ וּמַכְבִּיד הַמִּשְׁקָל. נִמּוּקֵי יוֹסֵף: לְפִי שֶׁיֵּשׁ לָהֶן בֵּית קִבּוּל נִדְבָּק בָּהֶן יוֹתֵר. הָרַ"שׁ:

פרק ה - משנה יא

בַּיָּבֵשׁ. כְּגוֹן פֵּרוֹת כַּמּוֹן וּפִלְפְּלִין אֵין צָרִיךְ לְקַנֵּחַ, שֶׁלֹּא נִדְבַּק בָּהֶן מְאוּמָה. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: וְהוּא הַדִּין לַעֲשָׂרָה לִיטְרִין בְּיַחַד, כִּדְאִיתָא בַּגְּמָרָא, הֶכְרֵעַ אֶחָד לְכֻלָּם. וּמוֹקֵי לָהּ בְּמָקוֹם שֶׁנָּהֲגוּ לְהַכְרִיעַ: מִשּׁוּם דְּנָפִישׁ יוֹתֵר מִדַּאי הֶכְרֵעַ טֶפַח בְּדָבָר קַל. נִרְאֶה לִי. הָרֹא"שׁ: נוֹתֵן כוּ'. דְּאָמַר קְרָא אֵיפָה שְׁלֵמָה וָצֶדֶק. וָצֶדֶק מְיֻתָּר הוּא, לְמִדְרָשׁ אַף עַל פִּי שֶׁאֵיפָה שְׁלֵמָה מָדַדְתָּ, צַדֵּק מִשֶּׁלְּךָ וְתֵן לוֹ. גְּמָרָא: וְכֵן הַגְּרָמָה דִּשְׁחִיטָה לְשׁוֹן הַכְרָעָה וְהַטָּיָה הִיא, שֶׁמַּטֶּה הַסַּכִּין חוּץ לִמְקוֹם שְׁחִיטָה. הָרַ"שׁ: מִדְּלֹא קָתָנֵי אֶחָד מֵעֲשָׂרָה כוּ', דַּיְקִינַן לָהּ בַּגְּמָרָא דַּחֲסוּרֵי מִחַסְּרָא וְהָכִי קָתָנֵי, אֶחָד מֵעֲשָׂרָה לַעֲשָׂרָה כוּ'. וְנִרְאֶה מִפְּנֵי שֶׁהַלַּח נִדְבָּק יוֹתֵר וְאֵינוֹ מַטִּיף לוֹ הַכֹּל, צָרִיךְ לְהוֹסִיף לוֹ יוֹתֵר מִבְּיָבֵשׁ. הָרַ"שׁ: וַאֲפִלּוּ בְּמָקוֹם שֶׁמּוֹחֲקִים, נַמִּי אִי אֶפְשָׁר לִמְחֹק כָּל כָּךְ בְּצִמְצוּם. הָרַ"שׁ: בַּדַּקָּה. דְּמַפְסִיד מוֹכֵר מִשּׁוּם שֶׁאֵינוֹ יָכוֹל לִמְחֹק כָּל כָּךְ, וּבַגַּסָּה מוֹחֵק יוֹתֵר כוּ'. רַ"שׁ: מִדִּכְתִיב אֵיפָה כוּ' וָצֶדֶק, לְאַטְפוּיֵי אָתָא, מִכָּל מָקוֹם, שֶׁאַף זֶה יָכוֹל לָבֹא לְרַמָּאוּת. דְּאִיכָּא דְּחָזֵי בַּמְּדִידָה וְלֹא יָדַע לֹא בַּפָּחוֹת וְלֹא בַּתּוֹסֶפֶת, וְיֹאמַר כָּךְ הַמִּנְהָג. הָרַ"שׁ:
