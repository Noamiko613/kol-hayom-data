אֶלָּא הַמּוֹצִיא מֵחֲבֵרוֹ עָלָיו הָרְאָיָה: חַיָּב כוּ'. וְהָנֵי מִלֵּי דְּלֹא צִמְּחוּ מֵחֲמַת עַצְמָן. אֲבָל מֵחֲמַת דָּבָר אַחֵר, כְּגוֹן שֶׁלָּקוּ בַּבָּרָד אוֹ כַּיּוֹצֵא בוֹ, אֵינוֹ חַיָּב בְּאַחֲרָיוּתָן. הָרִי"ף. וְאִם תֹּאמַר אִם כֵּן פְּשִׁיטָא. וְנִרְאֶה לִי דְּאַשְׁמְעִינַן דְּלֹא מָצֵי אָמַר לֵיהּ הַב לִי חִטַּי וְאַחְזִיר לְךָ דְּמֵיהֶם. בֵּית יוֹסֵף:

לְפִי גִּרְסַת הָרַ"ב, הַמְקַבֵּל, לֹא הָיָה צָרִיךְ לְפָרֵשׁ הַלּוֹקֵחַ, דְּהוּא הַמְקַבֵּל. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: כְּלוֹמַר דְּמַתְנִיתִין דִּתְנַן מִסְפָּר גָּדוֹל, לָאו דַּוְקָא, אֶלָּא הוּא הַדִּין מִסְפָּר קָטָן, וְלֹא אָמְרִינַן דְּמִסְפָּר קָטָן מַקְפִּיד וְרוֹצֶה שֶׁיִּהְיוּ כֻּלָּם טוֹבִים. וְכֵן כָּתְבוּ הָרַ"שׁ וְנִמּוּקֵי יוֹסֵף וְהַטּוּר, דְּהוּא הַדִּין אַחַת לְעֶשֶׂר. וּקְצָת קָשֶׁה מִפֶּרֶק ז' מִשְׁנָה ב' דְּאִצְטְרִיךְ לְמִתְנֵי מִסְפָּר קָטָן וְאִי לָאו הָכִי הֲוָה אֲמֵינָא דְּלֹא, וְאִם כֵּן הָכָא נַמִּי אֵימָא דְּאִין הָכִי נַמִּי דְּלֹא: כְּמוֹ וְאֶת פִּרְיָהּ יְקוֹסֵס וְיָבֵשׁ . הָרַ"שׁ: צָרִיךְ לוֹמַר מַרְתֵּף זֶה שֶׁל יַיִן כוּ' אֲבָל בְּיַיִן סְתָם וְלֹא אָמַר לְמִקְפָּה, פְּלִיגֵי אָמוֹרָאֵי בַּגְּמָרָא. וּפָסְקוּ הַפּוֹסְקִים כְּמַאן דְּאָמַר דִּמְקַבֵּל עָלָיו עֶשֶׂר לְמֵאָה. וְהָכִי אוֹקְמָהּ הָרִי"ף לְמַתְנִיתִין: כְּלוֹמַר שֶׁשּׁוֹאֲבוֹת הַיַּיִן לְפִי שֶׁמַּזִּיעוֹת אוֹתוֹ דֶּרֶךְ דָּפְנוֹתֵיהֶן:

דְּאִיכָּא לְמֵימַר כֵּלָיו גָּרְמוּ. הָרַ"שׁ: דִּבְהָכִי מַיְרֵי כְּדִמְפָרֵשׁ הָרַ"ב בַּסֵּיפָא וְאִם יָדוּעַ. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: מַחְמִיץ. בְּכָל שָׁנָה. רַ"שׁ: מְבֻשָּׂם. שֶׁיִּתְקַיֵּם כְּדֶרֶךְ הַמִּתְקַיְּמִין. רַ"שׁ: עַד הָעֲצֶרֶת. קָשֶׁה דְּהָא בְּכֵלָיו דְּלוֹקֵחַ אַיְרִינַן כוּ'. וְיֵשׁ לוֹמַר דְּוַדַּאי אִם נִמְצָא חָמֵץ מָצֵי אָמַר לֵיהּ קַנְקַנֶּךָ גָּרְמוּ. אֲבָל הָכָא לֹא אַיְרֵי בְּהָכִי אֶלָּא אַשְׁמְעִינַן דְּצָרִיךְ לִתֵּן לוֹ יַיִן הָרָאוּי לְהִתְקַיֵּם עַד הָעֲצֶרֶת. תּוֹסָפוֹת. וְכֵן כָּתְבוּ הַטּוּר וְהָרֹא"שׁ. וּבָרַ"שׁ לֹא נִרְאֶה כֵּן. וְהַבֵּית יוֹסֵף לֹא הֶעְתִּיק מִכָּל זֶה. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: וְהוּא שֵׁם מֻרְכָּב, שַׁתָּא קָדְמֵיתָא. עָרוּךְ: שָׁלֹשׁ שָׁנִים. מִשָּׁנָה דְּקֹדֶם אֶשְׁתָּקַד, דַּהֲוָה לְהוּ שָׁלֹשׁ שָׁנִים עִם שָׁנָה שֶׁאָנוּ עוֹמְדִים בָּהּ. רַ"שׁ:

לִבְנוֹת. דְּלָא טְרִיחַ בִּבְצִיר מֵאַרְבַּע עַל שֵׁשׁ. אֲבָל לַחֲלֻקַּת אַחִין אוֹ שֻׁתָּפִין בְּאַרְבַּע עַל אַרְבַּע הֲוֵי בַּיִת. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: בֵּית כוּ'. גְּמָרָא, וְלִתְנֵי בֵּית חַתְנוּת לִבְנוֹ וּלְבִתּוֹ. מִלְּתָא אַגַּב אוֹרְחֵיהּ קָא מַשְׁמַע לָן, דְּלָא דַרְכָּא דְּחַתְנָא לְמֵידָר בֵּית חֲמוֹהִי. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: הָרוֹצֶה כוּ'. אִיכָּא דְּאָמְרֵי בַּגְּמָרָא דְּרַבִּי עֲקִיבָא נַמִּי מוֹדֶה בָּזֶה, אֶלָּא שֶׁסְּבִירָא לֵיהּ שֶׁלִּפְעָמִים אָדָם עוֹשֶׂה דִּירָתוֹ רֶפֶת בָּקָר. עַיֵּן תּוֹסְפוֹת יוֹם טוֹב: וּבְשֶׁל בַּיִת רִאשׁוֹן קָאָמְרֵי, כַּכָּתוּב בִּמְלָכִים א' ו'. דְּאִלּוּ בַּיִת שֵׁנִי תְּנַן פֶּרֶק ד' דְּמִדּוֹת שֶׁהָיָה גָּבְהוֹ אַרְבָּעִים. וְאִם תֹּאמַר אִי הָכִי לֵיזַל בָּתַר אוֹתוֹ בִנְיָן. וְיֵשׁ לוֹמַר דְּאֵין לָנוּ לֵילֵךְ אֶלָּא אַחַר הַפָּחוֹת. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב:

בּוֹר. דַּוְקָא נָקַט בּוֹר, דְּאִלּוּ חֶדֶר דַּרְכּוֹ לִכָּנֵס שָׁם בֵּין בַּיּוֹם בֵּין בַּלַּיְלָה. נִמּוּקֵי יוֹסֵף. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: גְּמָרָא. דְּמִשּׁוּם גְּנֵבָה שֶׁלֹּא יִגְנֹב חֲפָצָיו שֶׁל בַּעַל הַבַּיִת לֹא חָיְשִׁינַן, שֶׁהָאִשָּׁה מְשַׁמֶּרֶת הַבַּיִת. רַ"שׁ. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב:

לְשׁוֹן הָרַ"שׁ, כְּגוֹן יוֹרְשִׁים אוֹ שֻׁתָּפִים שֶׁחִלְּקוּ וְנִתְרַצָּה כוּ'. עַד כָּאן. אֲבָל בְּלוֹקֵחַ גִּנָּה לִפְנִים מִגִּנָּתוֹ שֶׁל חֲבֵרוֹ, נִרְאֶה דְּאֵין צָרִיךְ לְרִצּוּי, אֶלָּא מִמֵּילָא יֵשׁ לוֹ דֶּרֶךְ, דְּהָכָא נַמִּי אִיכָּא לְמֵימַר דִּבְעַיִן יָפָה מָכַר כְּמוֹ בְּבוֹר. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: וְכֵן בְּמַתְנִיתִין תְּנַן לְתוֹךְ שָׂדֶה אַחֶרֶת וְאֶפְשָׁר דְּבִלְשׁוֹן מִשְׁנָה נִכְלַל גִּנָּה בְּשֵׁם שָׂדֶה. וְעַיֵּן מִשְׁנָה ז' פֶּרֶק ד' : תַּגָּרִין. לִקְנוֹת מִיַּרְקוֹת גִּנָּתוֹ. רַ"שׁ: אַחֶרֶת. כְּלוֹמַר לֹא יִכָּנֵס בְּתוֹכָהּ לְכַתְּחִלָּה כְּשֶׁאֵין צָרִיךְ לְזוֹ הַגִּנָּה כְּלוּם, אֶלָּא לִכָּנֵס דֶּרֶךְ עָלֵיהָ לְשָׂדֶה אַחֶרֶת שֶׁלּוֹ כְּדֵי לְקַצֵּר אֶת דַּרְכּוֹ לְהַכְנָסַת תַּגָּרִין. רַ"שׁ: נָתְנוּ כוּ'. כְּלוֹמַר נָתְנוּ לוֹ בֵּית דִּין דֶּרֶךְ מִן הַצַּד מִדַּעַת שְׁנֵיהֶם כוּ'. וְזֶה וְזֶה אֵין רַשָּׁאִים לְזָרְעָהּ הוֹאִיל כוּ', וּבָחֲרוּ לָהֶם דֶּרֶךְ זֶה עַל יְדֵי בֵּית דִּין. רַ"שׁ. וְהַטּוּר הֶעְתִּיק נָתַן וְכוּ': וְלֹא כוּ'. וְאַף עַל פִּי שֶׁנָּתְנוּ לוֹ מִדַּעַת שְׁנֵיהֶם, לְפִי שֶׁהוּא מַרְבֶּה עֲלֵיהֶם הַדֶּרֶךְ, וְלֹא נְתָנוֹ לוֹ אֶלָּא לְצֹרֶךְ גִּנָּתוֹ. הָרַמְבַּ"ם:
