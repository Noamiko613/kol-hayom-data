פרק ד - משנה א

כְּלוֹמַר שֶׁיִּתְחַיֵּב לוֹ בְּדִינְרֵי כֶּסֶף שֶׁהִתְנָה עִמּוֹ. וְלֹא בְתוֹרַת חֲלִיפִין דְּהַיְנוּ שֶׁכְּשֶׁמָּשַׁךְ זֶה נִקְנִים לוֹ דִּינְרֵי כֶסֶף וּבִרְשׁוּתוֹ הֵם, שֶׁאֵין קוֹנִין בְּפֵרוֹת, וְאֵין הַמַּטְבֵּעַ נִקְנֶה בַּחֲלִיפִין. אֶלָּא בְּתוֹרַת דָּמִים אַיְרִינַן, כְּלוֹמַר לְחַיְּבוֹ בְּדִינְרֵי כֶסֶף. וְתָנֵי הַזָּהָב מְחַיֵּב. גְּמָרָא: כְּמוֹ שֶׁאָמַר לוֹ. אִי אָמַר לוֹ מֵאַרְנָקִי חֲדָשָׁה יָהֵיבְנָא לָךְ, לֹא מָצֵי יָהֵיב לֵיהּ מֵאַרְנָקִי יְשָׁנָה, אַף עַל גַּב דַּעֲדִיפֵי מִנַּיְהוּ וְכוּ'. גְּמָרָא: הַנְּחֹשֶׁת כוּ'. אַף עַל גַּב דְּכָל שֶׁכֵּן הוּא מֵרֵישָׁא. אִצְטְרִיךְ, סַלְקָא דַּעְתָּךְ אֲמֵינָא הָנֵי פְרִיטֵי בְּאַתְרָא דְּסָגְיֵי אִנְהוּ חָרִיפֵי טְפֵי מִכַּסְפָּא אֵימָא טִבְעָא הָוֵי, קָא מַשְׁמַע לָן כֵּיוָן דְּאִיכָּא דּוּכְתָּא דְּלֹא סָגְיֵי בָּהּ, פֵּירֵי הָווּ. גְּמָרָא: כָּתְבוּ הַתּוֹסָפוֹת, תֵּימַהּ דְּכָל שֶׁכֵּן הִיא מֵרֵישָׁא. לְכָךְ נִרְאֶה דְּמָעוֹת הָרָעוֹת הַיְנוּ שֶׁיּוֹצְאוֹת קְצָת: אִצְטְרִיךְ, אַף עַל גַּב דְּאֵינוֹ חָסֵר כִּי אִם מְעַט וּבְדָבָר מֻעָט יַטְבִּיעוּ הַרְבֵּה מֵהֶם. תּוֹסָפוֹת: מִטַּלְטְלִין כוּ'. אַשְׁמְעִינַן דְּקוֹנִין אֲפִלּוּ מַטְבֵּעַ, כְּמוֹ מָעוֹת רָעוֹת וַאֲסִימוֹן, דְּמִקְרֵי פֵּירָא לְגַבֵּי כַּסְפָּא. וְהֵם אֵינָן קוֹנִין מִטַּלְטְלִין, דְּמִקְרֵי טִבְעָא לְגַבֵּי מִטַּלְטְלִין. תּוֹסָפוֹת: אֵינוֹ מוֹסִיף אַדִּלְעֵיל, דְּלָאו בְּמִטַּלְטְלִין בִּלְבַד הוּא אֶלָּא בְּמִטַּלְטְלִין וּמַטְבֵּעַ בְּיַחַד. וְכֵן נִרְאֶה גִּרְסַת רַשִׁ"י דְּכָתַב וְהַאי כָּל כוּ' וְלֹא כָּתְבוּ וְהַאי זֶה הַכְּלָל. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב:

פרק ד - משנה ב

כֵּיצַד. הוּא פֵּרוּשָׁא דְּזָהָב וּנְחֹשֶׁת קוֹנִים לַכֶּסֶף, וְכֶסֶף אֵינוֹ קוֹנֶה אוֹתָם. וּמְפָרֵשׁ מִשּׁוּם שֶׁאֵלּוּ פֵּרוֹת לְגַבֵּי כֶסֶף, וּבְפֵרוֹת עוֹשִׂין הַמְּשִׁיכָה. נִמּוּקֵי יוֹסֵף: וְגָמַר הֶדְיוֹט מֵהֶקְדֵּשׁ. רַשִׁ"י. וְהַנִמּוּקֵי יוֹסֵף כָּתַב דְּלֹא גָּמַר כוּ', וְהָכָא לָאו מִקְּרָא יָלֵיף אֶלָּא מִסְּבָרָא בְּעָלְמָא, דִּבְדִין הוּא שֶׁיְּהוּ קוֹנוֹת, כֵּיוָן שֶׁרֹב הַקִּנְיָנִים נַעֲשִׂים בְּכֶסֶף. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: כְּלוֹמַר בְּאֹנֶס, דְּלֵיכָּא לְאִשְׁתַּלּוֹמֵי מִבַּעַל הַדְּלֵקָה. גְּמָרָא: אֲבָל בְּמָעוֹת לֹא יוּכַל לוֹמַר נִשְׂרְפוּ, דִּלְכָל הַפָּחוֹת יִהְיֶה עֲלֵיהֶן כְּשׁוֹמֵר חִנָּם וְאֵין לָהֶם שְׁמִירָה אֶלָּא בַּקַּרְקַע, אִי נַמִּי דְּאֵין טֹרַח בְּהַצָּלָתָן. תּוֹסָפוֹת: וּמִשּׁוּם הָכִי הִפְקִיעוּ חֲכָמִים קִנְיָן דְּאוֹרַיְתָא כִּדְקָתָנֵי רֵישָׁא מָשַׁךְ וְלֹא נָתַן מָעוֹת קָנָה, (וְלֹא הִצְרִיכוּ מָעוֹת וּמְשִׁיכָה), דְּאִי מְשִׁיכָה לְבַדּוֹ לֹא תִקְנֶה יֹאמַר לוֹקֵחַ לַמּוֹכֵר נִשְׂרְפוּ חִטֶּיךָ בַּעֲלִיָּה. תּוֹסָפוֹת: וְטַעְמָא דְּלוֹקֵחַ יָכוֹל גַּם כֵּן לַחֲזֹר בּוֹ, דְּכָל שֶׁכֵּן כִּי מוֹקְמַת לַחֲזָרָה נַמִּי בִּרְשׁוּת לוֹקֵחַ מָסַר נַפְשֵׁיהּ הַמּוֹכֵר וְטָרַח וּמַצִּיל, שֶׁאִם יִרְאֶה הַלּוֹקֵחַ דְּלֵקָה בָּאָה יֹאמַר חוֹזְרַנִּי בִּי. רַשִׁ"י. וְהָרַ"ב תְּחִלַּת הַתַּקָּנָה נָקַט. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: שֶׁפָּרַע. לְפִי שֶׁהַשֵּׁם יִתְבָּרַךְ אֵינוֹ מְקַבֵּל הַפֵּרָעוֹן, שֶׁאִם חָטְאוּ לְפָנָיו מַה פָּעֲלוּ לוֹ. לְפִיכָךְ לֹא אָמְרוּ מִי שֶׁנִּפְרַע כוּ' אֶלָּא שֶׁפָּרַע, כְּלוֹמַר שֶׁפָּרַע מֵהֶם חוֹבָם, וְלֹא לוֹ: וְקָשֶׁה מַאי תֵּבַת כָּל. וְנִרְאֶה לִי דְּהָכִי קָאָמַר, כָּל נוֹשֵׂא שֶׁיִּזְדַּמֵּן מִן הָרָאוּי הוּא מִי שֶׁהַכֶּסֶף בְּיָדוֹ כוּ', הִלְכָּךְ מוֹכֵר שֶׁהַכֶּסֶף בְּיָדוֹ מָצֵי הֲדַר בֵּיהּ. בֵּית יוֹסֵף: כָּל דְּבָרָיו דִּבְרֵי תֵּימַהּ הֵם. רֵישׁ דְּבָרָיו שֶׁכָּתַב דְּזִמְנִין כוּ', וּמַשְׁמָעוּת לִישְׁנָא דְּרַבִּי שִׁמְעוֹן דִּלְעוֹלָם כָּל שֶׁהַכֶּסֶף כוּ'. וְעוֹד קָשֶׁה וְלָמָּה יוּכַל הַמּוֹכֵר לַחֲזֹר בּוֹ כֵּיוָן דְּבָטֵל טַעֲמָא דִּגְזֵרָה. וְגַם בְּמַה שֶּׁכָּתַב וְאֵין הֲלָכָה כוּ' אֶלָּא כוּ' יָכוֹל לַחֲזֹר בֵּין לוֹקֵחַ בֵּין מוֹכֵר, קָשֶׁה, אַמַּאי, כֵּיוָן דְּלֵיכָּא טַעֲמָא דִגְזֵרָה. וְאֶפְשָׁר דְּתָנָא קַמָּא לֹא סְבִירָא לֵיהּ כְּוָתֵיהּ דְּזִמְנִין כוּ', אֶלָּא שְׁנֵיהֶם יְכוֹלִין לַחֲזֹר כוּ'. וּבַגְּמָרָא מוּכָח דְּהָכָא לָאו בַּעֲלִיָּתוֹ שְׂכוּרָה כוּ' אֶלָּא דְּרַבִּי שִׁמְעוֹן סְבִירָא לֵיהּ דְּהַתַּקָּנָה לֹא הָיְתָה אֶלָּא עַל הַמּוֹכֵר, אֲבָל הַלּוֹקֵחַ אֵינוֹ יָכוֹל לַחֲזֹר בּוֹ (דְּהַיְנוּ בְּתַקָּנָה זוֹ), אֲבָל בַּעֲלִיָּה מֻשְׂכֶּרֶת כֻּלֵּי עָלְמָא סְבִירָא לְהוּ דִּשְׁנֵיהֶם אֵינָם יְכוֹלִים לַחֲזֹר בָּהֶם. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב:

פרק ד - משנה ג

אַרְבָּעָה כוּ'. הָוֵי מָצֵי לְמִתְנֵי כֶּסֶף מִשֵּׁשׁ כֶּסֶף לַדִּינָר. וְיֵשׁ לוֹמַר מִשּׁוּם מַתְנִיתִין ז' דְּפוֹחֵת וְהוֹלֵךְ עַיֵּן שָׁם. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: כְּלוֹמַר שֶׁהוּא שָׁוֶה סֶלַע. וְלֵיכָּא לְפֵרוּשֵׁי דְּזַבְּנֵיהּ בְּעֶשְׂרִים, דִּלְפִי זֶה נִתְאַנָּה הַמּוֹכֵר, וּבַסֵּיפָא תְּנַן עַד מָתַי מֻתָּר לְהַחֲזִיר וּמְפָרְשִׁים לָהּ בְּלוֹקֵחַ, דְּמוֹכֵר לְעוֹלָם חוֹזֵר, אֶלָּא דְּזַבְּנֵיהּ בְּעֶשְׂרִים וּשְׁמֹנָה. דְּבִכְלָל שְׁתוּת לַמִּקָּח בֵּין שֶׁנִּתְאַנָּה מוֹכֵר דְּזַבֵּין שָׁוֶה עֶשְׂרִים וְאַרְבָּעָה בְּעֶשְׂרִים, וּבֵין דְּנִתְאַנָּה לוֹקֵחַ דְּזָבִין שָׁוֶה עֶשְׂרִים וְאַרְבָּעָה בְּעֶשְׂרִים וּשְׁמֹנָה. הָכִי מַסְּקִינַן בַּגְּמָרָא אַלִּיבָּא דְּרַב דְּסָבַר שְׁתוּת מִקָּח דַּוְקָא. אֲבָל שְׁמוּאֵל סָבַר שְׁתוּת מָעוֹת נַמִּי שָׁנִינוּ. וּלְפִי דְּלִישְׁנָא דַּיֵּק טְפֵי אַלִּיבָּא דְרַב, פֵּרֵשׁ רַשִׁ"י וְהָרַ"ב אַלִּיבֵּיהּ אַף עַל גַּב דַּהֲלָכָה כִּשְׁמוּאֵל בְּדִינֵי: שְׁתוּת. דְּאִלּוּ פָּחוֹת מִשְּׁתוּת הָוֵי מְחִילָה, לְפִי שֶׁכֵּן הוּא דֶּרֶךְ מִקָּח וּמִמְכָּר, שֶׁאֵין הַלּוֹקֵחַ וְהַמּוֹכֵר יְכוֹלִים לְכַוֵּן דְּמֵי הַמִּקָּח בְּצִמְצוּם. וְדֶרֶךְ הָעוֹלָם לִמְחֹל טָעוּת עַד שְׁתוּת. טוּר:

פרק ד - משנה ד

מְפֹרָשׁ בַּגְּמָרָא דְּהַיְנוּ תַּגָּר סַפְסָר (שֶׁהוּא קוֹנֶה וּמוֹכֵר מִיָּד תָּמִיד), דְּמֵידַע זְבִינְתֵּיהּ כַּמָּה שָׁוֶה (שֶׁהֲרֵי לֹא שָׁהָה בֵּין קְנִיָּתָהּ לִמְכִירָתָהּ. רַשִׁ"י) וְאַחוּלֵי אַחִיל כוּ': וְהוּא הַדִּין לַמּוֹכֵר, וְשִׁיּוּרֵי שַׁיְּרֵיהּ. גְּמָרָא:

פרק ד - משנה ה

הַסֶּלַע. דְּלֹא דָּמֵי לְטַלִּית, דְּסֶלַע דְּכֵיוָן דְּלֹא סַגִּי לֹא מָחַל. גְּמָרָא: שְׁנֵי כוּ'. וְהָא דְּלֹא אָמַר מָעָה לְדִינָר, מִשּׁוּם דְּרַבִּי יְהוּדָה נָקַט פֻּנְדְּיוֹן קָאָמַר אִיהוּ שְׁתֵּי פֻנְדְּיוֹן:

פרק ד - משנה ו

עַד כוּ'. דְּלֹא דָּמֵי לְטַלִּית דְּכָל אֱינַשׁ קִים לֵיהּ בְּגַוֵּיהּ. גְּמָרָא:

פרק ד - משנה ז

אֲבָל בַּגְּמָרָא פָּרֵיךְ הָא נַמִּי תָנִינָא כוּ'. וּמְשָׁנֵי חָמֵשׁ פְּרוּטוֹת אִצְטְרִיךְ לֵיהּ, וּרְצוֹנוֹ לוֹמַר דִּמְצִיאָה, דְּאִינָךְ כֻּלְּהוּ תָנִינָא. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: כְּלוֹמַר יוֹדֶה לוֹ כָּל הַשְּׁאָר:

פרק ד - משנה ח

חֲמִשָּׁה כוּ'. כֻּלְּהוּ מַתְנְיָין בְּדוּכְתַּיְהוּ אֶלָּא לְאַשְׁמְעִינַן דְּתוּ לֹא: שֶׁל דְּמַאי. בַּגְּמָרָא אָמְרִינַן טַעֲמָא דְּמַתְנִיתִין רַבִּי מֵאִיר הִיא דְּאָמַר עָשׂוּ חִזּוּק לְדִבְרֵיהֶם כְּשֶׁל תּוֹרָה אֲבָל הָרַמְבַּ"ם וְהָרַ"ב דְּלֹא כָּתְבוּ שֶׁאֵין הֲלָכָה כֵן (דְּהָא בְּרֵישׁ פֶּרֶק ב' דְּעָרְלָה הִשְׁמִיט הָרַמְבַּ"ם וּתְרוּמַת מַעֲשֵׂר שֶׁל דְּמַאי) נִרְאֶה דִסְבִירָא לֵיהּ דְּכֻלֵּי עָלְמָא הִיא. וְטַעְמָא כִּדְפֵרֵשׁ בַּיְרוּשַׁלְמִי מִשּׁוּם גְּזֵרָה שֶׁלֹּא יִנְהֲגוּ בָּהּ בִּקְדֻשָּׁה. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: לֹא יָדַעְתִּי אַמַּאי הֻצְרְכוּ לְפָרֵשׁ שֶׁחַיָּב קָרְבָּן. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: וְהַגּוֹזֵל כוּ'. וְכֵן כּוֹפֵר בַּפִּקָּדוֹן כִּדִתְנִינַן לְהוּ בְּמִשְׁנָה ז' פֶּרֶק ט' דְּבָבָא קַמָּא. וְהוֹאִיל וּבְחַד קְרָא כְּתִיבֵי בַּהֲדֵי הֲדָדֵי לֹא הֻצְרַךְ לִשְּׁנוֹתָן כְּלָל. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב:

פרק ד - משנה ט

וְהַשְּׁטָרוֹת. הַמּוֹכֵר שְׁטָרוֹת לִגְבוֹת חוֹב שֶׁבְּתוֹכָן. רַשִׁ"י: וְהַקַּרְקָעוֹת. בְּמַשִּׂיג גְּבוּל. אִי נַמִּי שֶׁגָּנַב גְּפָנִים טְעוּנוֹת. תּוֹסָפוֹת: וּבְרֵישׁ מְרֻבָּה מְמַעֵט מִוְּגֻנַּב מִבֵּית הָאִישׁ. וְיֵשׁ לוֹמַר כוּ'. אִי נַמִּי הָתָם בְּגַנָּב עַצְמוֹ וְהָכָא אִצְטְרִיךְ לְטוֹעֵן טַעֲנַת גַּנָּב. תּוֹסָפוֹת. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: אֵינוֹ נִשְׁבָּע. הוּא הַדִּין נַמִּי שֶׁאֵינוֹ מְשַׁלֵּם כוּ' אִם פָּשַׁע בָּהּ. אֶלָּא מִשּׁוּם דְּבַכָּתוּב לֹא נֶאֶמְרוּ תַּשְׁלוּמִין כְּלָל בְּשׁוֹמֵר חִנָּם, מִשּׁוּם הָכִי לֹא הִזְכִּירוּ בַּמִּשְׁנָה אֶלָּא אֵינוֹ נִשְׁבָּע. פּוֹסְקִים: אוֹנָאָה. וְכֵן חוֹלֵק בְּתַשְׁלוּמֵי אַרְבָּעָה וַחֲמִשָּׁה בְּמִשְׁנָה ד' פֶּרֶק ז' דְּבָבָא קַמָּא. וּמִמֵּילָא דְּחוֹלֵק נַמִּי בְּכֶפֶל, דְּאִי לֵיכָּא כְּפֵלָא לֵיכָּא אַרְבָּעָה וַחֲמִשָּׁה:

פרק ד - משנה י

שֶׁיָּכוֹל הוּא לוֹמַר לֹא עָשִׂיתִי כִּי אִם לְטוֹבָה, שֶׁהָיִיתִי חָפֵץ לִקְנוֹת מִקָּח זֶה. וְאֵין דַּיָּנֵי בְּנֵי אָדָם יְכוֹלִין לָדוּן עָלָיו. לְפִיכָךְ נֶאֱמַר וְיָרֵאת מֵאֱלֹהֶיךָ, הֱוֵי יָרֵא מִן הַיּוֹדֵעַ מַחֲשָׁבוֹת אִם לְטוֹבָה אִם לְאוֹנָאָה: מַעֲשֵׂה אֲבוֹתֶיךָ. הַיְנוּ זוֹ אַף זוֹ, דַּאֲפִלּוּ מַעֲשֵׂה אֲבוֹתָיו אֵין לוֹ לְהַזְכִּיר. תּוֹסָפוֹת:

פרק ד - משנה יא

שֶׁמִּתְקַיֵּם יוֹתֵר. הָרֹא"שׁ. וּבַגְּמָרָא, דְּבֵין הַגִּתּוֹת שָׁנוּ. וּפֵרֵשׁ רַשִׁ"י שֶׁתּוֹסְסִין זֶה עִם זֶה וְנַעֲשִׂין טַעַם אֶחָד. אֲבָל לְאַחַר הַגִּתּוֹת שֶׁכְּבָר כָּל אֶחָד קָלַט רֵיחוֹ וְטַעֲמוֹ, אֵין מַשְׁבִּיחוֹ אֶלָּא פּוֹגְמוֹ. וּמִכָּל מָקוֹם אָמְרִינַן בַּגְּמָרָא דִּמְקוֹם שֶׁהַכֹּל טוֹעֲמִים קֹדֶם שֶׁיִּקְנוּ שָׁרֵי, שֶׁהֲרֵי יִטְעֲמֶנּוּ תְּחִלָּה: לְפִי שֶׁשִּׁמְרֵי חָבִית זֶה מְקַלְקְלִין חָבִית אַחֵר. טוּר: שֶׁאָז קוֹלְטִים הַמַּיִם טַעַם הַיַּיִן, אֲבָל שֶׁלֹּא בֵּין הַגִּתּוֹת הוּא נֶחְשָׁב כְּאִלּוּ הוּא מָזוּג:

פרק ד - משנה יב

מַתִּירִין. דְּאָמַר לֵיהּ אֲנָא מְפָלֵגְנָא אַמְגּוּזֵי, וְאַתְּ פָּלֵיג שִׁיסְקֵי. גְּמָרָא: כְּגוֹנֵב. אֶפְשָׁר דִּלְהָכִי לֹא קָרֵי לֵיהּ גּוֹנֵב הָעַיִן מַמָּשׁ לוֹמַר שֶׁאַף עַל פִּי שֶׁמּוֹדִיעוֹ שֶׁלֹּא בֵרֵר אֶלָּא מֵעַל פִּי הַמְּגוּרָה דַּאֲפִלּוּ הָכִי אָסוּר לְפִי שֶׁאֶפְשָׁר שֶׁעַל יְדֵי שֶׁרוֹאֶה בִּשְׁעַת הַמְּכִירָה עַל פִּי הַמְּגוּרָה שֶׁכֻּלָּן יָפוֹת וְאֵין בָּהֶן פְּסֹלֶת מִתּוֹךְ כָּךְ לֹא יַעֲלֶה עַל לִבּוֹ תָּמִיד שֶׁיֵּשׁ פְּסֹלֶת בּוֹ וּלְפִיכָךְ לֹא קוֹרְאוֹ אֶלָּא כְּגוֹנֵב כוּ':
