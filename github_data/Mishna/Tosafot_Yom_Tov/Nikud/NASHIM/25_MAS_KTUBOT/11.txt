פרק יא

פרק יא - משנה א

הַיְנוּ אַחַר תַּקָּנַת הַגְּאוֹנִים דִּכְתֻבָּה נִגְבֵּית מִמִּטַּלְטְלֵי, וּתְנַאי כְתֻבָּה כִּכְתֻבָּה דָּמֵי. אֲבָל לֹא מִדִּין הַגְּמָרָא: מַעֲשֵׂה. הָכִי גָרְסֵי הַתּוֹסָפוֹת, וְלֹא גָרְסֵי וּמַעֲשֵׂה יָדֶיהָ בְּוָי"ו. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: וְאֵין חַיָּבִין בִּקְבוּרָתָהּ. מַסְקֵי הַתּוֹסָפוֹת דְּאִצְטָרִיךְ לְאַשְׁמוֹעִינַן אֲפִלּוּ הֵיכָא דְּלֹא שָׁקְלָה אַלְמָנָה כְּתֻבָּתָהּ, כְּגוֹן דְּלֵיכָּא אֶלָּא מִטַּלְטְלֵי, אֵינָן חַיָּבִין בִּקְבוּרָתָהּ, וְאַף עַל גַּב דְּהַשְׁתָּא הֲווֹ כְּיוֹרְשֵׁי כְתֻבָּתָהּ, כֵּיוָן שֶׁאֵינָהּ גּוֹבָה אוֹתָהּ מֵהֶם. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: הָרַמְבַּ"ם. כִּדְאַשְׁכְּחָן בְּשׁוֹמֶרֶת יָבָם דְּיוֹרְשֵׁי כְתֻבָּתָהּ וְהֵן יוֹרְשֵׁי הַבַּעַל חַיָּבִין בִּקְבוּרָתָהּ. אֲבָל הָרַאֲבַ"ד הִשִּׂיג עָלָיו דְּשָׁאנֵי שׁוֹמֶרֶת יָבָם דְּיוֹרְשֵׁי הַבַּעַל הֵן יוֹרְשֵׁי הַכְּתֻבָּה מַמָּשׁ, אֲבָל זוֹ שֶׁמִּפְּנֵי שֶׁלֹּא נִשְׁבְּעָה הוּא שֶׁאֵין יוֹרְשֶׁיהָ נוֹטְלִים הַכְּתֻבָּה, לֹא מִפְּנֵי כֵן נוּכַל לִקְרֹא לְיוֹרְשֵׁי הַבַּעַל יוֹרְשֵׁי כְתֻבָּתָהּ, וַהֲלֹא הֵם אוֹמְרִים שֶׁכְּבָר נִפְרְעָה וְנָטְלָה צְרָרֵי וְהוֹרִישָׁה, אֶלָּא וַדַּאי יוֹרְשֵׁי נְדוּנְיָתָהּ וְנִכְסֵי מְלוֹג שֶׁלָּהּ קוֹבְרִים אוֹתָהּ, וְאַף עַל פִּי שֶׁלֹּא יָרְשׁוּ כְלוּם:

פרק יא - משנה ג

לְאַחֵר. וְכָל שֶׁכֵּן כְּשֶׁמָּחֲלְתָּהּ לַיְתוֹמִים, דְּכֵיוָן שֶׁאֵין לָהּ עֲלֵיהֶן כְּתֻבָּה הָא אִסְתַּלְּקָא מִבֵּיתַיְהוּ וְאֵין לָהּ עֲלֵיהֶם מְזוֹנוֹת. הָרַ"ן: שֶׁלֹּא בְּבֵית דִּין. אַשְׁמְעִינַן שֶׁמּוֹכֶרֶת לִמְזוֹנוֹת וְלֹא אִבְּדָה. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: לִמְזוֹנוֹת. בַּגְּמָרָא מְפָרְשִׁים דְּעֵצָה טוֹבָה קָמַשְׁמַע לָן, דְּלֹא לִקְּרוֹ לָהּ רַעַבְתָּנוּתָא, שֶׁהָרוֹאֶה שֶׁמּוֹכֶרֶת כָּל קַרְקָעוֹת אֵלּוּ סָבַר שֶׁמּוֹכַרְתָּן לִמְזוֹנוֹת מוֹצִיא עָלֶיהָ שֵׁם רַעַבְתָּנוּת. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב:

פרק יא - משנה ד

שָׁוֶה כוּ'. מַחֲלֹקֶת הַפּוֹסְקִים, אִיכָּא דְסָבְרֵי דְּאֵין אוֹנָאָה לְקַרְקָעוֹת כְּלָל, וּלְדִידְהוּ מַתְנִיתִין דְּהָכָא כִּפְשָׁטָהּ. וְאִיכָּא דְסָבְרֵי דְּאֵין אוֹנָאָה עַד פְּלַג אֲבָל בִּפְלַג אִיתָא אוֹנָאָה. וּמְפָרֵשׁ בַּיְרוּשַׁלְמִי, דְּהָכָא שָׁוֶה מָנֶה בְּמָאתַיִם מַיְרֵי שֶׁהוּקַר וְעָמַד בְּמָאתַיִם וְהַלּוֹקֵחַ רוֹצֶה לְקַיֵּם הַמֶּקָּח, וְשָׁוֶה מָאתַיִם בְּמָנֶה כְּשֶׁהוּזַל וְעוֹמֵד עַל מָנֶה. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: דְּהַמּוֹתָר מַתָּנָה הִיא, וְיֵשׁ לוֹמַר לַשָּׁלִיחַ נִתְּנָה, וְיֵשׁ לוֹמַר לַמְשַׁלֵּחַ. רַשִׁ"י: וְדִינָר. יֵשׁ מְפָרְשִׁים דִּינָר דַּוְקָא. וְלֹא נְהִירָא, כֵּיוָן דַּאֲפִלּוּ פָּחוֹת מִכְּדֵי אוֹנָאָה בָּטֵל, אֵין חִלּוּק בֵּין דִּינָר לְפָחוֹת. הָרֹא"שׁ: מְסַיֵּם רַשִׁ"י, אוֹ הָאוֹנָאָה עַצְמָהּ תִּשְׁעָה קַבִּין. וְהוּא הַדִּין אִם נִשְׁאַר לַיְתוֹמִים שָׁם לְבַד הָאוֹנָאָה תִּשְׁעָה קַבִּין, דְּכֵיוָן דְּיֵשׁ לַיְתוֹמִים קַרְקַע שָׁם כְּשִׁעוּר שָׂדֶה, יֹאמְרוּ אֵין רְצוֹנֵנוּ לִמְכֹּר קַרְקַע הָרָאוּי לָנוּ. אֲבָל אִם אֵין הָאוֹנָאָה רָאוּי לְהִצְטָרֵף לִכְדֵי שָׂדֶה, לֹא הִפְסִידָתָן כְּלָל: רַבִּי עֲקִיבָא. שֶׁשָּׁמַעְנוּ מִמֶּנּוּ בְּסוֹף פֶּרֶק קַמָּא דְבָבָא בַתְרָא דְּבֵית רֹבַע הוּא שִׁעוּר גִּנָּה. רַשִׁ"י: אַרְבַּע מֵאוֹת. אִצְטָרִיךְ, דְּלֹא תֵימָא דַּוְקָא בְּרֵישָׁא דְּכִי מָכְרָה מָנֶה בְּמָאתַיִם דְּאִסְתַּלְּקָא מֵהַאי בֵיתָא לְגַמְרֵי, אֲבָל הָכָא נֵימָא דְּאִם טָעֲתָה בְּמָנֶה רִאשׁוֹן יְהֵא מִכְרָהּ בָּטֵל אַטּוּ מָנֶה אַחֲרוֹן, דְּלֹא נֵיתֵי לְמֵימַר מִכְרָהּ קַיָּם, קָמַשְׁמַע לָן. גְּמָרָא, וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב:

פרק יא - משנה ה

שׁוּם. בְּשִׁי"ן יְמָנִית. כְּמוֹ וְשָׁם דֶּרֶךְ אַרְאֶנּוּ בְּיֵשַׁע אֱלֹהִים, דְּשָׁיֵם אוֹרְחֵיהּ סַגֵּי שָׁוֶה. עָרוּךְ: שְׁתוּת. הָכָא בְּיַתְמֵי תַּקּוּן שֶׁיְּהֵא בָטֵל, אַף עַל גַּב דְּקַיְמָא לָן דִּשְׁתוּת קָנָה וּמַחֲזִיר אוֹנָאָה. כֵּן כָּתַב מוהר"ר פַאלְק כֹּהֵן: בָּטֵל. הַיְנוּ בִּדְבָרִים שֶׁאֵין צָרִיךְ הַכְרָזָה. וְאֵילוּ הֵם, עֲבָדִים, שֶׁמָּא יִשְׁמְעוּ וְיִבְרְחוּ, מִטַּלְטְלִים וּשְׁטָרוֹת, שֶׁמָּא יִגָּנְבוּ. גְּמָרָא. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: קַיָּם. עַד פַּלְגָא. וְתַנְיָא נַמִּי הָכִי. גְּמָרָא: אִגֶּרֶת. שְׁטָר. לְפִי שֶׁכָּל הַקּוֹנֶה נִיחָא לֵיהּ דְּלִקְרוֹיֵיהּ אִנְשֵׁי לְהַאי שְׁטָרָא, כִּי הֵיכִי דְּלֵידְעוּ כֻּלֵּי עַלְמָא שֶׁעַל יְדֵי הַבֵּית דִּין וּבְדִקְדּוּק הָיְתָה הַקְּנִיָּה כוּ', וּלְפִיכָךְ נֹחַ לוֹ בְּיוֹתֵר שֶׁזֶּה הַשְּׁטָר יְהֵא גָלוּי לַכֹּל, לָכֵן קְרָאוּהָ אִגֶּרֶת, שֶׁכֵּן דֶּרֶךְ אִגֶּרֶת לְפָשְׁטָהּ וְלֹא לְסָתְמָהּ, וּכְמוֹ שֶׁכָּתְבוּ הַפּוֹסְקִים גַּבֵּי הַמְּגִלָּה. תּוֹסְפוֹת יוֹם טוֹב: מָנֶה בְּמָאתַיִם וְכוּ'. דַּוְקָא מָאתַיִם בְּמָנֶה. אֲבָל בְּפָחוֹת מִמָּנֶה לֹא. הָרַמְבַּ"ן. וְאֶפְשָׁר שֶׁזֶּה כְדַעַת הַסּוֹבְרִים דְּהָא דִבְקַרְקָעוֹת אֵין לָהֶם אוֹנָאָה הַיְנוּ עַד מֶחֱצָה דַּוְקָא. כֶּסֶף מִשְׁנֶה. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב פִּלְפּוּל נָאֶה:

הַשְּׁנִיָּה. בֵּין הִכִּיר בָּהּ בֵּין לֹא הִכִּיר בָּהּ: דְּאִשָּׁה רוֹצָה לְהִנָּשֵׂא יוֹתֵר מֵהָאִישׁ. רַשִׁ"י: וּבִמְמָאֶנֶת הַיְנוּ טַעֲמָא מִשּׁוּם דְּכֵיוָן דְּתַקּוּן רַבָּנָן נִשּׂוּאִין לִקְטַנָּה מִשּׁוּם תַּקַּנְתָּא כְּדֵי שֶׁלֹּא יִנְהֲגוּ בָהּ מִנְהַג הֶפְקֵר, זִכּוּ פֵּרוֹת נְכָסִים לַבַּעַל. וּבִשְׁנִיָּה מִשּׁוּם קְנָס. וּבְאַיְלוֹנִית הַיְנוּ טַעֲמָא מִשּׁוּם דְּאַף עַל גַּב דְּנִשּׂוּאֵי טָעוּת הוּא כֵּיוָן שֶׁלֹּא הִכִּיר בָּהּ, אֲפִלּוּ הָכִי כֵּיוָן שֶׁרָאֲתָה הָאִשָּׁה בַּעְלָהּ אוֹכֵל פֵּרוֹת וְשָׁתְקָה, מָחֲלָה. הָרַ"ן: אַיְלוֹנִית כִּשְׁנִיָּה. הָרַמְבַּ"ם. וְאִי בְּיוֹשֶׁבֶת תַּחְתָּיו לְאַחַר שֶׁנִּמְצֵאת אַיְלוֹנִית, אַמַּאי לֵית לָהּ, וַהֲרֵי הִיא כִּנְשָׂאָהּ מִתְּחִלָּה לְשֵׁם אַיְלוֹנִית, שֶׁיֶּשׁ לָהּ מְזוֹנוֹת. הָרַ"ן: דְּנִכְסֵי צֹאן בַּרְזֶל, יָכוֹל לוֹמַר שֶׁלִּי הוֹצֵאתִי, שֶׁהֲרֵי בִרְשׁוּת בֵּית דִּין נְשָׂאתִיהָ, וּכְשֶׁאוֹצִיאֶנָּה מִדַּעְתִּי אַחֲזִיר מַה שֶּׁקִּבַּלְתִּי. וְנִכְסֵי מְלוֹג, קַיְמָא לָן דְּעַיְלָא גְלִימָא פֵּרֵי הֲוֵי וּמְכַסֵּי בֵיהּ עַד דְּבָלֵי, וּלְפִיכָךְ כְּשֶׁהוֹצִיאָם בִּרְשׁוּת הוֹצִיאָם. הָרַ"ן: שֶׁזֶּה הַדִּין אָמְנָם קְנָס הוּא לַשְּׁנִיָּה עַל הָעֲבֵרָה, לְפִי שֶׁהַדַּעַת נוֹטָה אַחַר שֶׁאֵין לָהּ כְּתֻבָּה שֶׁלֹּא יִהְיֶה לַבַּעַל חֻקֵּי הַנִּשּׂוּאִין וְלֹא יֵהָנֶה מִנִּכְסֵי מְלוֹג, אֲבָל מִצַּד הַקְּנָס עָשִׂינוּ נִכְסֵי מְלוֹג שֶׁלָּהּ כְּנִכְסֵי שְׁאָר הַנָּשִׁים, שֶׁהָעִקָּר אֶצְלֵנוּ עַיְלָא גְלִימָא פֵּרֵי הֲוֵי. הָרַמְבַּ"ם: כְּתֻבָּה. וְהוּא הַדִּין לְכָל תְּנָאֶיהָ, דִּתְנַאי כְתֻבָּה כִּכְתֻבָּה. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: לְכֹהֵן גָּדוֹל. אַהִכִּיר בָּהּ דְּאַיְלוֹנִית קָאֵי, וְקָאָמַר דְּאַלְמָנָה כוּ' נַמִּי כִּי הִכִּיר בָּהּ יֶשׁ לָהּ כְּתֻבָּה. גְּמָרָא:
