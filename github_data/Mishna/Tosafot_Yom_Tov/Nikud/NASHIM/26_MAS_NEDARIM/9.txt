פרק ט

וְזֶה דָּבָר קַל וּפְרִיצוּת שֶׁכֵּן רְשָׁעִים דַּרְכָּן לִנְדֹּר, וְאִם אָמַר לָאו, דְּאִלּוּ הָיִיתִי יוֹדֵעַ לֹא הָיִיתִי נוֹדֵר, מַתִּירִין לוֹ. רַשִׁ"י: זֶהוּ פֵּרוּשׁ דְּאִם כֵּן אֵין נְדָרִים, וּכְאַבַּיֵּי בַּגְּמָרָא דְּפֵרֵשׁ אִם כֵּן אֵין נְדָרִים נִתָּרִים יָפֶה. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: דְּכֵיוָן דְּאַחֲצִיף כֻּלֵּי הַאי מֵעִקָּרָא דְּאַדְּרֵיהּ לְאָבִיו וְאִמּוֹ מִנְּכָסָיו, הָא אַחֲצִיף לֵיהּ, וְאִם אֵין לוֹ חֲרָטָה גְּמוּרָה יַחְצִיף פָּנָיו וְיֹאמַר שֶׁלֹּא הָיָה מַנִּיחַ בִּשְׁבִיל כְּבוֹדָם. גְּמָרָא, וּפֵרוּשׁ הָרֹא"שׁ:

בְּנוֹלָד. גְּמָרָא, מַאי טַעֲמָא דְּרַבִּי אֱלִיעֶזֶר, דְּאִלּוּ רַבָּנָן מִסְתַּבֵּר טַעֲמַיְהוּ כְּמוֹ שֶׁכָּתַב הָרַ"ב. אָמַר רַב חִסְדָּא דְּאָמַר קְרָא כִּי מֵתוּ כָּל הָאֲנָשִׁים , שֶׁהָיָה נִשְׁבַּע לְיִתְרוֹ שֶׁלֹּא לָשׁוּב מִצְרַיְמָה בִּשְׁבִיל אוֹתָן הָאֲנָשִׁים הַמְבַקְּשִׁים אֶת נַפְשׁוֹ, וּפָתַח לוֹ הַמָּקוֹם בְּנוֹלָד דְּמִיתָה. וְרַבָּנָן קָאָמְרֵי דְּמֵתוּ דִּקְרָא הַיְנוּ עָנִי כוּ' וְשָׁאנֵי עֲנִיּוּת דִּשְׁכִיחַ וְלֹא הָוֵי נוֹלָד. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: רַבִּי אֱלִיעֶזֶר כוּ'. מִשּׁוּם דְּנָקִיט גַּוְנֵי טוּבָא הָדַר וְתָנֵי, דִּבְכֻלְּהוּ רַבִּי אֱלִיעֶזֶר מַתִּיר וַחֲכָמִים אוֹסְרִין. הָרַ"ן:

וְאוֹמְרִים אִלּוּ הָיִיתָ יוֹדֵעַ כוּ' וְהוּא אוֹמֵר לָאו. רַשִׁ"י: מֵת כוּ'. וְהוּא הַדִּין אִפְּכָא אֶלָּא נָקַט מִלְּתָא דִּשְׁכִיחָא, דְּגַבֵּי כֶּלֶב לֹא שְׁכִיחָא הֲרִיגָה וּלְגַבֵּי נָחָשׁ שְׁכִיחָא טְפֵי הֲרִיגָה. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב:

וְעוֹד אָמַר רַבִּי מֵאִיר. קֻלָּא אַחֶרֶת. תּוֹסָפוֹת. וְלַגִּרְסָא דִּלְעֵיל וַחֲכָמִים מוֹדִים לוֹ הָכִי נַמִּי מוֹדִים לוֹ מִדְּקָתָנֵי וְעוֹד. כֶּסֶף מִשְׁנֶה: לֹא תִקֹּם. שֶׁאִם בִּקַּשְׁתָּ מִמֶּנּוּ כְּלִי וְלֹא הִשְׁאִילְךָ וְהִדַּרְתּוֹ הֲנָאָתְךָ, אוֹ שֶׁעָשָׂה לְךָ שֶׁלֹּא כַהֹגֶן וּמִתּוֹךְ שִׂנְאָה הִדַּרְתּוֹ, וְהַתּוֹרָה אָמְרָה לֹא תִשְׂנָא כוּ' הַתּוֹרָה הִזְקִיקָתְךָ לְפַרְנְסוֹ וּלְאָהֳבוֹ וּלְהַחְיוֹתוֹ. רַשִׁ"י: וְאֵין אַתָּה כוּ'. בַּגְּמָרָא פָּרֵיךְ דְּהָא מֻטָּל עַל הַגַּבָּאִים וְזֶה אֵינוֹ אָסוּר, דִּכְשֶׁנּוֹתֵן לַגַּבַּאי שֶׁיִּתֵּן לוֹ הָוֵי כְּנוֹתֵן מַתָּנָה לְאֶחָד כוּ', וּמְשָׁנֵי דְּכָל הַנּוֹפֵל אֵינוֹ נוֹפֵל לְיַד גַּבַּאי תְּחִלָּה, וּפֵרֵשׁ רַשִׁ"י אֶלָּא קְרוֹבָיו מִתְגַּלְגְּלִין עִמּוֹ: וְאֵין אָלָה אֶלָּא שְׁבוּעָה, דִּכְתִיב וַיָּבֵא אֹתוֹ בְּאָלָה, וּכְתִיב וְגַם בַּמֶּלֶך נְבוּכַדְנֶאצַּר מָרָד אֲשֶׁר הִשְׁבִּיעוֹ בֵאלֹקִים . גְּמָרָא. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב:

שְׂעַר רֹאשְׁךָ. הָעִקָּר אֶצְלֵנוּ מִטַּלְטְלֵי לֹא מִשְׁתַּעְבְּדֵי לִכְתֻבָּה אֶלָּא בְּתַקָּנַת בַּתְרָאֵי, וּלְכָךְ בֵּאֵר בַּגְּמָרָא שֶׁאֵלּוּ הַשְּׁמֹנֶה מֵאוֹת זוּז הָיָה לוֹ קַרְקַע שָׁוֶה זֶה הַסָּךְ וְאָמַר לוֹ אֲפִלּוּ לֹא יִשָּׁאֵר לְךָ שׁוּם דָּבָר אַחַר פֵּרְעוֹן הַכְּתֻבָּה אֶלָּא עַל דֶּרֶךְ מָשָׁל שֶׁתִּצְטָרֵךְ לִמְכֹּר שְׂעַר רֹאשְׁךָ לְצֹרֶךְ אֲכִילָה יֵשׁ לְךָ לִתֵּן כְּתֻבָּה. הָרַמְבַּ"ם. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב:

שֶׁמְּחֻיָּב בּוֹ בַּאֲכִילַת בָּשָׂר לְפִי שֶׁנֶּאֱמַר וְקָרָאתָ לַשַּׁבָּת עֹנֶג: מִקְצָתוֹ. עַל יְדֵי חֲרָטָה בְּהַתָּרַת חָכָם, אוֹ מֵעַצְמוֹ כְּגוֹן שֶׁהָיָה אָבִיו אֶחָד מֵהֶם. הָרֹא"שׁ: וּמִיהוּ לִמֵּד לֹא מַשְׁמַע דְּמִסְּבָרָא הוּא. וּבַיְרוּשַׁלְמִי יָלְפִינַן לַהּ מִקְּרָא דִּכְתִיב כְּכָל הַיּוֹצֵא מִפִּיו יַעֲשֶׂה, דְּמַשְׁמַע דַּוְקָא כְשֶׁכֻּלּוֹ קַיָּם. הָרַ"ן:

דְּאִי לֹא תֵימָא הָכִי שֶׁלֹּא אָמַר בְּכָ"ף אֶלָּא כְּלִישְׁנָא דְמַתְנִיתִין בְּוָי"ו, לֹא תְלָאָן זֶה בָּזֶה, אֶלָּא לְרַבָּנָן כָּל אֶחָד הָוֵי נֶדֶר בִּפְנֵי עַצְמוֹ כִּדְמַסִּיק הָרַ"ב בַּסֵּיפָא, וּלְרַבִּי שִׁמְעוֹן כְּלָלָא הָוֵי וְהֻתַּר לְאֶחָד מֻתָּר לְכֻלָּם. וּלְפִי זֶה רֵישָׁא דְּהָכָא רַבָּנָן, דְּאִי כְּרַבִּי שִׁמְעוֹן לְמַאי תְּנַן לְכֻלְּכֶם, לִתְנֵי לִרְבוּתָא שֶׁאֵינִי נֶהֱנֶה לָזֶה וְלָזֶה בְּלֹא כָּ"ף, וְתַבְרָא מִי שֶׁשָּׁנָה כוּ'. וּלְהָרַמְבַּ"ם שִׁיטָה אַחֶרֶת. עַיֵּן תּוֹסְפוֹת יוֹם טוֹב:

יַיִן כוּ'. צְרִיכָא כְּמוֹ שֶׁכָּתַב הָרַמְבַּ"ם, דְּתָנָא נֶדֶר שֶׁהֻתַּר מִקְצָתוֹ כוּ' לְעִנְיַן זְמַן הַנֶּדֶר וּלְעִנְיַן דְּבָרִים שֶׁנִּשְׁבָּעִים עֲלֵיהֶם וּמִן הָאָדָם שֶׁנִּשְׁבָּע עֲלֵיהֶם: יָפֶה. מְפָרֵשׁ בַּגְּמָרָא דַּחֲדָא וְעוֹד קָאָמַר, דְּהָא לֹא נָדַר אֶלָּא מִשּׁוּם שֶׁהוּא רַע וְנִמְצָא שֶׁכְּשֶׁאֵינוֹ רַע אֵין נִדְרוֹ נֶדֶר, אֶלָּא הָכִי נַמִּי קָאָמַר, חֲדָא דְּאֵין רַע וּבְהָכִי שָׁרֵי, וְעוֹד מִפְּנֵי שֶׁהוּא יָפֶה. וְכֵן מְפָרְשִׁינַן בְּבָצָל: וּבְכֻלְּהוּ נְדָרִים שֶׁהֻתְּרוּ מִקְצָתָן דִּינָא הָכִי, וְכֵן בְּנִמְצָא בָּהֶן אָבִיו וְאִמּוֹ דְּפֶרֶק ג', וְטַעֲמָא דִּכְשֶׁעַכְשָׁיו בַּחֲזָרָה עֲדַיִן הוּא אוֹמֵר הַכֹּל אָסוּר חוּץ כוּ' לֹא הוּרַע כֹּחַ הַנֶּדֶר:

בְּנוֹתֶיךָ. כְּלוֹמַר וְלֹא יִמְצְאוּ מִי שֶׁיִּשָּׂא אוֹתָן, דְּאָמְרֵי אֱינָשֵׁי רְחֵלָא בָּתַר רְחֵלָא אָזְלָא, כְּאִמָּהּ כֵּן בִּתָּהּ: מָה רָאֲתָה. וְיֵשׁ לְפָרֵשׁ כְּלוֹמַר מָה רָאֲתָה לַעֲשׂוֹת מַה שֶּׁעָשְׂתָה שֶׁעַל יְדֵי כֵן נִתְגָּרְשָׁה: מֻתָּר. וְצָרִיךְ בְּכָל זֶה הֲפָרָה אַחַר שְׁאֵלָה לְחָכָם. הָרַמְבַּ"ם:

וְאִם תֹּאמַר מַאי שְׁנָא מֵהָאוֹמֵר שֶׁאָבִיהָ רַע וְשָׁמַע שֶׁמֵּת אוֹ עָשָׂה תְשׁוּבָה דִּתְנַן בְּמִשְׁנָה ג' דַּהֲוֵי נֶדֶר טָעוּת, וְיֵשׁ לוֹמַר דְּהַיְנוּ טַעֲמָא מִשּׁוּם דְּאָדָם רַע עָשׂוּי לַעֲשׂוֹת תְּשׁוּבָה וְעָשׂוּי לָמוּת, הִלְכָּךְ אוֹמְדִין דַּעְתּוֹ דְּכִי אָמַר שֶׁאָבִיהָ רַע כָּל זְמַן שֶׁאָבִיהָ רַע קָאָמַר. מַה שֶּׁאֵין כֵּן בְּאִשָּׁה כְעוּרָה שֶׁאֵין עֲשׂוּיָה לִהְיוֹת נָאָה הִלְכָּךְ לֹא הָיָה בְדַעְתּוֹ שֶׁתְּהֵא מֻתֶּרֶת לוֹ לִכְשֶׁתְּהֵא נָאָה. הָרַשְׁבָּ"א זִכְרוֹנוֹ לִבְרָכָה. אִי נַמִּי אִי אָמַר שֶׁהִיא כְעוּרָה הָכִי נַמִּי אֲבָל הָכָא דְּלִפְלוֹנִית כְּעוּרָה קָאָמַר לֹא מַשְׁמַע כְּתוֹלֶה נִדְרוֹ בְּדָבָר. רַבִּי שְׁמוּאֵל בֶּן רַבִּי יְהוּדָה בַּר חִסְדַּאי זִכְרוֹנוֹ לִבְרָכָה: אֶל שָׁאוּל. וְסֵיפֵיהּ הַמַּלְבִּישְׁכֶם שָׁנִי עִם עֲדָנִים כוּ':
