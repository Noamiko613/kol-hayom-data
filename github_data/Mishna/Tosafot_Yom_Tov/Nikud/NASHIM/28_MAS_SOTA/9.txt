פרק ט - משנה א

כִּי יִמָּצֵא. הָכִי קָאָמַר, שֶׁנֶּאֱמַר וְעָנוּ וְאָמְרוּ כוּ', וְסֵדֶר עֶגְלָה עֲרוּפָה כֵּיצַד, כִּי יִמָּצֵא. גְּמָרָא: שְׁלֹשָׁה כוּ'. מְפָרֵשׁ בַּגְּמָרָא זְקֵנֶיךָ שְׁנַיִם, וְאֵין בֵּית דִּין שָׁקוּל, מוֹסִיפִין עֲלֵיהֶם עוֹד אֶחָד, הֲרֵי כָּאן שְׁלֹשָׁה: שְׁנַיִם. וְתַנָּא קַמָּא הַהוּא מִבַּעְיָא לֵיהּ לִמְיֻחָדִין שֶׁבְּשֹׁפְטֶיךָ. וְרַבִּי יְהוּדָה וָי"ו דִּוְשֹׁפְטֶיךָ דָרֵישׁ לְמִנְיָנָא לְהָבִיא עוֹד שְׁנַיִם, וְתַנָּא קַמָּא וָי"ו לֹא דָרֵישׁ. גְּמָרָא. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב:

פרק ט - משנה ב

בֵּית דִּין. דְּבָעִינַן זִקְנֵי הָעִיר וְלֵיכָּא. גְּמָרָא: דִּבְרֵי רַבִּי אֱלִיעֶזֶר. גְּמָרָא, קָסָבַר אִי אֶפְשָׁר לְצַמְצֵם, וּקְרוֹבָה וַאֲפִלּוּ קְרוֹבוֹת, פֵּרֵשׁ רַשִׁ"י כְּמוֹ וּבְהֵמָה רַבָּה :

פרק ט - משנה ג

דְּזֶה אֶחָד מֵעֲשָׂרָה תְּנָאִים שֶׁהִתְנָה יְהוֹשֻׁעַ . רַשִׁ"י:

פרק ט - משנה ד

לְפִי שֶׁהוּא אֶמְצָעוּת הַגּוּף, שֶׁכּוֹלֵל כְּלֵי הַמָּזוֹן שֶׁבָּהֶם נִמְשָׁךְ הַמַּאֲכָל. רַבִּי עֲקִיבָא אוֹמֵר מְקוֹם שְׁאִיפַת הָרוּחַ הוּא הַצֹּרֶךְ הַגָּדוֹל בְּחַיּוּת, לְפִי שֶׁנֶּאֱמַר כָּל וְכוּ'. הָרַמְבַּ"ם:

פרק ט - משנה ה

וְאֵין הַמּוּם כוּ'. דְּמַהוּ דְּתֵימָא נֵילַף מִקַּל וָחֹמֶר וּמַה פָּרָה כוּ', קָא מַשְׁמַע לָן דְּלֹא, דְּשָׁאנֵי הָתָם דְּאָמַר קְרָא אֲשֶׁר אֵין בָּהּ מוּם, בָּהּ מוּם פּוֹסֵל וְאֵין מוּם פּוֹסֵל בָּעֶגְלָה. כָּךְ לָמַדְתִּי מֵהַגְּמָרָא: קָשֶׁה. פֵּרֵשׁ הָרַמְבַּ"ם חָזָק הַהַמְשָׁכָה. וְכֵן כָּתַב בְּחִבּוּרוֹ, נַחַל שֶׁשּׁוֹטֵף בְּחָזְקָה, וְזֶהוּ אֵיתָן הָאָמוּר בַּתּוֹרָה. אֲבָל רַשִׁ"י מְפָרֵשׁ נַחַל אֵיתָן קָשֶׁה, שֶׁלֹּא נֶעֱבַד. לִדְבָרָיו נַחַל הָאָמוּר כָּאן אֵינוֹ שֶׁל מַיִם אֶלָּא עֵמֶק. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: וּקְצָת קָשֶׁה, דַּהֲוָה לֵיהּ לְמֵימַר וְאִם אֵינוֹ אֵיתָן כָּשֵׁר, שֶׁזֶּה לָשׁוֹן דִּיעֲבַד וְלָשׁוֹן קָצָר כוּ'. וְהָרַמְבַּ"ם פֵּרֵשׁ עִנְיָן אֵיתָן גָּדוֹל, כְּמוֹ שֶׁנֶּאֱמַר גּוֹי אֵיתָן הוּא , וְאוֹמֵר אִם הָיָה חָזָק הַהַמְשָׁכָה וַאֲפִלּוּ שֶׁלֹּא הָיָה גְּדוֹל הַשִּׁעוּר כָּשֵׁר. בָּא לוֹמַר שֶׁאֵיתָן יֵשׁ בּוֹ שְׁנֵי פֵּרוּשִׁים, גָּדוֹל וְחָזָק, וְשֶׁאָמַר הַתַּנָּא דִּקְפֵדַת הַכָּתוּב עַל הַחֹזֶק וְאַף עַל פִּי שֶׁאֵינוֹ גָּדוֹל. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: דְּגָמַר עֲרִיפָה עֲרִיפָה מֵחַטַּאת הָעוֹף. גְּמָרָא:

פרק ט - משנה ו

בִּמְקוֹם. בַּגְּמָרָא יָלֵיף לֵיהּ: בְּלֹא מָזוֹן. וְהֻצְרַךְ לְלַסְטֵם אֶת הַבְּרִיּוֹת וּלְכָךְ נֶהֱרַג. רַשִׁ"י: בְּלֹא לְוָיָה. יְחִידִי בְּלֹא חֲבוּרָה. רַשִׁ"י: אוֹמְרִים. כְּדִכְתִיב לְעֵיל וְנִגְּשׁוּ הַכֹּהֲנִים בְּנֵי לֵוִי, וְלֹא פֵּרֵשׁ עַל מָה הֵם נִגָּשִׁים. רַשִׁ"י: לֹא הָיוּ כוּ'. כְּלוֹמַר וְנִכַּפֵּר דִּכְתִיב בַּקְּרָא לָאו אַוְעָנוּ וְאָמְרוּ קָאֵי שֶׁיְּהֵא אַף זֶה בִּכְלַל אֲמִירָה, אֶלָּא מִלְּתָא בְּאַנְפֵּי נַפְשָׁהּ הִיא שֶׁרוּחַ הַקֹּדֶשׁ בִּשְּׂרָהּ כוּ'. רַשִׁ"י:

פרק ט - משנה ז

וְתִרְעֶה. פֵּרֵשׁ רַשִׁ"י דִּסְבִירָא לֵיהּ הַאי תַּנָּא דְּאֵין יְרִידָתָהּ לְנַחַל אֵיתָן אוֹסַרְתָּהּ אֶלָּא עֲרִיפָתָהּ. וְהָרַמְבַּ"ם פָּסַק דִּירִידָתָהּ אוֹסַרְתָּהּ בַּהֲנָאָה, וַאֲפִלּוּ הָכִי תֵּצֵא כוּ' דְּאַדַּעְתָּא דְּהָכִי לֹא אַקְדְּשָׁהּ. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב:

פרק ט - משנה ט

כִּדְתָנוּ רַבָּנָן לֹא נוֹדַע מִי הִכָּהוּ, הָא נוֹדַע מִי הִכָּהוּ אֲפִלּוּ אֶחָד בְּסוֹף הָעוֹלָם הִכִּיר בּוֹ לֹא הָיוּ עוֹרְפִין: מִשֶּׁבָּא כוּ'. וְרַצְחָן מְפֻרְסָם הָיָה: בֶּן הָרַצְחָן. בְּרִיָּה קָטִילָא. יְרוּשַׁלְמִי: לֹא אֶפְקוֹד כוּ'. תָּנוּ רַבָּנָן וְנִקָּה הָאִישׁ מֵעָוֹן, בִּזְמַן שֶׁהָאִישׁ מְנֻקֶּה מֵעָוֹן הַמַּיִם בּוֹדְקִין אֶת אִשְׁתּוֹ, וְאוֹמֵר לֹא אֶפְקוֹד וְגוֹ', מַאי וְאוֹמֵר, וְכִי תֵימָא עָוֹן דִּידֵיה אִין, דְּבָנֵיה וְדִבְנָתֵיהּ לֹא, תָּא שְׁמַע לֹא אֶפְקוֹד וְגוֹ', וְכִי תֵימָא עָוֹן אֵשֶׁת אִישׁ אִין עָוֹן דִּפְנוּיָה לֹא, תָּא שְׁמַע כִּי הֵם עִם הַזּוֹנוֹת יִפָּרֵדוּ , הֵם עַצְמָם נוֹאֲפִים כִּפְרָדִים הַלָּלוּ. רַשִׁ"י: וְשִׁכְחָה וּמַחֲלֹקֶת. וְעַד יְמֵיהֶן לֹא הָיָה מַחֲלֹקֶת בְּחַכְמֵי יִשְׂרָאֵל, כֻּלָּן הָיוּ אוֹמְרִים דְּבָרִים כִּנְתִינָתָן לְמֹשֶׁה מִסִּינַי. וְהֵן הָרִאשׁוֹנִים שֶׁנֶּחְלְקוּ בִּסְמִיכַת קָרְבְּנוֹת יוֹם טוֹב כִּדְאִיתָא בַּחֲגִיגָה, וְהוּא מַחֲלֹקֶת רִאשׁוֹן שֶׁהָיָה בְּיִשְׂרָאֵל בְּדִבְרֵי תּוֹרָה. רַשִׁ"י: אֵין אֶשְׁכּוֹל וְגוֹ'. לָרְאָיָה שֶׁאֶשְׁכּוֹל נֶאֱמַר לְכִנּוּי אָדָם הֶחָשׁוּב בְּעֵינֵי אֱלֹהִים יִתְבָּרַךְ:

פרק ט - משנה יא

הַשִּׁיר כוּ'. פֵּרְשׁוּ הַפּוֹסְקִים בְּשֵׁם רַב הַאי גָּאוֹן זִכְרוֹנוֹ לִבְרָכָה דְּהַיְנוּ שִׁיר שֶׁל אַהֲבָה שֶׁבֵּין אָדָם לַחֲבֵרוֹ וּלְשַׁבֵּחַ יָפֶה בְּיָפְיוֹ. אֲבָל שִׁירוֹת וְתִשְׁבָּחוֹת וְזִכְרוֹן חֲסָדִים שֶׁל הַקָּדוֹשׁ בָּרוּךְ הוּא מִנְהָג כָּל יִשְׂרָאֵל לְאָמְרָן בְּבֵית חֲתֻנָּה וּבְבֵית הַמִּשְׁתָּאוֹת בְּקוֹל נְגִינוֹת וּבְקוֹל שִׂמְחָה: מִשּׁוּם דִּמְנָלָן דְּמִשֶּׁבָּטְלָה סַנְהֶדְרִין כְּתִיב כִּדְאִיתָא בַּגְּמָרָא, וּמִשּׁוּם דִּבְהַאי קְרָא לֹא כְּתִיב בֵּיהּ דְּאָסוּר אֶלָּא מַשְׁמַע דְּנִשְׁבַּת מֵאֵלָיו, לְהָכִי מַיְתֵי מַתְנִיתִין קְרָא בַּשִּׁיר לֹא יִשְׁתּוּ, דִּלְמֵיסַר מַשְׁמַע: וְהֵם הַזְּקֵנִים שֶׁאָמַר הַקָּדוֹשׁ בָּרוּךְ הוּא עֲלֵיהֶן תִּתֵּן לְךָ בְּכָל שְׁעָרֶיךָ. הָרַמְבַּ"ם:

פרק ט - משנה יב

וּמִדִּקְדּוּק הַלָּשׁוֹן שֵׁם הָעֶצֶם לֹא יִתְרַבֶּה, וְאִם כֵּן הָאֵיךְ אָמְרוּ צוּפִים. וְנִרְאֶה לְפָרֵשׁ לְדִבְרֵי הָרַ"ב שֶׁהַמָּקוֹם הוּא נִקְרָא צוֹפִים בִּלְשׁוֹן מִקְרָא, כְּמָה דְּאַתְּ אָמַר הָרָמָתַיִם צוֹפִים , וְכֵן נִקְרָא בִּימֵי הַתַּנָּאִים, אֶלָּא שֶׁנִּשְׁתַּבֵּשׁ אַחֲרֵי כֵן בְּפִי הֶהָמוֹן וְהָיוּ קוֹרְאִין אוֹתָהּ צוֹפִיאָה. וּלְשׁוֹן הָרַמְבַּ"ם, דְּבַשׁ הַבָּא מֵאֶרֶץ יִשְׂרָאֵל נִקְרָא צוּפִים. מַשְׁמַע שֶׁרְצוֹנוֹ לוֹמַר שֶׁהַדְּבַשׁ נִקְרָא צוּפִים. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: לְוַתֵּר מָמוֹנָם לְנוֹי הִדּוּר מִצְוָה וְלִצְדָקָה וּלְהוֹצָאַת שַׁבָּתוֹת וְיָמִים טוֹבִים. רַשִׁ"י: גָּמַר. וְסֵיפֵיהּ כִּי פַסּוּ אֱמוּנִים מִבְּנֵי אָדָם. וְאַף עַל גַּב דְּדָוִד אֲמָרוֹ, מָצִינוּ לוֹ כַּמָּה מִזְמוֹרִים שֶׁאֲמָרָן עַל הֶעָתִיד:

פרק ט - משנה יג

וְסֵיפֵיהּ דִּקְרָא וְכָל חֵלֶב תִּירוֹשׁ וְדָגָן. וְאַף עַל גַּב דִּקְרָא בִּתְרוּמָה מִשְׁתָּעֵי, הֻשְׁאַל שֵׁם מַעַשְׂרוֹת לְכָל הַמַּתָּנוֹת. וְנָקַט דָּגָן לְפִי שֶׁהוּא צֹרֶךְ לִבְנֵי אָדָם יוֹתֵר מֵהַכֹּל, וְהוּא הַדִּין לְתִירוֹשׁ וְיִצְהָר:

פרק ט - משנה יד
