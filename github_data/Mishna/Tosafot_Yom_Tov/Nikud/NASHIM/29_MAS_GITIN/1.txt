פרק א

גֵּט. לְשׁוֹן אֲרַמִּי. סֵפֶר כְּרִיתוּת (דְּבָרִים כד) מְתֻרְגָּם גֵּט פִּטּוּרִין. וּלְכָךְ נָהֲגוּ לִכְתֹּב בּוֹ שְׁתֵּים עֶשְׂרֵה שׁוּרוֹת כְּמִנְיַן גֵּט. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: וְכֵן פָּסַק רַשִׁ"י. וְהָרַ"ן פֵּרֵשׁ דְּחוּץ לָאָרֶץ שֶׁל צַד מַעֲרָבִי שֶׁל אֶרֶץ יִשְׂרָאֵל נָקַט, לְפִי שֶׁבְּאוֹתוֹ צַד לֵיכָּא עֲיָרוֹת מֻבְלָעוֹת וּסְמוּכוֹת שֶׁהֲרֵי הַיָּם מַפְסִיק, וּבִצְדָדִים אֲחֵרִים לֹא פְּסִיקָא לֵיהּ מִלְּתָא מִשּׁוּם דְּאִיכָּא עֲיָרוֹת סְמוּכוֹת וּמֻבְלָעוֹת (וּלְקַמָּן אָמְרִינַן דְּתַנָּא קַמָּא מַפִּיק אֵלּוּ מִכְּלָל). וּמִשּׁוּם הָכִי אָמַר מִמְּדִינַת הַיָּם שֶׁהִיא מַעֲרַב אֶרֶץ יִשְׂרָאֵל, וְהוּא הַדִּין לְכָל צְדָדִים אֲחֵרִים מֵאוֹתָן עֲיָרוֹת שֶׁהֵן דּוֹמוֹת לְאֵלּוּ שֶׁבִּמְדִינַת הַיָּם, שֶׁאֵינָן לֹא מֻבְלָעוֹת וְלֹא סְמוּכוֹת. וְאֶפְשָׁר שֶׁלָּזֶה כִּוֵּן רַשִׁ"י זִכְרוֹנוֹ לִבְרָכָה בְּמַה שֶּׁסִּיֵּם, כְּלוֹמַר כָּל חוּץ לָאָרֶץ שֶׁהִיא דוֹמָה לְחוּץ לָאָרֶץ שֶׁבִּמְדִינַת הַיָּם: וּקְרָא כְּתִיב וְכָתַב לָהּ, דְּהַיְנוּ לִשְׁמָהּ. וְאֵין בְּקִיאִין הַיְנוּ דְּאֵין חוֹשְׁשִׁין לְהַךְ דְּרָשָׁה, דְּאִי לָאו הָכִי מַאי שְׁנָא מִשְּׁאָר הִלְכוֹת גִּטִּין דְּלֹא חָיְשִׁינַן לְהוּ דְּלֹא בְּקִיאֵי. תּוֹסָפוֹת: וְרַשִׁ"י פֵּרֵשׁ דִּסְתָמָא כִּי מַסְהִיד בְּפָנַי נִכְתַּב וּבְפָנַי נֶחְתַּם לִשְׁמָהּ קָמַסְהִיד. וְסִיֵּם הָרַ"ן דְּמֵימַר אָמַר הַאי שָׁלִיחַ לָמָּה לְהוּ לְרַבָּנָן לְמִשְׁאָל אִם נִכְתַּב וְנֶחְתַּם בְּפָנַי, שְׁמַע מִנַּהּ שֶׁיֵּשׁ כְּתִיבָה וַחֲתִימָה בְּגֵט שֶׁהִיא פְסוּלָה, וְאִלּוּ לֹא נִכְתַּב לִשְׁמָהּ הָיָה אוֹמֵר הַשָּׁלִיחַ כָּךְ אֵרַע מַעֲשֶׂה דִלְאַחֵר נִכְתַּב. וְהָא דְסַגִּי בְּעֵד אֶחָד, מִשּׁוּם דְּרֹב בְּקִיאִין הֵן. וַאֲפִלּוּ לְרַבִּי מֵאִיר דְּחָיֵישׁ לְמִעוּטָא, הָכָא הָוֵי מִעוּטָא דְמִעוּטָא, וְרַבָּנָן הוּא דְאַצְרוּךְ, וּמִשּׁוּם עִגּוּנָא הֵקֵלּוּ. גְּמָרָא. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: וּפָרֵיךְ בַּגְּמָרָא אִם כֵּן בְּפָנַי נִכְתַּב לָמָּה לִי, וּמְשָׁנֵי דִּלְמָא אָתֵי לְאַחְלוֹפֵי בְּקִיּוּם שְׁטָרוֹת דְּעָלְמָא בְּעֵד אֶחָד: דְּעֵדִים הַחֲתוּמִים עַל הַשְּׁטָר כְּמוֹ שֶׁנֶּחְקְרָה עֵדוּתָן בְּבֵית דִּין, וְרַבָּנָן הוּא דְאַצְרוּךְ, וּמִשּׁוּם עִגּוּנָא הֵקֵלּוּ. גְּמָרָא: הַחֶגֶר. בַּגְּמָרָא מְפָרֵשׁ דִּבְהָא פְלִיגֵי, דְתַנָּא קַמָּא סָבַר דְּמֵרֶקֶם וְחֶגֶר אַף עַל פִּי שֶׁהֵן מִחוּץ לָאָרֶץ אֵין צָרִיךְ שֶׁיֹּאמַר בְּפָנַי נִכְתַּב וּבְפָנַי נֶחְתַּם, לְפִי שֶׁהֵן עֲיָרוֹת הַסְּמוּכוֹת לְאֶרֶץ יִשְׂרָאֵל, וְכָל שֶׁכֵּן מִכְּפַר לוּדִים לְלוֹד שֶׁאֵין צָרִיךְ, לְפִי שֶׁהוּא מֻבְלָע בִּתְחוּם אֶרֶץ יִשְׂרָאֵל כוּ'. וְאָתֵי רַבָּן גַּמְלִיאֵל לָאַפְלוֹגֵי עֲלַהּ וּלְמֵימַר דְּדַוְקָא בְּמֻבְלָעוֹת כְּמוֹ מִכְּפַר לוּדִים גְּמִירֵי וּשְׁכִיחֵי, אֲבָל סְמוּכוֹת כְּגוֹן רֶקֶם לֹא גְמִירֵי וְלֹא שְׁכִיחֵי. וַאֲתָא רַבִּי אֱלִיעֶזֶר לְמֵימַר דַּאֲפִלּוּ מִכְּפַר לוּדִים שֶׁהוּא מֻבְלָע צָרִיךְ לוֹמַר בְּפָנַי נִכְתַּב וּבְפָנַי נֶחְתַּם, דִּנְהִי דְמוֹדֶה לְרַבָּנָן קַמָּאֵי דְמֻבְלָעִים גְּמִירֵי וּשְׁכִיחֵי, אֲפִלּוּ הָכִי מַצְרִיךְ בְּהוּ לוֹמַר בְּפָנַי נִכְתַּב וּבְפָנַי נֶחְתַּם, כְּדֵי שֶׁלֹּא תַחְלֹק בִּמְדִינַת הַיָּם. הָרַ"ן: מִכְּפַר כוּ'. מִקְרֵי כְּפַר לוּדִים, שֶׁבְּנֵי לוֹד מְצוּיִין בָּהּ תָּמִיד. וּמִשּׁוּם הָכִי נָקַט לְלוֹד לִרְבוּתָא. הָרַ"ן: וְהַמּוֹלִיךְ. וּבְהָא פְלִיגֵי אַתַּנָּא קַמָּא, דִּלְתַנָּא קַמָּא הַמּוֹלִיךְ מֵאֶרֶץ יִשְׂרָאֵל לְחוּץ לָאָרֶץ אֵין צָרִיךְ, לְפִי שֶׁבְּנֵי אֶרֶץ יִשְׂרָאֵל בְּקִיאִין לִשְׁמָהּ, וַחֲכָמִים גָּזְרוּ מוֹלִיךְ אַטּוּ מֵבִיא. וּלְמַאן דְּאָמַר טַעֲמָא מִשּׁוּם קִיּוּם, חֲכָמִים לִפְרוּשֵׁי מִלְּתָא דְּתַנָּא קַמָּא אָתוּ דְּמֵבִיא לָאו דַּוְקָא, אֶלָּא הוּא הַדִּין מוֹלִיךְ. גְּמָרָא: בִּמְדִינַת הַיָּם. לְטַעֲמָא דְאֵין בְּקִיאִין הוּא הַדִּין בְּאוֹתָהּ מְדִינָה נַמִּי צָרִיךְ שֶׁיֹּאמַר, אֶלָּא דְּנָקַט מִמְּדִינָה לִמְדִינָה לְמֵידַק מִנַּהּ דְּהָא מִמְּדִינָה לִמְדִינָה בְּאֶרֶץ יִשְׂרָאֵל לֹא צָרִיךְ, וַאֲפִלּוּ לְכַתְּחִלָּה. וּלְאִידָךְ טַעֲמָא נַמִּי בְּאֶרֶץ יִשְׂרָאֵל דִּקְבִיעֵי בֵית דִּין שְׁכִיחֵי שַׁיָּרוֹת. גְּמָרָא: הֶגְמוֹנְיָא. פְּקִידוּת: גְּמָרָא. וּפֵרֵשׁ רַשִׁ"י שֶׁלֹּא יֵלֵךְ מִזּוֹ לְזוֹ. וּבְאֶרֶץ יִשְׂרָאֵל נַמִּי לְטַעֲמָא דְקִיּוּם, דְּמַקְפִּידִין. וּלְמַאן דְּאָמַר לִשְׁמָהּ נַמִּי, דְּאִיהוּ סְבִירָא לֵיהּ נַמִּי טַעֲמָא דְקִיּוּם, דְּאִלּוּ בְּחוּץ לָאָרֶץ אֲפִלּוּ בְּאוֹתָהּ הֶגְמוֹנְיָא כִדְאִיתָא בַּגְּמָרָא:

לַמִּזְרָח. אֲבָל מַעֲרַב אֶרֶץ יִשְׂרָאֵל אֵין צָרִיךְ גְּבוּל, שֶׁהַיָּם הַגָּדוֹל גְּבוּלָהּ, דִּכְתִיב וּגְבוּל יָם וְהָיָה לָכֶם הַיָּם הַגָּדוֹל . רַשִׁ"י. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: וְאַשְׁקְלוֹן. וּבְסֵפֶר יְהוֹשֻׁעַ מוּכָח דְּמֵאֶרֶץ יִשְׂרָאֵל הִיא. גַּם בַּסִּפְרִי חָשִׁיב פָּרָשַׁת אַשְׁקְלוֹן מִכִּבּוּשׁ עוֹלֵי בָּבֶל. וְהָרַ"ן תֵּרֵץ דְּפָרָשַׁת אַשְׁקְלוֹן אֵינוֹ אֶלָּא פָּרָשַׁת דְּרָכִים שֶׁלָּהּ, וְשֶׁכֵּן הוּא בַּיְרוּשַׁלְמִי:

אֵינוֹ צָרִיךְ כוּ'. דְּמִשּׁוּם עִגּוּנָא הֵקֵלּוּ וְלֹא הִצְרִיכוּ קִיּוּם כָּל שֶׁלֹּא עִרְעֵר. תּוֹסָפוֹת: דְּאִי לֹא תֵּימַהּ הָכִי הָא קָיְמָא לָן (כְּתֻבּוֹת פ"ב מ"ח) דְּאֵין עִרְעוּר פָּחוֹת מִשְּׁנַיִם. וְאֶלָּא תְּרֵי, תְּרֵי וּתְרֵי נִינְהוּ, מַה חָזִית דִּסְמוֹךְ אַהָנֵי, סְמוֹךְ אַהָנֵי, וְאוֹקְמַהּ בְּחֶזְקַת אֵשֶׁת אִישׁ. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: מִשּׁוּם דְּלֹא קְבִיעֵי בֵית דִּין וְלֹא שְׁכִיחֵי שַׁיָּרוֹת: מִשּׁוּם דְּקָיְמָא לָן כְּמַאן דְּאָמַר דְּטַעְמָא מִשּׁוּם קִיּוּם הוּא: דִּלְאַחַר נְתִינָה תּוֹךְ כְּדֵי דִבּוּר מָצֵי לְמֵימַר בְּפָנַי נִכְתַּב וּבְפָנַי נֶחְתָּם. הָרַ"ן: עֵדִים. דְּקָיְמָא לָן עֵדֵי מְסִירָה כָרְתֵי: יִתְקַיֵּם. מְפָרֵשׁ בַּגְּמָרָא לְמַאן דְּאָמַר דְּלֹא בְקִיאֵי כוּ' דְּהָכָא לְאַחַר שֶׁלָּמְדוּ, כְּלוֹמַר לְאַחַר שֶׁקִּבְּלוּהָ לַדְּרָשָׁא דְּלִשְׁמָהּ. וַאֲפִלּוּ הָכִי בְּיָכוֹל צָרִיךְ לוֹמַר בְּפָנַי נִכְתַּב וּבְפָנַי נֶחְתָּם, דְּלֹא עָקְרוּ לְתַקַּנְתַּיְהוּ כוּ'. אֲבָל פִּקֵּחַ וְנִתְחָרֵשׁ מִלְּתָא דְלָא שְׁכִיחַ הִיא וְלֹא גָזְרוּ בָּהּ רַבָּנָן:

דִּכְתִיב אוֹ חֻפְשָׁה לֹא נִתַּן לָהּ. עַד שֶׁיִּכְתֹּב לִשְׁמָהּ. הָרַמְבַּ"ם. וּלְמַאן דְּאָמַר מִשּׁוּם קִיּוּם וַדַּאי דְבָעֵי קִיּוּם, מִידֵי דַהֲוָה אַשְּׁאַר שְׁטָר. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: אַחַד כוּ'. לוֹמַר שֶׁיֵּשׁ עוֹד דְּרָכִים אֲחֵרִים שֶׁשָּׁוִים וּכְדִלְקַמָּן: גִּטֵּי נָשִׁים. אִיפְּכָא הֲוָה לֵיהּ לְמֵימַר, שֶׁשָּׁווּ שִׁחְרוּרֵי כוּ', דְּעִקַּר דִּין מוֹלִיךְ וּמֵבִיא הָוֵי בְּאִשָּׁה, אֶלָּא דְאַגַּב דְּעַד הַשְׁתָּא אַיְרֵי בְּגִטִּין נַקְטֵיהּ בָּרֵישָׁא. תּוֹסָפוֹת:

כָּל גֵּט. כָּל שְׁטָר קָרוּי גֵט. הָרַ"ב: מְפָרֵשׁ בַּגְּמָרָא כְּגוֹן דְּחָתַם יִשְׂרָאֵל לְבַסּוֹף, כְּלוֹמַר אַחֲרֵי הַכּוּתִי, דְאִי לָאו דְּכוּתִי חָבֵר הוּא לֹא מַחְתִּים לֵיהּ מִקַּמֵּיהּ, דְּאֵין עֵדֵי גֵּט חוֹתְמִין זֶה בְלֹא זֶה, גְּזֵרָה מִשּׁוּם הָאוֹמֵר כֻּלְּכֶם. אֲבָל שְׁאָר שְׁטָרוֹת דְּחָתְמֵי זֶה בְלֹא זֶה אֵימַר רַוְחָא שָׁבֵיק לְמַאן דְּקַשִּׁישׁ מִנֵּיהּ, וַאֲתָא כּוּתִי קָתָנֵי, וְרַבָּן גַּמְלִיאֵל מַכְשִׁיר בִּשְׁנַיִם, וּמַעֲשֶׂה נַמִּי כוּ'. גְּמָרָא. וּלְדִידֵיהּ דְּאַחְזוּק בְּהָא, כְּלוֹמַר בְּגִטִּין וְשִׁחְרוּרִין, וְלֹא אַחְזוּק בְּהָא, בִּשְׁאָר שְׁטָרוֹת. וְשִׁטְרֵי הַלְוָאוֹת לֹא אִתְּמַר בַּגְּמָרָא, וְדַעַת הָרַ"מ לְהַשְׁווֹתָן לְשִׁטְרֵי מֶכֶר, וּפֵרֵשׁ הַמַּגִּיד שֶׁכֵּיוַן שֶׁעִקַּר הַשִּׁעְבּוּד חָל מַחֲמַת קַבָּלַת הַמָּעוֹת, לְעִנְיָן שֶׁלֹּא כוּ'. דְּיָכוֹל לוֹמַר לְעַבְדּוֹ עֲשֵׂה עִמִּי וְאֵינִי זָנְךָ. גְּמָרָא. וּפֵרֵשׁ רַשִׁ"י אֶלָּא חֲזֹר עַל הַפְּתָחִים. וּמְסַיֵּם הָרַמְבַּ"ם, שֶׁיִּשְׂרָאֵלִים מְצֻוִּים לְהַחֲיוֹת עֲבָדִים שֶׁבֵּינֵיהֶם: הַתְּרוּמָה. וְעֶבֶד יִשְׂרָאֵל מַפְסִידוֹ מִשִּׁפְחָה כְנַעֲנִית. וְאַף עַל גַּב דְּמַתִּירוֹ בְּבַת חוֹרִין, עַבְדָּא בְּהֶפְקֵרָא נִיחָא לֵיהּ זִילָא לֵיהּ לְזַלְזֵל בָּהּ לְכָל תַּאֲוָתוֹ וּמִתְנַהֶגֶת עִמּוֹ בִּפְרִיצוּת. וְרַבָּנָן, הָא דְמַתִּירוֹ בְּבַת חוֹרִין עֲדִיפָא לְהוּ. גְּמָרָא: וּקְצָת קָשֶׁה, דְּאַף עַל גַּב דִּבְהֵמָה נַמִּי אוֹכֶלֶת, מִכָּל מָקוֹם כֵּיוָן שֶׁאִם יִהְיֶה בֶּן חוֹרִין לֹא יֹאכַל, הָוֵי זְכוּת מַה שֶּׁאוֹכֵל עַכְשָׁיו. וּבַגְּמָרָא הָכִי אִיתָא מִפְּנֵי כוּ', דְּאִי בָּעֵי שָׁקִיל אַרְבָּעָה זוּזִי מִיִּשְׂרָאֵל וּפָסֵיל לֵיהּ כָּל הֵיכִי דְּאִיתֵיהּ. תּוֹסְפוֹת יוֹם טוֹב. וְעַיֵּן עוֹד: תְּנוּ. גָּרְסִינַן וְלֹא גָרְסִינַן תֵּן. שֶׁלֹּא מְסָרוֹ לְיַד שָׁלִיחַ בְּחַיָּיו, לְפִיכָךְ לֹא נֶחְלְקוּ חֲכָמִים בַּדָּבָר לוֹמַר שֶׁמִּשָּׁעָה רִאשׁוֹנָה זָכָה לוֹ הַשָּׁלִיחַ לָעֶבֶד לִהְיוֹת מְשֻׁחְרָר. רַשִׁ"י: וְלֹא אָמְרִינַן שֶׁמָּא מָנֶה קָבוּר בַּקַּרְקַע יֵשׁ לוֹ וְהַהוּא דְיָהִיב לֵיהּ:
