פרק ב

וַאֲפִלּוּ לְמַאן דְּאָמַר מִשּׁוּם קִיּוּם, וְצָרִיךְ לוֹמַר בְּפָנַי נִכְתַּב מִשּׁוּם דִּלְמָא אָתֵי לְאַחְלוֹפֵי, בְּחֶצְיוֹ הָרִאשׁוֹן שֶׁיֵּשׁ בּוֹ שֵׁם כוּ' שֶׁהוּא עִקַּר הַגֵּט מִנְכַּר מִלְּתָא וְלֹא אָתֵי לְאַחְלוֹפֵי. תּוֹסָפוֹת: וְאֵין צָרִיךְ שֶׁיִּהְיוּ שְׁנֵיהֶם אֲדוּקִים וַחֲתוּמִים בַּגֵּט, אֶלָּא שֶׁשְּׁנֵיהֶם אוֹמְרִים שֶׁהַבַּעַל עָשָׂה שְׁנֵיהֶם שְׁלוּחִים עַל כָּךְ וְאָמַר לָהֶם שְׁנֵיכֶם הוֹלִיכוּ גֵט לְאִשְׁתִּי, וּנְתָנוֹ הָאֶחָד בְּמַעֲמַד חֲבֵרוֹ כוּ'. תּוֹסָפוֹת: לְמַאן דְּאָמַר מִשּׁוּם קִיּוּם הָאִיכָּא שְׁנַיִם שֶׁמְּצוּיִין לְקַיְּמוֹ, וּלְמַאן דְּאָמַר מִשּׁוּם לִשְׁמָהּ, מַיְרֵי לְאַחַר שֶׁלָּמְדוּ, וּבִתְרֵי לֹא שְׁכִיחֵי, הִלְכָּךְ לֹא גָזְרִינַן שֶׁמָּא יַחֲזֹר הַדָּבָר לְקִלְקוּלוֹ: וְאֶחָד. שְׁלִישִׁי. רַשִׁ"י: דְּכֵיוָן דְּהַהוּא דְאָמַר בְּפָנַי נֶחְתַּם לֹא אָמַר בְּפָנַי נִכְתַּב, אָתֵי לְאַחְלוֹפֵי בְּקִיּוּם כוּ'. וּמַאן דְּאָמַר לִשְׁמָהּ, הָא סְבִירָא לֵיהּ נַמִּי טַעֲמָא דְּקִיּוּם. גְּמָרָא: מַכְשִׁיר. כֵּיוָן דְּאִיכָּא תְּרֵי דִמְסַהֲדִי אַכְּתִיבָה, תּוּ לֹא אָתֵי לְאַחְלוֹפֵי. גְּמָרָא: כָּשֵׁר. לְמַאן דִּסְבִירָא לֵיהּ טַעֲמָא דְקִיּוּם, אֲפִלּוּ אֵין שָׁם עֵד מֵעִיד עַל הַכְּתִיבָה כָּשֵׁר, שֶׁהֲרֵי נִתְקַיֵּם עַל פִּי שְׁנַיִם. וְלֹא נָקַט אֶחָד אוֹמֵר בְּפָנַי נִכְתַּב אֶלָּא אַגַּב רֵישָׁא דְתָנָא אֶחָד אוֹמֵר כוּ'. תּוֹסָפוֹת. וַאֲפִלּוּ יוֹצֵא מִתַּחַת יַד אֶחָד נַמִּי כָּשֵׁר. וּלְטַעֲמָא דְלִשְׁמָהּ דַּוְקָא בְּיוֹצֵא מִתַּחַת יְדֵי שְׁנֵיהֶם. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב:

וְנִרְאֶה לִי דְּנָקַט בַּת אֲחוֹתוֹ לְפִי שֶׁהִיא בַּת גִּילוֹ, כִּדְאִיתָא בְּסוֹף פֶּרֶק ה' דִּנְדָרִים. וּבַיְרוּשַׁלְמִי מַעֲשֶׂה בְּאֶחָד שֶׁנָּשָׂא בַת אֲחוֹתוֹ וְזִנְּתָה כוּ'. וּלְפִי זֶה יֵשׁ לוֹמַר דְּמִשּׁוּם מַעֲשֶׂה שֶׁהָיָה נָקַט בַּת אֲחוֹתוֹ: דְּלֹא אָמְרִינַן אוֹקְמַהּ אַחֶזְקַת אֵשֶׁת אִישׁ וְהַשְׁתָּא הוּא דְאִגָּרְשָׁה, כֵּיוָן שֶׁהִיא גְרוּשָׁה לְפָנֵינוּ. וְעוֹד אַדְּרַבָּה אוֹקְמַהּ בְּחֶזְקַת כְּשֵׁרָה שֶׁלֹּא נִבְעֲלָה כְשֶׁהִיא אֵשֶׁת אִישׁ. תּוֹסָפוֹת. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: דְּאִי מִשּׁוּם זְנוּת, לֹא שְׁכִיחַ. וְרַבָּנָן סָבְרֵי דְמִשּׁוּם פֵּירֵי לֹא הֻצְרְכוּ לְתַקֵּן זְמַן, דִּכְשֶׁאֵין בּוֹ זְמַן נַמִּי כְּשֶׁיָּבֹא הַגֵּט לְיַד הָאִשָּׁה יְכוֹלָה לָבֹא לְבֵית דִּין אוֹ לִפְנֵי עֵדִים וְתַרְאֶה לָהֶם גִּטָּהּ וְיִכְתְּבוּ לָהּ שֶׁמֵּאוֹתוֹ הַיּוֹם נִתְגָּרְשָׁה. תּוֹסָפוֹת:

שֶׁל קְיָמָא. לַאֲתוֹיֵי מֵי טַרְיָא (פֵּרוּשׁ גְּשָׁמִים שֶׁנִּשְׁאָרִים עַל הַגַּגּוֹת. לִישָׁנָא אַחֲרִינָא מַיִם שֶׁשּׁוֹרִין בָּהֶן פֵּירֵי) וְעַפְצָא: שֶׁל זַיִת. דַּוְקָא עָלֶה שֶׁל זַיִת וְכַיּוֹצֵא בּוֹ דְּמִתְקַיֵּם כוּ', אֲבָל שֶׁל כְּרֵשִׁין וּבְצָלִים וַעֲלֵה וְרָדִין וִירָקוֹת וְכָל דָּבָר שֶׁאֵינוֹ שֶׁל קְיָמָא פָּסוּל. תּוֹסָפוֹת: אַף עַל גַּב דִּבְמִשְׁנָה ד' פֶּרֶק ב' דְּסוֹטָה לְעִנְיַן מְגִלַּת סוֹטָה דִכְתִיב בַּסֵּפֶר לֹא דָרְשִׁינַן הָכִי, לֹא קָשְׁיָא, דְּהָכָא שָׁאנֵי, דִּכְלָל פְּרָט וּכְלָל דָּרֵישׁ, דִּוְנָתַן חָזַר וְכָלַל, אֲבָל הָתָם כְּלָל וּפְרָט וְאֵין בַּכְּלָל אֶלָּא מַה שֶּׁבַּפְּרָט. אִי נַמִּי הָכָא קְרָא יְתֵרָא קָדָרֵישׁ, וְכָתַב לָהּ תְּרֵי זִמְנֵי. תּוֹסָפוֹת:

וּלְשׁוֹן (טֹפֶס) הוּא, שֶׁהוּא הֶעְתֵּק וְתֹאַר הַגֵּט. וְתוֹרֵף, הוּא גִלּוּי הַגֵּט, מִלְּשׁוֹן יַנִּיחֶנָּהּ בְּמָקוֹם הַתֻּרְפָּה: וְלֵיכָּא לְמֵימַר דְּדִיעֲבַד אֲפִלּוּ תוֹרֵף כָּשֵׁר, דְּהָא כֵּיוָן דִּמְחֻבָּר פְּסוּלָא דְאוֹרַיְתָא הוּא אֲפִלּוּ דִיעֲבַד אִית לָן לְמִפְסְלֵיהּ. הָרַ"ן: יֵשׁ לְעַיֵּן כֵּיוָן דִּכְתִיבַת זְמַן בַּגֵּט לֵיתָא אֶלָּא מִדְּרַבָּנָן, כָּל שֶׁשְּׁאָר הַתּוֹרֵף נִכְתַּב כְּהִלְכָתוֹ, זְמַן לֹא יִהְיֶה לוֹ אֶלָּא דִין טֹפֶס. וְאֶפְשָׁר דְּאַלְּמוּהוּ כְּאִלּוּ הָיָה שֶׁל תּוֹרָה. בֵּית יוֹסֵף: וְיַעֲבִיר תְּנַאי שֶׁהָיָה בוֹ. רַשִׁ"י: וְאַף עַל גַּב דְּאֵינוֹ דּוֹמֶה נִמְחַק פַּעַם אֶחָד לְנִמְחָק שְׁתֵּי פְעָמִים, וְיִהְיֶה נִכָּר אֵצֶל מַחַק דְּעֵדִים, סָבַר תַּנָּא קַמָּא דְגָזְרוּ דִּלְמָא דָמִי הַאי מַחְקָא לְהַאי מַחְקָא. וַחֲכָמִים מַכְשִׁירִין, דְּאִי אִיכָּא חֲשָׁשָׁא אָזְלִינַן לְעֵדֵי מְסִירָה. בֵּית יוֹסֵף, וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: מַכְשִׁירִין. אֵין לִתְמֹהַּ אֵיךְ מַכְשִׁיר רַבִּי אֱלִיעֶזֶר בְּדָבָר שֶׁיָּכוֹל לְהִזְדַּיֵּף, וְהָא מוֹדֶה רַבִּי אֱלִיעֶזֶר בִּמְזֻיָּף מִתּוֹכוֹ שֶׁהוּא פָּסוּל. דְּהָא מוֹדֶה נַמִּי הֵיכָא דְּלֹא אָתֵי לְמִסְמָךְ עֲלַיְהוּ, כְּגוֹן בְּשֵׁמוֹת מֻבְהָקִים, דְּכָשֵׁר, הָכָא נַמִּי כָּשֵׁר. הַגָּהוֹת אָשְׁרִי: וְאַף עַל גַּב דְּבִשְׁאָר שְׁטָרוֹת נַמִּי קָיְמָא לָן דְּעַל יְדֵי עֵדֵי מְסִירָה גּוֹבֶה מִמְּשֻׁעְבָּדִים, מִכָּל מָקוֹם כֵּיוָן דִּלְגוֹבַיְנָא קָיְמָא דִלְמָא לֹא מַפִּיק לְהוּ עַד זְמַן אָרֹךְ וְשָׁכְחוּ עֵדֵי מְסִירָה אֶת הַדָּבָר וְסַמְכִינַן אַעֵדֵי חֲתִימָה, וְהוּא מְזַיֵּף כְּתָב הָעֶלְיוֹן וְחוֹתַם הָעֵדִים עוֹמֵד בִּמְקוֹמוֹ. וְכָתְבוּ הַתּוֹסָפוֹת דְּהַיְנוּ דַּוְקָא בִשְׁטָרוֹת הָעוֹמְדִים לִרְאָיָה, אֲבָל שְׁטַר מֶכֶר וּמַתָּנָה דָמִי לְגִטִּין. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב:

הַיְנוּ טֹפֶס: מִשּׁוּם דְּלֹא שְׁכִיחַ כְּלָל שַׁרְיָא אֲפִלּוּ לְכַתְּחִלָּה. וְלֹא דָמִי לִכְתָבוֹ בִּמְחֻבָּר בְּמַתְנִיתִין דִּלְעֵיל. הָרַ"ן: אֶת גִּטָּהּ. וּמַקְנָתוֹ לַבַּעַל וְחוֹזֵר וּמוֹסְרוֹ לָהּ לְגֵרוּשִׁין. רַשִׁ"י. וְאַף עַל גַּב דִּמְחֻסַּר הַקְנָאָה בֵין כְּתִיבָה לִנְתִינָה לֹא הָוֵי מְחֻסָּר מַעֲשֶׂה, כֵּיוָן שֶׁאֵינוֹ מַעֲשֶׂה בְגוּפוֹ שֶׁל גֵּט דּוּמְיָא דִקְצִיצָה. הָרַ"ן: שׁוֹבְרוֹ. שׁוֹבֵר פֵּרָעוֹן כְּתֻבָּה, שֶׁהָאִשָּׁה הָיָה לָהּ לְכָתְבוֹ וּלְמָסְרוֹ לוֹ לִהְיוֹת לוֹ לִרְאָיָה. רַשִׁ"י: בְּחוֹתְמָיו. הַיְנוּ עֵדֵי מְסִירָה. הָרַ"ן. וְהָא דְנָקַט חוֹתְמָיו, מִשּׁוּם דְּרֹב עֵדֵי מְסִירָה הֵם הַחֲתוּמִים בּוֹ, וְרֹב גִּטִּין חֲתוּמִים הֵם, אֲפִלּוּ לְרַבִּי אֱלִיעֶזֶר, מִפְּנֵי תִקּוּן הָעוֹלָם. רַשִׁ"י: וְאַף עַל גַּב דְּאָמְרִינַן לְעֵיל דְּסַגִּי בְּקוֹל קֻלְמוֹסָא, הַיְנוּ לְמַאן דְּיָכוֹל לְמִחְזֵי, דְמִסְתְּפֵי סוֹפֵר מִנֵּיהּ דִּלְמָא חָזֵי לֵיהּ, אֲבָל בְּסוּמָא כֵּיוָן דְּלֹא אֶפְשָׁר לְמִחְזֵי לֹא מִסְתְּפֵי מִנֵּיהּ, הִלְכָּךְ לֹא מָצֵי אָמַר. רַ"ן וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: הַיְנוּ לְמַאי דְפַסְקִינַן כְּמַאן דְּאָמַר דְּטַעְמָא מִשּׁוּם קִיּוּם:

הַקָּטָן. מַה דִּשְׁכִיחַ טְפֵי תָּנֵי לֵיהּ בָּרֵישָׁא. דְּגַדְלוּת הַקָּטָן וַדַּאי שְׁכִיחַ, וְכֵן פִּקְּחוּת הַחֵרֵשׁ שְׁכִיחַ יוֹתֵר מִפִּתּוּחַ הַסּוּמָא, וּפִתּוּחַ הַסּוּמָא שְׁכִיחַ טְפֵי מִשִּׁפּוּי הַשּׁוֹטֶה: דְּהַהֲלָכָה כְּמַאן דְּאָמַר דְּמִשּׁוּם קִיּוּם הוּא: בְּדַעַת. לוֹמַר דִּבְדַעַת תַּלְיָא מִלְּתָא אֲבָל לֹא בְכַשְׁרוּת, וְסוּמָא שֶׁהָיָה פָתוּחַ בִּשְׁעַת קַבָּלָה לֹא בָּעִינַן שֶׁיְּהֵא סוֹפוֹ בְכַשְׁרוּת. גְּמָרָא:

נֶאֱמָנוֹת. מְפָרֵשׁ בַּגְּמָרָא בְּחוּץ לָאָרֶץ, דְּכֵיוָן שֶׁצְּרִיכָה לוֹמַר בְּפָנַי נִכְתַּב וּבְפָנַי נֶחְתָּם וְאִי אֲתָא בַעַל מְעַרְעֵר לֹא מַשְׁגְּחִינַן בֵּיהּ, מְהֵימְנָא, דְּלֵיכָּא לְמֵימַר דִּלְקִלְקוּלָא מִכַּוְּנָה. אֲבָל בְּאֶרֶץ יִשְׂרָאֵל דְּאֵינָהּ אוֹמֶרֶת בְּפָנַי נִכְתַּב וּבְפָנַי נֶחְתָּם וְאִי אֲתָא בַּעַל וּמְעַרְעֵר מַשְׁגִּיחִין בֵּיהּ, אֵינָהּ נֶאֱמֶנֶת, דְּמִכַּוְּנָא לְקַלְקָלָה: לָאו דַּוְקָא שֶׁיַּתְנֶה הוּא שֶׁתֹּאמְרִי. וְזֶה לְשׁוֹן הָרַ"ן, וְקָא מַשְׁמַע לָן מַתְנִיתִין שֶׁאַף עַל פִּי שֶׁאִלּוּ רָצְתָה הָאִשָּׁה לוֹמַר כְּבָר נִתְגָּרַשְׁתִּי בַגֵּט שֶׁבְּיָדִי לֹא הָיְתָה צְרִיכָה לוֹמַר בְּפָנַי נִכְתַּב וּבְפָנַי נֶחְתָּם, אֲפִלּוּ הָכִי כָּל שֶׁהִיא אוֹמֶרֶת שֶׁנֶּעֶשְׂתָה שָׁלִיחַ צָרִיךְ לוֹמַר בְּפָנַי נִכְתַּב וּבְפָנַי נֶחְתָּם, וְלֹא מִפַּטְּרָה בְּמִגּוֹ דְּאִי בַּעְיָא אָמְרָה נִתְגָּרַשְׁתִּי: וַאֲפִלּוּ שֶׁאֵין הַגֵּט מְקֻיָּם. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב:
