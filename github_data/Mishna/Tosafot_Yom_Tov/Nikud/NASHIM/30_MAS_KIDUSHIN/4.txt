פרק ד - משנה א

יוֹחֲסִין. שֵׁם הַיַּחַס הוּנַח עַל יְדִיעַת הָאִישׁ לְבֵית אֲבוֹתָיו לָדַעַת עִנְיָנוֹ, עַל אֵיזֶה דֶרֶךְ שֶׁנּוֹלַד. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: מִדְּלֹא תְנַן עָלוּ לְאֶרֶץ יִשְׂרָאֵל אֶלָּא עָלוּ מִבָּבֶל, לְאַשְׁמְעִינַן שֶׁכֻּלָּם עָלוּ מִבָּבֶל. גְּמָרָא: כְּלוֹמַר סַנְהֶדְרִי גְדוֹלָה שֶׁהָיוּ יוֹשְׁבִים וְדָנִין אֶת הַכְּהֻנָּה. כְּדִתְנַן בְּסוֹף מַסֶּכֶת מִדּוֹת: כַּהֲנֵי כוּ'. מְנָלָן דִּסְלִיקוּ כוּ', וְיָלֵיף לְהוּ מִקְּרָא בַגְּמָרָא: לְוִיֵּי. וְהָא דִכְתִיב וּמִבְּנֵי לֵוִי לֹא מָצָאתִי שָׁם, זֶה הָיָה בְעוֹדוֹ בְּבָבֶל, דּוֹק בַּפְּסוּקִים. וְהָא דְעֶזְרָא קְנָסָם שֶׁלֹּא עָלוּ עִמּוֹ (כְּמוֹ שֶׁכָּתַב הָרַ"ב סוֹף מַעֲשֵׂר שֵׁנִי), הַיְנוּ עַל שֶׁלֹּא נִתְעוֹרְרוּ מֵעַצְמָם לַעֲלוֹת. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: וְלֹא מָנֵי חֲלוּצָה, מִשּׁוּם דְּדִבְרֵי סוֹפְרִים הוּא. וּמִכָּל מָקוֹם הַוָּלָד חָלָל מִדִּבְרֵיהֶם. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: שֶׁנֶּאֱמַר וַיִּתְּנֵם יְהוֹשֻׁעַ חֹטְבֵי עֵצִים וְגוֹ' . וְחַיָּבִים יִשְׂרָאֵל לָמוּל עַבְדֵיהֶם הַכְּנַעֲנִים, וְהַנְהוּ הֲווּ בִּכְלַל מִקְנַת כַּסְפֶּךָ (וְעַיֵּן מַה שֶּׁכָּתַבְתִּי רֵישׁ פֶּרֶק ג דְּמַכּוֹת): לְוִיֵּי כוּ'. אֲבָל כַּהֲנֵי בַּחֲלָלֵי גֵּרֵי וַחֲרוּרֵי לֹא, שֶׁחֲלָלָה בְלָאו. וּמְשֻׁחְרֶרֶת בְּחֶזְקַת זוֹנָה. רַשִׁ"י: וַחֲרוּרֵי. מַשְׁמַע אֲבָל עֲבָדִים לֹא. אַלִיבָּא דְרַבִּי טַרְפוֹן הוּא הַדִּין לְמַמְזֵרִים דְּמֻתָּרִין בַּעֲבָדִים, אֶלָּא מִשּׁוּם אִינָךְ שֶׁאֲסוּרִים בַּעֲבָדִים תָּנָא חֲרוּרֵי. הָרַ"ן: מֻתָּרִים כוּ'. וְאַף דִּנְתִינִים קָיְמֵי בְּלֹא תִתְחַתֵּן דְּקָאֵי אַשִּׁבְעָה עֲמָמִין בְּגֵרוּתָן, דְּיֵשׁ לוֹמַר דְּלֹא מִחַיְּבֵי בְּלָאו דְּחִתּוּן אֶלָּא הֵיכָא דְאִיכָּא חֲשָׁשָׁא דְכִי יָסִיר, לְפִי שֶׁלֹּא חָשְׁשָׁה תוֹרָה אֶלָּא אַמְּיֻחָסִים, אֲבָל אַפְּסוּלִים לֹא. אֶלָּא דִבְלֹא גֵרוּת אֲסוּרִים מִשּׁוּם חֲשַׁשׁ הֲסָרָה, אֲבָל נְתִינֵי שֶׁהֵם גֵּרִים וְחַיָּבִים בְּכָל הַמִּצְוֹת לָאו בְּנֵי הֲסָרָה נִינְהוּ: וְהָכִי נַמִּי אִיכָּא לְמִדְרַשׁ מַמְזֵר וַדַּאי הוּא דְלֹא יָבֹא, הָא סָפֵק מַמְזֵר יָבֹא, וְלִישְׁרֵי בְיִשְׂרָאֵל שְׁתוּקֵי וַאֲסוּפֵי. וְהָכִי נַמִּי אָמְרִינַן בַּגְּמָרָא, אֶלָּא דַחֲכָמִים הֵם שֶׁעָשׂוּ מַעֲלָה בְיוֹחֲסִין וְלֹא רָצוּ לְהַכְשִׁירָם לְיִשְׂרָאֵל, אֲבָל זֶה בָזֶה לֹא עָשׂוּ מַעֲלָה:

פרק ד - משנה ג

אָסוּר. דִּסְבִירָא לֵיהּ עָשׂוּ מַעֲלָה בְיוֹחֲסִין אֲפִלּוּ בְּהָנֵי:

הַנּוֹשֵׂא כוּ'. פֵּרוּשׁ כֹּהֵן שֶׁבָּא לִשָּׂא אִשָּׁה כֹהֶנֶת. וְקָאָמַר דְּצָרִיךְ לִבְדֹּק מִשּׁוּם שֶׁאָנוּ רוֹצִים שֶׁבְּנָהּ יְשַׁמֵּשׁ עַל גַּבֵּי הַמִּזְבֵּחַ. אֲבָל אֵין לְפָרֵשׁ דְּמַיְרֵי בְיִשְׂרָאֵל כוּ', דְּאִם כֵּן אַמַּאי נָקַט כֹּהֶנֶת, אֲפִלּוּ חֲלָלָה נַמִּי הֲוֵי בְּנָהּ כָּשֵׁר לִכְהֻנָּה. תּוֹסָפוֹת. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב דִּלְהָרַ"ב דִּמְפָרֵשׁ דְּדַוְקָא בְּעִרְעוּר דִּין הַבְּדִיקָה מִשּׁוּם עַצְמוֹ, שֶׁלֹּא יִשָּׂא לַמֻּזְהָר עָלֶיהָ. וַאֲפִלּוּ הָכִי אֵין לְפָרֵשׁ בְּיִשְׂרָאֵל, דְּכָל שֶׁאֵין כָּאן פְּסוּל חֲלָלוּת אֵין צָרִיךְ בְּדִיקָה אֲפִלּוּ לְמַמְזֵרוּת. וְכֵן כָּתְבוּ הַתּוֹסָפוֹת דְּעִקָּר הַבְּדִיקָה מִשּׁוּם חֲלָלוּת, וְאַגַּב כוּ'. וְעַיֵּן עוֹד: שְׁמֹנֶה. נָקַט הַאי לִישָׁנָא לְפִי שֶׁמִּפְּנֵי בְדִיקַת הָאַרְבַּע צָרִיךְ לִבְדֹּק הָאֲחֵרוֹת, שֶׁאֵין הָאַרְבַּע נִבְדָּקוֹת אֶלָּא אִם כֵּן הוּא בוֹדֵק שֶׁלְּמַעְלָה מֵהֶן. הָרַ"ן: וּלְשׁוֹן רַשִׁ"י, בּוֹדְקִים שֶׁלֹּא הָיוּ מַמְזֵרוֹת אוֹ אַחַת מֵהַפְּסוּלוֹת לַקָּהָל: לְוִיָּה כוּ'. דְּאַף עַל גַּב דְּאֵין בָּהֶן חֲשַׁשׁ חֲלָלוּת, שֶׁאָבִיהָ הוּא לֵוִי וְיִשְׂרָאֵל אֲפִלּוּ כְּשֶׁאִמָּהּ חֲלָלָה כְּשֵׁרָה לִכְהֻנָּה כְּדִתְנַן בְּמִשְׁנָה ו', מִכָּל מָקוֹם הוֹאִיל וְהִצְרִיכוּ לְכֹהֵן לִבְדֹּק מִשּׁוּם פְּסוּל חֲלָלוּת בְּכֹהֶנֶת דְּשָׁיְכֵי בָּהּ, הִלְכָּךְ הִצְרִיכוּהוּ לִבְדֹּק הֵיכָא דְלָא שַׁיָּךְ אֶלָּא פְסוּל מַמְזֵרוּת, כְּגוֹן לְוִיָּה וְיִשְׂרְאֵלִית. תּוֹסָפוֹת: וְכֵן לְשׁוֹן רַשִׁ"י. וְאַף עַל גַּב דִּבְכֹהֵן אָיְרֵי, הֻצְרְכוּ לְפָרֵשׁ כָּאן שֶׁלֹּא נִטְעֶה, דְּבֵין בָּרֵישָׁא בֵּין בַּסֵּיפָא הַנּוֹשֵׂא הוּא כְמוֹ הַנִּשֵּׂאת: מוֹסִיפִין כוּ'. לְפִי שֶׁהַתַּעֲרֹבוֹת בָּהֶם יוֹתֵר. הָרַמְבַּ"ם: שֶׁיָּעִידוּ שְׁנַיִם שֶׁנִּתְעָרֵב בָּהֶם מַמְזֵר אוֹ חָלָל אוֹ שֶׁיֵּשׁ בָּהֶן עַבְדוּת. הָרַמְבַּ"ם: וְלֹא קָשְׁיָא דְּהָא הִשְׁוָה הַכָּתוּב אִשָּׁה כוּ' לְכָל עֳנָשִׁין, דְּהָא אָמְרִינַן בַּגְּמָרָא דְּכָל הֵיכָא דְהוּא מֻזְהָר עָלֶיהָ הִיא מֻזְהֶרֶת עָלָיו שֶׁלֹּא תִשָּׂאֶנּוּ, וְאִם כֵּן הֲרֵי הִיא שָׁוָה לוֹ: וְכֵיוָן שֶׁלֹּא הִקְפִּידָה תוֹרָה עֲלֵיהֶם לְיוֹחֲסִין, רַבָּנָן נַמִּי לֹא עֲבוּד בְּהוּ מַעֲלָה לְאַצְרְכִינְהוּ בְדִיקָה, וַאֲפִלּוּ מֵחֲמַת מַמְזֵרוּת וּשְׁתִיקוּת, אֶלָּא כִי אִתְחַזַּק אִסּוּרָא בֵּיהּ, אֲבָל לַחֲשָׁשָׁא וּלְמִבְדַּק לֹא אַצְרְכִינְהוּ. רַשִׁ"י:

הַמִּזְבֵּחַ. מַיְרֵי אֲפִלּוּ בַּעֲבוֹדוֹת הַכְּשֵׁרוֹת בְּזָרִים כְּגוֹן שְׁחִיטָה כוּ', דִּלְכַתְּחִלָּה לֹא עָבְדֵי כִּי אִם מְיֻחָסִים וְהַיְנוּ דְּלֹא תָנֵי סֵפֶר תּוֹרָה וּתְרוּמָה וּנְשִׂיאַת כַּפַּיִם, מִשּׁוּם דְּמִלְּתָא דִפְשִׁיטָא נִינְהוּ דְּאֵינָן אֶלָּא בְּכֹהֲנִים. וְשִׁיֵּר נַמִּי שְׁטָרוֹת, דְּאִיכָּא מַאן דְּאָמַר דְּמַעֲלִין. תּוֹסָפוֹת. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: וְהוּא הַדִּין בְּאָבִיהָ שֶׁאֵין צָרִיךְ לִבְדֹּק עוֹד לֹא בְאִמּוֹ וְלֹא בְּאֵם אֵם אָבִיהָ, וְכֵן בְּכָל אָב: הַדּוּכָן כוּ'. דְּאָמַר מַר שֶׁשָּׁם (בְּלִשְׁכַּת הַגָּזִית. רַשִׁ"י) הָיוּ יוֹשְׁבִין סַנְהֶדְרִי גְדוֹלָה מְיֻחָסֵי כְהֻנָּה מְיֻחָסֵי לְוִיָּה. גְּמָרָא: תֵּימַהּ, דְּהָא אֵין כְּשֵׁרִים לְדִינֵי נְפָשׁוֹת אֶלָּא הַמַּשִּׂיאִין לִכְהֻנָּה, וְכָל סַנְהֶדְרִין הָיוּ דָּנִין דִּינֵי נְפָשׁוֹת, וְכָל עִיר שֶׁיֵּשׁ בָּהּ מֵאָה וְעֶשְׂרִים רְאוּיָה לְסַנְהֶדְרִין. אֶלָּא עִקָּרָא דְמִלְּתָא דְסַנְהֶדְרִין בְּכָל מָקוֹם. וּתְנַאי דְבִירוּשָׁלַיִם, אֲמָרוּהוּ בַגְּמָרָא אַשּׁוֹטְרֵי הָרַבִּים, מִשּׁוּם דִּלְדִינֵי מָמוֹנוֹת הַכֹּל כְּשֵׁרִים וַאֲפִלּוּ מַמְזֵר. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: מִשּׁוֹטְרֵי. הַיְנוּ דְּפֵרֵשׁ הָרַ"ב שֶׁהָיוּ דָּנִין דִּינֵי מָמוֹנוֹת. וְכֵן פֵּרֵשׁ רַשִׁ"י: לְשׁוֹן רַשִׁ"י, בְּעַרְכֵי, שֶׁהָיָה כָּתוּב בְּסִדְרֵי הַדַּיָּנִין שֶׁהָיוּ רְגִילִין לְיַחֵס אֶת הָרְאוּיִים לָדוּן שֶׁהָיוּ מְיֻחָסִים, וְכוֹתְבִין עַל הַסֵּדֶר פְּלוֹנִי וּפְלוֹנִי מְיֻחָסִים הֵן, שֶׁאֵין רְגִילִים בְּנֵי הַמָּקוֹם לְמַנּוֹת דַּיָּן שֶׁאֵינוֹ מְיֻחָס: מֶלֶךְ. דְּאָמַר קְרָא וְהִתְיַחְשָׂם בַּצָּבָא בַּמִּלְחָמָה : הַיְנוּ לוֹמַר דִּכְמוֹ שֶׁמָּצִינוּ דְּבָעִינַן זְכוּתָן כִּדְאִיתָא בְמִשְׁנָה ה' פֶּרֶק ט' דְּסוֹטָה, הָכִי נַמִּי הִצְרִיךְ דָּוִד הַמֶּלֶךְ עָלָיו הַשָּׁלוֹם שֶׁיִּהְיֶה לָהֶם זְכוּת אֲבוֹתָם:

פרק ד - משנה ו

לְעוֹלָם. מַהוּ דְתֵימָא מִידֵי דַּהֲוָה אַמִּצְרִי וַאֲדוֹמִי, מַה לְּהַלָּן לְאַחַר שְׁלֹשָׁה דוֹרוֹת כוּ', קָא מַשְׁמַע לָן. גְּמָרָא: כְּשֵׁרָה כוּ'. בַּגְּמָרָא יָלֵיף לֵיהּ מִקְּרָא. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב:

פרק ד - משנה ז

רַבִּי יוֹסֵי אוֹמֵר. הֲרֵי שָׁלֹשׁ מַחֲלוֹקוֹת בַּדָּבָר. לְרַבִּי יְהוּדָה דִלְעֵיל בַּת גֵּר זָכָר פְּסוּלָה וַאֲפִלּוּ נָשָׂא יִשְׂרְאֵלִית. לְרַבִּי אֱלִיעֶזֶר בֶּן יַעֲקֹב אִיכָּא צַד אֶחָד יִשְׂרָאֵל מַכְשִׁיר, וּמִיהוּ גֵּר וְגִיּוֹרֶת בִּתּוֹ פְסוּלָה. וּלְרַבִּי יוֹסֵי אֲפִלּוּ גֵר שֶׁנָּשָׂא גִּיּוֹרֶת בִּתּוֹ כְשֵׁרָה, וּמִיהוּ הוֹרָתָהּ וְלֵדָתָהּ בִּקְדֻשָּׁה בָּעִינַן. וּבַגְּמָרָא וְכֻלָּן מִקְרָא אֶחָד דָּרְשׁוּ כוּ'. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב:

פרק ד - משנה ח

וּתְמֵהַנִי, לָמָּה הֻצְרַךְ לְכָךְ, דְּכֵיוָן דִּבְחֶזְקַת כַּשְׁרוּת הוּא אֵין עִרְעוּר פָּחוֹת מִשְּׁנַיִם. הָרַ"ן. וְלִי נִרְאֶה דְמִשּׁוּם עֻבָּר שֶׁאֵין לוֹ חֲזָקָה דְכַשְׁרוּת הֻצְרַךְ לָזֶה: אַף עַל פִּי שֶׁלְּהַכְשִׁיר נֶאֱמֶנֶת כְּדִלְעֵיל בְּמִשְׁנָה ב' לְאַבָּא שָׁאוּל, שָׁאנֵי, דִּלְהַכְשִׁיר דְּבַר תּוֹרָה אֵין אָנוּ צְרִיכִין לָהּ, דְּהָא מִדִּינָא לֹא הֲוֵי אֶלָּא סָפֵק מַמְזֵר, וְסָפֵק מַמְזֵר מִן הַתּוֹרָה מֻתָּר לָבֹא בַקָּהָל כְּמוֹ שֶׁאִיתָא בְּמִשְׁנָה ב' אֶלָּא דְרַבָּנָן גָּזְרוּ עֲלֵיהֶם, וְהֶאֱמִינוּהָ בְּשֶׁל דִּבְרֵיהֶם. אֲבָל לִפְסֹל, כֵּיוָן דְּמִדְּאוֹרַיְתָא מֻתָּר לָבֹא בַקָּהָל וְאָסוּר בְּמַמְזֶרֶת וַדָּאִית, וְאִם הָיוּ מַאֲמִינִים אוֹתָהּ וּמַתִּירִין אוֹתוֹ בְּמַמְזֶרֶת וַדָּאִית, הָיוּ מְקִלִּים בְּשֶׁל תּוֹרָה, וְלֹא נָתְנָה תוֹרָה נֶאֱמָנוּת אֶלָּא לָאָב. וַאֲפִלּוּ לָאָב נֶחְלְקוּ חֲכָמִים. הָרַב הַמַּגִּיד מִשְׁנֶה: וְהוֹאִיל וְאֵינוֹ נֶאֱמָן לִפְסֹל לִבְנֵי הַבֵּן. אַף אַבְּנוֹ אֵינוֹ נֶאֱמָן, שֶׁאִם נַאֲמִינֵהוּ עַל הַבֵּן, מִמֵּילָא יִהְיוּ נִפְסָלִין בְּנֵי הַבֵּן. תּוֹסָפוֹת וְהָרַ"ן:

פרק ד - משנה ט

שְׁנֵיהֶם כוּ'. אִם בָּאת לִנָּשֵׂא לְאַחֵר. רַשִׁ"י:

פרק ד - משנה י

וְהַתּוֹסָפוֹת הִקְשׁוּ פְּשִׁיטָא כֵּיוָן דְּנִבְדֶּקֶת כְּבָר שׁוּב אֵין צָרִיךְ רְאָיָה. וְעוֹד דְּאֵינוֹ מֵבִיא רְאָיָה דְּסֵיפָא מְיֻתָּר כוּ'. וְלִי נִרְאֶה דְיֵשׁ לוֹמַר דְּאִין הָכִי נַמִּי דְּכֻלָּהּ מַתְנִיתִין לֹא אִיצְטְרִיךְ אֶלָּא מִשּׁוּם רְאָיָה דְעַל הַבָּנִים, דְּבָרֵישָׁא אֵין צָרִיךְ כֵּיוָן דִּכְרוּכִים, וּבַסֵּיפָא כֵּיוָן דְּמֵתָה צָרִיךְ, וְאַגַּב נָסֵיב נַמִּי בֵּין בָּרֵישָׁא בֵּין בַּסֵּיפָא דְאֵין צָרִיךְ רְאָיָה עַל הָאִשָּׁה. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: נִדְבָּקִים אֶצְלָהּ.רַשִׁ"י: עַל הַבָּנִים. צָרִיךְ לוֹמַר דְּמַתְנִיתִין אֲפִלּוּ לְרַבִּי יְהוּדָה דְמִשְׁנָה ח' שֶׁסּוֹבֵר דְּדִין תּוֹרָה הָאָב נֶאֱמָן לְפָסְלוֹ וְכָל שֶׁכֵּן לְהַכְשִׁירוֹ וּכְמוֹ שֶׁכָּתוּב לְעֵיל, אֲפִלּוּ הָכִי מִשּׁוּם מַעֲלָה דְיוֹחֲסִין אָמְרוּ חֲכָמִים שֶׁלֹּא לְהַאֲמִינוֹ. וְכָל זֶה הוּא רַק נֶגֶד מִשְׁפָּחוֹת הַבְּדוּקוֹת שֶׁרוֹצִים שֶׁתִּהְיֶה מִשְׁפַּחְתָּן בְּדוּקָה, אֲבָל בְּלָאו הָכִי אֵין צָרִיךְ לְהָבִיא רְאָיָה, דְּכָל אָדָם בְּחֶזְקַת כַּשְׁרוּת עַד שֶׁיִּפָּסֵל (עַל יְדֵי עִרְעוּר). הַמַּגִּיד:

פרק ד - משנה יא

הַבָּנִים. הַקְּטַנִּים הַכְּרוּכִים אַחֲרֶיהָ. גְּמָרָא:

פרק ד - משנה יב

לֹא יִתְיַחֵד. רֶמֶז לְיִחוּד מִן הַתּוֹרָה מִנַּיִן, שֶׁנֶּאֱמַר כִּי יְסִיתְךָ אָחִיךָ בֶן אִמֶּךָ , וְכִי בֶן אֵם מֵסִית בֶּן אָב אֵינוֹ מֵסִית, אֶלָּא לוֹמַר לְךָ, בֵּן מִתְיַחֵד עִם אִמּוֹ (וּלְהָכִי נָקַט בֶּן אִמֶּךָ, שֶׁהַבֵּן מָצוּי אֵצֶל הָאֵם), וְאָסוּר לְהִתְיַחֵד עִם כָּל עֲרָיוֹת שֶׁבַּתּוֹרָה. גְּמָרָא. וּלְפִי שֶׁלֹּא אָמְרוּ אֶלָּא רֶמֶז, כָּתַב הָרַמְבַּ"ם, וְאִסּוּר יִחוּד הָעֲרָיוֹת מִפִּי הַקַּבָּלָה. וְכָתַב עוֹד, דָּוִד וּבֵית דִּינוֹ גָּזְרוּ עַל יִחוּד הַפְּנוּיָה, שַׁמַּאי וְהִלֵּל גָּזְרוּ עַל יִחוּד כּוּתִית. עַד כָּאן. וּמִיהוּ כָּל זֶה בְּיוֹתֵר מִשָּׁלֹשׁ שָׁנִים וְיוֹם אֶחָד. יְרוּשַׁלְמִי. וְכֵן פָּסַק הָרַמְבַּ"ם. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: וְרַשִׁ"י וְהָרַמְבַּ"ם כָּתְבוּ נַמִּי בַּת חֲמוֹתָהּ: כְּלוֹמַר, מַה הִיא בִּיאָה, שֶׁתֵּדַע לְסַפֵּר דְּבָרִים בַּשּׁוּק. וְאֵינָהּ מוֹסֶרֶת כוּ' מִתּוֹךְ קָטְנָהּ עֲדַיִן לֹא לְבָשָׁהּ יֵצֶר, הִלְכָּךְ אֵין כָּאן מִשּׁוּם דַּעְתָּהּ קַלָּה, שֶׁלֹּא תְהֵא נוֹחָה לְהִתְפַּתּוֹת. רַשִׁ"י: שְׁנֵיהֶם בִּכְלָל. הָרַמְבַּ"ם: שֶׁאִשְׁתּוֹ. הִקְשׁוּ בַּתּוֹסָפוֹת דִּבְרֵישׁ פֶּרֶק ב' דַּעֲבוֹדָה זָרָה מַשְׁמַע שֶׁאִשָּׁה אַחַת מִתְיַחֶדֶת עִם אִישׁ שֶׁאִשְׁתּוֹ עִמּוֹ, וְהָכָא מַשְׁמַע דְּדַוְקָא שְׁתֵּי נָשִׁים, וּמִכָּל שֶׁכֵּן לְרַבָּנָן, אִם כֵּן מַתְנִיתִין דְּהָתָם דְּלֹא כְמַאן כוּ'. לְכָךְ כָּתְבוּ דְּהָכִי גָרְסִינַן, רַבִּי שִׁמְעוֹן אוֹמֵר אַף אִישׁ אֶחָד מִתְיַחֵד עִם שְׁתֵּי נָשִׁים. וּבִזְמַן שֶׁאִשְׁתּוֹ עִמּוֹ יָשֵׁן עִמָּהֶן בְּפוּנְדְּקִי, מִפְּנֵי שֶׁאִשְׁתּוֹ מְשַׁמַּרְתּוֹ. וַאֲתָאן אֲפִלּוּ לְרַבָּנָן. וְכֵן גִּרְסַת הָרִי"ף:

פרק ד - משנה יג

בְּיָדוּעַ שֶׁיֵּשׁ בָּהּ טַעַם בִּיאָה. רַשִׁ"י. כְּלוֹמַר, וְיֵשׁ לָחוּשׁ שֶׁמּוֹסֶרֶת עַצְמָהּ לְבִיאָה. וְאֵין זֶה כְּטַעַם בִּיאָה דִלְעֵיל, דְּהָתָם אֵינָה מוֹסֶרֶת. וּמִשׁוּם הָכִי לְעֵיל אָמְרִינַן שֶׁיּוֹדַעַת, וְהָכָא שֶׁיֵּשׁ בָּהּ: רַוָּק. תַּרְגּוּם שֶׁל וּבַחוּרִים כָּשׁוֹל יִכָּשֵׁלוּ הוּא רַוָּקִים. תּוֹסָפוֹת וְהָרַמְבַּ"ם: דְּאִלּוּ תִּינוֹקוֹת עַצְמָן לֹא נֶחְשְׁדוּ עַל הַזָּכוּר. גְּמָרָא: אִשָּׁה. מִפְּנֵי אֲבוֹתֵיהֶם שֶׁבָּאִים לְהָבִיא אֶת בְּנֵיהֶם. גְּמָרָא: אֶצְלוֹ בְּבֵית הַסֵּפֶר. וְטַעְמָא דְתַנָּא קַמָּא, דְּאַף עַל פִּי שֶׁהִיא בְּבֵיתָהּ מִקְּרֵי פַת בְּסַלּוֹ. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב:

פרק ד - משנה יד

וְעַל הַבְּהֵמָה. רַשִׁ"י וְרַ"ן: כְּלוֹמַר דַּהֲוָה אֲמִינָא דְלִשְׁרֵי לְהִתְיַחֵד עִם שְׁתֵּי נָשִׁים מִשּׁוּם כְּדֵי חַיָּיו. וְכָתַב הָרַמְבַּ"ם וְכֵיצַד יַעֲשֶׂה, יִתְעַסֵּק עִמָּהֶן וְאִשְׁתּוֹ עִמּוֹ, אוֹ יִפְנֶה לִמְלָאכָה אַחֶרֶת: נְקִיָּה כוּ'. בַּגְּמָרָא, מַחֲטָא דְתַלְמִיּוּתָא, כְּלוֹמַר תּוֹפֵר בְּגָדִים קְרוּעִים. וְהַיְנוּ תַלְמִיּוּתָא שֶׁהַתְּפִירוֹת עֲשׂוּיוֹת שׁוּרוֹת שׁוּרוֹת כִּתְלָמִים שֶׁל מַחֲרֵשָׁה. רַשִׁ"י. וְהֶעָרוּךְ מְסַיֵּם נְקִיָּה מִגָּזֵל וְקַלָּה דְּאֵין בָּהּ חֶסְרוֹן כִּיס: לְמִי כוּ'. לְשׁוֹן רַשִׁ"י, וְאַל יֹאמַר בְּלִבּוֹ אֻמָּנוּת זוֹ אֵינָהּ מְעַשֶּׁרֶת, אֶלָּא יְבַקֵּשׁ רַחֲמִים לְמִי שֶׁהָעֹשֶׁר שֶׁלּוֹ: שֶׁלּוֹ. שֶׁנֶּאֱמַר לִי הַכֶּסֶף וְלִי הַזָּהָב נְאוּם ה' צְבָאוֹת . גְּמָרָא: לְפִי זְכוּתוֹ. וְקַשְׁיָא לִי דְּאִם הַכֹּל לְפִי זְכוּתוֹ, אִם כֵּן תְּפִלָּה מַאי מַהֲנְיָא לֵיהּ. שֶׁאִם זְכוּתוֹ כְּדַאי אֵין הַקָּדוֹשׁ בָּרוּךְ הוּא מְקַפֵּחַ שְׂכָרוֹ, וְאִם זְכוּתוֹ אֵינוֹ כְּדַאי הָא אָמְרָן הַכֹּל לְפִי זְכוּתוֹ. וּמַסִּיק שֶׁעַל יְדֵי הַתְּפִלָּה זְכוּתוֹ יִגְרֹם לְשַׁנּוֹת מַזָּלוֹ כְּמוֹ עַל יְדֵי זְכוּת גָּדוֹל. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: רַבִּי שִׁמְעוֹן בֶּן אֶלְעָזָר כוּ'. כְּלוֹמַר רַבִּי מֵאִיר הֵבִיא רְאָיָה מִן הַשָּׂכָר. וְרַבִּי שִׁמְעוֹן בֶּן אֶלְעָזָר הֵבִיא רְאָיָה מִן הָעֹנֶשׁ: שֶׁהֲרֵעוֹתִי כוּ'. שֶׁנֶּאֱמַר עֲוֹנוֹתֵיכֶם הִטּוּ . גְּמָרָא: סַפָּר. בַּגְּמָרָא וּבַיְרוּשַׁלְמִי גָּרַס קַדָּר. וּפֵרֵשׁ רַשִׁ"י בַּעֲלֵי קְרָנוֹת: רְשָׁעִים. מִפְּנֵי לִסְטִיּוּת: כְּשֵׁרִים. שֶׁפּוֹרְשִׁין לַמִּדְבָּרוֹת לִמְקוֹם גְּדוּדֵי חַיּוֹת וְלִסְטִים וִירֵאִים לְנַפְשָׁם וּמְשַׁבְּרִים לִבָּן לַמָּקוֹם: חֲסִידִים. שֶׁפּוֹרְשִׁים לִמְקוֹם הַסַּכָּנָה וְתָמִיד הֵם בִּרְעָדָה יוֹתֵר מִן הַגַּמָּלִים: אֵינָהּ כֵּן. שֶׁאַף הַבָּא לִידֵי זִקְנָה אוֹ יִסּוּרִין וְאֵינוֹ יָכוֹל לַעֲסֹק בָּהּ הוּא אוֹכֵל מִמַּתַּן שְׂכָרָהּ. דִּכְתִיב עוֹד יְנוּבוּן בְּשֵׂיבָה. רַשִׁ"י. וּמִמֵּילָא דְהוּא הַדִּין יִסּוּרִין, דְּכָל שֶׁאֵינוֹ יָכוֹל לַעֲסֹק בָּהּ, מַה לִּי זִקְנָה וּמַה לִּי יִסּוּרִין: אַחֲרִית. וְהוּא מֵעִנְיַן אַחֲרָיוּת, שֶׁפֵּרוּשׁוֹ מֵיטָב. וְכֵן מְסַיֵּם דְּשֵׁנִים וְרַעֲנַנִּים יִהְיוּ: בְּשֵׂיבָה. לְשׁוֹן רַשִׁ"י, יִצְמְחוּ צֶמַח וְיִתְיַחֵד שְׂכָרָן לְעֵת שֵׂיבָתָן
