נִקְנֵית. לְבַעְלָהּ לְהַצְרִיכָהּ מִמֶּנּוּ גֵט. הָרַ"ן: וְאַכַּתִּי קָשְׁיָא דְלִתְנֵי מִתְקַדֶּשֶׁת. וּמְתָרֵץ בַּגְּמָרָא מִשּׁוּם דְּקָא בָּעֵי לְמִתְנֵי כֶסֶף, דְּיָלְפִינַן קִיחָה קִיחָה, וְקִיחָה אִקְּרֵי קִנְיָן דִּכְתִיב הַשָּׂדֶה אֲשֶׁר קָנָה אַבְרָהָם. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: בְּשָׁלֹשׁ כוּ'. מִנְיָנָא לְמַעוֹטֵי חֲלִיפִין. דְּלֹא תֵימָא הוֹאִיל וְגָמְרִינַן מִשְּׂדֵה עֶפְרוֹן, מָה הָתָם בַּחֲלִיפִין אַף כָּאן בַּחֲלִיפִין, קָא מַשְׁמַע לָן דְּלֹא, דְּאִשָּׁה בְּפָחוֹת מִשָּׁוֶה פְרוּטָה לֹא מַקְנְיָא נַפְשָׁהּ, וַחֲלִיפִין אֲפִלּוּ בְּפָחוֹת מִשָּׁוֶה פְרוּטָה. וְלֹא דִּבֵּר הַכָּתוּב בְּקִיחָה דְאִשָּׁה אֶלָּא בְקִנְיַן דָּמִים. גְּמָרָא: וְקוֹנָה. לִהְיוֹת בִּרְשׁוּתָהּ לְהִנָּשֵׂא לְאַחֵר. רַשִׁ"י: בִּשְׁתֵּי. מִנְיָנָא לְמַעוֹטֵי חֲלִיצָה דְּלָא תֵּיתֵי מִקַּל וָחֹמֶר מִיבָמָה שֶׁאֵינָהּ יוֹצְאָה בְגֵט כוּ', תַּלְמוּד לוֹמַר כוּ'. גְּמָרָא: בְּכֶסֶף. וְצָרִיךְ שֶׁיִּהְיֶה לְשֵׁם קִדּוּשִׁין וְלִפְנֵי עֵדִים. וְתֵימַהּ עַל הָרַ"ב וְהָרַמְבַּ"ם שֶׁלֹּא פֵרְשׁוּ. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: אִיבְרָא דְבַגְּמָרָא הָכִי אִיתָא, אֲבָל הָרַ"ב מְסַיֵּם וּנְתָנוֹ לָהּ, מִשּׁוּם דְּהוּא הַדִּין בְּמִתְקַדֶּשֶׁת שֶׁלֹּא עַל יְדֵי אָבִיהָ אָיְרֵי. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: וְלָאו דַוְקָא, אֶלָּא נִתְיַחֵד עִמָּהּ: בְּדִינָר. טַעֲמַיְיהוּ דְבֵית שַׁמַּאי שֶׁלֹּא יְהוּ בְנוֹת יִשְׂרָאֵל כְּהֶפְקֵר. גְּמָרָא. וּמָצָאנוּ שִׁעוּר דִּינָר שֶׁהוּא שִׁעוּר חָשׁוּב: וּבְשָׁוֶה. וְהַטַּעַם דְּכֵיוָן שֶׁהִיא מִתְרַצֵּית בַּדָּבָר וְנִיחָא לָהּ בְּשָׁוֶה כֶסֶף כְּכֶסֶף דָּמֵי, וְאֵין צָרִיךְ קְרָא לְרַבּוֹיֵי. הָרַ"ן: וְכַמָּה כוּ'. הָא דְלָא מְפָרֵשׁ כַּמָּה הוּא דִינָר מִשּׁוּם דְּדִינָר הָיוּ יוֹדְעִים כַּמָּה הָיָה כוּ'. וְעוֹד דְּדִבְרֵי בֵית הִלֵּל דְּקָיְמָא לָן כְּוָתֵיהּ אִיבָּעֵי לֵיהּ לְפָרוֹשֵׁי טְפֵי. תּוֹסָפוֹת: הָאִיטַלְקִי. הִיא אִיטַלְיָא שֶׁל יָוָן. הָרַ"ב: בְּגֵט כוּ'. בַּגְּמָרָא יָלְפִינַן לְהוּ מִקְּרָאֵי: כִּדְיָלֵיף לַהּ בַּגְּמָרָא מִקְּרָאֵי: בַּחֲלִיצָה. יָלֵיף לֵיהּ בַּגְּמָרָא מִקְּרָא: וּבְמִיתַת כוּ'. קַל וָחֹמֶר, וּמָה אֵשֶׁת אִישׁ שֶׁהִיא בְחֶנֶק מִיתַת הַבַּעַל מַתִּירָתָהּ, יְבָמָה שֶׁהִיא בְלָאו לֹא כָל שֶׁכֵּן כוּ'. גְּמָרָא:

בְּכֶסֶף. וְלֹא פֵרְשָׁה מִשְׁנָתֵנוּ זוֹ בְּכַמָּה. וְהָרַמְבַּ"ם כָּתַב בְּאָמָה, וְאֵינָהּ נִקְנֵית בִּפְרוּטָה מִפְּנֵי שֶׁצָּרִיךְ לִקְנוֹתָהּ בְּדָמִים שֶׁרְאוּיִים לְגֵרָעוֹן כְּדֵי שֶׁתִּגְרַע פִּדְיוֹנָהּ וְתֵצֵא. וּמָצָאתִי לוֹ רְאָיָה כוּ', וּבְעֶבֶד עִבְרִי לֹא כָתַב כֵּן. וְכָתַב הַכֶּסֶף מִשְׁנֶה מַשְׁמַע דְּדִינוֹ בִּפְרוּטָה, וְקָשֶׁה דְהָא אִתַּקַּשׁ לְעִבְרִיָּה: וְכֵן כָּתַב הָרַמְבַּ"ם. וּבַגְּמָרָא, אַשְׁכְּחַן עֶבֶד עִבְרִי הַנִּמְכָּר לְנָכְרִי כוּ', נִמְכָּר לְיִשְׂרָאֵל מְנָלָן. אָמַר קְרָא וְהֶפְדָּהּ, מְלַמֵּד שֶׁמְּגָּרֵעַ פִּדְיוֹנָהּ וְיוֹצְאָה. וּפֵרֵשׁ רַשִׁ"י אַלְמָא בְּכֶסֶף נִקְנָה, דְּאִי לֹא קְנָאָהּ בְּדָמִים מַאי מְגָרְעָא. וְהֻקַּשׁ עִבְרִי לְעִבְרִיָּה. אַשְׁכְּחַן מְכָרוּהוּ בֵּית דִּין הוֹאִיל וְנִמְכָּר בְּעַל כָּרְחוֹ, מוֹכֵר עַצְמוֹ מְנָלָן, יָלֵיף שָׂכִיר שָׂכִיר. וּפֵרֵשׁ רַשִׁ"י נֶאֱמַר בְּמוֹכֵר עַצְמוֹ כְּשָׂכִיר כְּתוֹשָׁב יִהְיֶה עִמָּךְ, וְנֶאֱמַר בִּמְכָרוּהוּ בֵּית דִּין כִּי מִשְׁנֶה שְׂכַר שָׂכִיר: כְּלוֹמַר קִיחַת קִדּוּשִׁין. וּלְפִי זֶה שְׁטָר דְּאָמָה נַמִּי הַקּוֹנֶה כוֹתֵב בִּתְּךָ קְנוּיָה לִי. וּבַגְּמָרָא פְלִיגֵי בָהּ. וְהָרַמְבַּ"ם פּוֹסֵק כְּמַאן דְּאָמַר דְּכוֹתֵב בִּתִּי קְנוּיָה לְךָ: וְהַךְ קְרָא בִּמְכָרוּהוּ בֵּית דִּין כְּתִיב, אֲבָל בְּמוֹכֵר עַצְמוֹ דִּינוֹ שֶׁמּוֹכֵר עַצְמוֹ אֲפִלּוּ לְעֶשְׂרִים שָׁנָה, וְאִם פָּגַע בּוֹ יוֹבֵל אֲפִלּוּ אַחַר שָׁנָה אַחַת יוֹצֵא, שֶׁנֶּאֱמַר עַד שְׁנַת הַיּוֹבֵל יַעֲבֹד עִמָּךְ. וְנִרְאֶה לִי דְּהוֹאִיל וּמִכָּל מָקוֹם יוֹצֵא בַּשָּׁנִים שֶׁקָּצַב בִּמְכִירָתוֹ, שַׁפִּיר תָּנֵי סְתָם דְּעֶבֶד עִבְרִי יוֹצֵא בַּשָּׁנִים, וְאֶלָּא מִיהָא הָא כִדְאִיתָא וְהָא כִדְאִיתָא: וּבַיּוֹבֵל. בַּגְּמָרָא יָלֵיף לְהוּ מִקְּרָא: וּבְגִרְעוֹן כֶּסֶף. תָּנָא וְקוֹנֶה אֶת עַצְמוֹ בְכֶסֶף וּבְשָׁוֶה כֶסֶף. וְיָלֵיף לֵיהּ מִקְּרָא כוּ'. וְדַע דְּבַבְּרַיְתָא תָנֵי וּבִשְׁטָר, כְּלוֹמַר שְׁטַר שִׁחְרוּר. וְכָתְבוּ הַתּוֹסָפוֹת דְּתַנָּא דְמַתְנִיתִין לֹא חָשֵׁיב אֶלָּא יְצִיאוֹת שֶׁהֵם בְּעַל כָּרְחוֹ שֶׁל הָאָדוֹן. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: מֵהֶקֵּישָׁא דְעִבְרִי וְעִבְרִיָּה. הָרַמְבַּ"ם: וְכֵיוָן דְּאִיתָא נַמִּי בָּאִישׁ לֹא קָתָנֵי. גְּמָרָא:

בְּכֶסֶף כוּ'. בַּגְּמָרָא דְּהוּא הַדִּין בַּחֲלִיפִין וּבִמְשִׁיכָה, וּמִלְּתָא דְאִיתָא בְּמִטַּלְטְלֵי לֹא קָתָנֵי: וְהַאי קְרָא בְפָרָשַׁת בְּהַר כְּתִיב: בְּכֶסֶף. וּבַיּוֹבֵל לֹא. וְאַף עַל גַּב דְּאִתַּקִּישׁ לְקַרְקָעוֹת לִשְׂדֵה אֲחֻזָּה, וּשְׂדֵה אֲחֻזָּה יוֹצֵאת בַּיּוֹבֵל כַּמְּפֹרָשׁ בַּכָּתוּב, בְּעֶבֶד כְּתִיב לְעוֹלָם בָּהֶם תַּעֲבֹדוּ, כִּדְאִיתָא בַגְּמָרָא:

וְעִקֵּם הַתַּנָּא לְשׁוֹנוֹ לוֹמַר דְּמִצְוֹת הַבֵּן הֵם, שֶׁאִם לֹא עֲשָׂאָן לוֹ אָבִיו חַיָּב הוּא בְעַצְמוֹ לַעֲשׂוֹתָן כִּדְאִיתָא בַגְּמָרָא. וּבַסֵּיפָא אַגַּב רֵישָׁא נָקַט הַאי לִישָׁנָא: כֻּלְּהוּ יָלְפִינַן לְהוּ בַגְּמָרָא מִקְּרָאֵי. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: חַיּוּתֵיהּ הוּא, שֶׁמָּא יִפְרֹשׁ בַּסְּפִינָה וְתִטְבַּע וְיִסְתַּכֵּן אִם אֵינוֹ יוֹדֵעַ לָשׁוּט: כְּלָלִים הֵם וּלְדֻגְמָא נַקְטִינְהוּ, כְּמוֹ שֶׁכָּתַב הָרַמְבַּ"ם: אִם הָיָה חוֹלֵק עִם אַחֵר בִּדְבַר הֲלָכָה לֹא יֹאמַר נִרְאִים לִי דִבְרֵי פְלוֹנִי. רַשִׁ"י. וְהָרָמָ"ה כָּתַב, זֶה אֵין צָרִיךְ, דְּהַיְנוּ הַחוֹלְקִים יָשִׁיב. טוּר: חַיָּבִין. דִּכְתִיב תִּירָאוּ. וּמַה תַּלְמוּד לוֹמַר אִישׁ, שֶׁהוּא סִפֵּק בְּיָדוֹ לַעֲשׂוֹת (שֶׁאֵין מוֹחֶה בְיָדוֹ). אִשָּׁה רְשׁוּת אֲחֵרִים (בַּעְלָהּ. רַשִׁ"י) עָלֶיהָ. גְּמָרָא: וְאַף דְּבַיְרוּשַׁלְמִי אָמְרִינַן כְּסוּת יוֹם שֶׁלְּבָשָׁהּ בַּלַּיְלָה חַיֶּבֶת בְּצִיצִית, אֲפִלּוּ הָכִי כֵּיוָן שֶׁחִיּוּבוֹ תָלוּי בְּמַה שֶּׁהַזְּמַן גָּרַם לְבִישָׁתוֹ
