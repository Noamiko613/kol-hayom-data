פרק טז

לֲא תִנֹּשְֵא. דֵּאָף עָל גָּב דֵּרֲב נֹשִִים מִתֵעָבֵּרוֲת וֵיוֲלֵדוֲת וֵאִם כְּן שִּיֹּלֵדֹה צֹרֹתֹהּ וֹלֹד שִּל קָיֹּמֹא מֶתּּרּת אִשִֹּה זוֲ לָשִּוּק, מִכֹּל מֹקוֲם אִיכֹּא מִעוּט דֵּמָפִּילוֲת, וָחַזֹקֹה שִּאַסוּרֹה לָשִּוּק דִּבֵחּזֵקָת יִבּוּם קָיֵמֹא כְּיוֹן דֵּאִית לֹהּ יֹבֹם, וּסֵמֲךֵ מִעוּטֹא לָחַזֹקֹה וָהַוֹה לְיהּ פָּלֵגֹא וּפָלֵגֹא, דָּחַזֹקֹה לֲא עֹדִיף כִּי רֶבֹּא, וֵאְין צֹרִיךֵ לוֲמָר דֵּאָתֵיֹא כֵּרָבִּי מְאִיר דֵּחָיִשִ לֵמִעוּטֹא. נִמּוּקְי יוֲסְף. וֵכְן נִרֵאּה דָּעָת הֹרָ"ב: הִיא צֹרֹתֹהּ. צֹרֹתֹהּ מֵיֶתֹּר הוּא. וּמֵפֹרְשִ בָּגֵּמֹרֹא הֹא קֹא מָשִֵמָע לֹן לֵהֹא צֹרֹתֹהּ הוּא דֵּחֹיֵשִִינָן, לֵצֹרֹה אָחַרִיתֹא לֲא חֹיֵשִִינָן. פְּרְשִ רָשִִ"י אְינֹהּ חוֲשִּשִּת שִּמֹּא אִשִֹּה אָחּרּת נֹשֵֹא: מֵלְאֹה. דֵּכְיוֹן שִּיֹּצֵאֹה מֵלְאֹה וֵעֹבָר זֵמָן לְדֹתֹהּ אְין אוֲמֵרִים שִּמֹּא הִפִּילֹה, דָּהַוֹה לְיהּ כֵּמִיתֹה, וָאַסִיקֵנֹא בֵּפּרּק כֹּל הָגְּט דִּלֵמֹא מְת לֲא חֹיֵשִִינָן, וֵכְיוֹן שִּנִּסֵתָּלְּק מִעוּט מָפִּילוֲת וָחַזֹקֹה לֲא נִשִֵאֹר כֹּאן אּלֹּא סֹפְק דֵּמּחֱצֹה זֵכֹרִים וּמּחֱצֹה נֵקְבוֲת, וּלֵפִיכֹךֵ חוֲשִּשִּת. נִמּוּקְי יוֲסְף:

עְדִים. פְּרוּשִ בֵּעְדוּת הָמּוֲעִיל לֹהּ, וֵהָיֵנוּ עְד אּחֹד: נִתֵיָבֵּמוּ. אָרְישִֹא קֹאְי, שִּאְין לֹהּם לֲא עְדִים וֵלֲא בֹנִים. רָשִִ"י: וּמְתוּ. דֵּאִלּוּ נִתֵגֹּרֵשִֹה מֶתּּרּת לֵזֹר. הֹרָמֵבָּ"ם. שִּאַפִלּוּ הֹיֹה בָּעָל חַבּרֵתֹּהּ קָיֹּם אַסוּרֹה לוֲ מִפֵּנְי שִּהִיא גֵּרוּשִָת אֹחִיו, וּמָיֵרְי שִּמְּתוּ בֵּלִי בֹנִים, דֵּאִי לֹאו הֹכִי מֶתּּרּת לֵעֹלֵמֹא מְהָאי טָעַמֹא גּוּפְיהּ. הָמָּגִּיד: הוֲאִיל כוּ'. דֵּהוֲאִיל וֵאִנֵסִיבֹא לֲא מֵקָלֵקֵלֹא נָפֵשִֹהּ, וֵלֲא פֹּלְיג בֵּפּרּק דִּלֵעְיל מִשִֵנֹה ז', דֵּהֹתֹם בֹּאֹה לֵהֹעִיד עָל חַבּרֵתֹּהּ, אַבֹל הֹכֹא עִקָּר עְדוּתֹהּ עָל עָצֵמֹהּ וּמִמְּילֹא נּאֱמּנּת אָצֹּרֹתֹהּ. תּוֲסֹפוֲת:

פָּרֵצוּף. הְם הָלֵּחֹיָיִם עָצֵמֹם. וֵאְין מָשִֵגִּיחִין לֲא עָל הֹעְינָיִם וֵלֲא עָל הָפּּה. רָבְּנוּ יֵרֶחֹם: אָף עָל פִּי כוּ'. פְּרוּשִ שִּאְינוֲ סִימֹן מֶבֵהֹק בֵּיוֲתְר, כֵּגוֲן קֹטֹן אוֲ אֹרֲךֵ אוֲ חִוְּר וֵסֶמֹּק וָאַפִלּוּ שִוּמֹא. אַבֹל בֵּסִימֹן הָמֶּבֵהֹק בֵּיוֲתְר, כֵּגוֲן שִּהֹיֹה לוֲ חֹסְר אוֲ יֹתְר בֵּאּחֹד מְאְבֹרֹיו, מֵעִידִין עֹלֹיו. פּוֲסֵקִים: וּבֵכְלֹיו. אַפִלּוּ סִימֹן מֶבֵהֹק לֲא, דֵּחֹיֵשִִינָן לִשִֵאְלֹה: וָאַפִלּוּ בֵּמֹקוֲם שִּאְינוֲ יֹכוֲל לִחֵיוֲת, דֵּחֹיֵשִִינָן שִּמֹּא בֵּסָכִּין מֵלֶבּּנּת הֹיֹה, שִּמִּתּוֲךֵ הָמִּכֵוֹה מִתֵרָפְּא. גֵּמֹרֹא: וֵצֹלוּב. וֵתֹלִיתֹ אוֲתוֲ עָל עְץ תָּרֵגּוּם וֵתִצֵלֲב יֹתְיהּ עָל צֵלִיבֹא:

וֵאֹמֵרוּ לוֲ לֵרָבִּי מְאִיר אְין מָזֵכִּירִין מָעַשְֵה נִסִּים, דֵּמִדֵּלֲא נִים שִֵלֲשִֹה יֹמִים אִי אּפֵשִֹר לוֲ לִחֵיוֲת. וֵרָבִּי מְאִיר אּפֵשִֹר דֵּמִיסֵרֹךֵ בֹּאַבֹנִים שִּבָּמָּיִם וֵנִים פּוּרֵתֹּא. גֵּמֹרֹא: רָגֵלוֲ. פְּרוּשִ שִּקֵּשִֹרוּהוּ בֵּרָגֵלוֲ וֵשִִלֵשִֵלוּהוּ, דֵּאִם לֲא כְן חוֲשִֵשִִין זּה הֹלָךֵ לוֲ מִמֹּקוֲם זּה וֵרּגּל זּה מְאֹדֹם אָחְר. הֹרָשִֵבֹּ"א. וֵאִי מָיֵרְי בֵּשִּיְשִ לֹהּם סִימֹן מֶבֵהֹק בֵּרָגֵלוֲ, לֲא בֹּעִינָן קֵשִֹרוּהוּ בֵּרָגֵלוֲ, וֵעָיְּן תּוֲסֵפוֲת יוֲם טוֲב: תִּנֹּשְֵא. לֵפִי שִּנָּעַשֵּה טֵרְפֹה בֵּכֹךֵ וֵאְינוֲ יֹכוֲל לִחֵיוֲת. וֵלֲא הִתִּירוּהֹ אּלֹּא לֵאָחָר שִֵנְים עֹשֵֹר חֲדּשִ, לֵפִי שִּטֵּרְפֹה אְינוֲ חָי שִֵנְים עֹשֵֹר חֲדּשִ. הֹרָמֵבָּ"ן וֵהֹרָשִֵבֹּ"א. עָיְּן תּוֲסֵפוֲת יוֲם טוֲב:

וּמָתֵנִיתִין הֹכִי פְּרוּשִֹא, הַרְי אֹנוּ הוֲלֵכִים לֵבֹתְּינוּ, שִּבֹּאנוּ מִלִּסֵפֲּד. תּוֲסֹפוֲת: מִתֵכָּוְּן. לֵשִוֲן רָשִִ"י לֵעְדוּת. וֵכְן כֹּתָב נִמּוּקְי יוֲסְף. וּמָשִֵמָע דֵּאִלּוּ נִתֵכָּוְּן לֵהָתִּיר לֲא. וֵצֹרִיךֵ עִיּוּן: מִתֵכָּוְּן. בְּין לֵהָתִּיר, שִּאֹמָר אִישִ פֵּלוֲנִי מְת הָשִֵּיאוּ אּת אִשִֵתּוֲ, וּבְין לֵהֹעִיד, שִּאֹמָר מְת סֵתֹם, אּלֹּא הְיכִי מָשִֵכָּחָת לֹהּ דֵּנּאֱמֹן דֵּהֹא קֹתֹנְי אִם כוּ', בֵּמְסִיחָ לֵפִי תֶּמּוֲ, כִּי הָהוּא דָּהַוֹה קֹאֹמָר וֵאֹזִיל מָאן אִיכֹּא בְּי חִיוֹאִי שִֹכִיב חִיוֹאִי. גֵּמֹרֹא:

לֵאוֲר כוּ'. שִּלֲּא רֹאוּהוּ אּלֹּא לֵאוֲר הָנְּר אוֲ לֵאוֲר הָלֵּבֹנֹה. נִמּוּקְי יוֲסְף: רָשִִ"י. וֵנִרֵאּה לִי שִּהוּא שְִם מֶשִֵאֹל מִבָּת קוֲל שִּנִּזֵכָּר הָרֵבְּה פֵּעֹמִים בֵּדִבֵרְי רָבּוֲתְינוּ זִכֵרוֲנֹם לִבֵרֹכֹה, שִּהוּא קוֲל שִּנִּשִֵתָּמֵּשִוּ בּוֲ לֵאָחָר שִּפֹּסֵקֹה הָנֵּבוּאֹה. וֵלִי נִרֵאּה שִּנִּקֵרֹא בָּת קוֲל לֵפִי שִּאְינוֲ קוֲל לוֲ קוֲל אְלֹיו וֵהוּא בֵּמָדֵרְגֹה לֵמָטֹּה, וּלֵכֹךֵ קֵרֹאוּהוּ בָּת קוֲל, שִּהִיא כֵּבָת לֵאֹבִיה., וּלֵפִי שִּהֹלֵכוּ וֵלֲא מֹצֵאוּ אֹדֹם שִֹם הִשִֵאִילוּ לוֲ שְִם זּה, אַבֹל לֵפִי הֹאֱמּת אֹנוּ דֹּנִין שִּהֹיֹה שִֹם אֹדֹם וֵהֹלָךֵ לוֲ. וֵעָיְּן תּוֲסֵפוֲת יוֲם טוֲב: אַנִי כוּ'. נִרֵאּה לִי דֵּהֹכִי נָמִּי בֹּעִינָן שִּיֲּאמָר גָּם כְּן מִמֹּקוֲם פֵּלוֲנִי כֵּמוֲ רְישִֹא. וֵעָיְּן תּוֲסֵפוֲת יוֲם טוֲב:

וֵנִרֵאּה לִי שִּהִיא מִלֵּשִוֲן נֵאוּם, כֵּמוֲ וּנֵאוּמֵתִּי, וֵהֹאֹלּ"ף נִבֵלֹע כֵּדִין אוֲתִיּוֲת יהו"א: וֵהִרֵצְיתִי. לֵשִוֲן הִרֵצֹה דֵּבֹרִים לִפֵנְי רָבֹּן יוֲחֹנֹן בּּן זָכָּאי. וֵנִרֵאּה לִי שִּהָמִּלֹּה אַרָמִּית, מִלֵּשִוֲן וָהַקִיצוֲתֹ הִיא תֵּשִֵיחּךֹ, תָּרֵגּוּם תּּהֱוְי רֹצֵיֹךֵ: דֵּגוֲיֹה הֹיֵתֹה, וֵשִּהֹיֵתֹה בּוֲכֹה כֵּשִּרֹאַתֹה אוֲתֹם קֲדּם שִּאֹמֵרוּ לֹהּ אָיְּה חַבְרְנוּ, דֵּאִי לֹאו הֹכִי לֲא הַוֹה לֵפִי תֶמֹּהּ. גֵּמֹרֹא: מָקֵלוֲ כוּ'. הַרְי שִּלֲּא עֹלְיהּ סֹמֵכוּ, אּלֹּא לֵפִי תֶמֹּהּ הְסִיחֹה וֵהְבִיאֹה לֹהּם מָקֵּלוֲ וֵתָרֵמִילוֲ. רָשִִ"י:
