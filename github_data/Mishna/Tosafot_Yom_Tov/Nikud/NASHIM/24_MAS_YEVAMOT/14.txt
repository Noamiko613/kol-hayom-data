פרק יד

יְקַיֵּם. הָא פְּשִׁיטָא, אֶלָּא אַגַּב דְּתָנֵי יוֹצִיא תָּנָא יְקַיֵּם. תּוֹסָפוֹת: מַה טַּעַם קָאָמַר, כְּלוֹמַר כְּקִדּוּשִׁין. רַשִׁ"י וְנִמּוּקֵי יוֹסֵף: וְהַיְנוּ טַעֲמָא דְּהָכָא לֹא קָתָנֵי עוֹלָמִית כְּמוֹ בְּנִשְׁתַּטָּה הוּא דְּקָתָנֵי עוֹלָמִית. גְּמָרָא. וְכָתַב נִמּוּקֵי יוֹסֵף לְפִי שֶׁאֵין גֵּרוּשָׁיו כְּלוּם מִן הַתּוֹרָה קָתָנֵי הַאי לִישָׁנָא, שֶׁאֵין לַגֵּרוּשִׁין תַּקָּנָה עוֹלָמִית, שֶׁאֲפִלּוּ נִשֵּׂאת בְּגֵט זֶה וְיֵשׁ לָהּ בָּנִים תֵּצֵא, מַה שֶּׁאֵין כֵּן בְּחֵרֶשֶׁת דִּבְדִיעֲבַד אִם נִשֵּׂאת לֹא תֵּצֵא וְאַף עַל פִּי שֶׁאֵין לָהּ בָּנִים. וְאֶפְשָׁר שֶׁאִם נִתְגָּרְשָׁה תִּנָּשֵׂא אֲפִלּוּ לְכַתְּחִלָּה. הָרִיטְבָּ"א וְנִמּוּקֵי יוֹסֵף: מִפְּנֵי מָה כוּ'. מְפָרֵשׁ בִּגְמָרָא דִּלְדִבְרֵיהֶם קָאָמַר, לְדִידִי כִּי הֵיכִי דְּאִישׁ לֹא מָצֵי מְגָרֵשׁ, אִשָּׁה לֹא מָצֵי מְגָרְשָׁהּ, אֶלָּא לְדִידְכוּ מַאי שְׁנָא אִשָּׁה וּמַאי שְׁנָא אִישׁ:

הֵעִיד כוּ'. רַבָּנָן הֱבִיאוּהָ לְבֵית הַמִּדְרָשׁ לְהָבִיא רְאָיָה לְדִבְרֵיהֶן, וְאָמְרוּ לוֹ אַף זוֹ כַּיּוֹצֵא בָּהּ. תּוֹסָפוֹת: וּרְצוֹנוֹ לוֹמַר כְּשֶׁעֲדַיִן לֹא בָּגְרָה, דְּהָא אַף בְּנַעֲרוּתָהּ יָכוֹל אָבִיהָ לְקַדְּשָׁהּ:

לִשְׁנֵי כוּ'. הַיְנוּ רֵישָׁא שְׁנֵי אַחִים חֵרְשִׁים כוּ' לִשְׁתֵּי אֲחָיוֹת חֵרְשׁוֹת, אֶלָּא דְּתָנֵי כֻּלֵּי גַּוְנֵי בְּדִידֵיהּ וּבְדִידָהּ: וְאַף עַל גַּב דִּקְרִיאָה אֵינָהּ מְעַכֶּבֶת כְּדִתְנַן בְּפֶרֶק י"ב מִשְׁנָה ג', כָּל הָרָאוּי לְבִלָּה אֵין בִּלָּה מְעַכֶּבֶת בּוֹ וְכָל שֶׁאֵין רָאוּי לְבִלָּה בִּלָּה מְעַכֶּבֶת בּוֹ: רַשִׁ"י. וְהָכִי קָאָמַר, דְּמַפְקָע נִשּׂוּאִין דִּידֵיהּ שֶׁהָיוּ מִכֹּחַ זִקַּת אָחִיו דְּהַוְיָא בִּרְמִיזָה, אֲבָל זִקָּה עַצְמָהּ אֵין גֵּט דּוֹחֶה:

מַה יַּעֲשֶׂה חֵרֵשׁ כוּ'. וּפָרְכִינַן בַּגְּמָרָא וְתֵיתֵב גַּבֵּיהּ, קָטָן אוֹכֵל אִסּוּרִין הוּא, וְאֵין בֵּית דִּין מְצֻוִּין לְהַפְרִישׁוֹ. וּמְשָׁנֵי, גְּזֵרָה מִשּׁוּם הַתָּרַת יְבָמָהּ לַשּׁוּק שֶׁיֹּאמְרוּ נִפְטְרָה מִשּׁוּם אֲחוֹת אִשָּׁה, וְלֹא יָדְעֵי אֱנָשֵׁי דְּטַעֲמָא מִשּׁוּם קָטָן אוֹכֵל אִסּוּרִין הוּא: רַשִׁ"י. וּרְצוֹנוֹ לוֹמַר דְּלֹא מֻפְקָע נִשּׂוּאָיו שֶׁהָיוּ מִכֹּחַ זִקָּה כוּ':
