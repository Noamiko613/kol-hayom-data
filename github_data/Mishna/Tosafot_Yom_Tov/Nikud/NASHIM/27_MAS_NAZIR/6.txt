פרק ו

הָעֲנָבִים. נַמִּי מְחַיֵּב עַל זַג וְחַרְצָן. אֶלָּא הָא אֲתָא לְאַשְׁמְעִינַן דְּאֵינוֹ חַיָּב אֶלָּא בְכַזַּיִת, וְהַאי דְקָתָנֵי עֲנָבִים לַאֲפוֹקֵי מֵרַבִּי אֱלִיעֶזֶר דְּאָמַר אֲפִלּוּ עָלִין וְלוּלָבִין. רַשִׁ"י: וְהַיְנוּ כְּמִשְׁנָה הָאַחֲרוֹנָה. וְרַבִּי עֲקִיבָא נַמִּי הָכִי סְבִירָא לֵיהּ, דְּאָמַר כְּדֵי לְצָרֵף כַּזַּיִת. וְהַתּוֹסָפוֹת כָּתְבוּ דְּמַתְנִיתִין אַלִּיבָּא דְּכֻלֵּי עָלְמָא, דְּבַאֲכִילָה מוֹדוּ כֻּלֵּי עָלְמָא דִבְכַזַּיִת, וְלֹא פְלִיגֵי אֶלָּא בִשְׁתִיָּה, דְּמִשְׁנָה רִאשׁוֹנָה עַד שֶׁיִּשְׁתֶּה רְבִיעִית יַיִן כְּדֶרֶךְ כָּל שְׁתִיָּה וְלֹא גָּמַר שְׁתִיָּה מֵאֲכִילָה, וְרַבִּי עֲקִיבָא גָּמַר כוּ'. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב אֵיךְ מְשַׁעֲרִין שְׁתִיָּה בְּכַזַּיִת וַאֲכִילָה בִּרְבִיעִית: בַּגְּמָרָא יָלֵיף לֵיהּ מִקְּרָא.וְרַבָּנָן דָּרְשִׁי לֵיהּ לִדְרָשָׁא אַחֲרִיתִי. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב:

בִּפְנֵי עַצְמוֹ. וְנִרְאֶה דְעִקָּר חִדּוּשׁ לְתַנָּא קַמָּא בָּא לוֹמַר חַיָּב עַל הַזַּגִּים בִּפְנֵי עַצְמוֹ אִם אָכַל לְבַד מִן הַזַּגִּים, וְאַהַהִיא אָתָא לְאַפְלוֹגֵי רַבִּי אֶלְעָזָר בֶּן עֲזַרְיָה כוּ'. תּוֹסָפוֹת: וְעַל שְׁנֵיהֶם הַכָּתוּב אוֹמֵר זַג אֶחָד: אַף עַל גַּב דִּלְפִי לְשׁוֹן הַמִּשְׁנָה אֵינוֹ חַיָּב עַד שֶׁיֹּאכַל כוּ' מַשְׁמַע דִּלְהָקֵל בָּא וּכְלוֹמַר שֶׁאֲפִלּוּ אָכַל כַּזַּיִת מֵחַרְצַנִּים אוֹ מִן הַזַּגִּים אֵינוֹ חַיָּב עוֹד כוּ'. אֲבָל סוֹבֵר הָרַ"ב שֶׁכְּמוֹ שֶׁיֵּשׁ בְּדִבְרֵי רַבִּי אֶלְעָזָר בֶּן עֲזַרְיָה לְהָקֵל כֵּן יֵשׁ בִּדְבָרָיו לְהַחְמִיר, שֶׁאִם אוֹכֵל חַרְצַנִּים וְזַגִּין שֶׁיִּהְיֶה חַיָּב אֲפִלּוּ אִם הוּא בְּפָחוֹת מִכַּזַּיִת, שֶׁסּוֹבֵר שֶׁהַתּוֹרָה הִקְפִּידָה שֶׁיִּהְיוּ חַרְצַנִּים שְׁנַיִם וְזַג אֶחָד וְשֶׁאִם אוֹכֵל כָּךְ הֲרֵי הוּא חַיָּב וַאֲפִלּוּ אֵין בָּהֶם כַּזַּיִת, וְיֵשׁ בַּדָּבָר לְהָקֵל וּלְהַחְמִיר, וּכְלַפֵּי הַךְ סְבָרָא הוּא דְּכָתַב שֶׁאֵין הֲלָכָה אֶלָּא אֵינוֹ לוֹקֶה עַד שֶׁיֹּאכַל כַּזַּיִת:

לוֹמַר שֶׁזֶּה שֶׁסּוֹתֵר הַיְנוּ אִם אֵין מִיּוֹם שֶׁגִּלַּח עַד סוֹף יְמֵי נְזִירוּתוֹ שְׁלֹשִׁים יוֹם, כְּגוֹן שֶׁגִּלְּחוּהוּ בְּיוֹם מְלֹאת, צָרִיךְ שֶׁיַּמְתִּין שְׁלֹשִׁים יוֹם, אֲבָל אִם יֵשׁ לוֹ עוֹד שְׁלֹשִׁים יוֹם אֵינוֹ סוֹתֵר, כְּלוֹמַר כֵּיוָן דְּאִיכָּא עֲדַיִן גִּדּוּל שֵׂעָר שְׁלֹשִׁים יוֹם. וְכֵן כָּתְבוּ הַתּוֹסָפוֹת. וְדַעַת הָרַמְבַּ"ם דַּאֲפִלּוּ בִּנְזִירוּת מְרֻבָּה סוֹתֵר שְׁלֹשִׁים יוֹם וְכָל דִּקְדּוּקֵי נְזִירוּת עָלָיו, אֶלָּא שֶׁאֵין עוֹלִין לוֹ מִן הַמִּנְיָן: דַּהֲוָה לֵיהּ לְמִכְתַּב בַּתַּעַר לֹא יְגַלֵּחַ כוּ'. וּלְמַסְקָנַת הַגְּמָרָא אֵינוֹ חַיָּב אֶלָּא כְעֵין תַּעַר, לְפֵרוּשׁ רַשִׁ"י שֶׁמַּעֲבִירָן סָמוּךְ לְעִקָּרָן. וְהַתּוֹסָפוֹת כָּתְבוּ שֶׁעוֹקֵר הַשֵּׂעָר וּמַשְׁחִיתוֹ מֵעִקָּרוֹ. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: כְּלוֹמַר שֶׁלֹּא יַנִּיחַ מִמֶּנּוּ שֶׁיָּכוֹף רֹאשׁוֹ לְעִקָּרוֹ: כְּלוֹמַר וְלֹא בְּכָל אֲדָמָה אָמַר רַבִּי יִשְׁמָעֵאל וַאֲפִלּוּ שֶׁאֵינָהּ מַשֶּׁרֶת וְלִגְזֹר אַטּוּ מַשֶּׁרֶת, וְאַף עַל גַּב דְּבַגְּמָרָא אִבַּעְיָא לָן כוּ' פֵּרֵשׁ הָרַ"ב לְהָקֵל. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב:

אֶלָּא אֶחָת. אֵין זֶה אֶלָּא בְּדִינֵי אָדָם, שֶׁאֵינוֹ מְחֻיָּב מַלְקוֹת אֶלָּא אַחַר הַתְרָאָה, וְלָכֵן צָרִיךְ הַתְרָאָה עַל כָּל מַלְקוֹת. אֲבָל בְּדִינֵי שָׁמָיִם כָּל זְמַן שֶׁשָּׁתָה רְבִיעִית יַיִן עֲבֵרָה, וְעַל כָּל גִּלּוּחַ וְכָל טֻמְאָה עֲבֵרָה. וְעַל זֶה תַּעֲשֶׂה הֶקֵּשׁ לְכָל אִסּוּרִים שֶׁבַּתּוֹרָה, אֵין הֶפְרֵשׁ בֵּין לָאו אוֹ כָּרֵת אוֹ מִיתָה בְּסַנְהֶדְרִין, אֲבָל תֵּדַע שֶׁהַבָּא עַל הָעֶרְוָה חַיָּב עַל כָּל עֲבֵרָה וַעֲבֵרָה בִּכְלַל, וְאִם אֲנַחְנוּ לֹא נוּכַל לְהַעֲנִישׁ עָלָיו אֶלָּא עֹנֶשׁ אֶחָד הַקָּדוֹשׁ בָּרוּךְ הוּא מַעֲנִישׁ אוֹתוֹ עַל כָּל עֲבֵרָה וַעֲבֵרָה, וְאִם הֵמִיתוּ אוֹתוֹ בְּסַנְהֶדְרִין עַל אֶחָד מֵאֵלּוּ הָעֲבֵרוֹת וְהִתְוַדָּה נִתְכַּפֵּר עַל הַכֹּל מִצַּד הַתְּשׁוּבָה, לֹא מִצַּד שֶׁהָעֹנֶשׁ הָאֶחָד שֶׁמַּעֲנִישִׁין אוֹתוֹ הַסַּנְהֶדְרִין יְכַפֵּר עֲוֹנוֹת הַרְבֵּה. הָרַמְבַּ"ם:

אֲבָל בַּטֻּמְאָה לֹא אָמַר לֹא יִטַּמֵּא בִלְבַד אֶלָּא אָמַר לְאָבִיו וְגוֹ'. הָרַמְבַּ"ם. מַשְׁמַע דְּמִדְּסָתַם וְכָתַב מִיַּיִן וְשֵׁכָר יַזִּיר יָלְפִינַן וְהַכֹּל בִּכְלַל וַאֲפִלּוּ דְּמִצְוָה. הָרַמְבַּ"ם כוּ'. וּבַגְּמָרָא יַיִן דְּמִצְוָה מַאי הִיא, קִדּוּשָׁא וְאַבְדַּלְתָּא לֵיכָּא לְמֵימַר, שֶׁאַף עַל פִּי שֶׁקִּדּוּשׁ יוֹם הַשַּׁבָּת בִּדְבָרִים דְּבַר תּוֹרָה הוּא, זֶה שֶׁמְּקַדְּשִׁין עַל הַיַּיִן מִדִּבְרֵי סוֹפְרִים הוּא כוּ'. וּמְשָׁנֵי כְּגוֹן שֶׁנִּשְׁבַּע לִשְׁתּוֹת, וְקָא מַשְׁמַע לָן דְּחַיִּל עֲלֵיהּ נְזִירוּת אַף עַל גַּב שֶׁנִּשְׁבַּע לִשְׁתּוֹת: וְתֵימַהּ דְּהָא אִיכָּא נַמִּי עֲשֵׂה דְּגַדֵּל פֶּרַע שְׂעַר רֹאשׁוֹ וְאֵין עֲשֵׂה דּוֹחֶה לֹא תַעֲשֶׂה וַעֲשֵׂה. וּבַגְּמָרָא פָּרֵיךְ וְתִגְלַחַת לֹא תֻתַּר מִכְּלָלָהּ קַל וָחֹמֶר מִיַּיִן כוּ', וּמְשָׁנֵי אָמַר קְרָא רֹאשׁוֹ וְאָמַר קְרָא זְקָנוֹ. וּפֵרְשׁוּ הַתּוֹסָפוֹת דְּאִי לָאו קְרָא הֲוָה מְקַיְמִינַן הַגִּלּוּחַ בִּשְׁאָר גּוּפוֹ, אֲבָל רֹאשׁוֹ יִהְיֶה אָסוּר מִטַּעַם דְּאֵין עֲשֵׂה כוּ'. וְהָרַמְבַּ"ם מְיַשֵּׁב מִפְּנֵי שֶׁכְּבָר נִטְמָא הַנָּזִיר בְּצָרַעַת וִימֵי חִלּוּטוֹ אֵין עוֹלִין לוֹ, וַהֲרֵי אֵין קָדוֹשׁ בָּהֶן, וּבָטֵל הָעֲשֵׂה מֵאֵלָיו וְלֹא נִשְׁאַר אֶלָּא לֹא תַעֲשֶׂה. וְאֶפְשָׁר דִּסְבִירָא לֵיהּ כְּאִידָךְ תַּנָּא דְּרֹאשׁוֹ לַהַקָּפָה אָתֵי. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: וּבְמֵת מִצְוָה. תַּנְיָא אֵיזֶהוּ מֵת מִצְוָה, כָּל שֶׁאֵין לוֹ קוֹבְרִים. קוֹרֵא וַאֲחֵרִים עוֹנִים אוֹתוֹ אֵין זֶה מֵת מִצְוָה. גְּמָרָא:

אֶלָּא אִם כֵּן כוּ'. וְלִכְאוֹרָה קָאֵי אַמְּצֹרָע כוּ'. אִי נַמִּי יֵשׁ לְפָרֵשׁ דְּקָאֵי אַנָּזִיר, וְכֵן פֵּרֵשׁ רַשִׁ"י, וְכֵן נִרְאֶה מִן הָרַמְבַּ"ם בְּפֵרוּשׁוֹ וּבְחִבּוּרוֹ. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב:

וְשׁוֹחֵט כוּ'. לְשׁוֹן הַתּוֹסָפוֹת אַחֲרֵי שֶׁאָמַר הָכָא שֶׁעַל הַשְּׁלָמִים יְגַלֵּחַ דִּין הוּא שֶׁיַּקְדִּימוֹ, וְקָשֶׁה אִם כֵּן לָמָּה הִקְדִּים חַטָּאת וְעוֹלָה לַשְּׁלָמִים. אֲבָל הָרַ"ב הֶעְתִּיק שָׁחַט כוּ', יֵשׁ לוֹמַר דְּהָכִי קָאָמַר, אַשְּׁחִיטַת שְׁלָמִים מְגַלֵּחַ וְלֹא עַל הַקּוֹדְמִים, וְכֵן סִדּוּרָן בַּפָּרָשָׁה, בֵּין בַּהֲבָאָה בֵּין בָּעֲשִׂיָּה. וּלְפִיכָךְ נִרְאֶה לִי שֶׁכְּסִדְרָן בַּכָּתוּב כֵּן הוּא סִדְרָן, דִּמְנָלָן לְשַׁנּוֹת הַסֵּדֶר מִמַּה שֶּׁסְדוּרִין בַּכָּתוּב. וְכֵן הוּא לְשׁוֹן הָרַמְבַּ"ם בְּחִבּוּרוֹ. וּבְפֵרוּשׁוֹ פָּסַק כְּרַבִּי יְהוּדָה. וְלָשׁוֹן דִּבְכָל מָקוֹם שֶׁאָמַר רַבִּי אֱלִיעֶזֶר, לֹא אִיצְטְרִיךְ, אֶלָּא דְּהָאֱמֶת הוּא אוֹמֵר. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: וּמְגַלֵּחַ. שֶׁתִּהְיֶה תִגְלַחְתּוֹ אַחַר שְׁחִיטָתָן. הָרַמְבַּ"ם: דְּאִי פֶּתַח אֹהֶל מוֹעֵד מַמָּשׁ, אִי נַמִּי כְּנֶגֶד הַפֶּתַח, בִּזְיוֹן מִקְדָּשׁ הוּא, כִּדְאִיתָא בַּגְּמָרָא: וְאִם גִּלַּח כוּ'. אָתְיָא כְּכֻלֵּי עָלְמָא. וּבַיְרוּשַׁלְמִי יָלֵיף לַהּ מִקְּרָא:

הַיְנוּ בַּנָּזִיר דִּכְתִיב בֵּיהּ כִּבְשָׂה וְכֶבֶשׂ וְאַיִל אֲבָל בְּעָלְמָא רְאוּיָה הִיא לַשְּׁלָמִים: וּלְעֵיל דָּרְשִׁינַן דְּבִשְׁלָמִים הַכָּתוּב מְדַבֵּר וְהַכֹּל בַּמַּשְׁמָע: וְיָלֵיף לָהּ בַּגְּמָרָא מִזֶּבַח, לְרַבּוֹת חַטָּאת בְּנָזִיר טָהוֹר וְאָשָׁם בְּנָזִיר טָמֵא. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב:

אוֹ שׁוֹלְקָן. בְּפֶרֶק ב' דִּפְסָחִים מִשְׁנָה ו' פֵּרֵשׁ הָרַ"ב דִּשְׁלִיקָה בִּשּׁוּל יוֹתֵר מִדַּאי עַד שֶׁנִּמּוֹחַ, וְלֹא הָוֵי זוֹ וְאֵין צָרִיךְ לוֹמַר זוֹ, דְּקָא מַשְׁמַע לָן דְּבִשְׁלִיקָה אֵינוֹ יוֹצֵא מִתּוֹרַת בִּשּׁוּל: אַחַר קָרְבָּן וְאַחַר הַתִּגְלַחַת, וְתִגְלַחַת מְעַכֶּבֶת. רַשִׁ"י. וְהַיְנוּ דְּלֹא תָּנֵי הֻתַּר לְגַלֵּחַ לְפִי שֶׁכְּבָר גִּלַּח שֶׁהֲרֵי גִּלּוּחַ מְעַכֶּבֶת: דִּלְאַחַר גִּלּוּחַ עֲבִיד תְּנוּפָה, אַף כָּאן אַחַר מַעֲשֶׂה יְחִידִי אַחַר קָרְבָּן אַף עַל פִּי שֶׁעֲדַיִן לֹא גִּלַּח, וְתִגְלַחַת לֹא מְעַכֶּבֶת. רַשִׁ"י. וּבַגְּמָרָא אֵימָא עַד דְּאִיכָּא תַּרְוַיְהוּ, דְּמַה הָתָם אַחַר הַתִּגְלַחַת כְּדִכְתִיב אַחַר הִתְגַּלְּחוֹ, אַף כָּאן נֵימָא אַחַר הִתְגַּלְּחוֹ. וּמְשָׁנֵי אִי הָכִי גְּזֵרָה שָׁוָה לָמָּה לִי: הֻתַּר. מַשְׁמַע וַדַּאי מִיָּד הֻתַּר. וְעַיֵּן סוֹף פֶּרֶק ח': דְּשָׁנָה רַבִּי לִדְרַבִּי שִׁמְעוֹן בִּלְשׁוֹן חֲכָמִים לְקַמָּן . וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב:

לְפוּם רָהִיטָא כָּתַב כֵּן, דִּסְתָמָא תְּנַן לְעֵיל מִשְׁנָה ג' דְּסוֹתֵר שְׁלֹשִׁים יוֹם. וּמִיהוּ בַּגְּמָרָא אָמְרִינַן דִּלְרַבִּי אֱלִיעֶזֶר סוֹתֵר שִׁבְעָה יָמִים בִּלְבַד, וְטַעְמָא כִּי הֵיכִי דְּגַדְּלֵי שְׂעָרוֹת, דְּקִים לְהוּ לְרַבָּנָן כָּל שִׁבְעָה יוֹמִין אָתְיָא מַזְיָא כְּדֵי לָכוֹף רֹאשׁוֹ לְעָקְרוֹ: וְיָלֵיף לַהּ מִקְּרָא. וּפֵרוּשׁ יָצָא הַיְנוּ מִצְוַת גִּלּוּחוֹ. אֲבָל יְדֵי נִדְרוֹ דְּקָרְבַּן נָזִיר לֹא יָצָא, וּכְדִתְנַן בְּרֵישׁ זְבָחִים שֶׁלֹּא עָלוּ לַבְּעָלִים לְשֵׁם חוֹבָה. וְהַיְנוּ דְקָאָמַר אוֹתוֹ זֶבַח לֹא עָלָה לוֹ:

דְּהָא אָמַר כָּל לְאַחַר מְלֹאת שִׁבְעָה סוֹתֵר. גְּמָרָא: קְצָת יֵשׁ לְדַקְדֵּק מִדְּכָתַב וּמֻתָּר לְגַלֵּחַ שֶׁסּוֹבֵר כְּמוֹ שֶׁאִיתָא לְעֵיל בְּמִשְׁנָה ז' דִּלְרַבִּי יְהוּדָה עִקַּר מִצְוַת הַגִּלּוּחַ אַחַר הַשְּׁלָמִים שֶׁקָּרֵב בָּאַחֲרוֹנָה. אֲבָל מִכָּל מָקוֹם קָשֶׁה לָמָּה נַקְטֵיהּ כְּלָל, וְשָׁנֵי מִלִּישְׁנָא דְּרַבִּי אֶלְעָזָר וְרַבִּי שִׁמְעוֹן דְּלֹא נַקְטֵי דְּמִצְוָתוֹ קוֹדֶמֶת, וּלְרַבִּי אֶלְעָזָר מְעַכֵּב נַמִּי: שֶׁמֵּתָה. וְנִטְמֵאת עָלֶיהָ. רַשִׁ"י:
