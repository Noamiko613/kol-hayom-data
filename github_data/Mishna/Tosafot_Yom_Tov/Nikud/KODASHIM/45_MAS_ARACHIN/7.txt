פרק ז

וֵהִיא שֵֹדּה שִּיֹּרָשִ הֹאֹדֹם מִמּוֲרִישִֹיו. הֹרָמֵבָּ"ם: כֵּדִכֵתִיב אִם מִשִֵּנָת הָיּוֲבְל יָקֵדִּישִ שֵֹדְהוּ וֵגוֲ', מִשִֵּנָת וֵלֲא שִֵנָת הָיּוֲבְל בִּכֵלֹל, אָלֵמֹא בָּשִֹּנֹה שִּלֵּאָחָר הָיּוֲבְל קֹאְי קֵרֹא, וֵקֹאֹמָר יִתְּן חַמִשִִּים: וֵתְימָהּ, דֵּפֶנֵדֵיוֲן הוּא חַצִי מֹעֹה, וּתֵנָן בֵּסוֲף פּּרּק קָמֹּא דִּשִֵקֹלִים דִּלֵמָחַצִית שִּקּל קָלֵבּוֲן מֹעֹה כּּסּף וָחַכֹמִים אוֲמֵרִים חַצִי. וּמִפֵּנְי כְן כֹּתֵבוּ הָתּוֲסֹפוֲת דֵּלֹאו דָּוֵקֹא נֹקָט הֹכֹא לֵשִוֲן קָלֵבּוֲן, וֵאְינוֲ אּלֹּא לֵהָשִֵווֲת הָשִֹּנִים שִּיִּהֵיּה סּלָע וּפֶנֵדֵיוֲן לֵאָרֵבֹּעִים וֵתְשִָע שִֹנִים שִּל יוֲבְל שִּחֹסְר מְהּם פֶּנֵדֵיוֲן אּחֹד. וֵשִּמֹּא כֵשִּנּוֲתְן חַמִשִִּים שִּקּל בֵּיָחָד, אֹז אְינוֲ נוֲתְן פֶּנֵדֵיוֲן וֵלֲא שִָיֹּךֵ שִֹם הּכֵרְעָ, דָהַוְי כֵעְין מִקֹּח וּמִמֵכֹּר, וֵלֲא שִָיֹּךֵ הָכֵרֹעֹה אּלֹּא כֵּשִּמָּחֵלִיפִים מֹעוֲת לָשִֶּלֵחֹנִי: לֵפִי שִּאִם הִקֵדִּישִ, מֶקֵדּּשִּת בֵּלִי סֹפְק, וָאַפִלּוּ יוֲם אּחֹד לִפֵנְי הָיּוֲבְל. הֹרָמֵבָּ"ם: גֵּמֹרֹא. וֵלֹאו דָּוֵקֹא נֹקָט דֵּאָקֵדֵּשִֹא, דֵּלֲא בָהּקֵדְּשִ תָּלֵיֹא מִלֵּתֹא כִּי אִם בָּגֵּאֶלֹּה. תּוֲסֹפוֲת: שִּהָמָּקֵדִּישִ שֵֹדְהוּ בִּשִֵנָת הָיּוֲבְל אְינֹהּ מֶקֵדּּשִּת. וּבֵרֲב סֵפֹרִים גֹּרֵסִינָן בִּשִֵנָת הָיּוֲבְל. וֵעָיְּן תּוֲסֵפוֲת יוֲם טוֲב: חֲמּר. הֹכִי כֵתִיב קֵרֹא. וּפְרְשִ הָתּוֲסֹפוֲת חַדֹשִִים, דֵּקֹשִּה לוֲ אָמָּאי נֹקָט תָּנֹּא חֲמּר. הַוֹה לְיהּ לֵמְימָר כּוֲר כֵּלִישִֵנֹא דֵרָבֹּנֹן: גֵּמֹרֹא, דּוּמֵיֹא דִסֵלֹעִים. וּפְרְשִ רָשִִ"י מִשִּוּם דֵּזּרָע כֵּתִיב: תְּימָהּ, דִּלֵמָאי נָפֵקֹא מִנָּהּ מִתֵחָשִֵּבִין בֵּאָפְּי נָפֵשִָיֵהוּ כוּ'. אַבֹל הֹרָמֵבָּ"ם כֹּתָב וּמֵחָשִֵּבִין לֹהּ מָה שִּּרֹאוּי לֹהּם כוּ'. וּלֵפִיכֹךֵ נִרֵאּה לִי לֵהָגִּיהָּ בֵּדִבֵרְי הֹרָ"ב וֵנִפֵדִּין בֵּשִֹוֵיֹן וֵהָשִֵּאֹר לֵפִי חּשִֵבּוֲן. וֵעוֲד יְשִ לוֲמָר, דֵּדָעֵתּוֲ כֵדָעָת הָתּוֲסֹפוֲת דֵּהָנֹּפֵקֹא מִנָּהּ שִּלֲּא חֹשִֵבִינָן לֵהוּ הָמִּדֵרוֲן. וֵעָיְּן תּוֲסֵפוֲת יוֲם טוֲב: פְּרוּשִ, הָכּּסּף שִּהוּא רוֲצּה לִתְּן בָּפִּדֵיוֲן חִיֵּבוֲ הָכֹּתוּב לִתֵּנוֲ כֵאּחֹד. אַבֹל אִם לֲא יִרֵצּה לִפֵדּוֲת כֹּל הָשֵֹּדּה, רָשִָּאי, דֵּגוֲאְל לָחַצֹאִין כֵּדִתֵנָן בֵּפּרּק ט' מִשִֵנֹה ב':

יוֲצֵאֹה כוּ'. הֹכִי גֹרֵסִינָן בָּמִּשִֵנֹה בָּגֵּמֹרֹא. פְּרְשִ רָשִִ"י כִּי הְיכִי דָהַוֹה נֹפְיק מִיֹּד דֵּהָהוּא קֹרוֲב דֵּאֹתְי הָאי מְחַמֹתְיהּ. רָשִִ"י, וֵעָיְּן תּוֲסֵפוֲת יוֲם טוֲב: לֲא כוּ'. וֵדִין הוּא, בֵּשִּל אַחְרִים אַנִי זוֲכּה , בֵּשִּל עָצֵמִי לֲא כֹּל שִּכְּן. תָּלֵמוּד לוֲמָר אַחֶזֹּתוֲ , אַחֶזֹּה שִּלּוֲ , וֵאְין זוֲ שִּלּוֲ. הֹא כְיצָד, יוֲצֵאֹה מִתָּחָת יֹדוֲ וּמִתֵחָלּּקּת לֵאּחֹיו הָכֲּהַנִים. גֵּמֹרֹא:

הָכֲּהַנִים כוּ'. דֵּהֹכִי כֵתִיב וֵאִם לֲא יִגֵאָל אּת הָשֵֹּדּה וֵאִם מֹכָר אּת הָשֵֹּדּה וֵגוֲ' וֵהֹיֹה הָשֵֹּדּה בֵּצְאתוֲ בָיֲּבְל וֵגוֲ' לָכֲּהְן תִּהֵיּה אַחֶזֹּתוֲ, וֵהֹכִי קֹאֹמָר, שִּאִם לֲא יִגֵאָל דֵּהָיֵנוּ הָבֵּעֹלִים אוֲ אִם מֹכָר וֵגוֲ', בִּשִֵנְי דֵרֹכִים אְלּוּ, לֲא יִגֹּאְל עוֲד, אּלֹּא וֵהֹיֹה בֵּצְאתוֲ בָיֲּבְל וֵגוֲ' לָכֲּהְן. תּוֲסֹפוֲת: וֵאִם לֲא יִפֵדּּה יֹקוּם הָבָּיִת לָהּקֵדְּשִ. הֹרָמֵבָּ"ם: רֹצֹה לוֲמָר, שִּאְין נוֲתֵנִין הָכֲּהַנִים לָהּקֵדְּשִ דֹּמִים. הֹרָמֵבָּ"ם: בֵּבֹבֹא מֵצִיעֹא דָּף ל"ט כוּ' מֵפֹרֵשִִינָן, נֵטוּשִִין, בֵּעָל כֹּרֵחְיהּ, שִּנִּשִֵבֹּה אוֲ שִּבֹּרָח מְחַמָת סָכֹּנֹה. רֵטוּשִִין, מִדָּעֵתֹּן, דִּכֵתִיב אְם עָל בֹּנִים רֶטֹּשִֹה כוּ'. וֵהֹכֹא אָף עָל גָּב דִּבֵעָל כֹּרֵחְיהּ נִשִֵאַרֹה בֵיָד הּקֵדְּשִ, לֲא מֹצְי קֹרְי לֵהוּ נֵטוּשִִין, כְּיוֹן דֵּמִתֵּחִלֹּה אָקֵדֵּשִָהּ מִדָּעֵתְּיהּ. וֵטָעַמֹא שִּנִּקֵרְאת בֵּשְִם זּה, כֵּדְי שִּתֵּהְא יֵדוּעֹה לָכֲּל, וֵלִכֵשִּיֹּבֲא אּחֹד וֵיִגֵאֹלּנֹּה וֵיָגִּיעָ אָחָר כֹּךֵ יוֲבְל תֵּהְא לָכֲּהַנִים:

לֹאו דָּוֵקֹא גֵאֹלֹהּ אָחְר וֵשִּאִם לֲא גֵאֹלֹהּ אָחְר לֲא, דֵּאִם כְּן אֹתֵיֹא כֵּרָבִּי אֱלִיעּזּר, וֵאְין הַלֹכֹה כֵּמוֲתוֲ. אּלֹּא הֹכִי קֹאֹמָר, וֵאִם לֲא גֵּאֹלֹהּ הוּא וֵאִי נָמִּי גֵאֹלֹהּ אָחְר: יֹצֹאת כוּ'. דֵּאִי לֲא רָבִּי לֹן קֵרֹא אּלֹּא מְת אֹבִיו וֵאָחָר כֹּךֵ הִקֵדִּישִֹהּ, לִכֵתֲּב קֵרֹא אַשִּר לֲא שֵֵדְה אַחֶזֹּתוֲ, אִי נָמִּי אַשִּר לֲא שֵֵדְה אַחֶזֹּה, וּמִתֵרָבֵּיֹא הֹא דֵרָבִּי מְאִיר דֵּדֹרֵשִִינָן יֹצֵאֹה זוֲ שִּהִיא כֵּבֹר אַחֶזֹּה. מָאי מִשֵֵּדְה אַחֶזֹּתוֲ, שֵֹדּה שִּאְינֹהּ רֵאוּיֹה כוּ'. גֵּמֹרֹא. וֵעָיְּן תּוֲסֵפוֲת יוֲם טוֲב: מִקֵנֹה. שֵֹדּה שִּלֵּקֹחֹהּ אוֲ זֹכֹה בֹהּ, הִיא נִקֵרְאת שֵֵדְה מִקֵנֹה. הֹרָמֵבָּ"ם: וֵרָשִִ"י בֵּפְרוּשִ הָחֶמֹּשִ כֹּתָב, דָּאַפִלּוּ לֲא גֵאֹלֹהּ אָחְר אּלֹּא שִּעַדָיִן הִיא בֵּיָד הָגִּזֵבֹּר, תְּצְא בֵּחִנֹּם בָּיּוֲבְל לָבֵּעֹלִים הֹרִאשִוֲנִים אַשִּר לוֲ אַחֶזָּת הֹאֹרּץ. וֵעָיְּן תּוֲסֵפוֲת יוֲם טוֲב: שִּאְין כוּ'. אָף עָל גָּב דִּקֵרֹא כֵתִיב בִּשִֵנָת הָיּוֲבְל יֹשִוּב הָשֵֹּדּה לָאַשִּר קֹנֹהוּ מְאִתּוֲ וֵגוֲ', אִצֵטֵרִיךֵ לֵטָעַמֹא, מִשִּוּם דֵּבִקֵרֹא לֲא מָיֵרְי אּלֹּא בֵמָקֵדִּישִ וּפוֲדְהוּ, דֵּהֹכִי כֵתִיב וֵחִשִָּב הָכֲּהְן וֵגוֲ' וֵנֹתָן אּת הֹעּרֵכֵּךֹ בָּיּוֲם הָהוּא קֲדּשִ לָה' בִּשִֵנָת הָיּוֲבְל יֹשִוּב הָשֵֹּדּה וֵגוֲ', וָאַנָן אֹיֵרִינָן בִּגֵאַלֹהּ אָחְר, וֵאִי נָמִּי לֲא נִגֵאַלֹה כֵלֹל:
