פרק ט

וּכְלוֹמַר דְּתָנֵי נַמִּי בֵּיהּ שְׁחִיטָה שֶׁאֵינָהּ רְאוּיָה: דַּאֲשֶׁר יִזָּרֵעַ לֹא אִצְטְרִיךְ, וּכְמוֹ שֶׁכָּתַבְתִּי בְּרֵישׁ עֻקְצִין. וְאַף עַל גַּב דְּלֹא שַׁיְּכָא אֶלָּא בִזְרָעִים, מַסְקִינַן בַּגְּמָרָא דִּתְלָתָא קְרָאֵי כְּתִיבֵי , חַד לְשׁוֹמֵר דִּזְרָעִים, וְחַד לְשׁוֹמֵר דְּאִילָנוֹת כְּלוֹמַר פֵּרוֹת דְּאִילָנוֹת, וְחַד לְשׁוֹמֵר בָּשָׂר בֵּיצִים וְדָגִים כוּ': פֵּרֵשׁ רַשִׁ"י, לֵיחָה הַיּוֹצֵא מִן הַבָּשָׂר שֶׁקּוֹרִין גלייר"א: פֵּרֵשׁ רַשִׁ"י, תְּבָלִין בְּאַנְפֵּי נַפְשַׁיְהוּ לֹא אָכְלֵי לְהוּ אֱינָשֵׁי. וּלְעֵיל פֶּרֶק ז' מִשְׁנָה ה' פֵּרֵשׁ דַּק דַּק: וְהָעֲצָמוֹת. שֶׁיֵּשׁ בָּהֶן מֹחַ וְהוּא אֹכֶל וְהָעֶצֶם שׁוֹמֵר לוֹ לְפִיכָךְ מִצְטָרֵף עִמּוֹ. רַשִׁ"י: וְהַגִּידִין. שֵׁם נוֹפֵל עַל הַגִּידִים הַדּוֹפְקִים, וְעַל שֶׁאֵין דּוֹפְקִים, וְעַל הַקְּשָׁרִים, וְעַל הַקְּרוּמִים, וּמֵתָרִים, וְהָעֲצַבִּים. הָרַמְבַּ"ם: וּמַשְׁמַע דְּאִלּוּ לְקַבֵּל טֻמְאָה מִתְטַמְּאִים אַף בְּפָחוֹת מִכַּבֵּיצָה. וְלֹא פֵרֵשׁ כֵּן בְּרֵישׁ פֶּרֶק ה' דִּתְרוּמוֹת: רַשִׁ"י. וּמְסַיֵּם וּבִטְהוֹרָה לְיִשְׂרָאֵל לֹא אִצְטְרִיךְ לְמִתְנֵי, דְּכָל שֶׁכֵּן דַּהֲוֵי אוֹכְלָא מִיָּד שֶׁשְּׁחָטָהּ: רַשִׁ"י. וְלִישְׁנָא דְסוּגְיָא דְּרֵישׁ פֶּרֶק ב' נָקְטֵי, וְשָׁם פֵּרֵשׁ רַשִׁ"י שֶׁלֹּא יַחְתֹּךְ כָּל הַמִּפְרֶקֶת לִשְׁנַיִם. וְעַיֵּן בְּמַסֶּכֶת כֵּלִים פֶּרֶק ד' מִשְׁנָה ב': כְּלוֹמַר בָּאֱלָל דְּרֵישָׁא קָאָמְרִינַן דִּבְכֻלְּהוּ אֱלָל קָאָמַר דְּמִצְטָרְפִין, וּבְאַפֵּי נַפְשַׁיְהוּ לֹא מְטַמְּאוּ. וְאָתָא רַבִּי יְהוּדָה לְמֵימַר בְּאֱלָל גִּיד מוֹדִינָא כוּ', אֲבָל בָּשָׂר שֶׁפְּלַטְתּוֹ כוּ'. רַשִׁ"י: וּבַגְּמָרָא, וְהוּא שֶׁכִּנְּסוֹ. וּפֵרֵשׁ רַשִׁ"י דְּגַלֵּי דַעְתֵּיהּ דְּלֹא בָטְלֵי מֵעִקָּרָא. אֲבָל נִתְכַּנֵּס מֵאֵלָיו, כְּגוֹן שֶׁכִּנְּסוּהוּ תִּינוֹקוֹת אַחַר הֶפְשֵׁט, לֹא, הוֹאִיל וּבָטֵל מִתְּחִלָּה וְנַעֲשָׂה עוֹר, תּוּ לֹא הָדַר הֲוֵי בָּשָׂר: רַשִׁ"י. דְּאִי אַאֲכִילָה מַאי בָּעֵי מְכֻנָּס דְּהַיְנוּ שֶׁכִּנְּסוֹ, הָא כֵּיוָן דַּאֲכָלוֹ אֵין כִּנּוּס גְּדוֹלָה מִזּוֹ. וְהָא דְלֹא נָקַט טָמֵא, דְּלֹא תֵימָא מִדְּרַבָּנָן, לְהָכִי קָאָמַר חַיָּבִים עָלָיו. תּוֹסָפוֹת:

עוֹר הָאָדָם. כְּדִמְפָרֵשׁ טַעֲמָא בְּרֵישׁ פֶּרֶק ז' דְּנִדָּה, דּוּמְיָא דְעֶצֶם, מַה עֶצֶם שֶׁנִּבְרָא עִמּוֹ וְאֵין גִּזְעוֹ מַחֲלִיף , אַף כָּל כוּ', וְעוֹר אֵין גִּזְעוֹ מַחֲלִיף, דִּמְקוֹמוֹ נַעֲשֶׂה צַלֶּקֶת . תּוֹסָפוֹת: וּבַגְּמָרָא, רְכוּבָה הַנִּמְכֶּרֶת עִם הָרֹאשׁ. וּפֵרֵשׁ רַשִׁ"י כָּל עוֹר שֶׁעַל עֶצֶם הַשּׁוֹק מֵרְכוּבָה הַתַּחְתּוֹנָה שֶׁחוֹתְכִים בִּשְׁעַת הַפְשָׁטַת בְּהֵמָה: כַּחֻלְדָּה. קַמָּא דְבַשְּׁרָצִים נָקֵיט: רַשִׁ"י. וְלִישְׁנָא דְמַתְנִיתִין שֶׁהִלֵּךְ, מַשְׁמַע דְּלֹא סַגִּי בִּשְׁטִיחָה בְּעַלְמָא. וְהָכִי מוּכָח בְּהֶדְיָא בַּגְּמָרָא, דְּאָמְרִינַן כַּמָּה כְּדֵי עֲבוֹדָה אַרְבָּעָה מִילִין. וְכֵן כָּתַב הָרַמְבַּ"ם: חוּץ כוּ'. וְאַף עַל גַּב דְּמֵת אָסוּר בַּהֲנָאָה וּבְכָל אָדָם אָסוּר לַעֲשׂוֹת מֵעוֹרוֹ שְׁטִיחִין, מִכָּל מָקוֹם לֹא הֲוֵי גָּזְרֵי רַבָּנָן מִשּׁוּם הַךְ חֲשָׁשָׁא שֶׁיְּהֵא טָמֵא, אֶלָּא מִשּׁוּם אָבִיו וְאִמּוֹ דְּחָמִיר טוּבָא. תּוֹסָפוֹת. אֲבָל בְּפֶרֶק ז' דְּנִדָּה כָּתְבוּ דְּעוֹר מֻתָּר בַּהֲנָאָה. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: אֲבָל לְעִנְיַן שַׁבָּת אַף תַּנָּא קַמָּא מוֹדֶה:

כְּלוֹמַר שֶׁאִם נִפְשַׁט יוֹתֵר מִשְּׁנֵי טְפָחִים וְנָגְעָה טֻמְאָה כוּ'. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: כְּלוֹמַר שֶׁהָעוֹר נִשְׁאַר כָּפוּל כְּמוֹת שֶׁהוּא וְלֹא שָׁטוּחַ. וּלְשׁוֹן הָרַמְבַּ"ם, עַד שֶׁיֵּצֵא כָל הָעוֹר שָׁלֵם בְּהֶקֵּפוֹ: רַשִׁ"י. וְאִם כֵּן לְרַבִּי יוֹחָנָן בֶּן נוּרִי הוּא דִמְפָרְשֵׁי הָכִי. וְהַיְנוּ דִּבְפֵרוּשׁ דִּבְרֵי רַבִּי יוֹחָנָן בֶּן נוּרִי מְסַיֵּם רַשִׁ"י, דְּהָא כֻּלּוֹ חִבּוּר דְּקָתָנֵי לָאו דַּוְקָא כֻּלּוֹ אֶלָּא עַד הֶחָזֶה. עַד כָּאן. וְיִתָּכֵן בְּעֵינַי שֶׁמְּפָרְשִׁים כֵּן כְּדֵי שֶׁלֹּא יְהֵא סְתָם וְאַחַר כָּךְ מַחֲלֹקֶת, דְּהַשְׁתָּא רַבִּי יוֹחָנָן בֶּן נוּרִי וַחֲכָמִים פְּלִיגֵי בְּפֵרוּשׁ דְּכֻלּוֹ חִבּוּר. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב:

וֶהֱסִיטָן. וְהוּא הַדִּין בְּלֹא תְחוּבִים וְנָגַע בָּהֶן דְּטָמֵא. דְּאִי לֹא תֵימָא הָכִי אֲפִלּוּ בְמַשָּׂא לֹא לִטָּמֵא, דְּאֶת שֶׁבָּא לִכְלַל מַגָּע בָּא לִכְלַל מַשָּׂא לְרַבִּי עֲקִיבָא, וְקָסָבַר יֵשׁ נוֹגֵעַ וְחוֹזֵר וְנוֹגֵעַ. אֶלָּא מִשּׁוּם דְּפָלֵיג רַבִּי עֲקִיבָא אַרַבִּי יִשְׁמָעֵאל בְּעוֹר שֶׁיֵּשׁ עָלָיו שְׁתֵּי חֲצָאֵי זֵיתִים, קָאָמַר דִּבְקֵיסָם כְּהַאי גַוְנָא מוֹדֶה לֵיהּ. תּוֹסָפוֹת. וְשִׁיטַת הָרַמְבַּ"ם בְּהֵפֶךְ. עַיֵּן תּוֹסְפוֹת יוֹם טוֹב:

לֹא מִשְּׁמֹנָה עָשָׂר דָּבָר. דְּהַהוּא בִּסְתָם יָדַיִם כְּמוֹ שֶׁכָּתַב הָרַ"ב בְּפֶרֶק קַמָּא דְשַׁבָּת, וְזוֹ בְּיָדַיִם שֶׁהֵן טְהוֹרִים בְּוַדַּאי סָמוּךְ לִנְטִילָתָן וְלֹא אַסַּח דַּעְתֵּיהּ, וּכְמוֹ שֶׁכָּתַבְתִּי בְּרֵישׁ פֶּרֶק ג' דְּיָדַיִם: קָאֵי אַפִּגּוּל. שֶׁלֹּא יְפַגְּלֵהוּ בְּרָצוֹן, כְּמוֹ שֶׁכָּתַבְתִּי בְּסוֹף פְּסָחִים: שֶׁנּוֹתַר בָּהֶן מֹחַ חוּץ לִזְמַנּוֹ וְשִׁמְּשׁוּהוּ עֲצָמוֹת הַלָּלוּ. רַשִׁ"י: וּבְשֶׁרֶץ נַמִּי דָּרֵישׁ בְּתוֹרַת כֹּהֲנִים פָּרָשַׁת שְׁמִינִי פֶּרֶק י', בְּנִבְלָתָם, לֹא מִן הָעֲצָמוֹת וְלֹא מִן הַשִּׁנַּיִם כוּ': הָכִי יָלֵיף לֵיהּ בַּגְּמָרָא:

זֶה לְשׁוֹן הָרַמְבַּ"ם, עִנְיַן מְרֻקֶּמֶת, שֶׁיֵּשׁ בָּהּ כְּבָר שִׂרְטוּט תֹּאַר בַּעֲלֵי חַיִּים וְאֵבָרָיו בְּתוֹךְ הַבֵּיצָה. כְּדִכְתִיב אֲשֶׁר עֻשֵּׂיתִי בַסֵּתֶר רֻקַּמְתִּי בְּתַחְתִּיּוֹת אָרֶץ . עַד כָּאן. וּבַגְּמָרָא מְרַבִּינַן לָהּ מֵהַטְּמֵאִים:

הַמְדֻלְדָּלִין. וְזֶה עַל מְנָת שֶׁיִּהְיוּ בְּעִנְיָן שֶׁאִי אֶפְשָׁר שֶׁיִּדָּבְקוּ וְלֹא שֶׁיֵּרָפְאוּ בְּשׁוּם פָּנִים. הָרַמְבַּ"ם: וְכֵן לְשׁוֹן רַשִׁ"י. דִּצְרִיכִים מַחֲשָׁבָה כְּדִתְנַן בְּמִשְׁנָה ב' פֶּרֶק ג' דְּעֻקְצִים. וְהָא דְפֵרְשׁוּ לְנָכְרִי, הוּא הַדִּין לְיִשְׂרָאֵל, דִּלְנָכְרִי אָסוּר אֵבָר מִן הַחַי כְּמוֹ לְיִשְׂרָאֵל. אֶלָּא דִּלְתַנָּא קַמָּא אֵין צָרִיךְ אֹכֶל שֶׁאַתָּה יָכוֹל כוּ'. וְלֹא פֵרְשׁוּ לְנָכְרִים אֶלָּא דְנָכְרִים אִיכָּא דְאָכְלֵי, אֲבָל בְּיִשְׂרָאֵל רְשִׁיעֵי לֹא עָסְקִינַן, דְּלֹא שְׁכִיחֵי: בַּגְּמָרָא יָלֵיף לָהּ מִקְּרָא: הֶכְשֵׁר. וְלֵיכָּא לְמֵימַר בְּהַאי אֵבָר שֶׁסּוֹפוֹ לְטַמֵּא טֻמְאָה חֲמוּרָה , דְּאֵינוֹ מְטַמֵּא טֻמְאַת אֵבָר מִן הַחַי עַד שֶׁיָּמוּת, וְשֶׁמָּא לֹא יָבֹא לִידֵי טֻמְאָה חֲמוּרָה דְּשֶׁמָּא יִשְׁחֲטֶנָּה וְאֵין שְׁחִיטָה עוֹשָׂה נִפּוּל. רַשִׁ"י: כְּלוֹמַר יְדֵי אֵבָר מִן הַחַי דִּמְטַמֵּא טֻמְאַת נְבֵלָה טָהֲרוּ: בַּגְּמָרָא יָלֵיף לְהוּ מִקְּרָא: בַּגְּמָרָא יָלֵיף לְהוּ מִקְּרָא: וְהוּא הַדִּין לַבָּשָׂר כָּל זְמַן כוּ', וּכְדִמְסַיֵּם לַאֲפוֹקֵי הָאֵבָר וְהַבָּשָׂר כוּ':

וְכֵן כָּתַב הָרַמְבַּ"ם. וְאַל תִּתְמַהּ, דִּגְדוֹלָה מִזּוֹ מָצִינוּ לְרַבִּי אֱלִיעֶזֶר שֶׁאָמַר שֶׁלֹּא אָמְרוּ עַל אֵבָר דִּמְטַמֵּא אֶלָּא אֵבָר מִן הַחַי, כְּדִתְנַן בְּפֶרֶק ו' דְּעֵדֻיּוֹת מִשְׁנָה ב'. וּמִכָּל מָקוֹם רַשִׁ"י וְתוֹסָפוֹת לֹא פֵרְשׁוּ כֵן. אֶלָּא דְּתַנָּא קַמָּא סְבִירָא לֵיהּ אוֹ כְּרַבִּי אֱלִיעֶזֶר אוֹ כְּרַבִּי נְחוּנְיָא דְּהָתָם מִשְׁנָה ג', וְרַבִּי שִׁמְעוֹן סְבִירָא לֵיהּ כְּרַבִּי יְהוֹשֻׁעַ דְּהָתָם. וְהָכִי מִסְתַּבֵּר, דְּלֹא לְהַפְלִיג בְּמַחֲלֹקֶת. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב:
