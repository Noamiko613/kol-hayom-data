יְשִ כוּ'. אְין מָאַמֹרוֲ בֵאְלּוּ הֹאָרֵבֹּעֹה חַלֹקִים עָל מִי שִּלֲּא קֵדֹמוֲ שִוּם דֹּבֹר בֵּשִוּם פֹּנִים שִּהוּא בֵכוֲר, לֵפִי שִּזּּה דֹבֹר מֵבֲאֹר, אַבֹל עָל מִי שִּהוּא בֵכוֲר וֵאָף עָל פִּי שִּקֹּדֵמֹה לוֲ לְידֹה, שִּאְין חוֲשִֵשִִין לֵאוֲתֹהּ קֵדִימֹה. הֹרָמֵבָּ"ם: שִּאַפִלּוּ נוֲלָד הָנְּפּל כֶּלּוֲ אְינוֲ מָפֵקִיעָ אּת הָבֹּא אָחַרֹיו, שִּאְין לִבּוֲ דֹוּה עֹלֹיו. וֵכֹל שִּכְּן אִם יֹצֹא רֲאשִוֲ מְת. וֵהָאי דֵּנֹקָט רֲאשִוֲ, מִשִּוּם בֵּכוֲר לָכֲּהְן נֹקָט לְיהּ, וּבִיצִיאָת רֲאשִוֲ הֹוְי יֹלוּד וּפוֲטְר אּת אֹחִיו מִבֵּכוֲר לָכֲּהְן. רָשִִ"י, וֵעָיְּן תּוֲסֵפוֲת יוֲם טוֲב: מָשִֵמָע דִּלֵיְשִ מֵפֹרֵשִִים נָמִּי סָנֵדֹּל חַתִיכָת בֹּשֵֹר. וֵלֲא מָשִֵמָע כְּן מִדֵּבֹרֹיו פּּרּק קָמֹּא דִּכֵרְתוֲת מִשִֵנֹה ג'. וֵעָיְּן תּוֲסֵפוֲת יוֲם טוֲב: הָבֹּא כוּ'. תָּנֵיְהּ מִשִּוּם בֹּבֹא דֵבֹתֵרֹא דֵפֹלְיג בֹּהּ רָבִּי יוֲסְי: וֵאְינוֲ כוּ'. דֵּאְינוֲ פּּטּר רּחּם. רָשִִ"י:

וֵנִרֵאּה עוֲד מִדֵּבֹרֹיו שִּבֵּהּכֵרְחָ תֹּמוּת: דּּרּךֵ לְידֹה. רָשִִ"י. וֵעָיְּן תּוֲסֵפוֲת יוֲם טוֲב: וּסֵבֹרֹא תֵמוּהֹה הִיא זוֲ, דֵּמִסֵתָּבֵּרֹא לֵשִוֲן לְידֹה מָשִֵמֹעוּתוֲ כֵדּרּךֵ לְידֹה שִּל כֹּל הֹעוֲלֹם. וּבָגֵּמֹרֹא, רָבִּי שִִמֵעוֲן לֵטָעַמְיהּ דֵּאֹמָר תְּלְד לֵרָבּוֲת יוֲצְא דֲפּן. וּפְרְשִ רָשִִ"י, לֵרָבּוֲת דָּהַוְי לְידֹה מֵעָלֵּיֹא וֵיוֲשִּבּת עֹלֹיו יֵמְי טֶמֵאֹה וֵטֹהֳרֹה, הִלֵכֹּךֵ גָּבְּי וֵיֹלֵדוּ לוֲ נָמִּי לְידֹה מֵעָלֵּיֹא חַשִִיב לְיהּ, דֵּגֹמֵרִינָן מְהֹתֹם:

וֵנֹקָט הֹאֹב מִשִּוּם סְיפֹא דֵבֹעְי לֵמִתֵנְי מְת הֹאֹב. רָשִִ"י: נֹתֵנוּ. כֵּלוֲמָר כָּדִּין נֹתֵנוּ. שִּהַרְי כֹּל שִּלֲּא חֹלֵקוּ הַרְי הָחוֲב עָל הָנֵּכֹסִים אָף לֵרָבִּי מְאִיר, וּמֵחֶיֹּבִים לֹתְת: וֵתְימָהּ, דֵּרָבִּי מְאִיר אִית לְיהּ בֵּרְרֹה . וֵיְשִ לֵחָלְּק בִּבֵרְרֹה הְיכֹא דֵמָתֵנְי בֵּפְרוּשִ כֵּגוֲן שִֵנְי לֶגִּין שִּאַנִי עֹתִיד לֵהָפֵרִישִ. תּוֲסֹפוֲת. וֵעָיְּן תּוֲסֵפוֲת יוֲם טוֲב: וֵכְן כֹּתָב הֹרָמֵבָּ"ם. וֵאִיכֹּא לֵמְידָק טוּבֹא, דֵּהֹא בֵמִשִֵנֹה ו' מֵפֹרְשִ דֵּרָבִּי יֵהוּדֹה סֹבָר מִלֵוּה הָכֵּתוּבֹה בָּתּוֲרֹה כִּכֵתוּבֹה בִּשִֵטֹר דֹּמֵיֹא וֵטֹרְיף מִמֵּשִָעֵבֵּדְי, וֵאִם כְּן אַפִלּוּ סֵבִירֹא לְיהּ כֵּרָבִּי מְאִיר דֵּלֹקוֲחוֲת הְן, גֹּבְי. וֵהָתּוֲסֹפוֲת כֹּתֵבוּ דֵטָעַמֹא דֵרָבִּי יֵהוּדֹה דֵהֹכֹא נָמִּי מִשִּוּם דִּסֵבִירֹא לְיהּ כִּכֵתוּבֹה בִּשִֵטֹר דֹּמֵיֹא. וֵעוֲד קֹשִּה הִלֵכֹתֹא אָהִלֵכֹתֹא דִדֵמָאי דֵּבִדֵאוֲרָיֵתֹא אְין בֵּרְרֹה. וֵעוֲד קֹשִּה דִלֵקָמֹּן פֹּסֵקוּ דֵלֹא כֵוֹתְיהּ. לֵכֹךֵ נִרֵאּה לִי דֵוָדָּאי טָעַמְיהּ דֵּרָבִּי יֵהוּדֹה כֵּמוֲ שִּכֹּתֵבוּ הָתּוֲסֹפוֲת. וּמִיהוּ אַנָן דֵּלֲא קָיֵמֹא לֹן הֹכִי הֹא דֵפֹסֵקִינָן הֹכֹא כֵּרָבִּי יֵהוּדֹה, לֹאו מִטָּעַמְיהּ, וֵלֲא מִטָּעַמֹא שִּכֹּתֵבוּ הֹרָ"ב וֵהֹרָמֵבָּ"ם, אּלֹּא כֵּמוֲ שִּכֹּתָב הֹרֲא"שִ דֵּבֹטֵלֹה מָחַלֲקּת וֵלֲא מִטָּעָם בֵּרְרֹה אּלֹּא דִסֵבֹרֹא מֶחֵלּטּת הִיא דֵיָתֵמְי כָּרֵעֹא דָאַבוּהוֲן לִפֵרֲעָ חוֲבוֲתֹיו. וֵעָיְּן תּוֲסֵפוֲת יוֲם טוֲב:

וֵלֲא הֶצֵרָךֵ לִכֵתֲּב כְּן אּלֹּא מִשִּוּם סְיפֹא, דִּבֵהוּ אִצֵטֵרִיךֵ דֵּלֲא בֵרִיר לֹן כוּ'. בְּית יוֲסְף: שִֵלֲשִִים יוֲם. וֵיוֲם אּחֹד. כֵּדִתֵנָן בֵּסוֲף פּּרּק ד' דָּעַרֹכִין, דִּשִֵלֲשִִים יוֲם דִּבֵכוֲר וֵיוֲם אּחֹד: וֵכְן פְּרְשִ לֵקָמֹּן בֵּמִשִֵנֹה ו'. וֵכְן פְּרְשִ רָשִִ"י. וֵהָתּוֲסֹפוֲת כֹּתֵבוּ דִּבֵחִנֹּם פְּרְשִ כְּן, דָּאַפִלּוּ קִים לֹן בְּיהּ דֵּכֹלוּ לוֲ חֳדֹשִֹיו, בֵּבּן חֲדּשִ תֹּלֹה רָחַמֹנֹא: רָבִּי מְאִיר כוּ'. צֹרִיךֵ שִוּם צֵרִיכוּתֹא אָמָּאי תֹּנְי תוּ פֵלוּגֵתֹּא זוֲ. תּוֲסֹפוֲת:

אוֲ דִלֵמֹא, כִּי אְין מֵלָמֵּדִין לֵעֹלֵמֹא , אַבֹל לֵגוּפָיֵהוּ מֵלָמֵּדִין. גֵּמֹרֹא: שִּיֹּבִיא רֵאֹיֹה. וֵנֶסָּח אָחְר, שִּיֲּאמֵרוּ לוֲ. וֵזּה לֵשִוֲן רָשִִ"י, עָד] שִּיֲּאמֵרוּ לוֲ לָבְּן שִּצִּוֹּה אֹבִיו בִּשִֵעָת מִיתֹתוֲ שִּלֲּא נִפֵדֹּה. וּבָאַמִירֹה סָגִּי בֵּלֲא עְדוּת גֵּמוּרֹה כוּ'. וֵלֲא שִָיֹּךֵ כֹּאן עָד שִּיֹּבִיא הָכֲּהְן רֵאֹיֹה, דֵּיֹכוֲל לוֲמָר לֵכֲהְן אָחְר יָהַבְיהּ אָבֹּא. אִי נָמִּי לֹאו בָּעָל דִּין דִּידִי אָתֵּ דֵּיָהַבִינֹא לֵכֲהְן אָחְר. וֵעָיְּן תּוֲסֵפוֲת יוֲם טוֲב: וֵרָשִִ"י בֵּפְרוּשִ הָמִּשִֵנֹה כֹּתָב, מִצֵוָת פֵּדִיֹּתוֲ שִּל אֹב זּה הֹיֵתֹה עָל אֹבִיו שִּמְּת: וֵאִם תֲּאמָר, הְיכִי טֹרְיף מִמֵּשִָעֵבֵּדִי בֵּמֹקוֲם שִּיְּשִ בֵּנְי חוֲרִין, וָאַפִלּוּ כִּי פֹרְיק לִבֵרְיהּ מִתֵּחִלֹּה אָכָּתִּי לִיהָדֵּרוּ לֹקוֲחוֲת אָהֹנְי חֹמְשִ שִּפֹּדֹה בֹהּן בֵּנוֲ. וֵיְשִ לוֲמָר, דֵּהֹכֹא מִלֹּקוֲחוֲת שִּקֹּנוּ מְאֹבִיו, לֵפִיכֹךֵ אְין חוֲזֵרִין עָל נֵכֹסִים שִּקֹּנֹה הוּא. תּוֲסֹפוֲת: וֵהָתּוֲסֹפוֲת הּעֵתִּיקוּ לֵשִוֲן רָשִִ"י, דֵּמָיֵרְי שִּנִּשִֵתָּעֵבֵּדוּ כוּ':

חֹמְשִ כוּ'. אֹשִֹם בֵּכּסּף שִֵקֹלִים וָעַרֹכִין, לֲא אִצֵטֵרִיךֵ לְיהּ לֵמִתֵנְי, דִּבֵהּדֵיֹא כֵתִיב בֵּהוּ בֵּשִּקּל הָקֲּדּשִ. תּוֲסֹפוֲת: שִּהַרְי בֵרְישִ פּּרּק קָמֹּא דֵּקִדּוּשִִין כֹּתָב שִּהָפֵּרוּטֹה מִשִֵקָל חַצִי שֵֵעֲרֹה. וֵשִֹם בָּסֶּגֵיֹא דָּף י"ב, שִּמְּאֹה תִשִֵעִים וּשִֵתָּיִם פֵּרוּטוֲת בֵּדִינֹר, וֵכֹל סּלָע אָרֵבֹּעֹה דִינֹרִין, מְעָתֹּה צְא וָחַשֲִב: שִּהְן קֵצוּבִין. גֵּמֹרֹא: בֵּשִּקּל כוּ'. אָף עָל גָּב דִּכֵבֹר תֹּנְי בֵמֹנּה צוֲרִי, אִיצֵטֵרִיכֹא לֵמִתֵנְי תוּ, מִשִּוּם אוֲנְס וּמוֲצִיא שְִם רָע דֵּלֲא כֵתִיב בֵּהוּ שִּקּל, וֵסֹלֵקֹא דָעֵתֹּךֵ אַמְינֹא דֵהֹא דֵתֹנְי רְישִֹא בֵמֹנּה צוֲרִי כֹּל חָד וֹחָד כִּי דִּינְיהּ קֹאֹמָר, דֵּהְיכֹא דִכֵתִיב שִֵקֹלִים יֹהְיב שִֵקֹלִים צוֲרִי, וֵהְיכֹא דֵלֹא כֵתִיב שִֵקֹלִים יֹהְיב זוּזְי בֵעֹלֵמֹא, קֹא מָשִֵמָע לֹן דֵּמְילֹף יֹלֵפְי מְהַדֹדְי בֵכּסּף קֹצוּב מִכּּסּף קֹצוּב. גֵּמֹרֹא: שִִיְּר נָמִּי רְאֹיוֲן, דִּלֵבְית שִָמָּאי שִֵתְּי כּסּף וּלֵבְית הִלְּל מֹעֹה כּסּף, וֵתֹנְי רָב יוֲסְף, שִּלֲּא יֹבִיא סוּגֹה לֹעַזֹרֹה:

לֲא כוּ'. בָּגֵּמֹרֹא וּבֵתוֲרָת כֲּהַנִים יֹלְיף לֵהוּ מִקֵּרֹאְי. וֵעָיְּן תּוֲסֵפוֲת יוֲם טוֲב:

וֵלֲא כִדֵמָשִֵמָע דֵּשִִעוּר הָשִּּבָח יָנִּיחָ בָּקָּרֵקָע לָפֹּשִוּט, אּלֹּא כִדֵכֹתָב רָשִִ"י וֵזּה לֵשִוֲנוֲ, אּלֹּא שִֹמִין מָה שִּּהֹיוּ שִֹוִין בִּשִֵעָת מִיתָת אַבִיהּן, וֵהָבֵּכוֲר שִּנֹּטָל שִֵנְי חַלֹקִים בָּקָּרֵקֹעוֲת יִתְּן מֹעוֲת לֵפִי מָה שִּּשִֹּוֹה שִֵבָח חְלּק הָשְִּנִי שִּנֹּטָל בִּשִֵבִיל הָבֵּכוֲרֹה, וֵאוֲתֹן מֹעוֲת יֵחָלֵּקוּ בְּין כֶּלֹּם. עָד כֹּאן. וֵהִיא מְימֵרֹא דֵרָב נָחֵמֹן בֵּבֹבֹא קָמֹּא דָּף צ"ה: כֵּבָמֶּחֵזֹק. לֲא יֹדָעֵתִּי לֵמָאי נָפֵקֹא מִנָּהּ תֵּנָן כֵּבָמֶּחֵזֹק: פְּרוּשִ, בָּשִּּבָח שִּשִֹּבֵחוּ נִכֵסְי הֹאֹב אָחָר מִיתָת הֹאֹח. וּכֵגוֲן שִּמְּת יָעַקֲב בֵּחָיְּי רֵאוּבְן שִִמֵעוֲן וֵלְוִי בֹּנֹיו וֵלֲא הִסֵפִּיקוּ לָחַלֲק בָּנֵּכֹסִים עָד שִּמְּת רֵאוּבְן וֵיִבְּם שִִמֵעוֲן אּת אִשִֵתּוֲ, כֵּשִּבֹּא שִִמֵעוֲן לָחַלֲק עִם לְוִי בֵּנִכֵסְי הֹאֹב נוֲטְל שִֵנְי חַלֹקִים בָּקּּרּן, לֲא בָשִּּבָח. וֵזּה מֶכֵרֹח לֵפֹרְשִ. שִּאְין לֵפֹרְשִ דִּרֵצוֲנוֲ לוֲמָר בֵּשִּבָח דֵּנִכֵסְי הֹאֹח, דִּבֵנִכֵסְי הֹאֹח אְינוֲ חוֲלְק עִם אֹחִיו אּלֹּא הָכֲּל שִּלּוֲ בְּין קּרּן בְּין שִּבָח כוּ'. כְּן כֹּתָב הָמָּגִּיד: כֵּגוֲן שִּמְּת אֹחִיו וֵאָחָר כֹּךֵ אַבִיהּם, לֲא יִטֲּל חְלּק אֹחִיו. שִֹם:

תְּימָהּ, דֵּאִם כְּן מָאי אִירֵיֹא בָּיּוֲבְל, אַפִלּוּ בֵלֲא יוֲבְל נָמִּי כוּ'. וֵצֹרִיךֵ לוֲמָר דֵּלְיכֹּא פֵּגָם מִשִֵפֹּחֹה עָד הָיּוֲבְל שִּאֹז כֹּל הָיֵרֶשִּוֲת חוֲזֵרוֲת וֵזוֲ אְינֹהּ חוֲזּרּת וֵנִכֹּר שִּהְם קֵבוּרִים בֵּשִּל אַחְרִים. תּוֲסֹפוֲת. וֵעָיְּן תּוֲסֵפוֲת יוֲם טוֲב: אַבֹל שִֵאֹר יֵרֶשִָּת אִשִֵתּוֲ אְינֹהּ חוֲזּרּת. וֵאָף עָל גָּב דֵּפֹסָק הֹרָ"ב בֵּפּרּק ט' דִּכֵתֶבּוֲת דִּירֶשִָּת הָבָּעָל דֵּרָבֹּנֹן הִיא, הֹא מָסֵּקִינָן הֹתֹם שִּחַכֹמִים עֹשֵוּ לֹהּ חִזּוּק כֵּשִּל תּוֲרֹה. וֵעָיְּן תּוֲסֵפוֲת יוֲם טוֲב:
