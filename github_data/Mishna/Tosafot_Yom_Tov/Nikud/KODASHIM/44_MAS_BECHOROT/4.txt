פרק ד - משנה א

וּפָרְכִינַן, אַדְרַבָּה דִּסְמִיךְ לֵיהּ לְדִסְמִיךְ לֵיהּ. אָמַר רַבָּה אָמַר קְרָא תַּעֲשֶׂה, הוֹסִיף לְךָ הַכָּתוּב עֲשִׂיָּה אַחֶרֶת בְּשׁוֹרְךָ. וְאֵימָא שִׁיתִין. לֹא מְסָרְךָ הַכָּתוּב אֶלָּא לַחֲכָמִים. פֵּרֵשׁ רַשִׁ"י, הוֹאִיל וְלֹא גִלָּה לְךָ שִׁעוּר עֲשִׂיָּה שֶׁהוֹסִיף, וַדַּאי לַחֲכָמִים מְסָרְךָ שֶׁיְּפָרְשׁוּ לְךָ הַטַּעַם, וְאַחֲרֵיהֶם תֵּלֵךְ כוּ'. גְּמָרָא: אַמְּלֵאָתְךָ בִּלְחוֹד הוּא דִּמְהַדַּר. וּכְמוֹ שֶׁכָּתַב הָרַ"ב בְּמִשְׁנָה ו' פֶּרֶק ג' דִּתְרוּמוֹת: קָשֶׁה, שֶׁהֲרֵי אֵין קְצִיר שְׂעֹרִים קוֹדֵם לָעֹמֶר שֶׁבְּפֶסַח וְהֵיאַךְ אֶפְשָׁר שֶׁהַחִטִּים גַּם כֵּן מְבֻשָּׁלִים בְּפֶסַח, וּמִקְרָא מָלֵא הוּא כִּי הַשְּׂעֹרָה אָבִיב וְגוֹ' וְהַחִטָּה וְגוֹ' אֲפִילוֹת הֵנָּה כוּ' . וְלָכֵן נִרְאֶה לִי שֶׁרוֹצֶה לוֹמַר כְּהַתּוֹסָפוֹת שָׁם שֶׁכָּתְבוּ שֶׁגְּמַר זְמַן בִּשּׁוּל הַתְּבוּאָה עַד הָעֲצֶרֶת שֶׁמְּבִיאִין כוּ'. תּוֹסְפוֹת יוֹם טוֹב. וְעַיֵּן תּוֹסָפוֹת חֲדָשִׁים: בְּדַקָּה כוּ'. צָרִיךְ לוֹמַר דְּרַבִּי יוֹסֵי לֵית לֵיהּ דְּרָשָׁא דִקְרָא, אֶלָּא סְבִירָא לֵיהּ דְּמִסְּבָרָא בְעָלְמָא אָמְרוּ חֲכָמִים שֶׁיְּטַפְּלוּ בוֹ יִשְׂרָאֵל, כְּדֵי שֶׁתְּהֵא הַנְּתִינָה דֶּרֶךְ גְּדֻלָּה וּכְדִכְתִיב לְמָשְׁחָה בָהֶם. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: שֶׁנֶּאֱמַר כוּ' בְשָׁנָה. וְכִי יִהְיֶה בוֹ מוּם בִּשְׁעָרֶיךָ תֹּאכֲלֶנּוּ. הָרַמְבַּ"ם:

פרק ד - משנה ב

שְׁנָתוֹ. דִּלְשָׁנָה דִידֵיהּ מָנִינַן, דְּאִם נוֹלַד בְּנִיסָן רַשַּׁאי לְקַיְּמוֹ עַד נִיסָן אַחֵר. וְלֹא אָמְרִינַן מִתִּשְׁרֵי שֶׁשָּׁלְמָה שְׁנַת מִנְיַן עוֹלָם שָׁלְמָה נַמִּי שְׁנַת הַבְּכוֹר. מְנָא הָנֵי מִלֵּי, דְּאָמַר קְרָא שָׁנָה בְשָׁנָה, אֵיזוֹהִי שָׁנָה שֶׁנִּכְנְסָה בְחַבְרוֹתֶיהָ הֱוֵי אוֹמֵר שָׁנָה שֶׁל בְּכוֹר, כְּלוֹמַר דְּכִי אָזְלַת אַחַר שְׁנַת הַבְּכוֹר הָוֵי נִכְנָס בַּחֲבֶרְתָּהּ, דְּכִי הָוֵי לַבְּכוֹר שָׁנָה אַחַת נִכְנֶסֶת שְׁנִיָּה לְמִנְיַן עוֹלָם מִתִּשְׁרֵי. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: שְׁלֹשִׁים יוֹם. מְפָרֵשׁ בַּבְּרַיְתָא, מִפְּנֵי הֲשָׁבַת אֲבֵדָה לַבְּעָלִים. וּפֵרֵשׁ רַשִׁ"י דְּהַיְנוּ כֹהֵן. דְּזִמְנִין דְּלֹא מַשְׁכַּח כֹּהֵן לְמֵיתְבֵיהּ לֵיהּ, וְאִם יִשְׁחָטֶנּוּ מִיָּד יִסְרַח, לְפִיכָךְ רַשַּׁאי לְקַיְּמוֹ שְׁלֹשִׁים יוֹם אַחַר שְׁנָתוֹ. וְהַתּוֹסָפוֹת פֵּרְשׁוּ דְּמַיְרֵי בְיַד כֹּהֵן וְאֵין צָרִיךְ עַכְשָׁיו לַבָּשָׂר, וְנָתְנוּ לוֹ חֲכָמִים זְמַן שְׁלֹשִׁים יוֹם. אֲבָל יִשְׂרָאֵל לְעוֹלָם מַמְתִּין עַד שֶׁיִּמְצָא כֹהֵן. עַד כָּאן. וּבָרֵישָׁא נַמִּי דִתְנַן מֻתָּר לְקַיְּמוֹ כָּל שְׁנֵים עָשָׂר חֹדֶשׁ, אִי נָפַל בֵּיהּ מוּם סָמוּךְ לִשְׁנֵים עָשָׂר חֹדֶשׁ כְּגוֹן חֲמִשָּׁה עָשָׂר יוֹם, מַשְׁלִימִין לוֹ עוֹד חֲמִשָּׁה עָשָׂר יוֹם אַחַר שְׁנָתוֹ שֶׁיְּהֵא לוֹ שְׁלֹשִׁים יוֹם מִיּוֹם שֶׁנָּפַל בּוֹ מוּם, כִּדְאִיתָא בַּגְּמָרָא:

פרק ד - משנה ג

וְלֹא מִשּׁוּם קְנָס. דְּהָא לֹא קָתָנֵי הוֹאִיל וּשְׁחָטוֹ, אֶלָּא הוֹאִיל וְנִשְׁחַט, מַשְׁמַע אֲפִלּוּ שְׁחָטוֹ אַחֵר, וּבְהָא לֹא שַׁיָּךְ קְנָסָא. הָרַמְבַּ"ן. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: מֻמְחֶה. פֵּרֵשׁ הָרַ"ב בְּמַתְנִיתִין דִּלְקַמָּן, וְהַשֵּׁם נִגְזַר מִן וּמָחָה עַל כֶּתֶּף יָם , שֶׁהוּא לְשׁוֹן מְיֻשָּׁר:

פרק ד - משנה ד

הוּא הָרֹאשׁ שֶׁבְּאֶרֶץ יִשְׂרָאֵל: לָאו מִשּׁוּם דְּתֵיתֵי כְסוּמְכוֹס. אֶלָּא תַּקָּנַת חֲכָמִים בְּעָלְמָא הִיא. וְאַף עַל גַּב דְּאַיְרֵי בְכָל מוּם אֲפִלּוּ יָדוּעַ שֶׁיַּתִּירֶנּוּ מֻמְחֶה עַל יָדוֹ, מִכָּל מָקוֹם חֲשִׁיב לֵיהּ סָפֵק קְצָת, דְּשֶׁמָּא לֹא מַזְּקִיק לֵיהּ מֻמְחֶה. וְגַם לֹא פְלִיגֵי בֵּין מוּם לְמוּם. וְאִי הֲוָה מַיְרֵי בְּדֻקִּין שֶׁבָּעַיִן דַּוְקָא הֲוָה נִיחָא טְפֵי. תּוֹסָפוֹת: הַיְנוּ רַבִּי מֵאִיר, וְאַשְׁכְּחַן דְּקָנֵיס שׁוֹגֵג אַטּוּ מֵזִיד, בְּגִטִּין דַּף נ"ג. תּוֹסָפוֹת. וּלְפִיכָךְ חַיָּב אַף עַל פִּי שֶׁלֹּא נָשָׂא וְנָתַן בַּיָּד. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: וְרֵישָׁא דְמַתְנִיתִין דְּקָתָנֵי וְנִשְׁחַט דְּלֹא מִתּוֹקְמָא בְּנָשָׂא וְנָתַן בַּיָּד, הָתָם קְנָסָא הוּא דְּקָנְסֵי רַבָּנָן דְּלֹא לֵיתֵי לְמִשְׁרֵי מִי שֶׁאֵינוֹ מֻמְחֶה, לְפִי שֶׁרְאִיַּת מוּמִין חָכְמָה יְתֵרָה הִיא, וְכִדְאִיתָא בַּגְּמָרָא. הָרֹא"שׁ: וְלָמָּה הָיָה לוֹ לַעֲשׂוֹת כָּךְ, הִלְכָּךְ הָוֵי כְּמוֹ מֵזִיד דְּמִשְׁנָה ד' פֶּרֶק ה' דְּגִטִּין. תּוֹסָפוֹת: וְכֵיוָן דְּאִי קַיְמָא פָּרָה הֲוָה מָצֵי לְמִהְדַּר בֵּיהּ וּלְמִשְׁרְיֵהּ, מִשְׁתְּכַח שֶׁמַּה שֶּׁאָמַר לָאו כְּלוּם הוּא זֶה, וְזֶה שֶׁהֶאֱכִילָהּ לַכְּלָבִים אִיהוּ דְאַפְסִיד אַנַּפְשֵׁיהּ. רַשִׁ"י. אֲבָל מִדִּבְרֵי הָרַ"ב רֵישׁ פֶּרֶק ג' דְּסַנְהֶדְרִין מַשְׁמַע אֲפִלּוּ נָשָׂא וְנָתַן בַּיָּד פָּטוּר. וְעַיֵּן מַה שֶּׁכָּתַבְתִּי שָׁם:

פרק ד - משנה ה

וּפָרְכִינַן בַּגְּמָרָא, אִי הָכִי כִּי אָמַר תָּם וְשָׁקֵיל אַגְרָא נַמִּי אָתוּ לְמִחְשְׁדֵיהּ דְּבַעַל מוּם הוּא, וְהַאי דְּלֹא קָשָׁרֵי, סָבַר כִּי הֵיכִי דְלִשְׁקַל אַגְרֵיהּ זִמְנָא אַחֲרִיתָא. וְאָמְרִינַן דְּלֹא תִקְּנוּ לוֹ עוֹד שָׂכָר אֶלָּא כִּי מַיְתֵי לֵיהּ בָּתַר הָכִי כַמָּה זִמְנֵי לֹא שָׁקֵיל מִידִי. וְכָתְבוּ הַתּוֹסָפוֹת, וְאִם תֹּאמַר אַכַּתִּי אָמְרוּ הַאי תָּם הוּא וְשָׁרֵי לֵיהּ כִּי הֵיכִי שֶׁלֹּא יַטְרִיחַ פַּעַם אַחֶרֶת. וְיֵשׁ לוֹמַר כוּ', וְעוֹד דְּבָזֶה לֹא יַחְשְׁדוּהוּ שֶׁיַּתִּיר הָאָסוּר בִּשְׁבִיל טִרְחָתוֹ שֶׁל פַּעַם שְׁנִיָּה:

פרק ד - משנה ו

דִּינָיו. וְלֹא תְנַן דִּינוֹ בָּטֵל וְעֵדוּתוֹ בָּטֵל. מַשְׁמַע דְּלֹא אוֹתוֹ עֵדוּת בִּלְבַד שֶׁיָּדוּעַ שֶׁנָּטַל בּוֹ שָׂכָר הוּא שֶׁבָּטְלָה אֶלָּא כָל עֵדֻיּוֹת שֶׁהֵעִיד וְכָל דִּינָיו שֶׁדָּן, אֶלָּא אִם כֵּן יָדוּעַ שֶׁלֹּא נָטַל בָּהֶן שָׂכָר. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: וְעַיֵּן מַה שֶּׁכָּתַבְתִּי פֶּרֶק ד' דְּאָבוֹת מִשְׁנָה ה'. וְכָתְבוּ הַתּוֹסָפוֹת, וְאִם תֹּאמַר וְשֹׁחַד לֹא תִקַּח דְּכָתַב רַחֲמָנָא לָמָּה לִּי, וְלָאו פִּרְכָא הוּא, דִּתְרֵי מִלֵּי נִינְהוּ, דִּבְתוֹרַת אַגְרָא הַיְנוּ שְׂכַר טָרְחוֹ, וּבְתוֹרַת שֹׁחַד הַיְנוּ לְהִשְׁתַּדֵּל לְדוּנוֹ יָפֶה וּלְהַפֵּךְ בִּזְכוּתוֹ, אֲפִלּוּ לְזַכּוֹת הַזַּכַּאי כוּ'. עַד כָּאן: וְכָל הַפּוֹסְקִים הִמְלִיצוּ בַּעֲדָם. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב בַּאֲרִיכוּת. וְעַיֵּן בְּרֵאשִׁית חָכְמָה בְּפֶרֶק ג' מִשַּׁעַר הַקְּדֻשָּׁה: דְּאִלּוּ לְהָבִיא וּלְמַלְּאֹת שָׁרֵי: מְעָרָה. אֵינָם מֻתָּרִים לְקִדּוּשׁ אֶלָּא מִן הַמַּיִם חַיִּים. הָרַמְבַּ"ם: וְטִמְּאָהוּ. גְּמָרָא, אִיהוּ גוּפֵיהּ הֵיכִי אָזֵיל, הָא קָעָבַר עַל לְנֶפֶשׁ לֹא יִטַּמָּא. וּמְשָׁנִינַן בִּשְׁאָר טֻמְאָה דְּלֹא מֻזְהָר עֲלַיְהוּ, כְּגוֹן נְבֵלוֹת וּשְׁרָצִים: וְאַף עַל פִּי שֶׁנִּמְצָא מַגִּיעַ לוֹ שָׂכָר לָדוּן, כֵּיוָן שֶׁמִּתְבַּטֵּל מִמְּלַאכְתּוֹ יָכוֹל לִטֹּל שָׂכָר. כָּךְ פָּסְקוּ הַתּוֹסָפוֹת. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב:

פרק ד - משנה ז

וְיִקְנְסוּהוּ כְפִי רִשְׁעוֹ. הָרַמְבַּ"ם: לְפִי שֶׁהַחִתּוּךְ נִכָּר, וְאֵינוֹ דוֹמֶה מַמָּשׁ לְנַקְבוּת. וּבַגְּמָרָא, וְאִידָךְ בֵּי עַכְבְּרֵי מֵידָע יְדִיעַ: וּבְגָדִים. וְנֻסְחָא אַחְרִינָא וְאָרִיג. וּרְצוֹנוֹ לוֹמַר דָּבָר מְחֻבָּר דּוֹמֶה לַאֲרִיג, רְצוֹנוֹ לוֹמַר הַלְּבָדִים:

פרק ד - משנה ח

פִּשְׁתָּן. יְרוּשַׁלְמִי, וּפִשְׁתָּן לָאו קִסְמִין הוּא. אָמַר רַבִּי חִינְנָא מִפְּנֵי זַרְעָהּ: וְאָרִיג. וּפֵרוּשׁ, כְּעֵין בְּגָדִים כְּגוֹן הַגְּדִילִים שֶׁהֵן מַעֲשֵׂה עֲבוֹת מִן הַפִּשְׁתִּים עַצְמוֹ:

פרק ד - משנה ט

רַבִּי שִׁמְעוֹן כוּ'. טַעֲמֵיהּ נַמִּי מִשּׁוּם קְנָס הוּא, דְּהָא רָבָא דַעֲבִיד עוֹבָדָא כְוָתֵיהּ וְקָאָמַר טַעֲמָא, בְּאִסּוּרָא דְגוּפֵיהּ קָנְסִינַן לֵיהּ: זִקַּת. שַׁיָּכוּת. וּכְבָר בֵּאַרְנוּ אוֹתוֹ בִּיבָמוֹת. הָרַמְבַּ"ם: תְּרוּמוֹת וּמַעַשְׂרוֹת. וְהָרַמְבַּ"ם הֶעְתִּיק תְּרוּמָה וּמַעֲשֵׂר. וְכָתַב, וְכֵן הֶחָשׁוּד לְמַעֲשֵׂר שֵׁנִי כוּ'. וּפֵרֵשׁ הַכֶּסֶף מִשְׁנֶה, דְּהָכִי מַשְׁמַע מִדְּקָתָנֵי רַבִּי שִׁמְעוֹן אוֹמֵר כָּל שֶׁיֵּשׁ בּוֹ זִקַּת תְּרוּמוֹת וּמַעַשְׂרוֹת, דְּאַשְּׁנֵי מַעַשְׂרוֹת קָאֵי. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב שֶׁהֶאֱרִיךְ, וּמַסִּיק, לְכָךְ נִרְאֶה לִי שֶׁהָרַמְבַּ"ם לֹא נִתְכַּוֵּן אֶלָּא לְמַעֲשֵׂר שֵׁנִי, וְהַאי וְכֵן כוּ' הוּא פֵּרוּשׁ דְּבָרָיו הָרִאשׁוֹנִים. וְהָא דְּתָנֵי מַעַשְׂרוֹת, רְצוֹנוֹ לוֹמַר דָּגָן וְתִירוֹשׁ וְיִצְהָר. וְהַשְׁתָּא נִיחָא נַמִּי גִרְסַת הַסֵּפֶר, תְּרוּמוֹת:

פרק ד - משנה י

אֵינוֹ כוּ'. גְּמָרָא, זוֹ דִּבְרֵי רַבִּי עֲקִיבָא. אֲבָל רַבִּי יְהוּדָה אוֹמֵר הֶחָשׁוּד עַל הַשְּׁבִיעִית חָשׁוּד עַל הַמַּעַשְׂרוֹת, דִּבְאַתְרֵיהּ דְּרַבִּי יְהוּדָה שְׁבִיעִית חֲמִירָא לְהוּ כְּדִתְנַן סוֹף פֶּרֶק ג' דִּדְמַאי: וּלְשׁוֹן הָרַמְבַּ"ם, כְּגוֹן טֻמְאַת אֳכָלִים וְטֻמְאַת יָדַיִם. וְזֶה לְשׁוֹנוֹ בְּפֶרֶק ח' מֵהִלְכוֹת שְׁמִטָּה, שֶׁהָאֹכֶל הַטָּמֵא הַזֶּה שֶׁמְּכָרוֹ בְּחֶזְקַת טָהוֹר, אֵינוֹ מְטַמֵּא אֲחֵרִים אֶלָּא מִדִּבְרֵי סוֹפְרִים: לֹא עַל זֶה כוּ'. לְפִי שֶׁהֶחָשׁוּד בְּמִלֵּי דְרַבָּנָן אֵין רָאוּי לַחֲשֹׁד אוֹתוֹ בְּמִלֵּי דְאוֹרַיְתָא. אֲבָל אִם נֶחֱשַׁד עַל טֻמְאָה דְאוֹרַיְתָא הֲרֵי הוּא גַם כֵּן חָשׁוּד עַל הַשְּׁבִיעִית וּמַעַשְׂרוֹת, וּלְפִיכָךְ אָמַר יֵשׁ שֶׁהוּא חָשׁוּד עַל הַטָּהֳרוֹת. . הָרַמְבַּ"ם: זֶה הַכְּלָל. קְצָת קָשֶׁה לְשׁוֹן זֶה הַכְּלָל, כֵּיוָן שֶׁאֵינוֹ כּוֹלֵל הַשְּׁנוּיִין כְּבָר אֶלָּא מִלְּתָא אַחֲרִיתָא קָפָסֵיק וְתָנֵי. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב:
