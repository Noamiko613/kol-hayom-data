פרק ה

לְשׁוֹן רַשִׁ"י, בִּתְחִלָּה שָׁחֹר וְאַחַר כָּךְ יוֹצֵא אָדֹם, וּשְׁתֵּיהֶן בִּשְׁתִיתָה בְּלֹא קִלּוּחַ בְּרָחוֹק, וְאַחַר כָּךְ מְקַלֵּחַ, וּבַסּוֹף כְּשֶׁהַדָּם מִתְמַעֵט שׁוֹתֵת וְיוֹרֵד. עַד כָּאן. וְהַאי דַּם הַקָּזָה רְצוֹנוֹ לוֹמַר הַקָּזַת בַּעַל חַי שֶׁאָסוּר דָּמוֹ. שֶׁדַּם הָאָדָם אֵין חַיָּבִים עָלָיו. הָרַמְבַּ"ם: בְּחַיָּה כוּ'. יָלְפִינַן לְהוּ בַּגְּמָרָא מִקְּרָאֵי: דַּם כוּ'. פֵּרֵשׁ הָרַ"ב בְּבָבָא קַמָּא וּבְחֻלִּין, נְחִירָה שֶׁתּוֹחֵב הַסַּכִּין בִּנְחִירָיו וְחוֹתֵךְ, וַעֲקִירָה שֶׁעוֹקֵר הַסִּימָנִין מִמְּקוֹם חִבּוּרָן. וְכָתְבוּ הַתּוֹסָפוֹת אֲבָל קָשֶׁה כֵּיוָן שֶׁאֵינָהּ מֵתָה עַל יְדֵי כָּךְ, וַהֲלָכָה לְמֹשֶׁה מִסִּינַי שֶׁהִיא פְּסוּלָה , אִם כֵּן דָּמָהּ אֵינָהּ בְּכָרֵת כֵּיוָן שֶׁאֵין הַנְּשָׁמָה יוֹצֵאת בּוֹ. לְכָךְ צָרִיךְ לוֹמַר דָּם מִשְּׁחִיטָה שֶׁהָיְתָה לְאַחַר עִקּוּר. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב בְּשֵׁם הָרַמְבַּ"ן: שֶׁהַנֶּפֶשׁ כוּ'. וְהוּא הַדִּין דִּבְדַם שְׁחִיטָה וּנְחִירָה וְעִקּוּר בָּעִינַן שֶׁהַנֶּפֶשׁ יוֹצְאָה בּוֹ. אֶלָּא לֹא אִצְטְרִיךְ לְמֵימַר רַק גַּבֵּי דַּם הַקָּזָה, שֶׁאֵין רָגִיל שֶׁיַּקִּיז דָּם לִבְהֶמְתּוֹ עַד שֶׁיֵּצֵא לָהּ דַּם הַנֶּפֶשׁ. תּוֹסָפוֹת. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: וְדַם דָּגִים טְמֵאִים אָסוּר. וְאֵין לוֹקִים עָלָיו מִשּׁוּם דָּם וְלֹא מִשּׁוּם שֶׁרֶץ הַמַּיִם. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: וְלֹא הִצְרִיךְ שׁוּם הוֹכָחָה לְדַם חֲגָבִים. וּבַגְּמָרָא נָמִי לֹא אִתְּמַר אֶלָּא בְּדָגִים. וְצָרִיךְ עִיּוּן אִם הוּא הַדִּין בַּחֲגָבִים אִי לֹא: מְחַיֵּב. וּמוֹדֶה רַבִּי יְהוּדָה לְעִנְיַן כַּפָּרָה. שֶׁנֶּאֱמַר כִּי הַדָּם הוּא בַּנֶּפֶשׁ יְכַפֵּר , דָּם שֶׁהַנֶּפֶשׁ יוֹצְאָה בּוֹ מְכַפֵּר כוּ', וְהָכָא הַיְנוּ טַעֲמָא דִּמְרַבֵּי לֵיהּ מִכָּל דָּם. גְּמָרָא:

מְחַיֵּב כוּ'. פְּלִיגֵי בִּדְרָשָׁא דִּקְרָאֵי בַּגְּמָרָא: פֵּרוּשׁ בְּחַטָּאת קְבוּעָה: דְּאִם סָפֵק מַאי קָאָמַר, פְּשִׁיטָא דְּסָפֵק הוּא לוֹ: שֶׁמִּמִּין כוּ'. לֹא בָּעֵי לַאֲתוֹיֵי אָשָׁם וַדַּאי, דְּלֹא בָּעֵי יְדִיעָה לְכַתְּחִלָּה כְּמוֹ בְּחַטָּאת. שֶׁאִם הָיָה סָפֵק מֵבִיא אָשָׁם תָּלוּי, וּלְכִי מִתְוַדַּע מֵבִיא חַטָּאתוֹ. גְּמָרָא:

וְזֶה כוּ'. דְּלֹא בָּעֵי חֲתִיכָה מִשְּׁנֵי חֲתִיכוֹת כוּ' דְּכֵיוָן דְּאִקְּבַע אִסּוּרָא מֵעִקָּרָא חַיָּב כוּ'. עַיֵּן תּוֹסְפוֹת יוֹם טוֹב: כְּלוֹמַר הַבָּאִים עַל חֵטְא כְּדִתְנַן בְּסוֹף פִּרְקִין, דְּדַוְקָא בַּחַטָּאת אֵין שְׁנַיִם כוּ', וְאָשָׁם תָּלוּי נַמִּי בִּמְקוֹם חַטָּאת הוּא בָּא:

חֲתִיכָה כוּ'. וְהוּא שֻׁמָּן דְּמִחַלַּף בְּחֵלֶב. וְהָכָא מִשּׁוּם דִּלְקַמָּן תָּנֵי קֹדֶשׁ תָּנֵי הָכָא חֻלִּין: גְּמָרָא. וּבַבְּרַיְתָא, דִּבְהֶדְיָא אָמַר רַבִּי יוֹסֵי הָכִי:

וְרַבִּי עֲקִיבָא (נַמִּי) מְחַיֵּב בָּזֶה, מֵחֲמַת הַחֵלֶב וּמֵחֲמַת הֶקְדֵּשׁ אַחַת, לְפִי שֶׁאָשָׁם תָּלוּי אֶחָד הוּא צָרִיךְ לְהָבִיא עַל שְׁנֵי הַסְּפֵקוֹת יַחַד. הָרַמְבַּ"ם:

גְּמָרָא, וְהָא יֵשׁ אוֹכֵל אֲכִילָה אַחַת רֵישׁ פֶּרֶק ג' מִשְׁנָה ד' דְּקָתָנֵי בֵּיהּ נוֹתָר, וְקָתָנֵי מֵבִיא אַרְבַּע חַטָּאוֹת וְאָשָׁם אֶחָד, אָמַר לוֹ הַהִיא בְּגַסָּה (בַּאֲכִילָה גַּסָּה) וְהַהִיא בְּדַקָּה. אִי נַמִּי הַהִיא בִּימוֹת הַגְּשָׁמִים, הָכָא בִּימוֹת הַחַמָּה דְּמַסְרַחַת הַבָּשָׂר: רַבִּי שִׁמְעוֹן כוּ'. דְּבַמֻּקְדָּשִׁין סְבִירָא לֵיהּ דְּגַלֵּי רַחֲמָנָא דְּאִסּוּר חָל עַל אִסּוּר. וְהַשְׁתָּא לֹא קַשְׁיָא מִמִּשְׁנָה ד' פֶּרֶק ג' דִּשְׁבוּעוֹת. וְתָמוּהַּ דְּבַגְּמָרָא הִקְשׁוּ זֶה מִבְּרַיְתָא וְלֹא הִקְשׁוּ מִמִּשְׁנָה דִּשְׁבוּעוֹת. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב:
