פרק א

שִֵלֲשִִים וֹשְִשִ כוּ'. מִנֵיֹנֹא לֹמֹּה לִי. שִּאִם עַשֵֹאֹן כֶּלֹּם בֵּהּעֵלְם אּחֹד חָיֹּב עָל כֹּל אָחָת וֵאָחָת. וּמָפִּיק לְיהּ מִקֵּרֹא כוּ'. וֵלֲא דָּוֵקֹא שִּיִּתֵחָיְּב שִֵלֲשִִים וֹשְִשִ חָטֹּאוֲת, דֵּהֹא פּּסָח וּמִילֹה אְין בֹּהּם חָטֹּאת: אַבֹל אִי אִיכֹּא הָתֵרֹאֹה יְשִ מְהּן בֵּחּנּק וֵיְשִ מְהּן בִּסֵקִילֹה וֵיְשִ מְהּן בֵּמָלֵקוֲת. וּבֵשִוֲגְג בֵּחָטֹּאת. רָשִִ"י: עָל הֹאְם כוּ'. טָעָם סִדּוּרֹן דֵּקֹחַשִִיב לֵהוּ הּחֹמוּר חֹמוּר בֵּרְישִֹא . רָשִִ"י: וֵעָל כוּ'. כֶּלֵּהוֲן מָיֵתְי לֵהוּ קֵרֹאְי בָּגֵּמֹרֹא: וֵהֹאִשִֹּה כוּ'. מָסִּיק בָּגֵּמֹרֹא דִּבֵגָבֵרְי קֹחַשִִיב לֵמִנֵיֹנֹא לֵחָיֵּבוֲ עָל כֶּלֹּם בֵּהּעֵלְם אּחֹד, וֵדָל מִנֵּהוֲן הֹאִשִֹּה כוּ', וֵתֹנֹא שִֵלֲשִִים וֵשִֹלֲשִ חָטֹּאוֲת בִּלֵבָד, וֵתֹנֹא הֹאִשִֹּה כוּ' וּפּסָח וּמִילֹה לֵסָיּוֲמִינֵהוּ כֵּרְתוֲת. דֵּאִי אֹמָרֵתֵּ בֵּנֹשְִי קֹחַשִִיב, דָּל מִנֵּהוֲן הָבֹּא עָל הָזֹּכֹר וֵעָל הָבֵּהְמֹה: וֵכְן פְּרְשִ לֵקָמֹּן גָּבְּי מֵחָלְּל שִָבֹּת בֵּאָחָת מְאַבוֲת מֵלֹאכוֲת. דֵּתָנֹּא לֲא מֵהָדָּר אּלֹּא אָמִּצֵוֲת שִּחָיֹּבִין עַלְיהּן כֹּרְת, וֵשְִם שִָבֹּת קֹתֹנְי וֵשְִם עַבוֲדֹה זֹרֹה קֹתֹנְי, וּלֵעוֲלֹם בֵּכֹל חַדֹא וָחַדֹא אִיכֹּא טוּבֹא. גֵּמֹרֹא: וֵהֹעוֲשֵֹן בֵּהּעֵלְם אּחֹד לֲא מִחָיָּב שִֵתָּיִם. רָשִִ"י. אַבֹל לֵעִנֵיָן מִיתֹה חִלֵּקֹן הָכֹּתוּב: וֵדִבֵרְי תְּימָהּ הְן, דִּבֵהּדֵיֹא שִֹנִינוּ בֵּמִשִֵנֹה ג' פּּרּק ב' דִּזֵבֹחִים חוּץ לִמֵקוֲמוֲ פֹּסוּל וֵאְין בּוֲ כֹּרְת: וֵאָף עָל פִּי שִּעָל הָסִּיכֹה עָצֵמֹהּ שִּיֹּסוּךֵ בּוֲ פֹּטוּר כֵּדִלֵקָמֹּן: לֹאו דָּוֵקֹא, דֵּגָבְּי קֵטֲרּת לֲא כֵּתִיב כֹּמוֲהוּ כֵּמוֲ גָּבְּי שִּמּן כוּ', אּלֹּא רֵצוֲנוֲ לוֲמָר עָד שִּיֵּהְא רֹאוּי לֵהָקֵטִיר אְיזּה שִִעוּר, וֵאְין צֹרִיךֵ שִּיָּעַשֵּה כֶּלֹּהּ: שִּכְּן הוּא בֵּפְרוּשִ בָּכֹּתוּב. אַבֹל הָמְּרִיחָ בָּקֵּטֲרּת שִּל בְּית הָמִּקֵדֹּשִ לֲא אָזֵהֹרֹה וֵלֲא כֹּרְת כוּ'. וֵלֲא דֹּמְי לֵסֹךֵ בֵּשִּמּן הָמִּשִֵחֹה, דֵּהֹתֹם אִית בְּיהּ מָעַשֵּה, וּמִשִּוּם הֹכִי אִית בְּיהּ אָזֵהֹרֹה. רָשִִ"י: דִּכֵתִיב אַשִּר יִתְּן מִמּּנּוּ עָל זֹר, מִמּּנּוּ מְאוֲתוֲ שִּל מֲשִּה. עָיְּן תּוֲסֵפוֲת יוֲם טוֲב: כֹּל מָה שִּּכֹּתָב הֹרָ"ב מָיֵתְי לֵהוּ קֵרֹאְי בָּגֵּמֹרֹא: וֵאְין מָלֵכְי יִשֵֵרֹאְל מוֲשִֵחִין . וֵהֹא דּּאֱלִישִֹע מֹשִָח לֵיְהוּא, מֵתֹרֵצִים בָּגֵּמֹרֹא בָּאַפָרֵסֵמֹא דֹּכֵיֹא וֵלֲא בֵּשִּמּן הָמִּשִֵחֹה: בָּגֵּמֹרֹא מִי הַוֹה שִּמּן הָמִּשִֵחֹה וֵהֹא יֲאשִִיֹּה גֵּנֹזוֲ, וּמֵשִֹנְי בָּאַפָרֵסֵמֹא דֹּכֵיֹא:

וֵהָבֹּא עָל הָזֹּכֹר וֵעָל הָבֵּהְמֹה כֵּסֹבוּר לֹבֲא עָל אִשִֵתּוֲ שִּלֲּא כֵּדָרֵכֹּהּ וּבֹא עָל הָזֹּכֹר וֵכוּ' וֵאָחָר כֹּךֵ נוֲדָע לוֲ דֵּזֹכֹר הוּא אוֲ בֵּהְמֹה הִיא וֵכְן בֵּכֶלֵּהוּ. נִדֹּה כֵּסֹבוּר טֵהוֲרֹה הִיא. מֵגָדְּף כֵּסֹבוּר מֵבֹרְךֵ שְִם בּּן שִֵתְּי אוֲתִיּוֲת וּבְרָךֵ שְִם בּּן אָרֵבָּע. מֲלּךֵ כֵּסֹבוּר לֵהָעַבִיר אֹדֹם אָחְר וֵהּעֱבִיר בֵּנוֲ. אוֲב נִתֵכָּוְּן לֵכִשִּוּף אָחְר. דֹּם כֵּסֹבוּר דָּם דֹּגִים. מָעַלּה בָּחוּץ כֵּסֹבוּר בֹּמוֲת אְלּוּ שִּל צִבּוּר מֶתֹּרוֲת וּמוֲקְי לְיהּ לִקֵרֹא בֵּבֹמָת יֹחִיד. שִּמּן וּקֵטֲרּת בֵּנִתֵכָּוְּן לֵסָמֵמֹנִים אַחְרִים: כֵּגוֲן תִּינוֲק שִּנִּשִֵבֹּה וֵלֲא שִֹמָע מְעוֲלֹם שִּיְּשִ שִָבֹּת. דֵּאִלּוּ שִֹמָע וֵעָכֵשִֹיו שִֵכֹחוֲ, חָיֹּב, כֵּמוֲ שִּכֹּתָב הֹרָ"ב בֵּרְישִ פּּרּק ז' דֵּשִָבֹּת: אָף כוּ'. אָדֵּרָבִּי מְאִיר מָהֵדֵּרְי דֵּאֹמָר חוּץ מִן הָמֵּטָמְּא מִקֵדֹּשִ וֵקֹדֹשִֹיו, וֵקֹאֹמֵרְי דֵּאָף הָמֵגָדְּף נָמִּי. וֵלֲא יֹדָעֵנֹא מָאי קָשִֵיֹא לְיהּ לֵרָשִִ"י אָדִּמֵפֹרְשִ בָּגֵּמֹרֹא דֵּרָבֹּנֹן מָהֵדֵּרִי אָיִּדֵעוֲנִי דֵּלֲא תֵּנָן לָהּ לֵפִי שִּאְין בּוֲ מָעַשֵּה וֵאֹמֵרְי לְיהּ מֵגָדְּף נָמִּי לְית בְּיהּ מָעַשֵּה, וֵכֹתָב תְּימָהּ לִי וּלֵרָבִּי יוֲחֹנֹן דֵּאֹמָר לֵעְיל הוֲאִיל וּשִֵנְיהּם בֵּלֹאו אּחֹד , מָאי אָף. עָד כֹּאן. דֵּהֹא מָתֵנִיתִין רָבִּי עַקִיבֹא הִיא דֵּלֲא בֹּעְי מָעַשֵּה וּמְהָאי טָעַמֹא לֲא אֹמָר רָבִּי יוֲחֹנֹן הֹכִי. וֵעָיְּן תּוֲסֵפוֲת יוֲם טוֲב:

יְשִ כוּ'. הִתֵחִיל לֵפֹרְשִ קֹרֵבָּן חָטֹּאת שִּלֲּא עָל יֵדְי חְטֵא: הָמָּפּּלּת כוּ'. מָתֵנִיתִין מִתֵפֹּרּשִּת יֹפּה בֵּדִבֵרְי הֹרָ"ב בֵּפּרּק ג' דֵּנִדֹּה מִשִֵנֹה ד'. שִּשִֹּם הִיא שִֵנוּיֹה לֵעִנֵיָן טֶמֵאָת לְידֹה. וֵכֹל הָטֵּמְאֹה לְידֹה הַרְי זוֲ מֵבִיאֹה קֹרֵבֹּן וֵחָטֹּאתֹהּ נּאֱכּלּת: לֲא דֹּק, דֵּתָנֹּאְי נִינֵהוּ לִשִֵנְי הָפְּרוּשִִים הָלֹּלוּ בֵּמָסּּכּת נִדֹּה דָּף כ"ה. עָיְּן תּוֲסֵפוֲת יוֲם טוֲב:

דֵּהֹכִי שִֹמֵעִינָן לְיהּ לֵרָבִּי יוֲסְי בֵּמִשִֵנֹה ה' פּּרּק ה' וֵכְן הַלֹכֹה, וֵכְן נָמִּי בֵּאֹשִֹם תֹּלוּי שִֹם בֵּמִשִֵנֹה ד', לֵפִי שִּבֹּא בִּמֵקוֲם חָטֹּאת:

דֹּגִים כוּ'. דֵּהֹא בֵּמִשִֵנֹה ג' פּּרּק ג' דֵּנִדֹּה אִם יְשִ עִמֹּהּן דָּם טֵמְאֹה, לֹאו מִשִּוּם לְידֹה אּלֹּא מִשִּוּם נִדֹּה:

דֵּחָיֹּב בֵּקֹרֵבֹּן שְִנִי כֵּמוֲ שִּכֹּתוּב בֵּפְרוּשִ הֹרָ"ב בֵּמָתֵנִיתִין דִּלֵקָמֹּן: לֵפִי שִּהוּא דָּם לְידֹה, לֲא דָּם טֲהָר. הֹרָמֵבָּ"ם:

מָשִֵמָע עָל לְידֹה אָחָת מֵבִיאֹה, דֵּהָיֵנוּ עָל כֹּל אָחָת וֵאָחָת. רָשִִ"י. וּדֵאְינֹהּ מֵבִיאֹה עָל כֹּל לְידֹה וֵלְידֹה דֵּתוֲךֵ מֵלֲאת עָיְּן בֵּפְרוּשִ הֹרָ"ב פּּרּק ב' מִשִֵנֹה ד': בֵּדִינֹרִין. דִּינֹרִין שִּל כּּסּף. דִּינֵרְי, שִּל זֹהֹב. עֹרוּךֵ:
