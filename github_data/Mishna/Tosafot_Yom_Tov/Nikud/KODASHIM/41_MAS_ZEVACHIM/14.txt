פרק יד

פרק יד - משנה א

שָׂעִיר הַמִּשְׁתַּלֵּחַ. לְאַחַר וִדּוּי. דְּאִי קֹדֶם וִדּוּי, בָּעֵי לִכָּנֵס לְהִתְוַדּוֹת עָלָיו. גְּמָרָא. וּכְלוֹמַר וַהֲרֵי הוּא רָאוּי לָבֹא:

פרק יד - משנה ב

מִשְׁכַּן ה'. וּבַגְּמָרָא פָּרֵיךְ וְהָכִי נַמִּי תֵּפּוּק לֵיהּ מִוְּאֶל פֶּתַח אֹהֶל מוֹעֵד לֹא הֱבִיאוֹ דִּלְעֵיל. וּמְשָׁנֵי, דְּמֵהָתָם לֵיכָּא לְמִשְׁמַע אֶלָּא כָּל שֶׁאֵינוֹ רָאוּי לָבֹא כְּלָל, וְשָׂעִיר מִשְׁתַּלֵּחַ נַמִּי מִשֶּׁיָּצָא לְאַחַר וִדּוּי אֵינוֹ רָאוּי לָבֹא, אֲבָל כִּי נִרְאָה וְנִדְּחָה כִי הָנֵי דְמַתְנִיתִין אֵימָא לֹא, אִצְטְרִיךְ לִפְנֵי מִשְׁכַּן ה': אָמַר לְהוּ מֹשֶׁה לְיִשְׂרָאֵל, כִּי עֲלִיתוּ לָאָרֶץ, אַרְבַּע עֶשְׂרֵה שָׁנִים שֶׁכָּבְשׁוּ וְשֶׁחִלְּקוּ, יְשָׁרוֹת תַּקְרִיבוּ, דְּהַיְנוּ נְדָרִים וּנְדָבוֹת. חוֹבוֹת לֹא תַקְרִיבוּ וַאֲפִלּוּ בְּבָמַת צִבּוּר, עַד שֶׁתָּבֹאוּ אֶל הַמְּנוּחָה, זוֹ שִׁילֹה, וְגִלְגָּל לְגַבֵּי שִׁילֹה מְחֻסַּר זְמַן הוּא, וְקָאָמַר לְהוּ מֹשֶׁה לֹא תַעֲשׂוּן. גְּמָרָא: לִרְוָחָא דְמִלְּתָא נָקְטֵיהּ הָכִי, וּבְלָאו הָכִי הֲוָה מָצֵי לְמֵימַר דְּאֵימָא מוֹדֶה כוּ': בְּלֹא תַעֲשֶׂה. אֵין הָעִנְיָן שֶׁחַיָּב כָּרֵת וּמַלְקוֹת, שֶׁזֶּה לֹא יָבֹא לְעוֹלָם כוּ'. אֲבָל הַכַּוָּנָה שֶׁהַלָּאו הַזֶּה הוּא אַזְהָרַת כָּרֵת שֶׁאֵין לוֹקִין עָלָיו. הָרַמְבַּ"ם:

פרק יד - משנה ג

וּמְצֹרָע. לְשׁוֹן רַשִׁ"י, לֹא גָרְסִינַן בְּמַתְנִיתִין. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: עֵרוּב פֵּרוּשִׁים כָּתַב הָרַ"ב. דְּמַה שֶּׁכָּתַב נְדָבָה, לְשׁוֹן רַשִׁ"י הוּא, וּלְדִידֵיהּ בְּקֹדֶם שֶׁהִקְרִיבוּ מַיְרֵי, דְּהָא מִתְקַבְּלוֹת בִּפְנִים לִנְדָבָה קָאָמַר. וְזֶה שֶׁכָּתַב לְאַחַר שֶׁהִקְרִיבוּ, לְשׁוֹן הָרַמְבַּ"ם הוּא. וְקָרוֹב לְהַגִּיהַּ בְּדִבְרֵי הָרַ"ב שֶׁצָּרִיךְ לוֹמַר אִי נַמִּי לְאַחַר כוּ'. וּפֵרוּשׁוֹ שֶׁזֶּה שֶׁכָּתַב נְדָבָה לִשְׁמָן, לְהָכִי מְדַקְדֵּק לִשְׁמָן מִשּׁוּם דְשֶׁלֹּא לִשְׁמָן אָמְרִינַן בַּגְּמָרָא דַּאֲפִלּוּ אָשָׁם נַמִּי הוֹאִיל וְרָאוּי שֶׁלֹּא לִשְׁמָן בִּפְנִים. תּוֹסְפוֹת יוֹם טוֹב. וְעַיֵּן עוֹד: זָרוּת. כָּל הֲנִי הַקְטָרוֹת בְּדָבָר שֶׁדַּרְכָּן לֶאֱכֹל לָאו הַקְטָרָה נִינְהוּ. וְכָל הָנָךְ עֲבוֹדוֹת הוֹאִיל וְיֵשׁ אַחֲרֵיהֶן עֲבוֹדָה, אֵין זָר חַיָּב עֲלֵיהֶן מִיתָה. כִּדְאָמַר בַּמַּסֶּכֶת יוֹמָא, עֲבוֹדַת מַתָּנָה, וְלֹא עֲבוֹדַת סִלּוּק. עֲבוֹדָה תַמָּה, וְלֹא עֲבוֹדָה שֶׁיֵּשׁ אַחֲרֶיהָ עֲבוֹדָה. וּפֵרוּשׁ סֵיפָא דִקְרָא וְהַזָּר הַקָּרֵב יוּמָת, עֲבוֹדַת מַתָּנָה שֶׁאַתָּה נוֹתֵן עַל הַמִּזְבֵּחַ, וְלֹא עֲבוֹדַת סִלּוּק שֶׁאַתָּה מְסַלֵּק וְסָר מֵעַל הַמִּזְבֵּחַ כְּגוֹן תְּרוּמַת הַדֶּשֶׁן. עֲבוֹדָה תַמָּה, עֲבוֹדָה שֶׁהִיא גוֹמֶרֶת וּמְתַמֶּמֶת אֶת הַדָּבָר, וְלֹא שֶׁיֵּשׁ אַחֲרֶיהָ עֲבוֹדָה, כְּגוֹן שְׁחִיטָה וְקַבָּלָה וְהוֹלָכָה שֶׁיֵּשׁ אַחֲרֶיהָ זְרִיקָה. אֲבָל זְרִיקָה עֲבוֹדָה אַחֲרוֹנָה שֶׁבַּדָּם כוּ'. רַשִׁ"י: טֻמְאָה. דְּאַזְהָרַת זָרוּת וְשִׁמּוּשׁ בְּטֻמְאָה מֵחַד קְרָא נָפֵק, מִוְּיִנָּזְרוּ וְלֹא יְחַלְּלוּ, וּבְכָל הַזְּבָחִים תִּנְיָנָא (זְבָחִים דף ט"ז), מַאי דִּמְחַיֵּב אַזָּרוּת מְחַיֵּב אַשִּׁמּוּשׁ טֻמְאָה. רַשִׁ"י: מְחֻסַּר בְּגָדִים. דְּאִינְהוּ נַמִּי בִזְמַן שֶׁאֵין בְּגָדִים עֲלֵיהֶם אֵין כְּהֻנָּתָן עֲלֵיהֶן וַהֲווּ לְהוּ זָרִים. רַשִׁ"י: רְחוּץ כוּ'. גָּמַר חֻקָּה חֻקָּה מִמְּחֻסַּר בְּגָדִים. הָתָם לְעִנְיַן אַחוֹלֵי עֲבוֹדָה, וּלְעִנְיַן מִיתָה נַמִּי אֲנַן גָּמְרִינַן לַהּ מֵהָתָם. רַשִׁ"י:

פרק יד - משנה ה

לַגִּלְגָּל. וִימֵי הַגִּלְגָּל אַרְבַּע עֶשְׂרֵה שָׁנָה שֶׁכָּבְשׁוּ וְחִלְּקוּ. מְפָרֵשׁ בַּגְּמָרָא:

פרק יד - משנה ו

כְּלוֹמַר כְּשֶׁתַּעַבְרוּ אֶת הַיַּרְדֵּן לֹא תַקְרִיבוּ בַּבָּמוֹת כְּלָל כְּכָל אֲשֶׁר אֲנַחְנוּ עוֹשִׂים פֹּה הַיּוֹם שֶׁאָנוּ מַקְרִיבִין בַּמִּשְׁכָּן חַטָּאוֹת וַאֲשָׁמוֹת, אֶלָּא אִישׁ כָּל הַיָּשָׁר בְּעֵינָיו תַּעֲשׂוּ, דְּהַיְנוּ נְדָרִים וּנְדָבוֹת, כִּי בְּעוֹד שֶׁתִּכְבְּשׁוּ וּתְחַלְּקוּ לֹא בָאתֶם עַד עַתָּה אֶל הַמְּנוּחָה. וַעֲבַרְתֶּם אֶת הַיַּרְדֵּן וִישַׁבְתֶּם בָּאָרֶץ, הַיְנוּ לְאַחַר יְרֻשָּׁה וִישִׁיבָה. וְהָיָה הַמָּקוֹם וְגוֹ' הִשָּׁמֶר לְךָ וְגוֹ' הַיְנוּ אִסּוּר בָּמוֹת. רַשִׁ"י: שֶׁל גַּג שֶׁל נְסָרִים. רַשִׁ"י:

פרק יד - משנה ז

לְנוֹב. כְּמוֹ שֶׁנֶּאֱמַר . אֵין לֶחֶם חֹל וְגוֹ' כִּי אִם לֶחֶם קֹדֶשׁ אֲשֶׁר לִפְנֵי ה', אַלְמָא שֻׁלְחָן וּמִזְבֵּחַ הָתָם הֲווּ. חָרְבָה נוֹב בִּימֵי שָׁאוּל, בָּאוּ לְגִבְעוֹן כְּדִכְתִיב כִּי הִיא הַבָּמָה הַגְּדוֹלָה. רַשִׁ"י: דִּכְתִיב וַיְּהִי מִיּוֹם שֶׁבֶת הָאָרוֹן בְּקִרְיַת יְעָרִים וְגוֹ' וַיִּהְיוּ עֶשְׂרִים שָׁנָה, וְהֵן מִשֶּׁחָרְבָה שִׁילֹה שֶׁנִּשְׁבָּה הָאָרוֹן וְהֻחְזַר לְקִרְיַת יְעָרִים, וְאָז לְקָחוֹ דָוִד מִקִּרְיַת יְעָרִים אַחֲרֵי שֶׁמָּלַךְ בְּחֶבְרוֹן שֶׁבַע שָׁנִים וּמָלַךְ בִּירוּשָׁלַיִם שְׁלֹשִׁים וְשָׁלֹשׁ שָׁנִים כִּמְפֹרָשׁ הַכֹּל בַּכָּתוּב, הֲרֵי חֲמִשִּׁים וְשָׁלֹשׁ שָׁנָה, וּשְׁלֹמֹה מָלַךְ אַרְבַּע שָׁנִים כְּשֶׁבָּנָה הַבַּיִת , הֲרֵי חֲמִשִּׁים וְשֶׁבַע שָׁנָה: עָרֵי יִשְׂרָאֵל. דְּהָא בְכָל מָקוֹם שֶׁהוּא שָׁם עוֹשֶׂה בָמָה וּמַקְרִיב. רַשִׁ"י. לְפִי שֶׁלֹּא הָיוּ לְיִשְׂרָאֵל עֲדַיִן עָרִים בִּרְשׁוּתָן אֲבָל הָיוּ עֲסוּקִין בְּכִבּוּשָׁן. הָרַמְבַּ"ם:

פרק יד - משנה ח

נַחֲלָה. שֶׁנִּתְקַיְּמָה קְדֻשָּׁתָהּ וַעֲמִידָתָהּ לְעוֹלָם, וְעָלֶיהָ אָמַר הַנָּבִיא וְנַחֲלָתוֹ לֹא יַעֲזֹב, לְפִי שֶׁאָמַר בִּתְחִלַּת דְּבָרָיו שֶׁהַשֵּׁם בָּחַר בִּירוּשָׁלַיִם לִשְׁכִינָתוֹ וּבָחַר בְּיִשְׂרָאֵל לִסְגֻלָּתוֹ. אַחַר כָּךְ אָמַר שֶׁהַקָּדוֹשׁ בָּרוּךְ הוּא יִשְׁמֹר הָעָם הַזֶּה אֲשֶׁר בָּחַר בּוֹ בַּמָּקוֹם הַהוּא אֲשֶׁר בָּחַר בּוֹ. שֶׁנֶּאֱמַר כִּי בָחַר ה' בְּצִיּוֹן. הָרַמְבַּ"ם. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: הַקְּלָעִים. לִפְנִים לְכָתְלֵי הָעֲזָרָה. הָרַמְבַּ"ם: הַחוֹמָה. שֶׁל יְרוּשָׁלַיִם, שֶׁהִיא בִּמְקוֹם מַחֲנֵה יִשְׂרָאֵל שֶׁבַּמִּדְבָּר, כְּדִכְתִיב וְאָכַלְתָּ שָׁם לִפְנֵי ה' אֱלֹהֶיךָ. רַשִׁ"י:

פרק יד - משנה ט

וּכְלוֹמַר דְּעִקַּר פָּרָשַׁת שְׁחוּטֵי חוּץ עַל אוֹתָן שֶׁבַּמִּדְבָּר נֶאֶמְרָה: הָכִי פֵּרְשׁוּ בַּבְּרַיְתָא, דְּכָל הָעִנְיָן שֶׁל אוֹתָהּ פָּרָשָׁה בִּתְחִלָּתָהּ עַד פָּסוּק וְלֹא יִזְבְּחוּ וְגוֹ' שֶׁעָנַשׁ כָּרֵת עַל הַשּׁוֹחֵט בַּחוּץ וְהַמַּעֲלֶה בַחוּץ, מְדַבֵּר בְּקָדָשִׁים שֶׁהִקְדִּישָׁן בִּשְׁעַת אִסּוּר בָּמוֹת וְהִקְרִיבָן בִּשְׁעַת אִסּוּר בָּמוֹת, דְּהַיְנוּ מִשֶּׁהוּקַם הַמִּשְׁכָּן. וּמִפָּסוּק וְלֹא יִזְבְּחוּ וּלְהַלָּן מְדַבֵּר בְּקָדָשִׁים שֶׁהִקְדִּישָׁן בִּשְׁעַת הֶתֵּר בָּמוֹת וּכְדִכְתִיב אֲשֶׁר הֵם זוֹבְחִים עַל פְּנֵי הַשָּׂדֶה, וְהָכִי מַשְׁמַע אֲשֶׁר הֵם זוֹבְחִים שֶׁהָיוּ עוֹמְדִים לִשָּׁחֵט בַּבָּמָה אִם הִקְרִיבָהּ בִּשְׁעַת אִסּוּר בָּמוֹת כְּאִלּוּ הִקְרִיב עַל פְּנֵי הַשָּׂדֶה. וְשָׁם נֶאֱמַר וֶהֱבִיאוּם לַה' מִצְוַת עֲשֵׂה. וְלֹא יִזְבְּחוּ עוֹד מִצְוַת לֹא תַעֲשֶׂה. יָכוֹל יְהֵא עָנוּשׁ כָּרֵת, תַּלְמוּד לוֹמַר זֹאת לָהֶם וְכוּ', זֹאת לָהֶם עֹנֶשׁ עֲשֵׂה וְלֹא תַעֲשֶׂה, וְלֹא אַחֶרֶת לָהֶם כָּרֵת:

פֵּרֵשׁ רַשִׁ"י וְזָבְחוּ זִבְחֵי שְׁלָמִים לַה' אוֹתָם, וְאוֹקְמִינָא לְהַאי קְרָא בְקָדָשִׁים שֶׁהִקְדִּישָׁן בִּשְׁעַת הֶתֵּר בָּמוֹת, וּכְתִיב אוֹתָם מִעוּטָא, אַלְמָא זְבָחִים הוּא דַּהֲווּ נְהִיגֵי בְהוּ, וְלֹא עוֹפוֹת וּמְנָחוֹת. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: הַזְּמָן. דְּיָלֵיף בַּבְּרַיְתָא מִוְּזֹאת תּוֹרַת זֶבַח הַשְּׁלָמִים, לַעֲשׂוֹת זְמַן בְּבָמָה קְטַנָּה כִּזְמַן בָּמָה גְדוֹלָה: וְהַנּוֹתָר. נַמִּי מִפִּגּוּל יָלֵיף מִגְּזֵרָה שָׁוָה דְעָוֹן עָוֹן. רַשִׁ"י: וּתְמֵהַנִי לָמָּה לֹא פֵרְשׁוֹ כְּמוֹ בְּכָל מָקוֹם דְּמַתְנִיתִין פִּגּוּל נוֹתָר טָמֵא, דְּטָמֵא לְעִנְיַן אֲכִילָה. וְכֵן פֵּרֵשׁ רַשִׁ"י, הָכָא וְטָמֵא נַמִּי בְּהַהִיא פַּרְשְׁתָא כְתִיב, וְהַנֶּפֶשׁ אֲשֶׁר תֹּאכַל וְגוֹ':
