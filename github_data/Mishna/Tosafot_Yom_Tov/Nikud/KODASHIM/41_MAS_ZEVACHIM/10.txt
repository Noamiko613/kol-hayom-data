פרק י

הַמְקֻדָּשׁ מֵחֲבֵרוֹ. שֶׁיֵּשׁ בּוֹ צַד רִבּוּי קְדֻשָּׁה מִמֶּנּוּ. רַשִׁ"י: וְכֵן כָּתַב רַשִׁ"י. וְנִרְאֶה לִי מִשּׁוּם דְּקַשְׁיָא לְהוּ הָא דִתְנַן בְּמִשְׁנָה ד' הַחַטָּאת קוֹדֶמֶת, וְלֹא תְנַן דַּם חַטָּאת, וְהָתָם אָמְרִינַן זֶה בָנָה אָב לְכָל הַחַטָּאוֹת. וּלְהָכִי פֵּרְשׁוּ דְּהָכָא דַּוְקָא כְשֶׁשְּׁנֵיהֶם שְׁחוּטִים וְנִזְרְקוּ וְהָתָם כְּשֶׁעוֹמְדִין לִשָּׁחֵט. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: שֶׁדָּמָהּ כוּ'. וְלֹא סַגִּי לֵיהּ מִפְּנֵי שֶׁהַחַטָּאת מְרַצָּה עַל חַיָּבֵי כְּרֵתוֹת דְּאִיכָּא בְאָשָׁם נַמִּי דְיֵשׁ לוֹ קִצְבָּה לְדָמִים כְּדִלְקַמָּן מִשְׁנָה ה'. אֲבָל חַטָּאת כָּל דְּהוּ, וְרִבּוּי דְמִזְבֵּחַ עָדִיף. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: בְּפֵרוּשׁ בַּכָּתוּב, אֶלָּא נִלְמָד בְּלִמּוּד, דְּשִׁירַיִם לַיְסוֹד בְּכֻלְּהוּ הָוֵי כְּמוֹ שֶׁכָּתַב רַשִׁ"י בַּגְּמָרָא דַּף נ"ב. וְכֵן כָּתַב הַמַּגִּיד מִשְׁנֶה: קָדְשֵׁי קָדָשִׁים. לֹא סַגִּי לֵיהּ בְּטַעֲמָא דִמְרַצֶּה, מִשּׁוּם דְּתוֹדָה וְאֵיל נָזִיר מְקֻדָּשִׁים דִּטְעוּנִין לֶחֶם, כִּדְאָמְרִינַן בַּגְּמָרָא. תּוֹסָפוֹת:

מִפְּנֵי שֶׁמִּצְוָתָן מְרֻבָּה, שֶׁכָּל הַקָּרְבָּנוֹת בָּאִין מֵהֶן. רַשִׁ"י: דָּמוֹ כוּ'. בִּשְׁלָמָא אֵמוּרִים לֵיתַנְהוּ בְּעוֹפוֹת, אֶלָּא דָם מִיהָא אִיתָא. אֶלָּא לְאַשְׁמְעִינַן אֵמוּרָיו כִּי דָמוֹ, מַה דָּמוֹ לִפְנֵי זְרִיקָה אַף אֵמוּרָיו קֹדֶם זְרִיקָה, דְּהָא קָרֵי לְהוּ קָדְשֵׁי קָדָשִׁים. וּמַה דָּמוֹ מִפְּסִיל בְּיוֹצֵא אַף אֵמוּרָיו מִפַּסְלֵי בְּיוֹצֵא. גְּמָרָא. וְנִרְאֶה לִי דְּמַתְנִיתִין אַלִּיבָּא דְרַבִּי אֱלִיעֶזֶר מִשְׁנָה ג' פֶּרֶק קַמָּא דִמְעִילָה, דִּסְבִירָא לֵיהּ דְּאֵמוּרָיו נִפְסָלִין בְּיוֹצֵא. וְכֵן מַשְׁמַע בַּסּוּגְיָא. גְּמָרָא:

חֵטְא. כְּלוֹמַר דִּמְכַפֶּרֶת: וְכֵיוָן דְּלֵית בָּהּ הַקְטָרָה לֹא מְהַנְיָא טַעֲמָא דִמְרַצֶּה כְּדִלְעֵיל מִשְׁנָה ב'. תּוֹסָפוֹת: כְּדִמְפָרֵשׁ בַּבְּרַיְתָא, שֶׁאֵין תַּלְמוּד לוֹמַר לְלַמֵּד שֶׁתִּקְרַב רִאשׁוֹנָה, שֶׁהֲרֵי כְּבָר נֶאֱמַר וְאֶת הַשֵּׁנִי יַעֲשֶׂה עוֹלָה: הָכִי קָאָמַר, אֲפִלּוּ חַטַּאת הָעוֹף לְעוֹלַת בְּהֵמָה. דִּבְתַרְוַיְהוּ בְּהֵמָה לֹא צְרִיכָא לְבִנְיַן אָב. גְּמָרָא: וְלָאו כְּלָלָא הוּא, דְּהָא בְּיוֹלֶדֶת כְּתִיב אֶחָד לְעוֹלָה וְאֶחָד לְחַטָּאת, הֲרֵי שֶׁהִקְדִּים עוֹלָה לְחַטָּאת. וְטַעֲמָא בַגְּמָרָא, לְפִי שֶׁכָּאן בָּאָה בִמְקוֹם בְּהֵמָה דַּעֲשִׁירוּת שֶׁהָיְתָה בְהֵמָה לְעוֹלָה. וּמִכָּל מָקוֹם מַסִּיק רָבָא דְּאַף בְּיוֹלֶדֶת לֹא הִקְדִּימָהּ הַכָּתוּב אֶלָּא לְמִקְרָאָה שֶׁתַּקְדִּים בִּקְרִיאַת שֵׁם וְלֹא לַעֲבוֹדָה. אֲבָל לְשׁוֹן רַשִׁ"י, לְמִקְרָאָה שֶׁתְּהֵא נִקְרֵאת בָּעִנְיָן תְּחִלָּה, אֲבָל בַּעֲבוֹדָתָן חַטָּאת קוֹדֵם. וּלְפִי זֶה כְּלָלָא הוּא, וְכֵן כָּתַב הָרַמְבַּ"ם. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב:

כְּלוֹמַר שֶׁנִּקְרְבוּ אֶמֶשׁ. וּדְלֹא כְמוֹ שֶׁכָּתַב בְּפֵרוּשׁ רַשִׁ"י שֶׁבָּאוּ אֶמֶשׁ לָעֲזָרָה וְשָׁהוּ עַד הַיּוֹם לְשָׁחֲטָן. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: אֶמֶשׁ כוּ'. פֵּרֵשׁ בְּקֻנְטְרֵס, דְּזִלְזוּל הוּא לְהַשְׁהוֹתָם כָּל כָּךְ שֶׁצּוּרַת הַבָּשָׂר מִתְקַלְקֶלֶת. וְעַל חִנָּם דָּחַק. אֶלָּא הַיְנוּ טַעֲמָא, דִּשְׁלָמִים שֶׁל אֶמֶשׁ אֵין לָהֶם הֶתֵּר אֲכִילָה אֶלָּא עַד הַלַּיְלָה, וְחַטָּאת וְאָשָׁם שֶׁל יוֹם עַד הַבֹּקֶר דְּנֶאֱכָלִין לְיוֹם וְלַיְלָה. תּוֹסָפוֹת: הַחַטָּאת כוּ'. וְהוּא הַדִּין אָשָׁם, דְּקָדְשֵׁי קָדָשִׁים הוּא. כֵּן כָּתְבוּ הַתּוֹסָפוֹת. וּבַתּוֹסֶפְתָּא שָׁנוּי בְּהֶדְיָא כוּ'. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב:

לְשַׁנּוֹת. נִרְאֶה לִי דַּאֲתָא לְאַשְׁמְעִינַן שֶׁאִם צְלָאוֹ יָכוֹל לְהוֹסִיף עָלָיו בְּתַבְלִין, וְכֵן אִם בִּשְּׁלוֹ יָכוֹל אַחַר כָּךְ לְשָׁלְקוֹ אוֹ לְצָלְאוֹ. דְּאִי לֹא תֵימָא הָכִי הָא תְנִינַן לֵיהּ בְּפֶרֶק אֵיזֶהוּ. וְגַם לִישְׁנָא דְּיָכוֹל לְשַׁנּוֹת דָּיֵק שַׁפִּיר טְפֵי. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: מַשְׁמַע דִּלְמַעוֹטֵי מִכָּרֵת דְּנוֹתָר. וְאִי אֶפְשָׁר לוֹמַר כֵּן, דִּבְוַדַּאי טַעַם כְּעִקָּר לְחַיְּבוֹ בְּכָרֵת. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב:

לִרְוָחָא דְמִלְּתָא כָתַב דְּרַבִּי שִׁמְעוֹן אָמַר. שֶׁכֵּן הוּא בַּבְּרַיְתָא מְנָחוֹת דַּף ע"ה. אֲבָל בְּמִשְׁנָה ג' פֶּרֶק ו' דִּמְנָחוֹת מִסְתָּם סָתֵם לָן כוּ': וְלֹג כוּ'. הָכִי פֵּרוּשׁוֹ, וּמוֹתַר לֹג. דְּמוֹתָר קַמָּא קָאֵי נַמִּי אַלֹּג, דְּהָא הִזָּה מִמֶּנּוּ כְּדִתְנַן פֶּרֶק י"ד דִּנְגָעִים מִשְׁנָה י': וּבְסוֹף פֶּרֶק י"ב דִּמְנָחוֹת כָּתַב, דְּקוֹמְצוֹ וּמַקְטִיר הַקֹּמֶץ וְשִׁירַיִם נֶאֱכָלִין. וְהֵן דִּבְרֵי שְׁמוּאֵל בַּגְּמָרָא דְהָכָא. וְכֵן פָּסַק הָרַמְבַּ"ם בְּחִבּוּרוֹ. וְכֵן נַמִּי בַּיַּיִן כָּתְבוּ שָׁם דִּמְנַסְּכוֹ לַשִּׁיתִין. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב:
