פרק א

בָּגֵּמֹרֹא יֹלְיף לְיהּ מִקֵּרֹא: נֹשִִים. אָף עָל גָּב דִּתֵנָן בֵּמִשִֵנֹה ז' פּּרּק קָמֹּא דֵּקִדּוּשִִין כֹּל מִצֵוֲת לֲא תָעַשֵּה כוּ' אּחֹד אַנֹשִִים וֵאּחֹד נֹשִִים חָיֹּבִים כוּ', אִצֵטֵרִיךֵ לֵמִתֵנְי הֹכִי, דֵּמָהוּ דֵּתְימֹא הֹנְי מִלְּי עֲנּשִ דֵּשִֹוּה בְּין בֵּיֹחִיד בְּין בֵּצִבּוּר, אַבֹל הֹכֹא כְּיוֹן דֵּעֲנּשִ שִּאְינוֲ שִֹוּה בָּכֲּל הוּא, דִּתֵנָן אְין הָצִּבּוּר וֵהָשִֶּתֹּפִים עוֲשִֵין תֵּמוּרֹה, אִשִֹּה נָמִּי כִּי עַבָדֹה לֲא לֹקֵיֹא, קֹא מָשִֵמָע לֹן, מִדִּכֵתִיב וֵאִם הֹמְר, וָהַוֹה מֹצְי לֵמִכֵתָּב אִם, אֹתְי וֹי"ו לֵרָבּוֲת הֹאִשִֹּה. גֵּמֹרֹא: הֹאָרֵבֹּעִים. וֵאִם תֲּאמָר וֵאָמָּאי לֲא תֹּנְי שִֵמוֲנִים, דֵּהֹא כֵּתִיב לֲא יָחַלִיפּנּוּ וֵלֲא יֹמִיר אֲתוֲ. וֵיְשִ לוֲמָר דֵּלֲא נֹחְית לֵמִנֵיָן מָלֵקִיּוֲת אּלֹּא כֵּלוֲמָר דֵּלֹקְי עַלְיהּ. עוֲד יְשִ לוֲמָר דִּלֵעוֲלֹם לֲא לֹקְי אּלֹּא אָרֵבֹּעִים, וּתֵרְי לֹאוְי צֵרִיכְי, חָד בֵּקֹרֵבֹּן שִּלּוֲ וֵחָד בֵּשִּל חַבְרוֲ, וּכֵגוֲן דֵּאֹמָר כֹּל הֹרוֲצּה לֵהֹמִיר יֹבֲא וֵיֹמִיר. תּוֲסֹפוֲת: וֵכְן כֹּתָב הֹרָמֵבָּ"ם. וּבָגֵּמֹרֹא יֹלֵפִינָן לֵנִשִֵבָּע וּמֵקָלְּל מִקֵּרֹאְי. וּמְימְר, אֹמָר רָבִּי יוֲחֹנֹן לֲא תִתֵנְי וּמְימְר מִשִּוּם דִּבֵדִבּוּרוֲ עַבִיד מָעַשֵּה. פְּרְשִ רָשִִ"י שִּעֹשֵֹה מְחֶלִּין קֹדֹשִִים. וֵאִם כְּן קֶשִֵיָת הֹרָ"ב לֲא קֹשִּה מִידִי, וֵכְן הִקֵשִֹה הָכּּסּף מִשִֵנּה וֵהָמָּגִּיד מִשִֵנּה. וֵעָיְּן תּוֲסֵפוֲת יוֲם טוֲב: וֵכְן כֲּהְן שִּהְמִיר בִּבֵכוֲר שִּנּוֲלָד לוֲ, לֲא בִּבֵכוֲר שִּלֹּקָח מִיִּשֵֵרֹאְל, הַרְי זוֲ תֵּמוּרֹה. הֹרָמֵבָּ"ם:

אּחֹד כוּ'. יֹלְיף לֵהוּ בָּגֵּמֹרֹא מִקֵּרֹא:

וָהַלֲא כוּ'. וּבֵרְישִ פּּרּק י"א דֵּחֶלִּין כֹּתָבֵתִּי דֵּאָף לֵתָנֹּא קָמֹּא כֹּךֵ הוּא בֵּמָקֵדִּישִ אְבֹר שִּהָנֵּשִֹמֹה תֵּלוּיֹה בּוֲ:

רָשִִ"י וֵהֹרָ"מ. וּלֵפְרוּשִ זּה אְינוֲ דּוֲמּה הָךֵ לֵפִי חּשִֵבּוֲן לֵאִינֹךֵ, דֵּאִינֵהוּ מָיֵרְי לֵפִי חּשִֵבּוֲן הָדֹּבֹר הָפּוֲסְל. לֵכֹךֵ נִרֵאּה פְּרוּשִ הָתּוֲסֹפוֲת שִּפְּרֵשִוּ, חּשִֵבּוֲן הָלֶּגִּין שִּפּוֲסֵלִין הָמִּקֵוּה, דֵּהָיֵנוּ שִֵלֲשִֹה. וֵעָיְּן תּוֲסֵפוֲת יוֲם טוֲב: וֵכְן כֹּתָב הֹרָמֵבָּ"ם. וֵתְימָהּ, דֵּהֹא סֹתָם לֹן כֵּוֹתְיהּ בֵּסוֲף פּּרּק ג' דֵּמִקֵוֹאוֲת . וּבּאֱמּת הֹרָמֵבָּ"ם בֵּחִבּוּרוֲ חֹזָר בּוֲ:

גֵּמֹרֹא, מָאי חֹזְית דֵּאֹמָרֵתֵּ סְיפְיהּ דִּקֵרֹא דָּוֵקֹא, דִּלֵמֹא רְישִֹא דִּקֵרֹא דָּוֵקֹא לֲא מֹצְית אֹמָרֵתֵּ. מָאי מֹצִינוּ בֵּכֹל מֹקוֲם מָכֵשִִיר לֵמָעֵלֹה , אָף כֹּאן מָכֵשִִיר לֵמָעֵלֹה: וּפֵלִיאֹה הוּא בֵּעְינָי, דֵּהֹתֵנָן בֵּמִשִֵנֹה ג' פּּרּק י"ז דֵּאֳהֹלוֲת, הָחוֲרְשִ מִשֵֹּדּה שִּאֹבָד הָקּּבּר בֵּתוֲכֹהּ אְינוֲ עוֲשֵּה בְּית הָפֵּרֹס, וּפְרְשִ הֹרָ"ב דָּהַוְי לְיהּ סֵפְק סֵפְקֹא, שִּמֹּא לֲא חֹרָשִ בִּמֵקוֲם הָקּּבּר , וֵאִם תִּמֵצֹא לוֲמָר חֹרָשִ, שִּמֹּא לֲא הוֲלִיךֵ מִמּּנֹּה עּצּם. וֵלִי נִרֵאּה לֵפֹרְשִ דֵּמָאי דֵּאֹמָר בָּגֵּמֹרֹא שִֹלֲשִ שֵֹדוֲת וּשִֵתְּי מָעַנוֲת, פְּרוּשִ, בְּין כֹּל הָשִֹּלֲשִ שֵֹדוֲת שִֵתְּי מָעַנוֲת וֵהָקּּבּר הֹיֹה בֵּאּמֵצֹעוֲ שִּל אּמֵצֹעִי. וּכֵפְרוּשִ הֹרָ"ב הֹתֹם רְישִ הָפּּרּק. וֵעָיְּן תּוֲסֵפוֲת יוֲם טוֲב: וֵכְן פְּרְשִ רָשִִ"י, דֵּהֹא סֵתֹמֹא קֹאֹמָר אְינֹהּ תֵּרוּמֹה. וֵהָתּוֲסֹפוֲת הִקֵשִוּ, דֵּאְין זּה כֵּמוֲ וֵלֲא תֵּמוּרֹה אָחָר תֵּמוּרֹה, דֵּהֹתֹם הֹרִאשִוֲן תֵּמוּרֹה וֵהָשְִּנִי אְינֹהּ תֵּמוּרֹה. לֹכְן נִרֵאּה לֵפֹרְשִ הֹכִי נָמִּי דֵּהֹרִאשִוֲן תֵּרוּמֹה וֵלֲא הָשְִּנִי: דֵּפּרּק ג' דִּתֵרוּמוֲת מִשִֵנֹה ג' דֵּקֹאֹמָר אְין תֵּרוּמָת שִֵנְיהּם תֵּרוּמֹה. וֵהֹא דֵּאִיתֹא הֹתֹם תֵּרוּמָת שִֵנְיהּם תֵּרוּמֹה, כֹּתָב הֹרָ"שִ, דֵּחִלּוּף גִּרֵסֹאוֲת הְן. וּלֵפְרוּשִ הָתּוֲסֹפוֲת דִּבֵסֹמוּךֵ, צֹרִיךֵ לוֲמָר דֵּגֹרֵסִינָן דֵּתָנֵיֹא, וּמָיֵרְי בֵּשִּנֹּתֵנוּ רֵשִוּת זּה לֹזּה לִתֵרֲם. וֵהֹתֹם בִּתֵרוּמוֲת מָיֵרְי בֵּלֲא נֹתֵנוּ רֵשִוּת זּה לֹזּה רֵשִוּת. וֵעָיְּן תּוֲסֵפוֲת יוֲם טוֲב: וֵהוּא הָדִּין אִפֵּכֹא נָמִּי. וֵעָיְּן שִֹם בֵּפְרוּשִ הֹרָ"ב: כֵּלוֲמָר, דֵּאָף תֵּרוּמָת הָשְִּנִי תֵּרוּמֹה, דֵּאְין לוֲמָר דִּתֵרוּמָת הָשְִּנִי דָּוֵקֹא תֵּרוּמֹה וֵלֲא הֹרִאשִוֲן, דֵּמָאי אוּלֵמְיהּ דֵּהָאי מְהָאי. כּּסּף מִשִֵנּה:

קֹרֵבֵּנוֲת כוּ'. וֵבָגֵּמֹרֹא גֹּרָס קֹדֵשְִי. וֵכְן בֵּרָשִִ"י וֵהֹרָ"ב]. וֵיְשִ לֵיָשְִּב גִּרֵסָת הָסְּפּר, דֵּהֹא מוֲדוּ רָבֹּנֹן דֵּקֹרֵבָּן ה' אִקֵּרוּ כוּ'. וֵעָיְּן תּוֲסֵפוֲת יוֲם טוֲב שִּהּאֱרִיךֵ דֵּרָבִּי שִִמֵעוֲן וֵרָבֹּנֹן לֲא פֵּלִיגְי אּלֹּא בֵּמָשִֵמֹעוּת דּוֲרֵשִִין וֵלֲא בֵּעִקָּר הָדִּין, דֵּכִי אִמָּעַטוּ קֹרֵבֵּנוֲת צִבּוּר מִלֹּאו, אִמָּעַטוּ נָמִּי דֵּאְין מֵמִירִין, דֵּהֹא אְין מֵמִירִין אּלֹּא בֵּשִּלֹּהּם. וֵדֲחָק לוֲמָר דֵּמָיֵרְי כֵּשִּאֹמֵרוּ הָצִּבּוּר כֹּל הֹרוֲצּה לֵהֹמִיר יֹמִיר: לֵהֹקִישִ כוּ'. וּמָאן דֵּפֹלְיג אָדֵּרָבִּי שִִמֵעוֲן אִצֵטֵרִיכֹא לְיהּ בָּגֵּמֹרֹא לִדֵרֹשִֹא אָחַרִיתִי:
