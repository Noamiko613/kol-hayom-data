פרק ד

אֲבָל בִּזְבָחִים דַּף ק"ט כָּתְבוּ הַתּוֹסָפוֹת דְּאֵין לְפָרֵשׁ לְהִתְחַיֵּב בַּחוּץ. דְּהָא לֹא תְנַן אֶלָּא עוֹלָה וְאֵמוּרִין, אֲבָל חֲמִשָּׁה דְבָרִים לֹא, דְּאַף עַל גַּב דִּמְחַיְּבוּ בַחוּץ הָנֵי מִלֵּי בִפְנֵי עַצְמוֹ, אֲבָל אִצְטָרוֹפֵי לֹא כוּ'. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: כָּתְבוּ הַתּוֹסָפוֹת דְּאֵין לְפָרֵשׁ לְעִנְיַן מַחֲשֶׁבֶת פִּגּוּל, דְּהָא אֵין אֲכִילָה וְהַקְטָרָה מִצְטָרְפִים. אֶלָּא מַיְרֵי לְעִנְיַן אֲכִילָה דְּמִחַיַּב הָאוֹכֵל מִשּׁוּם פִּגּוּל. וְאִי מֵרֵישָׁא דְקָתָנֵי קָדְשֵׁי מִזְבֵּחַ מִצְטָרְפִין, הֲוָה אֲמֵינָא בְמִין אֶחָד. קָא מַשְׁמַע לָן אֲפִלּוּ בִשְׁנֵי מִינִין כוּ'. וּקְצָת תֵּימַהּ דְּלֹא תָנֵי חֲמִשָּׁה דְבָרִים בִּשְׁלָמִים, וְיֵשׁ לוֹמַר מִשּׁוּם דְּאַיְרֵי נַמִּי בִמְעִילָה. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב:

דְּאֵין אִסּוּר חָל עַל אִסּוּר: כִּדְיָלֵיף לֵיהּ בַּגְּמָרָא מִקְּרָא. וּשְׁאָר שְׁרָצִים כְּגוֹן הַצְּפַרְדֵּעַ וְהַנָּחָשׁ וְהָעַקְרָב וְכַיּוֹצֵא בָהֶן אַף עַל פִּי שֶׁהֵן אֲסוּרִים בַּאֲכִילָה הֲרֵי הֵם טְהוֹרִים מִכְּלוּם וַאֲפִלּוּ כָאֳכָלִין טְמֵאִין אֵינָן. הָרַמְבַּ"ם: וְהַאי קְרָא מְיֻתָּר הוּא לִדְרָשָׁא לְמִדְרַשׁ בֵּיהּ הָכִי, דְהָא כְתִיב קְרָא אַחֲרִינָא אֵלֶּה הַטְּמֵאִים לָכֶם. רַשִׁ"י:

אֵין כוּ'. פֵּרוּשׁ לְטֻמְאָה. וּבַגְּמָרָא, לֹא שָׁנוּ אֶלָּא לְטֻמְאַת הַיָּדַיִם, אֲבָל לְעִנְיַן אֲכִילָה מִצְטָרְפִים. דְּתַנְיָא רַבִּי אֱלִיעֶזֶר אוֹמֵר לֹא יֵאָכֵל כִּי קֹדֶשׁ הוּא , כָּל שֶׁבַּקֹּדֶשׁ פָּסוּל, בָּא הַכָּתוּב לִתֵּן לֹא תַעֲשֶׂה עַל אֲכִילָתוֹ. וּפֵרֵשׁ רַשִׁ"י לְטֻמְאַת יָדַיִם שֶׁגָּזְרוּ בְנוֹתָר וּפִגּוּל שֶׁמְּטַמְּאִין אֶת הַיָּדַיִם מִשּׁוּם חַשְׁדֵי כְהֻנָּה, אוֹ מִשּׁוּם עֲצֵלֵי כְהֻנָּה. וְכֵיוָן דִּגְזֵרָה דְרַבָּנָן הוּא אֵינָן מִצְטָרְפִין דַּהֲוָה לֵיהּ גְּזֵרָה לִגְזֵרָה. אֲבָל לְעִנְיַן אֲכִילָה אִי אִיכָּא פָחוֹת מִכַּזַּיִת פִּגּוּל וְנוֹתָר מַשְׁלִים לְכַזַּיִת חַיָּבִים עָלָיו מִשּׁוּם פִּגּוּל, וְאִי רֹב דְּנוֹתָר כוּ' חַיָּבִים עָלָיו מִשּׁוּם נוֹתָר. וְאָמַר לִי רַבִּי דְּאִי הֲוָה חֲצִי זַיִת מִזֶּה וַחֲצִי כוּ' מִצְטָרְפִין לְחַיֵּב מִשּׁוּם נוֹתָר וּמִשּׁוּם פִּגּוּל. עַד כָּאן. וְנִרְאֶה לִי דְּלֹא לְחַיְּבוֹ בִּשְׁתַּיִם, שֶׁאִם כֵּן אַתָּה מְחַיְּבוֹ עַל פָּחוֹת מִכְּשִׁעוּר, אֶלָּא לְעִנְיַן מַלְקוֹת לְחַיְּבוֹ אִם הִתְרָה בוֹ מִשּׁוּם נוֹתָר אוֹ מִשּׁוּם פִּגּוּל: מִפְּנֵי כוּ'. אֵין זֶה עִקַּר הַטַּעַם, דְּהָא אֲמַרַן דִּלְעִנְיַן אֲכִילָה מִצְטָרְפִין וּמִשֵּׁם אֶחָד נַפְקָא וּלְטֻמְאַת יָדַיִם הוּא דְאֵין מִצְטָרְפִין, וְעִקַּר טֻמְאָה זוֹ אֵינָהּ אֶלָּא מִדְּרַבָּנָן, וְאִם כֵּן מַאי מִפְּנֵי שֶׁהֵם שְׁנֵי שֵׁמוֹת. אֶלָּא טַעֲמָא כִּדְכָתַב רַשִׁ"י דְלָא גָזְרִינַן גְּזֵרָה לִגְזֵרָה כוּ', וְיֵשׁ לוֹמַר דִּלְפִי שֶׁנִּמְצָא כַּמָּה פְּעָמִים שֶׁגָּזְרוּ שְׁתֵּי הַגְּזֵרוֹת כְּאַחַת תָּנֵי מִפְּנֵי כוּ', כְּלוֹמַר וּלְפִיכָךְ לֹא רָצוּ לִגְזֹר כְּאַחַת. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: מִצְטָרְפִים. תַּנְיָא רַבִּי שִׁמְעוֹן אוֹמֵר מַה טַּעַם כוּ' מִי גָרַם לַשֵּׁנִי לָאו רִאשׁוֹן. וְכֵיוָן דְּמִכֹּחַ רִאשׁוֹן אָתֵי לֵיהּ טֻמְאָה, הָוֵי כִּי רִאשׁוֹן, וּלְהָכִי מִצְטָרְפִין:

וְכָמוֹהָ בְּלֹא קְלִפָּתָהּ: כָּל כוּ'. וּבְפֶרֶק י' דְּמִקְוָאוֹת מִשְׁנָה ז' מְפָרֵשׁ הָרַ"ב דַּוְקָא הַשִּׁבְעָה מַשְׁקִין: וּבִמְלֹא כוּ'. וְהָרַ"ב הֶעְתִּיק וְכִמְלֹא. וְעַיֵּן פֶּרֶק ח' דְּיוֹמָא מִשְׁנָה ב':

וְתֵימַהּ דִּבְמִשְׁנָה ג' פֵּרֵשׁ הָרַ"ב כָּל הַנְּבֵלוֹת מִצְטָרְפוֹת לְטֻמְאָה אֲבָל לֹא לְמַלְקוֹת לְפִי שֶׁהֵן שְׁנֵי שֵׁמוֹת. וְצָרִיךְ לוֹמַר דְּמַתְנִיתִין דְּהָכָא פְלִיגָא. עַיֵּן תּוֹסְפוֹת יוֹם טוֹב: וְלִכְאוֹרָה לַפֵּרוּשׁ הָרִאשׁוֹן אֵין צָרִיךְ לוֹמַר דְּאֵינָן מִצְטָרְפוֹת לִלְקוֹת, כֵּיוָן שֶׁהֵם שְׁנֵי שֵׁמוֹת כְּדִתְנַן בְּמִשְׁנָה ג'. אֶלָּא דְבַגְּמָרָא אִפְּכָא שָׁמְעִינַן דְּפָרֵיךְ וּמִי צָרִיךְ לְרַבִּי שִׁמְעוֹן לְצָרוֹפֵי וְהָתַנְיָא רַבִּי שִׁמְעוֹן אוֹמֵר כָּל שֶׁהוּא לִמְכֹּר, וּמְשָׁנֵי תָנֵי אֵין צָרִיךְ לְצָרֵף: וְתֵימַהּ דִּלְעֵיל נַמִּי שֶׁהֵם שָׁוִים בְּדָבָר אֶחָד וְאַף עַל פִּי כֵן אֵינָן מִצְטָרְפִין. וְהָרַמְבַּ"ם מְסַיֵּם שֶׁאָנוּ חוֹשְׁשִׁין לְהַשְׁווֹת הַשִּׁעוּר לְטֻמְאָה בִּדְבָרִים שֶׁמְּטַמְּאִין לֹא בִדְבָרִים שֶׁמִּטַּמְּאִים. וּלְשׁוֹן רַשִׁ"י הַמְקַצֵּעַ מִכֻּלָּן יַחַד שֶׁחִבֵּר קִצּוּעֵיהֶן שֶׁקִּצֵּץ כָּל הַמִּינִין וְעָשָׂה מֵהֶן טֶפַח עַל טֶפַח וּתְפָרָן יַחַד. וְכֵן פֵּרֵשׁ רַשִׁ"י כָּאן. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב:
