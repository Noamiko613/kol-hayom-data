פרק ג - משנה א

וְהָתָם תָּנֵי לַהּ מִשּׁוּם תְּמוּרָה. וְהָכָא מִשּׁוּם מְעִילָה. גְּמָרָא:

פרק ג - משנה ב

לָאו דַּוְקָא דַּאֲפִלּוּ אָמַר אֵלּוּ לְחוֹבָתִי מְפֹרָשִׁין הֵן כְּמוֹ שֶׁתְּנַן בְּמִשְׁנָה ד' פֶּרֶק ד' דְּנָזִיר: לֹא נֶהֱנִין. מִדְּרַבָּנָן. אוֹ שֶׁמָּא דַּאֲפִלּוּ מִדְּאוֹרַיְתָא אָסוּר דְּקַיְמָא לִשְׁלָמִים. מִידִי דַּהֲוָה אַשְּׁלָמִים מֵחַיִּים דִּנְהִי דִמְעִילָה לֵיכָּא אִסּוּרָא דְאוֹרַיְתָא אִיכָּא. תּוֹסָפוֹת: רַשִׁ"י. וּלְפִי זֶה כְּשֶׁנֶּהֱנָה בְכֻלָּן מָעַל, וּמַתְנִיתִין סְתָמָא תְנַן וְלֹא מוֹעֲלִין דְּמַשְׁמַע דְּאֵין כָּאן מְעִילָה כְלָל, וְכֵן הָא דִתְנַן מִפְּנֵי שֶׁהֵן רְאוּיִין לָבֹא כֻּלָּן שְׁלָמִים מַשְׁמַע דִּבְכֻלָּן בְּיַחַד רָאוּי לָבֹא שְׁלָמִים. לָכֵן נִרְאֶה כְּפֵרוּשׁ הַתּוֹסָפוֹת שֶׁפֵּרְשׁוּ דְּכֵיוָן דְּלָא פֵרֵשׁ, אִם יִרְצֶה יָבִיא כָל הַמָּעוֹת לִשְׁלָמִים. כְּלוֹמַר וְהַשְּׁאָר יָבִיא מִבֵּיתוֹ. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: לִנְדָבָה. בַּגְּמָרָא פָרֵיךְ, וַהֲלֹא דְּמֵי חַטָּאת (וּשְׁלָמִים. תּוֹסָפוֹת) מְעֹרָבִין בָּהֶן. וּמְשָׁנֵי הֲלָכָה הוּא בְּנָזִיר. פֵּרְשׁוּ הַתּוֹסָפוֹת הֲלָכָה לְמשֶׁה הוּא דִּסְתוּמִים יִפְּלוּ לִנְדָבָה. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב:

פרק ג - משנה ג

וְהוּא לְשׁוֹן אֲרָמִית תְּסֻכֵּנִי בְּבֶטֶן אִמִּי , אַשְׁתִּית לִי. וְגִידִים תְּשׂוֹכְכֵנִי , וְגִידַיָא אַשְׁתֵּתַנִי:

פרק ג - משנה ד

הָכִי יָלֵיף לְהוּ בַגְּמָרָא מִקְּרָא: דְּאִי אֵפֶר כִּירָתוֹ מַאי בִתְחִלָּה, וְהַתּוֹסָפוֹת כָּתְבוּ דְּאֵפֶר כִּירָתוֹ פְּשִׁיטָא, דְּאַטּוּ גָרַע מִמַּקְדִּישׁ זֶבֶל אַשְׁפָּתוֹ דִּלְקַמָּן. דְּהָא וַדַּאי דְּהַכֹּל יָכוֹל אָדָם לְהַקְדִּישׁ לְבֶדֶק הַבַּיִת: קָשֶׁה לִי דְּאִם כֵּן כָּל הָנֵי דְאֵין מוֹעֲלִין בָּהֶן מִשֶּׁנַּעֲשֵׂית מִצְוָתָן דִּלְעֵיל נַמִּי, וּמַאי אִרְיָא בְּדִשּׁוּן דַּוְקָא. וְהַתּוֹסָפוֹת כָּתְבוּ דְּמַיְרֵי בְּאָדָם שֶׁנָּטַל מֵהַדֶּשֶׁן לְאַחַר שֶׁהוּרְמָה וְהִקְדִּישָׁהּ מוֹעֲלִין בָּהּ מִן הַתּוֹרָה. וְהָרַמְבַּ"ם נִרְאֶה שֶׁמְּפָרֵשׁ בְּאֵפֶר כִּירָה. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: נִרְאֶה לִי דְלָא דָק, שֶׁהֲוָה לֵיהּ לְמֵימַר כְּשֶׁנָּדַר זֶה: לֹא יָדַעְתִּי לָמָּה הֻצְרַךְ לִכְתֹּב לְאַלְתַּר. תּוֹסְפוֹת יוֹם טוֹב. וְלִי נִרְאֶה דְּקָא מַשְׁמַע לָן דַּאֲפִלּוּ אִם אַחַר שֶׁנֶּהֱנָה הֶחֱזִירָהּ לִמְקוֹמָהּ וְאִם כֵּן נוּכַל לְשַׁעֲרוֹ, מִכָּל מָקוֹם כֵּיוָן דִּזְמַן מָה לֹא הָיָה אֶפְשָׁר לְשַׁעֵר בִּשְׁבִילוֹ, מוֹעֲלִים בּוֹ. תּוֹסָפוֹת חֲדָשִׁים: וְכֵן בְּנֵי יוֹנָה שֶׁעָבַר זְמַנָּן לֹא דָמוּ לְחַטָּאת שֶׁעָבְרָה שְׁנָתָהּ דְּאָמַר בְּרֵישׁ פִּרְקִין דְּמוֹעֲלִין בּוֹ קֹדֶם כַּפָּרָה, דְּהָתָם מִשּׁוּם דְּאָזֵיל לִרְעִיָּה וְאִית לַהּ פִּדְיוֹן, אֲבָל הָכָא אֵין לְעוֹפוֹת פִּדְיוֹן וְלֹא אִקְרוּ קָדְשֵׁי ה'. תּוֹסָפוֹת: דְּתַמּוּת וְזַכְרוּת בִּבְהֵמָה וְאֵין תַּמּוּת וְזַכְרוּת בְּעוֹפוֹת:

פרק ג - משנה ה

אַף עַל גַּב דְּדָרְשִׁינַן בִּפְסוּלֵי הַמֻּקְדָּשִׁין בָּשָׂר וְלֹא חָלָב. תּוֹסְפוֹת יוֹם טוֹב. פֵּרוּשׁ אַף דְּחָלָב אָסוּר, לֹא הָוֵי קָדְשֵׁי ה'. וְעוֹד כֵּיוָן דְּאֵין חַיּוּת הַבְּהֵמָה תָּלוּי בוֹ דִלְהָוֵי חָשׁוּב כְּגוֹן הַבְּהֵמָה לִמְעֹל בָּהּ. תּוֹסָפוֹת. תּוֹסָפוֹת חֲדָשִׁים:

פרק ג - משנה ו

לַעֲשׂוֹת בּוֹ טִיט: שֶׁהֲרֵי הַקֶּרֶב וְהַכְּרָעַיִם לֹא הָיוּ רוֹחֲצִים אֶלָּא בְאַמַּת הַמַּיִם שֶׁבָּעֲזָרָה. תּוֹסָפוֹת: וּבוֹר הַיְנוּ מַיִם מְכֻנָּסִים: דְּשׁוֹבָךְ שָׁוֶה הַרְבֵּה, וְאֵין עוֹקְרִין אֲבָנָיו לָשׂוּם אוֹתוֹ בְּבֶדֶק הַבַּיִת. תּוֹסָפוֹת. וְכֵן כָּתַב הָרַשְׁבַּ"ם. וְכָתַב עוֹד דְּכָל הָנֵי דְתָנֵי וְלֹא לְבֶדֶק הַבַּיִת, אַמַּה שֶּׁבְּתוֹכָהּ קָאֵי: וְדִבְרֵי תֵימַהּ הֵן, דְּוַדַּאי דְּבִכּוּרִים לַמִּזְבֵּחַ נַמִּי לֹא הָווּ, שֶׁהֵן מִמַּתְּנוֹת כְּהֻנָּה. וְעוֹד מַאי בִכּוּרִים שָׁיְכָא בְשֶׁל הֶקְדֵּשׁ, דְּלֹא חִיְּבָה הַתּוֹרָה אֶלָּא בִכּוּרֵי אַדְמָתְךָ. וְרַשִׁ"י פֵרֵשׁ דְּהַאי אִילָן גֶּפֶן הוּא דְּרָאוּי לַמִּזְבֵּחַ לְנִסּוּךְ הַיַּיִן. וְהַתּוֹסָפוֹת הוֹסִיפוּ זַיִת דְּרָאוּי לְשֶׁמֶן לִמְנָחוֹת. וְאֵין רְאוּיָה לְבֶדֶק הַבַּיִת, דַּעֲצֵי גֶפֶן דַּקִּין הֵן, אִי נַמִּי דְאִם כֵּן הָיָה פּוֹחֵת דָּמָיו. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: בָּהֶם. פֵּרֵשׁ רַבֵּנוּ תָם שֶׁעָקַר חוּלְיָא וְנֶהֱנֶה מִמֶּנָּה דַּהֲוֵי תָלוּשׁ. דְּהָא אָמְרִינַן לְקַמָּן בַּבְּרַיְתָא דְאֵין מְעִילָה בִּמְחֻבָּר. תּוֹסָפוֹת: וְלֹא אָמְרִינַן דְּתִקְנֶה חֲצַר הֶקְדֵּשׁ כְּמוֹ חֲצַר הֶדְיוֹט, דְּחָצֵר מִשּׁוּם יָד אִתְרַבַּאי וְלֹא מָצִינוּ יָד לְהֶקְדֵּשׁ. תּוֹסָפוֹת. וְקָשֶׁה דְלִקְנֵי מִטַּעַם שְׁלִיחוּת דְּהָנֵי כָּהֲנֵי שְׁלוּחֵי שָׁמַיִם הֵם כוּ', וְעוֹד קָשֶׁה דְּמַאי שְׁנָא מִבֵּיצַת תַּרְנְגֹלֶת וַחֲלֵב חֲמוֹרָה דִלְעֵיל. וּמִיהוּ בָּזֶה יֵשׁ לוֹמַר דְּאִין הָכִי נַמִּי דְמַתְנִיתִין דִּלְעֵיל דְּלֹא כִּי הַאי תַּנָּא, וְהַיְנוּ דִּבְבָבָא בַתְרָא דַּף ע"ט גָּרַס בָּהּ דִּבְרֵי רַבִּי יְהוּדָה. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: וּלְשׁוֹן רַשִׁ"י שֶׁמַּתְנִין קֹדֶם לָכֵן עַל מְנָת כֵּן אֲנִי מַכְנִיס נְקֵבָה זוֹ לַדִּיר לְהִתְעַשֵּׂר שֶׁאִם תֻּקְדָּשׁ כוּ', וְשָׁמַעְתִּי דִלְהָכִי מַהֲנֵי תְנָאָה הוֹאִיל וִיכוֹלִין לְעַכֵּב שֶׁלֹּא תִכָּנֵס נְקֵבָה זוֹ לַדִּיר עַד לְאַחַר זְמַן, הִלְכָּךְ מַהֲנֵי תְנָאָה וְהָוֵי כִנְדָבָה. וּלְהָכִי קָרֵי לְהוּ אֲחֵרִים, שֶׁאֵינָן מַחְמִירִין עֲלֵיהֶן כָּרָאוּי. וְכֵן נַמִּי בִּוְלַד מֻקְדָּשִׁים לֹא הָיוּ מַקְדִּישִׁין חֲלֵב הָאֵם: מִטַּעַם שֶׁכָּתְבוּ הַתּוֹסָפוֹת דְּלִישָׁנָא לֹא מַשְׁמַע הָכִי וַהֲוָה לֵיהּ לְמֵימַר וַאֲחֵרִים מַתְנִים בְּכָךְ כוּ'. עַיֵּן תּוֹסְפוֹת יוֹם טוֹב: מַשְׁמַע דְּמִדְּרַבָּנָן הוּא. וְקָשֶׁה, דְּבַגְּמָרָא יָלְפִינַן לֵיהּ מִקְּרָא. וְהַתּוֹסָפוֹת הִקְשׁוּ עוֹד לָמָּה לִי קְרָא דִּפְשִׁיטָא דְאָסוּר דַּחֲלֵב הַמֻּקְדָּשִׁין לֹא נֶהֱנִין. וְעוֹד מַאי וַאֲחֵרִים כוּ' וּמַה נְּדָבָה הִיא זֹאת וְכִי הַהֶקְדֵּשׁ נֶהֱנֶה מִנִּדְבָתָהּ. לְכָךְ נִרְאֶה כוּ' דְּמַיְרֵי בִּוְלָדוֹת שֶׁיָּלְדוּ אַחַר שֶׁנַּעֲשׂוּ הֶקְדֵּשׁ וְהַוְּלָדוֹת קָדָשִׁים הֵן, וְאַף עַל פִּי כֵן אָסוּר לַהֲנִיקָן מִיַּלְפוּתָא דִקְרָא. וְנִיחָא גַם כֵּן נִדְבַת הָאֲחֵרִים. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: הֶקְדֵּשׁ. דְּכֶרֶם רֵעֲךָ אָמַר רַחֲמָנָא וְלֹא כֶרֶם שֶׁל הֶקְדֵּשׁ. רַשִׁ"י: הָרַמְבַּ"ם. וְרַשִׁ"י פֵרֵשׁ מִשּׁוּם דְּעִנְיָנָא כֻלַּהּ בְּחֻלִּין קָמַיְרֵי:

פרק ג - משנה ז

וְאִם תֹּאמַר וְלִתְנֵי אִידִי וְאִידִי תּוֹךְ שֵׁשׁ עֶשְׂרֵה וְלִתְנֵי סֵיפָא מוֹעֲלִין. אִי נַמִּי אִידִי וְאִידִי אַחַר שֵׁשׁ עֶשְׂרֵה וְלִתְנֵי בָרֵישָׁא מוֹעֲלִין. וְיֵשׁ לוֹמַר דְּתָנָא הָכִי לְאַשְׁמְעִינַן דִּבְתַרְוַיְהוּ בֵּין בְּתוֹךְ שֵׁשׁ עֶשְׂרֵה בֵּין חוּץ לְשֵׁשׁ עֶשְׂרֵה אַמָּה לֹא נֶהֱנִין וְלֹא מוֹעֲלִין. תּוֹסָפוֹת: זְקֵנִים. שֶׁלֹּא יָכְלוּ לִטְרֹחַ אַחַר עֲרָבָה. אִי נַמִּי לַחֲשִׁיבוּתָם הֻרְשׁוּ לִקַּח: דְּמִשֶּׁזָּקְפוּהָ דְּנַעֲשֵׂית מִצְוָתָהּ נֶהֱנִין מִמֶּנָּה. רַשִׁ"י: קָסָבַר מִצְוֹת לָאו לֵהָנוֹת נִתְּנוּ. תּוֹסָפוֹת. עַיֵּן תּוֹסְפוֹת יוֹם טוֹב:

פרק ג - משנה ח

וְאִם תֹּאמַר אַמַּאי לֹא נֶהֱנִין כֵּיוָן דְּאַיְתוּ מֵעָלְמָא. יֵשׁ לוֹמַר דְּגָזְרִינַן דִּלְמָא אָתֵי לְאַחְלוּפֵי בְקִנִּים הַמְחֻבָּרִין בָּאִילָן. תּוֹסָפוֹת: וְיֵשׁ לִתְמֹהַּ מַאי שְׁנָא מִשֶּׁל הֶקְדֵּשׁ דְּאֵין נֶהֱנִין. וּבַגְּמָרָא מַאי יַתִּיז, אֶפְרוֹחִים. וְכֵן כָּתַב הָרַמְבַּ"ם יַתִּיז בְּקָנֶה רְצוֹנוֹ לוֹמַר שֶׁיָּנִיס הָאֶפְרוֹחִים וְאָז יִקַּח אוֹתָם, וּמִכָּל מָקוֹם הִתִּיר הַקֵּן בָּאֲשֵׁרָה. אֲבָל הַתּוֹסָפוֹת כָּתְבוּ דַּאֲסוּרָה כְּמוֹ בְשֶׁל הֶקְדֵּשׁ כוּ', וְהָא דְתָנֵי יַתִּיז בַּקָּנֶה לְאַשְׁמְעִינַן דְּאַף עַל גַּב דַּהֲוֵי דָבָר מָאוּס אֲפִלּוּ הָכִי שָׁרֵי. וְרַשִׁ"י פֵרֵשׁ דִּלְהָכִי שָׁרֵי בַאֲשֵׁרָה, דְּלֵיכָּא לְמִגְזַר, מִשּׁוּם דִּבְדִילֵי מִנֵּיהּ. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: לֹא כוּ'. לְפִי שֶׁהַגִּזְבָּרִין לֹא קָנוּ וְהִקְדִּישׁוּ אֶלָּא דָבָר הַצָּרִיךְ לַבִּנְיָן. וְלֹא שִׁפּוּי וּנְבִיָּה. רַשִׁ"י: הָרַמְבַּ"ם כָּתַב נְמִיָּה הֵם הַחֲלָקִים הַקָּשִׁים שֶׁיֵּשׁ בָּעֵצִים הַנִּתָּזִים בִּשְׁעַת הַנִּגּוּר וְהַהַחְלָקָה כְשֶׁמְּנַגְּרִים וּמַחֲלִיקִין אוֹתוֹ יִשָּׁאֵר מְקוֹמָן כְּמִין גֻּמָּא בָעֵץ. וּמִקְצָת הַנּוּסְחָאוֹת נְוִיָּה, וְהָעִנְיָן אֶחָד, לְפִי שֶׁאוֹתָן הַחֲלָקִים דּוֹמֶה לְנָוֶה עוֹמֶדֶת בִּפְנֵי עַצְמָהּ בְּתוֹךְ הָעֵץ:
