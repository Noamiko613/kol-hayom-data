פרק ה

שֶׁיִּפְגֹּם כוּ'. יָלְפִינַן לְהוּ בַגְּמָרָא מִקְּרָא: קַטְלָא כוּ'. אוֹרְחָא דְמִלְּתָא נָקַט, שֶׁדֶּרֶךְ הַנָּשִׁים לְהִתְקַשֵּׁט וּלְהִשְׁתַּמֵּשׁ בְּשֶׁל זָהָב, וְדֶרֶךְ הָאֲנָשִׁים לְהִתְלַבֵּשׁ בְּחָלוּק וְטַלִּית וְלַעֲשׂוֹת מְלָאכָה: וְכֵיוָן שֶׁפְּגָמָהּ הִכְחִישָׁהּ וּפִחֵת מִדָּמֶיהָ. רַשִׁ"י: מָעַל. וּמְעִילָה זוֹ מִדִּבְרֵיהֶם. הָרַמְבַּ"ם:

וּמִלִּישָּׁנָא דְמַתְנִיתִין מַשְׁמַע שֶׁנֶּהֱנָה בְשָׁעָה שֶׁפָּגַם. וּלְשׁוֹן הָרַמְבַּ"ם, כְּגוֹן שֶׁקָּרַע קְצָת מִבֶּגֶד הֶקְדֵּשׁ וְנָתַן אוֹתָהּ חֲתִיכָה שֶׁקָּרַע בְּבִגְדוֹ וְנִתְקַשֵּׁט בּוֹ וְנֶהֱנָה מִמֶּנּוּ בְּשָׁוֶה פְרוּטָה בְתַכְשִׁיטוֹ וְלֹא פָגַם בַּחֲתִיכָה שׁוּם דָּבָר אֲבָל פָּגַם בְּבֶגֶד הַהֶקְדֵּשׁ שֶׁקָּרַע מִמֶּנּוּ בְשָׁוֶה פְרוּטָה, הֲרֵי זֶה נֶהֱנָה וּפָגַם. אֲבָל הַדָּבָר שֶׁפָּגַם לֹא נֶהֱנָה בְאוֹתוֹ הַפְּגָם עַצְמוֹ: לְשׁוֹן רַשִׁ"י, וְהָכָא מַיְרֵי בְדָבָר שֶׁיֵּשׁ בּוֹ פְּגָם:

וּכְלֵי שָׁרֵת. גְּמָרָא, קַל וָחֹמֶר אִם אֲחֵרִים מֵבִיא לִקְדֻשָּׁתָן , הוּא עַצְמוֹ לֹא כָל שֶׁכֵּן. פֵּרֵשׁ רַשִׁ"י שֶׁיְּהֵא קָדוֹשׁ כָּל כָּךְ שֶׁיְּהֵא בוֹ מוֹעֵל אַחַר מוֹעֵל: כְּרַבִּי יְהוּדָה בְּמִשְׁנָה ח' פֶּרֶק ב' דְּקִדּוּשִׁין דִּסְבִירָא לֵיהּ הֶקְדֵּשׁ בְּשׁוֹגֵג מִתְחַלֵּל: דְּהַשְׁתָּא יָכוֹל לִפְדּוֹתָן לְהַאֲכִיל לְאָדָם: וּפְלִיגֵי אַתַּנָּא קַמָּא וְרַבִּי שִׁמְעוֹן דְּמִשְׁנָה ג' פֶּרֶק ז' דִּתְמוּרָה, וְשָׁלֹשׁ מַחְלֹקוֹת בַּדָּבָר. תּוֹסָפוֹת. וְצָרִיךְ עִיּוּן דְּהָרַ"ב פָּסַק כְּרַבָּנָן דְּהָתָם וּדְהָכָא. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב:

וְאִם תֹּאמַר וּמִכָּל מָקוֹם הֲרֵי שָׁלַח בָּהּ יָד וְלִחַיַּב, דִּשְׁלִיחוּת יָד אֵינָהּ צְרִיכָה חִסָּרוֹן אֶלָּא שֶׁיִּתְכַּוֵּן לְחַסֵּר, וְהָכָא נַמִּי קָא מְכַוֵּן לִטְּלָהּ לְעַצְמוֹ. תֵּרְצוּ הַתּוֹסָפוֹת שֶׁיֵּשׁ לוֹמַר שֶׁאֵין אָדָם מוֹעֵל אֶלָּא בְמִתְכַּוֵּן לְהוֹצִיאוֹ מֵרְשׁוּת מִי שֶׁהוּא, וְהָכָא הוּא סָבַר שֶׁלּוֹ הוּא: וְנִמְצָא מְחַסֵּר לַהֶקְדֵּשׁ וּפְגָמוֹ, וְעוֹד דְּהַיְנוּ שִׁנּוּי מַמָּשׁ כוּ': מָעַל. בִּנְתִינַת הַמַּתָּנָה בְטוֹבַת הֲנָאָה הַבָּאָה לוֹ. הָרַמְבַּ"ם: מַשְׁמַע דְּהַיְנוּ נַמִּי פְּגָם שֶׁשִּׁנָּה אוֹתָהּ בַּבִּנְיָן כוּ'. וְהַתּוֹסָפוֹת הִקְשׁוּ דְהָא שִׁנּוּי הַחוֹזֵר לִבְרִיָּתוֹ לֹא הֲוָה שִׁנּוּי. וּבְלָשׁוֹן רִאשׁוֹן פֵּרֵשׁ רַשִׁ"י שֶׁפִּחֲתָהּ וַחֲתָכָהּ, וְכֵן פֵּרְשׁוּ הַתּוֹסָפוֹת: נָטַל כוּ'. קַשְׁיָא לִי הָא תוּ לָמָּה לִי. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: אוֹמֵר לוֹ כוּ'. וְלֹא מִחַסַּר מְשִׁיכָה, מַה שֶּׁאֵין כֵּן שְׁאָר אֻמָּנוּת דְּמִחַסְּרָן מְשִׁיכָה. וּלְדִידָן דְּקַיְמָא לָן דְּבַר תּוֹרָה מָעוֹת קוֹנוֹת מַתְנִיתִין בְּבַלָּן נָכְרִי, וּמִשּׁוּם הָכִי דַוְקָא בַּלָּן אֲבָל שְׁאָר אֻמָּנוּת בָּעֵי מְשִׁיכָה דְּכֵיוָן דִּבְיִשְׂרָאֵל מָעוֹת קוֹנוֹת בְּנָכְרִי קוֹנֶה בִמְשִׁיכָה, דְּדָרְשִׁינַן בַּגְּמָרָא רֵישׁ פֶּרֶק ב' דִּבְכוֹרוֹת לַעֲמִיתֶךָ בְּכֶסֶף, הָא לְעוֹבֵד כּוֹכָבִים בִּמְשִׁיכָה. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב:

וּבַגְּמָרָא תָנֵי אֲכִילָתוֹ וַאֲכִילַת חֲבֵרוֹ אֲפִלּוּ מִכָּאן וְעַד שָׁלֹשׁ שָׁנִים. וְהָרַמְבַּ"ם כָּתַב כֵּן גַּבֵּי הֲנָאַת עַצְמוֹ:
