פרק ה - משנה א

דַּהֲוָה לֵיהּ לְמֵימַר וְרָחַץ בַּמַּיִם בָּעֶרֶב, אֶלָּא לוֹמַר דְּעוֹד טָמֵא גַּם כֵּן כָּל כְּלִי שֶׁיִּגַּע בּוֹ. וּכְלֵי שֶׁטֶף הַיְנוּ כָּל שְׁאָר כֵּלִים חוּץ מִכְּלֵי חֶרֶס. עַל שֵׁם שֶׁיֵּשׁ לָהֶם טָהֳרָה בְּמִקְוֶה. רַשִׁ"י: דִּכְלִי שֶׁטֶף דּוֹמֶה קְצָת לְבֶגֶד, דַּהֲוֵי נַמִּי כְּלִי, מַה שֶּׁאֵין כֵּן הָאָדָם. וּכְלִי חֶרֶס אֵין דּוֹמֶה לוֹ, שֶׁאֵין לוֹ טָהֳרָה בְּמִקְוֶה: כָּל הַמְטַמֵּא כוּ'. כָּל אָדָם הַמְטַמֵּא קָאָמַר, דְּהָא מַשְׁכַּחַת כֵּלִים שֶׁאֵין מְטַמְּאִין כֵּלִים, וְאִם נָגְעוּ מְטַמְּאִין אֳכָלִים לִהְיוֹת תְּחִלָּה כִּדְמוּכָח בְּמַתְנִיתִין דִּלְקַמָּן. דְּדַוְקָא בְּאֵין נוֹגְעִים מַיְרֵי סֵיפָא, הָא נוֹגְעִים מְטַמְּאִין שְׁנַיִם. וְכֵן הוּא בְּהֶדְיָא בַתּוֹסֶפְתָּא. וְשָׁם מְפָרֵשׁ הָרַ"ב דְּמַתְנִיתִין כָּל אָדָם קָאָמַר: כְּלוֹמַר, לְעִנְיַן אֳכָלִים וּמַשְׁקִים. וְאַף עַל פִּי שֶׁאֵינוֹ אַב הַטֻּמְאָה לְעִנְיַן אָדָם וּכְלֵי חֶרֶס:

פרק ה - משנה ב

אֲבָל סֵיפָא כָּל שֶׁהַזָּב נִשָּׂא עָלָיו טָהוֹר, דַּוְקָא בְלֹא נָגַע. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: טָמֵא. וּלְקַמָּן מְפָרֵשׁ שֶׁאֵין כֻּלָּם שָׁוִין: טָהוֹר. לְגַמְרֵי. וְאֵין בָּהֶן אֲפִלּוּ טֻמְאַת מַדָּף, כִּדְאִיתָא בְתוֹרַת כֹּהֲנִים. תּוֹסָפוֹת. וְצָרִיךְ לוֹמַר דְּמַיְרֵי שֶׁלֹּא הוּסַט. דְּעַל יְדֵי הֶסֵּט הָא אֲמַרַן לְעֵיל דְּמִטַּמְאִין. וְהָרַ"שׁ פֵּרֵשׁ, דַּוְקָא עַל יְדֵי הֶסֵּט, וְטָהוֹר מִשּׁוּם דְּהַנּוֹשֵׂא אֵין בּוֹ רוּחַ חַיִּים, וּלְכָךְ תָּנֵי חוּץ מִן הָאָדָם. וְקָשֶׁה, דְּהֵיכִי מְטַהֲרִין הָכָא לְגַמְרֵי: דְּאֵין לְמֵדִין מִן הַכְּלָלוֹת: כֵּיצַד. אַרֵישָׁא נַמִּי קָאֵי: אֶצְבָּעוֹ. נִרְאֶה לִי שֶׁטֻּמְאָה זוֹ מִדִּבְרֵיהֶם. וּלְפִי זֶה צָרִיךְ לוֹמַר דְּמִשְׁכָּב וּמוֹשָׁב דְּסֵיפָא לֹא מַיְרֵי כְּשֶׁאֶצְבָּעוֹ בִּלְבַד מִלְמַעְלָן, דִּבְהוּ לְעוֹלָם רֻבּוֹ בָעִינַן. אֶלָּא מַתְנִיתִין דְּהָכָא וּדְהָתָם לָאו בַּחֲדָא שִׁיטָה, וְהָא כִדְאִיתָא וְהָא כִדְאִיתָא: שְׁנַיִם. לְפִי שֶׁהַמִּשְׁכָּב אַב הַטֻּמְאָה אֲפִלּוּ כְשֶׁנִּתְרוֹקֵן הַזָּב מִמֶּנּוּ. כִּדְיָלֵיף לֵיהּ בְּתוֹרַת כֹּהֲנִים:

פרק ה - משנה ג

וְאַף שֶׁלֹּא אָמְרוּ לְעֵיל אֶלָּא תַחַת הַזָּב, וַדַּאי דְּלֹא חֲמִירָא מִשְׁכָּב מִזָּב עַצְמוֹ, וּמִמֵּילָא נַשְׁמְעִינֵיהּ: וַאֲפִלּוּ לֹא הִפְסִיק בֵּינֵיהֶם אֶלָּא נְיָר. הָרַמְבַּ"ם: דְּאֵין לְךָ מִטַּמֵּא בְמַשָּׂא בִּנְבֵלָה אֶלָּא הָאָדָם: צָרִיךְ לוֹמַר וְנִשָּׂא. וְכֵן הוּא בָּרַ"שׁ. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: וּמִדְּלֹא תָנֵי גַּבֵּי מִשְׁכָּב חוּץ מִן הַמַּסִּיט, צָרִיךְ לוֹמַר דִּמְטַמֵּא בְלֹא מַסִּיט. וְטַעֲמָא נִרְאֶה לִי, מִשּׁוּם דְּאַחְמִיר רַחֲמָנָא בְּמִשְׁכָּב דַּאֲפִלּוּ הַנּוֹגֵעַ מְטַמֵּא בְגָדִים, וְגַבֵּי נְבֵלָה תְּנַן חֲשׂוּכֵי בְגָדִים בְּמַגָּע. הָרַ"שׁ: הַנּוֹשֵׂא. נַמִּי בְאָדָם. דּוּמְיָא דִנְבֵלָה: הַמַּאֲהִיל. בֵּין שֶׁהַמֵּת תַּחְתָּיו וְהָאָדָם נִשָּׂא בֵּין שֶׁהַמֵּת מַאֲהִיל עָלָיו וְהוּא נִשָּׂא, מְטַמֵּא טֻמְאַת אֹהֶל: לִרְבוּתָא כָּתַב כֵּן, דַּאֲפִלּוּ מִשּׁוּם מַשָּׂא שֶׁמַּכְבִּיד עָלָיו אֵינוֹ מְטַמֵּא, שֶׁאֵין בַּמֵּת מִשּׁוּם אֶבֶן מְסָמָא. וּמִכָּל מָקוֹם בְּאֵינָהּ רְצוּצָה מַיְרֵי. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: וְטַעֲמָא נִרְאֶה לִי, דְּטֻמְאַת מַשָּׂא דְּמֵת יָלְפִינַן בְּקַל וָחֹמֶר מִנְּבֵלָה, וְאָמְרִינַן דַּיּוֹ שֶׁלֹּא יְטַמֵּא מַשָּׂאוֹ אֶלָּא אִם כֵּן הִסִּיט, כְּמוֹ בִּנְבֵלָה. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב:

פרק ה - משנה ה

כִּדְיָלֵיף לְהוּ בְּסִפְרָא. הָרַמְבַּ"ם. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: הָרַ"שׁ. וּלְשִׁיטָתֵיהּ אָזֵיל כוּ', אֲבָל הָרַ"ב בְּחִנָּם נַקְטֵיהּ וְאֵין בּוֹ צֹרֶךְ לְפֵרוּשׁ הַמִּשְׁנָה. וְגַם בְּעִקַּר הַדִּין נִרְאֶה לִי דְּלָאו כְּהִלְכָתָא הוּא. וּרְאָיָה מִמִּשְׁנָה ה' ו' דְּפֶרֶק ד':

פרק ה - משנה ו

שְׁנַיִם. בְּמַגָּע וְלֹא בְמַשָּׂא: הַמַּסִּיט. כְּלוֹמַר, וְאֵינוֹ נוֹשֵׂא וְנִשָּׂא. וּבֵין שֶׁהַטָּהוֹר הִסִּיט לַטָּמֵא אוֹ אִפְּכָא. וְקָאֵי אַזָּב וְכוּ'. אֲבָל בִּמְצֹרָע וּמִשְׁכָּב וּמוֹשָׁב לֵיכָּא לְפָרוֹשֵׁי הָכִי, דְּהָא כָּל טֻמְאוֹת הַמַּסִּיט טְהוֹרוֹת חוּץ מִשֶּׁל זָב: בְּלֵידָה יָבֵשְׁתָּא: הַנּוֹשֵׂא כוּ'. כְּלוֹמַר וַאֲפִלּוּ אֵינוֹ מַסִּיט. מַה שֶּׁאֵין כֵּן נְבֵלָה וּמֵת דְּמִשְׁנָה ג' דְּבָעִינַן דַּוְקָא שֶׁיְּהֵא נִיסָט, עַיֵּן שָׁם:

פרק ה - משנה ז

הַמַּסִּיט. אֲבָל בְּמַשָּׂא בְּלֹא הֶסֵּט לֹא מְטַמֵּא, וְדָמוּ לִנְבֵלָה וּמֵת, דְּכֻלְּהוּ אִמְּעִיטוּ מִקְּרָא מֵאֶבֶן מְסָמָא. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: הָרַמְבַּ"ם. וּלְשִׁיטָתֵיהּ דִּסְבִירָא לֵיהּ דִּלְתַנָּא קַמָּא נַמִּי נוֹשֵׂא מְטַמֵּא בְּלֹא הֶסֵּט. אֲבָל הָרַ"שׁ מְפָרֵשׁ כָּאן כִּפְשׁוּטוֹ, דְּרַבִּי אֱלִיעֶזֶר פָּלֵיג וּסְבִירָא לֵיהּ דְּנוֹשֵׂא בְלֹא מַסִּיט נַמִּי מְטַמֵּא, דְּלֹא מְמַעֵט לְהוּ מִקְּרָא. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב:

פרק ה - משנה ח

וְהַמַּסִּיטוֹ. לַמֶּרְכָּב. אֲבָל מֶרְכָּב הַמַּסִּיט לַטָּהוֹר, לֹא, וּכְמוֹ בְמִשְׁכָּב:

פרק ה - משנה ט

טָהוֹר. הָאָדָם שֶׁהַנְּבֵלָה בְּפִיו וַעֲדַיִן לֹא בְלָעָהּ. וְקָא מַשְׁמַע לָן דְּלֹא תֵימָא דְּבִהְיוֹתָהּ נִסְתֶּרֶת וְנִכְנֶסֶת לַאֲוִיר הַתַּנּוּר תִּהְיֶה כְמוֹ בְלוּעָה, וְטָהוֹר הַתַּנּוּר, מַיְרֵי כְשֶׁהִיא בְלוּעָה. הָרַמְבַּ"ם: כְּלוֹמַר, דְּכָל אֵלּוּ דְַּאֲמַרַן דִּמְטַמְּאִין שְׁנַיִם מִשּׁוּם בְּגָדִים, כָּל אֵלּוּ אֵין כוּ':

פרק ה - משנה י

שֶׁאֵין כוּ'. דְּאִלּוּ יֵשׁ בָּהֶם, אַף הַנּוֹגֵעַ בָּהֶם מְטַמֵּא בְגָדִים, לְפִי שֶׁאִי אֶפְשָׁר שֶׁיִּגַּע בַּמַּיִם שֶׁלֹּא יַסִּיט אוֹתָם, וַהֲרֵי הוּא מְטַמֵּא שְׁנַיִם וּפוֹסֵל אֶחָד. אֲבָל אִם קָרְשׁוּ וְנַעֲשׂוּ כִגְלִיד דְּאֶפְשָׁר לִגַּע בְּלֹא הֶסֵּט, תָּנֵי בְפֶרֶק קַמָּא דְכֵלִים מִשְׁנָה ב' חֲשׂוּכֵי בְגָדִים בְּמַגָּע. וּכְמוֹ שֶׁכָּתַב הָרַ"ב בְּמִשְׁנָה ח'. וְהַתַּנָּא הָתָם וְהָכָא מִלְּתָא דִפְסִיקָא נָקַט: וְאֵין הַלָּשׁוֹן מַשְׁמַע כֵּן, דְּהָכִי הֲוָה לֵיהּ לְמִתְנֵי, בַּמֶּה דְבָרִים אֲמוּרִים בְּאָדָם, אֲבָל אֳכָלִים וְכֵלִים הַנּוֹגְעִים בְּאֶחָד כוּ'. וְהָרַמְבַּ"ם פֵּרֵשׁ, כָּל הַנּוֹגֵעַ בִּשְׁאָר אֲבוֹת הַטֻּמְאוֹת חוּץ מֵאֵלּוּ שֶׁמָּנָה עַד הֵנָּה . וְאָמְרוֹ חוּץ מִן הָאָדָם, רָצָה לוֹמַר הַמֵּת, שֶׁגַּם הוּא לֹא נִמְנָה כָאן, וְנוֹגֵעַ מְטַמֵּא שְׁנַיִם כוּ'. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב:

פרק ה - משנה יא

וּמִשּׁוּם דְּנוֹגֵעַ לֹא כְתִיב בְּהֶדְיָא תָּלֵי לֵיהּ בְּשֶׁרֶץ. אֲבָל בַּגְּמָרָא מַסִּיק דְּמַה שֶּׁרֶץ מְטַמֵּא בְאֹנֶס , אַף בַּעַל קֶרִי: אֲבָל בַּגְּמָרָא, דְּזֶהוּ מִלְּתָא דִפְשִׁיטָא הוּא, דִּקְרָא כְתִיב. וּלְמַחֲנוֹתָם אִצְטְרִיךְ, סָלְקָא דַעְתָּךְ אֲמֵינָא וּתְהִי נִדָּתָהּ עָלָיו כְּתִיב, דְּיִשְׁתַּלַּח חוּץ לִשְׁתֵּי מַחֲנוֹת כְּנִדָּה, קָא מַשְׁמַע לָן. וּבְבַעַל קֶרִי כְּמַגַּע שֶׁרֶץ לֵיכָּא לְמֵימַר הָכִי, דְּבַעַל קֶרִי חוּץ לִשְׁתֵּי מַחֲנוֹת, וְשֶׁרֶץ שָׁרֵי בְּמַחֲנֵה לְוִיָּה. אִי נַמִּי, הָכָא סֵיפָא אִצְטְרִיךְ לֵיהּ אֶלָּא כוּ': שֶׁחָמוּר כוּ'. וְקָשֶׁה, הָא חָמוּר נַמִּי טְמֵא מֵת מִבּוֹעֵל נִדָּה, שֶׁטָּעוּן הַזָּאָה שְׁלִישִׁי וּשְׁבִיעִי וְחֶרֶב הֲרֵי הוּא כְּחָלָל. וְשֶׁמָּא בַסֵּיפָא דִבְרַיְתָא חֲשִׁיב לְהוּ. תּוֹסָפוֹת. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב וְתוֹסָפוֹת חֲדָשִׁים:

פרק ה - משנה יב

דְּמִסְתַּבְּרָא דְּהַבָּא כוּ' נַמִּי בִשְׁלֹשָׁה לֻגִּין הוּא, כֵּיוָן דְּהָא בְהָא תַּלְיָא. וְכֵן כָּתַב הָרַמְבַּ"ם: דִּכְתִיב וּבָא הַשֶּׁמֶשׁ וְטָהֵר. גְּמָרָא:
