פרק ט

פרק ט - משנה א

בְּמֵת אֲשֶׁר הוּא מֵאֲבוֹת הַטֻּמְאוֹת כְּדִתְנַן בְּפֶרֶק קַמָּא דְאֳהָלוֹת. וְהוּא יְטַמֵּא כְלֵי חֶרֶס. הָרַמְבַּ"ם: כְּלוֹמַר שֶׁהַצָּמִיד פָּתִיל שֶׁעָלָיו אֵינוֹ כָרָאוּי וְנִטְמָא הַתַּנּוּר. הָרַמְבַּ"ם: כְּלוֹמַר שֶׁהָיָה הַצָּמִיד פָּתִיל כָּרָאוּי: כְּלוֹמַר דְּאִי לָאו הָכִי הָא קַיְמָא לָן דְּאֵין צָמִיד פָּתִיל מַצִּיל עַל כְּלֵי מַתָּכוֹת: מִצִּדֶּיהָ. כְּלוֹמַר שֶׁלֹּא עַל פִּיהָ שֶׁל הֶחָבִית: כְּלוֹמַר לִמְגוּפָה. הָרַאֲבַ"ד: פִּיהָ. הָכָא נַמִּי טַעֲמָא מִשּׁוּם דִּבְטִילֵי לְגַבֵּי מְגוּפָה כוּ', וּמְגוּפָה אֵינָהּ כְּחָבִית וְאֵינָהּ כְּלִי טָמֵא, אֲבָל הִיא נֶחְשֶׁבֶת לְגַבֵּי הֶחָבִית כְּמוֹ הַטְּפֵלָה לְגַבֵּי הַתַּנּוּר: כִּקְלִפַּת הַשׁוּם. הָכִי נַמִּי הֲוָה מָצֵי לְמִתְנֵי הָכִי בָרֵישָׁא גַבֵּי נְחֻשְׁתּוֹ שֶׁל תַּנּוּר, אֶלָּא דְהָתָם לֹא שְׁכִיחַ שֶׁיִּהְיֶה עָלָיו כִּקְלִפַּת הַשּׁוּם, מַה שֶּׁאֵין כֵּן הָכָא בִמְגוּפָה שְׁכִיחָא מִלְּתָא שֶׁהַמַּחַט אוֹ הַטַּבַּעַת נִגְבְּלוּ בְתוֹךְ הַטִּיט וְתַחְתֵּיהֶם כִּקְלִפַּת הַשּׁוּם. תּוֹסְפוֹת יוֹם טוֹב:

פרק ט - משנה ב

וּמַיְרֵי שֶׁהַמְּנִיקָה הִיא כֻלָּהּ תַּחַת צָמִיד פָּתִיל. עַיֵּן תּוֹסְפוֹת יוֹם טוֹב: כְּלוֹמַר בֵּין בְּשֶׁל חָבֵר בֵּין בְּשֶׁל עַם הָאָרֶץ. וּבְמִשְׁנָה ג' פֶּרֶק ה' דְּאֳהָלוֹת מַשְׁמַע דִּבְשֶׁל עַם הָאָרֶץ לְחוּד גָּזְרוּ וְלֹא בְשֶׁל חָבֵר. וּלְאוֹתוֹ פֵּרוּשׁ, הָא דִתְנַן לְעֵיל אִם בְּטָמֵא טְמֵאִין, הַיְנוּ שֶׁהוּא שֶׁל עַם הָאָרֶץ שֶׁהוּא טָמֵא. וְאִם בְּטָהוֹר, שֶׁאֵינוֹ שֶׁל עַם הָאָרֶץ. תּוֹסְפוֹת יוֹם טוֹב:

פרק ט - משנה ג

לְמַטָּה כוּ'. דְּאִי בִנְחֻשְׁתּוֹ כְדִתְנַן רֵישׁ פִּרְקִין, צָרִיךְ לְשַׁעֵר בְּבָצֵק כְּדִתְנַן הָתָם: מַחַט כוּ'. הַטָּמֵא בְמֵת. הָרַמְבַּ"ם. וּבְחִבּוּרוֹ כָתַב שֶׁכָּל הַכֵּלִים הַנִּמְצָאִים טְמֵאִים:

פרק ט - משנה ה

אִיכָּא לְמֵידַק, אַמַּאי דִיֵּק לְפָרוֹשֵׁי בְמַשְׁקִין חֲמוּרִין שֶׁהֵן אַב הַטֻּמְאָה, וַהֲרֵי כָל הַמַּשְׁקִים מְטַמְּאִים כֵּלִים בִּגְזֵרַת שְׁמֹנָה עָשָׂר דָּבָר כוּ'. וְהוּא פְלֻגְתָּא בַגְּמָרָא מַסֶּכֶת נִדָּה דַּף ס"ב. וּפָסַק הָרַמְבַּ"ם דְּמַתְנִיתִין אֶחָד קַלִּין וְאֶחָד חֲמוּרִין, אֶלָּא דְבַחֲמוּרִין דַּוְקָא כְשֶׁאֵינוֹ מַקְפִּיד עֲלֵיהֶן שֶׁיֵּצְאוּ הוּא דְבָעִינַן הֻסַּק הַתַּנּוּר, אֲבָל הִקְפִּיד עָלָיו וְיָכוֹל לָצֵאת הֲרֵי זֶה מְטַמֵּא הַתַּנּוּר אַף עַל פִּי שֶׁלֹּא הֻסַּק כוּ'. תּוֹסְפוֹת יוֹם טוֹב: חֲדָשָׁה. הַבָּאָה מִמַּשְׁקִים טְמֵאִים, שֶׁהֻסַּק בָּהּ הַתַּנּוּר, נִטְמָא, שֶׁסּוֹף מַשְׁקֶה לָצֵאת. הָרַמְבַּ"ם: שָׁלֹשׁ שָׁנִים. רְבוּתָא דְשָׁלֹשׁ שָׁנִים לַאֲפוֹקֵי מֵהַאי תַּנָּא דִמְטַהֵר לְאַחַר שָׁלֹשׁ שָׁנִים אַף עַל פִּי שֶׁלֹּא נִגְבָהּ, דְּלָא חֲשִׁיב שׁוּב מַשְׁקֶה כְלָל, וְקָא מַשְׁמַע לָן תַּנָּא דִידָן דְּכָל שֶׁלֹּא נִגְבָהּ אֲפִלּוּ לְאַחַר שָׁלֹשׁ שָׁנִים חָשׁוּב מַשְׁקֶה. תּוֹסְפוֹת יוֹם טוֹב: נִטְמָא. הַתַּנּוּר כְּשֶׁיֻּסַּק. הָרַמְבַּ"ם:

פרק ט - משנה ו

שֶׁמִּתְּחִלָּתָן כוּ'. שֶׁאִם נַעֲשׂוּ מִתְּחִלָּה בְטֻמְאָה וְיָצְאוּ מֵהֶן מַשְׁקִין, טְמֵאִין. וְטַעֲמָא, שֶׁכֵּיוָן שֶׁהָיוּ טְמֵאִין לֹא פָרְחָה טֻמְאָתָן, אֲבָל כְּשֶׁמִּתְּחִלָּה נַעֲשׂוּ בְטָהֳרָה וְלֹא הָיָה עֲלֵיהֶן שֵׁם טֻמְאָה, אֵין הִלּוּךְ הַטְּמֵאִים שֶׁעֲלֵיהֶם מְטַמֵּא לַמַּשְׁקֶה שֶׁיּוֹצֵא אַחַר כָּךְ, שֶׁאַף עַל פִּי שֶׁכְּבָר הֻכְשְׁרוּ בְמַשְׁקִין שֶׁיָּצְאוּ מֵהֶן כְּשֶׁנַּעֲשׂוּ בְטָהֳרָה, מִכָּל מָקוֹם הוֹאִיל וּבְטָהֳרָה נַעֲשׂוּ לֹא נִקְרָא עֲלֵיהֶן שֵׁם טֻמְאָה מֵעוֹלָם. תּוֹסְפוֹת יוֹם טוֹב, וְעַיֵּן עוֹד: אֲבָל לְפִי לְשׁוֹן הַמִּשְׁנָה מַשְׁמַע שֶׁהָיְתָה כְבָר לְבֵנָה קֹדֶם שֶׁנִּבְלְעָה. וּמַהֲרַ"ם כָּתַב, דֶּרֶךְ הַשָּׂרִים שֶׁקּוֹבְעִים טַבַּעַת בְּאַבְנֵי גָזִית כְּדֵי שֶׁיִּקְשְׁרוּ שָׁם סוּסֵיהֶם, וְכֵן עוֹשִׂין גַּם כֵּן תּוֹךְ בִּנְיָן שֶׁל לְבֵנִים: כָּךְ כָּתַב הָרַ"שׁ. וּלְשִׁיטָתוֹ דִסְבִירָא לֵיהּ דִּכְלִי מַתָּכוֹת נִצּוֹל בְּצָמִיד פָּתִיל כְּשֶׁהוּא שֶׁל חָבֵר, כְּמוֹ שֶׁכָּתַבְתִּי בְמִשְׁנָה ב', הֻצְרַךְ לְפָרֵשׁ כֵּן. אֲבָל לְשִׁיטַת הָרַמְבַּ"ם וּכְמוֹ שֶׁפֵּרֵשׁ הָרַ"ב שָׁם, דְּאַף בְּשֶׁל חָבֵר אָמְרוּ אֵין כְּלִי שֶׁטֶף נִצּוֹל בְּצָמִיד פָּתִיל, אֵין צֹרֶךְ לִסְבָרָא זוֹ לוֹמַר דְּדַוְקָא בִכְלִי שֶׁיֵּשׁ לוֹ תוֹךְ מַצִּיל. וְהָרַמְבַּ"ם כָּתַב בְּפֶרֶק כ' מֵהִלְכוֹת טֻמְאַת מֵת בְּטַעַם מִשְׁנָתֵנוּ שֶׁאֵין הַבְּלוּעִים בְּכֵלִים מַצִּילִים אֶלָּא בְצָמִיד פָּתִיל. עַד כָּאן. וּכְלוֹמַר דִּבְצָמִיד פָּתִיל יֵשׁ לוֹ הַצָּלָה עַל יְדֵי אַגַּב הֵיכָא דְבָטֵיל כְּמוֹ בִטְפֵלָה וּמְגוּפָה דְרֵישׁ פִּרְקִין. וְכֵן מוּכָח מֵהֶמְשֵׁךְ דְּבָרָיו שָׁם: טָהוֹר כוּ'. וְלֹא הֲוָה מָצֵי לְמִתְנֵי טָמֵא נִטְמְאוּ כְסִדְרָא דְרֵישָׁא מִשּׁוּם דְּאֵין כֵּלִים מִטַּמְּאִים מֵאֲוִיר כְּלִי חֶרֶס. מַהֲרַ"ם: וְאִם תֹּאמַר וְהָא לָאו פְּשׁוּטֵי כְלֵי עֵץ הוּא, דְּהָא כְלִי הַמְקַבֵּל טֻמְאָה חָשְׁבִינַן לֵיהּ כוּ'. אִיכָּא לְאוֹקְמֵי בִכְהַאי גַּוְנָא שֶׁאִי אֶפְשָׁר לְהוֹצִיא הַדָּרְבָן אֶלָּא אִם כֵּן יִשָּׁבֵר בֵּית קִבּוּלוֹ, וּלְכָךְ לָא חֲשִׁיב תּוּ בֵית קִבּוּל. תּוֹסָפוֹת:

פרק ט - משנה ז

כְּלוֹמַר מַה שֶּׁהוּא בֵין הַתַּנּוּר לַסְּרֵידָה. תּוֹסְפוֹת יוֹם טוֹב: עָגֹל. הַנֶּקֶב. וְקָשֶׁה מַאי אַשְׁמְעִינַן מִשְׁנָתֵנוּ אֵין רוֹאִין אוֹתוֹ אָרֹךְ, דְּמֵהֵיכִי תֵיתֵי לִרְאוֹתוֹ כְאִלּוּ הוּא אָרֹךְ, וַהֲרֵי פִי הַמַּרְדֵּעַ שֶׁבּוֹ מְשַׁעֲרִין עָגֹל הוּא כְמוֹ שֶׁכָּתַב הָרַ"ב שֶׁיֵּשׁ בְּהֶקֵפוֹ כוּ'. תּוֹסְפוֹת יוֹם טוֹב. וְלִי נִרְאֶה דְהָכִי קָאָמַר, שֶׁהַנֶּקֶב הוֹלֵךְ בָּעֳבִי הַסְּרֵידָה מִלְּמַעְלָה לְמַטָּה בְעַקְמִימוּת כְּמוֹ מַה שֶּׁאָנוּ קוֹרִין שווינד"ל שטע"ג, וּמִכֹּחַ זֶה אֵינוֹ נִכְנָס הַמַּרְדֵּעַ, אַף עַל פִּי שֶׁהַנֶּקֶב הוּא רָחָב בְּעַצְמוֹ אֵין רוֹאִין כוּ', פֵּרוּשׁ כְּאִלּוּ הָיָה בְשָׁוֶה מִלְּמַעְלָה לְמַטָּה. תּוֹסָפוֹת חֲדָשִׁים:

פרק ט - משנה ח

בַּחֲצָבִים גְּדוֹלִים. לִכְאוֹרָה תְמוּנָתָן מְשֻׁנָּה, שֶׁיֵּשׁ לָהֶן צְדָדִים בּוֹלְטִים. וְאִי נַמִּי יֵשׁ לְפָרֵשׁ דְּהָכִי קָאָמַר, בִּמְגוּפַת חֲצָבִים גְּדוֹלִים. תּוֹסְפוֹת יוֹם טוֹב: לְשִׁיטָתוֹ דְמִשְׁנָה ב' פֶּרֶק ג', דְּשֶׁל שֶׁמֶן דַּק מִשֶּׁל יַיִן. וְהוּא הַדִּין לִדְבַשׁ וְחָלָב. לְפִי שֶׁהַיַּיִן בְּמִקְצָת הָאֲרָצוֹת עָב יוֹתֵר. וְהַדְּבַשׁ רְצוֹנוֹ לוֹמַר מַשְׁקֶה שֶׁקּוֹרִין מע"ד. תּוֹסְפוֹת יוֹם טוֹב: כָּל שֶׁהוּא. בְּכוֹנֵס מַשְׁקֶה. הָרַמְבַּ"ם: וְיֵשׁ לִתֵּן טַעַם דִּכְתִיב וְכָל כְּלִי פָתוּחַ אֲשֶׁר אֵין צָמִיד פָּתִיל עָלָיו, וְדָרְשִׁינַן שֶׁטֻּמְאָתוֹ מַשְׁכֶּמֶת לְפִתְחוֹ, וְכִי נִקַּב כְּדֵי טָהֳרָתוֹ חֲשִׁיב כְּפֶתַח וּבָעֵי צָמִיד פָּתִיל. פָּחוּת מִכָּאן לֹא. הָרַ"שׁ: לְכָךְ כוּ'. וְשָׁנִינוּ שָׁלֹשׁ מִשְׁנָיוֹת בְּעִנְיָן אֶחָד, חֲדָא דִלְעֵיל בְּפֶרֶק ג', וְאִידָךְ בְּפֶרֶק ח', וְהַךְ דְּהָכָא. וְכֻלְּהוּ צְרִיכֵי, דְּקַמַּיְתָא מִתְנֵי לְעִנְיַן שִׁעוּר לִטַּהֵר, וְאִידָךְ לְהַצִּיל בְּאֹהֶל הַשֶּׁרֶץ, וְהַךְ דְּהָכָא לְהַצִּיל בְּאֹהֶל הַמֵּת. הָרַ"שׁ:
