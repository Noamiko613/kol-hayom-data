פרק יב

פרק יב - משנה א

דְּמִסְּבָרָא לֹא מְרַבִּינַן מִוְּכָל שֶׁכָּתַב בַּפֶּרֶק דִּלְעֵיל מִשְׁנָה ח' אֶלָּא טַבַּעַת אָדָם, מִשּׁוּם דְּתָפַסְתָּ מֻעָט תָּפַסְתָּ כוּ'. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב:

פרק יב - משנה ב

בַּעֲלֵי בָתִּים. תֵּימַהּ, מַאי שְׁנָא שֶׁל בַּעֲלֵי בָתִּים, שֶׁל סָרוֹקוֹת נַמִּי אֵינוֹ טָמֵא אֶלָּא כְשֶׁיֶּשׁ בּוֹ אֻנְקְיוֹת. וְנִרְאֶה לִי, דִּסְבִירָא לֵיהּ דְּסָרוֹקוֹת כֻּלָּן יֵשׁ לָהֶן אֻנְקְיוֹת, פֵּרוּשׁ כַּפּוֹת שֶׁל בֵּית קִבּוּל, אֲבָל שֶׁל בַּעֲלֵי בָתִּים שֶׁאֵינָן רְגִילִים לִשְׁקֹל כָּל כָּךְ, יֵשׁ שֶׁיֶּשׁ לָהֶם מֹאזְנַיִם שֶׁאֵין בֵּית קִבּוּל לְכַפּוֹת מֹאזְנַיִם כוּ', לְהָכִי נָקַט אִם יֵשׁ בָּהֶן אֻנְקְיוֹת טְמֵאָה. מַהֲרַ"ם: גִּרְסַת הָרַ"שׁ אֻנְקְיוֹת. וּפֵרֵשׁ, הֵם כַּף שֶׁל מֹאזְנַיִם וְיֵשׁ לָהֶן בֵּית קִבּוּל שֶׁעֲשׂוּיִין כְּמִין כּוֹס קָטָן: וְהָרַמְבַּ"ם בְּנֻסְחַת אֶרֶץ יִשְׂרָאֵל פֵּרֵשׁ אָקוֹן הוּא סַל הַצַּיָּדִין, וּתְמוּנָתָהּ מְפֻרְסֶמֶת אֶצְלָם, וְהוּא כְמוֹ תֵבָה מֵעֵץ בּוֹ אֻנְקְלָיוֹת, יִתְלוּ בוֹ כִּיס יִכָּנֵס הַדָּג בּוֹ וְלֹא יוּכַל לָצֵאת מִמֶּנּוּ:

פרק יב - משנה ג

וְהָרַמְבַּ"ם מְפָרֵשׁ, בַּרְזִלִּים קְבוּעִים שֶׁבַּמּוֹקֵד לִשְׁמֹר אוֹתוֹ מִן הַשְּׂרֵפָה בִּמְהִירוּת זְמָן. וְהַכֶּסֶף מִשְׁנֶה כָּתַב שֶׁנִּרְאֶה לוֹ שֶׁהֵם בַּרְזִלִּים הַתְּלוּיִים בַּמּוֹקֵד וּבְסוֹפָן מַזְלְגִים שֶׁהַקְּדֵרָה תְלוּיָה בָהֶם עַל הָאֵשׁ כְּדֵי שֶׁלֹּא תְהֵא יוֹשֶׁבֶת עַל הָאֵשׁ מַמָּשׁ וְתִשָּׂרֵף. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב כַּמָּה פֵּרוּשִׁים: לִתְלוֹת בּוֹ כֵלִים. הָרַ"שׁ:

פרק יב - משנה ד

וּלְשׁוֹן הָרַמְבַּ"ם, אֶבֶן הַשָּׁעוֹת, אֶבֶן תִּבָּנֶה בָּאָרֶץ וְיִרְשֹׁם בָּהּ קַוִּים יְשָׁרִים כְּתוּבִים עֲלֵיהֶם שְׁמוֹת הַשָּׁעוֹת, וְהִיא עֲגֻלָּה, וּבְמֶרְכַּז זֹאת הָעֲגֻלָּה מַסְמֵר נִצָּב עַל זָוִית נִצֶּבֶת, כָּל מַה שֶּׁיִּנָּטֶה בְּשִׁוּוּי צֵל זֶה הַמַּסְמֵר לְקַו מֵאֵלּוּ הַקַּוִּים, יֵדַע כַּמָּה שָׁעוֹת עָבְרוּ מֵהַיּוֹם. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: צָרִיךְ לוֹמַר וּמְחַבֵּר עָלָיו הַמַּטְוֶה. וּפֵרוּשׁוֹ עַל הַמַּסְמֵר. נִרְאֶה לִי: עוֹד יֵשׁ לוֹמַר, דִּבְמַסְמֵר שֶׁל אָרוֹן מַיְרֵי, שֶׁיֶּשׁ מַסְמֵר בָּרֵחַיִם שֶׁל גָּרוֹסוֹת, וּמְטַהֲרֵי רַבָּנָן מִשּׁוּם דְּאֵינוֹ עָשׂוּי אֶלָּא לְחִזּוּק. הָרַ"שׁ. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: מַתֶּכֶת כוּ'. וְאַף עַל גַּב דְּמַחְזֶקֶת הָרֵחַיִם אַרְבָּעִים סְאָה, טְמֵאָה, דְּבָתַר עֲגָלָה אָזְלִינַן שֶׁהִיא שֶׁל מַתָּכוֹת, וּכְלֵי מַתָּכוֹת לֹא אִתַּקְּשׁוּ לְשַׂק, וְלֹא בָעִינַן בְּהוּ מִטַּלְטֵל מָלֵא וְרֵיקָן. וַאֲפִלּוּ הָרֵחַיִם שֶׁל עֵץ מְקַבֶּלֶת טֻמְאָה, דְבָתַר עֲגָלָה גְרִירָא. מַהֲרַ"ם:

פרק יב - משנה ה

מַסְמֵר. מַסְמְרוֹת שֶׁמַּחֲזִיקִין בָּהֶם הַתִּקְרָה הָעֲשׂוּיִין לִתְקֹעַ אוֹתָן בָּעֵצִים, אֵינָן מְקַבְּלִים טֻמְאָה כוּ'. וְאִם הִתְקִינָן לִהְיוֹת כְּלִי בִּפְנֵי עַצְמוֹ, מְקַבֵּל טֻמְאָה, כֵּיצַד, מַסְמֵר כוּ'. הָרַמְבַּ"ם: פּוֹתֵחַ כוּ'. לְנַקֵּב אֶת הֶחָבִית לְהוֹצִיא מִמֶּנָּה יַיִן. הָרַ"שׁ: לְשׁוֹן הָרַאֲבַ"ד, לְפִי שֶׁהַמַּסְמֵר שֶׁנּוֹקְבִין בּוֹ אֶת הֶחָבִיּוֹת צָרִיךְ שֶׁיִּהְיֶה חָזָק הַרְבֵּה מִשְּׁאָר מַסְמְרוֹת, הִלְכָּךְ בַּהַתְקָנָה לְבַדָּהּ אֵינוֹ מְטַמֵּא עַד שֶׁיִּצְרְפֶנּוּ בָאֵשׁ וּבַמַּיִם. וְהָרַמְבַּ"ם פֵּרֵשׁ עַד שֶׁיַּכֵּהוּ בְּפַטִּישׁ וִיחַדֵּד אוֹתוֹ וִישִׁיבֵהוּ בְדִמְיוֹן הַמַּקְדֵּחַ. וְצוֹרֵף, שֵׁם לְאֻמַּן הַמַּתָּכוֹת:

פרק יב - משנה ו

וּתְלוֹי כוּ'. מַסְמְרִים שֶׁתּוֹלִים בָּהֶם הַמַּגְרֵדוֹת. וּלְפִי שֶׁהֵם מְשֻׁנִּים בְּצוּרָתָן מִשְּׁאָר מַסְמְרוֹת הַתְּקוּעִין, לְפִיכָךְ מְטַמֵּא רַבָּן גַּמְלִיאֵל. הָרַמְבַּ"ם: וְגֹלְמֵי כוּ'. שֶׁמָּא טַעֲמָא דְרַבָּן גַּמְלִיאֵל, כֵּיוָן דַּחֲזִי לְתַשְׁמִישׁ, טָמֵא, כִּי הֵיכָא דִפְשׁוּטִים טְמֵאִים הוּא הַדִּין גָּלְמֵי כְּלֵי מַתָּכוֹת. מַהֲרַ"ם: טָמֵא. וְאַף עַל פִּי שֶׁאֵינוֹ מְגֻפָּף לְבִזְבֵּזִים מֵאַרְבָּעָה רוּחוֹתָיו. דִּבְטַבְלָא לֵיכָּא קְפֵידָא כֻלֵּי הַאי שֶׁיְּהֵא מְגֻפָּף מִכָּל הָאַרְבַּע רוּחוֹת דּוּמְיָא דְכִירָה, דְּהָכִי נַמִּי אֵין עִקַּר תַּשְׁמִישָׁהּ לְבֵית קִבּוּל וְסַגִּי לֵיהּ בְּלִזְבֵּז שֶׁבְּשָׁלֹשׁ רוּחוֹת כְּדֵי לְהַחְזִיק מַה שֶׁמַּנִּיחִין עָלֶיהָ. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב:

פרק יב - משנה ז

וְהִתְקִינָהּ. אִם צְרִיכָה מַעֲשֶׂה כְּמוֹ חִתּוּךְ, אָתֵי שַׁפִּיר דְּנָקַט הִתְקִינָהּ, דְּמִידִי דִמְחֻסָּר מְלָאכָה אֵין טָמֵא בְמַחֲשָׁבָה. (כְּדִתְנַן בְּפֶרֶק כו מִשְׁנָה ד). וְאִם אֵינָהּ צְרִיכָה חִתּוּךְ, הָוֵי הִתְקִינָהּ לָאו דַּוְקָא. דְּכָל הַכֵּלִים יוֹרְדִין לִידֵי טֻמְאָתָן בְּמַחֲשָׁבָה. תּוֹסָפוֹת: וְהָא דִתְנַן שֶׁהוּא רַשַּׁאי לְקַיְּמָהּ עַד שְׁנֵי דִינָרִים, מוֹקֵי בַגְּמָרָא בִדְנָפְלָה לְנוּרָא וּפָחֲתָה לַהּ בְּחַד זִמְנָא:

פרק יב - משנה ח

וּבְנֻסְחַת אֶרֶץ יִשְׂרָאֵל כָּתַב, כּוֹתְשִׁים בּוֹ הַזֵּיתִים בְּמַעֲצָר. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: עַיֵּן תּוֹסְפוֹת יוֹם טוֹב פֵּרוּשִׁים שׁוֹנִים. וּלְעֵיל סוֹף פֶּרֶק ז':
