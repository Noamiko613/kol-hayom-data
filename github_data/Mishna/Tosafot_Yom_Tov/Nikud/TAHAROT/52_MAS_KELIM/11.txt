פרק יא

פרק יא - משנה א

וְלֹא סַגִּי לְהוּ דְלִגְזוֹר שֶׁשְּׁבִירָה צְרִיכָה הַעֲרֵב שֶׁמֶשׁ. דִּבְטֻמְאַת הַנֶּפֶשׁ בָּעֵי הַזָּאָה שְׁלִישִׁי וּשְׁבִיעִי, וּלְהָכִי גָזְרוּ בִסְתָם וְאָמְרוּ חָזְרוּ לְטֻמְאָתָן הַיְשָׁנָה. וּבַגְּמָרָא קָאָמַר, מִשּׁוּם גֶּדֶר מֵי חַטָּאת נָגְעוּ בָהּ. עַיֵּן תּוֹסְפוֹת יוֹם טוֹב:

פרק יא - משנה ב

וְהַצִּיר. הוּא צִיר הַשַּׁעַר אֲשֶׁר יְסוֹבֵב בְּעֶצֶם הַפּוֹתָה. וְהוּא יִהְיֶה מִבַּרְזֶל בְּשַׁעֲרֵי הַמְּדִינוֹת וְהַשְּׁעָרִים הַגְּדוֹלִים מְאֹד. כְּמוֹ הַדֶּלֶת תִּסֹּב עַל צִירָהּ (מִשְׁלֵי כו): וּבְפֵרוּשׁ הָרַמְבַּ"ם בְּנֻסְחַת אֶרֶץ יִשְׂרָאֵל כָּתוּב, הוֹאִיל וְעִקַּר עֲשִׂיָּתוֹ לִהְיוֹת דָּבֵק בָּאָרֶץ הִנֵּה הוּא לֹא יְקַבֵּל טֻמְאָה וַאֲפִלּוּ קֹדֶם חִבּוּרוֹ:

פרק יא - משנה ג

וְהוּא מִלְּשׁוֹן בַּרְזֶל עָשׁוֹת (יְחֶזְקֵאל כז). הָרַמְבַּ"ם: הַצִּפּוּיִין. הֵן הַחִפּוּיִין שֶׁיְּחַפּוּ הַכֵּלִים בְּלוּחוֹת זָהָב וְדוֹמֵיהֶן. הָרַמְבַּ"ם: וּבֶעָרוּךְ פֵּרֵשׁ כְּמוֹ אָזְנַיִם יֵשׁ לַכְּלִי, אֶלָּא שֶׁהָאֹזֶן יוֹצֵא מִגּוּף הַכְּלִי, וְאֹגֶן הוּא דָבָר שֶׁאֵינוֹ מִן הַכְּלִי אֶלָּא מְסֻבָּב עָלָיו וְיוֹצֵא. עַיֵּן תּוֹסְפוֹת יוֹם טוֹב: לְשַׁפְשֵׁף, בְּדָבָר הַמַּחְלִיקָן. לְשַׁבֵּץ, לְשׁוֹן מִשְׁבְּצוֹת, מְרַמְּצִין, עֲשׂוּיִין כְּמַזְלְגִים קְטַנִּים, כְּמוֹ וּמַפְּקִי לֵיהּ בְּרַמְצָא דְפַרְזְלָא בְמַסֶּכֶת נִדָּה, כְּלוֹמַר שֶׁמְּחֻסָּרִים לִתְקֹעַ בָּהֶן מַסְמְרוֹת וּמִשְׁבְּצוֹת לְנוֹי. רַשִׁ"י: כְּמוֹ שֶׁיְּגָרְדוּ כְלֵי מַתָּכוֹת אַחַר הַשִּׁיפָה. לְכַרְכֵּב, שֶׁיְּחַסֵּר הַלימא"ר כְּמוֹ שֶׁיַּעֲשֶׂה בַּבַּרְזֶל אַחַר הַגְּרִידָה. הָרַמְבַּ"ם. וְרַשִׁ"י פֵּרֵשׁ לְכַרְכֵּב, שֶׁחוֹקֵק בָּהֶן חֲקִיקוֹת גְּדוֹלוֹת וְכַרְכֹּב: נִרְאֶה לִי דְטַעֲמֵיהּ, דִּסְבִירָא לֵיהּ כְּאַבַּיֵי דְּאָמַר בְּמַסֶּכֶת שַׁבָּת דְּטַעֲמָא דְגָזְרוּ שֶׁיַּחְזְרוּ לְטֻמְאָתָן הַיְשָׁנָה דְחָשׁוּ שֶׁמָּא לֹא יִקְבֶנּוּ בִּכְדֵי טָהֳרָתוֹ. וְהַשְׁתָּא שֶׁקָּצַץ הַכְּלִי לַחֲתִיכוֹת לֵיכָּא לְמֵיחַשׁ לְהָכִי. תּוֹסְפוֹת יוֹם טוֹב. וְעַיֵּן עוֹד:

פרק יא - משנה ד

וְלָזֶה נִקְרָא חֲלָמָא. כְּמוֹ אִם יֶשׁ טַעַם בְּרִיר חַלָּמוּת (אִיּוֹב ו). הָרַמְבַּ"ם: וְצִפּוּי הַיְנוּ שֶׁהוּא כְמוֹ שֶׁמַּזְהִיבִין כְּלֵי כֶסֶף לְנוֹי: מִצַּד אֶחָד וְנִנְעַל וְנִסְגַּר. וּמִצַּד אַחֵר יֵשׁ בַּפּוּרְנָא נֶקֶב שֶׁמַּכְנִיסִים בּוֹ אֶת הַמַּפְתֵּחַ וּמַדְחִיק אֶת הַפִּין לְחֹרוֹ וְנִפְתַּח וְיוֹצֵא הַפִּין מִן הַפּוּרְנָא:

פרק יא - משנה ו

וַאֲפִלּוּ יֹאמַר אָדָם הַפִּיקָה, לֹא יֵדַע לְאֵיזֶה דָבָר כִּוֵּן, שֶׁזֶּה הַשֵּׁם נוֹפֵל עַל כָּל פִּלְכָּה עֲגֻלָּה אוֹ קְרוֹבַת הָעִגּוּל, עַד שֶׁיֹּאמַר פִּיקָה שֶׁל כּוּשׁ כוּ'. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: וְהָאִימָּה. מְפֹרָשׁ בְּפֵרוּשׁ הָרַ"ב פֶּרֶק כ"א. וּמַהֲרַ"ם כָּתַב בְּכָאן, הָאִימָּה, צִנּוֹרָא, כְּלִי תַשְׁמִישׁ לְמֶשִׁי:

פרק יא - משנה ז

עַיֵּן מַה שֶּׁכָּתַבְתִּי בְמַסֶּכֶת קִנִּים דִּקְרָנוֹת נִקְרְאוּ חֲצוֹצְרוֹת, וּלְשׁוֹן הָרַ"שׁ, כְּלֵי שִׁיר עֲשׂוּיִין מִקֶּרֶן, כְּדִכְתִיב קָל קַרְנָא מַשְׁרוֹקִיתָא (דָּנִיֵּאל ג). תּוֹסְפוֹת יוֹם טוֹב: כְּלוֹמַר מָקוֹם הַצַּר. שֶׁהַרְבֵּה פְעָמִים נִמְצָא בִּלְשׁוֹנָם שֶׁמְּשַׁמְּשִׁים בְּלָשׁוֹן קָצָר, וְהַמְּכֻוָּן בּוֹ צַר. וּלְפִי שֶׁאֵינוֹ רָחָב קְרָאוּהוּ קָצָר. תּוֹסְפוֹת יוֹם טוֹב: הַקַּב. וְהָרַ"ב הֶעְתִּיק הַקַּו כוּ'. אֲבָל הָרַ"שׁ הֶעְתִּיק כְּגִרְסַת הַסֵּפֶר. וּפֵרֵשׁ בְּשֵׁם הֶעָרוּךְ שֶׁהוּא נַרְתֵּק שֶׁלָּהּ. וּכְלִי הוּא. כְּמוֹ קַב הַקִּטֵּעַ בְּפֶרֶק בַּמֶּה אִשָּׁה מִשְׁנָה ח', שֶׁהוּא כְלִי וְנַרְתֵּק לָרֶגֶל: מְטַהֲרִין. נִרְאֶה לִי דְמִדִּתְנַן וּבִשְׁעַת חִבּוּרָן הַכֹּל טָמֵא, דְּהָא דִמְטַהֲרִים הַחֲכָמִים הַיְנוּ טַעֲמָא דְאֵין לַקַּו שֵׁם בִּפְנֵי עַצְמוֹ. וּפְלוּגְתַּיְהוּ כִּפְלוּגְתַּת רַבִּי עֲקִיבָא וַחֲכָמִים דִּלְעֵיל. תּוֹסְפוֹת יוֹם טוֹב:

פרק יא - משנה ח

הַכִּידוֹן. כְּלוֹמַר כְּגוֹן הַכִּידוֹן כוּ': כְּמוֹ נְטֵה בַּכִּידוֹן אֲשֶׁר בְּיָדְךָ (יְהוֹשֻׁעַ ח): וְנָשִׁים לָאו דַּוְקָא, שֶׁהֲרֵי מוֹנֶה אַף שֶׁל אֲנָשִׁים. אֶלָּא נָקַט דִּשְׁכִיחֵי טְפֵי: עִיר כוּ'. הוּבָא מֵהֶן קְצָתָן עַל צַד הַמָּשָׁל. כְּמוֹ שֶׁעָשָׂה בִכְלֵי הַמִּלְחָמָה. הָרַמְבַּ"ם: וּמִיהוּ הָכָא בְנִזְמֵי הָאֹזֶן, דְּהָא נִזְמֵי הָאַף תָּנֵי לְהוּ בַסֵּיפָא: וְטַבַּעַת כוּ'. כְּלוֹמַר וְטַבַּעַת שֶׁאָמַרְנוּ, בֵּין שֶׁיֵּשׁ עָלֶיהָ כוּ'. רַשִׁ"י: חוֹתָם. בַּגְּמָרָא פֶּרֶק ו' דְּשַׁבָּת שָׁם רָמִינַן מִמַּתְנִיתִין דְּהָתָם דְּלֹא תֵצֵא בְטַבַּעַת שֶׁיֵּשׁ עָלֶיהָ חוֹתָם וְאִם יָצָאתָה חַיֶּבֶת חַטָּאת, שְׁמַע מִנַּהּ דְּלָאו תַּכְשִׁיט הוּא לַהּ, וְאִלּוּ הָכָא קָתָנֵי לַהּ בַּהֲדֵי תַכְשִׁיטֵי נָשִׁים. וּמְשָׁנֵי רָבָא, דְּהָא דְקָתָנֵי הָכָא וְטַבַּעַת בֵּין שֶׁיֵּשׁ כוּ', לָאו אָרֵישָׁא דְאַיְרֵי בְתַכְשִׁיטֵי נָשִׁים קַיְּמֵי תַרְוַיְהוּ, אֶלָּא לִצְדָדִין קָתָנֵי, יֵשׁ עָלֶיהָ חוֹתָם תַּכְשִׁיטִין דְּאִישׁ, אֵין עָלֶיהָ חוֹתָם תַּכְשִׁיטִין דְּאִשָּׁה: שֶׁחֻלְיוֹת. לְשׁוֹן וַחֲלִי כָתֶם (מִשְׁלֵי כה). מַהֲרַ"ם: קְטַנָּה. נִרְאֶה לִי דְמַיְרֵי בְקַטְלָא שֶׁל פְּרָקִים שֶׁאֵין מַכְנִיסִין אוֹתָן בַּחוּט, וּמְשַׁעֵר תַּנָּא קַמָּא כִמְלֹא צַוָּאר קְטַנָּה. וְרַבִּי אֱלִיעֶזֶר קָאָמַר אֲפִלּוּ טַבַּעַת אַחַת דֶּרֶךְ לִתְלוֹת בַּצַּוָּאר. מַהֲרַ"ם. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב:

פרק יא - משנה ט

דְּאִי לֹא תֵימָא הָכִי אַף עַל גַּב דִּכְלֵי מַתָּכוֹת פְּשׁוּטֵיהֶן טְמֵאִין כְּדִתְנַן בְּרֵישׁ פִּרְקִין, זוֹ לָאו כְּלִי הִיא:
