פרק ג

דִּלְהָכִי שִׁנָּה מֵעִנְיָנָא דְרֵישָׁא. דְּאֵין דָּבָר אַחֵר מַאֲהִיל עָלָיו וְעַל כַּחֲצִי זַיִת אֶלָּא אִם כֵּן אֵינָהּ רְצוּצָה, וּמִכְּלַל דְּרֵישָׁא דְּלֹא קָתָנֵי הָכִי, דִּרְצוּצָה הִיא. מַהֲרַ"ם. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: מַאֲהִיל כוּ'. יֵשׁ לִתְמֹהַּ דְּלֹא תְנַן נַמִּי אוֹ כִשְׁנֵי חֲצָיֵי זֵיתִים מַאֲהִילִים עָלָיו כוּ'. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: אוֹ כוּ' טָהוֹר. תֵּימַהּ, הָא חַד שְׁמָא דְאֹהֶל הוּא. וּמַהֲרַ"ם פֵּרֵשׁ, דְּהַאי מַאֲהִיל עַל כַּחֲצִי זַיִת נַמִּי בִרְצוּצָה אַיְרֵי, וְהָוֵי נְגִיעָה וְאֹהֶל. וְזֶה דֹּחַק. וְנִרְאֶה לִי דְּהָרַ"ב לֹא גָרַס לֵיהּ וְכֵן בִּדְפוּס הַגְּמָרָא דְּעֵדֻיּוֹת לֹא גָרַס לֵיהּ. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: וְיִתָּכֵן דְּטַעֲמוֹ דַּחֲזִינַן לְאֹהֶל כְּמַאן דְּמָלֵי טֻמְאָה, מֵאַחַר שֶׁגְּזֵרַת הַכָּתוּב הוּא שֶׁכָּל אֲשֶׁר בָּאֹהֶל טָמֵא: הַכֹּל כוּ'. מַסְקָנָא דְּרַבָּנָן הִיא. מַהֲרַ"ם: וּלְתַנָּא קַמָּא הוּא שֶׁפֵּרֵשׁ כֵּן. דִּלְרַבִּי מֵאִיר אִיתָא נַמִּי מַגָּע וְאֹהֶל, דְּחַד שְׁמָא נִינְהוּ:

וְלִרְבוּתָא דְרַבָּנָן נָקַט כֵּן, דְּאַף שֶׁנִּתְפַּזֵּר וְגַג שֶׁכְּנֶגֶד זֶה מַאֲהִיל עַל חֲצִי שִׁעוּר וְכֵן כְּנֶגֶד זֶה וּשְׁנֵי אֳהָלוֹת הֵן, אֲפִלּוּ הָכִי מִצְטָרְפִין. רַשִׁ"י: לִישְׁנָא דִגְמָרָא. וּפֵרֵשׁ רַשִׁ"י, כְּלוֹמַר, הַמְגַלְגֵּל זֶה לָזֶה וְנִתְעָרֵב זֶה עִם זֶה: טָהוֹר. כְּלוֹמַר, הַכֵּלִים הַבָּאִים אַחַר הַבְּלִיעָה לְתוֹךְ הַבַּיִת, אֵינָם מִטַּמְּאִין: תּוֹסֶפְתָּא. וּבְמַיִם רִאשׁוֹנִים בִּלְבַד לֹא יוּכַל לְשַׁעֵר, שֶׁאֲפִלּוּ אִם לֹא יִמְצְאוּ רְבִיעִית יוֹתֵר, יֵשׁ לוֹמַר שֶׁיָּצָא רְבִיעִית דָּם וְנִשְׁאַר בַּבֶּגֶד מִן הַמַּיִם. אַךְ קָשֶׁה, דִּבְלֹא מְדִידַת מַיִם רִאשׁוֹנִים נַמִּי יָכוֹל לֵידַע, שֶׁיִּסְחֹט הַמַּיִם מִן הַכְּסוּת וְיִמְדֹּד אוֹתוֹ, וְיָבִיא מַיִם אֲחֵרִים עִם רְבִיעִית דָּם שֶׁיִּהְיֶה בֵּין הַכֹּל כְּשִׁעוּר הַמַּיִם שֶׁיָּצָא מִן הַכְּסוּת, וְאִם מַרְאֵיהֶן שָׁוֶה טְמֵאָה כוּ'. וְיֵשׁ לוֹמַר דְּאִין הָכִי נַמִּי, וּמַה שֶּׁהֻצְרְכוּ לְמִדָּה הַיְנוּ שֶׁאִם יִמְצְאוּ רְבִיעִית יוֹתֵר שֶׁלֹּא יִצְטָרְכוּ לְמַיִם אֲחֵרִים כוּ'. אֲבָל אִם לֹא מָצָא רְבִיעִית אֲזַי מֵבִיא אֲחֵרִים כְּמִדָּתָן, רוֹצֶה לוֹמַר כַּמִּדָּה שֶׁמָּצָא אַחַר הַכְּבִיסָה, אֶלָּא שֶׁנּוֹתֵן בָּהֶם רְבִיעִית דָּם וְעִם הָרְבִיעִית דָּם הוּא שֶׁיִּהְיֶה כְּמִדַּת הַמַּיִם מִן הַכְּבִיסָה. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: טְהוֹרָה. אֵינָהּ מְטַמְּאָה בְּאֹהֶל, וַהֲרֵי הִיא כִּכְסוּת שֶׁנָּגְעָה בַמֵּת. הָרַמְבַּ"ם. וְכֵן פֵּרְשׁוּ רַשִׁ"י וְתוֹסָפוֹת. וּמַהֲרַ"ז וָורְמַיְזָ"א הוֹכִיחַ דִּלְדַעַת הָרַ"ב דְּכֵלִים שֶׁנִּטְמְאוּ בְמֵת מְטַמְּאִין בְּאֹהֶל, צָרִיךְ לִדְחוֹת כְּמוֹ שֶׁכָּתְבוּ הַתּוֹסָפוֹת דְּמַיְרֵי שֶׁקֹּדֶם שֶׁנָּפַל רְבִיעִית נִבְלַע בַּכְּסוּת מַה שֶּׁאֵינוֹ יָכוֹל לָצֵאת וְנִתְבַּטֵּל בַּבֶּגֶד וְלֹא הָיָה רְבִיעִית שָׁלֵם מֵעוֹלָם: שֶׁאֵינוֹ יָכוֹל לָצֵאת. פֵּרוּשׁ, עַל יְדֵי כְּבִיסָה סְתָם, אַף עַל פִּי שֶׁעַל יְדֵי סַם יָכוֹל לָצֵאת. דְּאֵין דֶּרֶךְ לְהַקְפִּיד בִּכְהַאי גַּוְנָא. וּמֵהַאי טַעֲמָא לֹא קָשֶׁה מֵרֵישׁ פֶּרֶק ג' דְּעָרְלָה, בֶּגֶד שֶׁצְּבָעוֹ כוּ', דְּהָתָם בְּחָזוּתָא נֶהֱנֶה, אֲבָל בְּטֻמְאָה בִּקְפֵידָא תַּלְיָא מִלְּתָא. תּוֹסָפוֹת. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב:

וְקָשֶׁה, הֲרֵי הַבַּיִת מַאֲהִיל עַל הָאָדָם וְעַל הַטֻּמְאָה כֻלָּהּ וִיטַמְּאֵהוּ לָאָדָם כוּ'. וְהָרַמְבַּ"ם סִיֵּם, וְהֶאֱהִיל הַבַּיִת כוּ' וּכְמוֹ שֶׁכָּתַב הָרַ"ב בְּבָבָא שְׁנִיָּה. וּמַהֲרַ"ם פֵּרֵשׁ בְּחָצֵר שֶׁאֵין עָלָיו גַּג: קְטַפְרֵס. וְאַסְקֻפָּה דְּנָקַט, מִשּׁוּם כֵּלִים דְּקֹדֶם שְׁפִיכָה, שֶׁאִם נִשְׁפַּךְ בְּתוֹךְ הַבַּיִת טְמֵאִים. אֲבָל עַל הָאַסְקֻפָּה אֶפְשָׁר שֶׁלֹּא נִכְנַס כֻּלּוֹ בְּחִבּוּר אֲפִלּוּ שָׁעָה אַחַת. הָרַאֲבַ"ד. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: וְהָוֵי זוֹ וְאֵין צָרִיךְ לוֹמַר זוֹ. וְהָרֹא"שׁ פֵּרֵשׁ, בֵּין מַה שֶּׁיֵּשׁ מִן אַסְקֻפָּה לְתַחַת תִּקְרַת הַבַּיִת בֵּין מַה שֶּׁיֵּשׁ בַּחוּץ כֻּלָּהּ מְקוֹם מִדְרוֹן וְהַדָּם נִשְׁפָּךְ בְּכֻלָּהּ, נִמְצָא מִקְצָתוֹ תַּחַת הָאֹהֶל וּמִקְצָתוֹ חוּץ לָאֹהֶל, הַבַּיִת טָהוֹר, דְּאֵין קְטַפְרֵס חִבּוּר: וְאַף דְּבָשָׂר נַמִּי מַחֲלִיף, מִכָּל מָקוֹם נִכָּר בּוֹ חָרִיץ אוֹ גֻמָּא. גְּמָרָא. מַה שֶּׁאֵין כֵּן שֵׂעָר וְצִפֹּרֶן:

וְיָד לְמֵת יָלֵיף לֵיהּ בַּגְּמָרָא. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: כִּשְׁנֵי כוּ'. וּסְבִירָא לֵיהּ יֵשׁ יָד לְפָחוֹת מִכַּזַּיִת. אִי נַמִּי בְּעֶצֶם שֶׁיֵּשׁ בּוֹ מֹחַ וּמִשּׁוּם שׁוֹמֵר. מַהֲרַ"ם:

דַּם תְּבוּסָה. הֲלָכָה לְמֹשֶׁה מִסִּינַי דַּם הַמֵּת בִּרְבִיעִית, וְהַנָּזִיר מְגַלֵּחַ עָלָיו, וְחַיָּבִין עָלָיו עַל טֻמְאַת מִקְדָּשׁ וְקָדָשָׁיו. וּמִדִּבְרֵי סוֹפְרִים הוֹסִיפוּ דַּם תְּבוּסָה לְמַשָּׂא וּמַגָּע וְאֹהֶל לְבַד: שְׁמִינִית כוּ'. כְּלוֹמַר לַאֲפוֹקֵי רֻבּוֹ בְחַיָּיו, דְּלֹא גָזְרוּ בָזֶה. רַשִׁ"י: שֶׁדָּמוֹ שׁוֹתֵת. כְּלוֹמַר, שֶׁדֶּרֶךְ הַצָּלוּב שֶׁיְּהֵא דָּמוֹ שׁוֹתֵת. וְכֵן אֲבָל הַמֵּת כוּ', מִסְּתָמָא שֶׁדָּמוֹ מְנַטֵּף: דְּלֹא מִיבְטָל הַהוּא דִּלְאַחַר מִיתָה שֶׁהָיָה הַרְבֵּה. אֲבָל נוֹטֵף, קַמָּא קַמָּא בָּטֵל, וַאֲפִלּוּ הוּא מְרֻבֶּה בְּטֵלָה, שֶׁלֹּא נָפַל בְּיַחַד. רַשִׁ"י: אֲבָל מֵת מְטַמֵּא בִּמְנַטֵּף אֲפִלּוּ מֻנָּח בְּמִטָּה, דְּשָׁאנֵי מִטָּה דְּהוּא חָלוּל וְהַדָּם בּוֹקֵעַ וְיוֹרֵד וּמִתְאַסֵּף יַחַד. גְּמָרָא:

וְהוּא הַדִּין דַּהֲוָה מָצֵי לְפָרֵשׁ בְּכֵלִים שֶׁמֻּנָּחִים בַּחֲלַל הַחַלּוֹן אוֹ הַפֶּתַח: הַשִּׁדְרָה כוּ'. אַף עַל פִּי שֶׁאֵין כַּזַּיִת בָּשָׂר: אוּלַי שֶׁסּוֹבֵר דְּרַבִּי יוֹסֵי לֹא פָלֵיג. אֲבָל אֵינִי יוֹדֵעַ מִנַּיִן לוֹ. וְהָרַמְבַּ"ם כָּתַב שֶׁאֵינוֹ הֲלָכָה:

וְצָרִיךְ שֶׁיְּהֵא הֶחָלָל פָּנוּי טֶפַח, שֶׁאִם לֹא כֵן דְּהַטֻּמְאָה מְמַעֶטֶת אֶת הֶחָלָל טֶפַח, טֻמְאָה רְצוּצָה הִיא. הָרַ"א: פֵּרוּשׁ, בַּטָּמוּן בְּתוֹךְ הַקַּרְקַע. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: הָרַמְבַּ"ם. וְאָתֵי שַׁפִּיר דִּמְפָרֵשׁ נַמִּי לַמֵּבִיא אֶת הַטֻּמְאָה. וְהָרַ"שׁ שֶׁמְּפָרֵשׁ כֵּיצַד אַחוֹצֵץ וְלֹא אַמֵּבִיא כוּ', לֹא סִיֵּם כָּאן לַמֵּבִיא כוּ'. וְהָרַ"ב הִרְכִּיב שְׁנֵי הַפֵּרוּשִׁים וְלֹא דָק: סֻכָּה רֵישׁ פֶּרֶק ב'. וְעַיֵּן שָׁם בְּתוֹסָפוֹת: כְּלוֹמַר שֶׁיֵּשׁ בָּהֶם חוֹר גָּדוֹל. גְּמָרָא:
