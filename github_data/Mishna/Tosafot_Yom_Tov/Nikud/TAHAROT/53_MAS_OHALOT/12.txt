פרק יב

שִּעָל גָּבֹּיו כוּ'. אַבֹל שִּבֵּתוֲךֵ הָתָּנּוּר (וֵכֹל שִּכְּן תָּחֵתֹּיו) טֵמְאִים, מִידִי דָהַוֹה אָאֳכֹלִין שִּגִּבֵּלֹן בֵּטִיט. מָהַרָ"ם: דִּכֵהָאי גָוֵנֹא שִּהָטֶּמֵאֹה מִתֵפָּשִּּטּת סֹבִיב לָתָּנּוּר מִכֹּל צָד, אֹמֵרִינָן אַפִלּוּ מִן הָצָּד אֹדֹם וֵכְלִים כֵּמָאן דֵּלְיתָנֵהוּ דֹמְי לֵטָמְּא וֵלֲא לֵטָהְר, וּמִתֵפָּשִּּטּת הָטֶּמֵאֹה כֹּל תָּחָת הָנּּסּר אַפִלּוּ כֵנּגּד פִּי הָתָּנּוּר, וֵכְיוֹן שִּהָתָּנּוּר פֹּתוּחָ נִכֵנּסּת הָטֶּמֵאֹה לֵתוֲכוֲ. מָהַרָ"ם: כְּיוֹן דֵּאְין בְּין פִּי הָתָּנּוּר לָנּּסּר טּפָח, הֹוְי לְיהּ כֵּמִגֵדֹּל וֵכָוּּרּת שִּבָּפּּתָח וֵאְין בְּין פִּיהּם לָשִֵּקוֲף טּפָח, דֵּאְין טֶמֵאֹה נִכֵנּסּת לֹהּם, וֵלֲא בֹעְי צֹמִיד פֹּתִיל אּלֹּא הְיכֹא דֵאִיכֹּא טּפָח בְּין פִּי הָכֵּלִי לָגָּג שִּעֹלֹיו. אַבֹל בֵּכְלִים מוֲדּה דֵנָעַשֵֹה אֲהּל לֵטָמְּא. וֵתָנֹּא קָמֹּא סֵבִירֹא לְיהּ דֵּגֹזֵרוּ גָם בֵּתָנּוּר, דֵּלֲא לְימֵרוּ הָתָּנּוּר מָפֵסִיק בְּין הָטֶּמֵאֹה שִּלֲּא לָעַבֲר לֵצָד אָחְר. מָהַרָ"ם: טֵמְאִים. הָתָּנּוּרִים. וֵכֹל שִּכְּן כְּלִים שִּתָּחָת הָנּּסּר, דִּבֵעוֲדְף טּפָח מָיֵרְי. הֹרָאַבָ"ד:

וֵטָעַמֹא, דִּכֵשִּיְּשִ לֹהּ לִזֵבְּז הוּא דָחַשִִיבֹא לָחוֲץ וֵחוֲצּצּת, אַבֹל כֵּשִּאְין לֹהּ לִזֵבְּז לֹא חַשִִיבֹא וֵאְינֹהּ חוֲצּצּת. הֹרָ"שִ. וֵקֹשִּה דִּבֵמִשִֵנֹה ו' פּּרּק ה' שִֹמֵעִינָן דִּסֵרִידֹה שִּאְין לֹהּ גָּפָּיִם חוֲצּצּת וֵלֲא כֵשִּיְּשִ לֹהּ גָּפָּיִם. וּמָהַרָ"ם כֹּתָב עָל פִּי הָתּוֲסּפֵתֹּא, דֵּמָיֵרְי בִסֵרִידֹה שִּאְין לֹהּ לִזֵבְּז דֵּחָזֵיֹא לִשִֵכִיבֹה, וֵכֹל הָמִּטָּמְּא מִדֵרֹס מִטָּמְּא טֵמְא מְת, לֵכֹךֵ אְינֹהּ חוֲצּצּת, אַבֹל כֵּשִּיְּשִ לֹהּ לִזֵבְּז (וֵלֲא בֵאָרֵבָּע רוּחוֲתּיהֹ, דֵּאִם לֲא כְן טֹמְא מִטָּעָם כֵּלִי קִבּוּל) דֵּלֲא חָזֵיֹא לִשִֵכִיבֹה, חוֲצּצּת, וֵכְיוֹן דֵּהָתָּנּוּר מֶקֹּף צֹמִיד פֹּתִיל דֹּמְי לֵנִדֵבֹּךֵ הָנֹּתוּן עָל גָּבְּי אַבֹנִים דֵּרְישִ פּּרּק ו'. וֵעָיְּן תּוֲסֵפוֲת יוֲם טוֲב: הָכֲּל כוּ'. כֵּלוֲמָר, הָכֲּל טֹמְא חוּץ מִכֵּנּגּד אַוִירוֲ שִּל תָּנּוּר. מָהַרָ"ם: אַוִירוֲ כוּ'. כֵּלוֲמָר, עָל הָסֵּרְידֹה אַשִּר עָל אַוִיר הָתָּנּוּר. לֲא עָל אַוִיר הֹעִגּוּל הָיּוֲצְא מִפִּי הָתָּנּוּר. הֹרָמֵבָּ"ם:

כְּיוֹן דֵּהָזִּיז עֹשֵוּי לֵהָחַזִיק חֲם הֹאָמֵבָּטִי, בֹּטְל הֹאָמֵבָּטִי לֵגָבְּיהּ טֵפְי מִלֵּגָבְּי נּסּר. מָהַרָ"ם. וֵעָיְּן תּוֲסֵפוֲת יוֲם טוֲב:

מָשִֵמָע שִּנִּפֵחָת מְאְלֹיו. וֵקֹשִּה, דִּבֵרְישִ פּּרּק דִּלֵקָמֹּן מֵחָלֵּקִין בְּין עוֲשֵּה לֵנָעַשֵֹה. וֵהֹרָמֵבָּ"ם וֵהֹרָ"שִ כֹּתֵבוּ, סָנֵדֹּל שִּפֹּחָת הָמָּעַזִיבֹה, שִִעוּרוֲ בֵּפוֲתְחָ טּפָח. נִפֵחָת מְאְלֹיו, שִִעוּרוֲ בִּמֵלֲא אּגֵרוֲף. וֵכְן כֹּתָב מָהַרָ"ם דֵּמָתֵנִיתִין דָּוֵקֹא בֵּפֹחָת וֵלֲא בֵנִפֵחָת: הָכֲּל כוּ'. הֹעַלִיֹּה וֵכֹל מָה שִּּבֹּהּ מִכְּלִים. הֹרָ"ם: מוֲנִין כוּ'. וֵקֹשִּה, דֵּהֹא תֵנָן בֵּפּרּק י' מִשִֵנֹה ב' גָּבְּי אַרֶבֹּה שִּאְין בֹּהּ פּוֲתְחָ טּפָח, טֶמֵאֹה בָּבָּיִת נֹתָן אּת רָגֵלוֲ טֵהוֲרֹה. יְשִ לוֲמָר דֵּמָיֵרְי כֵשִּהָסָּנֵדֹּל נֹתוּן בִּפֵחָת מָעַזִיבֹה בֵּחּצֵיֹהּ שִּל מָטֹּה, דֵּנִדּוֲנּת מּחֱצֹה עָל מּחֱצֹה כֵּדִלֵעְיל בֵּפּרּק ו' מִשִֵנֹה ד', וָאַפִלּוּ הָטֶּמֵאֹה כֵנּגּד הָנּּקּב לֲא אֹמֵרִינָן עְרְב אּת הָטֶּמֵאֹה, שִּכֹּל דֹּבֹר שִּנִּכֵנָס בֹּעֳבִי הָמָּעַזִיבֹה וּמֵמָלְּא הָפֵּחֹת וֵיוֲשְִב שִֹם, מְהָתִּקֵרֹה הוּא חֹשִוּב, וֵאְין אוֲמֵרִים בּוֲ עְרְב אּת הָטֶּמֵאֹה. כּּסּף מִשִֵנּה בֵּשְִם הֹרִי"ק. וֵעָיְּן תּוֲסֵפוֲת יוֲם טוֲב:

מָעַזִיבֹה. וֵלֲא נֵסֹרִים. אּלֹּא דָחַדֹא נֹקָט. רָשִִ"י. וֵעָיְּן תּוֲסֵפוֲת יוֲם טוֲב: שִּגּוּד אָסִּיק וֵגוּד אָחִית הַלֹכֹה לֵמֲשִּה מִסִּינָי, וּבִתֵנָאי שִּיֵּהְא רֲחָב הָקּוֲרֹה טּפָח וּבְינְיהּן אַוִיר טּפָח. הֹרָמֵבָּ"ם:

כֶּלֹּהּ. אָף עָל פִּי שִּקֵּצֹתֹהּ פֵּחוּתֹה מִטּּפָח, מִפֵּנְי שִּקֵּצֹתֹהּ מִקֵצֹת אֲהּל הוּא. הֹרָמֵבָּ"ם. דֵּאִם לֲא כְן, לִתֵנְי מֵבִיאֹה אּת הָטֶּמֵאֹה וֵתוּ לֲא. כּּסּף מִשִֵנּה: מִשִּוּם דֵּלֲא סָגִּי בֵטָעַמֹא שִּהָמֵרֶבֹּע כוּ', דֵּאִיהוּ גוּפְיהּ קֹשִּה מֵנֹלֹן, לֵכֹךֵ מְבִיא מִקֵּרֹא. אָךֵ תֵּמְהָנִי דֵּהֹיֹה לוֲ לָהַבִיאוֲ מִמֵּלֹכִים א' ז' כ"ג דֵּקֹדְים: אָרֵבֹּעֹה. וֵקֹשִּה, דִּלֵגוּפְיהּ לֲא אִצֵטֵרִיךֵ, דִּכֵשִּמֹּדָד צָד אּחֹד לֹמֹּה לְיהּ לִמֵדֲּד עוֲד הָשִֵּלֲשִֹה צֵדֹדִים. וּלֵאָשִֵמֵעִינָן דִּמֵרֶבֹּע יֹתְר כוּ' אְינוֲ בֵמָשִֵמָע, דֵּאִם כְּן הַרְי זּה בֹּא לֵלָמְּד וֵנִמֵצֹא לֹמְד: רֵבִיעָ. דֵּהֹא כֹל עִגּוּל שִּרֹחֵבּוֲ אָמֹּה, הּקְּפוֲ שִֹלֲשִ, וּמֵרֶבֹּע שִּל אָמֹּה הּקְּפוֲ אָרֵבָּע. רָשִִ"י. וֵעָיְּן תּוֲסֵפוֲת יוֲם טוֲב:

כֵּלוֲמָר, מִמֵּקוֲם אּמֵצֹעוֲ שִּמִּשִֹּם מִתֵקָצְּר וֵהוֲלְךֵ וֵנָעַשֵּה מֹקוֲם חֹלוּל: כֵּלוֲמָר, וֵאָף עָל פִּי שִּהָטֶּמֵאֹה אְינוֲ בֵּמֹקוֲם שִּיְּשִ טּפָח עָל טּפָח: צֹרִיךֵ לוֲמָר דִּרֵצוֲנוֲ לוֲמָר דֵּהָתָּנֹּא לֲא דֹק, אַבֹל לֵדִינֹא אְין צֹרִיךֵ עּשֵֵרִים וֵאָרֵבֹּעֹה מָמֹּשִ, רָק שִּיּוּכָל לֵרָבְּעָ תָּחֵתֹּיו טּפָח עָל טּפָח. דֵּאִם לֲא כְן הְיכִי אֹמֵרִינָן לֲא דֹק לֵהֹקְל בֵּשִּל תּוֲרֹה. וֵלָחּשִֵבּוֲן הָמֵדֶיֹּק שִּל הֹרָמֵבָּ"ם אֹתְי שִָפִּיר דִּבֵדָוֵקֹא הוּא. עָיְּן תּוֲסֵפוֲת יוֲם טוֲב:

וּכֵשִִיטֵתְיהּ בֵּפּרּק י"א מִשִֵנֹה ז'. וֵרָבִּי יוֲסְי וֵרָבִּי יֵהוֲשִֶעָ בָּחַדֹא שִִיטֹה. וֵעָיְּן תּוֲסֵפוֲת יוֲם טוֲב: מִטּּפָח כוּ'. וֵכָיּוֲצְא בֹזּה מֹצֹאנוּ לוֲ בֵּפּרּק ט' מִשִֵנֹה ט"ו. וֵעָיְּן שִֹם:
