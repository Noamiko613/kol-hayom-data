פרק יב

דְּאִם לֹא כֵן הוּא מְנַעְנֵעַ כָּאן וְכָאן וְאֵינוֹ מְכַוֵּן לְהַזּוֹת עַל הַכְּלִי. רַשִׁ"י:

הַחֶרֶס. הַטָּהוֹר. וְסָלְקָא דַעְתָּךְ אֲמֵינָא דְּמִטַּהֶרֶת הַמַּחַט מִמִּצּוּי הַחֶרֶס כֵּיוָן שֶׁלֹּא הֻצְרַךְ לְהַזּוֹת עַל הַחֶרֶס, קָמַשְׁמַע לָן דְּלֹא: וּכְרַבָּנָן בְּרֵישׁ פֶּרֶק ט'. וְהָרַ"שׁ פֵּרֵשׁ, דְּלֹא אָמְרִינַן דִּלְמָא מֵחֲמַת הַדְּחָק יָצְאוּ מַיִם מִגּוּף הָאֵזוֹב וְנִתְעָרְבוּ בְמֵי חַטָּאת. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: לַאֲחֲרָיו. וְאַף שָׁם הָיוּ כֵלִים הַצְּרִיכִים הַזָּיָה. רַשִׁ"י: מִדִּכְתִיב וְהִזָּה הַטָּהֹר עַל הַטָּמֵא, וְלֹא כְתִיב עָלָיו, שְׁמַע מִנַּהּ שֶׁיְּהֵא מִתְכַּוֵּן לְטַהֲרוֹ. רַשִׁ"י וְהָרַ"ם. וְדַעַת הָרַאֲבַ"ד, דְּלָאו דַּוְקָא לְטַהֲרוֹ אֶלָּא שֶׁיִּתְכַּוֵּן לוֹ. וְנִרְאֶה שֶׁכֵּן הוּא דַעַת הָרַ"ב: כְּשֵׁרָה. דַּהֲוֵי כִלְפָנָיו. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב:

אִם יֵשׁ כוּ'. פֵּרוּשׁ, אַף עַל פִּי שֶׁיֵּשׁ בָּאֵזוֹב יִשְׁנֶה. הָרַמְבַּ"ם: הָאָדָם וְהִזָּה כוּ'. פֵּרוּשָׁא דְרֵישָׁא: בְּכַוָּנָה לְהַזּוֹת עַל דָּבָר הַמְקַבֵּל טֻמְאָה, וּבִתְנַאי שֶׁנָּטְפוּ בַכְּלִי. הָרַמְבַּ"ם:

מִשּׁוּם דְּלֹא הָוֵי מַמָּשׁ סְפֵק טֻמְאָה בִרְשׁוּת הָרַבִּים, דְּהָא נִטְמָא בְוַדַּאי, אֶלָּא שֶׁהַזָּיָתוֹ פְסוּלָה. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: רַבִּים. דִּבְּרוּ בַהֹוֶה, דִּבְחַלּוֹן שֶׁל יָחִיד אֵינוֹ בַנִּמְצָא כָּל כָּךְ מַיִם. וְהָרַ"ם כָּתַב, וְלֹא חָיְשִׁינַן שֶׁמָּא פְסוּלִין הֵם, כְּלוֹמַר וַעֲדַיִן מְטַמְּאִים. וּלְפִי זֶה דַּוְקָא נָקַט רַבִּים, דִּבְשֶׁל יָחִיד הָא צָרִיךְ לִבְדֹּק שֶׁמָּא פְסוּלִין הֵן: וְלֹא נִמְנָעִין. מִלָּבֹא אֶל הַמִּקְדָּשׁ וּמֵהִתְעַסֵּק בַּקָּדָשִׁים. הָרַמְבַּ"ם:

וּבִתְנַאי שֶׁלֹּא נִטְמָא הַקַּרְדֹּם בַּמֵּת עַצְמוֹ. הָרַמְבַּ"ם: כְּדֵי כוּ'. דִּבִכְלִי שֶׁהוּא מְטַבֵּל בּוֹ צְרִיכָא שִׁעוּר לְכֻלֵּי עָלְמָא. הָרַ"שׁ:

טָמֵא. פֵּרוּשׁ, לְחַטָּאת, לָכֵן הַמַּיִם פְּסוּלִין. אֲבָל אִי הָוֵי טָמֵא לְקֹדֶשׁ וְלִתְרוּמָה הָיוּ הַמַּיִם טְמֵאִין. הָרַמְבַּ"ם:

יָדָיו. וְלָאו דַּוְקָא לִשְׁתֵּיהֶן אֶלָּא בְאַחַת מֵהֶן סַגִּי:

חִבּוּר. לְטֻמְאָה וּלְהַזָּיָה. הָרַמְבַּ"ם. וְהָנֵי מִלֵּי כְּשֶׁמְּחֻבָּרִים, אֲבָל שֶׁלֹּא בִשְׁעַת חִבּוּר אִם נִטְמָא זֶה לֹא נִטְמָא זֶה. תּוֹסָפוֹת: בְּטַבָּעוֹת הַקְּבוּעִין בָּעֲרִיסָה הָעֲשׂוּיִין כְּעֵין פִּיקוֹת. הָרַ"שׁ: הַמַּלְבֵּן. פֵּרוּשׁ, דְּלֹא מְחֻבָּר לַמִּטָּה כְּלָל. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: פֵּרוּשׁ, דְּאִם נִטְמֵאת הַמִּטָּה לֹא נִטְמְאוּ הֵם. וְכֵן לְעִנְיַן הַזָּיָה. הָרַמְבַּ"ם:

וְהוּא כְלִי בַעַל חֻלְיוֹת יִתְרַכֵּב קְצָתָם בִּקְצָתָם. הָרַמְבַּ"ם. וְכֵן קְרָנִין הֵם בַּעֲלֵי חֻלְיוֹת: הַמַּפְתְּחוֹת. וְהַמַּפְתְּחוֹת נַעֲשׂוּ חִבּוּר עַל יְדֵי הַשַּׁלְשֶׁלֶת לַטֻּמְאָה וְלֹא לַהַזָּאָה. וְכָל זֶה אֵינוֹ אֶלָּא גְזֵרָה מִדִּבְרֵיהֶם. הָרַמְבַּ"ם:

הַכֹּל. לַאֲתוֹיֵי עָרֵל שֶׁמֵּתוּ אֶחָיו מֵחֲמַת מִילָה. כְּרַבִּי אֶלְעָזָר דִּסְבִירָא לֵיהּ דְּכָשֵׁר מִידִי דַּהֲוֵי אַטְּבוּל יוֹם שֶׁאָסוּר לִגַּע בִּתְרוּמָה וְכָשֵׁר לְפָרָה. גְּמָרָא. אֶלָּא דְקָשֶׁה, דִּבְפֶרֶק ה' מִשְׁנָה ד' כָּתַבְתִּי דְּטֻמְטוּם פָּסוּל לְקַדֵּשׁ שֶׁהוּא סְפֵק עָרֵל, וְאִם הַסְּבָרָא לְפָסְלוֹ לְקִדּוּשׁ כִּי טָמֵא, יִפָּסֵל נַמִּי לְהַזָּאָה, וְאִי יֵשׁ לְדַמּוֹת לִטְבוּל יוֹם, אֲפִלּוּ לְקִדּוּשׁ נַמִּי יֻכְשַׁר, וְהָכִי מוּכָח בַּגְּמָרָא. וְעוֹד קָשֶׁה לֵימָא דְהָכָא לַאֲתוֹיֵי אִשָּׁה לְרַבִּי יְהוּדָה, וּלְרַבָּנָן קָטָן, כְּמוֹ הָתָם. וְנִרְאֶה דְּמַתְנִיתִין דְּהָכָא וּדְהָתָם פְּלִיגֵי. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: דְּכֵיוָן דִּבְקִדּוּשׁ מַכְשִׁירִים זָר וְאִשָּׁה עַל כָּרְחֲךָ הָכָא לָאו לְאַכְשׁוֹרֵי זָר וְאִשָּׁה וּלְמִפְסַל קָטָן, דְּאִם כֵּן לִכְתּוֹב סְתָם וְלָקַח אֵזוֹב וְטָבַל בַּמַּיִם וְהִזָּה, אֶלָּא עַל כָּרְחֲךָ לְמִדְרַשׁ אִישׁ וְלֹא אִשָּׁה. רַשִׁ"י. וּלְרַבִּי יְהוּדָה דְּפָלֵיג הָתָם בְּקִדּוּשׁ פָּלֵיג נַמִּי הָכָא, דְּכֵיוָן דִּלְדִידֵיהּ אִשָּׁה פְסוּלָה בְקִדּוּשׁ וְקָטָן כָּשֵׁר, אֲתָא הַאי לְמֵימְרָא אִפְּכָא, דְּאִם לֹא כֵן לִשְׁתּוֹק מִנֵּיהּ: שֶׁאֵין כוּ'. דְּהָא הַזָּיָה צְרִיכָה כַוָּנָה , הִלְכָּךְ לֹא מְרַבִּינַן אֶלָּא קָטָן שֶׁיֵּשׁ בּוֹ דַעַת. וּמִמֵּילָא דִּלְקִדּוּשׁ דִּמְמַעֲטִינַן קָטָן גַּם יֶשׁ בּוֹ דַעַת פָּסוּל, דְּכֻלַּהּ פָּרָשָׁה מַשְׁמַע זֶה מוֹצִיא מִיַּד מַשְׁמַע. וְצָרִיךָ לוֹמַר דְּמַה שֶּׁכָּתַב הָרַ"ב לְעֵיל יָצְאוּ חֵרֵשׁ שׁוֹטֶה וְקָטָן שֶׁאֵין בָּהֶם דַּעַת, רוֹצֶה לוֹמַר דְּמִסְּתָמָא אֵין בָּהֶם דַּעַת וְלָכֵן פְּסָלָן הַכָּתוּב, וַאֲפִלּוּ מַשְׁכַּחַת נַמִּי קָטָן שֶׁיֵּשׁ בּוֹ דַעַת לֹא חִלֵּק, אֲבָל הָכָא אִי אֶפְשָׁר לוֹמַר לֹא חִלֵּק וּמְרַבֵּי אַף לְאֵין בּוֹ דַעַת, דְּלֹא עֲדִיף מִגָּדוֹל שֶׁצָּרִיךְ כַּוָּנָה. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: הָאִשָּׁה כוּ'. לְשׁוֹן הָרַמְבַּ"ם, קָטָן שֶׁיֵּשׁ בּוֹ דַעַת, הָאִשָּׁה מְסַעַדְתּוֹ. וְכֵן כָּתְבוּ הַתּוֹסָפוֹת. וְנָקְטֵי מִלְּתָא דִשְׁכִיחָא. וְלָאו דַּוְקָא, דְּהוּא הַדִּין לְגָדוֹל:

כָּשֵׁר. מִשּׁוּם דְּבָעֵי לְמִתְנֵי סֵיפָא פָּסוּל, תָּנֵי הָכָא כָּשֵׁר. אִי נַמִּי, אֲפִלּוּ שָׁהָה בֵינְתַיִם הַרְבֵּה. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: וְכֻלָּן. הֲוָה לֵיהּ לְמֵימַר וְאִם עֲשָׂאָן, כֵּיוָן דְּאֵינָן אֶלָּא שְׁתַּיִם, אֶלָּא דְמַסִּיק בַּלָּשׁוֹן דְּפֶרֶק ב' דִּמְגִלָּה מִשְׁנָה ד'. נִרְאֶה לִי:
