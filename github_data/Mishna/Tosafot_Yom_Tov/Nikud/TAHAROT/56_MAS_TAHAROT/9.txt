פרק ט

אֶלָּא אִם כֵּן זְפוּתָה. הָרַ"שׁ. וְקָשֶׁה, דְּאִם כֵּן לִגְזְרוּ גַּם כֵּן אַאֵינָהּ זְפוּתָה וּכְמוֹ שֶׁגָּזְרוּ בַּעֲנָבִים מֵהַאי טַעֲמָא כוּ'. וּלְמַה שֶּׁכָּתַב הָרַ"ב בְּפֶרֶק י' מִשְׁנָה ב' לֹא קָשֶׁה מִידִי:

לְשׁוֹן הָרַמְבַּ"ם בְּפֶרֶק קַמָּא דְּמַכְשִׁירִין, הַרְבֵּה אוֹמְרִים בַּמִּשְׁנָה הַזֹּאת טָמֵא בַּעֲבוּר מֻכְשָׁר, וְטָהוֹר עַל הַדָּבָר שֶׁאֵינוֹ מֻכְשָׁר:

וְלִכְאוֹרָה מַתְנִיתִין חַסּוֹרֵי מִחַסְּרָא וְצָרִיךְ לִגְרוֹס טְמֵאִין טְמֵאִין, אֲבָל נִרְאֶה דְּלֹא חָסְרִינַן לַהּ, דְּכִי קָתָנֵי מַשְׁקִין הַיְנוּ טְמֵאִים, וּטְמֵאִים דְּמַתְנִיתִין אַזֵּיתִים קָאֵי. וְעַיֵּן בַּמִּשְׁנָה דִּלְעֵיל בְּהָרַ"ב לַפֵּרוּשׁ הַשֵּׁנִי:

מִשּׁוּם דִּלְמָשְׁחָה כְּתִיב, וְאֵין זֶה דֶּרֶךְ גְּדֻלָּה לְעָשִׁיר. וּמִכָּל מָקוֹם קָשֶׁה, אַמַּאי מְהֵימְנִינַן לֵיהּ. וּמִסְתַּבְּרָא כְּמוֹ שֶׁכָּתַב הָרַמְבַּ"ם דְּמַפְרִישׁ בְּעוֹדָם זֵיתִים, מִדְּתָנֵי וְשִׁיֵּר קֻפָּה. וְכֵן מַשְׁמַע לְשׁוֹן הָרַ"ב בִּדְרַבִּי שִׁמְעוֹן. וְכֵן כָּתַב עוֹד הָרַמְבַּ"ם בְּיוֹלִיךְ לוֹ הַמַּפְתֵּחַ. יָבֹא הַכֹּהֵן וְיִסְחֲטֵם נֶגְדּוֹ וְיִקַּח מִמֶּנּוּ תְּרוּמַת שֶׁמֶן בְּטָהֳרָה:

וְקָשֶׁה, דְּהָא בְּרֵישׁ פִּרְקִין תְּנַן בֵּית הִלֵּל אוֹמְרִים מִשֶּׁיִּתְחַבְּרוּ כוּ' וְהַיְנוּ קֹדֶם שֶׁתִּגָּמֵר מְלַאכְתָּן. וְנִרְאֶה לִי דְּרַבָּן גַּמְלִיאֵל דִּלְעֵיל אֵינוֹ חוֹלֵק עַל בֵּית הִלֵּל, אֶלָּא תְּרֵי תַּנָּאֵי הֵן אַלִּיבָּא דְּבֵית הִלֵּל, תַּנָּא קַמָּא וְרַבָּן גַּמְלִיאֵל, וְהַךְ דְּהָכָא רַבָּן גַּמְלִיאֵל אֲמָרָהּ: זֵיתִים. שֶׁל תְּרוּמָה. הָרַמְבַּ"ם:

פֵּרוּשׁ, לְטָבְלָם. רַשִׁ"י:

הָרַמְבַּ"ם, שֶׁלֹּא הֻכְשְׁרוּ. וּבָרַ"שׁ, דְּלֹא: דִּכְשֶׁנִּשְׁבְּרוּ וְנִטְהֲרוּ, כְּשֶׁחָזַר וְעָשָׂה מֵהֶן כְּלִי חוֹזְרִין לְטֻמְאָתָן. עַיֵּן תּוֹסְפוֹת יוֹם טוֹב:

כְּלוֹמַר, שֶׁבַּמַּשְׁקֶה נִטְמְאוּ כֻּלָּם, שֶׁכֻּלָּם מְחֻבָּרִים לַמַּשְׁקֶה. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב:

הוּא לְשׁוֹן הַתּוֹסֶפְתָּא. וְהַיְנוּ פָּרוּד דְּמַתְנִיתִין: דְּכִי מָנִיתָ בֵּיהּ רִאשׁוֹן וְשֵׁנִי, הֲוֵי לֵיהּ שְׁאָר גּוּשִׁים שְׁלִישִׁי. הָרַ"שׁ: וְקָשֶׁה, דְּמַאי שְׁנָא נִמְצָא בַּמַּעֲטָן מִנִּמְצָא בַּגַּג, וְהָא הָכָא נַמִּי אִיכָּא לְמֵימַר שֶׁלְאַחַר שֶׁהֶעֱלָה לַגַּג נָפַל הַשֶּׁרֶץ בַּמַּעֲטָן, וְכֵן פֵּרֵשׁ הָרַ"ב בְּעַצְמוֹ בְּמִשְׁנָה ז' פֶּרֶק י'. וּבַתּוֹסֶפְתָּא תָּנֵי עֲלַהּ, דִּבְרֵי רַבִּי, וַחֲכָמִים אוֹמְרִים נִמְצָא בַּמַּעֲטָן הַגַּג טָהוֹר. וְכֵן פָּסַק הָרַמְבַּ"ם. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב: לִכְאוֹרָה מַשְׁמַע שֶׁאִלּוּ הָיָה טָמֵא מִקֹּדֶם לֹא פָּרְחָה הַטֻּמְאָה מִמֶּנָּה כוּ'. וְקָשֶׁה מִסּוֹף פֶּרֶק כ"ז דְּכֵלִים דִּכְשֶׁנִּמְהָה מִתְטַהֶרֶת. וּלְשׁוֹן רַשִׁ"י, טְהוֹרָה, דְּאָמְרִינַן הָכָא נָפְלָה עֲלַיְהוּ לְאַחַר שֶׁבָּלְתָה. וְעַיֵּן תּוֹסְפוֹת יוֹם טוֹב:
