פרק י

וְנַפְקָא מִנַּהּ בְּטֻמְאָה שֶׁמְּטַמֵּא בְּמַגָּע וְלֹא בְּהֶסֵּט. אֲבָל קָשֶׁה, דְּהָכָא מַיְרֵי שֶׁהָיוּ שָׁם כֵּלִים טְמֵאִין מִדְרָס, וְהֵם מְטַמְּאִים אַף בְּהֶסֵּט. וְאֶפְשָׁר דְּנַפְקָא מִנַּהּ דְּעַל הַנְּגִיעָה נֶאֱמָנִים לוֹמַר לֹא נָגַעְנוּ, וְעַל הַהֶסֵּט אִי אֶפְשָׁר שֶׁיִּהְיוּ נֶאֱמָנִים כֵּיוָן שֶׁאֵין חוֹשְׁשִׁין לָהּ. אֶלָּא דִּבְמַתְנִיתִין דִּלְקַמָּן פֵּרֵשׁ הָרַ"ב דְּאֵין נֶאֱמָנִין עַל הַנְּגִיעָה אֶלָּא הֵיכָא שֶׁיֵּשׁ חָבֵר עִמָּהֶם. וּלְכָךְ נִרְאֶה כְּפֵרוּשׁ הָרַמְבַּ"ם, דְּרַבִּי יוֹסֵי אַדְּרַבִּי שִׁמְעוֹן חוֹלֵק שֶׁסְּבִירָא לֵיהּ בֵּית הַבַּד טָהוֹר, וּפָלֵיג רַבִּי יוֹסֵי מִפְּנֵי שֶׁאַף עַל פִּי שֶׁלֹּא יִגְּעוּ בָּהֶן לִהְיוֹתָן טְמֵאִים, הִנֵּה יַסִּיטֵם לְפִי שֶׁאֵינָן יוֹדְעִים שֶׁהֶסֵּט יְטַמֵּא:

וְקָשֶׁה, דִּבְפֶרֶק ט' מִשְׁנָה ח' לֹא הִצְרִיכוּ שֶׁיְּהֵא הֶחָבֵר עוֹמֵד עֲלֵיהֶן. וְהָרַמְבַּ"ם לֹא הִצְרִיךְ כֵּן:

כֵּיוָן כוּ'. דְּכֵיוָן שֶׁנִּכְנְסוּ לִמְקוֹם הַטְּבִילָה עַל דַּעַת לִטְבֹּל, לֹא חָשְׁדִינַן לְהוּ שֶׁלֹּא יִטְבְּלוּ. וְלֹא הֶחְמִיר רַבִּי מֵאִיר בְּמַגָּעָן בְּרֵישׁ פִּרְקִין אֶלָּא מִשּׁוּם דְּאֵין חוֹשְׁשִׁים לִזָּהֵר, וְלֹא שֶׁנַּחֲזִיקֶנּוּ בְּרֶשַׁע:

טָהוֹר. שֶׁאַף עַל פִּי שֶׁהַגַּרְגֵּר נִטְמָא שֶׁהֲרֵי הֻכְשַׁר הוֹאִיל וְלַגַּת בְּצָרָן, מִכָּל מָקוֹם אֵין אֹכֶל מְטַמֵּא כְּלִי. וַאֲפִלּוּ אִם עַל יְדֵי דְּרִיכָה יָצָא מִמֶּנּוּ מַשְׁקֶה, אֵין הַמַּשְׁקֶה מְטַמֵּא הַכְּלִי כֵּיוָן שֶׁאֵין שִׁעוּר בָּאֹכֶל לְטַמֵּא הַמַּשְׁקֶה: דַּעֲנָבִים נַמִּי הָיוּ מִתְטַמְּאִים בְּכַבֵּיצָה:

בִּרְשׁוּת הַיָּחִיד. הָרַמְבַּ"ם:

טְמֵאוֹת. צָרִיךְ עִיּוּן, דְּהָא סְתָמָא קָתָנֵי, וְכִי לֹא בָּדַק וְלֹא כִּסָּה אֵימָא לְאַחַר שֶׁמִּלְּאָהּ נָפַל בָּהּ וְהָרִאשׁוֹנָה בִּלְבַד תִּהְיֶה טְמֵאָה:

בָּזוֹ כוּ'. נִכְנָסִים בְּרוּחַ זוֹ וְיוֹצְאִים בָּרוּחַ שֶׁכְּנֶגְדָּהּ. הָרַמְבַּ"ם: שֶׁאוֹתוֹ הַפְּסֹלֶת נִקְרָא גֶּפֶת: מִלְּשׁוֹן וְשִׁבֹּלֶת שְׁטָפָתְנִי . וּלְפִי שֶׁדֶּרֶךְ הַסְּפִינוֹת לָלֶכֶת שָׁם, נִקְרָא כֵּן, מִלְּשׁוֹן שְׁבִיל:
