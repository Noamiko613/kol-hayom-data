פרק יא

מִדֵּלֲא תֵנָן נָמִּי הֹכֹא חוּץ מִגְּר תּוֲשִֹב כֵּמוֲ בֵּרְישִ פּּרּק ג'. וֵעָיְּן תּוֲסֵפוֲת יוֲם טוֲב: וֵלֲא דֹמְי לֵהֹא דִבֵפּרּק ז' אְלּוּ בּהֹרוֲת טֵהוֲרוֲת שִּהֹיוּ בֵנֹכֵרִי קֲדּם שִּנִּתֵגָּיְּר. דֵּהֹתֹם בֵּגוּפוֲ, וּכֵשִּנִּתֵגָּיְּר כֵּקֹטֹן שִּנּוֲלָד הֹוְי, אַבֹל הֹכֹא הָאי בּּגּד הַוֹה חַזִי לִנֵגֹעִים כוּ'. הֹרָ"שִ: וֵלֲא יֹדָעֵתִּי מֹה עִנֵיָן יֹמִים לֵכֹאן, דֵּכֹל עוֲד שִּלֲּא רֹאֹהוּ הָכֲּהְן וֵגֹזָר עֹלֹיו טֶמֵאֹה כֵּאִלּוּ אְינוֲ הוּא, וֵלֲא דֹמְי לֵנִדֹּה אוֲ לֵנֹזִיר, זֹב וֵזֹבֹה, דִּתֵלוּיִין בֵּעָצֵמֹן: וֵצּמּר מִן בָּעַלְי חָיִּים הוּא, וֵהְם חָיִּים מִן הָצֵּמֹחִים. הֹרָמֵבָּ"ם: חִבְּר כוּ'. בֵּתוֲרָת כֲּהַנִים יֹלְיף לְיהּ. וֵעָיְּן תּוֲסֵפוֲת יוֲם טוֲב:

רֵחְלִים. שִּסֵּתֹם צּמּר הוּא שִּל רֵחְלִים: מּחֱצֹה כוּ'. אָף דִּסֵפְק נֵגֹעִים לֵהֹקְל, אַפִלּוּ הֹכִי טֹמְא. הֹרָ"שִ. וֵכְן בֵּכִלֵאָיִם דִּינֹא הֹכִי:

דִּבֵרְי רָבִּי מְאִיר כוּ'. בֵּתוֲרָת כֲּהַנִים פֵּלִיגְי בִדֵרֹשְִי דִקֵרֹאְי. וֵעָיְּן תּוֲסֵפוֲת יוֲם טוֲב: שִּאְין דּּרּךֵ לֵצֹבֵעוֲ. הֹרָמֵבָּ"ם. אָף כוּ'. צֵבוּעִים בִּידְי שִֹמָיִם מִנָּיִן, תָּלֵמוּד לוֲמָר לָפִּשִֵתִּים כוּ': דֵּלְית בְּיהּ צֹבוּעָ בִּידְי שִֹמָיִם. קֹרֵבָּן אָהַרֲן. וֵעָיְּן תּוֲסֵפוֲת יוֲם טוֲב:

שִּבָּיֵרֶקִּים כוּ'. מִדִּכֵתִיב יֵרָקֵרָק וָאַדָמֵדָּם יֹלְיף לָהּ בֵּתוֲרָת כֲּהַנִים. וֵכֹתָב הֹרָמֵבָּ"ם, וֵשֲִרּשִ מִשִֹּרֵשְִי הָלֹּשִוֲן שִּלֲּא יִהֵיּה נוֲסֹף אוֲת אוֲ אוֲתִיּוֲת אּלֹּא לֵזֹרְז וּלֵהָפֵלִיג כוּ'. וֵעָיְּן תּוֲסֵפוֲת יוֲם טוֲב: אַבֹל בֵּפֹתוּךֵ אְינֹן מִטָּמֵּאִין. וֵכֹל זּה בֵּתוֲרָת כֲּהַנִים. וֵעָיְּן תּוֲסֵפוֲת יוֲם טוֲב: כֵּלוֲמָר מֵכָבֵּסוֲ וּמָסֵגִּירוֲ כֵּדִלֵקָמֹּן. וּבֵתוֲרָת כֲּהַנִים יֹלְיף לְיהּ. וּמָשִֵמָע נָמִּי בֵּעוֲמְד שִֵנְי שִֹבוּעוֲת שִּיִּשֵֹּרְף. וֵאְין כְּן דָּעָת הֹרָמֵבָּ"ם. וֵעָיְּן תּוֲסֵפוֲת יוֲם טוֲב:

כֵּלוֲמָר לֲא הָבּּגּד כֶּלּוֲ וֵלֲא הָנּּגָע בִּלֵבָד, אּלֹּא מֵעָט מִן הָבּּגּד עִמּוֲ. מִדִּכֵתִיב וֵכִבֵּסוּ אּת אַשִּר בּוֲ הָנֹּגָע, וֵלֲא כֵתִיב וֵכִבֵּסוּ הָנּּגָע. תּוֲרָת כֲּהַנִים: דֵּאִלּוּ יֹרֲק חָלֹּשִ שִּהוּא מָרֵאּה שִֵלִישִִי, טֹהוֲר לֵגָמֵרְי, אַפִלּוּ בֹרִאשִוֲן אוֲ בָשְִּנִי, דִּכֵתִיב וֵהִנְּה כְּהֹה הָנּּגָע, מָשִֵמָע דָּעַדָיִן קֵצֹת מָרֵאְה נּגָע בּוֲ. תּוֲרָת כֲּהַנִים: וּבֹרִאשִוֲן עֹמָד. וֵאִם כֹּהֹה בֹרִאשִוֲן וֵחֹזָר בָּשְִּנִי לֵמָרֵאִיתוֲ, יִשֵֹּרְף. וֵאִם כֹּהֹה גָם בָּשְִּנִי, אַפִלּוּ פֹּשֵֹה בָכְּהּה, מֵכָבֵּסוֲ. הֹרָ"שִ. וֵכֶלֵּהוּ דִינְי דֵפִרֵקִין בֵּתוֲרָת כֲּהַנִים מִדֵּרֹשְִי דִקֵרֹא. וֵעָיְּן תּוֲסֵפוֲת יוֲם טוֲב: אְינוֲ צֹרִיךֵ כוּ'. וֵהָאי עוֲד מִבֹּעְי לְיהּ, מֹה רִאשִוֲנֹה כָּגֵּרִיס, אָף שִֵנִיֹּה. וֵרָבֹּנֹן תָּרֵתְּי שִֵמָע מִנָּהּ. תּוֲרָת כֲּהַנִים:

וּלֵהֹכִי לֲא שִָיֹּךֵ בְּיהּ לֵשִוֲן הָצֹּלֹה: הֹרָ"שִ מְהָתּוֲסּפֵתֹּא. וֵנִרֵאּה לִי דֵּטָעַמֹא דֵּהָמָּטֵלִית אְינוֲ נִמֵשִֹךֵ לֵהָבּּגּד, כְּיוֹן דִּמֵחֶבֹּר לָטֹּהוֲר. אַבֹל בֵּסְיפֹא, הָבּּגּד שִּאְינוֲ מֵחֶבֹּר לָטֹּהוֲר נִמֵשִֹךֵ אָחַרְי הָמָּטֵלִית. אּלֹּא דֵצֹרִיךֵ עִיּוּן דֵּמָאי שִֵנֹא מְהֹא דֵפּרּק י"ג מִשִֵנֹה ה' דֵּחוֲלְץ הֹאַבֹנִים. עָיְּן שִֹם. וֵהֹרָמֵבָּ"ם גּוֲרְס שֵוֲרְף. וֵכְן הוּא בֵּתוֲרָת כֲּהַנִים. וֵעָיְּן תּוֲסֵפוֲת יוֲם טוֲב:

שִּיּּשִ בּוֲ כָגֵּרִיס וֵלֲא יוֲתְר. הֹרָ"שִ. כֵּלוֲמָר, דֵּאְין צֹרִיךֵ שִּיֵּהְא בוֲ שִֹלֲשִ עָל שִֹלֲשִ, דֵּכֶלּוֲ בּּגּד אּחֹד הוּא. וֵעָיְּן תּוֲסֵפוֲת יוֲם טוֲב: וֵדָוֵקֹא מִצָּד אּחֹד, אָף שִּרֵחוֲקִים, אַבֹל מִצָּד זּה לֵצָד אָחְר לֲא, אּלֹּא אִם כְּן בֵּעוֲרוֲת:

שִֹלֲשִ. דֵּיֹלְיף לְיהּ נָמִּי בָּמּּה מֹצִינוּ מִבּּגּד. תּוֲרָת כֲּהַנִים: אַפִלּוּ כוּ'. דֵּאְין צֹרִיךֵ שִּיִּהֵיּה שִֵתִי וֹעְרּב, דֵּאוֲ לֵחָלְּק. קֹרֵבָּן אָהַרֲן:

רוֲצּה לוֲמָר לִשִֵתִי הֹעוֲמְד מִצֵטֹרְף: וֵקֹשִּה, דִּבֵפּרּק י"ב מִשִֵנֹה ב' תֹּנְי דִּתֵחִלָּת הָבּּגּד צֹרִיךֵ שִֹלֲשִ עָל שִֹלֲשִ. וֵעוֲד, מֹה עִנֵיָן בֹּתְּי נִירִין לֵשִֹלֲשִ עָל שִֹלֲשִ. וֵנִרֵאּה לִי דֵּהֹכִי גֹרֵסִינָן, עָד שִּיּּאֱרֲג בּוֲ שִֵלִישִִי, וֵלֲא גֹרֵסִינָן עָל שִֹלֲשִ. וֵעָיְּן תּוֲסֵפוֲת יוֲם טוֲב:

שִּהַוֹה לְיהּ לֵמְימָר הָצּּמּר כוּ', וּמָאַמֹרוֲ בָּצּּמּר, יוֲרּה שִּיְּשִ בֹּהּן מָה שִּּיִּשֵֵרֲף וּמָה שִּּיָּעַזֲב, וּבֹאֹה הָקָּבֹּלֹה שִּיָּעַזֲב מְהּן הֹאֶמֵרִיּוֲת. הֹרָמֵבָּ"ם: שִּאְינוֲ רֹאוּי לֵקָבְּל נּגָע, וֵאָף עָל פִּי שִּיְּשִ בֵּמִינוֲ נּגָע. הֹכִי יֹלְיף לְיהּ בֵּתוֲרָת כֲּהַנִים:

כֵּגוֲן כוּ'. וֵלֲא יֹדָעֵתִּי לֹמֹּה פְּרָט אְלּוּ, וֵרָבִּים הְם הָשִֵּנוּיִים בֵּכְלִים: וֵקֹשִּה, דִּסֵבֹכֹה גוּפֹא הַוֹה מֹצְי לֵמִתֵנְי. וֵהֹרָמֵבָּ"ם נִרֵאּה דֵּלֲא גֹרָס וֵשִֹבִיס:

וֵכְן פְּרְשִ הֹרָ"שִ, מִשִּוּם דֵּמוּכֹח מִתּוֲרָת כֲּהַנִים דֵּאָיֵרְי אָף בִּימְי הּסֵגְּרוֲ, וֵלֲא בֵאוֲתוֲ שִּנִּטֵהָר בִּקֵרִיעָת הָנּּגָע. וֵעָיְּן תּוֲסֵפוֲת יוֲם טוֲב: כֵּדִלֵעְיל בֵּסוֲף פּּרּק ה' לֵעִנֵיָן אֹדֹם. הֹרָ"שִ. וֵאָף דֵּהֹתֹם מִקֵּרֹא יֹלְיף, אְין לֵהָחֵמִיר בֵּנִגֵעְי בֵגֹדִים יוֲתְר מִבֵּנִגֵעְי אֹדֹם: הֹרָמֵבָּ"ם וֵהֹרָ"שִ. וֵתְימָהּ, לֹמֹּה לֲא פְרֵשִוּ מִדִּכֵתִיב מָמֵאּרּת. וֵעוֲד, שִּגָּם תֵּרוּמֹה טֵמְאֹה בִּשֵֵרְפֹה וּמֶתֹּר לְהֹנוֲת:
