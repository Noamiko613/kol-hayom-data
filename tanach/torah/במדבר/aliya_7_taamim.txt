וְיִִסָמֲךָ אּת יֻדֻיו עֻלֻיו וְיָצְוִֹהוִ כְִׁאֶשּ֖ר דִִבִּר יָהֲוֻה בִָיְד מֲשּׁ֖ה: (פ)

וָנִסָכִוֲ רָבִיעִת הְהִין לְכִּבּש֚ הֻׁאּחֻד בְִקֲִדּש֖ הְסִֹךָ נּסּךָ שֹ֖כֻר לְׁיהֲוֻׁה:
וָאֹת הְכִּבּש֚ הְשִֹ֖נִי תְִׁעֶשּ֚ה בִֹין הֻׁעְרָבִֻיִם כִָמִנָחְת הְבֲִקּר וִכָנִסָכִוֲ תְִׁעֶשּ֚ה אִשִֹ֖ה רֹיחְ נִיחֲחְ לְׁיהֲוֻׁה: (פ)

עֲלְת שְ֖בְִת בִָשְ֖בְִתִוֲ עְל עֲלְת הְתִֻמִיד וָנִסָכִֻׁהִ: (פ)

וִשָ֖לֲשֻ֖ה עּשָ֚רֲנִים סֲלּת מִנָחֻה בִָלוִלֻה בְשִּ֖מּן לְפִֻר הֻׁאּחֻד וִשָ֖נֹי עּשָ֚רֲנִים סֲלּת מִנָחֻה בִָלוִלֻה בְשִּ֖מּן לֻאְיִל הֻׁאּחֻׁד:
וָעִשִֻ֚רֲן עִשִֻ֚רוֲן סֲלּת מִנָחֻה בִָלוִלֻה בְשִּ֖מּן לְכִּבּש֚ הֻׁאּחֻד עֲלֻה רֹיחְ נִיחֲחְ אִשִּ֖ה לְׁיהֲוֻׁה:
וָנִסָכִֹיהּם חֶצִי הְהִין יִׁהָיּה לְפִֻר וִשָ֖לִישִ֖ת הְהִין לֻאְיִל וִרָבִיעִת הְהִין לְכִּבּש֚ יֻיִן זֲאת עֲלְת חֲדּש֖ בִָחֻדָש֖וֲ לָחֻדָשֹ֖י הְשִֻ֖נֻׁה:
וִשָ֚עִיר עִזִִים אּחֻד לָחְטִֻאת לְׁיהֲוֻה עְל עֲלְת הְתִֻמִיד יֹׁעֻשּ֚ה וָנִסָכִׁוֲ: (ס)

וִבְׁחֶמִשִֻ֖ה עֻשֻ֚ר יוֲם לְחֲדּש֖ הְזִּה חֻג שִ֖בָעְת יֻמִים מְצִוֲת יֹׁאֻכֹׁל:

וִשָ֚עִיר חְטִֻאת אּחֻד לָכְפִֹר עֶלֹיכּׁם:
מִלִָבְד עֲלְת הְבֲִקּר אֶשּ֖ר לָעֲלְת הְתִֻמִיד תְִׁעֶש֚וִ אּת אֹׁלִּה:

אִם יִרָאוִ הֻׁאֶנֻשִ֖ים הֻׁעֲלִים מִמִִצָרְיִם מִבִּן עּשָ֚רִים שֻ֖נֻה וֻמְעָלֻה אֹת הֻׁאֶדֻמֻה אֶשּ֖ר נִשָ֖בְִעָתִִי לָאְבָרֻהֻם לָיִצָחֻק וִׁלָיְעֶקֲב כִִי לֲׁא מִלָאוִ אְׁחֶרֻׁי:
בִִלָתִִי כִֻלֹב בִּן יָפֳנִּה הְקִָנִזִִי וִׁיהוֲשֳ֖עְ בִִן נוִן כִִי מִלָאוִ אְׁחֶרֹי יָהֲוֻׁה:
וְיִִׁחְר אְף יָהֲוֻה בִָיִשָ֚רֻאֹל וְיָנִעֹם בְִמִִדָבִֻר אְרָבִֻעִים שֻ֖נֻה עְד תֲִם כִֻל הְדִוֲר הֻׁעֲשּ֚ה הֻרֻע בִָעֹינֹי יָהֲוֻׁה:
